\section{Context and Problem}
% ============================================================================

\begin{frame}{From autocomplete to autonomous agents}
\begin{center}
\resizebox{!}{3.85cm}{\input{figures/fig_timeline.tex}}
\end{center}
\vspace{0.02cm}
\begin{itemize}\footnotesize
  \item First-generation tools suggested the \textbf{next line}; second-generation \textbf{agents} plan, edit, run tests, and revise --- \textbf{without} a human in the loop, following the ReAct (reason + act) pattern.
  \item That is a change in \emph{kind}, not degree --- why function-level benchmarks stop being sufficient.
\end{itemize}
\end{frame}

\begin{frame}{Vibe coding, and its riskier cousin}
\begin{itemize}
  \item \textbf{Vibe coding}: the developer keeps intent and direction; the agent handles implementation.
  \item \textbf{Vibe architecting} \citep{konrad2026vibe}: structural decisions shaped implicitly by \emph{how a prompt happens to be worded}, not by deliberate design.
  \item Their evidence: the \emph{same} task, different prompts, produced systems from \textbf{141 lines / 2 files} to \textbf{827 lines / 6 files} --- no architecture decision records, no rationale.
  \item Agent-authored code is not hypothetical: the AIDev dataset \citep{li2025aidev} alone catalogues \textbf{789{,}364 merged} pull requests from autonomous agents, and it is already being studied at scale \citep{cynthia2026agents}.
\end{itemize}
\keyfinding{Speed and pass-rate are one narrow slice of quality. SOLID, code smells, and architectural metrics exist precisely because functional correctness and good design are different things.}
\end{frame}

\begin{frame}{The benchmark--reality gap}
\begin{itemize}
  \item \textbf{HumanEval / SWE-bench} \citep{jimenez2024swebench} ask: does the code pass tests? Nothing about structure, coherence, or maintainability.
  \item \textbf{METR (2026)} \citep{metr2026sweverified}: $\sim$half of SWE-bench-Verified passes would \textbf{not} be merged by the repository's own maintainers --- a $\sim$24-point gap, roughly \textbf{2$\times$} the human reference rejection rate.
  \item \textbf{SlopCodeBench} \citep{slopcodebench2026}: quality erodes across successive checkpoints; agent code runs $\sim$2.2$\times$ more verbose than comparable open source; the best checkpoint solve rate was \textbf{17.2\%}.
  \item \textbf{NITR} \citep{zhu2026nitr}: \textbf{13.3\%} of AI-generated code passes every functional test while failing an architectural-quality check --- a \emph{structural oracle failure} invisible to test suites.
\end{itemize}
\end{frame}

\begin{frame}{Research objectives}
\begin{description}[leftmargin=1.9em,labelsep=0.5em,itemsep=9pt,topsep=2pt]
  \item[\textbf{\color{issatgold!80!black}O1.}] Build a repository-scale framework measuring all \textbf{five SOLID principles} on \textbf{real merged pull requests}, via deterministic structural proxies.
  \item[\textbf{\color{issatgold!80!black}O2.}] Turn construct validity into a \textbf{measured quantity} --- validate every proxy against an independent reading judge (precision/recall), not an assumption.
  \item[\textbf{\color{issatgold!80!black}O3.}] Apply the framework to \textbf{five leading agents} --- Claude Code, Copilot, Cursor, Devin, OpenAI Codex --- pooled and repository-mix-controlled.
  \item[\textbf{\color{issatgold!80!black}O4.}] Locate \textbf{where risk concentrates} (inside one unit vs.\ between units), and test whether \textbf{local ML} beats a deterministic proxy where its precision ceiling is genuinely semantic.
\end{description}
\end{frame}

\begin{frame}{Research questions}
\begin{description}[leftmargin=2.3cm,style=nextline]
\item[\textcolor{issatblue}{RQ1}] Do agents differ in SOLID adherence on real merged PRs, and where do the differences concentrate?
\item[\textcolor{issatblue}{RQ2}] Can SOLID be measured at scale, how faithfully do the proxies agree with a reading judge, and with independently built analysers?
\item[\textcolor{issatblue}{RQ3}] Is risk intra-unit or inter-unit; can local ML beat a deterministic proxy; and what actually ports to a second language?
\end{description}
\vspace{0.3cm}
\keyfinding{RQ1 is answered from the \textbf{Python} corpus only. AIDev supplies almost no Java from any agent but OpenAI Codex, so the Java arm is a within-agent replication, not a second comparison.}
\end{frame}

\section{State of the Art}
% ============================================================================

\begin{frame}{Foundations: three layers of design quality}
\begin{columns}[T]
\column{0.33\linewidth}
\textbf{\color{srpcolor}Design principles}\\[3pt]
{\small Martin's SOLID \citep{martin2000solid} --- five heuristics for where responsibility, extension, substitutability, and dependency should sit.}
\column{0.33\linewidth}
\textbf{\color{ocpcolor}Code smells}\\[3pt]
{\small Fowler's taxonomy \citep{fowler1999refactoring} --- God Class, Feature Envy, Data Clumps --- a shared vocabulary for \emph{what} is wrong.}
\column{0.34\linewidth}
\textbf{\color{dipcolor}Architectural metrics}\\[3pt]
{\small CK suite \citep{chidamber1994metrics} and Martin's $A$/$I$/$D$ \citep{martin2003agile} --- the quantitative vocabulary that makes design measurable.}
\end{columns}
\vspace{0.4cm}
These layers are not independent --- SOLID violations manifest as smells, and smells are what CK-style metrics are calibrated to catch. This thesis works at the \textbf{principle} layer, because it is the most design-relevant and least reliably measured of the three.
\end{frame}

\begin{frame}{What the literature already shows}
\begin{itemize}
  \item \citet{pehlivan2025solid} --- prompting LLMs to \emph{detect} SOLID violations: SRP/ISP detected reliably; \textbf{DIP is consistently the hardest}; accuracy falls as code complexity rises.
  \item NITR \citep{zhu2026nitr} --- agents succeed on dependency-control tasks only \textbf{4.3\%} of the time, on responsibility-decomposition only \textbf{15.2\%}.
  \item \citet{zhu2026smells} --- a \emph{machine signature} of AI defects (broken abstractions, systematic duplication, over-engineering); a \textbf{Volume-Quality Inverse Law}: code volume predicts structural degradation almost perfectly, and neither test coverage nor prompt detail stops it.
  \item ``Beyond Functional Correctness'' --- large AI-IDE projects that pass their tests still carry SRP-shaped smells.
\end{itemize}
\keyfinding{The common thread: passing tests and being well-designed are different things --- and DIP is the principle every study finds hardest, before this thesis even starts measuring.}
\end{frame}

\begin{frame}{Where the literature leaves a gap}
\begin{columns}[T]
\column{0.33\linewidth}
\textbf{\color{issatblue}Scale, with attribution}\\[3pt]
{\small ProjDevBench: 20 problems \citep{lu2026projdevbench}. SWE-bench: issue-level, not design. This thesis: \textbf{546 Python PRs / 147 repos} + \textbf{867 Java PRs / 27 repos}, each attributable to one agent.}
\column{0.33\linewidth}
\textbf{\color{issatblue}Depth + measured validity}\\[3pt]
{\small Most studies \emph{assert} their metric stands in for the principle. This thesis \textbf{measures} the gap: precision/recall against a reading judge, for all five principles.}
\column{0.34\linewidth}
\textbf{\color{issatblue}Agent-level specificity}\\[3pt]
{\small The literature varies the \emph{model}, holding scaffolding constant. This thesis isolates the \textbf{agent} --- Claude Code, Copilot, Cursor, Devin, Codex --- as the unit of comparison.}
\end{columns}
\end{frame}

\section{SOLID: the Principles, and Why}
% ============================================================================

\begin{frame}{The five SOLID principles}
\begin{center}
\vspace{0.1cm}
\resizebox{!}{5.0cm}{\input{figures/fig_solid_wheel.tex}}
\end{center}
\end{frame}

\begin{frame}{Why a validated proxy, not an assumed one}
\begin{itemize}
  \item SOLID (Martin, 2000) is the most cited normative OO design framework --- and, per the literature, the dimension \textbf{least reliably measured at repository scale}.
  \item None of the five principles is directly observable in source text: each is a statement about \emph{intent} --- the reasons a module may change, the variants a design anticipates, the promises a subtype makes.
  \item This study's stance (\textbf{post-positivist}): treat every measurement as a \emph{proxy}, and \textbf{quantify the proxy-to-principle gap} --- precision/recall against an independent reading judge --- rather than assert fidelity.
\end{itemize}
\keyfinding{\small Every number here is a statement about the proxy first, and about the principle only \emph{through} it --- why Part 5 reports agreement scores beside every rate.}
\end{frame}

\begin{frame}{The suppressors --- exact definitions}
\footnotesize
\textbf{\color{srpcolor}SRP --- LCOM4 (cohesion) and C3 (the semantic suppressor).}\\[1pt]
LCOM4: one node per non-constructor method, connected if two share an attribute or call each other --- \textbf{LCOM = connected components.} C3: each method $\to$ a smoothed TF-IDF vector $v_i$; mean pairwise cosine similarity ($C3\!\geq\!0.4$ un-flags the class):
\vspace{-0.32cm}
\[
\cos(a,b) = \frac{a \cdot b}{\lVert a\rVert\,\lVert b\rVert}
\qquad\quad
C3 = \operatorname*{mean}_{i<j}\ \cos(v_i,v_j)
\]
\vspace{-0.15cm}
\textbf{\color{ocpcolor}OCP --- the closed-set suppressor.}\\[1pt]
A chain is flagged only if wide \emph{and} not over a closed set:
\vspace{-0.32cm}
\[
\text{smell} = (\text{branch\_count} \geq \text{threshold}) \;\wedge\; \lnot\,\text{is\_closed\_type\_set}(\text{types})
\]
\vspace{-0.15cm}
\texttt{is\_closed\_type\_set}: every type a Python builtin, \textbf{or} sharing one fixed grammar prefix (e.g.\ \texttt{ast.*}) --- exhaustive by construction.
\end{frame}

\begin{frame}{Martin's plane --- and the sixth, descriptive evaluator}
\begin{columns}[T]
\column{0.44\linewidth}
\resizebox{\linewidth}{!}{\input{figures/fig_martin_plane.tex}}
\column{0.56\linewidth}
\small
\begin{itemize}
  \item $I = C_e/(C_e+C_a)$ instability; $A = N_a/N_c$ abstractness (share of abstract classes).
  \item $D = |A+I-1|$ --- distance from the ``main sequence'' good packages sit near.
  \item \textbf{Zone of pain}: concrete \& stable --- everything depends on it, it can't change. \textbf{Zone of uselessness}: abstract, used by nobody.
\end{itemize}
\end{columns}
\vspace{0.15cm}
\keyfinding{\footnotesize No delta, no violation flag: validation scored the $D$-as-change signal at precision \textbf{0.000}. Package \emph{position} stays a sound descriptor; package \emph{change} does not.}
\end{frame}

\begin{frame}{Validating the instrument}
\begin{itemize}
  \item A \textbf{reading judge} (an LLM, Claude) labels a stratified sample --- flagged and unflagged, blind to the rubric's own verdict --- by reading the actual source.
  \item Three of five proxies (SRP, OCP, LSP) start as \textbf{high-recall screens}; a \textbf{suppressor} then removes a \emph{nameable} false-positive idiom without cost to recall.
  \item Every candidate change passes a \textbf{validation gate}: it must raise precision at the same recall, on a sample \textbf{it was not tuned on}.
  \item Every comparative result is reported \textbf{pooled} and \textbf{macro-averaged per repository} --- macro leads, because one repository can otherwise masquerade as an agent property.
\end{itemize}
\end{frame}

\begin{frame}{Three more cross-cutting findings}
\begin{itemize}
  \item \textbf{Finding 4 --- porting is not re-parsing.} Three of Python's five LSP signals \textbf{cannot occur} in code \texttt{javac} accepts (invariant parameters, no narrowing). The Java detector needed a different \emph{signal set}, not just a re-parsed one.
  \item \textbf{Finding 5 --- Codex writes tests, not architecture.} Of 439 new Java classes Codex authored, \textbf{422 (96.1\%) are tests}; of 17 new production classes, \textbf{0 are flagged} as a god class --- while 12.9\% of the 976 production classes it \emph{touches} are, and every worst offender \textbf{predates} the PR.
  \item \textbf{Finding 8 --- half the corpus has nothing to invert into.} At package level, \textbf{median abstractness $A=0.000$ in both languages}: half of all placed packages declare no abstract class, interface, or protocol at all.
\end{itemize}
\end{frame}

\section{Conclusion}
% ============================================================================

\begin{frame}{Answering the research questions}
\small
\begin{description}[leftmargin=1.7cm,style=nextline]
\item[\textcolor{issatblue}{RQ1}] Agents differ, but only on \textbf{intra-unit} principles --- roughly 8 macro points on SRP (Devin best, Codex worst); OCP shows a real but tiny signal (11 PRs). Inter-unit principles show $\approx$0 per-PR signal for everyone.
\item[\textcolor{issatblue}{RQ2}] Yes, measurable --- at \textbf{two} operating points: crisp detectors (ISP, LSP after suppressors, P$\approx$1.0) and high-recall screens (SRP, OCP, DIP, P\,=\,0.33--0.68). Java's independent cross-check places the class instrument \emph{inside} the spread published tools show among themselves.
\item[\textcolor{issatblue}{RQ3}] Risk is \textbf{intra-unit}, replicated in Java; deterministic rules beat off-the-shelf ML on every gated experiment; portability is bounded by \textbf{language semantics} (what the compiler enforces) before it is bounded by tooling.
\end{description}
\end{frame}

\begin{frame}{Contributions}
\begin{description}[leftmargin=1.8em,labelsep=0.5em,itemsep=6pt,topsep=1pt]
  \item[\textbf{\color{issatgold!80!black}C1.}] A \textbf{validated, repository-scale instrument} for all five SOLID principles, in two languages, with precision/recall reported rather than assumed.
  \item[\textbf{\color{issatgold!80!black}C2.}] An \textbf{explicit map} of where structural SOLID measurement works (crisp detectors) and where it only screens (semantic ceilings) --- falsifiable, not a single pooled number.
  \item[\textbf{\color{issatgold!80!black}C3.}] An \textbf{empirical account} of where agent-authored risk concentrates: inside single units, not across them --- replicated in a language chosen to break the claim.
  \item[\textbf{\color{issatgold!80!black}C4.}] A \textbf{gated ML-vs-deterministic comparison}, with the negative ML result reported, not dropped.
  \item[\textbf{\color{issatgold!80!black}C5.}] A \textbf{demonstrated language port}: what carries over (pipeline shape, two metric cores) and what needed re-deriving (thresholds, one principle's signal set).
\end{description}
\end{frame}

\begin{frame}{Practical implications}
\begin{itemize}
  \item \textbf{Choosing between agents:} SOLID risk is a \emph{second axis} beside functional pass-rate --- Devin has the lowest God-Class rate among the five once repo mix and tests are controlled; Codex's Java output is conservative in a specific sense (96\% of new classes are tests).
  \item \textbf{Building a review gate on top of an agent:} an \textbf{LSP or ISP} flag can be trusted close to face value; an \textbf{SRP, OCP, or DIP} flag should trigger a human look, not an automatic rejection.
  \item The single cheapest, most reliable check available today: \texttt{refused\_bequest} (P\,=\,1.00, and independently corroborated by developers' own \texttt{\# type: ignore} comments).
\end{itemize}
\end{frame}

\begin{frame}{Limitations}
\begin{itemize}
  \item \textbf{Survivorship bias} --- the corpus is \emph{merged}, permissively-licensed, $\geq$25-star PRs only. This cannot separate ``agents rarely err'' from ``review catches their errors.''
  \item \textbf{The Java arm is one agent} (OpenAI Codex) --- a property of AIDev's coverage, not a design choice; no cross-language agent ranking is claimed.
  \item \textbf{Several counts are small} --- 11 OCP PRs, 10 DIP PRs, 4 Java DIP flags --- reported as counts, not smoothed into rates a single PR could move by several points.
\end{itemize}
\end{frame}

\begin{frame}{Future work}
\begin{itemize}
  \item \textbf{Measure rejected and abandoned PRs} --- the direct answer to the survivorship-bias limitation, and the single highest-value extension.
  \item \textbf{Draw the four outstanding Java holdouts} (OCP, LSP, ISP, DIP), so every Java figure becomes an out-of-sample estimate, as SRP's already is.
  \item \textbf{A churn-based, width-independent OCP signal} --- a class edited across many PRs is, by definition, not closed for modification; both corpora already carry the history.
  \item \textbf{A module- or function-scoped SRP proxy} for repositories where the class is not the dominant unit, motivated by Finding 8's Python package result (48.2\% have no class-ratio denominator).
\end{itemize}
\end{frame}

\begin{frame}[c]{Closing remark}
\begin{center}
\begin{minipage}{0.86\linewidth}
\large
The design faults autonomous coding agents leave in real, merged code are concentrated \textbf{\color{issatblue}inside single units} --- a class doing too much, a conditional that should have been an extension point --- and almost absent from the \textbf{\color{dipcolor}relations between units} that a single change rarely gets to author in full.

\vspace{0.4cm}
\normalsize\color{accentgray}\itshape
That distinction held across five agents and two languages chosen specifically to stress it. What remains open is whether it still holds at the scale where architecture is actually won or lost: a repository maintained over months, by an agent and by the humans who edit its output afterwards.
\end{minipage}
\end{center}
\end{frame}
