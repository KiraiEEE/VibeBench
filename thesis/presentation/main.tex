\section{Problem}
\begin{frame}{The benchmark--reality gap}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.52\textwidth}
\textbf{What we can already measure}
\begin{itemize}\itemsep3pt
  \item Agents now open pull requests on real repositories, unattended.
  \item Benchmarks answer one question: \textbf{do the tests pass?}
        (HumanEval, SWE-bench).
\end{itemize}
\vspace{6pt}
\textbf{What nobody measures}
\begin{itemize}\itemsep3pt
  \item Whether the code is \emph{structured} so the next person can change it.
  \item Code can be functionally correct and still be tightly coupled, poorly
        factored and expensive to extend.
\end{itemize}
\end{column}
\begin{column}{0.46\textwidth}
\begin{block}{The gap}
Functional correctness is a \textbf{floor}, not a design claim.\\[4pt]
A merged pull request tells you a maintainer accepted it. It does not tell you
what it did to the architecture.
\end{block}
\vspace{4pt}
{\footnotesize\color{accentgray} And most studies look only at \emph{merged}
code, which measures what reviewers accepted rather than what the agent wrote.}
\end{column}
\end{columns}
\end{frame}

% ===========================================================================
\begin{frame}{Research questions}
\begin{description}
  \item[\textbf{RQ1}] Do the three agents differ in SOLID adherence, and does
        the outcome of review (merged / rejected / open) relate to it?
  \item[\textbf{RQ2}] \textbf{Can SOLID be measured at scale, and how
        faithfully?}\\
        {\footnotesize\color{accentgray} This is the question the thesis takes
        most seriously: a proxy is only worth reporting if its fidelity is
        measured rather than assumed.}
  \item[\textbf{RQ3}] Where does the risk sit, \emph{inside} a unit or
        \emph{between} units?
\end{description}
\vspace{8pt}
\begin{center}
\keynum{4{,}245}{analysable Java PRs}\quad
\keynum{1{,}010}{repositories}\quad
\keynum{1{,}400}{judged entities}\quad
\keynum{3}{coding agents}
\end{center}
\end{frame}

\section{Method}
\begin{frame}{Five principles, five deterministic proxies}
\begin{center}\scriptsize
\begin{tabular}{@{}L{1.4cm}L{4.4cm}L{4.6cm}C{2.3cm}@{}}
\toprule
& \textbf{The forbidden smell} & \textbf{What is measured} & \textbf{Scope} \\
\midrule
\textcolor{srpcolor}{\textbf{SRP}} & god class &
seven-metric rubric + conceptual-cohesion suppressor & class \\
\textcolor{ocpcolor}{\textbf{OCP}} & type/tag switch &
dispatch-chain width + enum suppressor & dispatch chain \\
\textcolor{lspcolor}{\textbf{LSP}} & non-substitutable override &
the three faults \texttt{javac} cannot catch & override pair \\
\textcolor{ispcolor}{\textbf{ISP}} & fat interface &
contract width + implementer stub ratio & contract \\
\textcolor{dipcolor}{\textbf{DIP}} & concretion dependency &
newly cyclic module pairs & pull request \\
\bottomrule
\end{tabular}
\end{center}
\vspace{6pt}
\begin{block}{Why Liskov is only three signals in Java}
\small
\texttt{javac} already rejects a changed parameter list (an \emph{overload}), a
narrowed parameter type, and reduced visibility. Code that compiles cannot
contain those faults, so the proxy targets only what the compiler cannot see:
\textbf{refused bequest}, \textbf{strengthened precondition}, \textbf{weakened
postcondition}.
\end{block}
\end{frame}

  \item Merged, rejected \textbf{and} still-open requests.
\end{itemize}
\end{column}
\begin{column}{0.44\textwidth}
\begin{block}{A corpus of merged code is survivorship-biased}
It measures \textbf{what maintainers accepted}, not what the agent produced.
Including rejected and open requests makes the review filter itself observable.
\end{block}
\vspace{2pt}
{\footnotesize\color{accentgray}\textbf{Caveat, stated throughout:}
``rejected'' is mostly \texttt{closed\_unreviewed}, that is superseded or
stale, not a maintainer's objection on quality.}
\end{column}
\end{columns}

\section{RQ2, instrument validation}
\begin{frame}{The instrument has to be validated, not assumed}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.5\textwidth}
\textbf{Two independent axes}
\begin{enumerate}\small
  \item A \textbf{reading judge} labels 1{,}400 entities, 100 flagged and 100
        unflagged per principle, and records a reason for every verdict.
  \item \textbf{CK, PMD and DesigniteJava} run over the \emph{same} source
        trees, because a judge can be wrong in the same direction as the rubric.
\end{enumerate}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Two guards that make it trustworthy}
\small
\textbf{Both strata are sampled.} A sample drawn only from flagged rows can
measure precision but can \emph{never} discover a violation the detector
missed.\\[4pt]
\textbf{An audit gate.} Every judged row must trace back to the source file the
judge was shown, and no batch may reveal the detector's own answer.
\end{block}
\end{column}
\end{columns}
\end{frame}

% ===========================================================================
\begin{frame}{RQ2: what the instrument actually does}
\vspace{-4pt}
\begin{center}
\resizebox{0.86\textwidth}{!}{\input{figures/fig_validation_pr.tex}}
\end{center}
\vspace{-6pt}
{\footnotesize Precision below 0.5 for four principles is \textbf{the finding},
not a defect: a structural proxy asks a syntactic question and the principles
are semantic. The gap is what these numbers quantify.}
\end{frame}

% ===========================================================================
\begin{frame}{The holdout: two refinements, opposite fates}
Two quantities were \emph{fitted} to the first sample. Scoring them on that
same sample reports optimism, so both were re-scored on \textbf{200 fresh
entities each}, excluding every previously judged row.
\vspace{6pt}
\begin{center}\small
\begin{tabular}{@{}L{6.0cm}ccc@{}}
\toprule
\textbf{Refinement} & \textbf{Fitted sample} & \textbf{Holdout} & \textbf{Pooled} \\
\midrule
LSP restricted to \emph{refused bequest} &
1.000 \ (31/31) & \textcolor{goodgreen}{\textbf{1.000}} \ (33/33) & 1.000 \ (64/64) \\
ISP restricted to $\geq$30 members &
0.862 \ (25/29) & \textcolor{ispcolor}{\textbf{0.704}} \ (19/27) & 0.786 \ (44/56) \\
\bottomrule
\end{tabular}
\end{center}
\vspace{6pt}
\begin{block}{Why they came apart}
\textbf{Refused bequest is categorical}: a body that is nothing but a
\texttt{throw} either is or is not that shape. There is nothing to tune.\\
\textbf{``30 members'' is a numeric threshold} read off the false positives it
was then scored against, so it absorbed some of that sample's noise. Its fitted
value lies \emph{outside} the holdout's own interval (0.515--0.841).
\end{block}
\end{frame}

% ===========================================================================
\begin{frame}{Recall must be weighted, and that exposed a blind spot}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.47\textwidth}
Each sample is half flagged, half unflagged. \textbf{The corpus is not.}
For Liskov: 1{,}546 flagged override pairs against \textbf{79{,}689}
unflagged, so one sampled unflagged row stands for $\approx$797 entities and a
flagged one for 15.
\vspace{5pt}
\begin{center}\small
\begin{tabular}{@{}lcc@{}}
\toprule
& \textbf{raw} & \textbf{weighted} \\
\midrule
SRP & 0.972 & 0.875 \\
OCP & 0.943 & 0.733 \\
ISP & 0.914 & 0.552 \\
DIP & 1.000 & 1.000 \\
\textbf{LSP} & 0.966 & \textcolor{ispcolor}{\textbf{0.352}} \\
\bottomrule
\end{tabular}
\end{center}
\vspace{3pt}
{\scriptsize Flag rates bound SRP, OCP and DIP, \emph{not} LSP and ISP.}
\end{column}
\begin{column}{0.51\textwidth}
\begin{block}{The three Liskov violations the detector missed}
\small
All three are \textbf{silent} refusals, not declared ones:
\begin{itemize}\itemsep2pt\small
  \item a Vaadin root node throws for \emph{every} non-null parent its base
        accepts;
  \item a CommCare controller overrides a method that disables a login button
        with an \textbf{empty body};
  \item an nd4j operation implements \texttt{setY} as \texttt{this.y = null},
        a setter that discards its argument.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

% ===========================================================================
\begin{frame}{Second axis: agreement with published tools}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.53\textwidth}
\footnotesize
150 repositories, \textbf{111{,}029} shared production types. No tool is
treated as ground truth.
\vspace{5pt}
\begin{tabular}{@{}L{4.6cm}c@{}}
\toprule
\textbf{Pair (single-responsibility)} & $\boldsymbol{\kappa}$ \\
\midrule
ours \textasciitilde{} PMD \texttt{GodClass}  & \textbf{0.574} \\
ours \textasciitilde{} DesigniteJava          & \textbf{0.531} \\
PMD \textasciitilde{} DesigniteJava           & 0.384 \\
\bottomrule
\end{tabular}
\vspace{6pt}

Rank correlation with CK: \textbf{wmc 0.977}, methods 0.969, loc 0.944.
\end{column}
\begin{column}{0.45\textwidth}
\begin{block}{The result that matters}
\small
The two \emph{established} tools agree with each other \textbf{less}
($\kappa=0.384$) than either agrees with the rubric built here.\\[4pt]
\textbf{68.0\%} of our flags (4{,}304 of 6{,}326) are corroborated by at least
one independent tool.
\end{block}
\vspace{2pt}
{\footnotesize\color{accentgray} Cohesion is the exception (0.288), but CK's own
two cohesion variants agree only at 0.490 with each other, so the disagreement
is in the concept rather than in our implementation.}
\end{column}
\end{columns}
\end{frame}

\begin{block}{Three qualifications, all stated in the thesis}
\small
\textbf{Era.} Median PR dates are seven months apart (Codex Aug 2025, Copilot
Jan 2026, Claude Code Mar 2026). Agent and era are confounded.\\
\textbf{Thin cells.} Codex contributes 770 PRs, so its intervals are wide.\\
\textbf{Change size.} See the next slide, this one is decisive.
\end{block}

\begin{column}{0.5\textwidth}
\begin{block}{Merged and rejected code is structurally indistinguishable}

\end{block}

\section{RQ3, locus of risk}
\begin{frame}{RQ3: risk is intra-unit, not inter-unit}
\begin{center}\small
\begin{tabular}{@{}lrrr@{}}
\toprule
\textbf{Principle} & \textbf{Flagged entities present} & \textbf{Est.\ genuine} &
\textbf{PRs that introduced one} \\
\midrule
SRP & 11{,}822 classes        & $\approx$4{,}100 & n/a \\
OCP & 2{,}577 dispatch chains & $\approx$850 & 286 \ (2.2\%) \\
ISP & 1{,}090 contracts       & $\approx$350 & 276 \ (2.1\%) \\
LSP & 681 override pairs      & $\approx$270 & \textbf{16} \ (0.4\%) \\
DIP & 195 cycles              & $\approx$175 & 195 \ (4.1\%) \\
\bottomrule
\end{tabular}
\end{center}
\vspace{4pt}
{\scriptsize\color{accentgray} Both columns corrected for measured precision.}
\vspace{6pt}
\begin{block}{The structural explanation}
\small
An \textbf{intra-unit} smell (SRP, OCP) lives inside one class, so a single
pull request can author it. An \textbf{inter-unit} smell (LSP, ISP, DIP) exists
only in a \emph{relation} between two units, and one change rarely opens both
ends. 681 refused-bequest relationships exist in the code the agents touched;
they introduced \textbf{16}.
\end{block}
\end{frame}

\section{Conclusion}
\begin{frame}{Answering the research questions}
\begin{description}
  \item[\textbf{RQ1}] The agents differ, Claude Code highest on all five and
        Codex lowest on four, \textbf{but the ordering is conditional on change
        size} and agent is confounded with era. Review does not select on these
        properties.
  \item[\textbf{RQ2}] SOLID \emph{can} be measured at scale, with an operating
        characteristic that must be stated rather than collapsed into one
        number. Precision 0.330--0.890; weighted recall 0.352--1.000. Two
        detectors are repairable from their own labels, and only the
        \textbf{categorical} repair survived a holdout.
  \item[\textbf{RQ3}] Risk is \textbf{intra-unit}. Agents author god classes
        and type switches; they almost never author a broken subtype relation,
        because that takes two coordinated changes.
\end{description}
\end{frame}

% ===========================================================================
\begin{frame}{Contributions}
\begin{enumerate}
  \item A \textbf{benchmark over real, attributable units}: 4{,}245 Java pull
        requests across 1{,}010 repositories, with merged, rejected and open
        requests alike.
  \item \textbf{All five SOLID principles} operationalised as deterministic
        proxies, each with \emph{measured} construct validity rather than
        assumed proxy fidelity.
  \item An \textbf{out-of-sample holdout} that separates a refinement which
        generalises from one that was partly fitted, and a demonstration that
        stratified recall must be weighted before it means anything.
  \item The \textbf{intra/inter-unit distinction} as an empirical result:
        per-change measurement systematically cannot see inter-unit damage.
\end{enumerate}
\end{frame}

% ===========================================================================
\begin{frame}{Limitations}
\begin{itemize}\itemsep5pt\small
  \item \textbf{Agreement is rubric versus reading judge.} LLM judging of SOLID
        is a validated technique (Pehlivan et al., 2025), but the judge can err
        in the same direction as the rubric. The external-tool axis is the
        defence, and it covers SRP only, since no published source-only Java
        tool has an OCP or ISP rule.
  \item \textbf{The judge and one agent are both Claude.} Tested directly: on
        four of five principles Claude Code's flags are confirmed \emph{more}
        often, which is the opposite of leniency.
  \item \textbf{``Rejected'' mostly means superseded}, not refused on quality.
  \item \textbf{Source-only analysis} cannot resolve supertypes declared
        outside the repository: median resolution 87.0\%, so LSP and ISP
        under-count in framework-heavy projects. The direction is favourable,
        an unresolved supertype removes a candidate rather than inventing one.
  \item \textbf{Weighted recall for LSP and ISP rests on few observations}, so
        its intervals are wide.
  \item \textbf{Java only.} Nothing here is claimed for other languages.
\end{itemize}
\end{frame}

% ===========================================================================
\begin{frame}{Future work}
\begin{itemize}\itemsep6pt
  \item \textbf{A churn-based Open/Closed signal.} Openness is uncorrelated
        with dispatch width, so the obvious threshold is the wrong instrument;
        a class edited across many pull requests is by definition not closed.
  \item \textbf{A closed-set suppressor that recognises external
        specifications} (JDBC type codes, protobuf scalars, Unicode blocks),
        which are closed in fact but open to the compiler.
  \item \textbf{Narrow the recall intervals} for the two blind detectors with a
        sample drawn only from unflagged entities, aimed at \emph{silent}
        refusals.
  \item \textbf{A second language} with the same agent coverage: the framework
        is language-parameterised, only fact extraction is language-specific.
\end{itemize}
\end{frame}
