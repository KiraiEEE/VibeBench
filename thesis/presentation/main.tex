% ============================================================== context
\begin{frame}{Coding agents now submit real changes}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.56\textwidth}
\begin{itemize}\setlength{\itemsep}{5pt}
  \item Claude Code, GitHub Copilot and OpenAI Codex take a task and open a pull request.
  \item The AIDev dataset records \textbf{7.69 million} agent pull requests in 957k repositories.
  \item They are evaluated mainly on whether tests pass (HumanEval, SWE-bench).
  \item Passing tests says nothing about \textbf{design}: whether the next developer can change
        the code.
\end{itemize}
\end{column}
\begin{column}{0.4\textwidth}
\keynum[4.6cm]{about half}{of SWE-bench Verified patches that pass the tests would not be merged
by maintainers (METR, 2026)}\\[6pt]
\keynum[4.6cm]{77\%}{of long-horizon agent trajectories show growing structural erosion
(SlopCodeBench, 2026)}
\end{column}
\end{columns}
\end{frame}

% ============================================================== gap + RQ
\begin{frame}{The gap, and what this thesis does}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\textbf{What existing work does}
\begin{itemize}\setlength{\itemsep}{3pt}\small
  \item Benchmarks on functions or small task sets
  \item LLMs detecting SOLID violations in examples
  \item Smell counts on merged code only
  \item Metrics assumed to capture a principle
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{What this thesis does}
\begin{itemize}\setlength{\itemsep}{3pt}\small
  \item Real pull requests from three agents
  \item All five SOLID principles, one detector each
  \item Merged, rejected \emph{and} open requests
  \item Every detector's precision and recall measured
  \item The whole study repeated in a second language
\end{itemize}
\end{column}
\end{columns}
\vspace{6pt}
\takeaway{Unit of analysis: one pull request. Object of measurement: what that pull request did to
the design of the code it touched.}
\end{frame}

\begin{frame}{Where this study sees it differently}
\scriptsize
\begin{tabular}{@{}L{3.1cm}L{4.5cm}L{3.9cm}@{}}
\toprule
\textbf{Earlier finding} & \textbf{What we measured} & \textbf{Outcome} \\
\midrule
SRP easiest to detect, DIP hardest \textit{(Pehlivan et al.)} &
For \emph{structural} detectors the order reverses: DIP is the most precise
(\textbf{0.892} Java, \textbf{1.000} Python), SRP the least (\textbf{0.350}, \textbf{0.110}) &
\textcolor{ispcolor}{\textbf{Reversed}}, because a model reads examples while a static rule computes
a whole-repository graph \\
\addlinespace
Agents succeed at only 4.3\% of dependency-control tasks \textit{(Zhu, Haichao, NITR)} &
Agents \textbf{seldom introduce} dependency or responsibility violations into existing repositories &
\textcolor{issatgold!80!black}{\textbf{Different question}}: that benchmark asks agents to
\emph{repair} structure, we ask whether they \emph{damage} it \\
\addlinespace
Erosion grows across 77\% of long-horizon trajectories \textit{(SlopCodeBench)} &
The time coefficient is \textbf{inert} for every delta principle, in both languages &
\textcolor{accentgray}{\textbf{Not tested}}: our unit is one pull request, not a trajectory of
successive edits \\
\bottomrule
\end{tabular}
\vspace{4pt}
\takeaway{Taken together with Pehlivan: a graph-based detector and a reading model fail on opposite
principles, so they should be combined, not chosen between.}
\end{frame}

% ============================================================== SRP
\begin{frame}{SRP: God Class rubric with a cohesion suppressor}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.56\textwidth}
\small
Six size and complexity metrics; one \emph{hit} per threshold reached:
\[ h=\#\{m:\ \text{metric}_m \ge \tau_m\} \]
Cohesion, LCOM4 over the method graph (edge = shared field or a call), constructors excluded:
\[ \mathrm{LCOM}=\frac{c-1}{n-1} \]
Conceptual cohesion over TF-IDF vectors of method token bags:
\[ C3=\operatorname*{mean}_{i<j}\cos(v_i,v_j) \]
\[ \text{flag}=(h\ge 2)\ \wedge\ \neg\,(C3\ge 0.4 \wedge h<3) \]
\end{column}
\begin{column}{0.4\textwidth}
\footnotesize
\begin{tabular}{@{}lcc@{}}
\toprule
\textbf{Threshold} & \textbf{Java} & \textbf{Python} \\
\midrule
class lines & 362 & 250 \\
weighted methods & 45 & 36 \\
methods & 20 & 10 \\
method complexity & 11 & 12 \\
parameters & 5 & 7 \\
LCOM & 0.86 & 0.8 \\
\bottomrule
\end{tabular}\\[6pt]
{\scriptsize Each value is the 90th percentile of that corpus, fixed before any label existed.}\\[6pt]
\keynum[3.6cm]{0.350 / 0.110}{precision, Java / Python}
\end{column}
\end{columns}
\end{frame}

% ============================================================== OCP
\begin{frame}{OCP: dispatch width with a closed-set suppressor}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.56\textwidth}
\small
A dispatch chain is an \texttt{if}/\texttt{else if} ladder or a \texttt{switch} branching on one
subject. With $b$ branches:
\[ \text{smell}=(b\ge\tau)\ \wedge\ \neg\,\mathrm{closed}(\text{subject}) \]
$\mathrm{closed}$ holds when the subject's resolved type is an enumeration: a client cannot add a
value.\\[4pt]
A pull request is flagged when it \textbf{creates} or \textbf{widens} such a chain.
\end{column}
\begin{column}{0.4\textwidth}
\footnotesize
\begin{tabular}{@{}lcc@{}}
\toprule
$\tau$ & \textbf{Java} & \textbf{Python} \\
\midrule
type chains & 3 & 3 \\
tag / value chains & 8 & 5 \\
\bottomrule
\end{tabular}\\[8pt]
\keynum[3.6cm]{0.330 / 0.390}{precision, Java / Python}\\[6pt]
{\scriptsize Width does not predict truth: precision stays between 0.270 and 0.400 across every
band. The sets that are closed in practice are fixed by specifications, not by the language.}
\end{column}
\end{columns}
\end{frame}

% ============================================================== LSP
\begin{frame}{LSP: override compared with its nearest ancestor}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.54\textwidth}
\small
For each pair (subclass method, nearest ancestor method with the same signature and a real body):
\[ \text{violation}=\exists\,s\in \text{signals}(pair):\ s\in S \]
\textbf{Java} $S$ = refused bequest, strengthened precondition, weakened postcondition.\\[3pt]
\textbf{Python} $S$ adds what no compiler checks: parameter removed, required parameter added,
parameter type narrowed, return type widened.
\end{column}
\begin{column}{0.42\textwidth}
\footnotesize
\begin{tabular}{@{}lc@{}}
\toprule
\textbf{Java signal} & \textbf{precision} \\
\midrule
refused bequest & 31/31 \\
weakened postcondition & 6/36 \\
strengthened precondition & 3/33 \\
\bottomrule
\end{tabular}\\[6pt]
{\scriptsize The shipped Java flag keeps refused bequest only: 64 of 64 across two samples.}\\[6pt]
\keynum[3.6cm]{0.420 / 0.510}{precision of the full signal set, Java / Python}
\end{column}
\end{columns}
\end{frame}

% ============================================================== ISP
\begin{frame}{ISP: contract width and implementer stubs}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.56\textwidth}
\small
A contract is an interface or abstract class with at least one abstract member and at least one
implementer inside the repository. Width $w$ = abstract members; for each implementer,
\[ s=\frac{\#\text{stubbed members}}{\#\text{members it provides}} \]
\textbf{Java:} contract fat if $w\ge 13$; forced stub if stubs $\ge 2$ and $s\ge 0.5$.
Flag if the request creates a fat contract, widens one, or leaves a forced stub.\\[3pt]
\textbf{Python:} flag if at least \textbf{two} of: $w>7$; mean implementer coverage $<0.75$ over
$\ge2$ implementers; an implementer with forced stubs.
\end{column}
\begin{column}{0.4\textwidth}
\footnotesize
\begin{tabular}{@{}lrr@{}}
\toprule
\textbf{Members} & \textbf{judged} & \textbf{precision} \\
\midrule
0--17 & 43 & 0.116 \\
18--29 & 28 & 0.071 \\
30--49 & 14 & 0.786 \\
50--99 & 10 & 0.900 \\
$\geq$100 & 5 & 1.000 \\
\bottomrule
\end{tabular}\\[6pt]
{\scriptsize Width does separate violations here, which is why 13 was too low a bar.}\\[6pt]
\keynum[3.6cm]{0.335 / 0.425}{precision, Java / Python}
\end{column}
\end{columns}
\end{frame}

% ============================================================== DIP
\begin{frame}{DIP: new cycles in the import graph}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.56\textwidth}
\small
Build the internal import graph before and after the change and take strongly connected components
with Tarjan's algorithm. With $P(G)$ the set of module pairs inside one component:
\[ \text{new}=P(G_{\text{after}})\setminus P(G_{\text{before}}) \]
\[ \text{flag}=\exists (u,v)\in\text{new}:\ u\ \text{or}\ v\ \text{was changed} \]
Also computed, as description only: $I=\frac{C_e}{C_a+C_e}$, $A=\frac{N_a}{N_c}$,
$D=|A+I-1|$.
\end{column}
\begin{column}{0.4\textwidth}
\keynum[3.9cm]{0.892 / 1.000}{precision, Java / Python}\\[6pt]
{\scriptsize Java precision is exact: all 195 flagged pull requests were judged. In Python, 100 of
173 were judged and all were correct.}\\[8pt]
{\scriptsize The distance $D$ was tested as a change signal and confirmed \textbf{none} of 17
flags, so it describes packages and never raises a flag.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Findings}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
\begin{itemize}\setlength{\itemsep}{4pt}\small
  \item[\textbf{1}] What a compiler enforces decides which violations can exist: 46 of 51 confirmed
        Python Liskov violations are faults \texttt{javac} rejects.
  \item[\textbf{2}] Width predicts Interface Segregation violations and not Open/Closed ones,
        because one principle is about size and the other about whether a set can grow.
  \item[\textbf{3}] The closed sets that matter are fixed by specifications, not by the language.
\end{itemize}
\end{column}
\begin{column}{0.49\textwidth}
\begin{itemize}\setlength{\itemsep}{4pt}\small
  \item[\textbf{4}] Most flagged problems in the touched code predate the change.
  \item[\textbf{5}] Classes the agents wrote are flagged less often than the classes they edit.
  \item[\textbf{6}] Merged code is not a biased sample of unreviewed agent code in Java, but it is
        cleaner than abandoned code in Python.
\end{itemize}
\end{column}
\end{columns}

\begin{columns}[T,onlytextwidth]

\textbf{Contributions}

  \item A validated SOLID instrument over real agent pull requests, all five principles
  \item A corpus with merged, rejected and open requests
  \item A map of which principles a rule measures and which it only screens
  \item Evidence in two languages on where design risk sits
  \item A replication with the same agents in a second language
\end{itemize}

\begin{itemize}\setlength{\itemsep}{3pt}\small
  \item Add \textbf{refused bequest}: 64/64 in Java, 6/7 in Python
  \item In dynamic languages, also check that an override still accepts the base's calls
  \item Do not rely on interface-width thresholds
  \item Treat SRP and OCP flags as prompts for discussion, not as blockers
  \item Structural checks cannot see relation-level damage: that needs repository history

\begin{frame}{Limitations}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
\begin{itemize}\setlength{\itemsep}{3pt}\small
  \item SOLID adherence is not measured design quality or cost; nothing here observes an outcome
  \item Labels come from an LLM judge of the same family as one agent; 89.7\% self-agreement bounds
        every figure
  \item No independent-tool check for OCP, ISP, or for Python at all
  \item Agents chose their own tasks: these are associations, not causes
\end{itemize}
\end{column}
\begin{column}{0.49\textwidth}
\begin{itemize}\setlength{\itemsep}{3pt}\small
  \item Few reviewed rejections (171 Java, 196 Python)
  \item One snapshot per pull request: slow erosion is invisible
  \item Both corpora come from one dataset; 7.0\% of Python source is gone
  \item Small counts for the relation-level principles, so their intervals are wide
  \item Python Interface Segregation is defined differently, so raw rates are not comparable
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Backup: why recall must be weighted}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.52\textwidth}
\small
Each sample is half flagged, the population is not. For Java Liskov it holds 574 flagged override
pairs and 25,594 unflagged ones.
\[ R=\frac{TP\cdot N_f/n_f}{TP\cdot N_f/n_f+FN\cdot N_u/n_u} \]
Counting the strata as equally common gives 0.966. Weighting gives \textbf{0.386}.\\[4pt]
{\footnotesize The population is the one the sample was drawn from: production entities, counted
once each.}
\end{column}
\begin{column}{0.44\textwidth}
\begin{itemize}\setlength{\itemsep}{3pt}\small
  \item The intervals are wide because a weighted recall rests on the few violations found in the
        unflagged stratum (three for Liskov, in 200 rows).
  \item This is why more labelling would help recall, not precision.
\end{itemize}
\end{column}
