\newcommand{\gearicon}[3]{
  \begin{scope}[shift={#1}]
    \foreach \a in {0,45,...,315}{
      \fill[#3,rotate=\a] (-0.075,#2*0.60) rectangle (0.075,#2*0.60+0.19);
    }
    \fill[#3] (0,0) circle (#2*0.60);
    \fill[white] (0,0) circle (#2*0.60*0.42);
  \end{scope}
}
