% Figure: the five SOLID principles, redesigned as a "rulebook" of five
% numbered rule cards (badge + imperative + rationale), instead of a wheel.
\begin{tikzpicture}[
  every node/.style={font=\footnotesize},
  card/.style={draw,thick,rounded corners=3pt,fill=white,minimum width=11.4cm,
               minimum height=1.15cm,anchor=north west},
  badge/.style={circle,draw,very thick,minimum size=0.95cm,font=\bfseries\large},
]
\node[card,draw=srpcolor,fill=srpcolor!6] (c0) at (0,0) {};
\node[badge,draw=srpcolor,fill=srpcolor!16,text=srpcolor] at (c0.west) {S};
\node[anchor=west,font=\bfseries\small,text=srpcolor] at ($(c0.west)+(1.15,0.28)$) {Single Responsibility};
\node[anchor=west,font=\scriptsize,text=darktext,align=left,text width=9.7cm] at ($(c0.west)+(1.15,-0.22)$)
  {A class shall have one, and only one, reason to change.};

\node[card,draw=ocpcolor,fill=ocpcolor!6] (c1) at (0,-1.35) {};
\node[badge,draw=ocpcolor,fill=ocpcolor!16,text=ocpcolor] at (c1.west) {O};
\node[anchor=west,font=\bfseries\small,text=ocpcolor] at ($(c1.west)+(1.15,0.28)$) {Open/Closed};
\node[anchor=west,font=\scriptsize,text=darktext,align=left,text width=9.7cm] at ($(c1.west)+(1.15,-0.22)$)
  {Open for extension, closed for modification --- add behaviour, don't edit it.};

\node[card,draw=lspcolor,fill=lspcolor!6] (c2) at (0,-2.70) {};
\node[badge,draw=lspcolor,fill=lspcolor!16,text=lspcolor] at (c2.west) {L};
\node[anchor=west,font=\bfseries\small,text=lspcolor] at ($(c2.west)+(1.15,0.28)$) {Liskov Substitution};
\node[anchor=west,font=\scriptsize,text=darktext,align=left,text width=9.7cm] at ($(c2.west)+(1.15,-0.22)$)
  {A subtype must be usable anywhere its base type is expected.};

\node[card,draw=ispcolor,fill=ispcolor!6] (c3) at (0,-4.05) {};
\node[badge,draw=ispcolor,fill=ispcolor!16,text=ispcolor] at (c3.west) {I};
\node[anchor=west,font=\bfseries\small,text=ispcolor] at ($(c3.west)+(1.15,0.28)$) {Interface Segregation};
\node[anchor=west,font=\scriptsize,text=darktext,align=left,text width=9.7cm] at ($(c3.west)+(1.15,-0.22)$)
  {No client shall be forced to depend on methods it does not use.};

\node[card,draw=dipcolor,fill=dipcolor!6] (c4) at (0,-5.40) {};
\node[badge,draw=dipcolor,fill=dipcolor!16,text=dipcolor] at (c4.west) {D};
\node[anchor=west,font=\bfseries\small,text=dipcolor] at ($(c4.west)+(1.15,0.28)$) {Dependency Inversion};
\node[anchor=west,font=\scriptsize,text=darktext,align=left,text width=9.7cm] at ($(c4.west)+(1.15,-0.22)$)
  {Depend on abstractions --- never on concrete implementation details.};
\end{tikzpicture}
