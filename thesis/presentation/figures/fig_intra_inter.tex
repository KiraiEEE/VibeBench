% Figure: Finding 1 (risk is intra-unit, not inter-unit), redesigned as a
% radar/spider chart of each principle's own flagged-population share on
% the Python corpus (452 analysable PRs; SRP is class-scoped, macro mean
% across the five agents) --- the dramatic SRP/OCP spike vs.\ the near-zero
% LSP/ISP/DIP axes IS the intra-vs-inter-unit dissociation, read off data
% rather than illustrated with an abstract box diagram.
\begin{tikzpicture}[font=\scriptsize]
\def\Rmax{2.55}
% background grid rings at 5/10/15/20 %
\foreach \frac/\lbl [count=\i from 1] in {0.25/5,0.5/10,0.75/15,1.0/20}{
  \draw[gray!28,thin] (90:\Rmax*\frac) -- (18:\Rmax*\frac) -- (-54:\Rmax*\frac)
        -- (234:\Rmax*\frac) -- (162:\Rmax*\frac) -- cycle;
}
\node[font=\tiny,text=accentgray,anchor=west] at (0.06,0.63*\Rmax) {10\%};
\node[font=\tiny,text=accentgray,anchor=west] at (0.06,1.0*\Rmax) {20\%};

% axis spokes
\foreach \a in {90,18,-54,234,162}{
  \draw[gray!45,thin] (0,0) -- (\a:\Rmax);
}

% data polygon: SRP 17.5% (macro mean, class-scoped) / OCP 2.4% / LSP 0.22% / ISP 0.44% / DIP 2.2%
\coordinate (vsrp) at (90:{\Rmax*17.5/20});
\coordinate (vocp) at (18:{\Rmax*2.4/20});
\coordinate (vlsp) at (-54:{\Rmax*0.22/20});
\coordinate (visp) at (234:{\Rmax*0.44/20});
\coordinate (vdip) at (162:{\Rmax*2.2/20});
\fill[issatblue!16] (vsrp) -- (vocp) -- (vlsp) -- (visp) -- (vdip) -- cycle;
\draw[issatblue,thick] (vsrp) -- (vocp) -- (vlsp) -- (visp) -- (vdip) -- cycle;

\filldraw[srpcolor] (vsrp) circle (2.4pt);
\filldraw[ocpcolor] (vocp) circle (2.4pt);
\filldraw[lspcolor] (vlsp) circle (2.4pt);
\filldraw[ispcolor] (visp) circle (2.4pt);
\filldraw[dipcolor] (vdip) circle (2.4pt);

% axis end labels + values
\node[font=\scriptsize\bfseries,text=srpcolor,anchor=south,align=center] at (90:\Rmax+0.42) {SRP\\\footnotesize\normalfont 17.5\%};
\node[font=\scriptsize\bfseries,text=ocpcolor,anchor=south west,align=center] at (18:\Rmax+0.42) {OCP\\\footnotesize\normalfont 2.4\%};
\node[font=\scriptsize\bfseries,text=lspcolor,anchor=north west,align=center] at (-54:\Rmax+0.5) {LSP\\\footnotesize\normalfont 0.22\%};
\node[font=\scriptsize\bfseries,text=ispcolor,anchor=north east,align=center] at (234:\Rmax+0.5) {ISP\\\footnotesize\normalfont 0.44\%};
\node[font=\scriptsize\bfseries,text=dipcolor,anchor=south east,align=center] at (162:\Rmax+0.42) {DIP\\\footnotesize\normalfont 2.2\%};
\end{tikzpicture}
