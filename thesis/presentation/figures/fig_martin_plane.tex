% Figure: Martin's Abstractness/Instability plane with zones.
%
% Layout note: the four corner descriptors sit OUTSIDE the square. Inside, they
% collided with the main-sequence band at the top left and with the zone labels
% at both other corners, and the two of them nearest the border were clipped by
% it. Only one "intermediate" label is drawn, placed off the band, because two
% of them invited the reading that there are two distinct intermediate regions.
\begin{tikzpicture}[scale=5.8]
\draw[-{Latex[length=2mm]},thick] (0,0) -- (1.16,0) node[right,font=\footnotesize] {Instability $I$};
\draw[-{Latex[length=2mm]},thick] (0,0) -- (0,1.16) node[above,font=\footnotesize] {Abstractness $A$};
\node[below left,font=\scriptsize] at (0,0) {0};
\node[below,font=\scriptsize] at (1,0) {1};
\node[left,font=\scriptsize] at (0,1) {1};

% main sequence band, A + I = 1 plus or minus 0.15
\begin{scope}
  \clip (0,0) rectangle (1,1);
  \fill[issatgold!30] (-0.15,1.15) -- (1.15,-0.15) -- (0.85,-0.15) -- (-0.15,0.85) -- cycle;
\end{scope}
\draw[issatgold!70!black,thick] (0,1) -- (1,0);

% zone of pain, corner (0,0); zone of uselessness, corner (1,1)
\begin{scope}
  \clip (0,0) rectangle (1,1);
  \fill[ispcolor!22] (0,0) circle (0.5);
  \fill[lspcolor!22] (1,1) circle (0.5);
\end{scope}

\draw[thick] (0,0) rectangle (1,1);

% zone labels, kept clear of the band and of the corner descriptors
\node[align=center,font=\scriptsize\bfseries,text=ispcolor!70!black] at (0.17,0.15)
  {zone of\\pain};
\node[align=center,font=\scriptsize\bfseries,text=lspcolor!55!black] at (0.83,0.85)
  {zone of\\uselessness};
\node[align=center,font=\scriptsize\bfseries,rotate=-45,text=issatgold!55!black] at (0.5,0.55)
  {main sequence};
% Two intermediate regions exist, not one: the band splits the square, and each
% half keeps a slice outside its corner zone. Both labels are placed where
% A+I clears the band by more than 0.15 and the point is more than 0.5 from the
% zone corner, so neither sits on the band nor inside a shaded circle.
\node[align=center,font=\scriptsize,text=accentgray] at (0.70,0.55) {intermediate};
\node[align=center,font=\scriptsize,text=accentgray] at (0.30,0.45) {intermediate};

% corner descriptors, OUTSIDE the square so nothing is clipped or overlapped
\node[font=\tiny\itshape,align=right,anchor=south east,text=accentgray] at (-0.02,1.00)
  {abstract,\\stable};
\node[font=\tiny\itshape,align=left,anchor=south west,text=accentgray] at (1.02,1.00)
  {abstract,\\unstable};
\node[font=\tiny\itshape,align=right,anchor=north east,text=accentgray] at (-0.02,-0.03)
  {concrete,\\stable};
\node[font=\tiny\itshape,align=left,anchor=north west,text=accentgray] at (1.02,-0.03)
  {concrete,\\unstable};
\end{tikzpicture}
