% Figure: Martin's Abstractness/Instability plane, redesigned with
% corner-hugging quarter-disc zones (instead of full circles clipped by the
% square), a hatched main-sequence band (instead of a flat fill), and axis
% tick marks for a more "measured/plotted" look.
\begin{tikzpicture}[scale=5.7]
\draw[-{Latex[length=2mm]},thick] (0,0) -- (1.14,0) node[right,font=\footnotesize] {Instability $I$};
\draw[-{Latex[length=2mm]},thick] (0,0) -- (0,1.14) node[above,font=\footnotesize] {Abstractness $A$};
\foreach \t in {0.25,0.5,0.75,1}{
  \draw[accentgray,thin] (\t,0) -- (\t,-0.018) node[below,font=\tiny] {\t};
  \draw[accentgray,thin] (0,\t) -- (-0.018,\t) node[left,font=\tiny] {\t};
}
\node[below left,font=\scriptsize] at (0,0) {0};

% hatched main-sequence band A + I = 1, +/- 0.15
\begin{scope}
  \clip (0,0) rectangle (1,1);
  \foreach \o in {-0.13,-0.09,-0.05,-0.01,0.03,0.07,0.11}{
    \draw[issatgold!65,line width=0.55pt] (-0.2+\o,1.2) -- (1.2+\o,-0.2);
  }
\end{scope}
\draw[issatgold!75!black,very thick] (0,1) -- (1,0);

% zone of pain: quarter disc hugging (0,0)
\begin{scope}
  \clip (0,0) rectangle (1,1);
  \fill[ispcolor!24] (0,0) -- (0.62,0) arc[start angle=0,end angle=90,radius=0.62] -- cycle;
\end{scope}
% zone of uselessness: quarter disc hugging (1,1)
\begin{scope}
  \clip (0,0) rectangle (1,1);
  \fill[lspcolor!24] (1,1) -- (0.38,1) arc[start angle=180,end angle=270,radius=0.62] -- cycle;
\end{scope}

\draw[thick] (0,0) rectangle (1,1);

\node[align=center,font=\scriptsize\bfseries,text=ispcolor!70!black] at (0.22,0.22) {zone of\\pain};
