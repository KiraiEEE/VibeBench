% ---------------------------------------------------------------------------
% 5. Chapter / section heading style
% ---------------------------------------------------------------------------
% Adapted from TeX.SX #192940 (user2478, CC BY-SA 3.0) — boxed-numeral
% panel with outer frame, italic title, and horizontal rule.
% Source: https://tex.stackexchange.com/a/192940
\makeatletter
\def\@makechapterhead#1{%
  \vspace*{10\p@}%
  {\parindent\z@
   \hbox{%
     \vbox{%
       \hsize=7mm%
       \begin{tabular}{@{}p{7mm}@{}}
         \makebox[7mm]{\scshape\strut\small\chaptername}\\
         \makebox[7mm]{\cellcolor{black}\Huge\color{white}\bfseries\strut\thechapter\rule[-4cm]{0pt}{4cm}}%
       \end{tabular}%
       \makebox(0,0){\put(-10,-100){\fbox{\phantom{\rule[-4cm]{7mm}{4cm}}}}}%
     }%
     \kern-2pt
     \vbox to 0pt{%
       \tabular[t]{@{}p{1cm}p{\dimexpr\hsize-2.1cm}@{}}\hline
         & \Huge\itshape\rule{0pt}{1.5\ht\strutbox}#1\endtabular}%
   }%
  }%
  \markboth{\MakeUppercase{#1}}{\MakeUppercase{#1}}%
}
\def\@makeschapterhead#1{%
  \vspace*{10\p@}%
  {\parindent\z@\raggedright\normalfont
   \Huge\bfseries\color{issatblue}\headingfont #1\par\nobreak
   \vskip4pt{\color{issatgold}\rule{\linewidth}{1.2pt}}%
   \vskip 20\p@}%
}
\makeatother

% Unnumbered chapter heading for front-matter pages.
\newcommand{\simplechapter}[1]{%
  \clearpage
  \thispagestyle{plain}%
  \phantomsection
  {\normalfont\Huge\bfseries\color{issatblue}\headingfont #1\par}
  \vskip4pt{\color{issatgold}\rule{\linewidth}{1.2pt}}
  \vskip20pt
  \markboth{#1}{#1}%
}
