% Figure: Single Responsibility flag rate for classes the agent authored against
% existing classes it edited, macro-averaged over repositories, production code.
% Dumbbell per agent, one panel per language. Authored is always lower, so its
% value label goes left of its dot and the edited label right of its dot.
% Java (Table tab:authored-edited and report_java.py):
%   Claude Code 8.4 / 22.2, Copilot 3.5 / 22.0, Codex 7.4 / 18.7, all 5.6 / 20.9
% Python (report_python.py production classes, recomputed 2026-09-15):
%   Claude Code 11.0 / 23.1, Copilot 10.3 / 25.1, Codex 8.1 / 23.1, all 9.7 / 23.6
% Scale: 0..30 % over 5.6 cm per panel.
\begin{tikzpicture}[
  font=\scriptsize,
  grid/.style={draw=accentgray!22,line width=0.4pt},
  bell/.style={draw=accentgray!45,line width=2pt,line cap=round},
  auth/.style={circle,fill=srpcolor,draw=white,line width=0.6pt,inner sep=0pt,minimum size=8pt},
  edit/.style={circle,fill=accentgray!70,draw=white,line width=0.6pt,inner sep=0pt,minimum size=8pt},
  val/.style={font=\tiny,inner sep=1.5pt},
  agent/.style={font=\scriptsize,anchor=east},
  ptitle/.style={font=\footnotesize\bfseries,text=issatblue,anchor=south},
]
\def\ps{0.18667} % cm per percentage point (5.6 cm / 30)
\newcommand{\bell}[4]{% #1 x0 of panel, #2 y, #3 authored, #4 edited
  \draw[bell] ({#1+\ps*#3},#2) -- ({#1+\ps*#4},#2);
  \node[auth] at ({#1+\ps*#3},#2) {};
  \node[edit] at ({#1+\ps*#4},#2) {};
  \node[val,anchor=east,text=srpcolor!80!black] at ({#1+\ps*#3-0.14},#2) {#3};
  \node[val,anchor=west,text=accentgray] at ({#1+\ps*#4+0.14},#2) {#4};
}
\newcommand{\panel}[2]{% #1 x0, #2 title
  \foreach \v in {0,10,20,30}{
    \draw[grid] ({#1+\ps*\v},0.35) -- ({#1+\ps*\v},-3.25);
    \node[font=\tiny,text=accentgray,anchor=north] at ({#1+\ps*\v},-3.3) {\v\%};
  }
  \node[ptitle] at ({#1+2.8},0.4) {#2};
}
