% Figure: AI coding tools, 2021-2025 (axis runs to 2026).
% Main axis: one year = 2.8 cm. The first half of 2025, when the three agents
% studied in the thesis appeared, is enlarged in a strip below the axis, so the
% 2025 releases do not compete for the same 1.4 cm of axis.
% Release months: Copilot preview Jun 2021; CodeWhisperer preview Jun 2022;
% Devin Mar 2024; Claude Code Feb 2025; Codex CLI Apr 2025; OpenAI Codex,
% Copilot coding agent and Jules public beta May 2025.
% Geometry checked: the left guide line passes 0.5 cm below the Devin box; the
% May stack starts 0.2 cm right of the Codex CLI box; all within 15.6 cm.
\begin{tikzpicture}[
  font=\footnotesize,
  axis/.style={draw=accentgray,line width=0.9pt},
  tick/.style={draw=accentgray,line width=0.6pt},
  yr/.style={font=\scriptsize,text=accentgray,anchor=north},
  dot/.style={circle,fill=#1,draw=white,line width=0.8pt,inner sep=0pt,minimum size=6.5pt},
  tool/.style={draw=#1!70!black,fill=#1!10,rounded corners=3pt,inner sep=3pt,
               align=center,font=\scriptsize,minimum height=0.6cm},
  studied/.style={tool=issatgold,line width=1.1pt,fill=issatgold!18},
  stem/.style={draw=#1!60,line width=0.6pt},
  lane/.style={font=\scriptsize\bfseries,text=#1,anchor=west},
  note/.style={font=\tiny\itshape,text=accentgray},
]
% ---------------------------------------------------------------- main axis
\draw[axis,-{Latex[length=2.2mm]}] (-0.2,0) -- (14.6,0);
\foreach \y [count=\i from 0] in {2021,...,2026}{
  \draw[tick] ({\i*2.8},0.09) -- ({\i*2.8},-0.09);
  \node[yr] at ({\i*2.8},-0.14) {\y};
}
\fill[issatgold!25] (11.2,-0.11) rectangle (12.6,0.11);
\draw[issatgold!80!black,line width=0.6pt] (11.2,-0.11) rectangle (12.6,0.11);

\node[lane=srpcolor!80!black] at (-0.2,1.72) {Assistants \textmd{\textit{suggest code in the editor}}};
\node[lane=issatblue] at (-0.2,-1.0) {Agents \textmd{\textit{take a task, return a finished change}}};

% ------------------------------------------------------------- assistants
\node[dot=srpcolor] (d1) at (1.17,0) {};
\node[tool=srpcolor,anchor=south] (copilot) at (1.17,0.6) {GitHub Copilot\\[-1pt]\tiny code completion};
\draw[stem=srpcolor] (d1) -- (copilot);

\node[dot=srpcolor] (d2) at (3.97,0) {};
\node[tool=srpcolor,anchor=south west] (cw) at (3.35,0.6) {Amazon CodeWhisperer\\[-1pt]\tiny code completion};

% ------------------------------------------------------------------ agents
\node[dot=issatblue] (d3) at (8.87,0) {};
\node[tool=issatblue,anchor=north] (devin) at (8.87,-0.5) {Devin\\[-1pt]\tiny shell, browser and editor};
\draw[stem=issatblue] (d3) -- (devin);

% ------------------------------------------------------- enlarged 2025 strip
\def\sx{7.0}      % x of 1 Jan 2025 in the strip
\def\sm{1.25}     % cm per month
\def\sy{-3.05}    % y of the strip axis
\draw[issatgold!70!black,densely dashed,line width=0.5pt] (11.2,-0.11) -- (\sx,\sy+0.1);
\draw[issatgold!70!black,densely dashed,line width=0.5pt] (12.6,-0.11) -- ({\sx+6*\sm},\sy+0.1);
\fill[issatgold!10] (\sx,\sy-0.1) rectangle ({\sx+6*\sm},\sy+0.1);
\draw[axis] (\sx,\sy) -- ({\sx+6*\sm},\sy);
\foreach \m [count=\i from 0] in {Jan,Feb,Mar,Apr,May,Jun}{

  \node[font=\tiny,text=accentgray,anchor=south] at ({\sx+(\i+0.5)*\sm},\sy+0.12) {\m};
}
\draw[tick] ({\sx+6*\sm},\sy+0.08) -- ({\sx+6*\sm},\sy-0.08);
\node[font=\scriptsize\bfseries,text=issatgold!70!black,anchor=east] at (\sx-0.2,\sy) {2025};
