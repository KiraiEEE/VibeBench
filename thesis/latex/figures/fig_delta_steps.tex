\begin{tikzpicture}[
  font=\tiny,
  panel/.style={draw=issatblue!30,fill=lightblue!25,rounded corners=5pt,line width=0.6pt},

  ttl/.style={font=\scriptsize\bfseries,text=issatblue,anchor=north},

% ---------------------------------------------------------------- step 1
\begin{scope}[shift={(1.35,-1.35)}]

   \textbf{4} base and head from the API};
\node[note,anchor=north,text width=4.3cm] at (2.35,-3.35)
