% Figure: Martin's abstractness/instability plane (left) and how each coordinate
% is computed (right).
% Left: square of side 5.4 cm. Zone of pain = radius 0.5 around (0,0); zone of
% uselessness = radius 0.5 around (1,1); main sequence band = |A + I - 1| <= 0.15 (A+I from 0.85 to 1.15).
% An example package P at (I, A) = (0.25, 0.20) sits in the zone of pain; its
% normalised distance D = |0.20 + 0.25 - 1| = 0.55; the drawn perpendicular has length
% D/sqrt(2) = 0.39 in plane units, and is labelled accordingly.
% Right: three rows, each with a small glyph and its formula; the rows sit at
% y = 4.6, 2.75 and 0.9 and never reach the square (x >= 7.4).
\begin{tikzpicture}[
  font=\scriptsize,
  pkg/.style={draw=issatblue!70,fill=lightblue!60,rounded corners=3pt,line width=0.7pt,
              minimum width=1.25cm,minimum height=0.85cm},
  cls/.style={draw=accentgray,fill=white,minimum width=0.34cm,minimum height=0.24cm,inner sep=0pt},
  acls/.style={cls,dashed,fill=dipcolor!12,draw=dipcolor},
  dep/.style={-{Latex[length=1.5mm]},draw=accentgray,line width=0.6pt},
  frm/.style={font=\footnotesize,anchor=west},
  expl/.style={font=\tiny,text=accentgray,anchor=north west,align=left,text width=4.15cm,
              execute at begin node={\hyphenpenalty=10000}},
]
\def\s{5.2}
% ================================================================== plane
\begin{scope}
  \clip (0,0) rectangle (\s,\s);
  \fill[issatgold!28] (-0.15*\s,1.30*\s) -- (1.30*\s,-0.15*\s) -- (1.0*\s,-0.15*\s) -- (-0.15*\s,1.0*\s) -- cycle;
  \fill[ispcolor!20] (0,0) circle (0.5*\s);
  \fill[lspcolor!20] (\s,\s) circle (0.5*\s);
\end{scope}
\draw[issatgold!70!black,line width=0.9pt] (0,\s) -- (\s,0);
\draw[line width=0.8pt] (0,0) rectangle (\s,\s);
\draw[-{Latex[length=2mm]},line width=0.8pt] (\s,0) -- (\s+0.55,0) node[right,font=\footnotesize] {$I$};
\draw[-{Latex[length=2mm]},line width=0.8pt] (0,\s) -- (0,\s+0.55) node[above,font=\footnotesize] {$A$};
\foreach \v in {0.5,1}{
  \draw (\v*\s,0) -- (\v*\s,-0.07) node[below,font=\tiny] {\v};
  \draw (0,\v*\s) -- (-0.07,\v*\s) node[left,font=\tiny] {\v};
}
\node[below left,font=\tiny] at (0,0) {0};
\node[font=\scriptsize,anchor=north] at (0.5*\s,-0.38) {instability};
\node[font=\scriptsize,anchor=south,rotate=90] at (-0.62,0.5*\s) {abstractness};

\node[align=center,font=\scriptsize\bfseries,text=ispcolor!75!black] at (0.12*\s,0.36*\s) {zone of\\pain};
\node[align=center,font=\scriptsize\bfseries,text=lspcolor!55!black] at (0.84*\s,0.66*\s) {zone of\\uselessness};
% labels: "main sequence" at A+I = 0.90 (inside the band, off the line and off
% the P-to-line segment); each "intermediate" outside the band and both zones.
\node[font=\scriptsize\bfseries,rotate=-45,text=issatgold!45!black,inner sep=1pt]
  at (0.27*\s,0.63*\s) {main sequence};
\node[font=\tiny,text=accentgray] at (0.42*\s,0.92*\s) {intermediate};
\node[font=\tiny,text=accentgray] at (0.66*\s,0.08*\s) {intermediate};

% example package P and its distance to the main sequence
\coordinate (P) at (0.25*\s,0.20*\s);
\coordinate (F) at (0.525*\s,0.475*\s);
\draw[issatblue,densely dotted,line width=1pt] (P) -- (F);
\fill[issatblue] (P) circle (1.8pt);
\node[font=\tiny\bfseries,text=issatblue,anchor=north] at (P) {$P$};
\node[font=\tiny,text=issatblue,anchor=south east,inner sep=1pt] at ($(P)!0.55!(F)$) {$D/\sqrt{2}$};

% ============================================================ definitions
\def\rx{6.9}

% --- distance
\draw[issatgold!70!black,line width=0.9pt] (\rx+0.15,1.45) -- (\rx+1.85,0.35);

\node[font=\tiny,text=issatblue,anchor=east] at (\rx+0.7,0.84) {$D/\sqrt{2}$};
\node[frm] at (\rx+3.45,1.15) {$D = |A + I - 1|$};
\node[expl] at (\rx+3.45,0.72) {normalised distance from the main sequence (perpendicular distance $=D/\sqrt{2}$); recorded for description, not used as a flag};
\end{tikzpicture}
