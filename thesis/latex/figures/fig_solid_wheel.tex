% Figure: the five SOLID principles as five cards, each with a small drawing of
% the violation shape its detector looks for, the rule, and what is measured.
% A brace underneath groups the cards by where the violation lives (RQ3):
% inside one unit (S, O) or in a relation between units (L, I, D).
% Layout: card width 2.95 cm, gap 0.18 cm, centres 1.475 + 3.13 k; total 15.47 cm.
% Every glyph is drawn inside the card's glyph band (y -0.95 to -2.45).
\begin{tikzpicture}[
  font=\scriptsize,
  card/.style={draw=#1!75!black,fill=white,rounded corners=4pt,line width=0.8pt},
  head/.style={fill=#1,rounded corners=4pt},
  cls/.style={draw=accentgray,fill=white,line width=0.5pt,inner sep=0pt},
  bar/.style={fill=#1,rounded corners=0.6pt},
  lab/.style={font=\tiny,text=accentgray},
  bad/.style={draw=ispcolor,line width=1.1pt,-{Latex[length=1.6mm]}},
  rule/.style={font=\scriptsize\itshape,text=black!80,align=center,text width=2.7cm},
  meas/.style={font=\tiny,text=accentgray!80!black,align=center,text width=2.7cm,
              execute at begin node={\hyphenpenalty=10000\exhyphenpenalty=10000}},
]
\newcommand{\solidcard}[5]{% #1 colour, #2 letter, #3 name, #4 rule, #5 measured
  \draw[card=#1] (-1.475,0) rectangle (1.475,-4.45);
  \begin{scope}
    \clip[rounded corners=4pt] (-1.475,0) rectangle (1.475,-4.45);
    \fill[#1] (-1.475,0) rectangle (1.475,-0.72);
  \end{scope}
  \node[circle,fill=white,text=#1!80!black,font=\small\bfseries,inner sep=0pt,
        minimum size=0.44cm] at (-1.15,-0.36) {#2};
  \node[anchor=west,text=white,font=\fontsize{7.2}{8}\selectfont\bfseries,align=left,inner sep=1pt]
        at (-0.88,-0.36) {#3};
  \node[rule,anchor=north] at (0,-2.55) {#4};
  \draw[#1!40,line width=0.4pt] (-1.2,-3.28) -- (1.2,-3.28);
  \node[meas,anchor=north] at (0,-3.36) {#5};
}
