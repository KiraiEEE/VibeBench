% Deck figure: how a detector is validated, in one chain.
% population -> seeded stratified sample -> audit gate -> reading judge -> score,
% with the two variants underneath (holdout, census). The weighted-recall
% formula is in the score box, since that is where the stratum sizes are used.
% Boxes of 2.75 cm at x = 1.5, 4.55, 7.6, 10.65, 13.7; arrows in the 0.3 cm gaps.
\begin{tikzpicture}[
  font=\tiny,
  box/.style={draw=accentgray!70,fill=white,rounded corners=3pt,align=center,inner sep=3.5pt,
              line width=0.6pt,text width=2.55cm,font=\tiny},
  gate/.style={box,draw=ispcolor,line width=1.1pt,fill=ispcolor!8},
  key/.style={box,draw=issatblue,line width=1pt,fill=lightblue!70},
  arr/.style={-{Latex[length=2mm]},draw=issatgold!90!black,line width=1.1pt},
  thin/.style={-{Latex[length=1.7mm]},draw=accentgray!80,line width=0.6pt},
  ttl/.style={font=\scriptsize\bfseries},
  side/.style={draw=accentgray!55,fill=accentgray!6,rounded corners=3pt,align=left,inner sep=4pt,
               text width=5.6cm,font=\tiny},
  note/.style={font=\tiny\itshape,text=accentgray,align=center},
]
% ------------------------------------------------------------------ chain
\node[box] (pop) at (1.5,0)
  {{\scriptsize\bfseries{}population}\\[2pt]every entity the detector scored\\[2pt]
   {\itshape Java LSP: 1,546 flagged,\\79,689 unflagged}};
\node[box] (samp) at (4.55,0)
  {{\scriptsize\bfseries{}stratified sample}\\[2pt]100 flagged\\100 unflagged\\[2pt]{\itshape seed recorded}};
\node[gate] (gate) at (7.6,0)
  {{\scriptsize\bfseries{}audit gate}\\[2pt]every row traces to the source shown,\\no batch reveals the verdict};
\node[box] (judge) at (10.65,0)
  {{\scriptsize\bfseries{}reading judge}\\[2pt]reads the source, writes a label \textbf{and a reason}};
\node[key] (score) at (13.7,0)
  {{\scriptsize\bfseries{}score}\\[2pt]precision on the flagged side, recall weighted by stratum size};

\draw[arr] (pop) -- (samp);
\draw[arr] (samp) -- (gate);
\draw[arr] (gate) -- (judge);
\draw[arr] (judge) -- (score);

% ------------------------------------------------------------- the formula
\node[side,anchor=north,text width=6.6cm] (eq) at (11.3,-1.35)
  {\textbf{why recall is weighted}\\[3pt]
   the sample is half flagged, the corpus is not:\\[5pt]
   \mbox{\quad$R=\dfrac{TP\cdot N_f/n_f}{TP\cdot N_f/n_f+FN\cdot N_u/n_u}$}\\[5pt]
   $N$ = population, $n$ = sample, $f$ = flagged, $u$ = unflagged.
   Unweighted, Java LSP would read 0.966 instead of 0.352.};

% ------------------------------------------------------------- variants
\node[side,anchor=north west,text width=5.2cm] (hold) at (1.35,-1.35)
  {\textbf{holdout} (LSP, ISP)\\[2pt]
   a second sample, new seed, excluding every row judged before, judged before the
   two rounds are compared. A rule derived from false positives is only trusted
   after this.};
\node[side,anchor=north west,text width=5.2cm] (cens) at (1.35,-3.05)
  {\textbf{census} (DIP in Java, ISP in Python)\\[2pt]
   when a detector flags few entities, every flag is judged: 195 and 80. Precision
   is then exact, with no sampling interval.};

% one spine down the left, so no connector crosses a box
\draw[thin,rounded corners=3pt] (pop.south) -- (1.5,-0.78) -- (0.62,-0.78) -- (0.62,-1.95) -- (hold.west);
\draw[thin,rounded corners=3pt] (0.62,-1.95) -- (0.62,-3.62) -- (cens.west);
\end{tikzpicture}
