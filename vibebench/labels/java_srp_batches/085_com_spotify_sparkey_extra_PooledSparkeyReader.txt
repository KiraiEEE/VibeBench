       94    protected PooledSparkeyReader(SparkeyReader baseReader, int requestedPoolSize) {
       95      this.baseReader = baseReader;
       96  
       97      // Round up to next power of 2 for efficient bit-masking
       98      int poolSize = nextPowerOfTwo(requestedPoolSize);
       99      this.pool = new SparkeyReader[poolSize];
      100      this.mask = poolSize - 1;
      101      this.busy = new AtomicIntegerArray(poolSize);
      102  
      103      // Create pool of reader duplicates
      104      try {
      105        for (int i = 0; i < poolSize; i++) {
      106          pool[i] = baseReader.duplicate();
      107        }
      108      } catch (Exception e) {
      109        // Cleanup on failure
      110        for (int i = 0; i < poolSize; i++) {
      111          if (pool[i] != null) {
      112            pool[i].close();
      113          }
      114        }
      115        baseReader.close();
      116        throw new RuntimeException("Failed to create reader pool", e);
      117      }
      118    }

      188    /**
      189     * Hash thread ID to ensure uniform distribution across pool slots.
      190     *
      191     * <p>Without hashing, workloads where only even (or odd) thread IDs access Sparkey
      192     * would use only half the pool slots, doubling contention. This hash function
      193     * scrambles the bits to ensure full pool utilization.
      194     *
      195     * @param threadId the thread ID to hash
      196     * @return hashed thread ID
      197     */
      198    private static int hashThreadId(long threadId) {
      199      // MurmurHash3 finalizer - provides excellent avalanche properties
      200      // Ensures uniform distribution even for sequential input values
      201      long h = threadId;
      202      h ^= (h >>> 33);
      203      h *= 0xff51afd7ed558ccdL;
      204      h ^= (h >>> 33);
      205      return (int)h;
      206    }

      210    /**
      211     * Execute an operation on a pooled reader with lock-free slot acquisition.
      212     *
      213     * <p>Tries to acquire a slot via CAS: first the thread-affinity slot for cache
      214     * locality, then random slots. If all attempts fail (high contention), delegates
      215     * to a lazily-created overflow pool with 2x capacity. The overflow pool uses the
      216     * same strategy, so capacity grows exponentially until contention resolves.
      217     *
      218     * <p>This approach never blocks or parks — it always makes progress by expanding
      219     * capacity to match actual concurrent demand.
      220     */
      221    private <T> T executeOnPooledReader(ReaderOperation<T> operation) throws IOException {
      222      // Best-effort closed check — a concurrent close() may race past this point.
      223      // This is harmless: the worst case is one stale read attempt on an already-closed
      224      // reader, which will fail with an IOException from the underlying reader.
      225      if (closed) {
      226        throw new IllegalStateException("Reader is closed");
      227      }
      228  
      229      // Start with affinity slot based on thread ID, then try random slots
      230      int slot = hashThreadId(Thread.currentThread().getId()) & mask;
      231      for (int i = 0; i < CAS_ATTEMPTS; i++) {
      232        if (busy.compareAndSet(slot, 0, 1)) {
      233          try {
      234            return operation.execute(pool[slot]);
      235          } finally {
      236            busy.set(slot, 0);
      237          }
      238        }
      239        slot = ThreadLocalRandom.current().nextInt(pool.length);
      240      }
      241  
      242      // All attempts failed — delegate to overflow pool (lazily created, 2x size)
      243      return getOrCreateOverflow().executeOnPooledReader(operation);
      244    }

      246    private PooledSparkeyReader getOrCreateOverflow() {
      247      PooledSparkeyReader o = overflow;
      248      if (o != null) {
      249        return o;
      250      }
      251      synchronized (this) {
      252        if (closed) {
      253          throw new IllegalStateException("Reader is closed");
      254        }
      255        if (overflow == null) {
      256          overflow = new PooledSparkeyReader(baseReader, pool.length * 2);
      257        }
      258        return overflow;
      259      }
      260    }

      441    private static int nextPowerOfTwo(int n) {
      442      if (n <= 1) return 1;
      443      // Handle edge case: if n is already power of 2
      444      if ((n & (n - 1)) == 0) return n;
      445      return 1 << (32 - Integer.numberOfLeadingZeros(n - 1));
      446    }
