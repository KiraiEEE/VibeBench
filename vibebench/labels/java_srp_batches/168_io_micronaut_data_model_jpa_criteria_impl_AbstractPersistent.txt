       83          PersistentProperty persistentProperty = getPersistentEntity().getPropertyByName(attributeName);
       84  
       85          if (persistentProperty == null && attributeName.contains(".")) {
       86              int periodIndex = attributeName.indexOf(".");
       87              String owner = attributeName.substring(0, periodIndex);
       88              PersistentAssociationPath<E, ?> persistentAssociationPath;
       89              PersistentAssociationPath<E, ?> existingJoin = joins.get(owner);
       90              if (existingJoin != null) {
       91                  persistentAssociationPath = existingJoin;
       92              } else {
       93                  persistentAssociationPath = join(owner, type);
       94              }
       95              String remainingJoinPath = attributeName.substring(periodIndex + 1);
       96              return alias == null ? (PersistentAssociationPath<X, Y>) persistentAssociationPath.join(remainingJoinPath, type)
       97                  : (PersistentAssociationPath<X, Y>) persistentAssociationPath.join(remainingJoinPath, type, alias);
       98          }
       99  
      100          if (!(persistentProperty instanceof Association association)) {
      101              throw new IllegalStateException("Expected an association for attribute name: " + attributeName);
      102          }
      103  
      104          PersistentAssociationPath<E, ?> path = joins.computeIfAbsent(attributeName, a -> createJoinAssociation(association, type, alias));
      105  
      106          if (type != null && type != io.micronaut.data.annotation.Join.Type.DEFAULT) {
      107              path.setAssociationJoinType(type);
