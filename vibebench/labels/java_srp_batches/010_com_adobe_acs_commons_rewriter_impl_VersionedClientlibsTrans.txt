      181      private Attributes versionClientLibs(final String elementName, final Attributes attrs, final SlingHttpServletRequest request) {
      182          if (SaxElementUtils.isCss(elementName, attrs)) {
      183              return this.rebuildAttributes(new AttributesImpl(attrs), attrs.getIndex("", ATTR_HREF),
      184                      attrs.getValue("", ATTR_HREF), LibraryType.CSS, request);
      185  
      186          } else if (SaxElementUtils.isJavaScript(elementName, attrs)) {
      187              String attributeName = StringUtils.equals(elementName, "script") ? ATTR_SRC : ATTR_HREF;
      188              return this.rebuildAttributes(new AttributesImpl(attrs), attrs.getIndex("", attributeName),
      189                      attrs.getValue("", attributeName), LibraryType.JS, request);
      190  
      191          } else {
      192              return attrs;
      193          }

      196      private Attributes rebuildAttributes(final AttributesImpl newAttributes, final int index, final String path,
      197                                           final LibraryType libraryType, final SlingHttpServletRequest request) {
      198          final String contextPath = request.getContextPath();
      199          String libraryPath = path;
      200          if (StringUtils.isNotBlank(contextPath)) {
      201              libraryPath = path.substring(contextPath.length());
      202          }
      203  
      204          String versionedPath = this.getVersionedPath(libraryPath, libraryType, request);
      205  
      206          if (StringUtils.isNotBlank(versionedPath)) {
      207              if(StringUtils.isNotBlank(contextPath)) {
      208                  versionedPath = contextPath + versionedPath;
      209              }
      210              log.debug("Rewriting to: {}", versionedPath);
      211              newAttributes.setValue(index, versionedPath);
      212          } else {
      213              log.debug("Versioned Path could not be created properly");
      214          }
      215  
      216          return newAttributes;

      219      private String getVersionedPath(final String originalPath, final LibraryType libraryType,
      220              final SlingHttpServletRequest request) {
      221          if (originalPath.startsWith(PROXY_PREFIX) && originalPath.contains(PROXIED_STATIC_RESOURCE_PATH)) {
      222              log.debug("Static resource accessed via the clientlib proxy: '{}'", originalPath);
      223              return null;
      224          }
      225          try {
      226              boolean appendMinSelector = false;
      227              String libraryPath = StringUtils.substringBeforeLast(originalPath, ".");
      228              if (libraryPath.endsWith(MIN_SELECTOR_SEGMENT)) {
      229                  appendMinSelector = true;
      230                  libraryPath = StringUtils.substringBeforeLast(libraryPath, ".");
      231              }
      232  
      233              final HtmlLibrary htmlLibrary = getLibrary(libraryType, libraryPath, request);
      234  
      235              if (htmlLibrary != null) {
      236                  StringBuilder builder = new StringBuilder();
      237                  builder.append(libraryPath);
      238                  builder.append(".");
      239  
      240                  if (appendMinSelector) {
      241                      builder.append(MIN_SELECTOR).append(".");
      242                  }
      243                  if (enforceMd5) {
      244                      builder.append(MD5_PREFIX);
      245                  }
      246                  builder.append(getMd5(htmlLibrary));
      247                  builder.append(libraryType.extension);
      248  
      249                  return builder.toString();
      250              } else {
      251                  log.debug("Could not find HtmlLibrary at path: {}", libraryPath);
      252                  return null;
      253              }
      254          } catch (Exception ex) {
      255              // Handle unexpected formats of the original path
      256              log.error("Attempting to get a versioned path for [ {} ] but could not because of: {}", originalPath,
      257                      ex.getMessage());
      258              return originalPath;
      259          }
      260      }

      267      private String resolvePath(LibraryType libraryType, String libraryPath, SlingHttpServletRequest request) {
      268          if (!libraryPath.startsWith(PROXY_PREFIX)) {
      269              Resource libraryResource = request.getResourceResolver().resolve(request, libraryPath);
      270              if (libraryResource != null && !(libraryResource instanceof NonExistingResource)) {
      271                  return libraryResource.getPath();
      272              }
      273              // Default behavior, to keep consistency with previous implementation and to not return a null path in case
      274              // the resolver can't find the clientlib
      275              return libraryPath;
      276          }
      277          return resolveProxiedClientLibrary(libraryType, libraryPath, request.getResourceResolver(), true);
      278      }

      280      private String resolveProxiedClientLibrary(LibraryType libraryType, String proxiedPath, ResourceResolver resourceResolver, boolean refreshCacheIfNotFound) {
      281          final String relativePath = proxiedPath.substring(PROXY_PREFIX.length());
      282          for (final String prefix : resourceResolver.getSearchPath()) {
      283              final String absolutePath = prefix + relativePath;
      284              // check whether the ClientLibrary exists before calling HtmlLibraryManager#getLibrary in order
      285              // to avoid WARN log messages that are written when an unknown HtmlLibrary is requested
      286              if (hasProxyClientLibrary(libraryType, absolutePath)) {
      287                  return absolutePath;
      288              }
      289          }
      290  
      291          if (refreshCacheIfNotFound) {
      292              // maybe the library has appeared and our copy of the cache is stale
      293              log.info("Refreshing client libraries cache, because {} could not be found", proxiedPath);
      294              clientLibrariesCache = null;
      295              return resolveProxiedClientLibrary(libraryType, proxiedPath, resourceResolver, false);
      296          }
      297          return null;
      298      }

      324      @NotNull private String calculateMd5(@NotNull final HtmlLibrary htmlLibrary, boolean isMinified) throws IOException {
      325          // make sure that the minified version is being request in case minification is globally enabled
      326          // as this will reset the dirty flag on the clientlib
      327          try (InputStream input = htmlLibrary.getInputStream(isMinified)) {
      328              return DigestUtils.md5Hex(input);

      398      @NotNull
      399      UriInfo getUriInfo(@Nullable final String uri, @NotNull SlingHttpServletRequest request) {
      400          if (uri != null) {
      401              Matcher matcher;
      402              if (enforceMd5) {
      403                  matcher = FILTER_PATTERN_ENFORCE_MD5.matcher(uri);
      404              } else {
      405                  matcher = FILTER_PATTERN.matcher(uri);
      406              }
      407              if (matcher.matches()) {
      408                  final String libraryPath = matcher.group(1);
      409                  final String md5 = matcher.group(2);
      410                  final String extension = matcher.group(3);
      411  
      412                  LibraryType libraryType;
      413                  if (LibraryType.CSS.extension.substring(1).equals(extension)) {
      414                      libraryType = LibraryType.CSS;
      415                  } else {
      416                      libraryType = LibraryType.JS;
      417                  }
      418  
      419                  final HtmlLibrary htmlLibrary = getLibrary(libraryType, libraryPath, request);
      420                  return new UriInfo(libraryPath + "." + extension, md5, libraryType, htmlLibrary);
      421              }
      422          }
      423  
      424          return new UriInfo("", "", null, null);
      425      }

      429          @Override
      430          @SuppressWarnings("squid:S3776")
      431          public void doFilter(final ServletRequest request,
      432                               final ServletResponse response,
      433                               final FilterChain filterChain) throws IOException, ServletException {
      434              if (request instanceof SlingHttpServletRequest && response instanceof SlingHttpServletResponse) {
      435                  final SlingHttpServletRequest slingRequest = (SlingHttpServletRequest) request;
      436                  final SlingHttpServletResponse slingResponse = (SlingHttpServletResponse) response;
      437                  String uri = slingRequest.getRequestURI();
      438                  UriInfo uriInfo = getUriInfo(uri, slingRequest);
      439                  if (uriInfo.cacheKey != null) {
      440                      if ("".equals(uriInfo.md5)) {
      441                          log.debug("MD5 is blank for '{}' in Versioned ClientLibs cache, allowing {} to pass", uriInfo.cleanedUri, uri);
      442                          filterChain.doFilter(request, response);
      443                          return;
      444                      }
      445  
      446                      String md5FromCache = null;
      447                      try {
      448                          md5FromCache = getCacheEntry(uriInfo.cacheKey);
      449                      } catch (Exception e) {
      450                          log.warn("Failed to get cache entry for '{}'", uriInfo.cacheKey);
      451                      }
      452  
      453                      // this static value "Invalid cache key parameter." happens when the cache key can't be
      454                      // found in the cache
      455                      if ("Invalid cache key parameter.".equals(md5FromCache)) {
      456                          md5FromCache = calculateMd5(uriInfo.htmlLibrary, htmlLibraryManager.isMinifyEnabled());
      457                      }
      458  
      459                      if (md5FromCache == null) {
      460                          // something went bad during the cache access
      461                          log.warn("Failed to fetch data from Versioned ClientLibs cache, allowing {} to pass", uri);
      462                          filterChain.doFilter(request, response);
      463                      } else {
      464                          // the file is in the cache, compare the md5 from cache with the one in the request
      465                          if (md5FromCache.equals(uriInfo.md5)) {
      466                              log.debug("MD5 equals for '{}' in Versioned ClientLibs cache, allowing {} to pass", uriInfo.cleanedUri, uri);
      467                              filterChain.doFilter(request, response);
      468                          } else {
      469                              log.info("MD5 differs for '{}' in Versioned ClientLibs cache. Expected {}. Sending 404 for '{}'",
      470                                      uriInfo.cleanedUri, md5FromCache, uri);
      471                              slingResponse.sendError(HttpServletResponse.SC_NOT_FOUND);
      472                          }
      473                      }
      474                  } else {
      475                      filterChain.doFilter(request, response);
      476                  }
      477              } else {
      478                  filterChain.doFilter(request, response);
      479              }
      480          }
