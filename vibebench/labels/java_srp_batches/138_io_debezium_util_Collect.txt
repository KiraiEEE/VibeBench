       27       * Create a fixed sized Map that removes the least-recently used entry when the map becomes too large. The supplied
       28       * {@code maximumNumberOfEntries} should be a power of 2 to efficiently make efficient use of memory. If not, the resulting
       29       * map will be able to contain no more than {@code maximumNumberOfEntries} entries, but the underlying map will have a
       30       * capacity that is the next power of larger than the supplied {@code maximumNumberOfEntries} value so that it can hold
       31       * the required number of entries.
       32       *
       33       * @param maximumNumberOfEntries the maximum number of entries allowed in the map; should be a power of 2
       34       * @return the map that is limited in size by the specified number of entries; never null
       35       */
       36      public static <K, V> Map<K, V> fixedSizeMap(int maximumNumberOfEntries) {
       37          return new LinkedHashMap<K, V>(maximumNumberOfEntries + 1, .75F, true) { // throws illegal argument if < 0
       38              private static final long serialVersionUID = 1L;
       39              final int evictionSize = maximumNumberOfEntries - 1;
       40  
       41              @Override
       42              public boolean removeEldestEntry(Map.Entry<K, V> eldest) {
       43                  return size() > evictionSize;
       44              }
       45          };
       46      }

      273      /**
      274       * Set the value at the given position in the list, expanding the list as required to accommodate the new position.
      275       * <p>
      276       * This is not a thread-safe operation
      277       *
      278       * @param list the list to be modified
      279       * @param index the index position of the new value
      280       * @param value the value
      281       * @param defaultValue the value used for intermediate positions when expanding the list; may be null
      282       */
      283      public static <T> void set(List<T> list, int index, T value, T defaultValue) {
      284          while (list.size() <= index) {
      285              list.add(defaultValue);
      286          }
      287          list.set(index, value);
      288      }
