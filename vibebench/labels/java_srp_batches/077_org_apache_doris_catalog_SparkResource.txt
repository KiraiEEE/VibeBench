      206      public synchronized SparkRepository.SparkArchive prepareArchive() throws LoadException {
      207          String remoteRepositoryPath = workingDir + "/" + Env.getCurrentEnv().getClusterId()
      208                  + "/" + SparkRepository.REPOSITORY_DIR + name;
      209          BrokerDesc brokerDesc = new BrokerDesc(broker, getBrokerPropertiesWithoutPrefix());
      210          SparkRepository repository = new SparkRepository(remoteRepositoryPath, brokerDesc);
      211          // This checks and uploads the remote archive.
      212          repository.prepare();
      213          SparkRepository.SparkArchive archive = repository.getCurrentArchive();
      214          // Normally, an archive should contain a DPP library and a SPARK library
      215          Preconditions.checkState(archive.libraries.size() == 2);
      216          SparkRepository.SparkLibrary dppLibrary = archive.getDppLibrary();
      217          SparkRepository.SparkLibrary spark2xLibrary = archive.getSpark2xLibrary();
      218          if (dppLibrary == null || spark2xLibrary == null) {
      219              throw new LoadException("failed to get libraries from remote archive");
      220          }
      221          return archive;
      222      }

      257      private void updateProperties(Map<String, String> properties) throws DdlException {
      258          // update spark configs
      259          if (properties.containsKey(SPARK_MASTER)) {
      260              throw new DdlException("Cannot change spark master");
      261          }
      262          sparkConfigs.putAll(getSparkConfig(properties));
      263  
      264          // update working dir and broker
      265          if (properties.containsKey(WORKING_DIR)) {
      266              workingDir = properties.get(WORKING_DIR);
      267          }
      268          if (properties.containsKey(BROKER)) {
      269              broker = properties.get(BROKER);
      270          }
      271          brokerProperties.putAll(getBrokerProperties(properties));
      272          Map<String, String> env = getEnvConfig(properties);
      273          if (env.size() > 0) {
      274              if (envConfigs == null) {
      275                  envConfigs = env;
      276              } else {
      277                  envConfigs.putAll(env);
      278              }
      279          }
      280          LOG.info("updateProperties,{},{}", properties, envConfigs);
      281      }

      283      @Override
      284      protected void setProperties(Map<String, String> properties) throws DdlException {
      285          Preconditions.checkState(properties != null);
      286  
      287          // get spark configs
      288          sparkConfigs = getSparkConfig(properties);
      289          envConfigs = getEnvConfig(properties);
      290          LOG.info("setProperties,{},{}", properties, envConfigs);
      291          // check master and deploy mode
      292          if (getMaster() == null) {
      293              throw new DdlException("Missing " + SPARK_MASTER + " in properties");
      294          }
      295          String deployModeStr = sparkConfigs.get(SPARK_SUBMIT_DEPLOY_MODE);
      296          if (deployModeStr != null) {
      297              DeployMode deployMode = DeployMode.fromString(deployModeStr);
      298              if (deployMode == null) {
      299                  throw new DdlException("Unknown deploy mode: " + deployModeStr);
      300              }
      301          } else {
      302              throw new DdlException("Missing " + SPARK_SUBMIT_DEPLOY_MODE + " in properties");
      303          }
      304          // if deploy machines do not set HADOOP_CONF_DIR env, we should set these configs blow
      305          if (isYarnMaster()) {
      306              if (!sparkConfigs.containsKey(SPARK_FS_DEFAULT_FS)) {
      307                  throw new DdlException("Missing (" + SPARK_FS_DEFAULT_FS + ") in yarn master");
      308              }
      309  
      310              String haEnabled = sparkConfigs.get(SPARK_YARN_RESOURCE_MANAGER_HA_ENABLED);
      311              if (StringUtils.isNotEmpty(haEnabled) && "true".equals(haEnabled)) {
      312                  if (StringUtils.isEmpty(SPARK_YARN_RESOURCE_MANAGER_HA_RMIDS)) {
      313                      throw new DdlException("Missing (" + SPARK_YARN_RESOURCE_MANAGER_HA_RMIDS + ") in yarn master, "
      314                          + "when " + SPARK_YARN_RESOURCE_MANAGER_HA_ENABLED + "=true.");
      315                  }
      316  
      317                  String[] haIds = sparkConfigs.get(SPARK_YARN_RESOURCE_MANAGER_HA_RMIDS).split(",");
      318                  for (String haId : haIds) {
      319                      String addressKey = String.format(YARN_RESOURCE_MANAGER_ADDRESS_FOMART, haId);
      320                      String hostnameKey = String.format(YARN_RESOURCE_MANAGER_HOSTNAME_FORMAT, haId);
      321                      if (!sparkConfigs.containsKey(addressKey) && !sparkConfigs.containsKey(hostnameKey)) {
      322                          throw new DdlException("Missing " + addressKey + " or " + hostnameKey + " in yarn master, "
      323                              + "when " + SPARK_YARN_RESOURCE_MANAGER_HA_ENABLED + "=true.");
      324                      }
      325                  }
      326              } else if (!sparkConfigs.containsKey(SPARK_YARN_RESOURCE_MANAGER_ADDRESS)) {
      327                  throw new DdlException("Missing (" + SPARK_YARN_RESOURCE_MANAGER_ADDRESS + ") in yarn master, "
      328                      + "or not turned on ha.");
      329              }
      330          }
      331  
      332          // check working dir and broker
      333          workingDir = properties.get(WORKING_DIR);
      334          broker = properties.get(BROKER);
      335          if ((workingDir == null && broker != null) || (workingDir != null && broker == null)) {
      336              throw new DdlException("working_dir and broker should be assigned at the same time");
      337          }
      338          // check broker exist
      339          if (broker != null && !Env.getCurrentEnv().getBrokerMgr().containsBroker(broker)) {
      340              throw new DdlException("Unknown broker name(" + broker + ")");
      341          }
      342          brokerProperties = getBrokerProperties(properties);
      343      }

      417              result.addRow(Lists.newArrayList(name, lowerCaseType, entry.getKey(), entry.getValue()));
      418          }
