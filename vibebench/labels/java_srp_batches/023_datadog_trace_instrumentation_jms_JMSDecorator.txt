      159      if (null != resourceName) {
      160        span.setResourceName(resourceName);
      161      }

      163      try {
      164        final long produceTime = message.getJMSTimestamp();
      165        if (produceTime > 0) {
      166          final long consumeTime = TimeUnit.NANOSECONDS.toMillis(span.getStartTime());
      167          span.setTag(RECORD_QUEUE_TIME_MS, Math.max(0L, consumeTime - produceTime));

      197    /**
      198     * Sanitizes destination names to remove Kafka Connect schema-derived suffixes. When Kafka
      199     * Connect's IBM MQ connectors are used with schema converters (Protobuf/JSON Schema), union or
      200     * optional fields may get index suffixes like _messagebody_0 appended to the queue name.
      201     */
      202    private static String sanitizeDestinationName(String name) {
      203      if (name == null || name.isEmpty()) {
      204        return name;
      205      }
      206  
      207      int len = name.length();
      208  
      209      // Check if name ends with digits (the schema index suffix)
      210      if (!Character.isDigit(name.charAt(len - 1))) {
      211        return name;
      212      }
      213  
      214      // Find the underscore before the trailing digits
      215      int underscoreBeforeDigits = name.lastIndexOf('_');
      216      if (underscoreBeforeDigits <= 0) {
      217        return name;
      218      }
      219  
      220      // Verify all characters after the underscore are digits
      221      for (int i = underscoreBeforeDigits + 1; i < len; i++) {
      222        if (!Character.isDigit(name.charAt(i))) {
      223          return name;
      224        }
      225      }
      226  
      227      // Find the underscore before the suffix word
      228      int underscoreBeforeSuffix = name.lastIndexOf('_', underscoreBeforeDigits - 1);
      229      if (underscoreBeforeSuffix < 0) {
      230        return name;
      231      }
      232  
      233      // Check if the suffix word is one of our known Kafka Connect schema suffixes (case insensitive)
      234      int suffixStart = underscoreBeforeSuffix + 1;
      235      int suffixLen = underscoreBeforeDigits - suffixStart;
      236  
      237      if (isKnownKafkaConnectSuffix(name, suffixStart, suffixLen)) {
      238        return name.substring(0, underscoreBeforeSuffix);
      239      }
      240  
      241      return name;
      242    }

      244    private static boolean isKnownKafkaConnectSuffix(String name, int start, int len) {
      245      return (len == 11 && name.regionMatches(true, start, "messagebody", 0, 11))
      246          || (len == 4 && name.regionMatches(true, start, "text", 0, 4))
      247          || (len == 5 && name.regionMatches(true, start, "bytes", 0, 5))
      248          || (len == 5 && name.regionMatches(true, start, "value", 0, 5))
      249          || (len == 3 && name.regionMatches(true, start, "map", 0, 3));
      250    }

      252    public CharSequence toResourceName(String destinationName, boolean isQueue) {
      253      if (null == destinationName) {
      254        return isQueue ? queueTempResourceName : topicTempResourceName;
      255      }
      256      Function<CharSequence, CharSequence> joiner =
      257          isQueue ? queueResourceJoiner : topicResourceJoiner;
      258      // some systems may have queues and topics with the same name - since we won't know which was
      259      // cached first we check the character after the initial prefix to see if it's Q (for Queue) -
      260      // if that's what we expect we can use the cached value, otherwise generate the correct name
      261      CharSequence resourceName = resourceNameCache.computeIfAbsent(destinationName, joiner);
      262      if ((resourceName.charAt(resourcePrefix.length()) == 'Q') == isQueue) {
      263        return resourceName;
      264      }
      265      return joiner.apply(destinationName);
      266    }

      268    public String getDestinationName(Destination destination) {
      269      String name = null;
      270      try {
      271        if (destination instanceof Queue) {
      272          // WebLogic mixes all JMS Destination interfaces in a single base type which means we can't
      273          // rely on instanceof and have to instead check the result of getQueueName vs getTopicName
      274          if (!(destination instanceof TemporaryQueue) || isWebLogicDestination(destination)) {
      275            name = ((Queue) destination).getQueueName();
      276          }
      277        }
      278        // check Topic name if Queue name is null because this might be a WebLogic destination
      279        if (null == name && destination instanceof Topic) {
      280          if (!(destination instanceof TemporaryTopic) || isWebLogicDestination(destination)) {
      281            name = ((Topic) destination).getTopicName();
      282          }
      283        }
      284      } catch (Exception e) {
      285        log.debug("Unable to get jms destination name", e);
      286      }
      287      if (null != name && !name.startsWith(TIBCO_TMP_PREFIX)) {
      288        // Sanitize Kafka Connect schema-derived suffixes from queue/topic names
      289        return sanitizeDestinationName(name);
      290      }
      291      return null;
      292    }

      294    public boolean isQueue(Destination destination) {
      295      try {
      296        // handle WebLogic by treating everything as a Queue unless it's a Topic with a name
      297        return !(destination instanceof Topic) || null == ((Topic) destination).getTopicName();
      298      } catch (Exception e) {
      299        return true; // assume it's a Queue if we can't check the details
      300      }
      301    }
