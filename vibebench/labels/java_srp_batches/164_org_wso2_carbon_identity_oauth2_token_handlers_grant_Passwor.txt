      118      private String getFullQualifiedUsername(OAuth2AccessTokenReqDTO tokenReq, ServiceProvider serviceProvider) {
      119  
      120          boolean isEmailUserNameEnabled = MultitenantUtils.isEmailUserName();
      121          boolean isSaasApp = serviceProvider.isSaasApp();
      122          boolean isLegacySaaSAuthenticationEnabled = IdentityTenantUtil.isLegacySaaSAuthenticationEnabled();
      123          String usernameFromRequest = tokenReq.getResourceOwnerUsername();
      124          String tenantDomainFromContext = IdentityTenantUtil.resolveTenantDomain();
      125  
      126          if (!isSaasApp) {
      127              /*
      128              For non-Saas app tenant domain from context is appended to the username from request.
      129              When using tenant qualified URLs, providing tenant-aware username is expected.
      130               */
      131              return UserCoreUtil.addTenantDomainToEntry(usernameFromRequest, tenantDomainFromContext);
      132          } else if (isLegacySaaSAuthenticationEnabled) { // isSaasApp && isLegacySaaSAuthenticationEnabled.
      133              return usernameFromRequest;
      134          } else { // isSaasApp && !isLegacySaaSAuthenticationEnabled.
      135  
      136              /*
      137              If !isEmailUserNameEnabled, then username containing '@' symbol and a username containing
      138              a tenant domain can't be distinguished.
      139              Hence, tenant-qualified username is expected.
      140               */
      141              String tenantDomainFromUser = MultitenantUtils.getTenantDomain(usernameFromRequest);
      142              if (isEmailUserNameEnabled && StringUtils.equalsIgnoreCase(tenantDomainFromUser,
      143                      MultitenantConstants.SUPER_TENANT_DOMAIN_NAME) &&
      144                      !usernameFromRequest.endsWith(MultitenantConstants.SUPER_TENANT_DOMAIN_NAME)) {
      145                  return UserCoreUtil.addTenantDomainToEntry(usernameFromRequest, tenantDomainFromContext);
      146              }
      147              return usernameFromRequest;
      148          }
      149      }

      176      private AuthenticatedUser validateUserCredentials(OAuth2AccessTokenReqDTO tokenReq, ServiceProvider serviceProvider)
      177              throws IdentityOAuth2Exception {
      178  
      179          boolean isPublishPasswordGrantLoginEnabled = Boolean.parseBoolean(
      180                  IdentityUtil.getProperty(PUBLISH_PASSWORD_GRANT_LOGIN));
      181          try {
      182              // Get the user store preference order supplier.
      183              UserStorePreferenceOrderSupplier<List<String>> userStorePreferenceOrderSupplier =
      184                      FrameworkUtils.getUserStorePreferenceOrderSupplier(null, serviceProvider);
      185              UserMgtContext userMgtContext = new UserMgtContext();
      186              userMgtContext.setUserStorePreferenceOrderSupplier(userStorePreferenceOrderSupplier);
      187              if (userStorePreferenceOrderSupplier != null) {
      188                  UserCoreUtil.setUserMgtContextInThreadLocal(userMgtContext);
      189                  if (log.isDebugEnabled()) {
      190                      log.debug("UserMgtContext had been set as the thread local.");
      191                  }
      192              }
      193  
      194              String username = tokenReq.getResourceOwnerUsername();
      195              if (!IdentityUtil.isEmailUsernameValidationDisabled()) {
      196                  FrameworkUtils.validateUsername(username);
      197                  username = FrameworkUtils.preprocessUsername(username, serviceProvider);
      198              }
      199  
      200              String tenantAwareUserName = MultitenantUtils.getTenantAwareUsername(username);
      201              String userTenantDomain = MultitenantUtils.getTenantDomain(username);
      202              ResolvedUserResult resolvedUserResult =
      203                      FrameworkUtils.processMultiAttributeLoginIdentification(tenantAwareUserName, userTenantDomain);
      204              String userId = null;
      205              if (resolvedUserResult != null &&
      206                      ResolvedUserResult.UserResolvedStatus.SUCCESS.equals(resolvedUserResult.getResolvedStatus())) {
      207                  tenantAwareUserName = resolvedUserResult.getUser().getUsername();
      208                  userId = resolvedUserResult.getUser().getUserID();
      209                  tokenReq.setResourceOwnerUsername(tenantAwareUserName + "@" + userTenantDomain);
      210              }
      211  
      212              AbstractUserStoreManager userStoreManager = getUserStoreManager(userTenantDomain);
      213              AuthenticationResult authenticationResult;
      214              if (userId != null) {
      215                  authenticationResult = userStoreManager.authenticateWithID(userId, tokenReq.getResourceOwnerPassword());
      216              } else {
      217                  authenticationResult = userStoreManager.authenticateWithID(
      218                          UserCoreClaimConstants.USERNAME_CLAIM_URI, tenantAwareUserName,
      219                          tokenReq.getResourceOwnerPassword(), UserCoreConstants.DEFAULT_PROFILE);
      220              }
      221  
      222              boolean authenticated = AuthenticationResult.AuthenticationStatus.SUCCESS
      223                      == authenticationResult.getAuthenticationStatus()
      224                      && authenticationResult.getAuthenticatedUser().isPresent();
      225              if (log.isDebugEnabled()) {
      226                  log.debug("user " + tokenReq.getResourceOwnerUsername() + " authenticated: " + authenticated);
      227              }
      228  
      229              if (authenticated) {
      230  
      231                  AuthenticatedUser authenticatedUser
      232                          = new AuthenticatedUser(authenticationResult.getAuthenticatedUser().get());
      233                  if (isPublishPasswordGrantLoginEnabled) {
      234                      publishAuthenticationData(tokenReq, true, serviceProvider, authenticatedUser);
      235                  }
      236                  return authenticatedUser;
      237  
      238              } else {
      239                  if (isPublishPasswordGrantLoginEnabled) {
      240                      publishAuthenticationData(tokenReq, false, serviceProvider);
      241                  }
      242  
      243                  throw new IdentityOAuth2Exception("Authentication failed");
      244              }
      245          } catch (UserStoreClientException e) {
      246              if (isPublishPasswordGrantLoginEnabled) {
      247                  publishAuthenticationData(tokenReq, false, serviceProvider);
      248              }

      250              if (StringUtils.isNotBlank(e.getErrorCode())) {
      251                  message = e.getErrorCode() + " " + e.getMessage();
      252              }
      253              throw new IdentityOAuth2Exception(message, e);
      254          } catch (UserStoreException e) {
      255              if (isPublishPasswordGrantLoginEnabled) {
      256                  publishAuthenticationData(tokenReq, false, serviceProvider);

      258              String message = e.getMessage();

      260              // Therefore, checking for possible client exception.
      261              Throwable rootCause = ExceptionUtils.getRootCause(e);

      268              }
      269              if (e.getCause() instanceof IdentityException) {
      270                  IdentityException identityException = (IdentityException) (e.getCause());
      271                  // Set error code to message if available.
      272                  if (StringUtils.isNotBlank(identityException.getErrorCode())) {
      273                      message = identityException.getErrorCode() + " " + e.getMessage();

      275              }
      276              throw new IdentityOAuth2Exception(message, e);
      277          } catch (AuthenticationFailedException e) {
      278              String message = "Authentication failed for the user: " + tokenReq.getResourceOwnerUsername();
      279              if (log.isDebugEnabled()) {
      280                  log.debug(message, e);
      281              }
      282              throw new IdentityOAuth2Exception("Authentication failed");
      283          } finally {
      284              UserCoreUtil.removeUserMgtContextInThreadLocal();
      285              if (log.isDebugEnabled()) {
      286                  log.debug("UserMgtContext had been remove from the thread local.");
      287              }
      288          }

      315      protected void publishAuthenticationData(OAuth2AccessTokenReqDTO tokenReq, boolean authenticated,

      322              Map<String, Object> paramMap = new HashMap<>();
      323              paramMap.put(FrameworkConstants.AnalyticsAttributes.USER, authenticatedUser);
      324              paramMap.put(REMOTE_IP_ADDRESS, IdentityUtil.getClientIpAddress(tokenReq.getHttpServletRequestWrapper()));
      325              Map<String, Object> unmodifiableParamMap = Collections.unmodifiableMap(paramMap);
      326              if (authenticated) {
      327                  authnDataPublisherProxy
      328                          .publishAuthenticationStepSuccess(null, authenticationContext, unmodifiableParamMap);
      329                  authnDataPublisherProxy.publishAuthenticationSuccess(null, authenticationContext, unmodifiableParamMap);
      330              } else {
      331                  authnDataPublisherProxy.
      332                          publishAuthenticationStepFailure(null, authenticationContext, unmodifiableParamMap);
      333                  authnDataPublisherProxy.publishAuthenticationFailure(null, authenticationContext,
      334                          unmodifiableParamMap);

      336          }

      346      private AuthenticationContext initializeAuthContext(AuthenticatedUser authenticatedUser,
      347                                                          ServiceProvider serviceProvider) {
      348  
      349          AuthenticationContext authenticationContext = new AuthenticationContext();
      350          String contextId = UUIDGenerator.generateUUID();
      351          authenticationContext.setContextIdentifier(contextId);
      352          authenticationContext.setTenantDomain(authenticatedUser.getTenantDomain());
      353          authenticationContext.setRequestType(OAUTH2);
      354          authenticationContext.setRememberMe(false);
      355          authenticationContext.setForceAuthenticate(true);
      356          authenticationContext.setPassiveAuthenticate(false);
      357          authenticationContext.setProperty(IS_INITIAL_LOGIN, true);
      358  
      359          // Setting sequenceConfig with authenticatedUser, serviceProvider.
      360          SequenceConfig sequenceConfig = new SequenceConfig();
      361          sequenceConfig.setAuthenticatedUser(authenticatedUser);
      362  
      363          // Setting applicationConfig with serviceProvider.
      364          ApplicationConfig applicationConfig = new ApplicationConfig(serviceProvider);
      365          sequenceConfig.setApplicationConfig(applicationConfig);
      366  
      367          sequenceConfig.setAuthenticatedIdPs(FrameworkConstants.LOCAL_IDP_NAME);
      368          authenticationContext.setSequenceConfig(sequenceConfig);
      369  
      370           /* Setting the authenticated IDP for currentAuthenticatedIDPs to get
      371           the tenant domain and other parameters when the login is a success. */
      372          AuthenticatedIdPData authenticatedIdPData = new AuthenticatedIdPData();
      373          authenticatedIdPData.setUser(authenticatedUser);
      374          authenticatedIdPData.setIdpName(FrameworkConstants.LOCAL_IDP_NAME);
      375  
      376          AuthenticatorConfig authenticatorConfig = new AuthenticatorConfig();
      377          authenticatorConfig.setName(PASSWORD_GRANT_AUTHENTICATOR_NAME);
      378          authenticatedIdPData.addAuthenticator(authenticatorConfig);
      379          authenticationContext.getCurrentAuthenticatedIdPs().put(FrameworkConstants.LOCAL_IDP_NAME,
      380                  authenticatedIdPData);
      381  
      382          // Setting serviceProviderName from applicationConfig.
      383          authenticationContext.setServiceProviderName(sequenceConfig.getApplicationConfig().getApplicationName());
      384  
      385          return authenticationContext;
      386      }

      432          String tenantAwareUsername = MultitenantUtils.getTenantAwareUsername(tokenReq.getResourceOwnerUsername());
      433          String userTenantDomain = MultitenantUtils.getTenantDomain(tokenReq.getResourceOwnerUsername());
      434          String userNameWithTenant = tenantAwareUsername + UserCoreConstants.TENANT_DOMAIN_COMBINER + userTenantDomain;
      435          if (!userNameWithTenant.contains(CarbonConstants.DOMAIN_SEPARATOR) &&
      436                  StringUtils.isNotBlank(UserCoreUtil.getDomainFromThreadLocal())) {
      437              if (log.isDebugEnabled()) {
      438                  log.debug("User store domain is not found in username. Adding domain: " +
      439                          UserCoreUtil.getDomainFromThreadLocal());
      440              }
      441              return UserCoreUtil.getDomainFromThreadLocal() + CarbonConstants.DOMAIN_SEPARATOR +
      442                      userNameWithTenant;
