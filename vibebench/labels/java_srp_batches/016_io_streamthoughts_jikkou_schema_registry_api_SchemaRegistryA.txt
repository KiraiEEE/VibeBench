       34          List<String> urls = config.urls();

       44      private static SchemaRegistryApi createForUrl(URI baseUri, SchemaRegistryClientConfig config) {
       45          LOG.info("Create new Schema Registry client for: {}", baseUri);
       46          RestClientBuilder builder = RestClientBuilder
       47                  .newBuilder()
       48                  .baseUri(baseUri)
       49                  .enableClientDebugging(config.debugLoggingEnabled());

       52              case BASICAUTH -> {

       55                  yield builder;
       56              }
       57              case SSL -> {

       60              }

       64          };
