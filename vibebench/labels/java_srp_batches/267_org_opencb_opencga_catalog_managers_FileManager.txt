      154              throw new CatalogException("Missing file entries.");

      158          QueryOptions queryOptions = options != null ? new QueryOptions(options) : new QueryOptions();

      160          queryCopy.append(FileDBAdaptor.QueryParams.STUDY_UID.key(), studyUid);
      161  
      162          // TODO: This should be using getFieldFilter method.
      163          Function<File, String> fileStringFunction = File::getPath;
      164          boolean canBeSearchedAsName = true;
      165          List<String> correctedFileList = new ArrayList<>(uniqueList.size());
      166          Map<String, String> entriesIdMap = new HashMap<>();
      167          FileDBAdaptor.QueryParams idQueryParam = null;
      168          for (String entry : uniqueList) {
      169              final String correctedEntryId;
      170              FileDBAdaptor.QueryParams param = FileDBAdaptor.QueryParams.PATH;
      171              if (UuidUtils.isOpenCgaUuid(entry)) {
      172                  correctedEntryId = entry;
      173                  param = FileDBAdaptor.QueryParams.UUID;
      174                  fileStringFunction = File::getUuid;
      175              } else {
      176                  String fileName = entry;
      177                  if (!entry.contains("/")) {
      178                      // Treat as a file so we may need to revert the usage of ":" as part of the path
      179                      fileName = entry
      180                              .replace("\\:", "__OPENCGA__") // Pre-replacement to preserve original ":" from subpath
      181                              .replace(":", "/")             // Replace ":" by "/"
      182                              .replace("__OPENCGA__", ":");  // Revert replacement to original ":"
      183                  }
      184                  if (fileName.startsWith("/")) {
      185                      // Remove the starting /. Absolute paths are not supported.
      186                      fileName = fileName.substring(1);
      187                  }
      188                  correctedEntryId = fileName;
      189  
      190                  if (fileName.contains("/")) {
      191                      canBeSearchedAsName = false;
      192                  }
      193              }
      194              if (idQueryParam == null) {
      195                  idQueryParam = param;
      196              }
      197              correctedFileList.add(correctedEntryId);
      198              entriesIdMap.put(correctedEntryId, entry);

      200              if (idQueryParam != param) {
      201                  throw new CatalogException("Found uuids and paths in the same query. Please, choose one or do two different queries.");
      202              }
      203          }
      204          queryCopy.put(idQueryParam.key(), correctedFileList);
      205  
      206          // Ensure the field by which we are querying for will be kept in the results
      207          queryOptions = keepFieldInQueryOptions(queryOptions, idQueryParam.key());
      208  
      209          OpenCGAResult<File> fileDataResult = getFileDBAdaptor(organizationId).get(studyUid, queryCopy, queryOptions, user);
      210          if (fileDataResult.getNumResults() != correctedFileList.size() && idQueryParam == FileDBAdaptor.QueryParams.PATH
      211                  && canBeSearchedAsName) {
      212              // We also search by name
      213              queryCopy = query == null ? new Query() : new Query(query);

      246              }

      249                  queryCopy = query == null ? new Query() : new Query(query);

      270              } else if (entry.contains(":") && !entry.contains("/")) {
      271                  param = FileDBAdaptor.QueryParams.ID;

      273                  param = FileDBAdaptor.QueryParams.PATH;

      276              }
      277              if (idQueryParam == null) {

      329      public void matchUpVariantFiles(String studyStr, List<File> transformedFiles, String token) throws CatalogException {
      330          JwtPayload tokenPayload = catalogManager.getUserManager().validateToken(token);
      331          CatalogFqn studyFqn = CatalogFqn.extractFqnFromStudy(studyStr, tokenPayload);
      332          String organizationId = studyFqn.getOrganizationId();
      333          String userId = tokenPayload.getUserId(organizationId);
      334          Study study = studyManager.resolveId(studyStr, userId, organizationId);
      335  
      336          for (File transformedFile : transformedFiles) {
      337              authorizationManager.checkFilePermission(organizationId, study.getUid(), transformedFile.getUid(), userId,
      338                      FilePermissions.WRITE);
      339              String variantPathName = getMainVariantFile(transformedFile.getPath());
      340              if (variantPathName == null) {
      341                  // Skip the file.
      342                  logger.debug("The file {} is not a variant transformed file", transformedFile.getName());
      343                  continue;
      344              }
      345  
      346              // Search in the same path
      347              logger.info("Looking for vcf file in path {}", variantPathName);
      348              Query query = new Query()
      349                      .append(FileDBAdaptor.QueryParams.STUDY_UID.key(), study.getUid())
      350                      .append(FileDBAdaptor.QueryParams.PATH.key(), variantPathName)
      351                      .append(FileDBAdaptor.QueryParams.BIOFORMAT.key(), File.Bioformat.VARIANT);
      352  
      353              List<File> fileList = getFileDBAdaptor(organizationId).get(query, new QueryOptions()).getResults();
      354  
      355              if (fileList.isEmpty()) {
      356                  // Search by name in the whole study
      357                  String variantFileName = getMainVariantFile(transformedFile.getName());
      358                  logger.info("Looking for vcf file by name {}", variantFileName);
      359                  query = new Query()
      360                          .append(FileDBAdaptor.QueryParams.STUDY_UID.key(), study.getUid())
      361                          .append(FileDBAdaptor.QueryParams.NAME.key(), variantFileName)
      362                          .append(FileDBAdaptor.QueryParams.BIOFORMAT.key(), File.Bioformat.VARIANT);
      363                  fileList = new ArrayList<>(getFileDBAdaptor(organizationId).get(query, new QueryOptions()).getResults());
      364  
      365                  // In case of finding more than one file, try to find the proper one.
      366                  if (fileList.size() > 1) {
      367                      // Discard files already with a transformed file.
      368                      fileList.removeIf(file -> file.getInternal().getVariant().getIndex() != null
      369                              && file.getInternal().getVariant().getIndex().hasTransform()
      370                              && !file.getInternal().getVariant().getIndex().getTransform().getFileId().equals(transformedFile.getId()));
      371                  }
      372                  if (fileList.size() > 1) {
      373                      // Discard files not transformed nor indexed.
      374                      fileList.removeIf(file -> FileInternal.getVariantIndexStatusId(file.getInternal()).equals(VariantIndexStatus.NONE));
      375                  }
      376              }
      377  
      378              if (fileList.size() != 1) {
      379                  // VCF file not found
      380                  logger.warn("The vcf file corresponding to the file " + transformedFile.getName() + " could not be found");
      381                  continue;
      382              }
      383              File vcf = fileList.get(0);
      384  
      385              // Look for the json file. It should be in the same directory where the transformed file is.
      386              String jsonPathName = getVariantMetadataFile(transformedFile.getPath());
      387              query = new Query()
      388                      .append(FileDBAdaptor.QueryParams.STUDY_UID.key(), study.getUid())
      389                      .append(FileDBAdaptor.QueryParams.PATH.key(), jsonPathName)
      390                      .append(FileDBAdaptor.QueryParams.FORMAT.key(), File.Format.JSON);
      391              fileList = getFileDBAdaptor(organizationId).get(query, new QueryOptions()).getResults();
      392              if (fileList.size() != 1) {
      393                  // Skip. This should not ever happen
      394                  logger.warn("The json file corresponding to the file " + transformedFile.getName() + " could not be found");
      395                  continue;
      396              }
      397              File json = fileList.get(0);
      398  
      399              /* Update relations */
      400              FileRelatedFile producedFromRelation = new FileRelatedFile(vcf, FileRelatedFile.Relation.PRODUCED_FROM);
      401  
      402              // Update json file
      403              logger.debug("Updating json relation");
      404              List<FileRelatedFile> relatedFiles = ParamUtils.defaultObject(json.getRelatedFiles(), ArrayList::new);
      405              // Do not add twice the same relation
      406              if (!relatedFiles.contains(producedFromRelation)) {
      407                  relatedFiles.add(producedFromRelation);
      408                  ObjectMap params = new ObjectMap(FileDBAdaptor.QueryParams.RELATED_FILES.key(), relatedFiles);
      409                  getFileDBAdaptor(organizationId).update(json.getUid(), params, QueryOptions.empty());
      410              }
      411  
      412              // Update transformed file
      413              logger.debug("Updating transformed relation");
      414              relatedFiles = ParamUtils.defaultObject(transformedFile.getRelatedFiles(), ArrayList::new);
      415              // Do not add twice the same relation
      416              if (!relatedFiles.contains(producedFromRelation)) {
      417                  relatedFiles.add(producedFromRelation);
      418                  transformedFile.setRelatedFiles(relatedFiles);
      419                  ObjectMap params = new ObjectMap(FileDBAdaptor.QueryParams.RELATED_FILES.key(), relatedFiles);
      420                  getFileDBAdaptor(organizationId).update(transformedFile.getUid(), params, QueryOptions.empty());
      421              }
      422  
      423              // Update vcf file
      424              logger.debug("Updating vcf relation");
      425              FileInternalVariantIndex index = vcf.getInternal().getVariant().getIndex();
      426              if (!index.hasTransform()) {
      427                  index.setTransform(new FileInternalVariantIndex.Transform(transformedFile.getId(), json.getId()));
      428              }
      429              String status;
      430              if (index.getStatus() != null && index.getStatus().getId() != null) {
      431                  status = index.getStatus().getId();
      432              } else {
      433                  status = VariantIndexStatus.NONE;
      434              }
      435              if (VariantIndexStatus.NONE.equals(status)) {
      436                  // If TRANSFORMED, TRANSFORMING, etc, do not modify the index status
      437                  index.setStatus(new VariantIndexStatus(VariantIndexStatus.TRANSFORMED, "Found transformed file"));
      438              }
      439              ObjectMap params = new ObjectMap(FileDBAdaptor.QueryParams.INTERNAL_VARIANT_INDEX.key(), index);
      440              getFileDBAdaptor(organizationId).update(vcf.getUid(), params, QueryOptions.empty());
      441  
      442              // Update variant stats
      443              Path statsFile = Paths.get(json.getUri().getRawPath());
      444              try (InputStream is = FileUtils.newInputStream(statsFile)) {
      445                  VariantFileMetadata fileMetadata = getDefaultObjectMapper().readValue(is, VariantFileMetadata.class);
      446                  VariantSetStats stats = fileMetadata.getStats();
      447  
      448                  catalogManager.getFileManager()
      449                          .update(studyStr, vcf.getPath(),
      450                                  new FileUpdateParams().setQualityControl(
      451                                          new FileQualityControl().setVariant(
      452                                                  new VariantFileQualityControl(stats, null))),
      453                                  new QueryOptions(),
      454                                  token);
      455              } catch (IOException e) {
      456                  throw new CatalogException("Error reading file \"" + statsFile + "\"", e);
      457              }
      458          }
      459      }
