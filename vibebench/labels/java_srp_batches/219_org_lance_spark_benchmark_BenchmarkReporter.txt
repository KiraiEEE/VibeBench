      227    private static String formatBytes(long bytes) {
      228      if (bytes < 1024) {
      229        return bytes + "B";
      230      } else if (bytes < 1024 * 1024) {
      231        return String.format("%.0fKB", bytes / 1024.0);
      232      } else if (bytes < 1024L * 1024 * 1024) {
      233        return String.format("%.0fMB", bytes / (1024.0 * 1024));
      234      } else {
      235        return String.format("%.1fGB", bytes / (1024.0 * 1024 * 1024));
      236      }
      237    }
