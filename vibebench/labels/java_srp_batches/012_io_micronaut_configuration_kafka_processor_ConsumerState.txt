      145      void close() {
      146          if (closedState == ConsumerCloseState.POLLING) {
      147              final Instant start = Instant.now();
      148              Instant silentTime = start;
      149              do {
      150                  if (LOG.isTraceEnabled()) {
      151                      final Instant now = Instant.now();
      152                      if (now.isAfter(silentTime)) {
      153                          LOG.trace("Consumer {} is not closed yet (waiting {})", info.clientId, Duration.between(start, now));
      154                          // Inhibit TRACE messages for a while to avoid polluting the logs
      155                          silentTime = now.plusSeconds(5);
      156                      }
      157                  }
      158              } while (closedState == ConsumerCloseState.POLLING);
      159          }
      160          LOG.debug("Consumer {} is closed", info.clientId);
      161      }

      213      private void pollAndProcessRecords() {
      214          failed = true;
      215          // We need to retrieve current offsets in case we need to retry the current record or batch
      216          final Map<TopicPartition, OffsetAndMetadata> currentOffsets = getCurrentOffsets();
      217          // Poll records
      218          pauseTopicPartitions();
      219          final ConsumerRecords<?, ?> consumerRecords = pollRecords(currentOffsets);
      220          closedState = ConsumerCloseState.POLLING;
      221          if (!pollingStarted) {
      222              pollingStarted = true;
      223              kafkaConsumerProcessor.publishStartedPollingEvent(kafkaConsumer);
      224          }
      225          resumeTopicPartitions();
      226          if (consumerRecords == null || consumerRecords.isEmpty()) {
      227              return; // No consumer records to process
      228          }
      229          // Support Kotlin coroutines
      230          if (info.method.isSuspend()) {
      231              Argument<?> lastArgument = info.method.getArguments()[info.method.getArguments().length - 1];
      232              boundArguments.put(lastArgument, null);
      233          }
      234          processRecords(consumerRecords, currentOffsets);
      235          if (failed) {
      236              return;
      237          }
      238          if (info.offsetStrategy == OffsetStrategy.SYNC) {
      239              try {
      240                  kafkaConsumer.commitSync();
      241              } catch (CommitFailedException e) {
      242                  handleException(e, consumerRecords, null);
      243              }
      244          } else if (info.offsetStrategy == OffsetStrategy.ASYNC) {
      245              kafkaConsumer.commitAsync(resolveCommitCallback());
      246          }
      247      }

      249      private synchronized void pauseTopicPartitions() {
      250          if (pauseRequests == null || pauseRequests.isEmpty()) {
      251              return;
      252          }
      253          // Only attempt to pause active assignments
      254          Set<TopicPartition> validPauseRequests = new HashSet<>(pauseRequests);
      255          validPauseRequests.retainAll(assignments);
      256          if (validPauseRequests.isEmpty()) {
      257              return;
      258          }
      259          LOG.trace("Pausing Kafka consumption for Consumer [{}] from topic partition: {}", info.clientId, validPauseRequests);
      260          kafkaConsumer.pause(validPauseRequests);
      261          LOG.debug("Paused Kafka consumption for Consumer [{}] from topic partition: {}", info.clientId, kafkaConsumer.paused());
      262          if (pausedTopicPartitions == null) {
      263              pausedTopicPartitions = new HashSet<>();
      264          }
      265          pausedTopicPartitions.addAll(validPauseRequests);
      266      }

      268      private synchronized void resumeTopicPartitions() {
      269          Set<TopicPartition> paused = kafkaConsumer.paused();
      270          if (paused.isEmpty()) {
      271              return;
      272          }
      273          final List<TopicPartition> toResume = paused.stream()
      274              .filter(topicPartition -> pauseRequests == null || !pauseRequests.contains(topicPartition))
      275              .toList();
      276          if (!toResume.isEmpty()) {
      277              LOG.debug("Resuming Kafka consumption for Consumer [{}] from topic partition: {}", info.clientId, toResume);
      278              kafkaConsumer.resume(toResume);
      279          }
      280          if (pausedTopicPartitions != null) {
      281              toResume.forEach(pausedTopicPartitions::remove);
      282          }
      283      }

      302      private Publisher<RecordMetadata> sendToDestination(Object value, ConsumerRecord<?, ?> consumerRecord, ConsumerRecords<?, ?> consumerRecords) {
      303          if (value == null || info.sendToTopics.isEmpty()) {
      304              return Flux.empty();
      305          }
      306          final Object key = consumerRecord.key();
      307          final Producer<?, ?> kafkaProducer;
      308          if (info.shouldSendOffsetsToTransaction) {
      309              kafkaProducer = kafkaConsumerProcessor.getTransactionalProducer(
      310                  info.producerClientId,
      311                  info.producerTransactionalId,
      312                  byte[].class,
      313                  Object.class
      314              );
      315          } else {
      316              kafkaProducer = kafkaConsumerProcessor.getProducer(
      317                  Optional.ofNullable(info.producerClientId).orElse(info.groupId),
      318                  (Class<?>) (key != null ? key.getClass() : byte[].class),
      319                  value.getClass()
      320              );
      321          }
      322          Flux<RecordMetadata> result = Flux.create(emitter -> sendToDestination(emitter, kafkaProducer, key, value, consumerRecord, consumerRecords));
      323          return result.onErrorResume(error -> handleSendToError(error, consumerRecords, consumerRecord));
      324      }

      326      private void sendToDestination(FluxSink<RecordMetadata> emitter, Producer<?, ?> kafkaProducer, Object key, Object value, ConsumerRecord<?, ?> consumerRecord, ConsumerRecords<?, ?> consumerRecords) {
      327          try {
      328              if (info.shouldSendOffsetsToTransaction) {
      329                  beginTransaction(kafkaProducer);
      330              }
      331              sendToDestination(kafkaProducer, new FluxCallback(emitter), key, value, consumerRecord);
      332              if (info.shouldSendOffsetsToTransaction) {
      333                  endTransaction(kafkaProducer, consumerRecords);
      334              }
      335              emitter.complete();
      336          } catch (Exception e) {
      337              if (info.shouldSendOffsetsToTransaction) {
      338                  abortTransaction(kafkaProducer, e);
      339              }
      340              emitter.error(e);
      341          }
      342      }

      375      private void endTransaction(Producer<?, ?> kafkaProducer, ConsumerRecords<?, ?> consumerRecords) {
      376          final Map<TopicPartition, OffsetAndMetadata> offsetsToCommit = new HashMap<>();
      377          for (TopicPartition partition : consumerRecords.partitions()) {
      378              List<? extends ConsumerRecord<?, ?>> partitionedRecords = consumerRecords.records(partition);
      379              long offset = partitionedRecords.get(partitionedRecords.size() - 1).offset();
      380              offsetsToCommit.put(partition, new OffsetAndMetadata(offset + 1));
      381          }
      382          sendOffsetsToTransaction(kafkaProducer, offsetsToCommit);
      383      }
