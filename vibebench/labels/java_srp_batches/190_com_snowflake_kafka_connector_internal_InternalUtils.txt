       85    static Properties makeJdbcDriverPropertiesFromConnectorConfiguration(
       86        Map<String, String> connectorConfiguration, SnowflakeURL url) {
       87      Properties properties = new Properties();
       88  
       89      // decrypt rsa key
       90      String privateKey = "";
       91      String privateKeyPassphrase = "";
       92      String role = "";
       93  
       94      for (Map.Entry<String, String> entry : connectorConfiguration.entrySet()) {
       95        // case insensitive
       96        switch (entry.getKey().toLowerCase()) {
       97          case KafkaConnectorConfigParams.SNOWFLAKE_DATABASE_NAME:
       98            properties.put(JDBC_DATABASE, entry.getValue());
       99            break;
      100          case KafkaConnectorConfigParams.SNOWFLAKE_PRIVATE_KEY:
      101            privateKey = entry.getValue();
      102            break;
      103          case KafkaConnectorConfigParams.SNOWFLAKE_SCHEMA_NAME:
      104            properties.put(JDBC_SCHEMA, entry.getValue());
      105            break;
      106          case KafkaConnectorConfigParams.SNOWFLAKE_USER_NAME:
      107            properties.put(JDBC_USER, entry.getValue());
      108            break;
      109          case KafkaConnectorConfigParams.SNOWFLAKE_PRIVATE_KEY_PASSPHRASE:
      110            privateKeyPassphrase = entry.getValue();
      111            break;
      112          case KafkaConnectorConfigParams.SNOWFLAKE_ROLE_NAME:
      113            role = entry.getValue();
      114            break;
      115          default:
      116            // ignore unrecognized keys
      117        }
      118      }
      119  
      120      properties.put(SFSessionProperty.AUTHENTICATOR.getPropertyKey(), SNOWFLAKE_JWT);
      121      if (isBlank(privateKey)) {
      122        throw SnowflakeErrors.ERROR_0013.getException();
      123      }
      124      properties.put(
      125          JDBC_PRIVATE_KEY, PrivateKeyTool.parsePrivateKey(privateKey, privateKeyPassphrase));
      126  
      127      // set role for JDBC connection (SNOW-3029864)
      128      if (!isBlank(role)) {
      129        properties.put(SFSessionProperty.ROLE.getPropertyKey(), role);
      130      }
      131  
      132      // set ssl
      133      if (url.sslEnabled()) {
      134        properties.put(JDBC_SSL, "on");
      135      } else {
      136        properties.put(JDBC_SSL, "off");
      137      }
      138      // put values for optional parameters
      139      properties.put(JDBC_SESSION_KEEP_ALIVE, "true");
      140      // SNOW-989387 - Set query resultset format to JSON as a workaround
      141      properties.put(JDBC_QUERY_RESULT_FORMAT, "json");
      142      properties.put(SFSessionProperty.ALLOW_UNDERSCORES_IN_HOST.getPropertyKey(), "true");
      143  
      144      if (!properties.containsKey(JDBC_SCHEMA)) {
      145        throw SnowflakeErrors.ERROR_0014.getException();
      146      }
      147  
      148      if (!properties.containsKey(JDBC_DATABASE)) {
      149        throw SnowflakeErrors.ERROR_0015.getException();
      150      }
      151  
      152      if (!properties.containsKey(JDBC_USER)) {
      153        throw SnowflakeErrors.ERROR_0016.getException();
      154      }
      155  
      156      return properties;
      157    }

      159    /**
      160     * Helper method to decide whether to add any properties related to proxy server. These property
      161     * is passed on to snowflake JDBC while calling put API, which requires proxyProperties
      162     *

      166    protected static Properties generateProxyParametersIfRequired(Map<String, String> conf) {
      167      Properties proxyProperties = new Properties();
      168      // Set proxyHost and proxyPort only if both of them are present and are non null
      169      if (conf.get(KafkaConnectorConfigParams.JVM_PROXY_HOST) != null
      170          && conf.get(KafkaConnectorConfigParams.JVM_PROXY_PORT) != null) {
      171        proxyProperties.put(SFSessionProperty.USE_PROXY.getPropertyKey(), "true");
      172        proxyProperties.put(
      173            SFSessionProperty.PROXY_HOST.getPropertyKey(),
      174            conf.get(KafkaConnectorConfigParams.JVM_PROXY_HOST));
      175        proxyProperties.put(
      176            SFSessionProperty.PROXY_PORT.getPropertyKey(),
      177            conf.get(KafkaConnectorConfigParams.JVM_PROXY_PORT));
      178  
      179        // nonProxyHosts parameter is not required. Check if it was set or not.
      180        if (conf.get(KafkaConnectorConfigParams.JVM_NON_PROXY_HOSTS) != null) {
      181          proxyProperties.put(
      182              SFSessionProperty.NON_PROXY_HOSTS.getPropertyKey(),
      183              conf.get(KafkaConnectorConfigParams.JVM_NON_PROXY_HOSTS));
      184        }
      185  
      186        // For username and password, check if host and port are given.
      187        // If they are given, check if username and password are non null
      188        String username = conf.get(KafkaConnectorConfigParams.JVM_PROXY_USERNAME);
      189        String password = conf.get(KafkaConnectorConfigParams.JVM_PROXY_PASSWORD);
      190  
      191        if (username != null && password != null) {
      192          proxyProperties.put(SFSessionProperty.PROXY_USER.getPropertyKey(), username);
      193          proxyProperties.put(SFSessionProperty.PROXY_PASSWORD.getPropertyKey(), password);
      194        }
      195      }
      196      return proxyProperties;
      197    }
