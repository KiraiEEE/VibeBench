      237      private void analyzeCommon(Analyzer analyzer) throws AnalysisException {
      238          // check function name
      239          functionName.analyze(analyzer, this.type);
      240  
      241          // check operation privilege
      242          if (!Env.getCurrentEnv().getAccessManager().checkGlobalPriv(ConnectContext.get(), PrivPredicate.ADMIN)) {
      243              ErrorReport.reportAnalysisException(ErrorCode.ERR_SPECIFIC_ACCESS_DENIED_ERROR, "ADMIN");
      244          }
      245          // check argument
      246          argsDef.analyze(analyzer);
      247  
      248          // alias function does not need analyze following params
      249          if (isAlias) {
      250              return;
      251          }
      252  
      253          returnType.analyze(analyzer);
      254          if (intermediateType != null) {
      255              intermediateType.analyze(analyzer);
      256          } else {
      257              intermediateType = returnType;
      258          }
      259  
      260          String type = properties.getOrDefault(BINARY_TYPE, "JAVA_UDF");
      261          binaryType = getFunctionBinaryType(type);
      262          if (binaryType == null) {
      263              throw new AnalysisException("unknown function type");
      264          }
      265          if (type.equals("NATIVE")) {
      266              throw new AnalysisException("do not support 'NATIVE' udf type after doris version 1.2.0,"
      267                                      + "please use JAVA_UDF or RPC instead");
      268          }
      269  
      270          userFile = properties.getOrDefault(FILE_KEY, properties.get(OBJECT_FILE_KEY));
      271          //        if (Strings.isNullOrEmpty(userFile)) {
      272          //            throw new AnalysisException("No 'file' or 'object_file' in properties");
      273          //        }
      274          if (!Strings.isNullOrEmpty(userFile) && binaryType != TFunctionBinaryType.RPC) {
      275              try {
      276                  computeObjectChecksum();
      277              } catch (IOException | NoSuchAlgorithmException e) {
      278                  throw new AnalysisException("cannot to compute object's checksum. err: " + e.getMessage());
      279              }
      280              String md5sum = properties.get(MD5_CHECKSUM);
      281              if (md5sum != null && !md5sum.equalsIgnoreCase(checksum)) {
      282                  throw new AnalysisException("library's checksum is not equal with input, checksum=" + checksum);
      283              }
      284          }
      285          if (binaryType == TFunctionBinaryType.JAVA_UDF) {
      286              FunctionUtil.checkEnableJavaUdf();
      287  
      288              // always_nullable the default value is true, equal null means true
      289              Boolean isReturnNull = parseBooleanFromProperties(IS_RETURN_NULL);
      290              if (isReturnNull != null && !isReturnNull) {
      291                  returnNullMode = NullableMode.ALWAYS_NOT_NULLABLE;
      292              }
      293              // static_load the default value is false, equal null means false
      294              Boolean staticLoad = parseBooleanFromProperties(IS_STATIC_LOAD);
      295              if (staticLoad != null && staticLoad) {
      296                  isStaticLoad = true;
      297              }
      298              String expirationTimeString = properties.get(EXPIRATION_TIME);
      299              if (expirationTimeString != null) {
      300                  long timeMinutes = Long.parseLong(expirationTimeString);
      301                  if (timeMinutes <= 0) {
      302                      throw new AnalysisException("expirationTime should greater than zero: ");
      303                  }
      304                  this.expirationTime = timeMinutes;
      305              }
      306          }
      307      }

      309      private Boolean parseBooleanFromProperties(String propertyString) throws AnalysisException {
      310          String valueOfString = properties.get(propertyString);
      311          if (valueOfString == null) {
      312              return null;
      313          }
      314          if (!valueOfString.equalsIgnoreCase("false") && !valueOfString.equalsIgnoreCase("true")) {
      315              throw new AnalysisException(propertyString + " in properties, you should set it false or true");
      316          }
      317          return Boolean.parseBoolean(valueOfString);

      320      private void computeObjectChecksum() throws IOException, NoSuchAlgorithmException {
      321          if (FeConstants.runningUnitTest) {
      322              // skip checking checksum when running ut
      323              return;
      324          }
      325  
      326          try (InputStream inputStream = Util.getInputStreamFromUrl(userFile, null, HTTP_TIMEOUT_MS, HTTP_TIMEOUT_MS)) {
      327              MessageDigest digest = MessageDigest.getInstance("MD5");
      328              byte[] buf = new byte[4096];
      329              int bytesRead = 0;
      330              do {
      331                  bytesRead = inputStream.read(buf);
      332                  if (bytesRead < 0) {
      333                      break;
      334                  }
      335                  digest.update(buf, 0, bytesRead);
      336              } while (true);
      337  
      338              checksum = Hex.encodeHexString(digest.digest());
      339          }
      340      }

      368      private void analyzeUda() throws AnalysisException {
      369          AggregateFunction.AggregateFunctionBuilder builder
      370                  = AggregateFunction.AggregateFunctionBuilder.createUdfBuilder();
      371          URI location;
      372          if (!Strings.isNullOrEmpty(userFile)) {
      373              location = URI.create(userFile);
      374          } else {
      375              location = null;
      376          }
      377          builder.name(functionName).argsType(argsDef.getArgTypes()).retType(returnType.getType())
      378                  .hasVarArgs(argsDef.isVariadic()).intermediateType(intermediateType.getType())
      379                  .location(location);
      380          String initFnSymbol = properties.get(INIT_KEY);
      381          if (initFnSymbol == null && !(binaryType == TFunctionBinaryType.JAVA_UDF
      382                  || binaryType == TFunctionBinaryType.RPC)) {
      383              throw new AnalysisException("No 'init_fn' in properties");
      384          }
      385          String updateFnSymbol = properties.get(UPDATE_KEY);
      386          if (updateFnSymbol == null && !(binaryType == TFunctionBinaryType.JAVA_UDF)) {
      387              throw new AnalysisException("No 'update_fn' in properties");
      388          }
      389          String mergeFnSymbol = properties.get(MERGE_KEY);
      390          if (mergeFnSymbol == null && !(binaryType == TFunctionBinaryType.JAVA_UDF)) {
      391              throw new AnalysisException("No 'merge_fn' in properties");
      392          }
      393          String serializeFnSymbol = properties.get(SERIALIZE_KEY);
      394          String finalizeFnSymbol = properties.get(FINALIZE_KEY);
      395          String getValueFnSymbol = properties.get(GET_VALUE_KEY);
      396          String removeFnSymbol = properties.get(REMOVE_KEY);
      397          String symbol = properties.get(SYMBOL_KEY);
      398          if (binaryType == TFunctionBinaryType.RPC && !userFile.contains("://")) {
      399              if (initFnSymbol != null) {
      400                  checkRPCUdf(initFnSymbol);
      401              }
      402              checkRPCUdf(updateFnSymbol);
      403              checkRPCUdf(mergeFnSymbol);
      404              if (serializeFnSymbol != null) {
      405                  checkRPCUdf(serializeFnSymbol);
      406              }
      407              if (finalizeFnSymbol != null) {
      408                  checkRPCUdf(finalizeFnSymbol);
      409              }
      410              if (getValueFnSymbol != null) {
      411                  checkRPCUdf(getValueFnSymbol);
      412              }
      413              if (removeFnSymbol != null) {
      414                  checkRPCUdf(removeFnSymbol);
      415              }
      416          } else if (binaryType == TFunctionBinaryType.JAVA_UDF) {
      417              if (Strings.isNullOrEmpty(symbol)) {
      418                  throw new AnalysisException("No 'symbol' in properties of java-udaf");
      419              }
      420              analyzeJavaUdaf(symbol);
      421          }
      422          function = builder.initFnSymbol(initFnSymbol).updateFnSymbol(updateFnSymbol).mergeFnSymbol(mergeFnSymbol)
      423                  .serializeFnSymbol(serializeFnSymbol).finalizeFnSymbol(finalizeFnSymbol)
      424                  .getValueFnSymbol(getValueFnSymbol).removeFnSymbol(removeFnSymbol).symbolName(symbol).build();
      425          function.setLocation(location);
      426          function.setBinaryType(binaryType);
      427          function.setChecksum(checksum);
      428          function.setNullableMode(returnNullMode);
      429      }
