      441      private TableMetadata toTableMetadata(String db, GetTableResponse response) {
      442          TableSchema schema = TableSchema.create(response.getSchemaId(), response.getSchema());
      443          Map<String, String> options = new HashMap<>(schema.options());
      444          options.put(PATH.key(), response.getPath());
      445          response.putAuditOptionsTo(options);
      446          Identifier identifier = Identifier.create(db, response.getName());
      447          if (identifier.getBranchName() != null) {
      448              options.put(BRANCH.key(), identifier.getBranchName());
      449          }
      450          return new TableMetadata(schema.copy(options), response.isExternal(), response.getId());
      451      }
