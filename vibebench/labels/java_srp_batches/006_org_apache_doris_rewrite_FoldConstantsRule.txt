       91      @Override
       92      public Expr apply(Expr expr, Analyzer analyzer, ExprRewriter.ClauseType clauseType) throws AnalysisException {
       93          // evaluate `case when expr` when possible
       94          if (expr instanceof CaseExpr) {
       95              return CaseExpr.computeCaseExpr((CaseExpr) expr);
       96          }
       97  
       98          // Avoid calling Expr.isConstant() because that would lead to repeated traversals
       99          // of the Expr tree. Assumes the bottom-up application of this rule. Constant
      100          // children should have been folded at this point.
      101          for (Expr child : expr.getChildren()) {
      102              if (!child.isLiteral() && !(child instanceof CastExpr) && !((child instanceof FunctionCallExpr
      103                      || child instanceof ArithmeticExpr || child instanceof TimestampArithmeticExpr
      104                      || child instanceof VariableExpr))) {
      105                  return expr;
      106              }
      107          }
      108  
      109          if (expr.isLiteral() || !expr.isConstant()) {
      110              return expr;
      111          }
      112  
      113          // Do not constant fold cast(null as dataType) because we cannot preserve the
      114          // cast-to-types and that can lead to query failures, e.g., CTAS
      115          if (expr instanceof CastExpr) {
      116              CastExpr castExpr = (CastExpr) expr;
      117              if (castExpr.isNotFold()) {
      118                  return castExpr;
      119              }
      120              if (castExpr.getChild(0) instanceof NullLiteral) {
      121                  return castExpr.getChild(0);
      122              }
      123          }
      124  
      125          // Analyze constant exprs, if necessary. Note that the 'expr' may become non-constant
      126          // after analysis (e.g., aggregate functions).
      127          if (!expr.isAnalyzed()) {
      128              expr.analyze(analyzer);
      129              if (!expr.isConstant()) {
      130                  return expr;
      131              }
      132          }
      133          // it may be wrong to fold constant value in inline view
      134          // so pass the info to getResultValue method to let predicate itself
      135          // to decide if it can fold constant value safely
      136          return expr.getResultValue(expr instanceof SlotRef ? false : analyzer.isInlineViewAnalyzer());
      137      }

      209      public void getConstExpr(Expr expr, Map<String, TExpr> constExprMap, Map<String, Expr> oriConstMap,
      210              Analyzer analyzer, Map<String, Expr> sysVarMap, Map<String, Expr> infoFnMap)
      211              throws AnalysisException {
      212          if (expr.isConstant()) {
      213              // Do not constant fold cast(null as dataType) because we cannot preserve the
      214              // cast-to-types and that can lead to query failures, e.g., CTAS
      215              if (expr instanceof CastExpr) {
      216                  CastExpr castExpr = (CastExpr) expr;
      217                  if (castExpr.getChild(0) instanceof NullLiteral) {
      218                      return;
      219                  }
      220              }
      221              // skip literal expr
      222              if (expr instanceof LiteralExpr) {
      223                  return;
      224              }
      225              // skip BetweenPredicate need to be rewrite to CompoundPredicate
      226              if (expr instanceof BetweenPredicate) {
      227                  return;
      228              }
      229              // collect sysVariableDesc expr
      230              if (expr.contains(Predicates.instanceOf(VariableExpr.class))) {
      231                  getSysVarDescExpr(expr, sysVarMap);
      232                  return;
      233              }
      234              // collect InformationFunction
      235              if (expr.contains(Predicates.instanceOf(InformationFunction.class))) {
      236                  getInfoFnExpr(expr, infoFnMap);
      237                  return;
      238              }
      239              constExprMap.put(expr.getId().toString(), expr.treeToThrift());
      240              oriConstMap.put(expr.getId().toString(), expr);
      241          } else {
      242              recursiveGetChildrenConstExpr(expr, constExprMap, oriConstMap, analyzer, sysVarMap, infoFnMap);
      243  
      244          }

      330       * find and replace constant child expr of a expr by literal expr
      331       * @param expr
      332       * @param key
      333       * @param literalExpr
      334       * @return
      335       */
      336      private Expr replaceExpr(Expr expr, String key, LiteralExpr literalExpr) {
      337          if (expr.getId().toString().equals(key)) {
      338              return literalExpr;
      339          }
      340          // ATTN: make sure the child order of expr keep unchanged
      341          for (int i = 0; i < expr.getChildren().size(); i++) {
      342              Expr child = expr.getChild(i);
      343              if (!(child instanceof LiteralExpr) && literalExpr.equals(replaceExpr(child, key, literalExpr))) {
      344                  literalExpr.setId(child.getId());
      345                  expr.setChild(i, literalExpr);
      346                  break;
      347              }
      348          }
      349          return expr;
      350      }
