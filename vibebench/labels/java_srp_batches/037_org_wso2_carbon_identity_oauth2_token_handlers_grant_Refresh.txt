      165      @Override
      166      public boolean validateScope(OAuthTokenReqMessageContext tokReqMsgCtx)
      167              throws IdentityOAuth2Exception {
      168  
      169          if (!super.validateScope(tokReqMsgCtx)) {
      170              return false;
      171          }
      172  
      173          /*
      174            The requested scope MUST NOT include any scope
      175            not originally granted by the resource owner, and if omitted is
      176            treated as equal to the scope originally granted by the
      177            resource owner
      178           */
      179          String[] requestedScopes = tokReqMsgCtx.getOauth2AccessTokenReqDTO().getScope();
      180          String[] grantedScopes = tokReqMsgCtx.getScope();
      181          String[] grantedInternalScopes = tokReqMsgCtx.getAuthorizedInternalScopes();
      182          if (ArrayUtils.isNotEmpty(requestedScopes)) {
      183              if (ArrayUtils.isEmpty(grantedScopes) && ArrayUtils.isEmpty(grantedInternalScopes)) {
      184                  return false;
      185              }
      186              List<String> grantedScopeList = Stream
      187                      .concat(Arrays.stream(grantedScopes), Arrays.stream(grantedInternalScopes))
      188                      .collect(Collectors.toList());
      189              for (String scope : requestedScopes) {
      190                  if (!grantedScopeList.contains(scope)) {
      191                      if (log.isDebugEnabled()) {
      192                          log.debug("scope: " + scope + "is not granted for this refresh token");
      193                      }
      194                      return false;
      195                  }
      196              }
      197              tokReqMsgCtx.setScope(requestedScopes);
      198          }
      199          return true;
      200      }

      354      private void updateCacheIfEnabled(OAuthTokenReqMessageContext tokReqMsgCtx, AccessTokenDO accessTokenBean,
