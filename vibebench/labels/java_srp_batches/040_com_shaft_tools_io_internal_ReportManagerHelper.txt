      133      public static String prepareIssuesLog() {
      134          if (!listOfNewIssuesForFailedTests.isEmpty()) {
      135              listOfNewIssuesForFailedTests.forEach(issue -> logIssue("Test Method '" + issue.get(0) + "." + issue.get(1)
      136                      + "' failed. Please investigate and open a new Issue if needed.\n"));
      137          }
      138          if (!listOfOpenIssuesForPassedTests.isEmpty()) {
      139              listOfOpenIssuesForPassedTests.forEach(issue -> {
      140                  if (issue.get(3) != null && !issue.get(3).trim().isEmpty()) {
      141                      logIssue("Test Method '" + issue.get(0) + "." + issue.get(1)
      142                              + "' passed. Please validate and close this open issue '" + issue.get(2) + "': '"
      143                              + issue.get(3) + "'.\n");
      144                  } else {
      145                      logIssue("Test Method '" + issue.get(0) + "." + issue.get(1)
      146                              + "' passed. Please validate and close this open issue '" + issue.get(2) + "'.\n");
      147                  }
      148  
      149              });
      150          }
      151          if (!listOfOpenIssuesForFailedTests.isEmpty()) {
      152              listOfOpenIssuesForFailedTests.forEach(issue -> {
      153                  if (issue.get(3) != null && !issue.get(3).trim().isEmpty()) {
      154                      logIssue("Test Method '" + issue.get(0) + "." + issue.get(1) + "' failed with open issue '"
      155                              + issue.get(2) + "': '" + issue.get(3) + "'.\n");
      156                  } else {
      157                      logIssue("Test Method '" + issue.get(0) + "." + issue.get(1) + "' failed with open issue '"
      158                              + issue.get(2) + "'.\n");
      159                  }
      160              });
      161          }
      162  
      163          if (!issuesLog.get().trim().isEmpty()) {
      164              return "Issue Summary: Total Issues = " + (issueCounter.get() - 1) + ", New issues for Failed Tests = "
      165                      + failedTestsWithoutOpenIssuesCounter.get() + ", Open issues for Passed Tests = "
      166                      + openIssuesForPassedTestsCounter.get() + ", Open issues for Failed Tests = "
      167                      + openIssuesForFailedTestsCounter.get() + ". Kindly check the attached Issue details.";
      168          } else {
      169              return "";
      170          }
      171      }

      386          var callingMethodFullName = new StringBuilder();
      387          for (var i = 1; i < callingStack.length; i++) {
