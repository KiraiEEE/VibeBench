      106          public void alphaBlendOver(final BufferedImage image,

      108  
      109              for (int y = 0; y < height; y++) {
      110                  for (int x = 0; x < width; x++) {
      111                      int underRGB = rgb[x][y];
      112                      int overRGB = image.rgb[x][y];
      113                      int underRed   = (underRGB >>> 16) & 0xFF;
      114                      int underGreen = (underRGB >>>  8) & 0xFF;
      115                      int underBlue  =  underRGB         & 0xFF;
      116                      int overRed    = (overRGB >>> 16) & 0xFF;
      117                      int overGreen  = (overRGB >>>  8) & 0xFF;
      118                      int overBlue   =  overRGB         & 0xFF;
      119                      int red   = (int) ((  underRed * (1.0 - alpha)) + (  overRed * alpha));
      120                      int green = (int) ((underGreen * (1.0 - alpha)) + (overGreen * alpha));
      121                      int blue  = (int) (( underBlue * (1.0 - alpha)) + ( overBlue * alpha));
      122                      int newRgb = (red << 16) | (green << 8) | blue;
      123                      this.rgb[x][y] = newRgb;
      124                  }
      125              }
      126          }
