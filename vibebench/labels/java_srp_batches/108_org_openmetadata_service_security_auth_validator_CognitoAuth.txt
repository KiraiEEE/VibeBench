      115    private CognitoDetails validateAndExtractCognitoDetails(String authority) {
      116      // Validate authority format for Cognito
      117      if (!authority.matches("https://cognito-idp\\.[a-z0-9-]+\\.amazonaws\\.com/[a-zA-Z0-9_-]+")) {
      118        throw new IllegalArgumentException(
      119            "Invalid Cognito authority format. Expected: https://cognito-idp.{region}.amazonaws.com/{userPoolId}");
      120      }
      121  
      122      LOG.debug("Extracting Cognito details from authority: {}", authority);
      123      return parseCognitoUrl(authority);

      126    private CognitoDetails parseCognitoUrl(String sourceUrl) {
      127      // Parse Cognito URL to extract region and user pool ID
      128      // Authority format: https://cognito-idp.{region}.amazonaws.com/{userPoolId}
      129      // Discovery format:
      130      // https://cognito-idp.{region}.amazonaws.com/{userPoolId}/.well-known/openid-configuration
      131  
      132      String baseUrl;
      133      if (sourceUrl.contains("/.well-known/openid-configuration")) {
      134        baseUrl = sourceUrl.replace("/.well-known/openid-configuration", "");
      135      } else {
      136        baseUrl = sourceUrl;
      137      }
      138  
      139      if (!baseUrl.matches("https://cognito-idp\\.[a-z0-9-]+\\.amazonaws\\.com/[a-zA-Z0-9_-]+")) {
      140        throw new IllegalArgumentException(
      141            "Invalid Cognito URL format. Expected: https://cognito-idp.{region}.amazonaws.com/{userPoolId}");
      142      }
      143  
      144      // Extract region and user pool ID
      145      String[] parts = baseUrl.split("/");
      146      String domain = parts[2]; // cognito-idp.{region}.amazonaws.com
      147      String region = domain.split("\\.")[1];
      148      String userPoolId = parts[3];
      149  
      150      // Validate region
      151      if (!region.matches("^[a-z]{2}-[a-z]+-[0-9]$")) {
      152        throw new IllegalArgumentException("Invalid AWS region format: " + region);
      153      }
      154  
      155      // Validate user pool ID format
      156      if (!userPoolId.matches("^[a-zA-Z0-9][a-zA-Z0-9_-]+$")) {
      157        throw new IllegalArgumentException("Invalid Cognito user pool ID format: " + userPoolId);
      158      }
      159  
      160      String discoveryUri = baseUrl + COGNITO_WELL_KNOWN_PATH;
      161      return new CognitoDetails(region, userPoolId, discoveryUri);
      162    }

      172              ValidationErrorBuilder.FieldPaths.OIDC_DISCOVERY_URI,
      173              "Cognito user pool '"
      174                  + cognitoDetails.userPoolId
      175                  + "' not found in region '"
      176                  + cognitoDetails.region
      177                  + "'. Please verify the user pool ID and region.");
      178        } else if (response.getStatusCode() != 200) {
      179          return ValidationErrorBuilder.createFieldError(
      180              ValidationErrorBuilder.FieldPaths.OIDC_DISCOVERY_URI,
      181              "Failed to access Cognito discovery endpoint. HTTP response: "
      182                  + response.getStatusCode());
      183        }
      184  
      185        JsonNode discoveryDoc = JsonUtils.readTree(response.getBody());
      186  
      187        if (!discoveryDoc.has("issuer") || !discoveryDoc.has("authorization_endpoint")) {
      188          return ValidationErrorBuilder.createFieldError(
      189              ValidationErrorBuilder.FieldPaths.OIDC_DISCOVERY_URI,
      190              "Invalid Cognito discovery document format");
      191        }
      192  
      193        // Validate issuer format
      194        String issuer = discoveryDoc.get("issuer").asText();
      195        String expectedIssuer =
      196            String.format(
      197                "https://cognito-idp.%s.amazonaws.com/%s",
      198                cognitoDetails.region, cognitoDetails.userPoolId);
      199        if (!issuer.equals(expectedIssuer)) {
      200          return ValidationErrorBuilder.createFieldError(
      201              ValidationErrorBuilder.FieldPaths.OIDC_DISCOVERY_URI,
      202              "Unexpected issuer in Cognito discovery document. Expected: " + expectedIssuer);
      203        }

      209          return ValidationErrorBuilder.createFieldError(
      210              ValidationErrorBuilder.FieldPaths.OIDC_DISCOVERY_URI,
      211              "Missing required Cognito endpoints in discovery document");

      214        return null; // Success - Cognito user pool validated
      215      } catch (Exception e) {
      216        return ValidationErrorBuilder.createFieldError(
      217            ValidationErrorBuilder.FieldPaths.OIDC_DISCOVERY_URI,
      218            "User pool validation failed: " + e.getMessage());
