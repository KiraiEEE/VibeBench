      191    public void delete(boolean incidentResolved) {
      192      CommandContext commandContext = Context.getCommandContext();
      193  
      194      incrementSequenceCounter();
      195  
      196      // clean additional data related to this job
      197      JobHandler jobHandler = getJobHandler();
      198      if (jobHandler != null) {
      199        jobHandler.onDelete(getJobHandlerConfiguration(), this);
      200      }
      201  
      202      // fire delete event if this job is not being executed
      203      boolean executingJob = this.equals(commandContext.getCurrentJob());
      204      commandContext.getJobManager().deleteJob(this, !executingJob);
      205  
      206      // Also delete the job's exception byte array
      207      if (exceptionByteArrayId != null) {
      208        commandContext.getByteArrayManager().deleteByteArrayById(exceptionByteArrayId);
      209      }
      210  
      211      // remove link to execution
      212      ExecutionEntity exec = getExecution();
      213      if (exec != null) {
      214        exec.removeJob(this);
      215      }
      216  
      217      removeFailedJobIncident(incidentResolved);

      304    public void setRetries(int retries) {
      305      // if retries should be set to a negative value set it to 0
      306      if (retries < 0) {
      307        retries = 0;
      308      }
      309  
      310      // Assuming: if the number of retries will
      311      // be changed from 0 to x (x >= 1), means
      312      // that the corresponding incident is resolved.
      313      if (this.retries == 0 && retries > 0) {
      314        removeFailedJobIncident(true);
      315      }
      316  
      317      // If the retries will be set to 0, an
      318      // incident has to be created.
      319      if(retries == 0 && this.retries > 0) {
      320        createFailedJobIncident();
      321      }
      322      this.retries = retries;
      323    }

      330    protected void createFailedJobIncident() {
      331      final ProcessEngineConfigurationImpl processEngineConfiguration = Context.getProcessEngineConfiguration();
      332  
      333      if (processEngineConfiguration
      334          .isCreateIncidentOnFailedJobEnabled()) {
      335  
      336        String incidentHandlerType = Incident.FAILED_JOB_HANDLER_TYPE;
      337  
      338        // make sure job has an ID set:
      339        if(id == null) {
      340          id = processEngineConfiguration
      341              .getIdGenerator()
      342              .getNextId();
      343  
      344        } else {
      345          // check whether there exists already an incident
      346          // for this job
      347          List<Incident> failedJobIncidents = Context
      348              .getCommandContext()
      349              .getIncidentManager()
      350              .findIncidentByConfigurationAndIncidentType(id, incidentHandlerType);
      351  
      352          if (!failedJobIncidents.isEmpty()) {
      353            // update the historic job log id in the historic incidents (if available)
      354            for (Incident incident : failedJobIncidents) {
      355              HistoricIncidentEntity historicIncidentEvent = Context
      356                  .getCommandContext()
      357                  .getHistoricIncidentManager()
      358                  .findHistoricIncidentById(incident.getId());
      359              if (historicIncidentEvent != null) {
      360                historicIncidentEvent.setHistoryConfiguration(getLastFailureLogId());
      361                Context.getCommandContext().getDbEntityManager().merge(historicIncidentEvent);
      362              }
      363            }
      364            return;
      365          }
      366  
      367        }
      368  
      369  
      370        IncidentContext incidentContext = createIncidentContext();
      371        incidentContext.setActivityId(getActivityId());
      372        incidentContext.setHistoryConfiguration(getLastFailureLogId());
      373        incidentContext.setFailedActivityId(getFailedActivityId());
      374  
      375        IncidentHandling.createIncident(incidentHandlerType, incidentContext, exceptionMessage);
      376  
      377      }
      378    }
