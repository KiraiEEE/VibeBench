       88      public static void buildTileProviderList(final boolean force) {
       89          if (!(tileProviders.isEmpty() || force)) {
       90              return;
       91          }
       92          tileProviders.clear();
       93  
       94          // --------------------------------------------------------------------
       95          // online-based map providers
       96          // --------------------------------------------------------------------
       97  
       98          // Google Map based tile providers
       99          if (isGoogleMapsInstalled()) {
      100              registerTileProvider(new GoogleMapSource());
      101              registerTileProvider(new GoogleSatelliteSource());
      102              registerTileProvider(new GoogleTerrainSource());
      103          }
      104  
      105          // OSM online tile providers (Mapsforge)
      106          if (Settings.showMapsforgeInUnifiedMap()) {
      107              registerTileProvider(new OsmOrgSource());
      108              registerTileProvider(new OsmDeSource());
      109              registerTileProvider(new CyclosmSource());
      110              registerTileProvider(new OpenTopoMapSource());
      111  
      112              if (UserDefinedMapsforgeOnlineSource.isConfigured()) {
      113                  registerTileProvider(new UserDefinedMapsforgeOnlineSource());
      114              }
      115          }
      116  
      117          // OSM online tile providers (VTM)
      118          if (Settings.showVTMInUnifiedMap()) {
      119              registerTileProvider(new OsmOrgVTMSource());
      120              registerTileProvider(new OsmDeVTMSource());
      121              registerTileProvider(new CyclosmVTMSource());
      122              registerTileProvider(new OpenTopoMapVTMSource());
      123  
      124              if (UserDefinedMapsforgeVTMOnlineSource.isConfigured()) {
      125                  registerTileProvider(new UserDefinedMapsforgeVTMOnlineSource());
      126              }
      127          }
      128  
      129          // --------------------------------------------------------------------
      130          // offline-based map providers
      131          // --------------------------------------------------------------------
      132  
      133          // collect available offline map files
      134          final List<ImmutablePair<String, Uri>> temp =
      135                  CollectionStream.of(ContentStorage.get().list(PersistableFolder.OFFLINE_MAPS, true))
      136                          .filter(fi -> !fi.isDirectory && fi.name.toLowerCase(Locale.getDefault()).endsWith(FileUtils.MAP_FILE_EXTENSION) && isValidMapFile(fi.uri))
      137                          .map(fi -> new ImmutablePair<>(StringUtils.capitalize(StringUtils.substringBeforeLast(fi.name, ".")), fi.uri)).toList();
      138  
      139          // OSM offline tile providers (Mapsforge)
      140          if (Settings.showMapsforgeInUnifiedMap()) {
      141              if (temp.size() > 1) {
      142                  registerTileProvider(new MapsforgeMultiOfflineTileProvider(temp));
      143              }
      144  
      145              // sort according to displayName and register
      146              final List<ImmutablePair<String, AbstractMapsforgeOfflineTileProvider>> offlineMaps = new ArrayList<>();
      147              for (ImmutablePair<String, Uri> data : temp) {
      148                  final AbstractMapsforgeOfflineTileProvider tp = new AbstractMapsforgeOfflineTileProvider(data.left, data.right, 2, 18); // @todo: get actual values for zoomMin/zoomMax
      149                  offlineMaps.add(new ImmutablePair<>(tp.getDisplayName(data.left), tp));
      150              }
      151              Collections.sort(offlineMaps, (o1, o2) -> TextUtils.COLLATOR.compare(o1.left, o2.left));
      152              for (ImmutablePair<String, AbstractMapsforgeOfflineTileProvider> data : offlineMaps) {
      153                  registerTileProvider(data.right);
      154              }
      155          }
      156  
      157          // OSM offline tile providers (VTM)
      158          if (Settings.showVTMInUnifiedMap()) {
      159              if (temp.size() > 1) {
      160                  registerTileProvider(new MapsforgeVTMMultiOfflineTileProvider(temp));
      161              }
      162  
      163              // sort according to displayName and register
      164              final List<ImmutablePair<String, AbstractMapsforgeVTMOfflineTileProvider>> offlineMaps = new ArrayList<>();
      165              for (ImmutablePair<String, Uri> data : temp) {
      166                  final AbstractMapsforgeVTMOfflineTileProvider tp = new AbstractMapsforgeVTMOfflineTileProvider(data.left, data.right, 2, 18); // @todo: get actual values for zoomMin/zoomMax
      167                  offlineMaps.add(new ImmutablePair<>(tp.getDisplayName(data.left), tp));
      168              }
      169              Collections.sort(offlineMaps, (o1, o2) -> TextUtils.COLLATOR.compare(o1.left, o2.left));
      170              for (ImmutablePair<String, AbstractMapsforgeVTMOfflineTileProvider> data : offlineMaps) {
      171                  registerTileProvider(data.right);
      172              }
      173          }
      174  
      175          // --------------------------------------------------------------------
      176          // "no map" tile provider
      177          registerTileProvider(new NoMapMapsforgeTileProvider());
      178      }

      180      private static boolean isGoogleMapsInstalled() {
      181          // Check if API key is available
      182          final String mapsKey = LocalizationUtils.getString(R.string.maps_api2_key);
      183          if (StringUtils.length(mapsKey) < 30 || StringUtils.contains(mapsKey, "key")) {
      184              Log.w("No Google API key available.");
      185              return false;
      186          }
      187  
      188          // Check if API is available
      189          try {
      190              Class.forName("com.google.android.gms.maps.SupportMapFragment");
      191          } catch (final ClassNotFoundException ignored) {
      192              return false;
      193          }
      194  
      195          // Assume that Google Maps is available and working
      196          return true;
      197      }
