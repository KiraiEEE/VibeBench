      422                  // How the implementation looks depends on whether the type is a super type, subtype, both, or is a
      423                  // stand-alone type.
      424                  //
      425                  // Intermediate types, those that are both a super type and subtype, will pass null as the type

      439                  // as potential legal values for deserialization, confusing the Salmon deserializer. So, calling into
      440                  // Salmon
      441                  // will only attempt to handle Salmon and SmartSalmon, and same goes for Shark with Shark, Sawshark,
      442                  // GoblinShark, and Cookiecuttershark. It's only Fish that will handle all subtypes, as this is the most
      443                  // generic
      444                  // super type. This also creates a well-defined bounds for code generation in regard to type
      445                  // hierarchies.
      446                  //
      447                  // In a real scenario someone deserializing to Salmon should have an exception about discriminator types

      476          private void writeToJson(JavaClass classBlock, boolean isJsonMergePatch) {
      477              boolean callToJsonSharedForParentProperties = settings.isShareJsonSerializableCode()
      478                  && !isJsonMergePatch
      479                  && model.isAllPolymorphicModelsInSamePackage()
      480                  && !CoreUtils.isNullOrEmpty(model.getParentModelName());

      482                  && !isJsonMergePatch
      483                  && model.isAllPolymorphicModelsInSamePackage()
      484                  && model.isPolymorphicParent();
      485  
      486              classBlock.javadocComment(JavaJavadocComment::inheritDoc);
      487              addGeneratedAnnotation.accept(classBlock);
      488              classBlock.annotation("Override");
      489              classBlock.publicMethod("JsonWriter toJson(JsonWriter jsonWriter) throws IOException", methodBlock -> {
      490                  if (isJsonMergePatch) {
      491                      // If the model is the root parent use the JSON merge patch serialization tracking property
      492                      // directly,
      493                      // otherwise use the access helper to determine whether to use JSON merge patch serialization.
      494                      ClientModel rootParent = ClientModelUtil.getRootParent(model);
      495                      String ifStatement = (rootParent == model)
      496                          ? "jsonMergePatch"
      497                          : JSON_MERGE_PATCH_HELPER_CLASS_NAME + ".get" + rootParent.getName()
      498                              + "Accessor().isJsonMergePatch(this)";
      499  
      500                      methodBlock.ifBlock(ifStatement, ifBlock -> ifBlock.methodReturn("toJsonMergePatch(jsonWriter)"))
      501                          .elseBlock(elseBlock -> serializeJsonProperties(methodBlock, false, false, false));
      502                  } else {
      503                      // For polymorphic models, when all models are contained in the same package, a package-private
      504                      // 'toJsonShared'
      505                      // method will be generated by parent models that handles serialization of properties defined by the
      506                      // parent
      507                      // model.
      508                      // So, if we have that case, we skip serializing the properties defined by the parent model(s) and
      509                      // instead call
      510                      // the 'toJsonShared' method to serialize those properties. And, for properties defined by this
      511                      // model, if the
      512                      // model is a parent model generate a 'toJsonShared' method to serialize those properties.
      513                      //
      514                      // At this time, due to implementation complexity and the low value add, JSON merge patch
      515                      // serialization won't
      516                      // support 'toJsonMergePatchShared'. JSON merge patch serialization has low usage, and combined with
      517                      // the low
      518                      // usage of polymorphism as well, there isn't a large expectation for this feature.
      519                      serializeJsonProperties(methodBlock, false, callToJsonSharedForParentProperties,
      520                          callToJsonSharedForThisProperties);
      521                  }
      522              });
      523  
      524              // If this model is defining a 'toJsonShared' method, write it.
      525              if (callToJsonSharedForThisProperties) {
      526                  classBlock.packagePrivateMethod("void toJsonShared(JsonWriter jsonWriter) throws IOException",
      527                      methodBlock -> {
      528                          if (callToJsonSharedForParentProperties) {
      529                              methodBlock.line("super.toJsonShared(jsonWriter);");
      530                          }
      531  
      532                          serializeThisJsonProperties(
      533                              (property, fromSuper) -> serializeJsonProperty(methodBlock, property,
      534                                  property.getSerializedName(), fromSuper, true, false),
      535                              methodBlock, callToJsonSharedForParentProperties);
      536                      });
      537              }
