      231  
      232      final private List<T> list;

      235        this.list = list;
      236      }
      237  
      238      @Override
      239      public int compare(T arg1, T arg2) {
      240        String v1 = arg1.getVersion();
      241        String v2 = arg2.getVersion();
      242        if (v1 == null && v2 == null) {
      243          return Integer.compare(list.indexOf(arg1), list.indexOf(arg2)); // retain original order
      244        } else if (v1 == null) {
      245          return -1;
      246        } else if (v2 == null) {
      247          return 1;
      248        } else {
      249          String mm1 = VersionUtilities.getMajMin(v1);
      250          String mm2 = VersionUtilities.getMajMin(v2);
      251          if (mm1 == null || mm2 == null) {
      252            return v1.compareTo(v2);
      253          } else {
      254            return mm1.compareTo(mm2);
      255          }
      256        }
      257      }

      420    public void registerResourceFromPackage(CanonicalResourceProxy r, PackageInformation packageInfo) throws FHIRException {
      421      PackageHackerR5.fixLoadedResource(r, packageInfo);
      422  
      423      synchronized (lock) {
      424        definitionsChanged();
      425        if (packageInfo != null) {
      426          packages.put(packageInfo.getVID(), packageInfo);
      427        }
      428  
      429        String url = r.getUrl();
      430        if (!allowLoadingDuplicates && hasResourceVersion(r.getType(), url, r.getVersion()) && !packageInfo.isTHO()) {
      431          // special workaround for known problems with existing packages
      432          if (Utilities.existsInList(url, "http://hl7.org/fhir/SearchParameter/example")) {
      433            return;
      434          }
      435          // matchbox patch for duplicate resources, see https://github.com/ahdis/matchbox/issues/227 and issues/452
      436          CanonicalResource ex = fetchResourceWithException(r.getType(), url, VersionResolutionRules.defaultRule());
      437          boolean forceDrop = false;
      438          if (url.startsWith("http://terminology.hl7.org/") && ex.getVersion()!=null && ex.getSourcePackage()!=null && ex.getVersion().equals(ex.getSourcePackage().getVersion()) ) {
      439              forceDrop = true;
      440          }
      441          if (forceDrop || (packageInfo!=null && packageInfo.getVersion()!=null && !packageInfo.getVersion().equals(r.getVersion()))) {
      442              dropResource(r.getType(), r.getId());
      443              dropResource(r.getType(), url);
      444          } else {
      445              return;
      446          }
      447          // END matchbox patch
      448        }
      449        boolean added = registerResource(r, packageInfo);
      450        if (added) {
      451          registerInAllResourceIndex(r, packageInfo);
      452        }
      453      }

      456    private void registerInAllResourceIndex(CanonicalResourceProxy r, PackageInformation packageInfo) {
      457      if (r.getId() != null) {
      458        Map<String, ResourceProxy> map = allResourcesById.get(r.getType());
      459        if (map == null) {
      460          map = new HashMap<String, ResourceProxy>();
      461          allResourcesById.put(r.getType(), map);
      462        }
      463        if ((packageInfo == null || !packageInfo.isExamplesPackage()) || !map.containsKey(r.getId())) {
      464          map.put(r.getId(), new ResourceProxy(r));
      465        }
      466      }
      467      if (r.getUrl() != null) {
      468        List<ResourceProxy> list = allResourcesByUrl.get(r.getUrl());
      469        if (list == null) {
      470          list = new ArrayList<>();
      471          allResourcesByUrl.put(r.getUrl(), list);
      472        }
      473        list.add(new ResourceProxy(r));
      474      }
