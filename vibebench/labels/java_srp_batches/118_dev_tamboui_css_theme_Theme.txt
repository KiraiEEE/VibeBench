       94      /**
       95       * Infers whether theme is dark based on background/text colors.
       96       * Heuristic: if background is darker than text, it's a dark theme.
       97       */
       98      private static boolean inferDarkMode(Stylesheet stylesheet) {
       99          Optional<String> bg = stylesheet.resolveVariable(ThemePropertyNames.BACKGROUND);
      100          Optional<String> text = stylesheet.resolveVariable(ThemePropertyNames.TEXT);
      101  
      102          if (bg.isPresent() && text.isPresent()) {
      103              StyleEngine engine = StyleEngine.create();
      104              Optional<Color> bgColor = engine.parseColor(bg.get());
      105              Optional<Color> textColor = engine.parseColor(text.get());
      106  
      107              if (bgColor.isPresent() && textColor.isPresent()) {
      108                  double bgLightness = getLightness(bgColor.get());
      109                  double textLightness = getLightness(textColor.get());
      110                  return bgLightness < textLightness;
      111              }
      112          }
      113  
      114          return false;
      115      }

      117      /**
      118       * Calculates perceived lightness of a color (0.0 = black, 1.0 = white).
      119       * Uses standard relative luminance formula.
      120       */
      121      private static double getLightness(Color color) {
      122          Color.Rgb rgb = color.toRgb();
      123          double r = rgb.r() / 255.0;
      124          double g = rgb.g() / 255.0;
      125          double b = rgb.b() / 255.0;
      126  
      127          // Relative luminance
      128          return 0.2126 * r + 0.7152 * g + 0.0722 * b;
      129      }

      310      /**
      311       * Generates a complete color palette from the theme's base colors.
      312       *
      313       * <p>Inspired by Textual's ColorSystem.generate(). Creates a full palette including:</p>
      314       * <ul>
      315       *   <li>Lightened variants (primary-lighten-1, primary-lighten-2, primary-lighten-3)</li>
      316       *   <li>Darkened variants (primary-darken-1, primary-darken-2, primary-darken-3)</li>
      317       *   <li>Derived colors (if secondary/error/warning not defined, derive from primary)</li>
      318       *   <li>Muted variants (primary-muted, secondary-muted, error-muted)</li>
      319       * </ul>
      320       *
      321       * <p>Example:</p>
      322       * <pre>{@code
      323       * Theme theme = Theme.builder()
      324       *     .dark(true)
      325       *     .primary("#3498DB")
      326       *     .luminositySpread(0.15f)
      327       *     .build();
      328       *
      329       * Map<String, String> palette = theme.generate();
      330       * // Contains: primary, primary-lighten-1, primary-darken-1,
      331       * //           secondary (derived), error (derived), etc.
      332       * }</pre>
      333       *
      334       * @return map of color name → hex value for complete palette
      335       * @throws IllegalStateException if primary color is not defined
      336       */
      337      public Map<String, String> generate() {
      338          Map<String, String> palette = new LinkedHashMap<>();
      339  
      340          // Get base colors with fallbacks
      341          Color primary = getPrimary().orElseThrow(() ->
      342              new IllegalStateException("Primary color required for palette generation"));
      343          Color secondary = getSecondary().orElse(primary);
      344          Color accent = getAccent().orElse(primary);
      345          Color error = getError().orElse(primary.rotateHue(180));  // Complementary
      346          Color warning = getWarning().orElse(primary.rotateHue(30));  // Analogous
      347          Color success = getSuccess().orElse(primary.rotateHue(-30));  // Analogous
      348  
      349          // Generate shades for each color (-3, -2, -1, 0, +1, +2, +3)
      350          generateShades(palette, ThemePropertyNames.PRIMARY, primary);
      351          generateShades(palette, ThemePropertyNames.SECONDARY, secondary);
      352          generateShades(palette, ThemePropertyNames.ACCENT, accent);
      353          generateShades(palette, ThemePropertyNames.ERROR, error);
      354          generateShades(palette, ThemePropertyNames.WARNING, warning);
      355          generateShades(palette, ThemePropertyNames.SUCCESS, success);
      356  
      357          // Generate muted variants
      358          Color background = getBackground().orElse(dark ?
      359              Color.rgb(18, 18, 18) : Color.rgb(255, 255, 255));
      360          palette.put(ThemePropertyNames.PRIMARY + "-muted",
      361              primary.mix(background, 0.7f).toHex());
      362          palette.put(ThemePropertyNames.SECONDARY + "-muted",
      363              secondary.mix(background, 0.7f).toHex());
      364          palette.put(ThemePropertyNames.ERROR + "-muted",
      365              error.mix(background, 0.7f).toHex());
      366  
      367          return palette;
      368      }

      370      private void generateShades(Map<String, String> palette, String name, Color baseColor) {
      371          float step = luminositySpread / 2.0f;  // Each shade is half the spread
      372  
      373          palette.put(name, baseColor.toHex());
      374  
      375          for (int i = 1; i <= 3; i++) {
      376              float delta = step * i;
      377              palette.put(name + "-lighten-" + i, baseColor.lighten(delta).toHex());
      378              palette.put(name + "-darken-" + i, baseColor.darken(delta).toHex());
      379          }
      380      }

      398       * @return CSS content
      399       */
      400      public String toCss() {
      401          StringBuilder sb = new StringBuilder();
      402  
      403          // Add header comment with theme metadata
      404          sb.append("/* Theme: ").append(name);
      405          if (dark) {
      406              sb.append(" (dark)");
      407          } else {
      408              sb.append(" (light)");
      409          }
      410          sb.append(" */\n\n");
      411  
      412          // Generate variable declarations in definition order
      413          for (Map.Entry<String, String> entry : variables.entrySet()) {
      414              sb.append("$")
      415                      .append(entry.getKey())
      416                      .append(": ")
      417                      .append(entry.getValue())
      418                      .append(";\n");
      419          }
      420  
      421          return sb.toString();
      422      }
