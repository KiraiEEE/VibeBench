      120      /**
      121       * Processes a non-streaming chat request.
      122       *
      123       * @param request the HTTP request
      124       * @param response the HTTP response
      125       * @throws IOException if an I/O error occurs
      126       */
      127      protected void processChatRequest(final HttpServletRequest request, final HttpServletResponse response) throws IOException {
      128          if (!"POST".equalsIgnoreCase(request.getMethod())) {
      129              if (logger.isDebugEnabled()) {
      130                  logger.debug("Invalid method for chat request. method={}", request.getMethod());
      131              }
      132              writeJsonResponse(response, HttpServletResponse.SC_METHOD_NOT_ALLOWED, createErrorResponse("Method not allowed"));
      133              return;
      134          }
      135  
      136          try {
      137              final String message = request.getParameter("message");
      138              final String sessionId = request.getParameter("sessionId");
      139              final String clearParam = request.getParameter("clear");
      140  
      141              if (logger.isDebugEnabled()) {
      142                  logger.debug("Processing chat request. sessionId={}, messageLength={}, clear={}", sessionId,
      143                          message != null ? message.length() : 0, clearParam);
      144              }
      145  
      146              if (StringUtil.isBlank(message)) {
      147                  if ("true".equals(clearParam) && StringUtil.isNotBlank(sessionId)) {
      148                      ComponentUtil.getChatSessionManager().clearSession(sessionId);
      149                      if (logger.isDebugEnabled()) {
      150                          logger.debug("Session cleared. sessionId={}", sessionId);
      151                      }
      152                      writeJsonResponse(response, HttpServletResponse.SC_OK, createSuccessResponse(sessionId, "Session cleared", null));
      153                      return;
      154                  }
      155                  if (logger.isDebugEnabled()) {
      156                      logger.debug("Message is required but was empty");
      157                  }
      158                  writeJsonResponse(response, HttpServletResponse.SC_BAD_REQUEST, createErrorResponse("Message is required"));
      159                  return;
      160              }
      161  
      162              final String userId = getUserId(request);
      163              final ChatResult result = ComponentUtil.getChatClient().chat(sessionId, message, userId);
      164  
      165              if (logger.isDebugEnabled()) {
      166                  logger.debug("Chat request completed. sessionId={}, responseLength={}", result.getSessionId(),
      167                          result.getMessage().getContent() != null ? result.getMessage().getContent().length() : 0);
      168              }
      169  
      170              writeJsonResponse(response, HttpServletResponse.SC_OK,
      171                      createSuccessResponse(result.getSessionId(), result.getMessage().getContent(), result.getMessage().getSources()));
      172  
      173          } catch (final Exception e) {
      174              logger.warn("Failed to process chat request. message={}", e.getMessage(), e);
      175              writeJsonResponse(response, HttpServletResponse.SC_INTERNAL_SERVER_ERROR, createErrorResponse("Internal server error"));
      176          }

      179      /**
      180       * Processes a streaming chat request using Server-Sent Events (SSE).
      181       * Uses the enhanced multi-phase RAG flow with intent detection and result evaluation.
      182       *
      183       * @param request the HTTP request
      184       * @param response the HTTP response
      185       * @throws IOException if an I/O error occurs

      187      protected void processStreamRequest(final HttpServletRequest request, final HttpServletResponse response) throws IOException {
      188          if (!"POST".equalsIgnoreCase(request.getMethod()) && !"GET".equalsIgnoreCase(request.getMethod())) {
      189              if (logger.isDebugEnabled()) {
      190                  logger.debug("Invalid method for stream request. method={}", request.getMethod());
      191              }
      192              writeJsonResponse(response, HttpServletResponse.SC_METHOD_NOT_ALLOWED, createErrorResponse("Method not allowed"));
      193              return;
      194          }
      195  
      196          final String message = request.getParameter("message");
      197          final String sessionId = request.getParameter("sessionId");
      198  
      199          if (logger.isDebugEnabled()) {
      200              logger.debug("Processing stream request. sessionId={}, messageLength={}", sessionId, message != null ? message.length() : 0);
      201          }
      202  
      203          if (StringUtil.isBlank(message)) {
      204              if (logger.isDebugEnabled()) {
      205                  logger.debug("Message is required but was empty for stream request");
      206              }
      207              writeJsonResponse(response, HttpServletResponse.SC_BAD_REQUEST, createErrorResponse("Message is required"));
      208              return;
      209          }
      210  
      211          // Set SSE headers
      212          response.setContentType("text/event-stream");
      213          response.setCharacterEncoding("UTF-8");
      214          response.setHeader("Cache-Control", "no-cache");
      215          response.setHeader("Connection", "keep-alive");
      216          response.setHeader("X-Accel-Buffering", "no"); // Disable nginx buffering
      217  
      218          try (final PrintWriter writer = response.getWriter()) {
      219              final String userId = getUserId(request);
      220  
      221              // Send initial session info
      222              sendSseEvent(writer, "session", Map.of("sessionId", sessionId != null ? sessionId : ""));
      223              if (logger.isDebugEnabled()) {
      224                  logger.debug("SSE session event sent. sessionId={}", sessionId);
      225              }
      226  
      227              // Create phase callback for SSE events
      228              final ChatPhaseCallback phaseCallback = new ChatPhaseCallback() {
      229                  @Override
      230                  public void onPhaseStart(final String phase, final String phaseMessage) {
      231                      try {
      232                          sendSseEvent(writer, "phase", Map.of("phase", phase, "status", "start", "message", phaseMessage));
      233                          if (logger.isDebugEnabled()) {
      234                              logger.debug("SSE phase start event sent. phase={}, message={}", phase, phaseMessage);
      235                          }
      236                      } catch (final Exception e) {
      237                          if (logger.isDebugEnabled()) {
      238                              logger.debug("Failed to send phase start event. phase={}, error={}", phase, e.getMessage());
      239                          }
      240                      }
      241                  }
      242  
      243                  @Override
      244                  public void onPhaseComplete(final String phase) {
      245                      try {
      246                          sendSseEvent(writer, "phase", Map.of("phase", phase, "status", "complete"));
      247                          if (logger.isDebugEnabled()) {
      248                              logger.debug("SSE phase complete event sent. phase={}", phase);
      249                          }
      250                      } catch (final Exception e) {
      251                          if (logger.isDebugEnabled()) {
      252                              logger.debug("Failed to send phase complete event. phase={}, error={}", phase, e.getMessage());
      253                          }
      254                      }
      255                  }
      256  
      257                  @Override
      258                  public void onChunk(final String content, final boolean done) {
      259                      try {
      260                          if (!done && StringUtil.isNotBlank(content)) {
      261                              sendSseEvent(writer, "chunk", Map.of("content", content));
      262                          }
      263                      } catch (final Exception e) {
      264                          if (logger.isDebugEnabled()) {
      265                              logger.debug("Failed to send SSE chunk. error={}", e.getMessage());
      266                          }
      267                      }
      268                  }
      269  
      270                  @Override
      271                  public void onError(final String phase, final String errorMessage) {
      272                      try {
      273                          sendSseEvent(writer, "error", Map.of("phase", phase, "message", errorMessage));
      274                          if (logger.isDebugEnabled()) {
      275                              logger.debug("SSE error event sent. phase={}, error={}", phase, errorMessage);
      276                          }
      277                      } catch (final Exception e) {
      278                          if (logger.isDebugEnabled()) {
      279                              logger.debug("Failed to send error event. phase={}, error={}", phase, e.getMessage());
      280                          }
      281                      }
      282                  }
      283              };
      284  
      285              // Stream the response using enhanced flow
      286              final ChatResult result = ComponentUtil.getChatClient().streamChatEnhanced(sessionId, message, userId, phaseCallback);
      287  
      288              // Send sources
      289              final List<ChatSource> sources = result.getMessage().getSources();
      290              if (sources != null && !sources.isEmpty()) {
      291                  sendSseEvent(writer, "sources", Map.of("sources", sources));
      292                  if (logger.isDebugEnabled()) {
      293                      logger.debug("SSE sources event sent. sourcesCount={}", sources.size());
      294                  }
      295              }
      296  
      297              // Send completion event with HTML content
      298              final Map<String, Object> doneData = new HashMap<>();
      299              doneData.put("sessionId", result.getSessionId());
      300              final String htmlContent = result.getMessage().getHtmlContent();
      301              if (htmlContent != null) {
      302                  doneData.put("htmlContent", htmlContent);
      303              }
      304              sendSseEvent(writer, "done", doneData);
      305              if (logger.isDebugEnabled()) {
      306                  logger.debug("SSE stream completed. sessionId={}, hasHtmlContent={}", result.getSessionId(), htmlContent != null);
      307              }
      308  
      309          } catch (final Exception e) {
      310              logger.warn("Failed to process stream request. sessionId={}, message={}", sessionId, e.getMessage(), e);
      311              if (!response.isCommitted()) {
      312                  try (final PrintWriter writer = response.getWriter()) {
      313                      sendSseEvent(writer, "error", Map.of("message", "Internal server error"));
      314                  } catch (final IOException ioe) {
      315                      logger.warn("Failed to send error response. error={}", ioe.getMessage());
      316                  }
      317              }
      318          }
      319      }

      321      /**
      322       * Sends a Server-Sent Event (SSE) to the client.
      323       *
      324       * @param writer the print writer to write the event to
      325       * @param event the event name
      326       * @param data the event data to serialize as JSON
      327       */
      328      protected void sendSseEvent(final PrintWriter writer, final String event, final Map<String, Object> data) {
      329          try {
      330              writer.write("event: " + event + "\n");
      331              writer.write("data: " + objectMapper.writeValueAsString(data) + "\n\n");
      332              writer.flush();
      333          } catch (final JsonProcessingException e) {
      334              logger.warn("Error serializing SSE data", e);
      335          }
      336      }
