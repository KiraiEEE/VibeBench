      175    public void initByURI(final Uri videoURI) {
      176      mSourceUri = videoURI;

      178      if (isVideoType) {

      182        mVideoView.setOnPreparedListener(mp -> {
      183          mp.setVideoScalingMode(MediaPlayer.VIDEO_SCALING_MODE_SCALE_TO_FIT);
      184          mediaPlayer = mp;
      185          mediaPrepared();
      186        });
      187

      189        mVideoView.setOnCompletionListener(mp -> mediaCompleted());
      190      } else {

      195  
      196        mediaPlayer = new MediaPlayer();
      197        try {
      198          mediaPlayer.setDataSource(videoURI.toString());
      199          mediaPlayer.setOnPreparedListener(mp -> {
      200            mediaPrepared();
      201          });
      202          mediaPlayer.setOnCompletionListener(mp -> mediaCompleted());
      203          mediaPlayer.setOnErrorListener(this::onFailToLoadMedia);

      205          mediaPlayer.prepareAsync(); // use prepareAsync to avoid blocking the main thread
      206        } catch (IOException e) {
      207          e.printStackTrace();

      251    private void mediaPrepared() {
      252      mDuration = mediaPlayer.getDuration();
      253      mMaxDuration = Math.min(mMaxDuration, mDuration);
      254  
      255      if (isVideoType) {
      256        mediaMetadataRetriever = MediaMetadataUtil.getMediaMetadataRetriever(mSourceUri.toString());
      257        if (mediaMetadataRetriever == null) {
      258          String deviceInfo = Build.MANUFACTURER + " " + Build.MODEL + " (Android " + Build.VERSION.RELEASE + ")";
      259          String errorMessage = "Error when retrieving video info on " + deviceInfo + ". Please check LogCat for detailed error information.";
      260          Log.e(TAG, "Failed to get MediaMetadataRetriever for URI: " + mSourceUri.toString() + " on device: " + deviceInfo);
      261          mOnTrimVideoListener.onError(errorMessage, ErrorCode.FAIL_TO_GET_VIDEO_INFO);
      262          return;
      263        }
      264  
      265        // take first frame
      266        Bitmap bitmap = mediaMetadataRetriever.getFrameAtTime(0, MediaMetadataRetriever.OPTION_CLOSEST_SYNC);
      267  
      268        if (bitmap != null) {
      269          int bitmapHeight = bitmap.getHeight() > 0 ? bitmap.getHeight() : VideoTrimmerUtil.THUMB_HEIGHT;
      270          int bitmapWidth = bitmap.getWidth() > 0 ? bitmap.getWidth() : VideoTrimmerUtil.THUMB_WIDTH;
      271          VideoTrimmerUtil.mThumbWidth = VideoTrimmerUtil.THUMB_HEIGHT * bitmapWidth / bitmapHeight;
      272        }
      273  
      274        VideoTrimmerUtil.SCREEN_WIDTH_FULL = this.getScreenWidthInPortraitMode();
      275        VideoTrimmerUtil.VIDEO_FRAMES_WIDTH = VideoTrimmerUtil.SCREEN_WIDTH_FULL - RECYCLER_VIEW_PADDING * 2;
      276        VideoTrimmerUtil.MAX_COUNT_RANGE = Math.max((VIDEO_FRAMES_WIDTH / VideoTrimmerUtil.mThumbWidth), VideoTrimmerUtil.MAX_COUNT_RANGE);
      277  
      278        startShootVideoThumbs(mContext, VideoTrimmerUtil.MAX_COUNT_RANGE, 0, mDuration);
      279      }
      280  
      281      // Set initial handle positions if mMaxDuration < video duration
      282      if (mMaxDuration < mDuration) {
      283        endTime = mMaxDuration;
      284      } else {
      285        endTime = mDuration;
      286      }
      287      updateHandlePositions();
      288  
      289      loadingIndicator.setVisibility(View.GONE);
      290      mPlayView.setVisibility(View.VISIBLE);
      291      saveBtn.setVisibility(View.VISIBLE);
      292  
      293      if (jumpToPositionOnLoad > 0) {
      294        seekTo(jumpToPositionOnLoad > mDuration ? mDuration : jumpToPositionOnLoad, true);
      295      }
      296  
      297      if (autoplay) {
      298        playOrPause();
      299      }
      300  
      301      mOnTrimVideoListener.onLoad(mDuration);
      302    }
