      165              throws Throwable {

      196       * Dispatches the data change events for the records of a single replica-set.

      198      private void createDataEvents(ChangeEventSourceContext sourceContext,
      199                                    MongoDbSnapshotContext snapshotContext,
      200                                    SnapshotReceiver<MongoDbPartition> snapshotReceiver,
      201                                    MongoDbConnection mongo,
      202                                    SnapshottingTask snapshottingTask)
      203              throws Throwable {
      204          snapshotContext.lastCollection = false;
      205          snapshotContext.offset.startInitialSnapshot();
      206  
      207          LOGGER.info("Beginning snapshot at {}", snapshotContext.offset.getOffset());
      208  
      209          Set<Pattern> dataCollectionPattern = getDataCollectionPattern(snapshottingTask.getDataCollections());
      210          // mongo.collections() return a not sorted list and so not deterministic. Forcing the natural order.
      211          List<CollectionId> allCollections = mongo.collections().stream()
      212                  .sorted(Comparator.comparing(CollectionId::name))
      213                  .collect(Collectors.toList());
      214          final List<CollectionId> collections = determineDataCollectionsToBeSnapshotted(allCollections, dataCollectionPattern)
      215                  .collect(Collectors.toList());
      216          snapshotProgressListener.monitoredDataCollectionsDetermined(snapshotContext.partition, collections);
      217  
      218          // Since multiple snapshot threads are to be used, create a thread pool and initiate the snapshot.
      219          // The current thread will wait until the snapshot threads either have completed or an error occurred.
      220          final int numThreads = Math.min(collections.size(), connectorConfig.getSnapshotMaxThreads());
      221          final Queue<CollectionId> collectionsToCopy = new ConcurrentLinkedQueue<>(collections);
      222  
      223          LOGGER.info("Creating snapshot worker pool with {} worker thread(s)", numThreads);
      224          final ExecutorService executorService = Threads.newFixedThreadPool(MongoDbConnector.class, taskContext.getServerName(), "snapshot-main",
      225                  io.debezium.util.ThreadNameContext.from(connectorConfig), connectorConfig.getSnapshotMaxThreads());
      226  
      227          final AtomicBoolean aborted = new AtomicBoolean(false);
      228          final AtomicInteger threadCounter = new AtomicInteger(0);
      229  
      230          LOGGER.info("Preparing to use {} thread(s) to snapshot {} collection(s): {}", numThreads, collections.size(),
      231                  Strings.join(", ", collections));
      232  
      233          CompletionService<Void> completionService = new ExecutorCompletionService<>(executorService);
      234          for (int i = 0; i < numThreads; ++i) {
      235              completionService
      236                      .submit(() -> buildCallable(sourceContext, snapshotContext, snapshotReceiver, mongo, snapshottingTask, threadCounter, aborted, collectionsToCopy));
      237          }
      238  
      239          try {
      240              for (int i = 0; i < numThreads; i++) {
      241                  completionService.take().get();
      242              }
      243          }
      244          finally {
      245              executorService.shutdown();
      246          }
      247  
      248          snapshotContext.offset.stopInitialSnapshot();
      249      }

      251      private Void buildCallable(ChangeEventSourceContext sourceContext, MongoDbSnapshotContext snapshotContext, SnapshotReceiver<MongoDbPartition> snapshotReceiver,
      252                                 MongoDbConnection mongo, SnapshottingTask snapshottingTask, AtomicInteger threadCounter, AtomicBoolean aborted,
      253                                 Queue<CollectionId> collectionsToCopy) {
      254  
      255          taskContext.configureLoggingContext("snapshot" + threadCounter.incrementAndGet());
      256          CollectionId id = null;
      257          try {
      258              while (!aborted.get() && (id = collectionsToCopy.poll()) != null) {
      259                  if (!sourceContext.isRunning()) {
      260                      throw new InterruptedException("Interrupted while snapshotting");
      261                  }
      262  
      263                  if (collectionsToCopy.isEmpty()) {
      264                      snapshotContext.lastCollection = true;
      265                  }
      266  
      267                  createDataEventsForCollection(
      268                          sourceContext,
      269                          snapshotContext,
      270                          snapshotReceiver,
      271                          id,
      272                          mongo, snapshottingTask.getFilterQueries());
      273              }
      274          }
      275          catch (Throwable t) {
      276              // Do nothing so that this thread is stopped
      277              LOGGER.error("Snapshot failed", t);
      278              aborted.set(true);
      279              throw new ConnectException("Snapshotting of collection " + id + " failed", t);
      280          }
      281          return null;
      282      }

      297      private void createDataEventsForCollection(ChangeEventSourceContext sourceContext,
      298                                                 MongoDbSnapshotContext snapshotContext,
      299                                                 SnapshotReceiver<MongoDbPartition> snapshotReceiver,
      300                                                 CollectionId collectionId, MongoDbConnection mongo,
      301                                                 Map<DataCollectionId, String> snapshotFilterQueryForCollection)
      302              throws InterruptedException {
      303          long exportStart = clock.currentTimeInMillis();
      304          LOGGER.info("\t Exporting data for collection '{}'", collectionId);
      305          notificationService.initialSnapshotNotificationService().notifyTableInProgress(snapshotContext.partition, snapshotContext.offset, collectionId.namespace());
      306  
      307          mongo.execute("sync '" + collectionId + "'", client -> {
      308              final MongoDatabase database = client.getDatabase(collectionId.dbName());
      309              final MongoCollection<BsonDocument> collection = database.getCollection(collectionId.name(), BsonDocument.class);
      310  
      311              final int batchSize = taskContext.getConnectorConfig().getSnapshotFetchSize();
      312  
      313              long docs = 0;
      314              Optional<String> snapshotFilterForCollectionId = Optional.ofNullable(snapshotFilterQueryForCollection.get(collectionId));
      315              Bson filterQuery = Document.parse(snapshotFilterForCollectionId.orElse("{}"));
      316  
      317              try (MongoCursor<BsonDocument> cursor = collection.find(filterQuery).batchSize(batchSize).iterator()) {
      318                  snapshotContext.lastRecordInCollection = false;
      319                  if (cursor.hasNext()) {
      320                      while (cursor.hasNext()) {
      321                          if (!sourceContext.isRunning()) {
      322                              throw new InterruptedException("Interrupted while snapshotting collection " + collectionId.name());
      323                          }
      324  
      325                          BsonDocument document = cursor.next();
      326                          docs++;
      327  
      328                          snapshotContext.lastRecordInCollection = !cursor.hasNext();
      329  
      330                          if (snapshotContext.lastCollection && snapshotContext.lastRecordInCollection) {
      331                              snapshotContext.offset.markSnapshotRecord(SnapshotRecord.LAST);
      332                          }
      333  
      334                          dispatcher.dispatchSnapshotEvent(snapshotContext.partition, collectionId,
      335                                  getChangeRecordEmitter(snapshotContext, collectionId, document),
      336                                  snapshotReceiver);
      337                      }
      338                  }
      339                  else if (snapshotContext.lastCollection) {
      340                      // if the last collection does not contain any records we still need to mark the last processed event as last one
      341                      snapshotContext.offset.markSnapshotRecord(SnapshotRecord.LAST);
      342                  }
      343  
      344                  notificationService.initialSnapshotNotificationService().notifyCompletedTableSuccessfully(snapshotContext.partition, snapshotContext.offset,
      345                          collectionId.namespace());
      346                  LOGGER.info("\t Finished snapshotting {} records for collection '{}'; total duration '{}'", docs, collectionId,
      347                          Strings.duration(clock.currentTimeInMillis() - exportStart));
      348                  snapshotProgressListener.dataCollectionSnapshotCompleted(snapshotContext.partition, collectionId, docs);
      349              }
      350          });
      351      }
