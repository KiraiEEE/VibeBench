       71      public void addReplicationJob(ReplicationJob job) throws AlreadyExistsException {
       72          // Limit replication job size
       73          if (runningJobs.size() >= Config.replication_max_parallel_table_count) {
       74              throw new RuntimeException(
       75                      "The replication jobs exceeds the replication_max_parallel_table_count: "
       76                              + Config.replication_max_parallel_table_count);
       77          }
       78  
       79          // Limit replication replica count
       80          long replicationReplicaCount = getReplicatingReplicaCount();
       81          if (replicationReplicaCount >= Config.replication_max_parallel_replica_count) {
       82              throw new RuntimeException("The replicating replica count in all running replication jobs "
       83                      + replicationReplicaCount
       84                      + " exceeds replication_max_parallel_replica_count: "
       85                      + Config.replication_max_parallel_replica_count);
       86          }
       87  
       88          // Limit replication data size
       89          long replicatingDataSizeMB = getReplicatingDataSize() / 1048576;
       90          if (replicatingDataSizeMB >= Config.replication_max_parallel_data_size_mb) {
       91              throw new RuntimeException("The replicating data size in all running replication jobs "
       92                      + replicatingDataSizeMB
       93                      + "(MB) exceeds replication_max_parallel_data_size_mb: "
       94                      + Config.replication_max_parallel_data_size_mb);
       95          }
       96  
       97          if (runningJobs.putIfAbsent(job.getTableId(), job) != null) {
       98              throw new AlreadyExistsException("Replication job of table " + job.getTableId() + " is already running");
       99          }
      100  
      101          committedJobs.remove(job.getTableId()); // If the job is committed before, remove it from committed jobs
      102          abortedJobs.remove(job.getTableId()); // If the job is aborted before, remove it from aborted jobs
      103  
      104          LOG.info("Added replication job, database id: {}, table id: {}, "
      105                  + "replication data size: {}, current replicating data size: {}(MB)",
      106                  job.getDatabaseId(), job.getTableId(), job.getReplicationDataSize(), replicatingDataSizeMB);
      107      }

      157      public void replayReplicationJob(ReplicationJob replicationJob) {
      158          if (replicationJob.getState().equals(ReplicationJobState.COMMITTED)) {
      159              committedJobs.put(replicationJob.getTableId(), replicationJob);
      160              runningJobs.remove(replicationJob.getTableId());
      161          } else if (replicationJob.getState().equals(ReplicationJobState.ABORTED)) {
      162              abortedJobs.put(replicationJob.getTableId(), replicationJob);
      163              runningJobs.remove(replicationJob.getTableId());
      164          } else {
      165              runningJobs.put(replicationJob.getTableId(), replicationJob);
      166          }
      167      }
