      121      public void createExercisesExport(Path workingDirectory, User user) throws IOException {
      122          // retrieve all exercises as we cannot retrieve the exercises by course because a user might have participated in a course they are no longer a member of (they have

      125          var exerciseParticipationsPerCourse = allExerciseParticipations.stream().collect(Collectors.groupingBy(Exercise::getCourseViaExerciseGroupOrCourseMember));
      126          for (var entry : exerciseParticipationsPerCourse.entrySet()) {
      127              var course = entry.getKey();
      128              Path courseDir = retrieveCourseDirPath(workingDirectory, course);
      129              var exercises = entry.getValue();
      130              Path exercisesDir = courseDir.resolve("exercises");
      131              if (!exercises.isEmpty()) {
      132                  createDirectoryIfNotExistent(exercisesDir);
      133              }
      134              for (var exercise : exercises) {
      135                  if (exercise instanceof ProgrammingExercise programmingExercise) {
      136                      createProgrammingExerciseExport(programmingExercise, exercisesDir, user);
      137                  }
      138                  else {
      139                      createNonProgrammingExerciseExport(exercise, exercisesDir, user);
      140                  }
      141              }

      159          }

      161          RepositoryExportOptionsDTO repositoryExportOptions = new RepositoryExportOptionsDTO();
      162          var listOfProgrammingExerciseParticipations = programmingExercise.getStudentParticipations().stream()
      163                  .filter(studentParticipation -> studentParticipation instanceof ProgrammingExerciseStudentParticipation)
      164                  .map(studentParticipation -> (ProgrammingExerciseStudentParticipation) studentParticipation).toList();
      165  
      166          // we use this directory only to clone the repository and don't do this in our current directory because the current directory is part of the final data export

      168          var tempRepoWorkingDir = fileService.getTemporaryUniqueSubfolderPath(repoClonePath, 10);
      169          // NOTE: repositoryExportOptions.filterLateSubmissions must be false here because we do not want to filter any submissions for the data export
      170          programmingExerciseExportService.exportStudentRepositories(programmingExercise, listOfProgrammingExerciseParticipations, Map.of(), tempRepoWorkingDir, exerciseDir,

      173          createPlagiarismCaseInfoExport(programmingExercise, exerciseDir, user.getId());

      193      /**
      194       * Creates the export for the submission of the user to the given exercise.
      195       * <p>
      196       * Includes the submission information and the submission content and the results if the results are
      197       * published.
      198       * For quiz exercises it delegates the creation of the export to {@link DataExportQuizExerciseCreationService}.
      199       *
      200       * @param exercise    the exercise for which the export should be created
      201       * @param exerciseDir the directory in which the export should be created
      202       * @param user        the user for which the export should be created
      203       */
      204      private void createSubmissionsResultsExport(Exercise exercise, Path exerciseDir, User user) throws IOException {
      205          // quizzes do not have an assessment due date, so we need to check if they have ended according to their due date
      206          boolean isInstructor = authCheckService.isAtLeastInstructorForExercise(exercise, user);
      207          boolean includeResults = (exercise.isExamExercise() && exercise.getExam().resultsPublished())
      208                  || (exercise.isCourseExercise() && ExerciseDateService.isAfterAssessmentDueDate(exercise) && !(exercise instanceof QuizExercise))
      209                  || (exercise.isCourseExercise() && exercise instanceof QuizExercise quizExercise && quizExercise.isQuizEnded()) || isInstructor;
      210          for (var participation : exercise.getStudentParticipations()) {
      211              for (var submission : participation.getSubmissions()) {
      212                  createSubmissionCsvFile(submission, exerciseDir);
      213                  if (submission instanceof FileUploadSubmission fileUploadSubmission) {
      214                      copyFileUploadSubmissionFile(FilePathConverter.buildFileUploadSubmissionPath(exercise.getId(), submission.getId()), exerciseDir, fileUploadSubmission);
      215                  }
      216                  else if (submission instanceof TextSubmission textSubmission) {
      217                      storeTextSubmissionContent(textSubmission, exerciseDir);
      218                  }
      219                  else if (submission instanceof ModelingSubmission modelingSubmission) {
      220                      storeModelingSubmissionContent(modelingSubmission, exerciseDir);
      221                  }
      222                  else if (submission instanceof QuizSubmission) {
      223                      dataExportQuizExerciseCreationService.createQuizAnswersExport((QuizExercise) exercise, participation, exerciseDir, includeResults);
      224                  }
      225                  // for a programming exercise, we want to include the results that are visible before the assessment due date
      226                  if (includeResults || exercise instanceof ProgrammingExercise) {
      227                      boolean programmingExerciseBeforeAssessmentDueDate = exercise instanceof ProgrammingExercise && !ExerciseDateService.isAfterAssessmentDueDate(exercise);
      228                      createResultsAndComplaintFiles(submission, exerciseDir, user, programmingExerciseBeforeAssessmentDueDate, isInstructor);
      229                  }
      230              }
      231          }
      232      }

      241      private void storeModelingSubmissionContent(ModelingSubmission modelingSubmission, Path outputDir) throws IOException {
      242          if (modelingSubmission.getModel() == null) {

      244              return;
      245          }
      246          var fileName = "submission_" + modelingSubmission.getId();
      247          if (modelingApollonApi.isEmpty()) {
      248              log.warn("Cannot include modeling submission content in data export as pdf because apollon profile is not active. Going to include the json file");
      249              addModelJsonWithExplanationHowToView(modelingSubmission.getModel(), outputDir, fileName);
      250              return;
      251          }
      252  
      253          try (var modelAsPdf = modelingApollonApi.get().convertModel(modelingSubmission.getModel())) {
      254              FileUtils.writeByteArrayToFile(outputDir.resolve(fileName + PDF_FILE_EXTENSION).toFile(), modelAsPdf.readAllBytes());
      255          }
      256          catch (Exception e) {
      257              log.warn("Failed to include the model as pdf, going to include it as plain JSON file.", e);
      258              addModelJsonWithExplanationHowToView(modelingSubmission.getModel(), outputDir, fileName);
      259          }
      260      }

      298      /**
      299       * Creates a txt file containing the results with the score, the number of passed test cases if it is a programming exercise
      300       * and the feedbacks (both manual and automatic).
      301       *
      302       * @param submission                                 the submission for which the results should be stored
      303       * @param outputDir                                  the directory in which the results should be stored
      304       * @param user                                       the user for which the export should be created
      305       * @param programmingExerciseBeforeAssessmentDueDate whether the programming exercise is before the assessment due date
      306       * @param isInstructor                               whether the user is an instructor in the course the exercise belongs to
      307       * @throws IOException if the file cannot be written
      308       */
      309      private void createResultsAndComplaintFiles(Submission submission, Path outputDir, User user, boolean programmingExerciseBeforeAssessmentDueDate, boolean isInstructor)
      310              throws IOException {
      311          StringBuilder resultScoreAndFeedbacks = new StringBuilder();
      312          for (var result : submission.getResults()) {
      313              if (result != null) {
      314                  // Do not include the results if the assessment due date is in the future and the assessment is not automatic and the user is not an instructor
      315                  // We only consider programming exercises here because for other exercises this method is not called if the assessment due date is in the future
      316                  if (programmingExerciseBeforeAssessmentDueDate && result.getAssessmentType() != AssessmentType.AUTOMATIC && !isInstructor) {
      317                      continue;
      318                  }
      319                  resultService.filterSensitiveInformationIfNecessary(submission.getParticipation(), List.of(result), Optional.of(user));
      320                  var score = result.getScore();
      321                  if (score != null) {
      322                      resultScoreAndFeedbacks.append("Score of submission: ").append(score).append("%").append(" ")
      323                              .append(roundToNDecimalPlaces(score * submission.getParticipation().getExercise().getMaxPoints() / 100, 2)).append(" Points").append("\n");
      324                  }
      325                  if (submission instanceof ProgrammingSubmission && result.getPassedTestCaseCount() != null && result.getTestCaseCount() != null && result.getTestCaseCount() > 0) {
      326                      resultScoreAndFeedbacks.append("Passed test cases: ").append(result.getPassedTestCaseCount()).append("/").append(result.getTestCaseCount()).append("\n");
      327                  }
      328                  if (submission instanceof ProgrammingSubmission programmingSubmission && programmingSubmission.isBuildFailed()) {
      329                      resultScoreAndFeedbacks.append("Build failed").append("\n");
      330                  }
      331                  for (var feedback : result.getFeedbacks()) {
      332                      if (feedback != null) {
      333                          resultScoreAndFeedbacks.append("- Feedback: ");
      334  
      335                          // getTestCase() is null if it's manual feedback or if the test name was filtered out above
      336                          if (feedback.getTestCase() != null && feedback.getTestCase().getTestName() != null) {
      337                              resultScoreAndFeedbacks.append(feedback.getTestCase().getTestName()).append("\t");
      338                          }
      339                          // getText() contains, e.g., the file and line number for programming exercises
      340                          if (feedback.getText() != null) {
      341                              resultScoreAndFeedbacks.append(feedback.getText()).append("\t");
      342                          }
      343                          // getDetailText() contains the text message produced by the test case or written by the assessor.
      344                          // Usually null if the test case passes
      345                          if (feedback.getDetailText() != null) {
      346                              resultScoreAndFeedbacks.append(feedback.getDetailText()).append("\t");
      347                          }
      348                          if (feedback.getCredits() != null) {
      349                              resultScoreAndFeedbacks.append(feedback.getCredits());
      350                          }
      351                          resultScoreAndFeedbacks.append("\n");
      352                      }
      353                  }
      354                  FileUtils.writeStringToFile(outputDir.resolve("submission_" + submission.getId() + "_result_" + result.getId() + TXT_FILE_EXTENSION).toFile(),
      355                          resultScoreAndFeedbacks.toString(), StandardCharsets.UTF_8);
      356              }
      357              resultScoreAndFeedbacks = new StringBuilder();
      358          }
      359          var possibleComplaint = complaintRepository.findWithEagerComplaintResponseByResultSubmissionId(submission.getId());
      360          if (possibleComplaint.isPresent()) {
      361              addComplaintData(possibleComplaint.get(), outputDir);
      362          }
      363      }

      374      private void addComplaintData(Complaint complaint, Path outputDir) throws IOException {
      375          List<String> headers = new ArrayList<>();
      376          var dataStreamBuilder = Stream.builder();
      377          headers.add("id");
      378          dataStreamBuilder.add(complaint.getId());
      379          headers.add("submitted at");
      380          dataStreamBuilder.add(complaint.getSubmittedTime());
      381          headers.add("complaint type");
      382          dataStreamBuilder.add(complaint.getComplaintType());
      383          headers.add("complaint text");
      384          dataStreamBuilder.add(complaint.getComplaintText());
      385  
      386          if (complaint.getComplaintResponse() != null) {
      387              headers.add("complaint response");
      388              dataStreamBuilder.add(complaint.getComplaintResponse().getResponseText());
      389          }
      390          if (complaint.isAccepted() != null && ComplaintType.COMPLAINT == complaint.getComplaintType()) {
      391              headers.add("accepted");
      392              dataStreamBuilder.add(complaint.isAccepted());
      393          }
      394          CSVFormat csvFormat = CSVFormat.DEFAULT.builder().setHeader(headers.toArray(String[]::new)).get();
      395          var prefix = complaint.getComplaintType() == ComplaintType.COMPLAINT ? "complaint_" : "more_feedback_";
      396  
      397          try (final var printer = new CSVPrinter(Files.newBufferedWriter(outputDir.resolve(prefix + complaint.getId() + CSV_FILE_EXTENSION)), csvFormat)) {
      398              printer.printRecord(dataStreamBuilder.build());
      399              printer.flush();

      411      private void createPlagiarismCaseInfoExport(Exercise exercise, Path exercisePath, long userId) throws IOException {
      412          if (plagiarismCaseApi.isEmpty()) {

      416          var plagiarismCaseOptional = plagiarismCaseApi.get().findByStudentIdAndExerciseIdWithPostAndAnswerPost(userId, exercise.getId());
      417          if (plagiarismCaseOptional.isEmpty()) {

      441              dataStreamBuilder.add(plagiarismCase.getVerdictMessage());
      442          }

      446                  csvFormat)) {
      447              printer.printRecord(dataStreamBuilder.build());
      448              printer.flush();
      449          }
