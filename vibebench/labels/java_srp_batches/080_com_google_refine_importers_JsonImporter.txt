      114      final static private JsonNode parseForPreview(JsonParser parser, PreviewParsingState state, JsonToken token)
      115              throws JsonParseException, IOException {
      116          if (token != null) {
      117              switch (token) {
      118                  case START_ARRAY:
      119                      return parseArrayForPreview(parser, state);
      120                  case START_OBJECT:
      121                      return parseObjectForPreview(parser, state);
      122                  case VALUE_STRING:
      123                      return new TextNode(parser.getText());
      124                  case VALUE_NUMBER_INT:
      125                      return new LongNode(parser.getLongValue());
      126                  case VALUE_NUMBER_FLOAT:
      127                      return new DoubleNode(parser.getDoubleValue());
      128                  case VALUE_TRUE:
      129                      return BooleanNode.getTrue();
      130                  case VALUE_FALSE:
      131                      return BooleanNode.getFalse();
      132                  case VALUE_NULL:
      133                      return null;
      134                  case END_ARRAY:
      135                  case END_OBJECT:
      136                  case FIELD_NAME:
      137                  case NOT_AVAILABLE:
      138                  case VALUE_EMBEDDED_OBJECT:
      139                  default:
      140                      break;
      141              }
      142          }

      152      final static private ObjectNode parseObjectForPreview(JsonParser parser, PreviewParsingState state) throws IOException {
      153          ObjectNode result = ParsingUtilities.mapper.createObjectNode();
      154          loop: while (state.tokenCount < PREVIEW_PARSING_LIMIT) {
      155              JsonToken token = parser.nextToken();
      156              if (token == null) {
      157                  break;
      158              }
      159              state.tokenCount++;
      160  
      161              switch (token) {
      162                  case FIELD_NAME:
      163                      String fieldName = parser.getText();
      164                      JsonNode fieldValue = parseForPreview(parser, state);
      165                      JSONUtilities.safePut(result, fieldName, fieldValue);
      166                      break;
      167                  case END_OBJECT:
      168                      break loop;
      169                  default:
      170                      break loop;
      171              }
      172          }
      173          return result;
      174      }

      176      final static private ArrayNode parseArrayForPreview(JsonParser parser, PreviewParsingState state) throws IOException {
      177          ArrayNode result = ParsingUtilities.mapper.createArrayNode();
      178          loop: while (state.tokenCount < PREVIEW_PARSING_LIMIT) {
      179              JsonToken token = parser.nextToken();
      180              if (token == null) {
      181                  break;
      182              }
      183              state.tokenCount++;
      184  
      185              switch (token) {
      186                  case END_ARRAY:
      187                      break loop;
      188                  default:
      189                      JsonNode element = parseForPreview(parser, state, token);
      190                      result.add(element);
      191              }
      192          }
      193          return result;

      302          private Serializable getValue(JsonParser parser, JsonToken token) throws IOException {
      303              if (token != null) {
      304                  switch (token) {
      305                      case VALUE_STRING:
      306                          return parser.getText();
      307                      case VALUE_NUMBER_INT:
      308                          if (parser.getNumberType() == NumberType.INT || parser.getNumberType() == NumberType.LONG) {
      309                              return Long.valueOf(parser.getLongValue());
      310                          } else {
      311                              return parser.getNumberValue();
      312                          }
      313                      case VALUE_NUMBER_FLOAT:
      314                          if (parser.getNumberType() == NumberType.FLOAT) {
      315                              return Float.valueOf(parser.getFloatValue());
      316                          } else if (parser.getNumberType() == NumberType.DOUBLE) {
      317                              return Double.valueOf(parser.getDoubleValue());
      318                          } else {
      319                              return parser.getNumberValue();
      320                          }
      321                      case VALUE_TRUE:
      322                          return Boolean.TRUE;
      323                      case VALUE_FALSE:
      324                          return Boolean.FALSE;
      325                      case VALUE_NULL:

      335          @Override
      336          public Token next() throws TreeReaderException {
      337              JsonToken previous = current;
      338              current = next;
      339              next = null; // in case an exception is thrown
      340              try {
      341                  if (current != null) {
      342                      if (current.isScalarValue()) {
      343                          fieldValue = getValue(parser, current);
      344                      } else {
      345                          fieldValue = null;
      346                      }
      347                      if (current == JsonToken.FIELD_NAME) {
      348                          fieldName = parser.getText();
      349                      } else if (current == JsonToken.START_ARRAY
      350                              || current == JsonToken.START_OBJECT) {
      351                          // Use current field name for next level object
      352                          // ie elide one level of anonymous fields
      353                          if (previous != JsonToken.FIELD_NAME) {
      354                              fieldName = ANONYMOUS;
      355                          }
      356                      }
      357                  }
      358                  next = parser.nextToken();
      359              } catch (JsonParseException e) {
      360                  throw new TreeReaderException(e.getOriginalMessage());
      361              } catch (IOException e) {
      362                  throw new TreeReaderException(e);
      363              }
      364              return current();
