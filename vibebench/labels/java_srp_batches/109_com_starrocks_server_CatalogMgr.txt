      101      public void createCatalog(String type, String catalogName, String comment, Map<String, String> properties)
      102              throws DdlException {
      103          CatalogConnector connector = null;
      104          Catalog catalog = null;
      105          writeLock();
      106          try {
      107              if (Strings.isNullOrEmpty(type)) {
      108                  throw new DdlException("Missing properties 'type'");
      109              }
      110  
      111              Preconditions.checkState(!catalogs.containsKey(catalogName), "Catalog '%s' already exists", catalogName);
      112  
      113              try {
      114                  Preconditions.checkState(!catalogs.containsKey(catalogName), "Catalog '%s' already exists", catalogName);
      115  
      116                  connector = connectorMgr.createConnector(
      117                          new ConnectorContext(catalogName, type, properties), false);
      118                  if (null == connector) {
      119                      LOG.error("{} connector [{}] create failed", type, catalogName);
      120                      throw new DdlException("connector create failed");
      121                  }
      122                  long id = isResourceMappingCatalog(catalogName) ?
      123                          ConnectorTableId.CONNECTOR_ID_GENERATOR.getNextId().asInt() :
      124                          GlobalStateMgr.getCurrentState().getNextId();
      125                  catalog = new ExternalCatalog(id, catalogName, comment, properties);
      126                  catalogs.put(catalogName, catalog);
      127  
      128                  if (!isResourceMappingCatalog(catalogName)) {
      129                      GlobalStateMgr.getCurrentState().getEditLog().logCreateCatalog(catalog);
      130                  }
      131              } catch (StarRocksConnectorException e) {
      132                  LOG.error("connector create failed. catalog [{}] ", catalogName, e);
      133                  throw new DdlException(String.format("connector create failed {%s}", e.getMessage()));
      134              }
      135          } catch (Exception e) {
      136              if (connector != null && connectorMgr.connectorExists(catalogName)) {
      137                  connectorMgr.removeConnector(catalogName);
      138              }
      139  
      140              if (catalog != null) {
      141                  catalogs.remove(catalogName);
      142              }
      143  
      144              throw e;
      145          } finally {
      146              writeUnLock();
      147          }

      150      private void reCreatCatalog(Catalog catalog, Map<String, String> alterProperties, boolean isReplay) throws DdlException {
      151          String catalogName = catalog.getName();
      152          String type = catalog.getType();
      153          CatalogConnector newConnector = null;
      154  
      155          writeLock();
      156          try {
      157              Map<String, String> newProperties = new HashMap<>(catalog.getConfig().size() + alterProperties.size());
      158              newProperties.putAll(catalog.getConfig());
      159              newProperties.putAll(alterProperties);
      160  
      161              LOG.info("Recreate catalog [{}] with properties [{}]", catalogName, newProperties);
      162  
      163              newConnector = connectorMgr.createHiddenConnector(
      164                      new ConnectorContext(catalogName, type, newProperties), isReplay);
      165              if (null == newConnector) {
      166                  throw new DdlException("Create connector failed");
      167              }
      168  
      169              // drop old connector
      170              connectorMgr.removeConnector(catalogName);
      171  
      172              // replace old connector with new connector
      173              connectorMgr.addConnector(catalogName, newConnector);
      174  
      175              catalog.getConfig().putAll(alterProperties);
      176          } catch (Exception e) {
      177              LOG.error("Recreate catalog failed. catalog [{}] ", catalogName, e);
      178  
      179              if (newConnector != null) {
      180                  newConnector.shutdown();
      181              }
      182  
      183              throw new DdlException(String.format("Alter catalog failed, msg: [%s]", e.getMessage()), e);
      184          } finally {
      185              writeUnLock();
      186          }
      187      }

      243      private void alterCatalog(Catalog catalog, Map<String, String> properties, boolean isReplay) throws DdlException {
      244          Map<String, String> alterProperties = new HashMap<>(properties.size());
      245          Map<String, String> oldProperties = catalog.getConfig();
      246  
      247          for (String confName : properties.keySet()) {
      248              String oldVal = oldProperties.get(confName);
      249              String newVal = properties.get(confName);
      250              if (!oldProperties.containsKey(confName) || !Objects.equals(oldVal, newVal)) {
      251                  alterProperties.put(confName, newVal);
      252              }
      253          }
      254  
      255          if (alterProperties.isEmpty()) {
      256              return;
      257          }
      258  
      259          reCreatCatalog(catalog, alterProperties, isReplay);
      260      }

      299      public void replayCreateCatalog(Catalog catalog) throws DdlException {
      300          String type = catalog.getType();
      301          String catalogName = catalog.getName();
      302          Map<String, String> config = catalog.getConfig();
      303          CatalogConnector catalogConnector = null;
      304  
      305          try {
      306              if (Strings.isNullOrEmpty(type)) {
      307                  throw new DdlException("Missing properties 'type'");
      308              }
      309  
      310              // skip unsupported connector type
      311              if (!ConnectorType.isSupport(type)) {
      312                  LOG.error("Replay catalog [{}] encounter unknown catalog type [{}], ignore it", catalogName, type);
      313                  return;
      314              }
      315  
      316              readLock();
      317              try {
      318                  Preconditions.checkState(!catalogs.containsKey(catalogName), "Catalog '%s' already exists", catalogName);
      319              } finally {
      320                  readUnlock();
      321              }
      322  
      323              try {
      324                  catalogConnector = connectorMgr.createConnector(
      325                          new ConnectorContext(catalogName, type, config), true);
      326                  if (catalogConnector == null) {
      327                      LOG.error("{} connector [{}] create failed.", type, catalogName);
      328                      throw new DdlException("connector create failed");
      329                  }
      330              } catch (StarRocksConnectorException e) {
      331                  LOG.error("connector create failed [{}], reason {}", catalogName, e.getMessage());
      332                  throw new DdlException(String.format("connector create failed: %s", e.getMessage()));
      333              }
      334  
      335              writeLock();
      336              try {
      337                  catalogs.put(catalogName, catalog);
      338              } finally {
      339                  writeUnLock();
      340              }
      341          } catch (Exception e) {
      342              if (catalogConnector != null && connectorMgr.connectorExists(catalogName)) {
      343                  connectorMgr.removeConnector(catalogName);
      344              }
      345  
      346              catalogs.remove(catalogName);
      347              throw e;
      348          }

      389          List<Resource> resources = GlobalStateMgr.getCurrentState().getResourceMgr().getNeedMappingCatalogResources();
      390          for (Resource resource : resources) {
      391              Map<String, String> properties = Maps.newHashMap(resource.getProperties());
      392              String type = resource.getType().name().toLowerCase(Locale.ROOT);
      393              if (!NEED_MAPPING_CATALOG_RESOURCES.contains(type)) {
      394                  return;
      395              }

      397              String catalogName = getResourceMappingCatalogName(resource.getName(), type);
      398              properties.put("type", type);

      400              try {
