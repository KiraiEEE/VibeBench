      196      boolean updateData(String key, T cur) {
      197          MutableBoolean updated = new MutableBoolean();
      198          data.compute(key, (k, prev) -> {
      199              if (Objects.equals(prev, cur)) {
      200                  if (log.isDebugEnabled()) {
      201                      log.debug("{} skipped item key={} value={} prev={}",
      202                              name, key, cur, prev);
      203                  }
      204                  updated.setValue(false);
      205                  return prev;
      206              } else {
      207                  updated.setValue(true);
      208                  return cur;
      209              }
      210          });
      211          return updated.booleanValue();
      212      }

      214      public void handleSessionEvent(SessionEvent sessionEvent) {
      215          if (CollectionUtils.isEmpty(outdatedItemListeners)) {
      216              log.warn("{} Skipped handle metadata store session event {} because does not set itemOutdatedListeners",
      217                  name, sessionEvent);
      218              return;
      219          }
      220          if (sessionEvent == SessionEvent.SessionLost) {
      221              Map<String, T> snapshot = new HashMap<>(data);
      222              log.warn("{} clearing owned bundles because metadata store session lost {}",  name, snapshot);
      223              for (Map.Entry<String, T> entry : snapshot.entrySet()) {
      224                  for (var listener : outdatedItemListeners) {
      225                      try {
      226                          listener.accept(entry.getKey(), entry.getValue());
      227                          if (clearUntrustedData) {
      228                              data.remove(entry.getKey());
      229                          }
      230                      } catch (Throwable e) {
      231                          if (tableViewShutDownListener == null) {
      232                              log.warn("{} failed to listen item whose state is unknown because of metadata store"
      233                                      + " session lost. key:{}, val:{}", name, entry.getKey(), entry.getValue(), e);
      234                          } else {
      235                              log.warn("{} Shutdown table view, because failed to listen item whose state is unknown due"
      236                                      + " to metadata store session lost. key:{}, val:{}", name, entry.getKey(),
      237                                      entry.getValue(), e);
      238                              tableViewShutDownListener.accept(e);
      239                          }
      240                          break;
      241                      }
      242                  }
      243              }
      244          } else if (sessionEvent == SessionEvent.SessionReestablished) {
      245              log.info("{} Refilling bundle owner list after metadata store session reestablished",  name);
      246              // Since just get a session expired issue, we'd better to print the initialize detail logs.
      247              fillAsync(null, true).exceptionally(ex -> {
      248                  if (tableViewShutDownListener == null) {
      249                      log.warn("{} failed to fill existing items after session reestablished", name, ex);
      250                  } else {
      251                      log.error("{} Shutdown table view because failed to fill existing items after session"
      252                          + " reestablished", name, ex);
      253                      tableViewShutDownListener.accept(ex);
      254                  }
      255                  return null;
      256              });
      257          }
      258      }

      339      private void fill() throws MetadataStoreException {
      340          AtomicLong loadedCounter = new AtomicLong();
      341          long maxWaitTime = Math.min(timeoutInMillis, FILL_TIMEOUT_IN_MILLIS);
      342          try {
      343              fillAsync(loadedCounter, false).get(maxWaitTime, TimeUnit.MILLISECONDS);
      344              log.info("{} completed filling existing items with size:{}", name, loadedCounter.get());
      345          } catch (InterruptedException | ExecutionException | TimeoutException e) {
      346              String err = name + " failed to fill existing items in "
      347                      + TimeUnit.MILLISECONDS.toSeconds(maxWaitTime) + " secs. Filled count:"
      348                      + loadedCounter.get();
      349              log.error(err);
      350              throw new MetadataStoreException(err, FutureUtil.unwrapCompletionException(e));
      351          }
      352      }

      354      private CompletableFuture<Void> handleExistingLeafs(String rootPath, String path, @Nullable AtomicLong count,
      355                                                          boolean printDetails) {
      356          return store.getChildren(path).thenCompose(children -> {
      357              if (children.isEmpty()) {
      358                  // Skip root path.
      359                  if (rootPath.equals(path)) {
      360                      return CompletableFuture.completedFuture(null);
      361                  }
      362                  // Leaf node.
      363                  if (count != null) {
      364                      count.incrementAndGet();
      365                  }
      366                  if (printDetails) {
      367                      log.info("handling exist leaf {}", path);
      368                  }
      369                  return handleExisting(path);
      370              } else {
      371                  // Dir node.
      372                  // The limitation of max quantity of concurrency reading is not needed, because the metadata store
      373                  // client works in a single thread, and all OPs will be packaged into a batch, the more the faster.
      374                  // If metadata store receives too many requests, it will split them into multi requests itself.
      375                  if (printDetails) {
      376                      log.info("handling exist dir {}", path);
      377                  }
      378                  List<CompletableFuture<Void>> futureList = new ArrayList<>();
      379                  for (var child : children) {
      380                      futureList.add(handleExistingLeafs(rootPath, path + "/" + child, count, printDetails));
      381                  }
      382                  return FutureUtil.waitForAll(futureList);
      383              }
      384          });
      385      }

      411      public CompletableFuture<Void> put(String key, T value) {
      412          String path = getPath(key);
      413          return cache.readModifyUpdateOrCreate(path, (old) -> {
      414              if (conflictResolver.test(old.orElse(null), value)) {
      415                  return value;
      416              } else {
      417                  throw new ConflictException(
      418                          String.format("Failed to update from old:%s to value:%s", old, value));
      419              }
      420          }).thenCompose(__ -> doHandleNotification(path)) // immediately notify local tableview
      421          .exceptionally(e -> {
      422              if (e.getCause() instanceof MetadataStoreException.BadVersionException) {
      423                  throw FutureUtil.wrapToCompletionException(new ConflictException(
      424                          String.format("Failed to update to value:%s", value)));
      425              }
      426  
      427              throw FutureUtil.wrapToCompletionException(e.getCause());
      428          });
      429      }
