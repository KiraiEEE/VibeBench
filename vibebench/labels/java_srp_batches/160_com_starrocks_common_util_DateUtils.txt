       77          DateTimeFormatterBuilder builder = new DateTimeFormatterBuilder();
       78          boolean escaped = false;
       79          for (int i = 0; i < pattern.length(); i++) {
       80              char character = pattern.charAt(i);
       81              if (escaped) {

       84                          builder.appendValue(ChronoField.MONTH_OF_YEAR, 1, 2, SignStyle.NORMAL);
       85                          break;
       86                      case 'm': // %m Month, numeric (00..12)
       87                          if (isOutputFormat) {
       88                              builder.appendValue(ChronoField.MONTH_OF_YEAR, 2);
       89                          } else {
       90                              builder.appendValue(ChronoField.MONTH_OF_YEAR, 1, 2, SignStyle.NORMAL);
       91                          }
       92                          break;
       93                      case 'd': // %d Day of the month, numeric (00..31)
       94                          if (isOutputFormat) {
       95                              builder.appendValue(ChronoField.DAY_OF_MONTH, 2);
       96                          } else {
       97                              builder.appendValue(ChronoField.DAY_OF_MONTH, 1, 2, SignStyle.NORMAL);
       98                          }
       99                          break;
      100                      case 'e': // %e Day of the month, numeric (0..31)
      101                          builder.appendValue(ChronoField.DAY_OF_MONTH, 1, 2, SignStyle.NORMAL);
      102                          break;
      103                      case 'H': // %H Hour (00..23)
      104                          builder.appendValue(ChronoField.HOUR_OF_DAY, 2);
      105                          break;
      106                      case 'k': // %k Hour (0..23)
      107                          builder.appendValue(ChronoField.HOUR_OF_DAY, 1, 2, SignStyle.NORMAL);
      108                          break;
      109                      case 'h': // %h Hour (01..12)
      110                      case 'I': // %I Hour (01..12)
      111                          builder.appendValue(ChronoField.CLOCK_HOUR_OF_AMPM, 2);
      112                          break;
      113                      case 'l': // %l Hour (1..12)
      114                          builder.appendValue(ChronoField.CLOCK_HOUR_OF_AMPM, 1, 2, SignStyle.NORMAL);
      115                          break;
      116                      case 'i': // %i Minutes, numeric (00..59)
      117                          builder.appendValue(ChronoField.MINUTE_OF_HOUR, 2);
      118                          break;
      119                      case 'j': // %j Day of year (001..366)
      120                          builder.appendValue(ChronoField.DAY_OF_YEAR, 3);
      121                          break;
      122                      case 'S': // %S Seconds (00..59)
      123                      case 's': // %s Seconds (00..59)
      124                          builder.appendValue(ChronoField.SECOND_OF_MINUTE, 2);
      125                          break;

      128                                  .appendLiteral(':')

      132                          break;

      134                          builder.appendValue(ChronoField.ALIGNED_WEEK_OF_YEAR, 2);
      135                          break;
      136                      case 'Y': // %Y Year, numeric, four digits
      137                          builder.appendValue(ChronoField.YEAR, 4);

      141                                  .parseDefaulting(ChronoField.ERA, 1);

      153                      case 'p': // %p AM or PM, Java can't convert language
