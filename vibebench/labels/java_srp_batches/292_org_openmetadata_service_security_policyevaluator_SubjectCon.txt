      509      /** Policy iterator for a team */
      510      TeamPolicyIterator(UUID teamId, List<UUID> teamsVisited, boolean skipRoles) {
      511        Team team = Entity.getEntity(Entity.TEAM, teamId, TEAM_FIELDS, Include.NON_DELETED);
      512  
      513        // If a team is already visited (because user can belong to multiple teams
      514        // and a team can belong to multiple teams) then don't visit the roles/policies of that team
      515        // This also protects against circular dependencies in team hierarchy
      516        if (!teamsVisited.contains(teamId)) {
      517          teamsVisited.add(teamId);
      518          if (!skipRoles && team.getDefaultRoles() != null) {
      519            iterators.add(
      520                new RolePolicyIterator(Entity.TEAM, team.getName(), team.getDefaultRoles()));
      521          }
      522          if (team.getPolicies() != null) {
      523            iterators.add(new PolicyIterator(Entity.TEAM, team.getName(), null, team.getPolicies()));
      524          }
      525          for (EntityReference parentTeam : listOrEmpty(team.getParents())) {
      526            iterators.add(new TeamPolicyIterator(parentTeam.getId(), teamsVisited, skipRoles));
      527          }
      528        } else {
      529          LOG.warn(
      530              "Circular dependency detected in team hierarchy for team: {}. Skipping to prevent infinite loop.",
      531              team.getName());
      532        }
      533      }
