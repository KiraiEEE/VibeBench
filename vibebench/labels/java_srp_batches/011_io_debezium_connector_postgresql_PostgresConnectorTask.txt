      269      private SlotCreationResult tryToCreateSlot(Snapshotter snapshotter, PostgresConnectorConfig connectorConfig, SlotState slotInfo) {
      270  
      271          SlotCreationResult slotCreatedInfo = null;
      272          if (snapshotter.shouldStream()) {
      273              replicationConnection = createReplicationConnection(this.taskContext,
      274                      connectorConfig.maxRetries(), connectorConfig.retryDelay());
      275  
      276              // we need to create the slot before we start streaming if it doesn't exist
      277              // otherwise we can't stream back changes happening while the snapshot is taking place
      278              if (slotInfo == null) {
      279                  if (connectorConfig.isReadOnlyConnection()) {
      280                      LOGGER.warn("Connector is configured to be in read-only mode but replication slot was not found.\n" +
      281                              "The attempt to create it can fail. Please check you configuration in case.");
      282                  }
      283                  try {
      284                      slotCreatedInfo = replicationConnection.createReplicationSlot().orElse(null);
      285                  }
      286                  catch (SQLException ex) {
      287                      String message = "Creation of replication slot failed";
      288                      if (ex.getMessage().contains("already exists")) {
      289                          message += "; when setting up multiple connectors for the same database host, please make sure to use a distinct replication slot name for each.";
      290                      }
      291                      throw new DebeziumException(message, ex);
      292                  }
      293              }
      294          }
      295          return slotCreatedInfo;
      296      }

      312      public ReplicationConnection createReplicationConnection(PostgresTaskContext taskContext, int maxRetries, Duration retryDelay)
      313              throws ConnectException {
      314          final Metronome metronome = Metronome.parker(retryDelay, Clock.SYSTEM);
      315          short retryCount = 0;
      316          ReplicationConnection replicationConnection = null;
      317          while (retryCount <= maxRetries) {
      318              try {
      319                  return taskContext.createReplicationConnection(jdbcConnection);
      320              }
      321              catch (SQLException ex) {
      322                  retryCount++;
      323                  if (retryCount > maxRetries) {
      324                      LOGGER.error("Too many errors connecting to server. All {} retries failed.", maxRetries);
      325                      throw new ConnectException(ex);
      326                  }
      327  
      328                  LOGGER.warn("Error connecting to server; will attempt retry {} of {} after {} " +
      329                          "seconds. Exception message: {}", retryCount, maxRetries, retryDelay.getSeconds(), ex.getMessage());
      330                  try {
      331                      metronome.pause();
      332                  }
      333                  catch (InterruptedException e) {
      334                      LOGGER.warn("Connection retry sleep interrupted by exception: " + e);
      335                      Thread.currentThread().interrupt();
      336                  }
      337              }
      338          }
      339          return replicationConnection;
      340      }

      351      @Override
      352      protected void doStop() {
      353          // The replication connection is regularly closed at the end of streaming phase
      354          // in case of error it can happen that the connector is terminated before the stremaing
      355          // phase is started. It can lead to a leaked connection.
      356          // This is guard to make sure the connection is closed.
      357          try {
      358              if (replicationConnection != null) {
      359                  replicationConnection.close();
      360              }
      361          }
      362          catch (Exception e) {
      363              LOGGER.trace("Error while closing replication connection", e);
      364          }
      365  
      366          try {
      367              if (beanRegistryJdbcConnection != null) {
      368                  beanRegistryJdbcConnection.close();
      369              }
      370          }
      371          catch (Exception e) {
      372              LOGGER.trace("Error while closing JDBC bean registry connection", e);
      373          }
      374  
      375          if (jdbcConnection != null) {
      376              jdbcConnection.close();
      377          }
      378  
      379          if (schema != null) {
      380              schema.close();
      381          }
      382      }

      404      @Override
      405      public void performCommit() {
      406          boolean locked = commitLock.tryLock();
      407  
      408          if (locked) {
      409              try {
      410                  if (coordinator != null) {
      411                      Offsets<PostgresPartition, PostgresOffsetContext> offsets = this.getPreviousOffsets(this.partitionProvider, this.offsetContextLoader);
      412                      if (offsets.getOffsets() != null) {
      413                          offsets.getOffsets()
      414                                  .entrySet()
      415                                  .stream()
      416                                  .filter(e -> e.getValue() != null)
      417                                  .forEach(entry -> {
      418                                      Map<String, String> partition = entry.getKey().getSourcePartition();
      419                                      Map<String, ?> lastOffset = entry.getValue().getOffset();
      420                                      LOGGER.debug("Committing offset '{}' for partition '{}'", partition, lastOffset);
      421                                      coordinator.commitOffset(partition, lastOffset);
      422                                  });
      423                      }
      424                  }
      425              }
      426              finally {
      427                  commitLock.unlock();
      428              }
      429          }
      430          else {
      431              LOGGER.warn("Couldn't commit processed log positions with the source database due to a concurrent connector shutdown or restart");
      432          }
      433      }

      450      private static void checkWalLevel(PostgresConnection connection, SnapshotterService snapshotterService) throws SQLException {
      451          final String walLevel = connection.queryAndMap(
      452                  "SHOW wal_level",
      453                  connection.singleResultMapper(rs -> rs.getString("wal_level"), "Could not fetch wal_level"));
      454          if (!"logical".equals(walLevel)) {
      455  
      456              if (snapshotterService.getSnapshotter() != null && snapshotterService.getSnapshotter().shouldStream()) {
      457                  // Logical WAL_LEVEL is only necessary for CDC snapshotting
      458                  throw new SQLException("Postgres server wal_level property must be 'logical' but is: '" + walLevel + "'");
      459              }
      460              else {
      461                  LOGGER.warn("WAL_LEVEL check failed but this is ignored as CDC was not requested");
      462              }
      463          }
      464      }
