      337        String componentName = componentType == null ? null : req.getParams().get("componentName");
      338        boolean showParams = req.getParams().getBool("expandParams", false);
      339        Map<String, Object> map =
      340            Utils.convertToMap(this.req.getCore().getSolrConfig(), new LinkedHashMap<>());
      341        if (componentType != null && !SolrRequestHandler.TYPE.equals(componentType)) return map;
      342  
      343        @SuppressWarnings({"unchecked"})
      344        Map<String, Object> reqHandlers =
      345            (Map<String, Object>)
      346                map.computeIfAbsent(SolrRequestHandler.TYPE, k -> new LinkedHashMap<>());
      347        List<PluginInfo> plugins = this.req.getCore().getImplicitHandlers();
      348        for (PluginInfo plugin : plugins) {
      349          if (SolrRequestHandler.TYPE.equals(plugin.type)) {
      350            if (!reqHandlers.containsKey(plugin.name)) {
      351              reqHandlers.put(plugin.name, plugin);
      352            }
      353          }
      354        }
      355        if (!showParams) return map;
      356        for (Map.Entry<String, Object> e : reqHandlers.entrySet()) {
      357          if (componentName == null || e.getKey().equals(componentName)) {
      358            Map<String, Object> m = expandUseParams(req, e.getValue());

      360          }
      361        }

      446          switch (op.name) {

      464                  if (err != null) {
      465                    op.addError(err);
      466                    continue;
      467                  }
      468  
      469                  try {
      470                    val = (Map) entry.getValue();
      471                  } catch (Exception e1) {
      472                    op.addError("invalid params for key : " + key);
      473                    continue;
      474                  }
      475  
      476                  if (val.containsKey("")) {
      477                    op.addError("Empty keys are not allowed in params");
      478                    continue;
      479                  }
      480  
      481                  RequestParams.ParamSet old = params.getParams(key);
      482                  if (op.name.equals(UPDATE)) {
      483                    if (old == null) {
      484                      op.addError(formatString("unknown paramset {0} cannot update ", key));
      485                      continue;
      486                    }
      487                    params = params.setParams(key, old.update(val));
      488                  } else {
      489                    Long version = old == null ? 0 : old.getVersion() + 1;
      490                    params = params.setParams(key, RequestParams.createParamSet(val, version));
      491                  }
      492                }
      493                break;
      494              }
      495            case "delete":

      498                if (op.hasError()) break;
