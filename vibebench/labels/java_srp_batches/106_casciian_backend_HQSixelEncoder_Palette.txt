      204           * A bucket contains colors that will all be mapped to the same
      205           * weighted average color value.
      206           */
      207          private class Bucket {
      208  
      209              /**
      210               * The colors in this bucket.
      211               */
      212              private List<ColorIdx> colors;
      213  
      214              /**
      215               * The palette index for this bucket.  For now this points to a
      216               * simple average of all the colors, or black if no colors are in
      217               * this bucket.
      218               */
      219              public int index = 0;

      221              // The minimum and maximum, and "total" component values in this
      222              // bucket.
      223              private int minRed   = 0xFF;
      224              private int maxRed   = 0;
      225              private int minGreen = 0xFF;
      226              private int maxGreen = 0;
      227              private int minBlue  = 0xFF;
      228              private int maxBlue  = 0;

      231              private int lastAverage = -1;

      233              /**
      234               * Public constructor.
      235               *
      236               * @param n the expected number of colors that will be in this
      237               * bucket

      239              public Bucket(final int n) {
      240                  reset(n);
      241              }

      243              /**
      244               * Reset the stats.
      245               *
      246               * @param n the expected number of colors that will be in this
      247               * bucket
      248               */
      249              private void reset(final int n) {
      250                  colors      = new ArrayList<>(n);
      251                  minRed      = 0xFF;
      252                  maxRed      = 0;
      253                  minGreen    = 0xFF;
      254                  maxGreen    = 0;

      256                  maxBlue     = 0;

      271              /**
      272               * Add a color to the bucket.
      273               *
      274               * @param color the color to add
      275               */
      276              public void add(final ColorIdx color) {
      277                  colors.add(color);
      278  
      279                  int rgb   = color.color;
      280                  int red   = (rgb >>> 16) & 0xFF;
      281                  int green = (rgb >>>  8) & 0xFF;
      282                  int blue  =  rgb         & 0xFF;
      283                  if (red > maxRed) {
      284                      maxRed = red;
      285                  }
      286                  if (red < minRed) {
      287                      minRed = red;
      288                  }
      289                  if (green > maxGreen) {
      290                      maxGreen = green;
      291                  }
      292                  if (green < minGreen) {
      293                      minGreen = green;
      294                  }
      295                  if (blue > maxBlue) {
      296                      maxBlue = blue;
      297                  }
      298                  if (blue < minBlue) {
      299                      minBlue = blue;
      300                  }
      301              }

      303              /**
      304               * Partition this bucket into two buckets, split along the color
      305               * with the maximum range.
      306               *
      307               * @return the other bucket
      308               */
      309              public Bucket partition() {
      310                  int redDiff = Math.max(0, (maxRed - minRed));
      311                  int greenDiff = Math.max(0, (maxGreen - minGreen));
      312                  int blueDiff = Math.max(0, (maxBlue - minBlue));
      313                  if (verbosity >= 5) {
      314                      System.err.printf("partn colors %d Δr %d Δg %d Δb %d\n",
      315                          colors.size(), redDiff, greenDiff, blueDiff);
      316                  }
      317  
      318                  if ((redDiff > greenDiff) && (redDiff > blueDiff)) {
      319                      // Partition on red.
      320                      if (verbosity >= 5) {
      321                          System.err.println("    RED");
      322                      }
      323                      Collections.sort(colors, new Comparator<ColorIdx>() {
      324                          public int compare(ColorIdx c1, ColorIdx c2) {
      325                              int red1 = (c1.color >>> 16) & 0xFF;
      326                              int red2 = (c2.color >>> 16) & 0xFF;
      327                              return red1 - red2;
      328                          }
      329                      });
      330                  } else if ((greenDiff > blueDiff) && (greenDiff > redDiff)) {
      331                      // Partition on green.
      332                      if (verbosity >= 5) {
      333                          System.err.println("    GREEN");
      334                      }
      335                      Collections.sort(colors, new Comparator<ColorIdx>() {
      336                          public int compare(ColorIdx c1, ColorIdx c2) {
      337                              int green1 = (c1.color >>> 8) & 0xFF;
      338                              int green2 = (c2.color >>> 8) & 0xFF;
      339                              return green1 - green2;
      340                          }
      341                      });
      342                  } else {
      343                      // Partition on blue.
      344                      if (verbosity >= 5) {
      345                          System.err.println("    BLUE");
      346                      }
      347                      Collections.sort(colors, new Comparator<ColorIdx>() {
      348                          public int compare(ColorIdx c1, ColorIdx c2) {
      349                              int blue1 = c1.color & 0xFF;
      350                              int blue2 = c2.color & 0xFF;
      351                              return blue1 - blue2;
      352                          }
      353                      });
      354                  }
      355  
      356                  int oldN = colors.size();
      357  
      358                  List<ColorIdx> newBucketColors;
      359                  newBucketColors = colors.subList(oldN / 2, oldN);
      360                  Bucket newBucket = new Bucket(newBucketColors.size());
      361                  for (ColorIdx color: newBucketColors) {
      362                      newBucket.add(color);
      363                  }
      364  
      365                  List<ColorIdx> newColors;
      366                  newColors = colors.subList(0, oldN - newBucketColors.size());
      367                  reset(newColors.size());
      368                  for (ColorIdx color: newColors) {
      369                      add(color);
      370                  }
      371                  assert (newBucketColors.size() + newColors.size() == oldN);
      372                  return newBucket;
      373              }

      375              /**
      376               * Average the colors in this bucket.
      377               *
      378               * @return an averaged RGB value
      379               */
      380              public int average() {
      381                  if (lastAverage != -1) {
      382                      return lastAverage;
      383                  }
      384  
      385                  if (quantizationDone) {
      386                      if (colors.size() == 0) {
      387                          lastAverage = 0xFF000000;
      388                          return lastAverage;
      389                      }
      390                      int sixelColor = sixelColors.get(index);
      391                      if ((sixelColor == 0xFF000000)
      392                          || (sixelColor == 0xFF646464)
      393                      ) {
      394                          // This bucket is mapped to black or white.
      395                          lastAverage = sixelColor;
      396                          return lastAverage;
      397                      }
      398                  }
      399  
      400                  // Compute the average color.
      401                  long totalRed = 0;
      402                  long totalGreen = 0;
      403                  long totalBlue = 0;
      404                  long count = 0;
      405                  for (ColorIdx color: colors) {
      406                      int rgb = color.color;
      407                      int red   = (rgb >>> 16) & 0xFF;
      408                      int green = (rgb >>>  8) & 0xFF;
      409                      int blue  =  rgb         & 0xFF;
      410                      totalRed   += (long) color.count * red;
      411                      totalGreen += (long) color.count * green;
      412                      totalBlue  += (long) color.count * blue;
      413                      count += color.count;
      414                  }
      415                  if (count == 0) {
      416                      lastAverage = 0xFF000000;
      417                      return lastAverage;
      418                  }
      419                  totalRed   = clampSixel((int) (totalRed   / count));
      420                  totalGreen = clampSixel((int) (totalGreen / count));
      421                  totalBlue  = clampSixel((int) (totalBlue  / count));
      422  
      423                  lastAverage = (int) ((0xFF << 24) | (totalRed   << 16)
      424                                                    | (totalGreen <<  8)
      425                                                    |  totalBlue);
      426                  return lastAverage;
      427              }
