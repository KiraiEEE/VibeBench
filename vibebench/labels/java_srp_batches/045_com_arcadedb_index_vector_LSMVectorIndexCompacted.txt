      123    /**
      124     * Appends a vector entry during compaction using variable-sized encoding.
      125     * Entries are written sequentially without pointer tables for maximum space efficiency.
      126     *
      127     * @param currentPage                 The current page being written to (or null to create new)
      128     * @param compactedPageNumberOfSeries Counter for page series numbering
      129     * @param currentFileOffset           Tracks absolute file offset for entries (updated by this method)
      130     * @param vectorId                    The vector ID
      131     * @param rid                         The record ID
      132     * @param deleted                     Whether this entry is deleted
      133     *
      134     * @return CompactionAppendResult containing new pages and absolute file offset of the written entry
      135     */
      136    public CompactionAppendResult appendDuringCompaction(MutablePage currentPage,
      137        final AtomicInteger compactedPageNumberOfSeries, final AtomicLong currentFileOffset,
      138        final int vectorId, final RID rid, final boolean deleted) throws IOException, InterruptedException {
      139  
      140      final List<MutablePage> newPages = new ArrayList<>();
      141  
      142      if (currentPage == null) {
      143        // CREATE A NEW PAGE
      144        currentPage = createNewPage(compactedPageNumberOfSeries.getAndIncrement());
      145        newPages.add(currentPage);
      146      }
      147  
      148      // Calculate variable entry size (space needed for this specific entry)
      149      final int vectorIdSize = Binary.getNumberSpace(vectorId);
      150      final int bucketIdSize = Binary.getNumberSpace(rid.getBucketId());
      151      final int positionSize = Binary.getNumberSpace(rid.getPosition());
      152      final int entrySize = vectorIdSize + positionSize + bucketIdSize + 1; // +1 for deleted byte
      153  
      154      // Read page header using BasePage methods (accounts for PAGE_HEADER_SIZE automatically)
      155      int offsetFreeContent = currentPage.readInt(LSMVectorIndex.OFFSET_FREE_CONTENT);
      156      int numberOfEntries = currentPage.readInt(LSMVectorIndex.OFFSET_NUM_ENTRIES);
      157      int pageNum = currentPage.getPageId().getPageNumber();
      158  
      159      // Calculate space needed (no pointer table - just header + sequential entries)
      160      int headerSize = getHeaderSize(pageNum);
      161      final int availableSpace = currentPage.getMaxContentSize() - offsetFreeContent;
      162  
      163      if (availableSpace < entrySize) {
      164        // NO SPACE LEFT, CREATE A NEW PAGE AND FLUSH CURRENT ONE (NO WAL)
      165        // During compaction, pages are created fresh and don't need version tracking
      166        database.getPageManager().writePages(List.of(currentPage), true);
      167  
      168        currentPage = createNewPage(compactedPageNumberOfSeries.getAndIncrement());
      169        newPages.add(currentPage);
      170  
      171        // Reset for new page
      172        offsetFreeContent = currentPage.readInt(0);
      173        numberOfEntries = 0;
      174        pageNum = currentPage.getPageId().getPageNumber();
      175        headerSize = getHeaderSize(pageNum);
      176      }
      177  
      178      // Record the absolute file offset where this entry will be written
      179      final long entryFileOffset = currentFileOffset.get();
      180  
      181      // Write entry sequentially using variable-sized encoding (vectors stored in documents, not index)
      182      int bytesWritten = 0;
      183      bytesWritten += currentPage.writeNumber(offsetFreeContent + bytesWritten, vectorId);
      184      bytesWritten += currentPage.writeNumber(offsetFreeContent + bytesWritten, rid.getBucketId());
      185      bytesWritten += currentPage.writeNumber(offsetFreeContent + bytesWritten, rid.getPosition());
      186      bytesWritten += currentPage.writeByte(offsetFreeContent + bytesWritten, (byte) (deleted ? 1 : 0));
      187  
      188      // Update page header
      189      numberOfEntries++;
      190      offsetFreeContent += bytesWritten;
      191      currentPage.writeInt(LSMVectorIndex.OFFSET_FREE_CONTENT, offsetFreeContent);
      192      currentPage.writeInt(LSMVectorIndex.OFFSET_NUM_ENTRIES, numberOfEntries);
      193  
      194      // Advance file offset tracker
      195      currentFileOffset.addAndGet(bytesWritten);
      196  
      197      // Return the new pages and the absolute file offset where this entry was written
      198      return new CompactionAppendResult(newPages, entryFileOffset);
      199    }

      205    protected MutablePage createNewPage(final int compactedPageNumberOfSeries) {
      206      final int txPageCounter = getTotalPages();
      207      // Create MutablePage directly (compaction happens outside transaction context)
      208      final MutablePage currentPage = new MutablePage(new PageId(database, getFileId(), txPageCounter), pageSize);
      209  
      210      // Calculate header size first
      211      final int headerSize = getHeaderSize(txPageCounter);
      212  
      213      int pos = 0;
      214  
      215      // offsetFreeContent starts right after header (entries grow forward sequentially)
      216      pos += currentPage.writeInt(pos, headerSize);
      217      // numberOfEntries (initially 0)
      218      pos += currentPage.writeInt(pos, 0);
      219      // mutable flag (IMMUTABLE for compacted pages)
      220      pos += currentPage.writeByte(pos, (byte) 0);
      221  
      222      // compacted page number of series
      223      pos += currentPage.writeInt(pos, compactedPageNumberOfSeries);
      224  
      225      // If page 0, write metadata
      226      if (txPageCounter == 0) {
      227        pos += currentPage.writeInt(pos, dimensions);
      228        pos += currentPage.writeInt(pos, similarityFunction.ordinal());
      229        pos += currentPage.writeInt(pos, maxConnections);
      230        currentPage.writeInt(pos, beamWidth);
      231      }
      232  
      233      // Manually update page count (following LSMTreeIndexCompacted pattern)
      234      updatePageCount(txPageCounter + 1);
      235  
      236      return currentPage;

      243    public Map<Integer, VectorEntry> getAllVectors() {
      244      final Map<Integer, VectorEntry> vectors = new HashMap<>();
      245  
      246      try {
      247        final int totalPages = getTotalPages();
      248  
      249        for (int pageNum = 0; pageNum < totalPages; pageNum++) {
      250          final BasePage page = database.getTransaction().getPage(new PageId(database, getFileId(), pageNum), pageSize);
      251  
      252          // Read page header using BasePage methods (accounts for PAGE_HEADER_SIZE automatically)
      253          final int offsetFreeContent = page.readInt(0);
      254          final int numberOfEntries = page.readInt(4);
      255  
      256          if (numberOfEntries == 0)
      257            continue;
      258  
      259          final int headerSize = getHeaderSize(pageNum);
      260  
      261          // Parse variable-sized entries sequentially (no pointer table)
      262          int currentOffset = headerSize;
      263          for (int i = 0; i < numberOfEntries; i++) {
      264            // Read variable-sized vectorId
      265            final long[] vectorIdAndSize = page.readNumberAndSize(currentOffset);
      266            final int id = (int) vectorIdAndSize[0];
      267            currentOffset += (int) vectorIdAndSize[1];
      268  
      269            // Read variable-sized bucketId
      270            final long[] bucketIdAndSize = page.readNumberAndSize(currentOffset);
      271            final int bucketId = (int) bucketIdAndSize[0];
      272            currentOffset += (int) bucketIdAndSize[1];
      273  
      274            // Read variable-sized position
      275            final long[] positionAndSize = page.readNumberAndSize(currentOffset);
      276            final long position = positionAndSize[0];
      277            currentOffset += (int) positionAndSize[1];
      278  
      279            final RID rid = new RID(database, bucketId, position);
      280  
      281            // Read deleted flag (fixed 1 byte)
      282            final boolean deleted = page.readByte(currentOffset) == 1;
      283            currentOffset += 1;
      284  
      285            // Load vector from document (vectors are NOT stored in index pages)
      286            float[] vector = null;
      287            if (!deleted) {
      288              try {
      289                final var record = rid.asDocument(false);
      290                if (record != null) {
      291                  // Get vector from document property
      292                  final String vectorPropertyName = mainIndex.getPropertyNames() != null && !mainIndex.getPropertyNames().isEmpty() ?
      293                      mainIndex.getPropertyNames().get(0) : "vector";
      294                  final Object vectorObj = record.get(vectorPropertyName);
      295                  if (vectorObj instanceof float[]) {
      296                    vector = (float[]) vectorObj;
      297                  }
      298                }
      299              } catch (final Exception e) {
      300                // Skip entries where document is not accessible
      301              }
      302            }
      303  
      304            // Skip entries without valid vectors (unless deleted)
      305            if (vector != null || deleted) {
      306              final VectorEntry entry = new VectorEntry(id, rid, vector);
      307              entry.deleted = deleted;
      308              vectors.put(id, entry);
      309            }
      310          }
      311        }
      312  
      313      } catch (final Exception e) {

      315      }

      324      if (pageNum == 0) {

      326        return 4 + 4 + 1 + 4 + 4 + 4 + 4 + 4;
      327      } else {
      328        // other pages: offsetFree + count + mutable + series
      329        return 4 + 4 + 1 + 4;
