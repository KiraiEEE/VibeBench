      101      void syncButtonPressed() {
      102          if (!ConnectivityStatus.isNetworkAvailable(StandardHomeActivity.this)) {
      103              if (ConnectivityStatus.isAirplaneModeOn(StandardHomeActivity.this)) {
      104                  handleSyncNotAttempted(Localization.get("notification.sync.airplane.action"));
      105                  CommCareApplication.notificationManager().reportNotificationMessage(
      106                          NotificationMessageFactory.message(
      107                                  NotificationMessageFactory.StockMessages.Sync_AirplaneMode,
      108                                  AIRPLANE_MODE_CATEGORY));
      109              } else {
      110                  handleSyncNotAttempted(Localization.get("notification.sync.connections.action"));
      111                  CommCareApplication.notificationManager().reportNotificationMessage(
      112                          NotificationMessageFactory.message(
      113                                  NotificationMessageFactory.StockMessages.Sync_NoConnections,
      114                                  AIRPLANE_MODE_CATEGORY));
      115              }
      116              FirebaseAnalyticsUtil.reportSyncResult(
      117                      false,
      118                      AnalyticsParamValue.SYNC_TRIGGER_USER,
      119                      AnalyticsParamValue.SYNC_MODE_SEND_FORMS,
      120                      AnalyticsParamValue.SYNC_FAIL_NO_CONNECTION);
      121              return;
      122          }
      123          fetchJobProgressOverNetwork();
      124          CommCareApplication.notificationManager().clearNotifications(AIRPLANE_MODE_CATEGORY);
      125          sendFormsOrSync(true);
      126      }

      173      private static void preparePinMenu(Menu menu, boolean enableMenus) {
      174          boolean pinEnabled = enableMenus && DeveloperPreferences.shouldOfferPinForLogin();
      175          menu.findItem(R.id.action_set_pin).setVisible(pinEnabled);
      176          boolean hasPinSet = false;
      177  
      178          try {
      179              hasPinSet = CommCareApplication.instance().getRecordForCurrentUser().hasPinSet();
      180          } catch (SessionUnavailableException e) {
      181              Log.d(TAG, "Session expired and menu is being created before redirect to login screen");
      182          }
      183  
      184          if (hasPinSet) {
      185              menu.findItem(R.id.action_set_pin).setTitle(Localization.get("home.menu.pin.change"));
      186          } else {
      187              menu.findItem(R.id.action_set_pin).setTitle(Localization.get("home.menu.pin.set"));
      188          }

      192      public boolean onOptionsItemSelected(MenuItem item) {
      193          FirebaseAnalyticsUtil.reportOptionsMenuItemClick(this.getClass(),
      194                  menuIdToAnalyticsParam.get(item.getItemId()));
      195          int itemId = item.getItemId();
      196          if (itemId == R.id.action_update) {
      197              launchUpdateActivity(false);
      198              return true;
      199          } else if (itemId == R.id.action_saved_forms) {
      200              goToFormArchive(false);
      201              return true;
      202          } else if (itemId == R.id.action_change_language) {
      203              showLocaleChangeMenu(uiController);
      204              return true;
      205          } else if (itemId == R.id.action_preferences) {
      206              createPreferencesMenu(this);
      207              return true;
      208          } else if (itemId == R.id.action_advanced) {
      209              showAdvancedActionsPreferences();
      210              return true;
      211          } else if (itemId == R.id.action_about) {
      212              showAboutCommCareDialog();
      213              return true;
      214          } else if (itemId == R.id.action_set_pin) {
      215              launchPinAuthentication();
      216              return true;
      217          } else if (itemId == R.id.action_update_commcare) {
      218              startCommCareUpdate();
      219              return true;
      220          }
      221  
      222          return super.onOptionsItemSelected(item);
      223      }

      245      protected void handleDrawerItemClick(@NonNull BaseDrawerController.NavItemType itemType, String recordId) {
      246          switch (itemType) {
      247              case COMMCARE_APPS -> {
      248                  if(recordId != null) {
      249                      String currentSeatedId = CommCareApplication.instance().getCurrentApp().getUniqueId();
      250                      if(!recordId.equals(currentSeatedId)) {
      251                          //Navigate to LoginActivity for selected app
      252                          CommCareApplication.instance().closeUserSession();
      253                          Intent i = new Intent();
      254                          i.putExtra(EXTRA_APP_ID, recordId);
      255                          setResult(RESULT_OK, i);
      256                          finish();
      257                      }
      258                  }
      259              }
      260              case OPPORTUNITIES -> {
      261                  if(personalIdManagedLogin) {
      262                      ConnectNavHelper.INSTANCE.goToConnectJobsList(this);
      263                      closeDrawer();
      264                  } else {
      265                      navigateToConnectMenu();
      266                  }
      267              }
      268              case MESSAGING -> {
      269                  if(personalIdManagedLogin) {
      270                      ConnectNavHelper.INSTANCE.goToMessaging(this);
      271                      closeDrawer();
      272                  } else {
      273                      navigateToMessaging();
      274                  }
      275              }
      276              case WORK_HISTORY -> {
      277                  if(personalIdManagedLogin) {
      278                      ConnectNavHelper.INSTANCE.goToWorkHistory(this);
      279                      closeDrawer();
      280                  } else {
      281                      navigateToWorkHistory();
      282                  }
