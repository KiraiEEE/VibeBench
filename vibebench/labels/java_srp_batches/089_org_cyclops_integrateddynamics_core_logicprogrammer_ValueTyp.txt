      355      public static boolean slotClickCommon(int slotId, Slot slot, int mouseButton, ClickType clickType, Player player,
      356                                            List<ItemMatchProperties> inputStacks, int propertySlotCount,
      357                                            Consumer<Integer> setPropertySubGui,
      358                                            Consumer<Integer> refreshPropertiesGui) {
      359          if (slotId >= SLOT_OFFSET && slotId < propertySlotCount + SLOT_OFFSET) {
      360              if (clickType == ClickType.QUICK_MOVE && mouseButton == 0) {
      361                  if (player.level().isClientSide()) {
      362                      int id = slotId - SLOT_OFFSET;
      363                      setPropertySubGui.accept(id);
      364                  }
      365                  return true;
      366              } else {
      367                  // Similar logic as ContainerExtended.adjustPhantomSlot
      368                  ItemMatchProperties props = inputStacks.get(slotId - SLOT_OFFSET);
      369                  int quantityCurrent = props.getTagQuantity();
      370                  int quantityNew;
      371                  if (clickType == ClickType.QUICK_MOVE) {
      372                      quantityNew = mouseButton == 0 ? (quantityCurrent + 1) / 2 : quantityCurrent * 2;
      373                  } else {
      374                      quantityNew = mouseButton == 0 ? quantityCurrent - 1 : quantityCurrent + 1;
      375                  }
      376  
      377                  if (quantityNew > slot.getMaxStackSize()) {
      378                      quantityNew = slot.getMaxStackSize();
      379                  }
      380  
      381                  props.setTagQuantity(quantityNew);
      382                  if (!props.getItemStack().isEmpty()) {
      383                      props.getItemStack().setCount(quantityNew);
      384                  }
      385  
      386                  if (quantityNew <= 0) {
      387                      props.setItemTag(null);
      388                      props.setTagQuantity(1);
      389                      if (IModHelpers.get().getMinecraftHelpers().isClientSideThread()) {
      390                          refreshPropertiesGui.accept(slotId - SLOT_OFFSET);
      391                      }
      392                  }
      393              }
      394          }
      395  
      396          return false;
      397      }

      420      public static Optional<ItemStack> getRotatingItemFromTag(ItemMatchProperties props) {
      421          String tagName = props.getItemTag();
      422          if (tagName != null) {
      423              try {
      424                  Optional<HolderSet.Named<Item>> tag = BuiltInRegistries.ITEM.get(TagKey.create(Registries.ITEM, Identifier.parse(tagName)));
      425                  if (!tag.isEmpty()) {
      426                      List<Item> items = tag.stream()
      427                              .flatMap(holders -> holders.stream())
      428                              .map(holder -> holder.value()).toList();
      429                      int tick = ((int) Minecraft.getInstance().level.getGameTime()) / TICK_DELAY;
      430                      Item item = items.get(tick % items.size());
      431                      return Optional.of(new ItemStack(item, props.getTagQuantity()));
      432                  }
      433              } catch (IdentifierException e) {
      434                  // Ignore invalid tags
      435              }
      436          }
      437          return Optional.empty();
