      155      public void onSchedule() throws Exception {
      156          if (!inSchedule.compareAndSet(false, true)) {
      157              return;
      158          }
      159          try {
      160              switch (state.get()) {
      161                  case INIT:
      162                      Preconditions.checkState(false, "has not started");
      163                  case STARTED:
      164                      prepare();
      165                      break;
      166                  case PREPARING:
      167                  case PAUSED:
      168                  case FAILED:
      169                      Preconditions.checkState(false, "should not be scheduled");
      170                      break;
      171                  case RUN_EPOCH:
      172                      epoch.onSchedule();
      173                      epochCoordinator.runEpoch(epoch);
      174                      break;
      175                  default:
      176              }
      177          } finally {
      178              inSchedule.compareAndSet(true, false);
      179          }

      251              ConnectContext context = queryCoordinator.getConnectContext();

      260              dataSink.complete();

      264          List<ExecutionFragment> execFragments = queryCoordinator.getFragmentsInPreorder();
      265          TDescriptorTable descTable = queryCoordinator.getDescriptorTable();

      268          // Group all fragment instances by BE id, and package them into a task
      269          BiMap<Long, TNetworkAddress> taskId2Addr = HashBiMap.create();

      272          long taskIdGen = 0;
