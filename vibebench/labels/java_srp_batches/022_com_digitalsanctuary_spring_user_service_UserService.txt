      292  	private void savePasswordHistory(User user, String encodedPassword) {
      293  		if (user == null || !StringUtils.hasText(encodedPassword)) {
      294  			log.warn("Cannot save password history: user or password is null/empty.");

      299  		passwordHistoryRepository.save(entry);

      306  	/**
      307  	 * Cleans up old password history entries for a user, keeping only the most recent entries.
      308  	 * Uses SERIALIZABLE isolation to prevent race conditions when the same user changes
      309  	 * their password concurrently from multiple sessions.
      310  	 *
      311  	 * @param user the user whose password history should be cleaned up
      312  	 */
      313  	@Transactional(isolation = Isolation.SERIALIZABLE)
      314  	private void cleanUpPasswordHistory(User user) {
      315  		if (user == null || historyCount <= 0) {
      316  			return;
      317  		}
      318  
      319  		List<PasswordHistoryEntry> entries = passwordHistoryRepository.findByUserOrderByEntryDateDesc(user);
      320  		// Keep historyCount + 1 entries: the current password plus historyCount previous passwords
      321  		// This ensures we actually prevent reuse of the last historyCount passwords
      322  		int maxEntries = historyCount + 1;
      323  		if (entries.size() > maxEntries) {
      324  			List<PasswordHistoryEntry> toDelete = entries.subList(maxEntries, entries.size());
      325  			passwordHistoryRepository.deleteAll(toDelete);
      326  			log.debug("Cleaned up {} old password history entries for user: {}", toDelete.size(), user.getEmail());
      327  		}
      328  	}
