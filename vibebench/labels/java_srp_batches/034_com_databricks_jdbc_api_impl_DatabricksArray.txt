      119          case DatabricksTypeUtil.FLOAT:

      121          case DatabricksTypeUtil.DOUBLE:

      123          case DatabricksTypeUtil.DECIMAL:

      125          case DatabricksTypeUtil.BOOLEAN:

      127          case DatabricksTypeUtil.DATE:

      129          case DatabricksTypeUtil.TIMESTAMP:

      131          case DatabricksTypeUtil.TIME:

      133          case DatabricksTypeUtil.BINARY:

      135          case DatabricksTypeUtil.STRING:
      136          default:
      137            return value.toString();
      138        }
