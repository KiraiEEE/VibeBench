      171      private void performMigration(PartyBlockingFile partyFile, ClaimedChunkBlockingFile chunkFile, PlayerNameTrackerBlockingFile nameFile, AdminOverridesBlockingFile adminFile) {
      172          logger.at(Level.INFO).log("Starting migration to SQLite...");
      173  
      174          // Backup the entire folder
      175          try {
      176              Path source = Paths.get(FileUtils.MAIN_PATH);
      177              Path backup = Paths.get(FileUtils.MAIN_PATH + "_backup_" + System.currentTimeMillis());
      178              copyFolder(source, backup);
      179              logger.at(Level.INFO).log("Backup created at: " + backup.toAbsolutePath());
      180          } catch (IOException e) {
      181              logger.at(Level.SEVERE).log("Failed to create backup before migration: " + e.getMessage());
      182              return;
      183          }
      184  
      185          try {
      186              connection.setAutoCommit(false);
      187  
      188              // Migrate Parties
      189              for (PartyInfo party : partyFile.getParties().values()) {
      190                  saveParty(party);
      191              }
      192  
      193              // Migrate Claims
      194              for (Map.Entry<String, HashMap<String, ChunkInfo>> dimEntry : chunkFile.getChunks().entrySet()) {
      195                  String dimension = dimEntry.getKey();
      196                  for (ChunkInfo chunk : dimEntry.getValue().values()) {
      197                      saveClaim(dimension, chunk);
      198                  }
      199              }
      200  
      201              // Migrate Name Cache
      202              for (PlayerNameTracker.PlayerName name : nameFile.getTracker().getNames()) {
      203                  saveNameCache(name.getUuid(), name.getName(), name.getLastSeen(), name.getPlayTime());
      204              }
      205  
      206              // Migrate Admin Overrides
      207              for (UUID uuid : adminFile.getAdminOverrides()) {
      208                  saveAdminOverride(uuid);
      209              }
      210  
      211              connection.commit();
      212              connection.setAutoCommit(true);
      213  
      214              // Create migration marker
      215              new File(FileUtils.MAIN_PATH + File.separator + ".migrated").createNewFile();
      216              logger.at(Level.INFO).log("Migration to SQLite completed successfully.");
      217          } catch (Exception e) {
      218              try {
      219                  connection.rollback();
      220              } catch (SQLException ex) {
      221                  logger.at(Level.SEVERE).withCause(ex).log("Migration rollback failed");
      222              }
      223              logger.at(Level.SEVERE).withCause(e).log("Migration failed: " + e.getMessage());
      224          }
