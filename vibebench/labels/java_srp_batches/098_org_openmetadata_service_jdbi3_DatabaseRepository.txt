      327    private Map<UUID, List<EntityReference>> batchFetchDatabaseSchemas(List<Database> databases) {
      328      var schemasMap = new HashMap<UUID, List<EntityReference>>();
      329      if (databases == null || databases.isEmpty()) {
      330        return schemasMap;
      331      }
      332  
      333      // Initialize empty lists for all databases
      334      databases.forEach(database -> schemasMap.put(database.getId(), new ArrayList<>()));
      335  
      336      // Single batch query to get all schemas for all databases
      337      // Use Include.ALL to get all relationships including those for soft-deleted entities
      338      var records =
      339          daoCollection
      340              .relationshipDAO()
      341              .findToBatch(
      342                  entityListToStrings(databases),
      343                  Relationship.CONTAINS.ordinal(),
      344                  DATABASE_SCHEMA,
      345                  Include.ALL);
      346  
      347      // Collect all unique schema IDs first
      348      var schemaIds = records.stream().map(rec -> UUID.fromString(rec.getToId())).distinct().toList();
      349  
      350      // Batch fetch all schema entity references
      351      var schemaRefs = Entity.getEntityReferencesByIds(DATABASE_SCHEMA, schemaIds, Include.ALL);
      352      var schemaRefMap =
      353          schemaRefs.stream().collect(Collectors.toMap(EntityReference::getId, ref -> ref));
      354  
      355      // Group schemas by database ID
      356      records.forEach(
      357          record -> {
      358            var databaseId = UUID.fromString(record.getFromId());
      359            var schemaId = UUID.fromString(record.getToId());
      360            var schemaRef = schemaRefMap.get(schemaId);
      361            if (schemaRef != null) {
      362              schemasMap.get(databaseId).add(schemaRef);
      363            }
      364          });
      365  
      366      return schemasMap;
      367    }

      416      var serviceMap = new HashMap<UUID, EntityReference>();
      417      if (databases == null || databases.isEmpty()) {
      418        return serviceMap;
      419      }

      422      // Use Include.ALL to get all relationships including those for soft-deleted entities

      427                  entityListToStrings(databases), Relationship.CONTAINS.ordinal(), Include.ALL);
      428  
      429      // Collect all unique service IDs first

      431          records.stream()

      443      // Map databases to their services

      445          record -> {

      447              var databaseId = UUID.fromString(record.getToId());
