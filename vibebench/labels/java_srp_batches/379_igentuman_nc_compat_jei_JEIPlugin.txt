       85      private static HashMap<String, RecipeType<? extends NcRecipe>> getRecipeTypes() {
       86          if (recipeTypes == null) {
       87              recipeTypes = new HashMap<>();
       88              for (String name : RECIPE_CLASSES.keySet()) {
       89                  if (Processors.all().containsKey(name) && !Processors.all().get(name).isRegistered()) {
       90                      continue;
       91                  }
       92                  recipeTypes.put(name, new RecipeType<>(rl(name), RECIPE_CLASSES.get(name)));
       93              }
       94          }
       95          return recipeTypes;
       96      }

      269      private List<HeatSinkPlacementRecipe> heatSinkPlacementRecipes() {
      270          List<HeatSinkPlacementRecipe> recipes = new ArrayList<>();
      271          
      272          for (Map.Entry<String, HeatSinkDef> entry : FissionReactorRegistration.heatsinks.entrySet()) {
      273              String heatSinkName = entry.getKey();
      274              HeatSinkDef heatSinkDef = entry.getValue();
      275              
      276              // Skip empty and active heat sinks as they don't have placement rules
      277              if (heatSinkName.equals("empty") || heatSinkName.equals("active")) {
      278                  continue;
      279              }
      280              
      281              // Get the heat sink item
      282              String blockKey = heatSinkName + "_heat_sink";
      283              if (FissionReactorRegistration.FISSION_BLOCKS.containsKey(blockKey)) {
      284                  ItemStack heatSinkItem = new ItemStack(FissionReactorRegistration.FISSION_BLOCKS.get(blockKey).get());
      285                  recipes.add(new HeatSinkPlacementRecipe(rl(blockKey), heatSinkDef, heatSinkItem));
      286              }
      287          }
      288          
      289          return recipes;
      290      }

      315      private List<MultiblockStructureRecipe> loadMultiblockStructures() {
      316          List<MultiblockStructureRecipe> recipes = new ArrayList<>();
      317          ResourceManager resourceManager = Minecraft.getInstance().getResourceManager();
      318          List<String> structures = List.of(
      319              "fission_reactor.nbt",
      320              "fusion_reactor.nbt",
      321              "kugelblitz_chamber.nbt",
      322              "linear_accelerator.nbt",
      323              "target_chamber.nbt",
      324              "leacher.nbt",
      325              "turbine.nbt"
      326          );
      327  
      328          for (String file : structures) {
      329              Optional<Resource> structure = resourceManager.getResource(rl("structures/" + file));
      330              CompoundTag nbt = null;
      331              try {
      332                  nbt = NbtIo.readCompressed(structure.get().open(), net.minecraft.nbt.NbtAccounter.unlimitedHeap());
      333              } catch (IOException e) {
      334                  continue;
      335              }
      336              recipes.add(new MultiblockStructureRecipe(rl(file), nbt, file));
      337          }
      338          return recipes;
      339      }
