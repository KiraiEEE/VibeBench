       54      private static ImmutableMap<String, ImmutableList<ResourceTypeEntry>> discoverResourceTypes() {
       55          var loader = MoreObjects.firstNonNull(
       56                  ResourcePackages.class.getClassLoader(),
       57                  Thread.currentThread().getContextClassLoader()
       58          );
       59          final ClassPath classpath;
       60          try {
       61              classpath = ClassPath.from(loader);
       62          } catch (IOException e) {
       63              throw new IllegalStateException(String.format("Failed to read class path: %s", e.getMessage()), e);
       64          }
       65  
       66          return classpath.getAllClasses().stream()
       67                  // exclude early our dependencies and common packages almost certain to not contain what we want
       68                  .filter(ResourcePackages::excludePackages)
       69                  .map(c -> {
       70                      try {
       71                          return c.load();
       72                      } catch (LinkageError e) {
       73                          throw new IllegalStateException(String.format(
       74                                  "Failed to load class '%s' (package: '%s') from class path: %s",
       75                                  c, c.getPackageName(), e.getMessage()
       76                          ), e);
       77                      }
       78                  })
       79                  .filter(c -> c.isAnnotationPresent(ResourceType.class))
       80                  .filter(c -> {
       81                      /* must be a subclass of Resource */
       82                      if (Resource.class.isAssignableFrom(c)) {
       83                          /* must be static or standalone class (top level class) */
       84                          if (Reflection.isNestedClass(c)) {
       85                              throw new IllegalStateException(String.format(
       86                                      "Expected class '%s' annotated with '%s' to be static or standalone, but was nested. " +
       87                                              "Make the class static or standalone.",
       88                                      c.getTypeName(), ResourceType.class.getTypeName()
       89                              ));
       90                          }
       91                          return true;
       92                      } else {
       93                          throw new IllegalStateException(String.format(
       94                                  "Expected class '%s' to be annotated with '%s', to be assignable from '%s', but it was not.",
       95                                  c.getTypeName(), ResourceType.class.getTypeName(), Resource.class.getTypeName()
       96                          ));
       97                      }
       98                  })
       99                  .filter(c -> !Reflection.isNestedClass(c))
      100                  .map(c -> {
      101                      //noinspection unchecked
      102                      return (Class<Resource>) c; // checked in filters above
      103                  })
      104                  .map(c -> new ResourceTypeAndClass(c.getAnnotation(ResourceType.class), c))
      105                  .collect(
      106                          collectingAndThen(
      107                                  groupingBy(
      108                                          g -> g.annotation.type(),
      109                                          Collectors.mapping(
      110                                                  g -> new ResourceTypeEntry(Optionals.ofBlank(g.annotation.version()), g.type),
      111                                                  ImmutableList.<ResourceTypeEntry>toImmutableList()
      112                                          )
      113                                  ),
      114                                  ImmutableMap::copyOf
      115                          )
      116                  );
      117      }

      166      private static boolean excludePackagesFromProperties(ClassInfo c) {
      167          String packages = System.getProperty("pulumi.resourcepackages.excludes");
      168          if (packages != null) {
      169              String[] items = packages.split(",");
      170              for (String item: items){
      171                  if (c.getPackageName().startsWith(item)) {
      172                      return true;
      173                  }
      174              }
      175          }
      176          return false;

      183      @InternalUse
      184      Optional<Resource> tryConstruct(String type, String version, String urn) {
      185          this.log.excessive(
      186                  "Deserialize/ResourcePackages: searching for type=%s version=%s urn=%s",
      187                  type, version, urn
      188          );
      189          var resourceType = tryGetResourceType(type, version);
      190          if (resourceType.isEmpty()) {
      191              var message = String.format(
      192                      "Deserialize/ResourcePackages: can't find a resource: '%s'; version=%s urn=%s",
      193                      type, version, urn
      194              );
      195              this.log.debugOrExcessive(message, String.format(
      196                      "; Available resources type names:\n %s",
      197                      String.join("\n", ResourcePackages.resourceTypes.get().keySet())
      198              ));
      199              return Optional.empty();
      200          }
      201          log.excessive(String.format(
      202                  "Deserialize/ResourcePackages: found resource: '%s'; type=%s version=%s urn=%s",
      203                  resourceType, type, version, urn
      204          ));
      205  
      206          if (Reflection.isNestedClass(resourceType.get())) {
      207              throw new IllegalArgumentException(String.format(
      208                      "tryConstruct(String, String, String) cannot be used with nested classes, " +
      209                              "make class '%s' static or standalone",
      210                      resourceType.get().getTypeName()
      211              ));
      212          }
      213  
      214          var urnParsed = Urn.parse(urn);
      215          var urnName = urnParsed.name;
      216  
      217          // Looking for a constructor with the following signature:
      218          //
      219          //     (String name, SomeResourceArgs args, CustomResourceOptions options)
      220          //
      221          // The search is approximate. We may need to consider using annotations instead in future versions.
      222          var constructorInfo =
      223                  Arrays.stream(resourceType.get().getDeclaredConstructors())
      224                          .filter(c -> c.getParameterCount() == 3)
      225                          // Remove confusion of constructors with the second param of type:
      226                          //     Output<String> id
      227                          .filter(c -> !c.getParameterTypes()[1].equals(Output.class))
      228                          .collect(toSingleton(cause ->
      229                                  new IllegalArgumentException(String.format(
      230                                          "Resource provider error. Could not find a constructor for resource %s" +
      231                                                  " with the following signature:" +
      232                                                  " `(String name, SomeResourceArgs args, CustomResourceOptions options)`" +
      233                                                  ", got: `%s`",
      234                                          resourceType.get(), cause
      235                                  ))
      236                          ));
      237  
      238          constructorInfo.setAccessible(true);
      239  
      240          this.log.excessive(
      241                  "Deserialize/ResourcePackages: constructing using constructor=%s type=%s version=%s urn=%s",
      242                  constructorInfo, resourceType.get().getTypeName(), version, urn
      243          );
      244          var resourceOptions = resolveResourceOptions(resourceType.get(), urn);
      245          try {
      246              var resource = (Resource) constructorInfo.newInstance(new Object[]{urnName, null, resourceOptions});
      247              return Optional.of(resource);
      248          } catch (InstantiationException | IllegalAccessException | InvocationTargetException e) {
      249              throw new IllegalArgumentException(String.format(
      250                      "Couldn't instantiate class '%s' using constructor: '%s', for resource type: '%s'",
      251                      resourceType.get().getTypeName(), constructorInfo, type
      252              ), e);
      253          } finally {
      254              constructorInfo.setAccessible(false);
      255          }
      256      }

      258      private ResourceOptions resolveResourceOptions(Class<?> resourceType, String urn) {
      259          if (CustomResource.class.isAssignableFrom(resourceType)) {
      260              return CustomResourceOptions.builder().urn(urn).build();
      261          }
      262          if (ComponentResource.class.isAssignableFrom(resourceType)) {
      263              return ComponentResourceOptions.builder().urn(urn).build();
      264          }
      265          throw new IllegalStateException(String.format("Unexpected resource type: '%s'", resourceType.getTypeName()));
      266      }

      268      @InternalUse
      269      Optional<Class<Resource>> tryGetResourceType(String name, @Nullable String version) {
      270          requireNonNull(name);
      271  
      272          var minimalVersion = !isNullOrEmpty(version)
      273                  ? SemanticVersion.parse(version) : SemanticVersion.of(0);
      274  
      275          var types = Maps.tryGetValue(ResourcePackages.resourceTypes.get(), name);
      276          if (types.isEmpty()) {
      277              return Optional.empty();
      278          }
      279  
      280          return types.get().stream()
      281                  .map(resource -> new SemanticVersionAndClass(
      282                          resource.version.map(SemanticVersion::parse).orElse(minimalVersion), resource.type
      283                  ))
      284                  /* must be at least the minimal version */
      285                  .filter(resource -> resource.semanticVersion.isGreaterOrEqualTo(minimalVersion))
      286                  /* must have same major version */
      287                  .filter(resource ->
      288                          isNullOrEmpty(version) || resource.semanticVersion.getMajor() == minimalVersion.getMajor()
      289                  )
      290                  /* latest, filtered version */
      291                  .max(comparing(resource -> resource.semanticVersion))
      292                  .map(resource -> resource.type);
      293      }
