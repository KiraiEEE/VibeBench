       72      public void execute(Collection<SinkRecord> records) {
       73          final Map<CollectionId, Buffer> upsertBufferByTable = new LinkedHashMap<>();
       74          final Map<CollectionId, Buffer> deleteBufferByTable = new LinkedHashMap<>();
       75  
       76          for (SinkRecord kafkaSinkRecord : records) {
       77  
       78              JdbcSinkRecord record = new JdbcKafkaSinkRecord(kafkaSinkRecord, config.getPrimaryKeyMode(), config.getPrimaryKeyFields(), config.getFieldFilter(), dialect);
       79              LOGGER.trace("Processing {}", record);
       80  
       81              validate(record);
       82  
       83              Optional<CollectionId> optionalCollectionId = getCollectionIdFromRecord(record);
       84              if (optionalCollectionId.isEmpty()) {
       85  
       86                  LOGGER.warn("Ignored to write record from topic '{}' partition '{}' offset '{}'. No resolvable table name", record.topicName(), record.partition(),
       87                          record.offset());
       88                  continue;
       89              }
       90  
       91              final CollectionId collectionId = optionalCollectionId.get();
       92  
       93              if (record.isTruncate()) {
       94                  if (!config.isTruncateEnabled()) {
       95                      LOGGER.debug("Truncates are not enabled, skipping truncate for topic '{}'", record.topicName());
       96                      continue;
       97                  }
       98  
       99                  // Here we want to flush the buffer to let truncate having effect on the buffered events.
      100                  flushBuffers(upsertBufferByTable);
      101                  flushBuffers(deleteBufferByTable);
      102  
      103                  try {
      104                      final TableDescriptor table = checkAndApplyTableChangesIfNeeded(collectionId, record);
      105                      writeTruncate(dialect.getTruncateStatement(table));
      106                      continue;
      107                  }
      108                  catch (SQLException e) {
      109                      throw new ConnectException("Failed to process a sink record", e);
      110                  }
      111              }
      112  
      113              if (record.isDelete() || record.isTombstone()) {
      114                  if (!config.isDeleteEnabled()) {
      115                      LOGGER.debug("Deletes are not enabled, skipping delete for topic '{}'", record.topicName());
      116                      continue;
      117                  }
      118  
      119                  if (upsertBufferByTable.get(collectionId) != null && !upsertBufferByTable.get(collectionId).isEmpty()) {
      120                      // When a delete event arrives, update buffer must be flushed to avoid losing the delete
      121                      // for the same record after its update.
      122  
      123                      flushBufferWithRetries(collectionId, upsertBufferByTable.get(collectionId).flush());
      124                  }
      125  
      126                  Buffer tableIdBuffer = resolveBuffer(deleteBufferByTable, collectionId, record);
      127                  List<JdbcSinkRecord> toFlush = tableIdBuffer.add(record);
      128                  flushBufferWithRetries(collectionId, toFlush);
      129              }
      130              else {
      131                  if (deleteBufferByTable.get(collectionId) != null && !deleteBufferByTable.get(collectionId).isEmpty()) {
      132                      // When an insert arrives, delete buffer must be flushed to avoid losing an insert for the same record after its deletion.
      133                      // this because at the end we will always flush inserts before deletes.
      134                      flushBufferWithRetries(collectionId, deleteBufferByTable.get(collectionId).flush());
      135                  }
      136  
      137                  Stopwatch updateBufferStopwatch = Stopwatch.reusable();
      138                  updateBufferStopwatch.start();
      139  
      140                  Buffer tableIdBuffer = resolveBuffer(upsertBufferByTable, collectionId, record);
      141  
      142                  List<JdbcSinkRecord> toFlush = tableIdBuffer.add(record);
      143                  updateBufferStopwatch.stop();

      145                  LOGGER.trace("[PERF] Update buffer execution time {}", updateBufferStopwatch.durations());
      146                  flushBufferWithRetries(collectionId, toFlush);
      147              }
      148  
      149          }
      150  
      151          flushBuffers(upsertBufferByTable);
      152          flushBuffers(deleteBufferByTable);
      153      }

      175      private void flushBufferWithRetries(CollectionId collectionId, List<JdbcSinkRecord> toFlush) {
      176          int retries = 0;
      177          Exception lastException = null;
      178  
      179          LOGGER.debug("Flushing records in JDBC Writer for table: {}", collectionId.name());
      180          while (retries <= flushMaxRetries) {
      181              try {
      182                  if (retries > 0) {
      183                      LOGGER.warn("Retry to flush records for table '{}'. Retry {}/{} with delay {} ms",
      184                              collectionId.name(), retries, flushMaxRetries, flushRetryDelay.toMillis());
      185                      try {
      186                          Metronome.parker(flushRetryDelay, Clock.SYSTEM).pause();
      187                      }
      188                      catch (InterruptedException e) {
      189                          throw new ConnectException("Interrupted while waiting to retry flush records", e);
      190                      }
      191                  }
      192                  flushBuffer(collectionId, toFlush);
      193                  return;
      194              }
      195              catch (Exception e) {
      196                  lastException = e;
      197                  if (isRetriable(e)) {
      198                      retries++;
      199                  }
      200                  else {
      201                      throw new ConnectException("Failed to process a sink record", e);
      202                  }
      203              }
      204          }
      205          throw new ConnectException("Exceeded max retries " + flushMaxRetries + " times, failed to process sink records", lastException);
      206      }

      208      private void flushBuffer(CollectionId collectionId, List<JdbcSinkRecord> toFlush) throws SQLException {
      209          Stopwatch flushBufferStopwatch = Stopwatch.reusable();
      210          Stopwatch tableChangesStopwatch = Stopwatch.reusable();
      211          if (!toFlush.isEmpty()) {
      212              LOGGER.debug("Flushing records in JDBC Writer for table: {}", collectionId.name());
      213              tableChangesStopwatch.start();
      214              final TableDescriptor table = checkAndApplyTableChangesIfNeeded(collectionId, toFlush.get(0));
      215              tableChangesStopwatch.stop();
      216              String sqlStatement = getSqlStatement(table, toFlush.get(0));
      217              flushBufferStopwatch.start();
      218              recordWriter.write(toFlush, sqlStatement);
      219              flushBufferStopwatch.stop();
      220  
      221              LOGGER.trace("[PERF] Flush buffer execution time {}", flushBufferStopwatch.durations());
      222              LOGGER.trace("[PERF] Table changes execution time {}", tableChangesStopwatch.durations());
      223          }

      237      private TableDescriptor checkAndApplyTableChangesIfNeeded(CollectionId collectionId, JdbcSinkRecord record) throws SQLException {
      238          if (!hasTable(collectionId)) {
      239              // Table does not exist, lets attempt to create it.
      240              try {
      241                  return createTable(collectionId, record);
      242              }
      243              catch (SQLException ce) {
      244                  // It's possible the table may have been created in the interim, so try to alter.
      245                  LOGGER.warn("Table creation failed for '{}', attempting to alter the table", collectionId.toFullIdentiferString(), ce);
      246                  try {
      247                      return alterTableIfNeeded(collectionId, record);

      253                  }
      254              }
      255          }

      257              // Table exists, lets attempt to alter it if necessary.
      258              try {

      262                  LOGGER.error("Failed to alter the table '{}'.", collectionId.toFullIdentiferString(), ae);

      315              return table;

      322                  throw new SQLException(String.format(

      348      private String getSqlStatement(TableDescriptor table, JdbcSinkRecord record) {
      349          if (!record.isDelete()) {
      350              switch (config.getInsertMode()) {
      351                  case INSERT:
      352                      return dialect.getInsertStatement(table, record);
      353                  case UPSERT:
      354                      if (record.keyFieldNames().isEmpty()) {
      355                          throw new ConnectException("Cannot write to table " + table.getId().name() + " with no key fields defined.");
      356                      }
      357                      return dialect.getUpsertStatement(table, record);
      358                  case UPDATE:
      359                      return dialect.getUpdateStatement(table, record);
      360              }
      361          }
      362          else {
      363              return dialect.getDeleteStatement(table, record);
      364          }
      365  
      366          throw new DataException(String.format("Unable to get SQL statement for %s", record));

      388      private boolean isRetriable(Throwable throwable) {
      389          if (throwable == null) {
      390              return false;
      391          }
      392          for (Class<? extends Exception> e : dialect.getCommunicationExceptions()) {
      393              if (e.isAssignableFrom(throwable.getClass())) {
      394                  return true;
      395              }
      396          }
      397          return isRetriable(throwable.getCause());
