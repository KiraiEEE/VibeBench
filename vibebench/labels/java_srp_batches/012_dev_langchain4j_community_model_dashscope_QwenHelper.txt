      180      static Map<String, Object> toMultiModalContent(Content content) {
      181          switch (content.type()) {
      182              case IMAGE:
      183                  Image image = ((ImageContent) content).image();
      184                  String imageContent;
      185                  if (image.url() != null) {
      186                      imageContent = image.url().toString();
      187                      return Collections.singletonMap("image", imageContent);
      188                  } else if (Utils.isNotNullOrBlank(image.base64Data())) {
      189                      return Collections.singletonMap(
      190                              "image", "data:%s;base64,%s".formatted(image.mimeType(), image.base64Data()));
      191                  } else {
      192                      return Collections.emptyMap();
      193                  }
      194              case AUDIO:
      195                  Audio audio = ((AudioContent) content).audio();
      196                  String audioContent;
      197                  if (audio.url() != null) {
      198                      audioContent = audio.url().toString();
      199                      return Collections.singletonMap("audio", audioContent);
      200                  } else if (Utils.isNotNullOrBlank(audio.base64Data())) {
      201                      return Collections.singletonMap(
      202                              "audio", "data:%s;base64,%s".formatted(audio.mimeType(), audio.base64Data()));
      203                  } else {
      204                      return Collections.emptyMap();
      205                  }
      206              case VIDEO:
      207                  Video video = ((VideoContent) content).video();
      208                  String videoContent;
      209                  if (video.url() != null) {
      210                      videoContent = video.url().toString();
      211                      return Collections.singletonMap("video", videoContent);
      212                  } else if (Utils.isNotNullOrBlank(video.base64Data())) {
      213                      return Collections.singletonMap(
      214                              "video", "data:%s;base64,%s".formatted(video.mimeType(), video.base64Data()));
      215                  } else {
      216                      return Collections.emptyMap();
      217                  }
      218              case TEXT:
      219                  return Collections.singletonMap("text", ((TextContent) content).text());
      220              default:
      221                  return Collections.emptyMap();
      222          }
      223      }
