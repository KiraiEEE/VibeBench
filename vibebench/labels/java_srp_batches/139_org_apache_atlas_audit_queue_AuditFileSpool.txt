      308  		if (!initDone) {

      310  					+ queueProvider.getName());

      314  				+ ", consumer=" + consumerProvider.getName());

      318  
      319  		PrintWriter out = getOpenLogFileStream();

      322  			// complete

      324  				if (isWriting) {

      334  							+ queueProvider.getName() + ", consumer="
      335  							+ consumerProvider.getName());

      338  					out.close();

      341  					logger.debug("Error closing spool out file.", t);
      342  				}
      343  			}

      346  			if (destinationThread != null) {
      347  				destinationThread.interrupt();
