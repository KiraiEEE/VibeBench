      264    private void acquireLock(Resource resource, boolean isReadLock, ReadWriteLock lock,
      265                             long startWaitingTimeNanos) {
      266      if (isReadLock) {
      267        lock.readLock().lock();
      268        updateReadLockMetrics(resource, (ReentrantReadWriteLock) lock, startWaitingTimeNanos);
      269      } else {
      270        lock.writeLock().lock();
      271        updateWriteLockMetrics(resource, (ReentrantReadWriteLock) lock, startWaitingTimeNanos);
      272      }

      275    private OMLockDetails acquireLocks(Resource resource, boolean isReadLock,
      276        Function<Striped<ReadWriteLock>, Iterable<ReadWriteLock>> lockListProvider) {
      277      Pair<Map<Resource, Striped<ReadWriteLock>>, ResourceLockManager> resourceLockPair =
      278          resourcelockMap.get(resource.getClass());
      279      ResourceLockManager<Resource> resourceLockManager = resourceLockPair.getRight();
      280      resourceLockManager.clearLockDetails();
      281      if (!resourceLockManager.canLockResource(resource)) {
      282        String errorMessage = getErrorMessage(resource);
      283        LOG.error(errorMessage);
      284        throw new RuntimeException(errorMessage);
      285      }
      286  
      287      long startWaitingTimeNanos = Time.monotonicNowNanos();
      288  
      289      for (ReadWriteLock lock : lockListProvider.apply(resourceLockPair.getKey().get(resource))) {
      290        acquireLock(resource, isReadLock, lock, startWaitingTimeNanos);
      291      }
      292      return resourceLockManager.lockResource(resource);
      293    }

      453    private OMLockDetails releaseLock(Resource resource, boolean isReadLock,
      454        String... keys) {
      455      Pair<Map<Resource, Striped<ReadWriteLock>>, ResourceLockManager> resourceLockPair =
      456          resourcelockMap.get(resource.getClass());
      457      ResourceLockManager<Resource> resourceLockManager = resourceLockPair.getRight();
      458      resourceLockManager.clearLockDetails();
      459      ReentrantReadWriteLock lock = getLock(resourceLockPair.getKey(), resource, keys);
      460      if (isReadLock) {
      461        lock.readLock().unlock();
      462        updateReadUnlockMetrics(resource, lock);
      463      } else {
      464        boolean isWriteLocked = lock.isWriteLockedByCurrentThread();
      465        lock.writeLock().unlock();
      466        updateWriteUnlockMetrics(resource, lock, isWriteLocked);
      467      }
      468      return resourceLockManager.unlockResource(resource);
      469    }

      471    private OMLockDetails releaseLocks(Resource resource, boolean isReadLock,
      472        Function<Striped<ReadWriteLock>, Iterable<ReadWriteLock>> lockListProvider) {
      473      Pair<Map<Resource, Striped<ReadWriteLock>>, ResourceLockManager> resourceLockPair =
      474          resourcelockMap.get(resource.getClass());
      475      ResourceLockManager<Resource> resourceLockManager = resourceLockPair.getRight();
      476      resourceLockManager.clearLockDetails();
      477      List<ReadWriteLock> locks = StreamSupport.stream(lockListProvider.apply(resourceLockPair.getKey().get(resource))
      478              .spliterator(), false).collect(Collectors.toList());
      479      // Release locks in reverse order.
      480      Collections.reverse(locks);
      481      for (ReadWriteLock lock : locks) {
      482        if (isReadLock) {
      483          lock.readLock().unlock();
      484          updateReadUnlockMetrics(resource, (ReentrantReadWriteLock) lock);
      485        } else {
      486          boolean isWriteLocked = ((ReentrantReadWriteLock)lock).isWriteLockedByCurrentThread();
      487          lock.writeLock().unlock();
      488          updateWriteUnlockMetrics(resource, (ReentrantReadWriteLock) lock, isWriteLocked);
      489        }
      490      }
      491      return resourceLockManager.unlockResource(resource);
      492    }
