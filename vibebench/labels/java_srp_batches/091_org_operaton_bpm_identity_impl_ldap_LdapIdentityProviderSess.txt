      227    @Override
      228    public boolean checkPassword(String userId, String password) {
      229  
      230      // prevent a null password
      231      if (password == null) {
      232        return false;
      233      }
      234  
      235      // engine can't work without users
      236      if (userId == null || userId.isEmpty()) {
      237        return false;
      238      }
      239  
      240      /*
      241       * We only allow login with no password if anonymous login is set.
      242       * RFC allows such a behavior but discourages the usage, so we provide it for
      243       * user which have a ldap with anonymous login.
      244       */
      245      if (!ldapConfiguration.isAllowAnonymousLogin() && password.isEmpty()) {
      246        return false;
      247      }
      248  
      249      // first search for user using manager DN
      250      LdapUserEntity user = (LdapUserEntity) findUserById(userId);
      251      close();
      252  
      253      if (user == null) {
      254        return false;
      255      } else {
      256  
      257        LdapContext context = null;
      258  
      259        try {
      260          // bind authenticate for user + supplied password
      261          context = ldapClient.openContext(user.getDn(), password);
      262          return true;
      263  
      264        } catch (LdapAuthenticationException e) {
      265          if(ldapConfiguration.isPasswordCheckCatchAuthenticationException()) {
      266            return false;
      267          } else {
      268            throw e;
      269          }
      270  
      271        } finally {
      272          ldapClient.closeLdapCtx(context);
      273  
      274        }
      275  
      276      }
      277  
      278    }

      281  
      282      StringWriter search = new StringWriter();
      283      search.write("(&");
      284  
      285      // restrict to users
      286      search.write(ldapConfiguration.getUserSearchFilter());
      287  
      288      // add additional filters from query
      289      if (query.getId() != null) {
      290        addFilter(ldapConfiguration.getUserIdAttribute(), escapeLDAPSearchFilter(query.getId()), search);
      291      }
      292      if (query.getIds() != null && query.getIds().length > 0) {
      293        // wrap ids in OR statement
      294        search.write("(|");
      295        for (String userId : query.getIds()) {
      296          addFilter(ldapConfiguration.getUserIdAttribute(), escapeLDAPSearchFilter(userId), search);
      297        }
      298        search.write(")");
      299      }
      300      if (query.getEmail() != null) {
      301        addFilter(ldapConfiguration.getUserEmailAttribute(), query.getEmail(), search);
      302      }
      303      if (query.getEmailLike() != null) {
      304        addFilter(ldapConfiguration.getUserEmailAttribute(), query.getEmailLike(), search);
      305      }

      307        addFilter(ldapConfiguration.getUserFirstnameAttribute(), query.getFirstName(), search);

      428    protected <E extends DbEntity, T> List<T> retrieveResults(String baseDn,
      429                                                              String filter,
      430                                                              Function<SearchResult, E> transformEntity,
      431                                                              Predicate<String> resultCountPredicate,
      432                                                              int maxResults,
      433                                                              int firstResult,
      434                                                              boolean ignorePagination) {
      435      StringBuilder resultLogger = initializeResultLogger();
      436      List<T> entities = new ArrayList<>();
      437      int resultCount = 0;
      438  
      439      do {
      440        resultCount = processSearchPage(baseDn, filter, transformEntity, resultCountPredicate, 
      441                                        maxResults, firstResult, ignorePagination, 
      442                                        entities, resultCount, resultLogger);
      443      } while (paginationContinues(entities.size(), maxResults));
      444  
      445      logResults(resultLogger);
      446      return entities;
      447    }

      449    private <E extends DbEntity, T> int processSearchPage(String baseDn,
      450                                                          String filter,
      451                                                          Function<SearchResult, E> transformEntity,
      452                                                          Predicate<String> resultCountPredicate,
      453                                                          int maxResults,
      454                                                          int firstResult,
      455                                                          boolean ignorePagination,
      456                                                          List<T> entities,
      457                                                          int resultCount,
      458                                                          StringBuilder resultLogger) {
      459      try (LdapSearchResults searchResults = ldapClient.search(baseDn, filter)) {
      460        while (searchResults.hasMoreElements() && shouldContinueProcessing(entities, maxResults, ignorePagination)) {
      461          SearchResult result = searchResults.nextElement();
      462          resultCount = processSearchResult(result, transformEntity, resultCountPredicate, 
      463                                           firstResult, ignorePagination, entities, 
      464                                           resultCount, resultLogger);
      465        }
      466      }
      467      return resultCount;
      468    }
