       31     * This metadata file holds the scheme for the complete tileset, so it only has to be requested once from a map client for
       32     * the full session because it contains the scheme information for all tiles.
       33     * This is a POC approach for generating the MLT metadata from a MVT tile.

       35     * - Generate the metadata scheme from the MVT TileJson file -> but not all types of MLT (like float, double, ...)
       36     *   are available in the TileJSON file

      221    public static String createTilesetMetadataJSON(MltTilesetMetadata.TileSetMetadata pbMetadata) {
      222      var root = new TreeMap<String, Object>();
      223      final int version = 1;
      224      root.put("version", version);
      225      if (pbMetadata.hasName()) {
      226        root.put("name", pbMetadata.getName());
      227      }
      228      if (pbMetadata.hasDescription()) {
      229        root.put("description", pbMetadata.getDescription());
      230      }
      231      if (pbMetadata.hasAttribution()) {
      232        root.put("attribution", pbMetadata.getAttribution());
      233      }
      234      if (pbMetadata.hasMinZoom()) {
      235        root.put("minZoom", pbMetadata.getMinZoom());
      236      }
      237      if (pbMetadata.hasMaxZoom()) {
      238        root.put("maxZoom", pbMetadata.getMaxZoom());
      239      }
      240  
      241      var bounds = new ArrayList<Map<String, Object>>();
      242      for (int i = 0; i < (pbMetadata.getBoundsCount() / 4); ++i) {
      243        var bound = new TreeMap<String, Object>();
      244        bound.put("left", pbMetadata.getBounds(4 * i));
      245        bound.put("top", pbMetadata.getBounds((4 * i) + 1));
      246        bound.put("right", pbMetadata.getBounds((4 * i) + 2));
      247        bound.put("bottom", pbMetadata.getBounds((4 * i) + 3));
      248        bounds.add(bound);
      249      }
      250      if (!bounds.isEmpty()) {
      251        root.put("bounds", bounds);
      252      }
      253  
      254      var centers = new ArrayList<Map<String, Object>>();
      255      for (int i = 0; i < (pbMetadata.getCenterCount() / 2); ++i) {
      256        var center = new TreeMap<String, Object>();
      257        center.put("longitude", 2 * i);
      258        center.put("latitude", (2 * i) + 1);
      259        centers.add(center);
      260      }
      261      if (!centers.isEmpty()) {
      262        root.put("center", centers);
      263      }
      264  
      265      return new Gson().toJson(root);

      269    /// Takes the values individually because, despite having mostly
      270    /// the same information, the field and column are separate types.
      271    private static void writeColumnOrFieldType(
      272        DataOutputStream stream,
      273        String name,
      274        boolean isNullable,
      275        boolean vertexScope,
      276        @Nullable MltTilesetMetadata.ScalarType physicalScalarType,
      277        @Nullable MltTilesetMetadata.LogicalScalarType logicalScalarType,
      278        @Nullable MltTilesetMetadata.ComplexType physicalComplexType,
      279        @Nullable MltTilesetMetadata.LogicalComplexType logicalComplexType,
      280        @Nullable List<MltTilesetMetadata.Field> children)
      281        throws IOException {
      282      final boolean isComplex = physicalComplexType != null || logicalComplexType != null;
      283      final boolean isLogical = logicalScalarType != null || logicalComplexType != null;
      284      final boolean hasChildren = (children != null && !children.isEmpty());
      285  
      286      final var options =
      287          (isNullable ? ColumnOptions.NULLABLE : 0)
      288              | (isComplex ? ColumnOptions.COMPLEX_TYPE : 0)
      289              | (isLogical ? ColumnOptions.LOGICAL_TYPE : 0)
      290              | (hasChildren ? ColumnOptions.HAS_CHILDREN : 0)
      291              | (vertexScope ? ColumnOptions.VERTEX_SCOPE : 0);
      292      EncodingUtils.putVarInt(stream, options);
      293  
      294      EncodingUtils.putString(stream, name);
      295  
      296      if (physicalScalarType != null) {
      297        EncodingUtils.putVarInt(stream, physicalScalarType.getNumber());
      298      } else if (physicalComplexType != null) {
      299        EncodingUtils.putVarInt(stream, physicalComplexType.getNumber());
      300      } else if (logicalScalarType != null) {
      301        EncodingUtils.putVarInt(stream, logicalScalarType.getNumber());
      302      } else if (logicalComplexType != null) {
      303        EncodingUtils.putVarInt(stream, logicalComplexType.getNumber());
      304      } else {
      305        throw new IllegalArgumentException("invalid type specification");
      306      }
      307  
      308      if (hasChildren) {
      309        EncodingUtils.putVarInt(stream, children.size());
      310        for (var child : children) {
      311          final boolean complex = child.hasComplexField();
      312          final boolean logical =
      313              (complex && child.getComplexField().hasLogicalType())
      314                  || (!complex && child.getScalarField().hasLogicalType());
      315  
      316          writeColumnOrFieldType(
      317              stream,
      318              child.getName(),
      319              child.getNullable(),
      320              /* vertexScope= */ false,
      321              (!complex && !logical) ? child.getScalarField().getPhysicalType() : null,
      322              (!complex && logical) ? child.getScalarField().getLogicalType() : null,
      323              (complex && !logical) ? child.getComplexField().getPhysicalType() : null,
      324              (complex && logical) ? child.getComplexField().getLogicalType() : null,
      325              complex ? child.getComplexField().getChildrenList() : null);
      326        }
      327      }
      328    }

      333    public static byte[] createEmbeddedMetadata(MltTilesetMetadata.TileSetMetadata pbMetadata)
      334        throws IOException {
      335      try (var byteStream = new ByteArrayOutputStream()) {
      336        try (var dataStream = new DataOutputStream(byteStream)) {
      337          for (var table : pbMetadata.getFeatureTablesList()) {
      338            EncodingUtils.putString(dataStream, table.getName());
      339            EncodingUtils.putVarInt(dataStream, table.getColumnsCount());
      340            for (var column : table.getColumnsList()) {
      341              writeColumnOrFieldType(
      342                  dataStream,
      343                  column.getName(),
      344                  column.getNullable(),
      345                  (column.getColumnScope() == MltTilesetMetadata.ColumnScope.VERTEX),
      346                  column.hasScalarType() && column.getScalarType().hasPhysicalType()
      347                      ? column.getScalarType().getPhysicalType()
      348                      : null,
      349                  column.hasScalarType() && column.getScalarType().hasLogicalType()
      350                      ? column.getScalarType().getLogicalType()
      351                      : null,
      352                  column.hasComplexType() && column.getComplexType().hasPhysicalType()
      353                      ? column.getComplexType().getPhysicalType()
      354                      : null,
      355                  column.hasComplexType() && column.getComplexType().hasLogicalType()
      356                      ? column.getComplexType().getLogicalType()
      357                      : null,
      358                  column.hasComplexType() ? column.getComplexType().getChildrenList() : null);
      359            }
      360          }
      361        }
