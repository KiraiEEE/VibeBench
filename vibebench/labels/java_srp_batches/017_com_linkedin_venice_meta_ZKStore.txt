      315    /**
      316     * Set current serving version number of this store. If store is disabled to write, thrown {@link
      317     * StoreDisabledException}.
      318     *
      319     * When a new version is promoted to current, this method also sets the previousCurrentVersion
      320     * field on the NEW current version to track which version was current before.
      321     */
      322    @Override
      323    public void setCurrentVersion(int currentVersion) {
      324      checkDisableStoreWrite("setStoreCurrentVersion", currentVersion);
      325      // Update the latest version promotion to current timestamp, which is useful for backup version retention.
      326      setLatestVersionPromoteToCurrentTimestamp(System.currentTimeMillis());
      327  
      328      // Capture old current version before updating
      329      int oldCurrentVersion = getCurrentVersion();
      330  
      331      setCurrentVersionWithoutCheck(currentVersion);
      332  
      333      // Set previousCurrentVersion on the NEW current version
      334      if (oldCurrentVersion != Store.NON_EXISTING_VERSION && currentVersion != Store.NON_EXISTING_VERSION) {
      335        updateVersionPreviousCurrentVersion(currentVersion, oldCurrentVersion);
      336      }
      337    }

      340    private void updateVersionPreviousCurrentVersion(int versionNumber, int previousCurrentVersion) {
      341      for (StoreVersion storeVersion: storeProperties.versions) {
      342        if (storeVersion.number == versionNumber) {
      343          storeVersion.previousCurrentVersion = previousCurrentVersion;
      344          return;
      345        }
      346      }
