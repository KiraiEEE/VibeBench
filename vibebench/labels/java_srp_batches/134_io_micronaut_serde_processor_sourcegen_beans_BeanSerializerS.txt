      104          Map<String, String> keyFieldNames = new LinkedHashMap<>();

      106          Map<String, String> serializerFieldNames = new LinkedHashMap<>();
      107          List<FieldDef> fields = new ArrayList<>();
      108  
      109          int index = 0;
      110          for (BeanSerdeShape.BeanProperty property : beanSerdeShape.properties()) {
      111              String keyFieldName = constantFieldName("KEY", property.name(), index);
      112              String argumentFieldName = constantFieldName("ARGUMENT", property.name(), index);
      113              keyFieldNames.put(property.name(), keyFieldName);
      114              argumentFieldNames.put(property.name(), argumentFieldName);
      115  
      116              fields.add(FieldDef.builder(keyFieldName, STRING_TYPE)
      117                  .addModifiers(Modifier.PRIVATE, Modifier.STATIC, Modifier.FINAL)
      118                  .initializer(ExpressionDef.constant(property.name()))
      119                  .build());
      120              fields.add(FieldDef.builder(argumentFieldName, ARGUMENT_TYPE)
      121                  .addModifiers(Modifier.PRIVATE, Modifier.STATIC, Modifier.FINAL)
      122                  .initializer(BeanSerdeSourceGenUtils.argumentExpression(property.serializationType()))
      123                  .build());

      125                  String serializerFieldName = constantFieldName("SERIALIZER", property.name(), index);
      126                  serializerFieldNames.put(property.name(), serializerFieldName);
      127                  fields.add(FieldDef.builder(serializerFieldName, SERIALIZER_TYPE)
      128                      .addModifiers(Modifier.PRIVATE, Modifier.FINAL)
      129                      .addAnnotation(Nullable.class)
      130                      .build());
      131              }
      132              index++;
      133          }
      134

      136              .addModifiers(Modifier.PUBLIC, Modifier.FINAL)

      294      private StatementDef serializeProperty(VariableDef.This aThis,
      295                                             ClassTypeDef serializerClassTypeDef,
      296                                             VariableDef objectEncoder,
      297                                             VariableDef.MethodParameter context,
      298                                             VariableDef.MethodParameter type,
      299                                             VariableDef.MethodParameter value,
      300                                             BeanSerdeShape.BeanProperty property,
      301                                             int index,
      302                                             Map<String, String> argumentFieldNames,
      303                                             Map<String, String> serializerFieldNames) {
      304          ExpressionDef argumentExpression = serializerClassTypeDef.getStaticField(argumentFieldNames.get(property.name()), ARGUMENT_TYPE);
      305          Method scalarMethod = scalarEncoderMethod(property.serializationType());
      306          ExpressionDef propertyValue = value.invoke(property.readMethod());
      307          if (scalarMethod != null) {
      308              StatementDef scalarStatement;
      309              if (property.serializationType().isPrimitive() && !property.serializationType().isArray()) {
      310                  scalarStatement = objectEncoder.invoke(scalarMethod, propertyValue);
      311              } else {
      312                  StatementDef.DefineAndAssign propertyValueDef = propertyValue.newLocal(BeanSerdeSourceGenUtils.localName(VALUE_LOCAL_PREFIX, property.name(), index));
      313                  scalarStatement = StatementDef.multi(
      314                      propertyValueDef,
      315                      propertyValueDef.variable().isNull().ifTrue(
      316                          objectEncoder.invoke(ENCODE_NULL_METHOD),
      317                          StatementDef.multi(objectEncoder.invoke(scalarMethod, propertyValueDef.variable()))
      318                      )
      319                  );
      320              }
      321              return wrapWithPropertyPath(scalarStatement, type, property.name(), argumentExpression);
      322          }
      323          String serializerFieldName = serializerFieldNames.get(property.name());
      324          StatementDef.DefineAndAssign serializerDef = new StatementDef.DefineAndAssign(
      325              new VariableDef.Local(BeanSerdeSourceGenUtils.localName("serializer", property.name(), index), NULLABLE_SERIALIZER_TYPE),
      326              aThis.field(serializerFieldName, SERIALIZER_TYPE)
      327          );
      328          StatementDef initializeSerializerStatement = serializerDef.variable().isNull().ifTrue(
      329              serializerDef.variable().assign(context.invoke(FIND_SERIALIZER_METHOD, argumentExpression).invoke(CREATE_SPECIFIC_SERIALIZER_METHOD, context, argumentExpression))
      330          );
      331          VariableDef serializer = serializerDef.variable();
      332  
      333          StatementDef serializeStatement = StatementDef.multi(
      334              serializerDef,
      335              initializeSerializerStatement,
      336              serializer.invoke(
      337                  SERIALIZE_METHOD,
      338                  objectEncoder,
      339                  context,
      340                  argumentExpression,
      341                  propertyValue.cast(TypeDef.OBJECT)
      342              )
      343          );
      344          if (property.serializationType().isPrimitive() && !property.serializationType().isArray()) {
      345              return wrapWithPropertyPath(serializeStatement, type, property.name(), argumentExpression);
      346          }
      347          StatementDef.DefineAndAssign propertyValueDef = propertyValue.newLocal(BeanSerdeSourceGenUtils.localName(VALUE_LOCAL_PREFIX, property.name(), index));
      348          return wrapWithPropertyPath(StatementDef.multi(
      349              propertyValueDef,
      350              propertyValueDef.variable().isNull().ifTrue(
      351                  objectEncoder.invoke(ENCODE_NULL_METHOD),
      352                  StatementDef.multi(
      353                      serializerDef,
      354                      initializeSerializerStatement,
      355                      serializer.invoke(
      356                          SERIALIZE_METHOD,
      357                          objectEncoder,
      358                          context,
      359                          argumentExpression,
      360                          propertyValueDef.variable().cast(TypeDef.OBJECT)
      361                      )
      362                  )
      363              )
      364          ), type, property.name(), argumentExpression);
      365      }

      371      private Method scalarEncoderMethod(ClassElement type) {
      372          return switch (type.getName()) {
      373              case "boolean", "java.lang.Boolean" -> type.isArray() ? null : ENCODE_BOOLEAN_METHOD;
      374              case "byte", "java.lang.Byte" -> type.isArray() ? null : ENCODE_BYTE_METHOD;
      375              case "short", "java.lang.Short" -> type.isArray() ? null : ENCODE_SHORT_METHOD;
      376              case "char", "java.lang.Character" -> type.isArray() ? null : ENCODE_CHAR_METHOD;
      377              case "int", "java.lang.Integer" -> type.isArray() ? null : ENCODE_INT_METHOD;
      378              case "long", "java.lang.Long" -> type.isArray() ? null : ENCODE_LONG_METHOD;
      379              case "float", "java.lang.Float" -> type.isArray() ? null : ENCODE_FLOAT_METHOD;
      380              case "double", "java.lang.Double" -> type.isArray() ? null : ENCODE_DOUBLE_METHOD;
      381              case "java.lang.String" -> type.isArray() ? null : ENCODE_STRING_METHOD;
      382              case "java.math.BigInteger" -> type.isArray() ? null : ENCODE_BIG_INTEGER_METHOD;
      383              case "java.math.BigDecimal" -> type.isArray() ? null : ENCODE_BIG_DECIMAL_METHOD;
      384              default -> null;
      385          };
      386      }

      388      private StatementDef wrapWithPropertyPath(StatementDef statement,
      389                                                VariableDef.MethodParameter type,
      390                                                String propertyName,
      391                                                ExpressionDef argumentExpression) {
      392          return StatementDef.doTry(statement)
      393              .doCatch(ClassTypeDef.of(Throwable.class), exceptionVariable ->
      394                  ClassTypeDef.of(GeneratedSerdeExceptionUtil.class)
      395                      .invokeStatic(
      396                          WITH_PROPERTY_PATH_THROWABLE_METHOD,
      397                          exceptionVariable,
      398                          type,
      399                          ExpressionDef.constant(propertyName),
      400                          argumentExpression
      401                      )
      402                      .doThrow()
      403              );
      404      }
