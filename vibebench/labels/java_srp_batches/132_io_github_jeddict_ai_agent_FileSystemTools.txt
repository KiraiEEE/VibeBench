      118       * Searches for files in the file system given a directory and a regex pattern.
      119       * The tool scans all folders and subfolders of the given directory.
      120       *
      121       * @param path the directory path relative to the project to start the search from
      122       * @param regexPattern the regex pattern to match against the absolute path of the files

      125      @Tool("""
      126      Recursively find files in a given root folder whose file name matches a regex
      127      pattern. The pattern is matched against the full pathname. It returns a
      128      newline-separated list of relative pathnames, or an empty string if no matches
      129      are found. It returns an error message starting with "ERR:" if the starting
      130      path does not exist. If the pattern is empty, it matches all files.
      131      """)
      132      @ToolPolicy(READONLY)
      133      public String findFiles(
      134          @P("root folder path")
      135          final String path,
      136          @P("regex pattern matched against the path of each file found.")
      137          final String regexPattern
      138      ) throws Exception {
      139          progress("🔎 Searching for files matching '" + regexPattern + "' in directory '" + path + "'");
      140          Path startDir = fullPath(path);
      141  
      142          if (!Files.exists(startDir) || !Files.isDirectory(startDir)) {
      143               progress("❌ invalid directory: " + path);
      144               return "ERR: invalid directory " + path;
      145          }
      146  
      147          final Pattern pattern = ((regexPattern == null) || regexPattern.isBlank())
      148                                ? null : Pattern.compile(regexPattern);
      149  
      150          try (Stream<Path> stream = Files.walk(startDir)) {
      151              Stream<Path> fileStream = stream.filter(p -> !Files.isDirectory(p));
      152  
      153              if (pattern != null) {
      154                  fileStream = fileStream.filter(p -> pattern.matcher(p.toAbsolutePath().toString()).find());
      155              }
      156  
      157              List<String> matches = fileStream
      158                      .map(p -> basepath.relativize(p).toString())
      159                      .sorted()
      160                      .collect(Collectors.toList());
      161  
      162              if (matches.isEmpty()) {
      163                  progress("⚠️ No matches found for '" + regexPattern + "' in: " + path);
      164                  //
      165                  // TODO: back to return "" once https://github.com/langchain4j/langchain4j/issues/4300
      166                  //       will be fixed
      167                  //
      168                  return "No matches found for " + regexPattern;
      169              }
      170  
      171              String result = String.join("\n", matches);
      172              progress("✅ Found " + matches.size() + " matching files.");
      173              return result;
      174          } catch (IOException e) {
      175              progress("❌ Error searching files: " + e.getMessage());
      176              throw e;
      177          }
      178      }
