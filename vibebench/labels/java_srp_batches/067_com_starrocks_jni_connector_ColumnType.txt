      184      private void parseStruct(List<String> childNames, List<ColumnType> childTypeValues, StringScanner scanner) {
      185          while (scanner.peek() != '>') {

      199      private void parse(StringScanner scanner) {
      200          int p = scanner.indexOf('<', ',', '>', '(', ')');
      201          int end = scanner.indexOf(')') + 1;
      202          String t = scanner.substr(p);
      203          if (t.startsWith("decimal")) {
      204              t = scanner.substr(end);
      205              scanner.moveTo(end);
      206          } else if (t.startsWith("char") || t.startsWith("varchar")) {
      207              // right now this only used in hive scanner
      208              // for char(xx) and varchar(xx), we only need t to be char or varchar and skip (xx)
      209              // otherwise struct<c_char:char(30),c_varchar:varchar(200)> will get wrong result
      210              scanner.moveTo(end);
      211          } else {
      212              scanner.moveTo(p);
      213          }
      214          // assume there is no blank char in `type`.
      215          typeValue = null;
      216          switch (t) {
      217              case "array": {
      218                  // array<TYPE>
      219                  typeValue = TypeValue.ARRAY;
      220                  childNames = new ArrayList<>();
      221                  childTypes = new ArrayList<>();
      222                  parseArray(childTypes, scanner);
      223              }
      224              break;
      225              case "map": {
      226                  // map<TYPE1,TYPE2>
      227                  typeValue = TypeValue.MAP;
      228                  childNames = new ArrayList<>();
      229                  childTypes = new ArrayList<>();
      230                  parseArray(childTypes, scanner);
      231              }
      232              break;
      233              case "struct": {
      234                  // struct<F1:TYPE1,F2:TYPE2,F3:TYPE3..>
      235                  typeValue = TypeValue.STRUCT;
      236                  childNames = new ArrayList<>();
      237                  childTypes = new ArrayList<>();
      238                  parseStruct(childNames, childTypes, scanner);
      239              }
      240              break;
      241              default: {
      242                  if (t.startsWith("decimal")) {
      243                      typeValue = parseDecimal(t);
      244                  } else {
      245                      typeValue = PRIMITIVE_TYPE_VALUE_MAPPING.getOrDefault(t, null);
      246                  }
      247              }
      248          }
      249  
      250          if (typeValue == null) {
      251              throw new RuntimeException("Unsupported type: " + t);
      252          }
      253      }

      381      public void pruneOnSelectedFields(SelectedFields ssf) {
      382          // If no spec at all, then select all fields
      383          if (ssf == null) {
      384              fieldIndex = new ArrayList<>();
      385              for (int i = 0; i < childNames.size(); i++) {
      386                  fieldIndex.add(i);
      387              }
      388              pruneChildren(null);
      389              return;
      390          }
      391  
      392          // make index and prune on this struct.
      393          Map<String, Integer> index = new HashMap<>();
      394          for (int i = 0; i < childNames.size(); i++) {
      395              index.put(childNames.get(i), i);
      396          }
      397  
      398          List<String> fields = ssf.getFields();
      399          List<String> names = new ArrayList<>();
      400          List<ColumnType> types = new ArrayList<>();
      401          fieldIndex = new ArrayList<>();
      402          for (String f : fields) {
      403              Integer i = index.get(f);
      404              if (i == null) {
      405                  // field name mismatch.
      406                  fieldIndex.add(null);
      407                  types.add(new ColumnType(name + '.' + f, TypeValue.UNKNOWN));
      408              } else {
      409                  fieldIndex.add(i);
      410                  types.add(childTypes.get(i));
      411              }
      412              names.add(f);
      413          }
      414          childNames = names;
      415          childTypes = types;
      416          pruneChildren(ssf);
      417      }
