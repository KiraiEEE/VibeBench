      103    public String getSummary() {
      104      Map<Result, Long> counts = results.values().stream()
      105          .collect(Collectors.groupingBy(r -> r, () -> new EnumMap<>(Result.class), Collectors.counting()));
      106      long total = results.size();
      107      switch (action) {
      108        case CHECK:
      109          return String.format("License check: %d file(s) checked (OK: %d, Missing: %d, Unknown: %d)",
      110              total,
      111              counts.getOrDefault(Result.PRESENT, 0L),
      112              counts.getOrDefault(Result.MISSING, 0L),
      113              counts.getOrDefault(Result.UNKNOWN, 0L));
      114        case FORMAT:
      115          return String.format("License format: %d file(s) processed (Added: %d, Replaced: %d, OK: %d, Unknown: %d)",
      116              total,

      118              counts.getOrDefault(Result.REPLACED, 0L),
      119              counts.getOrDefault(Result.NOOP, 0L),
      120              counts.getOrDefault(Result.UNKNOWN, 0L));
      121        case REMOVE:
