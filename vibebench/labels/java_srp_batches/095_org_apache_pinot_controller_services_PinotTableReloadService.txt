      195    public SuccessResponse reloadSegmentsInTimeRange(String tableName, @Nullable String tableTypeStr,
      196        @Nullable String startTimestampStr, @Nullable String endTimestampStr, boolean excludeOverlapping,
      197        boolean forceDownload, @Nullable String targetInstance, HttpHeaders headers) {
      198      long startTimestamp = Long.MIN_VALUE;
      199      long endTimestamp = Long.MAX_VALUE;
      200      if (Strings.isNullOrEmpty(startTimestampStr) && Strings.isNullOrEmpty(endTimestampStr)) {
      201        throw new ControllerApplicationException(LOG, "At least one of startTimestamp or endTimestamp must be provided.",
      202            Response.Status.BAD_REQUEST);
      203      }
      204      try {
      205        if (!Strings.isNullOrEmpty(startTimestampStr)) {
      206          startTimestamp = Long.parseLong(startTimestampStr);
      207        }
      208        if (!Strings.isNullOrEmpty(endTimestampStr)) {
      209          endTimestamp = Long.parseLong(endTimestampStr);
      210        }
      211      } catch (NumberFormatException e) {

      214            Response.Status.BAD_REQUEST, e);
      215      }
      216      if (startTimestamp >= endTimestamp) {
      217        throw new ControllerApplicationException(LOG, String.format(
      218            "startTimestamp must be less than endTimestamp. Provided: start=%d, end=%d", startTimestamp, endTimestamp),
      219            Response.Status.BAD_REQUEST);
      220      }

      222      tableName = DatabaseUtils.translateTableName(tableName, headers);
      223      TableType tableTypeFromRequest = resolveTableTypeForReload(tableName, tableTypeStr, forceDownload);
      224      List<String> tableNamesWithType =
      225          ResourceUtils.getExistingTableNamesWithType(_pinotHelixResourceManager, tableName, tableTypeFromRequest, LOG);
      226      Map<String, Map<String, String>> perTableMsgData = new LinkedHashMap<>();
      227      for (String tableNameWithType : tableNamesWithType) {
      228        List<String> segments =
      229            _pinotHelixResourceManager.getSegmentsFor(tableNameWithType, true, startTimestamp, endTimestamp,
      230                excludeOverlapping);
      231        if (segments.isEmpty()) {
      232          continue;
      233        }
      234        Set<String> selectedSegments = new HashSet<>(segments);
      235        // TODO: For large tables, consider adding a HelixResourceManager helper that derives the per-instance
      236        //  segment map directly for the provided segment set, rather than fetching the full server-to-segments
      237        //  map and filtering.
      238        Map<String, List<String>> serverToSegmentsMap =
      239            _pinotHelixResourceManager.getServerToSegmentsMap(tableNameWithType, targetInstance, false);
      240        Map<String, List<String>> filteredInstanceToSegmentsMap = new HashMap<>();

      246          }
      247        }
      248        if (filteredInstanceToSegmentsMap.isEmpty()) {

      250        }

      265            .distinct()
      266            .sorted()

      282        } catch (Exception e) {

      291      }
      292      try {
      293        return new SuccessResponse(JsonUtils.objectToString(perTableMsgData));

      296            Response.Status.INTERNAL_SERVER_ERROR, e);

      334    /**
      335     * Resolves the effective table type for a reload request.
      336     * When forceDownload is requested and no table type is specified (neither in the table name nor the request),
      337     * defaults to OFFLINE since forceDownload only works with OFFLINE tables currently.
      338     */
      339    private TableType resolveTableTypeForReload(String tableName, String tableTypeStr, boolean forceDownload) {
      340      TableType tableTypeFromTableName = TableNameBuilder.getTableTypeFromTableName(tableName);
      341      TableType tableTypeFromRequest = Constants.validateTableType(tableTypeStr);
      342      // When rawTableName is provided but without table type, Pinot tries to reload both OFFLINE
      343      // and REALTIME tables for the raw table. But forceDownload option only works with
      344      // OFFLINE table currently, so we limit the table type to OFFLINE to let Pinot continue
      345      // to reload without being accidentally aborted upon REALTIME table type.
      346      // TODO: support to force download immutable segments from RealTime table.
      347      if (forceDownload && tableTypeFromTableName == null && tableTypeFromRequest == null) {
      348        return TableType.OFFLINE;
      349      }
      350      return tableTypeFromRequest;
      351    }

      357    private String getExistingTable(String tableName, String segmentName) {
      358      TableType tableType = TableNameBuilder.getTableTypeFromTableName(tableName);
      359      if (tableType == null) {
      360        // Derive table type from segment name if the given table name doesn't have type suffix
      361        tableType = LLCSegmentName.isLLCSegment(segmentName) ? TableType.REALTIME
      362            : (UploadedRealtimeSegmentName.isUploadedRealtimeSegmentName(segmentName) ? TableType.REALTIME
      363                : TableType.OFFLINE);
      364      }
      365      return ResourceUtils.getExistingTableNamesWithType(_pinotHelixResourceManager, tableName, tableType, LOG).get(0);
      366    }

      368    private Map<String, Map<String, String>> reloadSegmentsForTable(String tableNameWithType, boolean forceDownload,
      369        Map<String, List<String>> instanceToSegmentsMap) {
      370      long startTimeMs = System.currentTimeMillis();
      371      // Use a single shared job ID across all instances so the logical reload operation can be tracked
      372      // via a single ZK entry, consistent with the time-range reload path.
      373      String reloadJobId = UUID.randomUUID().toString();
      374      String targetInstance = instanceToSegmentsMap.size() == 1 ? instanceToSegmentsMap.keySet().iterator().next() : null;
      375      Map<String, Integer> instanceMsgCountMap =
      376          _pinotHelixResourceManager.reloadSegments(tableNameWithType, forceDownload, instanceToSegmentsMap, reloadJobId);
      377      int totalNumReloadMsgSent = 0;
      378      Map<String, Map<String, String>> instanceMsgData = new HashMap<>();
      379      for (Map.Entry<String, Integer> instanceMsgInfo : instanceMsgCountMap.entrySet()) {
      380        String instance = instanceMsgInfo.getKey();
      381        int numReloadMsgSent = instanceMsgInfo.getValue();
      382        if (numReloadMsgSent <= 0) {
      383          continue;
      384        }
      385        totalNumReloadMsgSent += numReloadMsgSent;
      386        Map<String, String> tableReloadMeta = new HashMap<>();
      387        tableReloadMeta.put("numMessagesSent", String.valueOf(numReloadMsgSent));
      388        tableReloadMeta.put("reloadJobId", reloadJobId);
      389        instanceMsgData.put(instance, tableReloadMeta);
      390      }
      391      if (totalNumReloadMsgSent > 0) {
      392        // Store a single ZK entry for the entire job (all instances share the same reloadJobId).
      393        Set<String> allSegments = new TreeSet<>();
      394        for (List<String> segments : instanceToSegmentsMap.values()) {
      395          allSegments.addAll(segments);
      396        }
      397        String segmentNamesStr = StringUtils.join(allSegments, SegmentNameUtils.SEGMENT_NAME_SEPARATOR);
      398        try {
      399          String instanceToSegmentsMapJson = JsonUtils.objectToString(instanceToSegmentsMap);
      400          if (_pinotHelixResourceManager.addNewReloadSegmentJob(tableNameWithType, segmentNamesStr, targetInstance,
      401              reloadJobId, startTimeMs, totalNumReloadMsgSent, instanceToSegmentsMapJson)) {
      402            instanceMsgData.values().forEach(meta -> meta.put("reloadJobMetaZKStorageStatus", "SUCCESS"));
      403          } else {
      404            instanceMsgData.values().forEach(meta -> meta.put("reloadJobMetaZKStorageStatus", "FAILED"));
      405            LOG.error("Failed to add batch reload job meta into zookeeper for table: {}", tableNameWithType);
      406          }
      407        } catch (Exception e) {
      408          instanceMsgData.values().forEach(meta -> meta.put("reloadJobMetaZKStorageStatus", "FAILED"));
      409          LOG.error("Failed to add batch reload job meta into zookeeper for table: {}", tableNameWithType, e);
      410        }
      411      }
      412      return instanceMsgData;
      413    }
