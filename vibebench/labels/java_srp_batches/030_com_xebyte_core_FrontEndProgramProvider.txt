       76      @Override
       77      public Program getProgram(String name) {
       78          if (name == null || name.trim().isEmpty()) {
       79              return getCurrentProgram();
       80          }
       81  
       82          String searchName = name.trim();
       83  
       84          // 1. ALWAYS check CodeBrowser programs first — if the user has a program open
       85          // in a CodeBrowser, we MUST use that instance so changes are visible immediately.
       86          // This is critical: using our own cached instance would modify a different object
       87          // in memory, and the CodeBrowser wouldn't see the changes until a reload.
       88          List<Program> cbPrograms = collectCodeBrowserPrograms();
       89  
       90          // Path-based match against CodeBrowser programs (e.g., "/Vanilla/1.13c/D2Client.dll")
       91          if (searchName.startsWith("/")) {
       92              for (Program prog : cbPrograms) {
       93                  DomainFile df = prog.getDomainFile();
       94                  String dfPath = (df != null) ? df.getPathname() : "null";
       95                  if (df != null && df.getPathname().equals(searchName)) {
       96                      return prog;
       97                  }
       98              }
       99          }
      100  
      101          // Name-based match against CodeBrowser programs
      102          for (Program prog : cbPrograms) {
      103              if (prog.getName().equalsIgnoreCase(searchName)) {
      104                  return prog;
      105              }
      106          }
      107          // Partial match on CodeBrowser programs
      108          for (Program prog : cbPrograms) {
      109              if (prog.getName().toLowerCase().contains(searchName.toLowerCase())) {
      110                  return prog;
      111              }
      112          }
      113  
      114          // 2. Check path-based cache (handles multi-version same-name DLLs not in CodeBrowser)
      115          if (searchName.startsWith("/")) {
      116              String cacheKey = pathToName.get(searchName);
      117              if (cacheKey != null) {
      118                  Program cached = openPrograms.get(cacheKey);
      119                  if (cached != null) {
      120                      return cached;
      121                  }
      122              }
      123          }
      124  
      125          // 3. Check our cache (only for non-path lookups to avoid collisions)
      126          if (!searchName.startsWith("/")) {
      127              Program cached = openPrograms.get(searchName);
      128              if (cached != null) {
      129                  return cached;
      130              }
      131              // Case-insensitive cache lookup
      132              for (Map.Entry<String, Program> entry : openPrograms.entrySet()) {
      133                  if (entry.getKey().equalsIgnoreCase(searchName)) {
      134                      return entry.getValue();
      135                  }
      136              }
      137          }
      138  
      139          // 4. Try to open from project by name or path
      140          return openFromProject(searchName);
      141      }

      186       * Collect programs from all running CodeBrowser instances, deduplicating
      187       * by object identity and name.
      188       *
      189       * @return Mutable list of unique programs across all CodeBrowsers
      190       */
      191      private List<Program> collectCodeBrowserPrograms() {
      192          List<Program> allPrograms = new ArrayList<>();
      193          for (ProgramManager pm : findAllCodeBrowserProgramManagers()) {
      194              for (Program prog : pm.getAllOpenPrograms()) {
      195                  // Deduplicate by object identity only — not by name,
      196                  // since multiple versions of the same DLL are distinct programs
      197                  boolean alreadyListed = false;
      198                  for (Program existing : allPrograms) {
      199                      if (existing == prog) {
      200                          alreadyListed = true;
      201                          break;
      202                      }
      203                  }
      204                  if (!alreadyListed) {
      205                      allPrograms.add(prog);
      206                  }
      207              }
      208          }
      209          return allPrograms;

      218       * Open a program from the active project by name or path.
      219       *
      220       * @param nameOrPath Program name (e.g., "D2Common.dll") or project path (e.g., "/LoD/1.00/D2Common.dll")
      221       * @return The opened program, or null if not found
      222       */
      223      @Override
      224      public Program openFromProject(String nameOrPath) {
      225          Project project = tool.getProject();
      226          if (project == null) {
      227              Msg.warn(this, "No active project");
      228              return null;
      229          }
      230  
      231          ProjectData projectData = project.getProjectData();
      232          if (projectData == null) {
      233              return null;
      234          }
      235  
      236          DomainFile domainFile = null;
      237  
      238          // Try as absolute path first
      239          if (nameOrPath.startsWith("/")) {
      240              domainFile = projectData.getFile(nameOrPath);
      241          }
      242  
      243          // If not found by path, search recursively by name
      244          if (domainFile == null) {
      245              domainFile = findFileByName(projectData.getRootFolder(), nameOrPath);
      246          }
      247  
      248          if (domainFile == null) {
      249              Msg.info(this, "File not found in project: " + nameOrPath);
      250              return null;
      251          }
      252  
      253          String projectPath = domainFile.getPathname();
      254          // Use project path as unique cache key (handles multiple versions of same DLL)
      255          String cacheKey = projectPath;
      256  
      257          try {
      258              // getDomainObject returns the SAME instance if already open in CodeBrowser
      259              // This is the key to seamless integration — shared domain objects
      260              Program program = (Program) domainFile.getDomainObject(consumer, false, false, monitor);
      261  
      262              // Release previous consumer reference if overwriting a cache entry
      263              // to prevent reference count leaks on the DomainObject
      264              Program previousProgram = openPrograms.get(cacheKey);
      265              if (previousProgram != null && previousProgram != program) {
      266                  try {
      267                      previousProgram.release(consumer);
      268                      Msg.info(this, "Released previous cached program for: " + cacheKey);
      269                  } catch (Exception ex) {
      270                      Msg.warn(this, "Error releasing previous program " + cacheKey + ": " + ex.getMessage());
      271                  }
      272              }
      273  
      274              openPrograms.put(cacheKey, program);
      275              pathToName.put(projectPath, cacheKey);
      276              // Also map the input path if different from project path
      277              if (!nameOrPath.equals(projectPath)) {
      278                  pathToName.put(nameOrPath, cacheKey);
      279              }
      280              if (currentProgram == null) {
      281                  currentProgram = program;
      282              }
      283              Msg.info(this, "Opened program from project: " + program.getName() +
      284                  " (" + projectPath + ")");
      285              return program;
      286          } catch (Exception e) {
      287              Msg.error(this, "Failed to open program: " + nameOrPath + " — " + e.getMessage());
      288              // Try read-only as fallback
      289              try {
      290                  Program program = (Program) domainFile.getImmutableDomainObject(consumer, DomainFile.DEFAULT_VERSION, monitor);
      291  
      292                  // Release previous consumer reference if overwriting
      293                  Program previousProgram = openPrograms.get(cacheKey);
      294                  if (previousProgram != null && previousProgram != program) {
      295                      try {
      296                          previousProgram.release(consumer);
      297                      } catch (Exception ex) {
      298                          Msg.warn(this, "Error releasing previous program " + cacheKey + ": " + ex.getMessage());
      299                      }
      300                  }
      301  
      302                  openPrograms.put(cacheKey, program);
      303                  pathToName.put(projectPath, cacheKey);
      304                  if (!nameOrPath.equals(projectPath)) {
      305                      pathToName.put(nameOrPath, cacheKey);
      306                  }
      307                  if (currentProgram == null) {
      308                      currentProgram = program;
      309                  }
      310                  Msg.info(this, "Opened program read-only: " + program.getName());
      311                  return program;
      312              } catch (Exception e2) {
      313                  Msg.error(this, "Failed to open program even read-only: " + nameOrPath + " — " + e2.getMessage());
      314                  return null;
      315              }
      316          }
      317      }

      320       * Search for a file by name recursively in the project folder tree.
      321       */
      322      private DomainFile findFileByName(DomainFolder folder, String name) {
      323          if (folder == null) {
      324              return null;
      325          }
      326  
      327          // Check files in this folder
      328          try {
      329              for (DomainFile file : folder.getFiles()) {
      330                  if (file.getName().equalsIgnoreCase(name)) {
      331                      return file;
      332                  }
      333              }
      334  
      335              // Recurse into subfolders
      336              for (DomainFolder subfolder : folder.getFolders()) {
      337                  DomainFile found = findFileByName(subfolder, name);
      338                  if (found != null) {
      339                      return found;
      340                  }
      341              }
      342          } catch (Exception e) {
      343              Msg.warn(this, "Error searching folder " + folder.getPathname() + ": " + e.getMessage());
      344          }
      345  
      346          return null;
      347      }

      350       * Find ProgramManagers from ALL running CodeBrowser tool instances.
      351       * When multiple CodeBrowsers are open (e.g., user double-clicks multiple
      352       * programs in FrontEnd), each has its own ProgramManager.
      353       *
      354       * @return List of ProgramManagers from all running CodeBrowsers (may be empty)

      356      private List<ProgramManager> findAllCodeBrowserProgramManagers() {
      357          List<ProgramManager> managers = new ArrayList<>();
      358  
      359          Project project = tool.getProject();
      360          if (project == null) {
      361              return managers;
      362          }
      363  
      364          try {
      365              ghidra.framework.model.ToolManager tm = project.getToolManager();
      366              if (tm == null) {
      367                  return managers;
      368              }
      369  
      370              for (PluginTool runningTool : tm.getRunningTools()) {
      371                  ProgramManager pm = runningTool.getService(ProgramManager.class);
      372                  if (pm != null) {
      373                      managers.add(pm);
      374                  }
      375              }
      376          } catch (Exception e) {
      377              // ToolManager may not be available in all contexts
      378          }
      379  
      380          return managers;
      381      }
