      211      public ProfilingExecPlan getProfilingPlan() {
      212          if (profilingPlan == null) {
      213              synchronized (this) {
      214                  if (profilingPlan == null) {
      215                      boolean needSetCtx = ConnectContext.get() == null && this.connectContext != null;
      216                      try {
      217                          if (needSetCtx) {
      218                              this.connectContext.setThreadLocalInfo();
      219                          }
      220                          profilingPlan = ProfilingExecPlan.buildFrom(this);
      221                      } finally {
      222                          if (needSetCtx) {
      223                              ConnectContext.remove();
      224                          }
      225                      }
      226                  }
      227              }
      228          }
      229          return profilingPlan;
      230      }

      232      public String getExplainString(TExplainLevel level) {
      233          StringBuilder str = new StringBuilder();
      234  
      235          if (level == TExplainLevel.VERBOSE || level == TExplainLevel.COSTS) {
      236              if (FeConstants.showFragmentCost) {
      237                  final String prefix = "  ";
      238                  AuditEvent auditEvent = connectContext.getAuditEventBuilder().build();
      239                  str.append("PLAN COST").append("\n")
      240                          .append(prefix).append("CPU: ").append(auditEvent.planCpuCosts).append("\n")
      241                          .append(prefix).append("Memory: ").append(auditEvent.planMemCosts).append("\n\n");
      242              }
      243          }
      244  
      245          if (level == null) {
      246              str.append(Explain.toString(physicalPlan, outputColumns));
      247          } else {
      248              if (planCount != 0) {
      249                  str.append("There are ").append(planCount).append(" plans in optimizer search space\n");
      250              }
      251              if (useBaseline > 0) {
      252                  str.append("Using baseline plan[").append(useBaseline).append("]\n\n");
      253              }
      254  
      255              for (int i = 0; i < fragments.size(); ++i) {
      256                  PlanFragment fragment = fragments.get(i);
      257                  if (i > 0) {
      258                      // a blank line between plan fragments
      259                      str.append("\n");
      260                  }
      261                  if (level.equals(TExplainLevel.NORMAL)) {
      262                      str.append("PLAN FRAGMENT ").append(i).append("\n");
      263                      str.append(fragment.getExplainString(TExplainLevel.NORMAL));
      264                  } else if (level.equals(TExplainLevel.COSTS)) {
      265                      str.append("PLAN FRAGMENT ").append(i).append("(").append(fragment.getFragmentId()).append(")\n");
      266                      str.append(fragment.getCostExplain());
      267                  } else {
      268                      str.append("PLAN FRAGMENT ").append(i).append("(").append(fragment.getFragmentId()).append(")\n");
      269                      str.append(fragment.getVerboseExplain());
      270                  }
      271              }
      272          }
      273          return str.toString();
      274      }
