      208      public void loadServerInfo() {
      209          try (QueryResponse response = this.query("SELECT currentUser() AS user, timezone() AS timezone, version() AS version LIMIT 1").get()) {
      210              try (ClickHouseBinaryFormatReader reader = this.newBinaryFormatReader(response)) {
      211                  if (reader.next() != null) {
      212                      String tmpDbUser = reader.getString("user");
      213                      if (tmpDbUser != null && !tmpDbUser.isEmpty()) {
      214                          this.dbUser = tmpDbUser;
      215                      }
      216                      this.configuration.put(ClientConfigProperties.SERVER_TIMEZONE.getKey(), reader.getString("timezone"));
      217                      serverVersion = reader.getString("version");
      218                  }
      219              }
      220          } catch (Exception e) {
      221              throw new ClientException("Failed to get server info", e);
      222          }
