      156    public void run() {
      157      setStatus(TaskStatus.PROCESSING);
      158  
      159      List<FeatureList> loadedFeatureLists = new ArrayList<>();
      160      try {
      161        Path tempDirectory = FileAndPathUtil.createTempDirectory(TEMP_FLIST_DATA_FOLDER);
      162  
      163        logger.info(() -> "Unzipping feature lists of project to " + tempDirectory.toString());
      164        ZipUtils.unzipDirectory(FeatureListSaveTask.FLIST_FOLDER, zip, tempDirectory.toFile());
      165        logger.info(() -> "Unzipping feature lists done.");
      166  
      167        File[] files = new File(tempDirectory.toFile(), FeatureListSaveTask.FLIST_FOLDER).listFiles(
      168            (dir, name) -> fileNamePattern.matcher(name).matches());

      170          logger.info("Did not find feature lists to load.");
      171          setStatus(TaskStatus.FINISHED);
      172          return;
      173        }
      174

      176

      178  
      179        // enable caching of mobility scans during project import.
      180        project.setProjectLoadImsImportCaching(true);
      181  
      182        for (File flistFile : files) {
      183          if (isCanceled()) {
      184            return;
      185          }
      186          rowCounter.set(0);
      187  
      188          final File metadataFile = new File(flistFile.toString()
      189              .replace(FeatureListSaveTask.DATA_FILE_SUFFIX,
      190                  FeatureListSaveTask.METADATA_FILE_SUFFIX));
      191          ModularFeatureList flist = createRows(storage, flistFile, metadataFile);
      192  
      193          if (flist == null) {
      194            logger.severe(
      195                () -> "Cannot load feature list from files " + flistFile.getAbsolutePath() + " and "
      196                    + metadataFile.getAbsolutePath());
      197            continue;
      198          }
      199          parseFeatureList(storage, project, flist, flistFile);
      200  
      201          // TODO maybe remove so that ModularFeatureList.getFeatureList can be unmodifiable
      202          // disable buffering after the import (replace references to CachedIMSRawDataFiles with IMSRawDataFiles
      203          flist.replaceCachedFilesAndScans();
      204  
      205          project.addFeatureList(flist);
      206          loadedFeatureLists.add(flist);

      216      }
      217  
      218      // disable caching on project level
      219      project.setProjectLoadImsImportCaching(false);
      220  
      221      //  group flists by date created, only use the latest set of feature lists in next batch step
      222      final Set<FeatureList> mostRecentStepFeatureLists = Set.copyOf(loadedFeatureLists.stream()

      224              Collectors.groupingBy(flist -> flist.getAppliedMethods().getLast().getModuleCallDate()))

      232    private void parseFeatureList(MemoryMapStorage storage, MZmineProject project,
      233        ModularFeatureList flist, File flistFile) {
      234      currentFlist = flist.getName();
      235      processedRows = 0;
      236      totalRows = flist.getNumberOfRows();
      237  
      238      try (InputStream fis = new FileInputStream(flistFile)) {
      239        final XMLInputFactory xif = XMLInputFactory.newInstance();
      240        final XMLStreamReader reader = xif.createXMLStreamReader(fis);
      241  
      242        while (reader.hasNext()) {
      243          if (isCanceled()) {
      244            return;
      245          }
      246  
      247          int type = reader.next();
      248          if (type == XMLEvent.START_ELEMENT) {
      249            final String localName = reader.getLocalName();
      250            if (CONST.XML_FEATURE_LIST_ELEMENT.equals(localName)) {
      251              if (!flist.getName().equals(reader.getAttributeValue(null, CONST.XML_FLIST_NAME_ATTR))
      252                  || !flist.getDateCreated()
      253                  .equals(reader.getAttributeValue(null, CONST.XML_DATE_CREATED_ATTR))) {
      254                throw new IllegalArgumentException(
      255                    "The name of the loaded feature list does not match the expected name. %s != %s Does a feature list with this name already exist?".formatted(
      256                        flist.getName(), reader.getAttributeValue(null, CONST.XML_FLIST_NAME_ATTR)));
      257              }
      258            } else if (CONST.XML_ROW_ELEMENT.equals(localName)) {
      259              parseRow(reader, storage, project, flist);
      260              processedRows++;
      261            }
      262          }
      263        }
      264  
      265      } catch (IOException | XMLStreamException e) {
      266        logger.log(Level.WARNING, "Error opening file " + flistFile.getAbsolutePath(), e);
      267      }
      268    }

      272     * {@link this#readMetadataCreateFeatureList(File, MemoryMapStorage)}.

      274     * Then passes the feature list data file once and creates the rows with the associated ids. No

      283    private ModularFeatureList createRows(MemoryMapStorage storage, File dataFile,
      284        File metadataFile) {
      285  
      286      ModularFeatureList flist = readMetadataCreateFeatureList(metadataFile, storage);
      287      if (flist == null) {
      288        throw new IllegalStateException("Cannot create feature list.");
      289      }
      290  
      291      try (InputStream fis = new FileInputStream(dataFile)) {
      292        final XMLInputFactory xif = XMLInputFactory.newInstance();
      293        final XMLStreamReader reader = xif.createXMLStreamReader(fis);
      294  
      295        logger.finest(
      296            () -> "Creating " + ModularFeatureListRow.class.getSimpleName() + "s for feature list "
      297                + flist.getName() + ".");
      298        while (reader.hasNext()) {
      299          final int type = reader.next();
      300          if (type == XMLEvent.START_ELEMENT && reader.getLocalName().equals(CONST.XML_ROW_ELEMENT)) {
      301            int id = Integer.parseInt(reader.getAttributeValue(null, idTypeUniqueID));
      302            flist.addRow(new ModularFeatureListRow(flist, id));
      303          }
      304        }
      305      } catch (IOException | XMLStreamException e) {
      306        logger.log(Level.WARNING, e.getMessage(), e);
      307        return null;
      308      }
      309  
      310      logger.finest(
      311          () -> "Created " + flist.getNumberOfRows() + " rows in feature list " + flist.getName());
      312      return flist;

      417    private void parseRow(XMLStreamReader reader, MemoryMapStorage storage, MZmineProject project,
      418        ModularFeatureList flist) throws XMLStreamException {
      419      if (!reader.getLocalName().equals(CONST.XML_ROW_ELEMENT)) {
      420        throw new IllegalStateException("Cannot parse row if current element is not a row element");
      421      }
      422  
      423      int id = Integer.parseInt(reader.getAttributeValue(null, idTypeUniqueID));
      424      final ModularFeatureListRow row = (ModularFeatureListRow) flist.getRow(rowCounter.get());
      425      if (id != row.getID()) {
      426        throw new IllegalStateException("Row ids do not match.");
      427      }
      428  
      429      while (!(reader.getEventType() == XMLEvent.END_ELEMENT && reader.getLocalName()
      430          .equals(CONST.XML_ROW_ELEMENT)) && reader.hasNext()) {
      431        if (reader.next() == XMLEvent.START_ELEMENT) {
      432          if (reader.getLocalName().equals(CONST.XML_FEATURE_ELEMENT)) {
      433            final String fileName = reader.getAttributeValue(null, CONST.XML_RAW_FILE_ELEMENT);
      434            final RawDataFile file = project.getCurrentRawDataFiles().stream()
      435                .filter(f -> f.getName().equals(fileName)).findFirst().orElse(null);
      436            if (file == null) {
      437              logger.warning(() -> "Cannot load feature for row id " + id + " for file " + fileName
      438                  + ". File does not exist in project.");
      439              continue;
      440            }
      441            parseFeature(reader, storage, project, flist, row, file);
      442          } else if (reader.getLocalName().equals(CONST.XML_DATA_TYPE_ELEMENT)) {
      443            DataType type = DataTypes.getTypeForId(
      444                reader.getAttributeValue(null, CONST.XML_DATA_TYPE_ID_ATTR));
      445            Object value = parseDataType(reader, type, project, flist, row, null, null);
      446            if (type != null && value != null) {
      447              try {
      448                row.set(type, value);
      449              } catch (RuntimeException e) {
      450                // TODO - maybe log?
      451                logger.log(Level.WARNING, () -> String.format(
      452                    "DataType %s and value %s were not set to row. Maybe incompatible during loading?",
      453                    type, value));
      454                // cannot set bound values. can go silent.
      455              }
      456            }
      457          }
      458        }
      459      }
      460      rowCounter.getAndIncrement();
      461    }
