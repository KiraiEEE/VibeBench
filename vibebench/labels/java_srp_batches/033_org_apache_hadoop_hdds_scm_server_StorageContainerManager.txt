      583    private void initializeCertificateClient() throws IOException {
      584      securityConfig = new SecurityConfig(configuration);
      585      if (OzoneSecurityUtil.isSecurityEnabled(configuration) &&
      586          scmStorageConfig.checkPrimarySCMIdInitialized()) {
      587        SCMSecurityProtocolClientSideTranslatorPB scmSecurityClient =
      588            getScmSecurityClientWithMaxRetry(configuration, getCurrentUser());

      590            securityConfig, scmSecurityClient, scmStorageConfig.getScmId(),
      591            scmStorageConfig.getClusterID(),
      592            scmStorageConfig.getScmCertSerialId(),
      593            getScmAddress(scmHANodeDetails, configuration).getHostName());
      594      }
