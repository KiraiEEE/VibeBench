      297    boolean canDoLightning(
      298        List<LineageRelationship> lineageRelationships,
      299        String input,
      300        Filter inputFilters,
      301        List<SortCriterion> sortCriteria) {
      302      boolean simpleFilters =
      303          inputFilters == null
      304              || inputFilters.getOr() == null
      305              || inputFilters.getOr().stream()
      306                  .allMatch(
      307                      criterion ->
      308                          criterion.getAnd().stream()
      309                              .allMatch(
      310                                  criterion1 ->
      311                                      "platform".equals(criterion1.getField())
      312                                          || "origin".equals(criterion1.getField())));
      313      return (lineageRelationships.size()
      314              > appConfig.getCache().getSearch().getLineage().getLightningThreshold())
      315          && input.equals("*")
      316          && simpleFilters
      317          && CollectionUtils.isEmpty(sortCriteria);
      318    }

      321    LineageSearchResult getLightningSearchResult(
      322        List<LineageRelationship> lineageRelationships,
      323        Filter inputFilters,
      324        int from,
      325        @Nullable Integer size,
      326        Set<String> entityNames) {
      327      size = ConfigUtils.applyLimit(_graphService.getGraphServiceConfig(), size);
      328  
      329      // Construct result objects
      330      LineageSearchResult finalResult =
      331          new LineageSearchResult().setMetadata(new SearchResultMetadata());
      332      LineageSearchEntityArray lineageSearchEntityArray = new LineageSearchEntityArray();
      333      AggregationMetadata entityTypeAgg = constructAggMetadata("Type", "entity");
      334      AggregationMetadata platformTypeAgg = constructAggMetadata("Platform", "platform");
      335      AggregationMetadata environmentAgg = constructAggMetadata("Environment", "origin");
      336  
      337      Map<String, Long> entityTypeAggregations = new HashMap<>();
      338      Map<String, Long> platformTypeAggregations = new HashMap<>();
      339      Map<String, Long> environmentAggregations = new HashMap<>();
      340  
      341      AggregationMetadataArray aggregationMetadataArray = new AggregationMetadataArray();
      342  
      343      // Aggregations supported by this model
      344      // entity type
      345      // platform
      346      // environment
      347      int start = 0;
      348      int numElements = 0;
      349      for (LineageRelationship relnship : lineageRelationships) {
      350        Urn entityUrn = relnship.getEntity();
      351        String entityType = entityUrn.getEntityType();
      352        // Apply platform, entity types, and environment filters
      353  
      354        String platform = getPlatform(entityType, entityUrn);
      355  
      356        String environment = null;
      357        if (entityType.equals(DATASET_ENTITY_NAME)) {
      358          environment = entityUrn.getEntityKey().get(2);
      359        }
      360  
      361        // Filters
      362        Set<String> platformCriteriaValues = null;
      363        Set<String> originCriteriaValues = null;
      364        if (inputFilters != null && inputFilters.getOr() != null) {
      365          platformCriteriaValues =
      366              inputFilters.getOr().stream()
      367                  .map(ConjunctiveCriterion::getAnd)
      368                  .flatMap(CriterionArray::stream)
      369                  .filter(criterion -> "platform".equals(criterion.getField()))
      370                  .flatMap(criterion -> criterion.getValues().stream())
      371                  .collect(Collectors.toSet());
      372          originCriteriaValues =
      373              inputFilters.getOr().stream()
      374                  .map(ConjunctiveCriterion::getAnd)
      375                  .flatMap(CriterionArray::stream)
      376                  .filter(criterion -> "origin".equals(criterion.getField()))
      377                  .flatMap(criterion -> criterion.getValues().stream())
      378                  .collect(Collectors.toSet());
      379        }
      380        boolean isNotFiltered =
      381            (entityNames.isEmpty() || entityNames.contains(entityUrn.getEntityType()))
      382                && (CollectionUtils.isEmpty(platformCriteriaValues)
      383                    || (platform != null && platformCriteriaValues.contains(platform)))
      384                && (CollectionUtils.isEmpty(originCriteriaValues)
      385                    || (environment != null && originCriteriaValues.contains(environment)));
      386  
      387        if (isNotFiltered) {
      388          start++;
      389          if ((start > from) && (numElements < size)) {
      390            lineageSearchEntityArray.add(
      391                new LineageSearchEntity()
      392                    .setEntity(entityUrn)
      393                    .setDegree(relnship.getDegree())
      394                    .setPaths(relnship.getPaths()));
      395            numElements++;
      396          }
      397  
      398          // entityType
      399          entityTypeAggregations.compute(entityType, (key, value) -> value == null ? 1L : ++value);
      400  
      401          // platform
      402          if (platform != null) {
      403            platformTypeAggregations.compute(platform, (key, value) -> value == null ? 1L : ++value);
      404          }
      405  
      406          // environment
      407          if (environment != null) {
      408            environmentAggregations.compute(
      409                environment, (key, value) -> value == null ? 1L : ++value);
      410          }
      411        }
      412      }
      413  
      414      aggregationMetadataArray.add(DEGREE_FILTER_GROUP);
      415      if (platformTypeAggregations.keySet().size() > 0) {
      416        for (Map.Entry<String, Long> platformCount : platformTypeAggregations.entrySet()) {
      417          try {
      418            platformTypeAgg
      419                .getFilterValues()
      420                .add(
      421                    new FilterValue()
      422                        .setValue(platformCount.getKey())
      423                        .setFacetCount(platformCount.getValue())
      424                        .setEntity(Urn.createFromString(platformCount.getKey())));
      425            platformTypeAgg.getAggregations().put(platformCount.getKey(), platformCount.getValue());
      426          } catch (URISyntaxException e) {
      427            log.warn("Unexpected exception: {}", e.getMessage());
      428          }
      429        }
      430        aggregationMetadataArray.add(platformTypeAgg);
      431      }
      432      if (entityTypeAggregations.keySet().size() > 0) {
      433        for (Map.Entry<String, Long> entityCount : entityTypeAggregations.entrySet()) {
      434          entityTypeAgg
      435              .getFilterValues()
      436              .add(
      437                  new FilterValue()
      438                      .setValue(entityCount.getKey())
      439                      .setFacetCount(entityCount.getValue()));
      440          entityTypeAgg.getAggregations().put(entityCount.getKey(), entityCount.getValue());
      441        }
      442        aggregationMetadataArray.add(entityTypeAgg);
      443      }
      444      if (environmentAggregations.keySet().size() > 0) {
      445        for (Map.Entry<String, Long> entityCount : environmentAggregations.entrySet()) {
      446          environmentAgg
      447              .getFilterValues()
      448              .add(
      449                  new FilterValue()
      450                      .setValue(entityCount.getKey())
      451                      .setFacetCount(entityCount.getValue()));
      452          environmentAgg.getAggregations().put(entityCount.getKey(), entityCount.getValue());
      453        }
      454        aggregationMetadataArray.add(environmentAgg);
      455      }
      456      finalResult.setEntities(lineageSearchEntityArray);
      457      finalResult.getMetadata().setAggregations(aggregationMetadataArray);
      458      finalResult.setNumEntities(start);
      459      return finalResult.setFrom(from).setPageSize(size);
