      174      public boolean checkForBatchTooLargeError(Throwable t) {
      175          Throwable currentCause = t;
      176          while (currentCause != null) {
      177              String message = currentCause.getMessage();
      178              if (message != null &&
      179                      message.contains("Batch too large") &&
      180                      currentCause.getClass().getSimpleName().equals("InvalidQueryException")) {
      181                  return true;
      182              }
      183              currentCause = currentCause.getCause();
      184          }
      185          return false;
      186      }
