      368      private void initRmMap() {
      369          if (!model.model.containsKey("g")) {
      370              return;
      371          }
      372  
      373          for (String ptype : model.model.get("g").keySet()) {
      374              if (rmMap.containsKey(ptype)) {
      375                  rmMap.get(ptype).clear();
      376                  continue;
      377              }
      378              Assertion assertion = model.model.get("g").get(ptype);
      379              int token_length = (assertion.tokens != null ? assertion.tokens.length : 0);
      380              int paramsToken_length = (assertion.paramsTokens != null ? assertion.paramsTokens.length : 0);
      381              if (token_length <= 2 && paramsToken_length == 0) {
      382                  assertion.rm = new DomainManager(10);
      383                  rmMap.put(ptype, assertion.rm);
      384              }
      385              if (token_length <= 2 && paramsToken_length != 0) {
      386                  assertion.condRM =new ConditionalRoleManager(10);
      387                  condRmMap.put(ptype, assertion.condRM);
      388                  rmMap.put(ptype, assertion.condRM);
      389              }
      390              if (token_length > 2) {
      391                  if (paramsToken_length == 0) {
      392                      assertion.rm = new DomainManager(10);
      393                      rmMap.put(ptype, assertion.rm);
      394                  } else {
      395                      assertion.condRM = new ConditionalRoleManager(10);
      396                      condRmMap.put(ptype, assertion.condRM);
      397                      rmMap.put(ptype, assertion.condRM);
      398                  }
      399                  String matchFun = "keyMatch(r_dom, p_dom)";
      400                  if (model.model.get("m").get("m").value.contains(matchFun)) {
      401                      addNamedDomainMatchingFunc(ptype, "g", BuiltInFunctions::keyMatch);
      402                  }
      403              }
      404          }
      405      }
