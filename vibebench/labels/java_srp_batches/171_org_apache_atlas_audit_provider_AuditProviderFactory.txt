      301  	private AuditHandler getProviderFromConfig(Properties props,
      302  			String propPrefix, String providerName, AuditHandler consumer) {
      303  		AuditHandler provider = null;
      304  		String className = MiscUtil.getStringProperty(props, propPrefix + "."
      305  				+ BaseAuditHandler.PROP_CLASS_NAME);
      306  		if (className != null && !className.isEmpty()) {
      307  			try {
      308  				Class<?> handlerClass = Class.forName(className);
      309  				if (handlerClass.isAssignableFrom(AuditQueue.class)) {
      310  					// Queue class needs consumer
      311  					handlerClass.getDeclaredConstructor(AuditHandler.class)
      312  							.newInstance(consumer);
      313  				} else {
      314  					provider = (AuditHandler) Class.forName(className)
      315  							.newInstance();
      316  				}
      317  			} catch (Exception e) {
      318  				LOG.fatal("Can't instantiate audit class for providerName="
      319  						+ providerName + ", className=" + className
      320  						+ ", propertyPrefix=" + propPrefix, e);
      321  			}
      322  		} else {
      323  			if (providerName.equalsIgnoreCase("elasticsearch")) {
      324  				provider = new ElasticSearchAuditDestination();
      325  			} else if (providerName.equalsIgnoreCase("log4j")) {
      326  				provider = new Log4JAuditDestination();
      327  			} else if (providerName.equalsIgnoreCase("batch")) {
      328  				provider = getAuditProvider(props, propPrefix, consumer);
      329  			} else if (providerName.equalsIgnoreCase("async")) {
      330  				provider = new AuditAsyncQueue(consumer);
      331  			} else {
      332  				LOG.error("Provider name doesn't have any class associated with it. providerName="
      333  						+ providerName + ", propertyPrefix=" + propPrefix);
      334  			}
      335  		}
      336  		if (provider != null && provider instanceof AuditQueue) {
      337  			if (consumer == null) {
      338  				LOG.fatal("consumer can't be null for AuditQueue. queue="
      339  						+ provider.getName() + ", propertyPrefix=" + propPrefix);
      340  				provider = null;
      341  			}
      342  		}
      343  		return provider;
      344  	}
