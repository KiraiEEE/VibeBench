      103    public void uninstall(String packageName, String version)
      104        throws IOException, SolrServerException {
      105      SolrPackageInstance packageInstance = getPackageInstance(packageName, version);
      106      if (packageInstance == null) {
      107        runtime.printError(
      108            "Package "
      109                + packageName
      110                + ":"
      111                + version
      112                + " doesn't exist. Use the install command to install this package version first.");
      113        runtime.exit(1);
      114      }
      115  
      116      // Make sure that this package instance is not deployed on any collection
      117      Map<String, String> collectionsDeployedOn = getDeployedCollections(packageName);
      118      for (String collection : collectionsDeployedOn.keySet()) {
      119        if (version.equals(collectionsDeployedOn.get(collection))) {
      120          runtime.printError(
      121              "Package "
      122                  + packageName
      123                  + " is currently deployed on collection: "
      124                  + collection
      125                  + ". Undeploy the package with undeploy <package-name> --collections <collection1>[,<collection2>,...] before attempting to uninstall the package.");
      126          runtime.exit(1);
      127        }
      128      }
      129  
      130      // Make sure that no plugin from this package instance has been deployed as cluster level
      131      // plugins
      132      Map<String, SolrPackageInstance> clusterPackages = getPackagesDeployedAsClusterLevelPlugins();
      133      for (String clusterPackageName : clusterPackages.keySet()) {
      134        SolrPackageInstance clusterPackageInstance = clusterPackages.get(clusterPackageName);
      135        if (packageName.equals(clusterPackageName)
      136            && version.equals(clusterPackageInstance.version)) {
      137          runtime.printError(
      138              "Package "
      139                  + packageName
      140                  + "is currently deployed as a cluster-level plugin ("
      141                  + clusterPackageInstance.getCustomData()
      142                  + "). Undeploy the package with undeploy <package-name> --collections <collection1>[,<collection2>,...] before uninstalling the package.");
      143          runtime.exit(1);
      144        }
      145      }
      146  
      147      // Delete the package by calling the Package API and remove the Jar
      148  
      149      runtime.printSuccess("Executing Package API to remove this package...");
      150      try {
      151        new PackageApi.DeletePackageVersion(packageName, version).process(solrClient);
      152        runtime.printSuccess("Package version deleted from Package API.");
      153      } catch (SolrServerException | IOException e) {
      154        throw new SolrException(ErrorCode.BAD_REQUEST, e);
      155      }
      156  
      157      runtime.printSuccess(
      158          "Executing Package Store API to remove the " + packageName + " package...");
      159  
      160      List<String> filesToDelete = new ArrayList<>(packageInstance.files);
      161      filesToDelete.add(
      162          String.format(Locale.ROOT, "/package/%s/%s/%s", packageName, version, "manifest.json"));
      163      for (String filePath : filesToDelete) {
      164        DistribFileStore.deleteZKFileEntry(zkClient, filePath);
      165        runtime.printSuccess("Deleting " + filePath);
      166        new FileStoreApi.DeleteFile(filePath).process(solrClient);
      167      }
      168  
      169      runtime.printSuccess("Package uninstalled: " + packageName + ":" + version + ":-)");
      170    }

      225    public Map<String, SolrPackageInstance> getPackagesDeployed(String collection) {
      226      Map<String, String> packages = null;
      227      try {
      228        NamedList<Object> result =
      229            solrClient.request(
      230                new GenericV2SolrRequest(
      231                        SolrRequest.METHOD.GET,
      232                        PackageUtils.getCollectionParamsPath(collection) + "/PKG_VERSIONS")
      233                    .setRequiresCollection(
      234                        false) /* Making a collection request, but already baked into path */);
      235        packages =
      236            (Map<String, String>)
      237                Objects.requireNonNullElse(result._get("/response/params/PKG_VERSIONS"), Map.of());
      238      } catch (PathNotFoundException ex) {
      239        // Don't worry if PKG_VERSION wasn't found. It just means this collection was never touched by
      240        // the package manager.
      241      } catch (SolrServerException | IOException ex) {
      242        throw new SolrException(ErrorCode.SERVER_ERROR, ex);
      243      }
      244      if (packages == null) return Map.of();
      245      Map<String, SolrPackageInstance> ret = new HashMap<>();
      246      for (String packageName : packages.keySet()) {
      247        if (!StrUtils.isNullOrEmpty(packageName)
      248            && // There can be an empty key, storing the version here
      249            packages.get(packageName)
      250                != null) { // null means the package was undeployed from this package before
      251          ret.put(packageName, getPackageInstance(packageName, packages.get(packageName)));
      252        }
      253      }
      254      return ret;
      255    }

      263    public Map<String, SolrPackageInstance> getPackagesDeployedAsClusterLevelPlugins() {
      264      Map<String, String> packageVersions = new HashMap<>();
      265      // map of package name to multiple values of pluginMeta(Map<String, String>)
      266      Map<String, Set<PluginMeta>> packagePlugins = new HashMap<>();
      267      Object pluginsValue;
      268      try {
      269        NamedList<Object> response =
      270            solrClient.request(
      271                new GenericV2SolrRequest(SolrRequest.METHOD.GET, PackageUtils.CLUSTERPROPS_PATH));
      272        Integer statusCode = (Integer) response._get(List.of("responseHeader", "status"), null);
      273        if (statusCode == null || statusCode == ErrorCode.NOT_FOUND.code) {
      274          // Cluster props doesn't exist, that means there are no cluster level plugins installed.
      275          pluginsValue = null;
      276        } else {
      277          pluginsValue = response.get(ContainerPluginsApi.PLUGIN);
      278        }
      279      } catch (SolrServerException | IOException ex) {
      280        throw new SolrException(ErrorCode.SERVER_ERROR, ex);
      281      }
      282      @SuppressWarnings({"unchecked"})
      283      Map<String, Object> clusterPlugins =
      284          pluginsValue != null ? (Map<String, Object>) pluginsValue : Map.of();
      285      for (Map.Entry<String, Object> entry : clusterPlugins.entrySet()) {
      286        PluginMeta pluginMeta;
      287        try {
      288          pluginMeta =
      289              PackageUtils.getMapper().readValue(Utils.toJSON(entry.getValue()), PluginMeta.class);
      290        } catch (IOException e) {
      291          throw new SolrException(
      292              ErrorCode.SERVER_ERROR,
      293              "Exception while fetching plugins from /clusterprops.json in ZK.",
      294              e);
      295        }
      296        if (pluginMeta.klass.contains(":")) {
      297          String packageName = pluginMeta.klass.substring(0, pluginMeta.klass.indexOf(':'));
      298          packageVersions.put(packageName, pluginMeta.version);
      299          packagePlugins.computeIfAbsent(packageName, k -> new HashSet<>()).add(pluginMeta);
      300        }
      301      }
      302      Map<String, SolrPackageInstance> ret = new HashMap<>();
      303      for (Map.Entry<String, String> entry : packageVersions.entrySet()) {
      304        String packageName = entry.getKey();
      305        String packageVersion = entry.getValue();
      306        // There can be an empty key, storing the version here
      307        // null means the package was undeployed from this package before
      308        if (!StrUtils.isNullOrEmpty(packageName) && packageVersion != null) {
      309          ret.put(packageName, getPackageInstance(packageName, packageVersion));
      310          ret.get(packageName).setCustomData(packagePlugins.get(packageName));
      311        }
      312      }
      313      return ret;
      314    }
