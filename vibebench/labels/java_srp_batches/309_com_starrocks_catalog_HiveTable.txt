      261      public void modifyTableSchema(String dbName, String tableName, HiveTable updatedTable) {
      262          ImmutableList.Builder<Column> fullSchemaTemp = ImmutableList.builder();
      263          ImmutableList.Builder<String> dataColumnNamesTemp = ImmutableList.builder();
      264  
      265          updatedTable.nameToColumn.forEach((colName, column) -> {
      266              Column baseColumn = nameToColumn.get(colName);
      267              if (baseColumn != null) {
      268                  column.setComment(baseColumn.getComment());
      269              }
      270          });
      271  
      272          fullSchemaTemp.addAll(updatedTable.fullSchema);
      273          dataColumnNamesTemp.addAll(updatedTable.dataColumnNames);
      274  
      275          Database db = GlobalStateMgr.getCurrentState().getLocalMetastore().getDb(dbName);
      276          if (db == null) {
      277              throw new StarRocksConnectorException("Not found database " + dbName);
      278          }
      279          Locker locker = new Locker();
      280          locker.lockDatabase(db.getId(), LockType.WRITE);
      281          try {
      282              this.fullSchema.clear();
      283              this.nameToColumn.clear();
      284              this.dataColumnNames.clear();
      285  
      286              this.fullSchema.addAll(fullSchemaTemp.build());
      287              updateSchemaIndex();
      288              this.dataColumnNames.addAll(dataColumnNamesTemp.build());
      289  
      290              if (GlobalStateMgr.getCurrentState().isLeader()) {
      291                  ModifyTableColumnOperationLog log = new ModifyTableColumnOperationLog(dbName, tableName, fullSchema);
      292                  GlobalStateMgr.getCurrentState().getEditLog().logModifyTableColumn(log);
      293              }
      294          } finally {
      295              locker.unLockDatabase(db.getId(), LockType.WRITE);
      296          }
      297      }

      300      public TTableDescriptor toThrift(List<ReferencedPartitionInfo> partitions) {
      301          Preconditions.checkNotNull(partitions);
      302  
      303          THdfsTable tHdfsTable = new THdfsTable();
      304          tHdfsTable.setHdfs_base_dir(tableLocation);
      305  
      306          // columns and partition columns
      307          Set<String> partitionColumnNames = Sets.newHashSet();
      308          List<TColumn> tPartitionColumns = Lists.newArrayList();
      309          List<TColumn> tColumns = Lists.newArrayList();
      310  
      311          for (Column column : getPartitionColumns()) {
      312              tPartitionColumns.add(column.toThrift());
      313              partitionColumnNames.add(column.getName());
      314          }
      315          for (Column column : getBaseSchema()) {
      316              if (partitionColumnNames.contains(column.getName())) {
      317                  continue;
      318              }
      319              tColumns.add(column.toThrift());
      320          }
      321          tHdfsTable.setColumns(tColumns);
      322          if (!tPartitionColumns.isEmpty()) {
      323              tHdfsTable.setPartition_columns(tPartitionColumns);
      324          }
      325  
      326          // partitions
      327          List<String> partitionNames = Lists.newArrayList();
      328          for (ReferencedPartitionInfo partition : partitions) {
      329              partitionNames.add(PartitionUtil.toHivePartitionName(getPartitionColumnNames(), partition.getKey()));
      330          }
      331          List<PartitionInfo> hivePartitions;
      332          try {
      333              hivePartitions = GlobalStateMgr.getCurrentState().getMetadataMgr()
      334                      .getPartitions(this.getCatalogName(), this, partitionNames);
      335          } catch (StarRocksConnectorException e) {
      336              LOG.warn("table {} gets partition info failed.", name, e);
      337              return null;
      338          }
      339  
      340          for (int i = 0; i < hivePartitions.size(); i++) {
      341              ReferencedPartitionInfo info = partitions.get(i);
      342              PartitionKey key = info.getKey();
      343              long partitionId = info.getId();
      344  
      345              THdfsPartition tPartition = new THdfsPartition();
      346              tPartition.setFile_format(hivePartitions.get(i).getFileFormat().toThrift());
      347  
      348              List<LiteralExpr> keys = key.getKeys();
      349              tPartition.setPartition_key_exprs(keys.stream().map(Expr::treeToThrift).collect(Collectors.toList()));
      350  
      351              THdfsPartitionLocation tPartitionLocation = new THdfsPartitionLocation();
      352              tPartitionLocation.setPrefix_index(-1);
      353              tPartitionLocation.setSuffix(hivePartitions.get(i).getFullPath());
      354              tPartition.setLocation(tPartitionLocation);
      355              tHdfsTable.putToPartitions(partitionId, tPartition);
      356          }
      357  
      358          tHdfsTable.setSerde_lib(hiveProperties.get(HIVE_TABLE_SERDE_LIB));
      359          tHdfsTable.setInput_format(hiveProperties.get(HIVE_TABLE_INPUT_FORMAT));
      360          tHdfsTable.setHive_column_names(hiveProperties.get(HIVE_TABLE_COLUMN_NAMES));
      361          tHdfsTable.setHive_column_types(hiveProperties.get(HIVE_TABLE_COLUMN_TYPES));
      362          tHdfsTable.setSerde_properties(serdeProperties);
      363          tHdfsTable.setTime_zone(TimeUtils.getSessionTimeZone());
      364  
      365          TTableDescriptor tTableDescriptor = new TTableDescriptor(id, TTableType.HDFS_TABLE, fullSchema.size(),
      366                  0, hiveTableName, hiveDbName);

      368          return tTableDescriptor;
