      364      private <T, R> Page<R> findPage(Session session, PagedQuery<T> pagedQuery) {
      365          if (pagedQuery instanceof PreparedQuery<?, ?> pq) {
      366              PreparedQuery<T, R> preparedQuery = (PreparedQuery<T, R>) pq;
      367              Pageable pageable = preparedQuery.getPageable();
      368              if (pageable.getMode() != Pageable.Mode.OFFSET) {
      369                  KeyedResultList<R> keyedResultList = getKeyedResult(preparedQuery, session, pageable);
      370                  List<Pageable.Cursor> cursors =
      371                      keyedResultList.getKeyList()
      372                          .stream()
      373                          .map(key -> Pageable.Cursor.of(key.toArray()))
      374                          .collect(toList());
      375                  return CursoredPage.of(keyedResultList.getResultList(), pageable, cursors, -1L);
      376              }
      377              ListResultCollector<R> resultCollector = new ListResultCollector<>();
      378              collectFindAll(session, preparedQuery, resultCollector);
      379              return Page.of(resultCollector.result, pageable, -1L);
      380          }
      381          ListResultCollector<T> collector = new ListResultCollector<>();
      382          collectPagedResults(sessionFactory.getCriteriaBuilder(), session, pagedQuery, collector);
      383          return Page.of((List<R>) collector.result, pagedQuery.getPageable(), -1L);
      384      }

      415      private static <T, R> KeyedPage<T> getKeyedPage(PreparedQuery<T, R> preparedQuery, Pageable pageable) {
      416          CursoredPageable cursoredPageable = (CursoredPageable) pageable;
      417          var unkeyedPage =
      418              page(pageable.getSize(), pageable.getNumber())
      419                  .keyedBy(getOrders(preparedQuery.getSort(), preparedQuery.getRootEntity()));
      420          return cursoredPageable.cursor()
      421                  .map(_cursor -> {
      422                      List<?> els = _cursor.elements();
      423                      var elements = (List<Comparable<?>>) els;
      424                      return switch (pageable.getMode()) {
      425                          case CURSOR_NEXT -> unkeyedPage.withKey(elements, KEY_OF_LAST_ON_PREVIOUS_PAGE);
      426                          case CURSOR_PREVIOUS -> unkeyedPage.withKey(elements, KEY_OF_FIRST_ON_NEXT_PAGE);
      427                          default -> unkeyedPage;
      428                      };
      429                  }).orElse(unkeyedPage);
      430      }

      432      @Override
      433      public <T> T persist(@NonNull InsertOperation<T> operation) {
      434          StoredQuery<T, ?> storedQuery = operation.getStoredQuery();
      435          return executeWrite(session -> {
      436              if (storedQuery != null) {
      437                  return executeUpdate(operation, session, storedQuery);
      438              }
      439              T entity = operation.getEntity();
      440              if (persistOrMergeOnSave) {
      441                  RuntimePersistentEntity<T> persistentEntity = getEntity(operation.getRootEntity());
      442                  RuntimePersistentProperty<T> identity = persistentEntity.getIdentity();
      443                  if (identity != null && identity.getProperty().get(entity) == null) {
      444                      session.persist(entity);
      445                  } else {
      446                      entity = session.merge(entity);
      447                  }
      448              } else {
      449                  session.persist(entity);
      450              }
      451              flushIfNecessary(session, operation.getAnnotationMetadata());
      452              return entity;
      453          });
      454      }

      497      private <T> BatchOperation<T> executeUpdate(BatchOperation<T> operation, Session session, StoredQuery<T, ?> storedQuery) {
      498          for (T entity : operation) {
      499              executeUpdate(session, storedQuery, operation.getInvocationContext(), entity);
      500          }
      501          if (flushIfNecessary(session, operation.getAnnotationMetadata())) {
      502              for (T entity : operation) {
      503                  session.remove(entity);
      504              }
      505          }
      506          return operation;
