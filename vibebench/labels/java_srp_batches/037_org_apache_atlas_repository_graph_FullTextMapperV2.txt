       85          String                       ret     = null;
       86          final AtlasEntityWithExtInfo entityWithExtInfo;
       87  
       88          if (followReferences) {
       89              entityWithExtInfo = getAndCacheEntityWithExtInfo(guid);
       90          } else {
       91              AtlasEntity entity = getAndCacheEntity(guid);
       92  
       93              entityWithExtInfo = entity != null ? new AtlasEntityWithExtInfo(entity) : null;
       94          }
       95  
       96          if (entityWithExtInfo != null) {
       97              StringBuilder sb = new StringBuilder();
       98  
       99              if (CollectionUtils.isNotEmpty(classifications)) {
      100                  for (AtlasClassification classification : classifications) {
      101                      final AtlasClassificationType classificationType = typeRegistry.getClassificationTypeByName(classification.getTypeName());
      102                      final Set<String>             excludeAttributes  = getExcludeAttributesForIndexText(classification.getTypeName());
      103  
      104                      sb.append(classification.getTypeName()).append(FULL_TEXT_DELIMITER);
      105  
      106                      mapAttributes(classificationType, classification.getAttributes(), entityWithExtInfo, sb, new HashSet<String>(), excludeAttributes, false); //false because of full text context.
      107                  }
      108              }
      109  
      110              ret = sb.toString();
      111          }
      112  
      113          if (LOG.isDebugEnabled()) {

      167      private void map(AtlasEntity entity, AtlasEntityExtInfo entityExtInfo, StringBuilder sb, Set<String> processedGuids, boolean isClassificationOnly) throws AtlasBaseException {
      168          if (entity == null || processedGuids.contains(entity.getGuid())) {
      169              return;
      170          }
      171  
      172          final AtlasEntityType entityType        = typeRegistry.getEntityTypeByName(entity.getTypeName());
      173          final Set<String>     excludeAttributes = getExcludeAttributesForIndexText(entity.getTypeName());
      174  
      175          processedGuids.add(entity.getGuid());
      176          if(!isClassificationOnly) {
      177              sb.append(entity.getTypeName()).append(FULL_TEXT_DELIMITER);
      178  
      179              mapAttributes(entityType, entity.getAttributes(), entityExtInfo, sb, processedGuids, excludeAttributes, isClassificationOnly);
      180          }
      181  
      182          final List<AtlasClassification> classifications = entity.getClassifications();
      183          if (CollectionUtils.isNotEmpty(classifications)) {
      184              for (AtlasClassification classification : classifications) {
      185                  final AtlasClassificationType classificationType              = typeRegistry.getClassificationTypeByName(classification.getTypeName());
      186                  final Set<String>             excludeClassificationAttributes = getExcludeAttributesForIndexText(classification.getTypeName());
      187  
      188  
      189                  sb.append(classification.getTypeName()).append(FULL_TEXT_DELIMITER);
      190  
      191                  mapAttributes(classificationType, classification.getAttributes(), entityExtInfo, sb, processedGuids, excludeClassificationAttributes, isClassificationOnly);
      192              }
      193          }
      194      }

      196      private List<Object> rearrangeAttributesOrder(List<Object> keys) {
      197          if (keys != null && keys.contains(SOURCE_TAG_QN)) {
      198              keys.remove(SOURCE_TAG_QN);
      199              keys.add(SOURCE_TAG_QN);
      200          }
      201          return keys;
      202      }

      204      @Override
      205      public void mapAttributes(AtlasStructType structType, Map<String, Object> attributes, AtlasEntityExtInfo entityExtInfo, StringBuilder sb,
      206                                 Set<String> processedGuids, Set<String> excludeAttributes, boolean isClassificationOnly) throws AtlasBaseException {
      207          if (MapUtils.isEmpty(attributes)) {
      208              return;
      209          }
      210  
      211          for (Map.Entry<String, Object> attributeEntry : attributes.entrySet()) {
      212              String attribKey = attributeEntry.getKey();
      213              Object attrValue = attributeEntry.getValue();
      214  
      215              if (attrValue == null || excludeAttributes.contains(attribKey)) {
      216                  continue;
      217              }
      218  
      219              if (!followReferences) {
      220                  AtlasAttribute attribute     = structType != null ? structType.getAttribute(attribKey) : null;
      221                  AtlasType      attributeType = attribute != null ? attribute.getAttributeType() : null;
      222  
      223                  if (attributeType == null) {
      224                      continue;
      225                  }
      226  
      227                  if (attributeType instanceof AtlasArrayType) {
      228                      attributeType = ((AtlasArrayType) attributeType).getElementType();
      229                  }
      230  
      231                  if (attributeType instanceof AtlasEntityType || attributeType instanceof AtlasBuiltInTypes.AtlasObjectIdType) {
      232                      continue;
      233                  }
      234              }
      235  
      236  
      237              sb.append(attribKey).append(FULL_TEXT_DELIMITER);
      238  
      239              mapAttribute(attrValue, entityExtInfo, sb, processedGuids, isClassificationOnly);
      240          }
      241      }

      243      private void mapAttribute(Object value, AtlasEntityExtInfo entityExtInfo, StringBuilder sb, Set<String> processedGuids, boolean isClassificationOnly) throws AtlasBaseException {
      244          if (value instanceof AtlasObjectId) {
      245              if (followReferences && entityExtInfo != null) {
      246                  AtlasObjectId objectId = (AtlasObjectId) value;
      247                  AtlasEntity   entity   = entityExtInfo.getEntity(objectId.getGuid());
      248  
      249                  if (entity != null) {
      250                      map(entity, entityExtInfo, sb, processedGuids, isClassificationOnly);
      251                  }
      252              }
      253          } else if (value instanceof List) {
      254              List<?> valueList = (List<?>) value;
      255  
      256              for (Object listElement : valueList) {
      257                  mapAttribute(listElement, entityExtInfo, sb, processedGuids, isClassificationOnly);
      258              }
      259          } else if (value instanceof Map) {
      260              Map<?, ?> valueMap = (Map<?, ?>) value;
      261              List<Object> keys = new ArrayList<>(valueMap.keySet());
      262  
      263              // Only rearrange if the key exists
      264              rearrangeAttributesOrder(keys);
      265  
      266              for (Object key : keys) {
      267                  mapAttribute(key, entityExtInfo, sb, processedGuids, isClassificationOnly);
      268                  mapAttribute(valueMap.get(key), entityExtInfo, sb, processedGuids, isClassificationOnly);
      269              }
      270          } else if (value instanceof Enum) {
      271              Enum<?> enumValue = (Enum<?>) value;
      272  
      273              sb.append(enumValue.name()).append(FULL_TEXT_DELIMITER);
      274          } else if (value instanceof AtlasStruct) {
      275              AtlasStruct atlasStruct = (AtlasStruct) value;
      276  
      277              for (Map.Entry<String, Object> entry : atlasStruct.getAttributes().entrySet()) {
      278                  sb.append(entry.getKey()).append(FULL_TEXT_DELIMITER);
      279                  mapAttribute(entry.getValue(), entityExtInfo, sb, processedGuids, isClassificationOnly);
      280              }
      281          } else {
      282              sb.append(String.valueOf(value)).append(FULL_TEXT_DELIMITER);
      283          }
      284      }

      292      public AtlasEntity  getAndCacheEntity(String guid, boolean includeReferences) throws AtlasBaseException {
      293          RequestContext context = RequestContext.get();
      294          AtlasEntity    entity  = context.getEntity(guid);
      295  
      296          if (entity == null) {
      297              entity = entityGraphRetriever.toAtlasEntity(guid, includeReferences);
      298  
      299              if (entity != null) {
      300                  context.cache(entity);
      301  
      302                  if (LOG.isDebugEnabled()) {
      303                      LOG.debug("Cache miss -> GUID = {}", guid);
      304                  }
      305              }
      306          }
      307

      331      private Set<String> getExcludeAttributesForIndexText(String typeName) {
      332          final Set<String> ret;
      333  
      334          if (excludeAttributesCache.containsKey(typeName)) {
      335              ret = excludeAttributesCache.get(typeName);
      336  
      337          } else if (configuration != null) {
      338              String[] excludeAttributes = configuration.getStringArray(FULL_TEXT_EXCLUDE_ATTRIBUTE_PROPERTY + "." +
      339                                                                                typeName + "." + "attributes.exclude");
      340  
      341              if (ArrayUtils.isNotEmpty(excludeAttributes)) {
      342                  ret = new HashSet<>(Arrays.asList(excludeAttributes));
      343              } else {
      344                  ret = Collections.emptySet();
      345              }
      346  
      347              excludeAttributesCache.put(typeName, ret);
      348          } else {
      349              ret = Collections.emptySet();
      350          }
      351  
      352          return ret;
      353      }
