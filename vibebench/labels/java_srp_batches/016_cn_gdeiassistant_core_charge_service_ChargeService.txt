       59      public ChargeVO ChargeRequest(String sessionId, int amount) throws Exception {
       60          UserCertificateEntity userCertificate = userCertificateService.getUserSessionCertificate(sessionId);
       61          CloseableHttpClient httpClient = null;
       62          CookieStore cookieStore = null;
       63          try {
       64              if (amount <= 0 || amount > 500) {
       65                  throw new AmountNotAvailableException("充值金额超过范围");
       66              }
       67              HttpClientSession httpClientSession = HttpClientUtils.getHttpClient(sessionId, false, 15);
       68              httpClient = httpClientSession.getCloseableHttpClient();
       69              cookieStore = httpClientSession.getCookieStore();
       70              //登录支付管理平台
       71              LoginCardSystem(httpClient, userCertificate.getUser().getUsername()
       72                      , userCertificate.getUser().getPassword(), false);
       73              //发送充值请求
       74              Map<String, String> ecardDataMap = SendChargeRequest(httpClient, amount);
       75              //确认充值请求
       76              Charge charge = ConfirmChargeRequest(httpClient, cookieStore, ecardDataMap);
       77              ChargeVO vo = new ChargeVO();
       78              vo.setAlipayURL(charge.getAlipayURL());
       79              vo.setCookieList(charge.getCookieList());
       80              return vo;
       81          } catch (AmountNotAvailableException e) {
       82              logger.error("校园卡充值异常：", e);
       83              throw new AmountNotAvailableException("用户充值金额超过范围");
       84          } catch (IOException e) {
       85              logger.error("校园卡充值异常：", e);
       86              throw new NetWorkTimeoutException("网络连接超时");
       87          } catch (Exception e) {
       88              logger.error("校园卡充值异常：", e);
       89              throw new ServerErrorException("支付管理系统异常");
       90          } finally {
       91              if (httpClient != null) {
       92                  try {
       93                      httpClient.close();
       94                  } catch (IOException e) {
       95                      logger.error("关闭校园卡充值 HttpClient 失败", e);
       96                  }
       97              }
       98              if (cookieStore != null) {
       99                  HttpClientUtils.syncHttpClientCookieStore(sessionId, cookieStore);
      100              }
      101          }
      102      }

      342      private Charge ConfirmChargeRequest(CloseableHttpClient httpClient, CookieStore cookieStore
      343              , Map<String, String> ecardDataMap) throws Exception {
      344          Charge charge = new Charge();
      345          List<BasicNameValuePair> basicNameValuePairs = new ArrayList<>();
      346          for (Map.Entry<String, String> entry : ecardDataMap.entrySet()) {
      347              basicNameValuePairs.add(new BasicNameValuePair(entry.getKey(), entry.getValue()));
      348          }
      349          HttpPost httpPost = new HttpPost("https://epay.gdei.edu.cn:8443/synpay/web/forwardPayTool");
      350          //绑定表单参数
      351          httpPost.setEntity(new UrlEncodedFormEntity(basicNameValuePairs, StandardCharsets.UTF_8));
      352          HttpResponse httpResponse = httpClient.execute(httpPost);
      353          Document document = Jsoup.parse(EntityUtils.toString(httpResponse.getEntity()));
      354          if (httpResponse.getStatusLine().getStatusCode() == 200) {
      355              Elements inputs = document.getElementsByTag("input");
      356              List<BasicNameValuePair> mapiDataList = new ArrayList<>();
      357              for (Element input : inputs) {
      358                  String inputName = input.attr("name");
      359                  String inputValue = input.attr("value");
      360                  mapiDataList.add(new BasicNameValuePair(inputName, inputValue));
      361              }
      362              httpPost = new HttpPost("https://mapi.alipay.com/gateway.do?_input_charset=utf-8");
      363              //绑定表单参数
      364              httpPost.setEntity(new UrlEncodedFormEntity(mapiDataList, StandardCharsets.UTF_8));
      365              //设置头信息
      366              httpPost.setHeader("Referer", "https://epay.gdei.edu.cn:8443/synpay/web/forwardPayTool");
      367              httpPost.setHeader("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/71.0.3578.98 Safari/537.36");
      368              httpPost.setHeader("Upgrade-Insecure-Requests", "1");
      369              httpResponse = httpClient.execute(httpPost);
      370              if (httpResponse.getStatusLine().getStatusCode() == 302) {
      371                  HttpGet httpGet = new HttpGet(httpResponse.getFirstHeader("Location").getValue());
      372                  httpResponse = httpClient.execute(httpGet);
      373                  if (httpResponse.getStatusLine().getStatusCode() == 302) {
      374                      //获取支付宝充值接口URL地址
      375                      String url = httpResponse.getFirstHeader("Location").getValue();
      376                      httpGet = new HttpGet(url);

      379                          //获取Cookies

      385                          charge.setCookieList(cookieList);
