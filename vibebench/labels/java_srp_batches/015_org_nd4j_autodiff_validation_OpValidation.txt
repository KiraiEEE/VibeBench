      405       * Validate the outputs of a single op

      407       * @param testCase Op test case to run
      408       * @return NULL if test is OK, or an error message otherwise

      410      public static String validate(OpTestCase testCase) {
      411          collectCoverageInformation(testCase);
      412  
      413          //Check shape function:
      414          List<DataBuffer> outShapes;
      415          try {
      416              outShapes = Nd4j.getExecutioner().calculateOutputShape(testCase.op());
      417          } catch (Throwable t) {
      418              throw new IllegalStateException("Error calculating output shapes during op validation", t);
      419          }
      420  
      421          if (outShapes.size() != testCase.testFns().size()) {
      422              return "Expected number of output shapes and number of outputs differ. " + outShapes.size() + " output shapes," +
      423                      " but OpTestCase specifies " + testCase.testFns().size() + " outputs expected";
      424          }
      425  
      426          for (int i = 0; i < outShapes.size(); i++) {
      427              val act = outShapes.get(i);
      428              val exp = testCase.expShapes().get(i);
      429              if(!Objects.equals(exp.dataType(), act.dataType())) {
      430                  return "Shape function check failed for output " + i + ": expected shape " + exp + ", actual shape " + act;
      431              }
      432              long[] shapeInfo = act.asLong();
      433              long[] actShape = new long[(int) shapeInfo[0]];
      434              for(int j = 1; j < shapeInfo.length; j++) {
      435                  actShape[j - 1] = shapeInfo[j];
      436              }
      437              if(!Arrays.equals(actShape, exp.getShape())) {
      438                  return "Shape function check failed for output " + i + ": expected shape " + exp + ", actual shape " + act;
      439              }
      440          }
      441  
      442          //Check the outputs:
      443          try {
      444              Nd4j.getExecutioner().execAndReturn(testCase.op());
      445          } catch (Throwable t) {
      446              throw new IllegalStateException("Error during op execution", t);
      447          }
      448  
      449          for (int i = 0; i < testCase.testFns().size(); i++) {
      450              String error;
      451              try {
      452                  error = testCase.testFns().get(i).apply(testCase.op().outputArguments().get(i));
      453              } catch (Throwable t) {
      454                  throw new IllegalStateException("Exception thrown during op output validation for output " + i, t);
      455              }
      456  
      457              if (error != null) {
      458                  return "Output " + i + " failed: " + error;
      459              }
      460          }
      461  
      462          return null;    //OK
      463      }
