      115      public boolean syncAddOrDropPartitions() throws LockTimeoutException {
      116          // collect mv partition items with lock
      117          Locker locker = new Locker();
      118          if (!locker.tryLockTableWithIntensiveDbLock(db.getId(), mv.getId(),
      119                  LockType.READ, Config.mv_refresh_try_lock_timeout_ms, TimeUnit.MILLISECONDS)) {
      120              throw new LockTimeoutException("Failed to lock database: " + db.getFullName() + " in syncPartitionsForList");
      121          }
      122  
      123          PartitionDiffResult result;
      124          try {
      125              result = differ.computePartitionDiff(null);
      126              if (result == null) {
      127                  logger.warn("compute list partition diff failed, result is null");
      128                  return false;
      129              }
      130          } finally {
      131              locker.unLockTableWithIntensiveDbLock(db.getId(), mv.getId(), LockType.READ);
      132          }
      133  
      134          // drop old partitions and add new partitions
      135          final PartitionDiff partitionDiff = result.diff;
      136          // We should delete the old partition first and then add the new one,
      137          // because the old and new partitions may overlap
      138          final Map<String, PCell> deletes = partitionDiff.getDeletes();
      139          for (String mvPartitionName : deletes.keySet()) {
      140              dropPartition(db, mv, mvPartitionName);
      141          }
      142          logger.info("The process of synchronizing materialized view [{}] delete partitions list [{}]",
      143                  mv.getName(), deletes);
      144  
      145          // add partitions
      146          final Map<String, String> partitionProperties = MvUtils.getPartitionProperties(mv);
      147          final DistributionDesc distributionDesc = MvUtils.getDistributionDesc(mv);
      148          final Map<String, PCell> adds = partitionDiff.getAdds();
      149          // filter by partition ttl
      150          filterPartitionsByTTL(adds, true);
      151          // add partitions for mv
      152          addListPartitions(db, mv, adds, partitionProperties, distributionDesc);
      153          logger.info("The process of synchronizing materialized view [{}] add partitions list [{}]",
      154                  mv.getName(), adds);
      155  
      156          // add into mv context
      157          result.mvPartitionToCells.putAll(adds);
      158          final Map<Table, Map<String, PCell>> refBaseTablePartitionMap = result.refBaseTablePartitionMap;
      159          // base table -> Map<partition name -> mv partition names>
      160          final Map<Table, Map<String, Set<String>>> baseToMvNameRef =
      161                  differ.generateBaseRefMap(refBaseTablePartitionMap, result.mvPartitionToCells);
      162          // mv partition name -> Map<base table -> base partition names>
      163          final Map<String, Map<Table, Set<String>>> mvToBaseNameRef =
      164                  differ.generateMvRefMap(result.mvPartitionToCells, refBaseTablePartitionMap);
      165          mvContext.setMVToCellMap(result.mvPartitionToCells);
      166          mvContext.setRefBaseTableMVIntersectedPartitions(baseToMvNameRef);
      167          mvContext.setMvRefBaseTableIntersectedPartitions(mvToBaseNameRef);
      168          mvContext.setRefBaseTableToCellMap(refBaseTablePartitionMap);
      169          mvContext.setExternalRefBaseTableMVPartitionMap(result.getRefBaseTableMVPartitionMap());
      170          return true;
      171      }

      174      public Expr generatePartitionPredicate(Table refBaseTable, Set<String> refBaseTablePartitionNames,
      175                                             List<Expr> mvPartitionSlotRefs) throws AnalysisException {
      176          Map<Table, Map<String, PCell>> basePartitionMaps = mvContext.getRefBaseTableToCellMap();
      177          if (basePartitionMaps.isEmpty()) {
      178              return null;
      179          }
      180          Map<String, PCell> baseListPartitionMap = basePartitionMaps.get(refBaseTable);
      181          if (baseListPartitionMap == null) {
      182              logger.warn("Generate incremental partition predicate failed, " +
      183                      "basePartitionMaps:{} contains no refBaseTable:{}", basePartitionMaps, refBaseTable);
      184              return null;
      185          }
      186          if (baseListPartitionMap.isEmpty()) {
      187              return new BoolLiteral(true);
      188          }
      189  
      190          // base table's partition columns, not mv's partition columns
      191          Map<Table, List<Column>> refBaseTablePartitionColumns = mv.getRefBaseTablePartitionColumns();
      192          if (refBaseTablePartitionColumns == null || !refBaseTablePartitionColumns.containsKey(refBaseTable)) {
      193              logger.warn("Generate incremental partition failed, partitionTableAndColumn {} contains no ref table {}",
      194                      refBaseTablePartitionColumns, refBaseTable);
      195              return null;
      196          }
      197          // base table's partition columns that are referred by mv
      198          List<Column> refPartitionColumns = refBaseTablePartitionColumns.get(refBaseTable);
      199          if (refPartitionColumns.size() != mvPartitionSlotRefs.size()) {
      200              logger.warn("Generate incremental partition failed, refPartitionColumns size {} != mvPartitionSlotRefs size {}",
      201                      refPartitionColumns.size(), mvPartitionSlotRefs.size());
      202              return null;
      203          }
      204          return genPartitionPredicate(refBaseTable, refBaseTablePartitionNames, mvPartitionSlotRefs,
      205                  refPartitionColumns, baseListPartitionMap);
      206      }

      209      public Expr generateMVPartitionPredicate(TableName tableName,
      210                                               Set<String> mvPartitionNames) throws AnalysisException {
      211          Map<String, PCell> mvToCellMap = mvContext.getMVToCellMap();
      212          if (mvToCellMap.isEmpty()) {
      213              return new BoolLiteral(true);
      214          }
      215          PartitionInfo partitionInfo = mv.getPartitionInfo();
      216          Preconditions.checkArgument(partitionInfo instanceof ListPartitionInfo);
      217  
      218          ListPartitionInfo listPartitionInfo = (ListPartitionInfo) partitionInfo;
      219          List<Expr> mvPartitionExprs = listPartitionInfo.getPartitionExprs(tableName, mv.getIdToColumn());
      220          List<Column> mvPartitionCols = mv.getPartitionColumns();
      221          Preconditions.checkArgument(mvPartitionExprs.size() == mvPartitionCols.size());
      222          if (mvPartitionCols.size() == 1) {
      223              boolean isContainsNullPartition = false;
      224              Column refPartitionColumn = mvPartitionCols.get(0);
      225              List<Expr> selectedPartitionValues = Lists.newArrayList();
      226              Type partitionType = refPartitionColumn.getType();
      227              for (String tablePartitionName : mvPartitionNames) {
      228                  PListCell cell = (PListCell) mvToCellMap.get(tablePartitionName);
      229                  for (List<String> values : cell.getPartitionItems()) {
      230                      if (mvPartitionCols.size() != values.size()) {
      231                          return null;
      232                      }
      233                      LiteralExpr partitionValue = new PartitionValue(values.get(0)).getValue(partitionType);
      234                      if (partitionValue.isConstantNull()) {
      235                          isContainsNullPartition = true;
      236                          continue;
      237                      }
      238                      selectedPartitionValues.add(partitionValue);
      239                  }
      240              }
      241              Expr mvPartitionExpr = mvPartitionExprs.get(0);
      242              Expr inPredicate = MvUtils.convertToInPredicate(mvPartitionExpr, selectedPartitionValues);
      243              // NOTE: If target partition values contain `null partition`, the generated predicate should
      244              // contain `is null` predicate rather than `in (null) or = null` because the later one is not correct.
      245              if (isContainsNullPartition) {
      246                  IsNullPredicate isNullPredicate = new IsNullPredicate(mvPartitionExpr, false);
      247                  return Expr.compoundOr(Lists.newArrayList(inPredicate, isNullPredicate));
      248              } else {
      249                  return inPredicate;
      250              }
      251          } else {
      252              List<Expr> partitionPredicates = Lists.newArrayList();
      253              for (String tablePartitionName : mvPartitionNames) {
      254                  PListCell cell = (PListCell) mvToCellMap.get(tablePartitionName);
      255                  for (List<String> values : cell.getPartitionItems()) {
      256                      if (mvPartitionCols.size() != values.size()) {
      257                          return null;
      258                      }
      259                      List<Expr> predicates = Lists.newArrayList();
      260                      for (int i = 0; i < mvPartitionCols.size(); i++) {
      261                          Column refPartitionColumn = mvPartitionCols.get(i);
      262                          Type partitionType = refPartitionColumn.getType();
      263                          LiteralExpr partitionValue = new PartitionValue(values.get(i)).getValue(partitionType);
      264                          Expr mvPartitionByExpr = mvPartitionExprs.get(i);
      265                          Expr predicate;
      266                          if (partitionValue.isConstantNull()) {
      267                              // NOTE: If target partition values contain `null partition`, the generated predicate should
      268                              // contain `is null` predicate rather than `in (null) or = null` because the later one is not correct.
      269                              predicate = new IsNullPredicate(mvPartitionByExpr, false);
      270                          } else {
      271                              predicate = MvUtils.convertToInPredicate(mvPartitionByExpr,
      272                                      Lists.newArrayList(partitionValue));
      273                          }
      274                          predicates.add(predicate);
      275                      }
      276                      partitionPredicates.add(Expr.compoundAnd(predicates));
      277                  }
      278              }
      279              return Expr.compoundOr(partitionPredicates);

      390                          mvToRefreshPartitionNames);
      391                  return mvToRefreshPartitionNames;
      392              }
      393              // because the relation of partitions between materialized view and base partition table is n : m,
      394              // should calculate the candidate partitions recursively.

      399                          mvContext.getRefBaseTableMVIntersectedPartitions(),

      401                          mvPotentialPartitionNames);

      419      public Set<String> getMVPartitionNamesWithTTL(MaterializedView mv,
      420                                                    MVRefreshParams mvRefreshParams,
      421                                                    boolean isAutoRefresh) {
      422          // if the user specifies the start and end ranges, only refresh the specified partitions
      423          boolean isCompleteRefresh = mvRefreshParams.isCompleteRefresh();
      424          Map<String, PCell> mvListPartitionMap = Maps.newHashMap();
      425          if (!isCompleteRefresh) {
      426              Set<PListCell> pListCells = mvRefreshParams.getListValues();
      427              Map<String, PListCell> mvPartitions = mv.getListPartitionItems();
      428              for (Map.Entry<String, PListCell> e : mvPartitions.entrySet()) {
      429                  PListCell mvListCell = e.getValue();
      430                  if (mvListCell.getItemSize() == 1) {
      431                      // if list value is a single value, check it directly
      432                      if (pListCells.contains(e.getValue()))  {
      433                          mvListPartitionMap.put(e.getKey(), e.getValue());
      434                      }
      435                  } else {
      436                      // if list values is multi values, split it into single values and check it then.
      437                      if (mvListCell.toSingleValueCells().stream().anyMatch(i -> pListCells.contains(i))) {
      438                          mvListPartitionMap.put(e.getKey(), e.getValue());
      439                      }
      440                  }
      441              }
      442          } else {
      443              mvListPartitionMap = mv.getPartitionCells(Optional.empty());
      444          }
      445          // filter all valid partitions by partition_retention_condition
      446          filterPartitionsByTTL(mvListPartitionMap, false);
      447          return mvListPartitionMap.keySet();
