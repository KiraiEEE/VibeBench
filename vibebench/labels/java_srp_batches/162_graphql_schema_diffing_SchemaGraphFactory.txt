       40      public SchemaGraph createGraph(GraphQLSchema schema) {
       41          Set<GraphQLSchemaElement> roots = new LinkedHashSet<>();
       42          roots.add(schema.getQueryType());
       43          if (schema.isSupportingMutations()) {
       44              roots.add(schema.getMutationType());
       45          }
       46          if (schema.isSupportingSubscriptions()) {
       47              roots.add(schema.getSubscriptionType());
       48          }
       49          roots.addAll(schema.getAdditionalTypes());
       50          roots.addAll(schema.getDirectives());
       51          roots.addAll(schema.getSchemaDirectives());
       52          roots.add(schema.getIntrospectionSchemaType());
       53          Traverser<GraphQLSchemaElement> traverser = Traverser.depthFirst(GraphQLSchemaElement::getChildren);
       54          SchemaGraph schemaGraph = new SchemaGraph();
       55          class IntrospectionNode {
       56  
       57          }
       58          traverser.traverse(roots, new TraverserVisitor<GraphQLSchemaElement>() {
       59              @Override
       60              public TraversalControl enter(TraverserContext<GraphQLSchemaElement> context) {
       61                  boolean isIntrospectionNode = false;
       62                  if (context.thisNode() instanceof GraphQLNamedType) {
       63                      if (Introspection.isIntrospectionTypes((GraphQLNamedType) context.thisNode())) {
       64                          isIntrospectionNode = true;
       65                          context.setVar(IntrospectionNode.class, new IntrospectionNode());
       66                      }
       67                  } else {
       68                      isIntrospectionNode = context.getVarFromParents(IntrospectionNode.class) != null;
       69                  }
       70                  if (context.thisNode() instanceof GraphQLObjectType) {
       71                      newObject((GraphQLObjectType) context.thisNode(), schemaGraph, isIntrospectionNode);
       72                  }
       73                  if (context.thisNode() instanceof GraphQLInterfaceType) {
       74                      newInterface((GraphQLInterfaceType) context.thisNode(), schemaGraph, isIntrospectionNode);
       75                  }
       76                  if (context.thisNode() instanceof GraphQLUnionType) {
       77                      newUnion((GraphQLUnionType) context.thisNode(), schemaGraph, isIntrospectionNode);
       78                  }
       79                  if (context.thisNode() instanceof GraphQLScalarType) {
       80                      newScalar((GraphQLScalarType) context.thisNode(), schemaGraph, isIntrospectionNode);
       81                  }
       82                  if (context.thisNode() instanceof GraphQLInputObjectType) {
       83                      newInputObject((GraphQLInputObjectType) context.thisNode(), schemaGraph, isIntrospectionNode);
       84                  }
       85                  if (context.thisNode() instanceof GraphQLEnumType) {
       86                      newEnum((GraphQLEnumType) context.thisNode(), schemaGraph, isIntrospectionNode);
       87                  }
       88                  if (context.thisNode() instanceof GraphQLDirective) {
       89                      // only continue if not applied directive
       90                      if (context.getParentNode() == null) {
       91                          newDirective((GraphQLDirective) context.thisNode(), schemaGraph);
       92                      }
       93                  }
       94                  return TraversalControl.CONTINUE;
       95              }
       96  
       97              @Override
       98              public TraversalControl leave(TraverserContext<GraphQLSchemaElement> context) {
       99                  return TraversalControl.CONTINUE;
      100              }
      101          });
      102          addSchemaVertex(schemaGraph, schema);
      103  
      104          ArrayList<Vertex> copyOfVertices = new ArrayList<>(schemaGraph.getVertices());
      105          for (Vertex vertex : copyOfVertices) {
      106              if (SchemaGraph.OBJECT.equals(vertex.getType())) {
      107                  handleObjectVertex(vertex, schemaGraph, schema);
      108              }
      109              if (SchemaGraph.INTERFACE.equals(vertex.getType())) {
      110                  handleInterfaceVertex(vertex, schemaGraph, schema);
      111              }
      112              if (SchemaGraph.UNION.equals(vertex.getType())) {
      113                  handleUnion(vertex, schemaGraph, schema);
      114              }
      115              if (SchemaGraph.INPUT_OBJECT.equals(vertex.getType())) {
      116                  handleInputObject(vertex, schemaGraph, schema);
      117              }
      118              if (SchemaGraph.DIRECTIVE.equals(vertex.getType())) {
      119                  handleDirective(vertex, schemaGraph, schema);
      120              }
      121          }
      122          return schemaGraph;
      123      }

      125      private void addSchemaVertex(SchemaGraph schemaGraph, GraphQLSchema graphQLSchema) {
      126          GraphQLObjectType queryType = graphQLSchema.getQueryType();
      127          GraphQLObjectType mutationType = graphQLSchema.getMutationType();
      128          GraphQLObjectType subscriptionType = graphQLSchema.getSubscriptionType();
      129          Vertex schemaVertex = new Vertex(SchemaGraph.SCHEMA, "schema");
      130          schemaVertex.add("name", SchemaGraph.SCHEMA);
      131          schemaGraph.addVertex(schemaVertex);
      132  
      133          Vertex queryVertex = schemaGraph.getType(queryType.getName());
      134          schemaGraph.addEdge(new Edge(schemaVertex, queryVertex, "query"));
      135          if (mutationType != null) {
      136              Vertex mutationVertex = schemaGraph.getType(mutationType.getName());
      137              schemaGraph.addEdge(new Edge(schemaVertex, mutationVertex, "mutation"));
      138          }
      139          if (subscriptionType != null) {
      140              Vertex subscriptionVertex = schemaGraph.getType(subscriptionType.getName());
      141              schemaGraph.addEdge(new Edge(schemaVertex, subscriptionVertex, "subscription"));
      142          }
      143          createAppliedDirectives(schemaVertex, graphQLSchema.getSchemaDirectives(), schemaGraph);
      144      }

      146      private void handleInputObject(Vertex inputObject, SchemaGraph schemaGraph, GraphQLSchema graphQLSchema) {
      147          GraphQLInputObjectType inputObjectType = (GraphQLInputObjectType) graphQLSchema.getType(inputObject.get("name"));
      148          List<GraphQLInputObjectField> inputFields = inputObjectType.getFields();
      149          for (GraphQLInputObjectField inputField : inputFields) {
      150              Vertex inputFieldVertex = schemaGraph.findTargetVertex(inputObject, vertex -> vertex.getType().equals("InputField") &&
      151                      vertex.get("name").equals(inputField.getName())).get();
      152              handleInputField(inputFieldVertex, inputField, schemaGraph, graphQLSchema);
      153          }

      156      private void handleInputField(Vertex inputFieldVertex, GraphQLInputObjectField inputField, SchemaGraph
      157              schemaGraph, GraphQLSchema graphQLSchema) {
      158          GraphQLInputType type = inputField.getType();
      159          GraphQLUnmodifiedType graphQLUnmodifiedType = GraphQLTypeUtil.unwrapAll(type);
      160          Vertex typeVertex = assertNotNull(schemaGraph.getType(graphQLUnmodifiedType.getName()));
      161          Edge typeEdge = new Edge(inputFieldVertex, typeVertex);
      162          String typeEdgeLabel = "type=" + GraphQLTypeUtil.simplePrint(type) + ";defaultValue=";
      163          if (inputField.hasSetDefaultValue()) {
      164              typeEdgeLabel += AstPrinter.printAst(ValuesResolver.valueToLiteral(inputField.getInputFieldDefaultValue(), inputField.getType(), GraphQLContext.getDefault(), Locale.getDefault()));
      165          }
      166  
      167          typeEdge.setLabel(typeEdgeLabel);
      168          schemaGraph.addEdge(typeEdge);

      362      private void createAppliedDirectives(Vertex from,
      363                                           List<GraphQLDirective> appliedDirectives,
      364                                           SchemaGraph schemaGraph) {
      365          Map<String, Integer> countByName = new LinkedHashMap<>();
      366          for (GraphQLDirective appliedDirective : appliedDirectives) {
      367              Vertex appliedDirectiveVertex = new Vertex(SchemaGraph.APPLIED_DIRECTIVE, debugPrefix + String.valueOf(counter++));
      368              appliedDirectiveVertex.add("name", appliedDirective.getName());
      369              for (GraphQLArgument appliedArgument : appliedDirective.getArguments()) {
      370                  if (appliedArgument.hasSetValue()) {
      371                      Vertex appliedArgumentVertex = new Vertex(SchemaGraph.APPLIED_ARGUMENT, debugPrefix + String.valueOf(counter++));
      372                      appliedArgumentVertex.add("name", appliedArgument.getName());
      373                      appliedArgumentVertex.add("value", AstPrinter.printAst(ValuesResolver.valueToLiteral(appliedArgument.getArgumentValue(), appliedArgument.getType(), GraphQLContext.getDefault(), Locale.getDefault())));
      374                      schemaGraph.addVertex(appliedArgumentVertex);
      375                      schemaGraph.addEdge(new Edge(appliedDirectiveVertex, appliedArgumentVertex));
      376                  }
      377              }
      378              schemaGraph.addVertex(appliedDirectiveVertex);
      379  
      380              // repeatable directives means we can have multiple directives with the same name
      381              // the edge label indicates the applied directive index
      382              Integer count = countByName.getOrDefault(appliedDirective.getName(), 0);
      383              schemaGraph.addEdge(new Edge(from, appliedDirectiveVertex, String.valueOf(count)));
      384              countByName.put(appliedDirective.getName(), count + 1);
      385          }
      386      }

      388      private void newDirective(GraphQLDirective directive, SchemaGraph schemaGraph) {
      389          Vertex directiveVertex = new Vertex(SchemaGraph.DIRECTIVE, debugPrefix + String.valueOf(counter++));
      390          directiveVertex.add("name", directive.getName());
      391          directiveVertex.add("repeatable", directive.isRepeatable());
      392          directiveVertex.add("locations", directive.validLocations());
      393          boolean graphqlSpecified = Directives.isBuiltInDirective(directive.getName());
      394          directiveVertex.setBuiltInType(graphqlSpecified);
      395          directiveVertex.add("description", desc(directive.getDescription()));
      396          for (GraphQLArgument argument : directive.getArguments()) {
      397              Vertex argumentVertex = newArgument(argument, schemaGraph, graphqlSpecified);
      398              schemaGraph.addVertex(argumentVertex);
      399              schemaGraph.addEdge(new Edge(directiveVertex, argumentVertex));
      400          }
      401          schemaGraph.addDirective(directive.getName(), directiveVertex);
      402          schemaGraph.addVertex(directiveVertex);
      403      }
