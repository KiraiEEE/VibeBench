      160      public boolean isAccessible(Map context, Object target, Member member, String propertyName) {
      161          LOG.debug("Checking access for [target: {}, member: {}, property: {}]", target, member, propertyName);
      162  
      163          if (member == null) {
      164              throw new IllegalArgumentException("Member cannot be null!");
      165          }
      166          if (target != null) {
      167              // Special case: Target is a Class object but not Class.class
      168              if (Class.class.equals(target.getClass()) && !Class.class.equals(target)) {
      169                  if (!isStatic(member) && !Constructor.class.equals(member.getClass())) {
      170                      throw new IllegalArgumentException("Member expected to be static or constructor!");
      171                  }
      172                  if (!member.getDeclaringClass().equals(target)) {
      173                      throw new IllegalArgumentException("Target class does not match member!");
      174                  }
      175                  target = null; // This information is not useful to us and conflicts with following logic which expects target to be null or an instance containing the member
      176              // Standard case: Member should exist on target
      177              } else if (!member.getDeclaringClass().isAssignableFrom(target.getClass())) {
      178                  throw new IllegalArgumentException("Member does not exist on target!");
      179              }
      180          }
      181  
      182          if (!checkProxyObjectAccess(target)) {
      183              LOG.warn("Access to proxy is blocked! Target [{}], proxy class [{}]", target, target.getClass().getName());
      184              return false;
      185          }
      186  
      187          if (!checkProxyMemberAccess(target, member)) {
      188              LOG.warn("Access to proxy is blocked! Member class [{}] of target [{}], member [{}]", member.getDeclaringClass(), target, member);
      189              return false;
      190          }
      191  
      192          if (!checkPublicMemberAccess(member)) {
      193              LOG.warn("Access to non-public [{}] is blocked!", member);
      194              return false;
      195          }
      196  
      197          if (!checkStaticFieldAccess(member)) {
      198              LOG.warn("Access to static field [{}] is blocked!", member);
      199              return false;
      200          }
      201  
      202          if (!checkStaticMethodAccess(member)) {
      203              LOG.warn("Access to static method [{}] is blocked!", member);
      204              return false;
      205          }
      206  
      207          if (!checkDefaultPackageAccess(target, member)) {
      208              return false;
      209          }
      210  
      211          if (!checkExclusionList(target, member)) {
      212              return false;
      213          }
      214  
      215          if (!checkAllowlist(target, member)) {
      216              return false;
      217          }
      218  
      219          if (!isAcceptableProperty(propertyName)) {
      220              return false;
      221          }
      222  
      223          return true;

      229      protected boolean checkAllowlist(Object target, Member member) {
      230          if (!enforceAllowlistEnabled) {
      231              return true;
      232          }
      233  
      234          if (!disallowProxyObjectAccess && ProxyUtil.isProxy(target)) {
      235              // If `disallowProxyObjectAccess` is not set, allow resolving Hibernate entities to their underlying
      236              // classes/members. This allows the allowlist capability to continue working and offer some level of
      237              // protection in applications where the developer has accepted the risk of allowing OGNL access to Hibernate
      238              // entities. This is preferred to having to disable the allowlist capability entirely.
      239              Object newTarget = ProxyUtil.getHibernateProxyTarget(target);
      240              if (newTarget != target) {
      241                  logAllowlistHibernateEntity(target, newTarget);
      242                  target = newTarget;
      243                  member = ProxyUtil.resolveTargetMember(member, newTarget);
      244              }
      245          }
      246  
      247          Class<?> memberClass = member.getDeclaringClass();
      248          if (!isClassAllowlisted(memberClass)) {
      249              LOG.warn("Declaring class [{}] of member type [{}] is not allowlisted! Add to '{}' or '{}' configuration.",
      250                      memberClass, member, STRUTS_ALLOWLIST_CLASSES, STRUTS_ALLOWLIST_PACKAGE_NAMES);
      251              return false;
      252          }
      253          if (target == null || target.getClass() == memberClass) {
      254              return true;
      255          }
      256          Class<?> targetClass = target.getClass();
      257          if (!isClassAllowlisted(targetClass)) {
      258              LOG.warn("Target class [{}] of target [{}] is not allowlisted! Add to '{}' or '{}' configuration.",
      259                      targetClass, target, STRUTS_ALLOWLIST_CLASSES, STRUTS_ALLOWLIST_PACKAGE_NAMES);
      260              return false;
      261          }
      262          return true;
      263      }

      415          List<String> packageParts = Arrays.asList(toPackageName(clazz).split("\\."));
      416          for (int i = 0; i < packageParts.size(); i++) {
      417              String parentPackage = String.join(".", packageParts.subList(0, i + 1));
