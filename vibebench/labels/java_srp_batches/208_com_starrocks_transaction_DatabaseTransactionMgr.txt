      322       * prepared transaction process as follows:

      331      public void prepareTransaction(long transactionId,
      332                                     long preparedTimeoutMs,
      333                                     List<TabletCommitInfo> tabletCommitInfos,
      334                                     List<TabletFailInfo> tabletFailInfos,
      335                                     TxnCommitAttachment txnCommitAttachment,
      336                                     boolean writeEditLog)
      337              throws StarRocksException {
      338          Preconditions.checkNotNull(tabletCommitInfos, "tabletCommitInfos is null");
      339          Preconditions.checkNotNull(tabletFailInfos, "tabletFailInfos is null");
      340          // 1. check status
      341          // the caller method already own db lock, we do not obtain db lock here
      342          Database db = globalStateMgr.getLocalMetastore().getDb(dbId);
      343          if (null == db) {
      344              throw new MetaNotFoundException("could not find db [" + dbId + "]");
      345          }
      346  
      347          TransactionState transactionState = getTransactionState(transactionId);
      348          if (transactionState == null) {
      349              throw new TransactionNotFoundException(transactionId);
      350          }
      351  
      352          transactionState.writeLock();
      353          try {
      354              if (transactionState.getTransactionStatus() == TransactionStatus.ABORTED) {
      355                  throw new TransactionCommitFailedException(transactionState.getReason());
      356              }
      357              if (transactionState.getTransactionStatus() == TransactionStatus.VISIBLE) {
      358                  LOG.debug("transaction is already visible: {}", transactionId);
      359                  return;
      360              }
      361              if (transactionState.getTransactionStatus() == TransactionStatus.COMMITTED) {
      362                  LOG.debug("transaction is already committed: {}", transactionId);
      363                  return;
      364              }
      365              if (transactionState.getTransactionStatus() == TransactionStatus.PREPARED) {
      366                  LOG.debug("transaction is already prepared: {}", transactionId);
      367                  return;
      368              }
      369              // For compatible reason, the default behavior of empty load is still returning
      370              // "No rows were imported from upstream" and abort transaction.
      371              if (Config.empty_load_as_error && tabletCommitInfos.isEmpty()
      372                      && transactionState.getSourceType() != TransactionState.LoadJobSourceType.INSERT_STREAMING) {
      373                  throw new TransactionCommitFailedException(ERR_NO_ROWS_IMPORTED.formatErrorMsg());
      374              }
      375  
      376              if (transactionState.getWriteEndTimeMs() < 0) {
      377                  transactionState.setWriteEndTimeMs(System.currentTimeMillis());
      378              }
      379  
      380              // update transaction state extra if exists
      381              if (txnCommitAttachment != null) {
      382                  transactionState.setTxnCommitAttachment(txnCommitAttachment);
      383              }
      384  
      385              Span txnSpan = transactionState.getTxnSpan();
      386              txnSpan.setAttribute("db", db.getFullName());
      387              txnSpan.addEvent("pre_commit_start");
      388  
      389              List<TransactionStateListener> stateListeners = populateTransactionStateListeners(transactionState, db);
      390              String tableNames = stateListeners.stream().map(TransactionStateListener::getTableName)
      391                      .collect(Collectors.joining(","));
      392              txnSpan.setAttribute("tables", tableNames);
      393  
      394              for (TransactionStateListener listener : stateListeners) {
      395                  listener.preCommit(transactionState, tabletCommitInfos, tabletFailInfos);
      396              }
      397  
      398              transactionState.beforeStateTransform(TransactionStatus.PREPARED);
      399              boolean txnOperated = false;
      400  
      401              Span unprotectedCommitSpan = TraceManager.startSpan("unprotectedPreparedTransaction", txnSpan);
      402  
      403              writeLock();
      404              try {
      405                  // transaction state is modified during check if the transaction could commit
      406                  if (transactionState.getTransactionStatus() != TransactionStatus.PREPARE) {
      407                      return;
      408                  }
      409  
      410                  // update transaction state version
      411                  transactionState.setTransactionStatus(TransactionStatus.PREPARED);
      412                  transactionState.setPreparedTimeAndTimeout(System.currentTimeMillis(), preparedTimeoutMs);
      413  
      414                  for (TransactionStateListener listener : stateListeners) {
      415                      listener.preWriteCommitLog(transactionState);
      416                  }
      417  
      418                  // persist transactionState
      419                  if (writeEditLog) {
      420                      unprotectUpsertTransactionState(transactionState);
      421                  }
      422  
      423                  txnOperated = true;
      424              } finally {
      425                  writeUnlock();

      427                  for (Map.Entry<Long, TableCommitInfo> entry : transactionState.getIdToTableCommitInfos().entrySet()) {
      428                      numPartitions += entry.getValue().getIdToPartitionCommitInfo().size();
      429                  }
      430                  txnSpan.setAttribute("num_partition", numPartitions);
      431                  unprotectedCommitSpan.end();

      433                  try {
      434                      transactionState.afterStateTransform(TransactionStatus.PREPARED, txnOperated, null);
      435                  } catch (Throwable t) {
      436                      LOG.warn("transaction after state transform failed: {}", transactionState, t);
      437                  }
      438              }
      439              if (writeEditLog) {
      440                  persistTxnStateInTxnLevelLock(transactionState);
      441              }
      442  
      443              LOG.debug("transaction:[{}] successfully prepare", transactionState);
      444          } finally {
      445              transactionState.writeUnlock();
      446          }
      447      }
