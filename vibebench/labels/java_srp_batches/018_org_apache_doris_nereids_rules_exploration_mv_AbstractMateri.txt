      177          // split the query top plan expressions to group expressions and functions, if can not, bail out.
      178          Pair<Set<? extends Expression>, Set<? extends Expression>> queryGroupAndFunctionPair

      188                          .flattenMap().get(0);

      191              if (queryTopPlanFunctionSet.contains(topExpression)) {
      192                  // if agg function, try to roll up and rewrite
      193                  Expression rollupedExpression = tryRewriteExpression(queryStructInfo, topExpression,
      194                          mvExprToMvScanExprQueryBased, aggregateFunctionMode, materializationContext,
      195                          "Query function roll up fail",
      196                          () -> String.format("queryExpression = %s,\n mvExprToMvScanExprQueryBased = %s",
      197                                  topExpression, mvExprToMvScanExprQueryBased));
      198                  if (rollupedExpression == null) {
      199                      return null;
      200                  }
      201                  finalOutputExpressions.add(new Alias(rollupedExpression));
      202              } else {
      203                  // if group by dimension, try to rewrite
      204                  Expression rewrittenGroupByExpression = tryRewriteExpression(queryStructInfo, topExpression,
      205                          mvExprToMvScanExprQueryBased, groupByMode, materializationContext,
      206                          "View dimensions doesn't not cover the query dimensions",
      207                          () -> String.format("mvExprToMvScanExprQueryBased is %s,\n queryExpression is %s",
      208                                  mvExprToMvScanExprQueryBased, topExpression));
      209                  if (rewrittenGroupByExpression == null) {
      210                      // group expr can not rewrite by view
      211                      return null;
      212                  }
      213                  NamedExpression groupByExpression = rewrittenGroupByExpression instanceof NamedExpression
      214                          ? (NamedExpression) rewrittenGroupByExpression : new Alias(rewrittenGroupByExpression);
      215                  finalOutputExpressions.add(groupByExpression);
      216                  finalGroupExpressions.add(groupByExpression);
      217              }
      218          }
      219          LogicalAggregate<Plan> queryAggregate = queryTopPlanAndAggPair.value();
      220          List<Expression> queryGroupByExpressions = queryAggregate.getGroupByExpressions();

      224                  if (queryTopPlanGroupBySet.contains(expression)) {
      225                      continue;
      226                  }

      228                          mvExprToMvScanExprQueryBased, groupByMode, materializationContext,

      274       * handle the scene that query top plan not use the group by in query bottom aggregate
      275       * If mv is select o_orderdate from  orders group by o_orderdate;
      276       * query is select 1 from orders group by o_orderdate.
      277       * Or mv is select o_orderdate from orders group by o_orderdate
      278       * query is select o_orderdate from  orders group by o_orderdate, o_orderkey;
      279       * if the slot which query top project use can not cover the slot which query bottom aggregate group by slot
      280       * should compensate group by to make sure the data is right.
      281       * For example:
      282       * mv is select o_orderdate from orders group by o_orderdate;
      283       * query is select o_orderdate from  orders group by o_orderdate, o_orderkey;
      284       *
      285       * @param queryGroupByExpressions query bottom aggregate group by is o_orderdate, o_orderkey
      286       * @param queryTopProject query top project is o_orderdate
      287       * @return need to compensate group by if true or not need
      288       *

      290      private static boolean needCompensateGroupBy(Set<? extends Expression> queryTopProject,
      291              List<Expression> queryGroupByExpressions) {
      292          Set<Expression> queryGroupByExpressionSet = new HashSet<>(queryGroupByExpressions);
      293          if (queryGroupByExpressionSet.size() != queryTopProject.size()) {
      294              return true;
      295          }
      296          Set<NamedExpression> queryTopPlanGroupByUseNamedExpressions = new HashSet<>();
      297          Set<NamedExpression> queryGroupByUseNamedExpressions = new HashSet<>();
      298          for (Expression expr : queryTopProject) {
      299              queryTopPlanGroupByUseNamedExpressions.addAll(expr.collect(NamedExpression.class::isInstance));
      300          }
      301          for (Expression expr : queryGroupByExpressionSet) {
      302              queryGroupByUseNamedExpressions.addAll(expr.collect(NamedExpression.class::isInstance));
      303          }
      304          // if the slots query top project use can not cover the slots which query bottom aggregate use
      305          // Should compensate.
      306          return !queryTopPlanGroupByUseNamedExpressions.containsAll(queryGroupByUseNamedExpressions);
      307      }

      341      @Override
      342      protected boolean canUnionRewrite(Plan queryPlan, AsyncMaterializationContext context,
      343              CascadesContext cascadesContext) {
      344          // Check query plan is contain the partition column
      345          // Query plan in the current rule must contain aggregate node, because the rule pattern is
      346          Optional<LogicalAggregate<Plan>> logicalAggregateOptional =
      347                  queryPlan.collectFirst(planTreeNode -> planTreeNode instanceof LogicalAggregate);
      348          if (!logicalAggregateOptional.isPresent()) {
      349              return true;
      350          }
      351          List<Expression> groupByExpressions = logicalAggregateOptional.get().getGroupByExpressions();
      352          if (groupByExpressions.isEmpty()) {
      353              // Scalar aggregate can not compensate union all
      354              return false;
      355          }
      356          MTMVPartitionInfo mvPartitionInfo = context.getMtmv().getMvPartitionInfo();
      357          List<BaseColInfo> pctInfos = mvPartitionInfo.getPctInfos();
      358          boolean canUnionRewrite = false;
      359          // Check the query plan group by expression contains partition col or not
      360          List<? extends Expression> groupByShuttledExpressions =
      361                  ExpressionUtils.shuttleExpressionWithLineage(groupByExpressions, queryPlan, new BitSet());
      362          for (Expression expression : groupByShuttledExpressions) {
      363              canUnionRewrite = !expression.collectToSet(expr -> {
      364                  if (!(expr instanceof SlotReference) || !((SlotReference) expr).isColumnFromTable()) {
      365                      return false;
      366                  }
      367                  String columnName = ((SlotReference) expr).getOriginalColumn().map(Column::getName).orElse(null);
      368                  BaseTableInfo baseTableInfo = ((SlotReference) expr).getOriginalTable().map(BaseTableInfo::new)
      369                          .orElse(null);
      370                  return pctInfos.stream().anyMatch(colInfo -> colInfo.getTableInfo().equals(baseTableInfo)
      371                          && colInfo.getColName().equals(columnName));
      372              }
      373              ).isEmpty();
      374              if (canUnionRewrite) {
      375                  break;
      376              }
      377          }
      378          return canUnionRewrite;
      379      }

      384      private static boolean checkCompatibility(
      385              StructInfo queryStructInfo,
      386              LogicalAggregate<Plan> queryAggregate, LogicalAggregate<Plan> viewAggregate,
      387              MaterializationContext materializationContext) {
      388          // if view is scalar aggregate but query is not. Or if query is scalar aggregate but view is not
      389          // Should not rewrite
      390          List<Expression> queryGroupByExpressions = queryAggregate.getGroupByExpressions();
      391          List<Expression> viewGroupByExpressions = viewAggregate.getGroupByExpressions();
      392          if (!queryGroupByExpressions.isEmpty() && viewGroupByExpressions.isEmpty()) {
      393              materializationContext.recordFailReason(queryStructInfo,
      394                      "only one the of query or view is scalar aggregate and "
      395                              + "can not rewrite expression meanwhile",
      396                      () -> String.format("query aggregate = %s,\n view aggregate = %s,\n",
      397                              queryAggregate.treeString(),
      398                              viewAggregate.treeString()));
      399              return false;
      400          }
      401          boolean viewHasGroupSets = viewAggregate.getSourceRepeat()
      402                  .map(repeat -> repeat.getGroupingSets().size()).orElse(0) > 0;
      403          // if both query and view has group sets, or query doesn't hava, mv have, not supported
      404          if (viewHasGroupSets) {
      405              materializationContext.recordFailReason(queryStructInfo,
      406                      "both query and view have group sets, or query doesn't have but view have, not supported",
      407                      () -> String.format("query aggregate = %s,\n view aggregate = %s,\n",
      408                              queryAggregate.treeString(),
      409                              viewAggregate.treeString()));
      410              return false;
      411          }
      412          return true;
      413      }
