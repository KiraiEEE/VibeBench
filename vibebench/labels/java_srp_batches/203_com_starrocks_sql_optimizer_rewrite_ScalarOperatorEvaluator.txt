       70      private synchronized void registerFunctions() {
       71          // double checked locking pattern
       72          // functions only need to init once
       73          if (functions != null) {
       74              return;
       75          }
       76  
       77          ImmutableMap.Builder<FunctionSignature, FunctionInvoker> mapBuilder = new ImmutableMap.Builder<>();
       78  
       79          Class<?> metaFunctions = MetaFunctions.class;
       80          Class<?> clazz = ScalarOperatorFunctions.class;
       81          for (Method method : ListUtils.union(
       82                  Lists.newArrayList(clazz.getDeclaredMethods()),
       83                  Lists.newArrayList(metaFunctions.getDeclaredMethods()))) {
       84              ConstantFunction annotation = method.getAnnotation(ConstantFunction.class);
       85              registerFunction(mapBuilder, method, annotation);
       86  
       87              ConstantFunction.List listAnnotation = method.getAnnotation(ConstantFunction.List.class);
       88  
       89              if (listAnnotation != null) {
       90                  for (ConstantFunction f : listAnnotation.list()) {
       91                      registerFunction(mapBuilder, method, f);
       92                  }
       93              }
       94          }
       95          this.functions = mapBuilder.build();
       96      }

      133       * evaluation a fe built-in function
      134       *
      135       * @param root CallOperator root
      136       * @return ConstantOperator if the CallOperator is effect (All child constant/FE builtin function support/....)

      138      public ScalarOperator evaluation(CallOperator root, boolean needMonotonic) {
      139          if (ConnectContext.get() != null
      140                  && ConnectContext.get().getSessionVariable().isDisableFunctionFoldConstants()) {
      141              return root;
      142          }
      143  
      144          for (ScalarOperator child : root.getChildren()) {
      145              if (!OperatorType.CONSTANT.equals(child.getOpType())) {
      146                  return root;
      147              }
      148          }
      149  
      150          Function fn = root.getFunction();
      151          if (fn == null) {
      152              return root;
      153          }
      154  
      155          // return Null directly iff:
      156          // 1. Not UDF
      157          // 2. Not in isNotAlwaysNullResultWithNullParamFunctions
      158          // 3. Has null parameter
      159          // 4. Not assert_true
      160          if (!GlobalStateMgr.getCurrentState()
      161                  .isNotAlwaysNullResultWithNullParamFunction(fn.getFunctionName().getFunction())
      162                  && !fn.isUdf()
      163                  && !FunctionSet.ASSERT_TRUE.equals(fn.getFunctionName().getFunction())) {
      164              for (ScalarOperator op : root.getChildren()) {
      165                  if (((ConstantOperator) op).isNull()) {
      166                      // Should return ConstantOperator.createNull(fn.getReturnType()),
      167                      // but for keep same with old StarRocks
      168                      // types in decimalv3-typed function instances are wild types(both precision and scale are -1)
      169                      // the wild types should never escaped outside of function instance resolution.
      170                      Type type = fn.getReturnType();
      171                      if (type.isDecimalV3()) {
      172                          return ConstantOperator.createNull(root.getType());
      173                      } else {
      174                          return ConstantOperator.createNull(fn.getReturnType());
      175                      }
      176                  }
      177              }
      178          }
      179  
      180          List<Type> argTypes = new ArrayList<>(Arrays.asList(fn.getArgs()));
      181  
      182          FunctionSignature signature =
      183                  new FunctionSignature(fn.functionName().toUpperCase(), argTypes, fn.getReturnType());
      184  
      185          FunctionInvoker invoker = functions.get(signature);
      186  
      187          if (invoker == null) {
      188              return root;
      189          }
      190  
      191          if (needMonotonic && !isMonotonicFunc(invoker, root)) {
      192              return root;
      193          }
      194  
      195          try {
      196              ConstantOperator operator = invoker.invoke(root.getChildren());
      197              // check return result type, decimal will change return type
      198              if (!operator.isNull() &&
      199                      operator.getType().getPrimitiveType() != fn.getReturnType().getPrimitiveType()) {
      200                  Preconditions.checkState(operator.getType().isDecimalOfAnyVersion());
      201                  Preconditions.checkState(fn.getReturnType().isDecimalOfAnyVersion());
      202                  if (operator.getType().isDecimal256()) {
      203                      Optional<ConstantOperator> res = operator.castTo(fn.getReturnType());
      204                      if (res.isPresent() && res.get().isConstantNull()) {
      205                          return root;
      206                      }
      207                  }
      208                  operator.setType(fn.getReturnType());
      209              }
      210              return operator;
      211          } catch (Exception e) {
      212              if (LOG.isDebugEnabled()) {
      213                  LOG.debug("failed to invoke", e);
      214              }
      215              if (invoker.isMetaFunction) {
      216                  throw new StarRocksPlannerException(ErrorType.USER_ERROR, ExceptionUtils.getRootCauseMessage(e));
      217              }
      218          }
      219          return root;
      220      }

      222      public boolean isMonotonicFunction(CallOperator call) {
      223          if (call instanceof CastOperator) {
      224              return true;
      225          }
      226          FunctionSignature signature;
      227          if (call.getFunction() != null) {
      228              Function fn = call.getFunction();
      229              List<Type> argTypes = Arrays.asList(fn.getArgs());
      230              signature = new FunctionSignature(fn.functionName().toUpperCase(), argTypes, fn.getReturnType());
      231          } else {
      232              List<Type> argTypes = call.getArguments().stream().map(ScalarOperator::getType).collect(Collectors.toList());
      233              signature = new FunctionSignature(call.getFnName().toUpperCase(), argTypes, call.getType());
      234          }
      235  
      236          FunctionInvoker invoker = functions.get(signature);
      237  
      238          return invoker != null && isMonotonicFunc(invoker, call);
      239      }

      256      private boolean isMonotonicFunc(FunctionInvoker invoker, CallOperator operator) {
      257          if (!invoker.isMonotonic) {
      258              return false;
      259          }
      260  
      261          final ImmutableSet<String> SUPPORTED = ImmutableSet.of(
      262                  FunctionSet.DATE_FORMAT,
      263                  FunctionSet.STR_TO_DATE,
      264                  FunctionSet.STR2DATE,
      265                  FunctionSet.FROM_UNIXTIME,
      266                  FunctionSet.FROM_UNIXTIME_MS,
      267                  FunctionSet.TO_DATETIME
      268          );
      269  
      270          if (SUPPORTED.contains(invoker.getSignature().getName().toLowerCase()) && operator.getChildren().size() == 2) {
      271              String pattern = operator.getChild(1).toString();
      272              if (pattern.isEmpty()) {
      273                  return true;
      274              }
      275  
      276              if (SUPPORT_JAVA_STYLE_DATETIME_FORMATTER.contains(pattern.trim())) {
      277                  return true;
      278              }
      279              Optional<String> stripedPattern = stripFormatValue(pattern);
      280              // "YmdHiSf" or "YmdHisf" ensure the format string value reserve the original date order
      281              // like date_format('2021-01-01', '%Y-%m-%d') is qualified but date_format('2021-01-01', '%m-%Y-%d') is not
      282              return stripedPattern.map(e -> "YmdHiSf".startsWith(e) || "YmdHisf".startsWith(e)).orElse(false);
      283          }
      284  
      285          return true;
      286      }
      287      private Optional<String> stripFormatValue(String pattern) {
      288          StringBuilder builder = new StringBuilder();
      289          boolean unsupportedFormat = false;
      290          for (char c : pattern.toCharArray()) {
      291              switch (c) {
      292                  case 'Y':
      293                  case 'm':
      294                  case 'd':
      295                  case 'H':
      296                  case 'i':
      297                  case 'S':
      298                  case 's':
      299                  case 'f':
      300                      builder.append(c);
      301                      break;
      302                  case 'a':
      303                  case 'b':
      304                  case 'c':
      305                  case 'D':
      306                  case 'e':
      307                  case 'h':
      308                  case 'I':
      309                  case 'j':
      310                  case 'k':
      311                  case 'l':
      312                  case 'M':
      313                  case 'p':
      314                  case 'r':
      315                  case 'T':
      316                  case 'U':
      317                  case 'u':
      318                  case 'V':
      319                  case 'v':
      320                  case 'W':
      321                  case 'w':
      322                  case 'X':
      323                  case 'x':
      324                  case 'y':
      325                      unsupportedFormat = true;
      326                      break;
      327                  default:
      328                      // do nothing
      329              }
      330          }
      331  
      332          return unsupportedFormat ? Optional.empty() : Optional.of(builder.toString());
      333      }

      359              final List<Object> invokeArgs = createInvokeArgs(args);

      363          private List<Object> createInvokeArgs(List<ScalarOperator> args) {
      364              final List<Object> invokeArgs = Lists.newArrayList();
      365              for (int index = 0; index < method.getParameterTypes().length; index++) {
      366                  final Class<?> argType = method.getParameterTypes()[index];
      367  
      368                  if (argType.isArray()) {
      369                      Preconditions.checkArgument(method.getParameterTypes().length == index + 1);
      370                      final List<ConstantOperator> variableArgs = Lists.newArrayList();
      371                      Set<PrimitiveType> checkSet = Sets.newHashSet();
      372  
      373                      for (int variableArgIndex = index; variableArgIndex < args.size(); variableArgIndex++) {
      374                          ConstantOperator arg = (ConstantOperator) args.get(variableArgIndex);
      375                          variableArgs.add(arg);
      376                          checkSet.add(arg.getType().getPrimitiveType());
      377                      }
      378  
      379                      // Array data must keep same kinds
      380                      if (checkSet.size() > 1) {
      381                          throw new IllegalArgumentException("Function's args does't match.");
      382                      }
      383  
      384                      ConstantOperator[] argsArray = new ConstantOperator[variableArgs.size()];
      385                      argsArray = variableArgs.toArray(argsArray);
      386                      invokeArgs.add(argsArray);
      387                  } else {
      388                      invokeArgs.add(args.get(index));
      389                  }
      390              }
      391              return invokeArgs;
      392          }
