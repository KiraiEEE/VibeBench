      376       * 检查 Headers tab 是否有内容
      377       */
      378      private boolean hasHeadersContent() {
      379          List<HttpHeader> headers = headersPanel.getHeadersListFromModel();
      380          if (headers == null || headers.isEmpty()) {
      381              return false;
      382          }
      383          // 检查是否有非空的 header
      384          return headers.stream().anyMatch(header ->
      385                  header.getKey() != null && !header.getKey().trim().isEmpty()
      386          );
      387      }

      391       */
      392      private boolean hasVariablesContent() {
      393          List<Variable> variables = variablesPanel.getVariableListFromModel();
      394          if (variables == null || variables.isEmpty()) {
      395              return false;
      396          }

      398          return variables.stream().anyMatch(variable ->
      399                  variable.getKey() != null && !variable.getKey().trim().isEmpty()
      400          );

      406      private boolean hasAuthContent() {
