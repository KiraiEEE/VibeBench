      246    /**
      247     * Emit metrics and conditional log, then restore MDC. Idempotent — safe to call multiple times;
      248     * only the first invocation emits metrics and logs. Never throws — metric/logging failures are
      249     * caught internally to avoid breaking the cascade operation.
      250     */
      251    @Override
      252    public void close() {
      253      if (!closed.compareAndSet(false, true)) {
      254        return; // already closed — skip duplicate metric/log emission
      255      }
      256      try {
      257        long durationNanos = System.nanoTime() - startNanos;
      258        long durationMs = TimeUnit.NANOSECONDS.toMillis(durationNanos);
      259        int processed = entitiesProcessed.get();
      260        int errors = errorCount.get();
      261        String status = errors > 0 ? "completed_with_errors" : "completed";
      262  
      263        int hops = hopsCompleted.get();
      264  
      265        // Emit metrics (always, regardless of log level)
      266        if (metricUtils != null) {
      267          metricUtils.recordTimer(
      268              metricPrefix + ".duration",
      269              durationNanos,
      270              "operation_type",
      271              operationType,
      272              "trigger_urn_type",
      273              triggerUrnType,
      274              "status",
      275              status);
      276          metricUtils.incrementMicrometer(
      277              metricPrefix + ".entities_processed",
      278              processed,
      279              "operation_type",
      280              operationType,
      281              "trigger_urn_type",
      282              triggerUrnType);
      283          if (hops > 0) {
      284            metricUtils.incrementMicrometer(
      285                metricPrefix + ".hops",
      286                hops,
      287                "operation_type",
      288                operationType,
      289                "trigger_urn_type",
      290                triggerUrnType);
      291          }
      292          if (errors > 0) {
      293            metricUtils.incrementMicrometer(
      294                metricPrefix + ".errors",
      295                errors,
      296                "operation_type",
      297                operationType,
      298                "trigger_urn_type",
      299                triggerUrnType,
      300                "error_type",
      301                lastErrorType != null ? lastErrorType : "unknown");
      302          }
      303        }
      304  
      305        // Conditional logging per PR #16577/#16578 guidelines
      306        if (errors > 0) {
      307          log.warn(
      308              "Cascade completed with errors: type={}, entities={}, hops={}, errors={}, duration={}ms",
      309              operationType,
      310              processed,
      311              hops,
      312              errors,
      313              durationMs);
      314        } else if (durationMs >= slowThresholdMs) {
      315          log.info(
      316              "Cascade completed (slow): type={}, entities={}, hops={}, duration={}ms",
      317              operationType,
      318              processed,
      319              hops,
      320              durationMs);
      321        } else {
      322          log.debug(
      323              "Cascade completed: type={}, entities={}, hops={}, duration={}ms",
      324              operationType,
      325              processed,
      326              hops,
      327              durationMs);
      328        }
      329      } catch (Exception e) {
      330        // Never propagate — observability failures must not break the cascade
      331        log.debug("Failed to emit cascade metrics", e);
      332      } finally {
      333        if (manageMDC) {
      334          restoreMDC(MDC_CASCADE_OPERATION_ID, prevOperationId);
      335          restoreMDC(MDC_CASCADE_TRIGGER_URN, prevTriggerUrn);
      336          restoreMDC(MDC_CASCADE_OPERATION_TYPE, prevOperationType);
      337        }
      338      }
      339    }
