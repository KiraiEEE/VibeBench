      287      private void fillViewFromFilter(final String inputFilter, final boolean forceAdvanced) {
      288          includeInconclusiveFilterCheckbox.setChecked(false);
      289          inverseFilterCheckbox.setChecked(false);
      290          andOrFilterCheckbox.setChecked(false);
      291  
      292          boolean setAdvanced = false;
      293          if (inputFilter != null) {
      294              try {
      295                  final List<IFilterViewHolder<?>> filterList = new ArrayList<>();
      296                  final GeocacheFilter filter = GeocacheFilter.checkConfig(inputFilter);
      297                  FilterUtils.setFilterText(binding.filterStorageName, filter.getNameForUserDisplay(), filter.isSavedDifferently());
      298                  includeInconclusiveFilterCheckbox.setChecked(filter.isIncludeInconclusive());
      299                  setAdvanced = filter.isOpenInAdvancedMode();
      300                  IGeocacheFilter filterTree = filter.getTree();
      301                  if (filterTree instanceof NotGeocacheFilter) {
      302                      inverseFilterCheckbox.setChecked(true);
      303                      filterTree = filterTree.getChildren().get(0);
      304                  }
      305                  if (filterTree instanceof LogicalGeocacheFilter) {
      306                      andOrFilterCheckbox.setChecked(filterTree instanceof OrGeocacheFilter);
      307                      for (IGeocacheFilter c : filterTree.getChildren()) {
      308                          filterList.add(FilterViewHolderCreator.createFor(c, this));
      309                      }
      310                  } else if (filterTree instanceof BaseGeocacheFilter) {
      311                      filterList.add(FilterViewHolderCreator.createFor(filterTree, this));
      312                  }
      313                  filterListAdapter.setItems(filterList);
      314                  adjustFilterEmptyView();
      315              } catch (ParseException pe) {
      316                  Log.w("Exception parsing input filter", pe);
      317              }
      318          }
      319  
      320  
      321          //set basic/advanced switch -> AFTER list is re-layouted
      322          if (!forceAdvanced && !setAdvanced && isBasicPossibleWithoutLoss()) {
      323              switchToBasic();
      324          } else {
      325              switchToAdvanced(false);
      326          }
      327      }

      393      private GeocacheFilter getFilterFromView() {
      394          IGeocacheFilter filter = null;
      395  
      396          if (filterListAdapter.getItemCount() > 0) {
      397              filter = andOrFilterCheckbox.isChecked() ? new OrGeocacheFilter() : new AndGeocacheFilter();
      398              for (IFilterViewHolder<?> f : filterListAdapter.getItems()) {
      399                  filter.addChild(FilterViewHolderCreator.createFrom(f));
      400              }
      401              if (inverseFilterCheckbox.isChecked()) {
      402                  final IGeocacheFilter notFilter = new NotGeocacheFilter();
      403                  notFilter.addChild(filter);
      404                  filter = notFilter;

      406          }

      408          return GeocacheFilter.create(
      409                  GeocacheFilter.getPurifiedFilterName(binding.filterStorageName.getText().toString()),
      410                  binding.filterBasicAdvanced.isChecked(),
