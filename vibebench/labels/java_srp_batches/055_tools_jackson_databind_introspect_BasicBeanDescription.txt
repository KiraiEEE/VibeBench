      272      public AnnotatedMember findAnySetterAccessor() throws IllegalArgumentException
      273      {
      274          if (_propCollector != null) {
      275              AnnotatedMethod anyMethod = _propCollector.getAnySetterMethod();
      276              if (anyMethod != null) {
      277                  // Also, let's be somewhat strict on how field name is to be
      278                  // passed; String, Object make sense, others not so much.
      279  
      280                  /* !!! 18-May-2009, tatu: how about enums? Can add support if
      281                   *  requested; easy enough for devs to add support within method.
      282                   */
      283                  Class<?> type = anyMethod.getRawParameterType(0);
      284                  if ((type != String.class) && (type != Object.class)) {
      285                      throw new IllegalArgumentException(String.format(
      286  "Invalid 'any-setter' annotation on method '%s()': first argument not of type String or Object, but %s",
      287  anyMethod.getName(), type.getName()));
      288                  }
      289                  return anyMethod;
      290              }
      291              AnnotatedMember anyField = _propCollector.getAnySetterField();
      292              if (anyField != null) {
      293                  // For now let's require a Map; in future can add support for other
      294                  // types like perhaps Iterable<Map.Entry>?
      295                  Class<?> type = anyField.getRawType();
      296                  if (!Map.class.isAssignableFrom(type)
      297                          && !JsonNode.class.isAssignableFrom(type)) {
      298                      throw new IllegalArgumentException(String.format(
      299  "Invalid 'any-setter' annotation on field '%s': type is not instance of `java.util.Map` or `JsonNode`",
      300  anyField.getName()));
      301                  }
      302                  return anyField;
      303              }
      304          }
      305          return null;
      306      }

      347      public Object instantiateBean(boolean fixAccess) {
      348          AnnotatedConstructor ac = _classInfo.getDefaultConstructor();
      349          if (ac == null) {
      350              return null;
      351          }
      352          if (fixAccess) {
      353              ac.fixAccess(_config.isEnabled(MapperFeature.OVERRIDE_PUBLIC_ACCESS_MODIFIERS));
      354          }
      355          try {
      356              return ac.call();
      357          } catch (Exception e) {
      358              Throwable t = e;
      359              while (t.getCause() != null) {
      360                  t = t.getCause();
      361              }
      362              ClassUtil.throwIfError(t);
      363              ClassUtil.throwIfRTE(t);
      364              throw new IllegalArgumentException("Failed to instantiate bean of type "+ClassUtil.nameOf(_classInfo.getAnnotated())
      365                      +": ("+t.getClass().getName()+") "+ClassUtil.exceptionMessage(t), t);
      366          }
