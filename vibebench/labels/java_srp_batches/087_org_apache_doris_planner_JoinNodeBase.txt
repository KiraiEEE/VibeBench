      102          if (slotDesc.isMaterialized()) {
      103              return true;
      104          }
      105          Expr child = smap.get(new SlotRef(slotDesc));
      106          if (child == null) {
      107              return false;

      110          child.collect(SlotRef.class, slotRefList);

      123          boolean copyRight = false;
      124          boolean leftNullable = false;
      125          boolean rightNullable = false;
      126          switch (joinOp) {
      127              case INNER_JOIN:

      130                  copyRight = true;
      131                  break;
      132              case LEFT_OUTER_JOIN:
      133                  copyLeft = true;
      134                  copyRight = true;
      135                  rightNullable = true;
      136                  break;
      137              case RIGHT_OUTER_JOIN:
      138                  copyLeft = true;
      139                  copyRight = true;
      140                  leftNullable = true;
      141                  break;
      142              case FULL_OUTER_JOIN:
      143                  copyLeft = true;
      144                  copyRight = true;
      145                  leftNullable = true;
      146                  rightNullable = true;
      147                  break;
      148              case LEFT_ANTI_JOIN:

      151                  copyLeft = true;
      152                  break;
      153              case RIGHT_ANTI_JOIN:
      154              case RIGHT_SEMI_JOIN:
      155                  copyRight = true;
      156                  break;
      157              default:
      158                  break;
      159          }
      160          ExprSubstitutionMap srcTblRefToOutputTupleSmap = new ExprSubstitutionMap();
      161          int leftNullableNumber = 0;
      162          int rightNullableNumber = 0;
      163          if (copyLeft) {
      164              //cross join do not have OutputTblRefIds
      165              List<TupleId> srcTupleIds = getChild(0) instanceof JoinNodeBase ? getChild(0).getOutputTupleIds()
      166                      : getChild(0).getOutputTblRefIds();
      167              for (TupleDescriptor leftTupleDesc : analyzer.getDescTbl().getTupleDesc(srcTupleIds)) {
      168                  // if the child is cross join node, the only way to get the correct nullable info of its output slots
      169                  // is to check if the output tuple ids are outer joined or not.
      170                  // then pass this nullable info to hash join node will be correct.
      171                  boolean needSetToNullable =
      172                          getChild(0) instanceof JoinNodeBase && analyzer.isOuterJoined(leftTupleDesc.getId());
      173                  for (SlotDescriptor leftSlotDesc : leftTupleDesc.getSlots()) {
      174                      if (!isMaterializedByChild(leftSlotDesc, getChild(0).getOutputSmap())) {
      175                          continue;
      176                      }
      177                      SlotDescriptor outputSlotDesc =
      178                              analyzer.getDescTbl().copySlotDescriptor(outputTupleDesc, leftSlotDesc);
      179                      if (leftNullable) {
      180                          outputSlotDesc.setIsNullable(true);
      181                          leftNullableNumber++;
      182                      }
      183                      if (needSetToNullable) {
      184                          outputSlotDesc.setIsNullable(true);
      185                      }
      186                      srcTblRefToOutputTupleSmap.put(new SlotRef(leftSlotDesc), new SlotRef(outputSlotDesc));
      187                  }
      188              }
      189          }
      190          if (copyRight) {
      191              List<TupleId> srcTupleIds = getChild(1) instanceof JoinNodeBase ? getChild(1).getOutputTupleIds()
      192                      : getChild(1).getOutputTblRefIds();
      193              for (TupleDescriptor rightTupleDesc : analyzer.getDescTbl().getTupleDesc(srcTupleIds)) {
      194                  boolean needSetToNullable =
      195                          getChild(1) instanceof JoinNodeBase && analyzer.isOuterJoined(rightTupleDesc.getId());
      196                  for (SlotDescriptor rightSlotDesc : rightTupleDesc.getSlots()) {
      197                      if (!isMaterializedByChild(rightSlotDesc, getChild(1).getOutputSmap())) {
      198                          continue;
      199                      }
      200                      SlotDescriptor outputSlotDesc =
      201                              analyzer.getDescTbl().copySlotDescriptor(outputTupleDesc, rightSlotDesc);
      202                      if (rightNullable) {
      203                          outputSlotDesc.setIsNullable(true);
      204                          rightNullableNumber++;
      205                      }
      206                      if (needSetToNullable) {
      207                          outputSlotDesc.setIsNullable(true);
      208                      }
      209                      srcTblRefToOutputTupleSmap.put(new SlotRef(rightSlotDesc), new SlotRef(outputSlotDesc));
      210                  }
      211              }
      212          }
      213  
      214          // add mark slot if needed
      215          if (isMarkJoin() && analyzer.needPopUpMarkTuple(innerRef)) {
      216              SlotDescriptor markSlot = analyzer.getMarkTuple(innerRef).getSlots().get(0);
      217              SlotDescriptor outputSlotDesc =
      218                      analyzer.getDescTbl().copySlotDescriptor(outputTupleDesc, markSlot);
      219              srcTblRefToOutputTupleSmap.put(new SlotRef(markSlot), new SlotRef(outputSlotDesc));
      220          }
      221  
      222          // 2. compute srcToOutputMap
      223          vSrcToOutputSMap = ExprSubstitutionMap.subtraction(outputSmap, srcTblRefToOutputTupleSmap, analyzer);
      224          for (int i = 0; i < vSrcToOutputSMap.size(); i++) {
      225              Preconditions.checkState(vSrcToOutputSMap.getRhs().get(i) instanceof SlotRef);
      226              SlotRef rSlotRef = (SlotRef) vSrcToOutputSMap.getRhs().get(i);
      227              if (vSrcToOutputSMap.getLhs().get(i) instanceof SlotRef) {
      228                  SlotRef lSlotRef = (SlotRef) vSrcToOutputSMap.getLhs().get(i);
      229                  rSlotRef.getDesc().setIsMaterialized(lSlotRef.getDesc().isMaterialized());
      230              } else {
      231                  rSlotRef.getDesc().setIsMaterialized(true);
      232                  rSlotRef.materializeSrcExpr();
      233              }
      234          }
      235  
      236          outputTupleDesc.computeStatAndMemLayout();
      237          // 3. add tupleisnull in null-side
      238          Preconditions.checkState(srcTblRefToOutputTupleSmap.getLhs().size() == vSrcToOutputSMap.getLhs().size());
      239          // Condition1: the left child is null-side
      240          // Condition2: the left child is a inline view
      241          // Then: add tuple is null in left child columns
      242          if (leftNullable && getChild(0).getTblRefIds().size() == 1
      243                  && analyzer.isInlineView(getChild(0).getTblRefIds().get(0))) {
      244              List<Expr> tupleIsNullLhs = TupleIsNullPredicate.wrapExprs(
      245                      vSrcToOutputSMap.getLhs().subList(0, leftNullableNumber), new ArrayList<>(), TNullSide.LEFT,
      246                      analyzer);
      247              tupleIsNullLhs
      248                      .addAll(vSrcToOutputSMap.getLhs().subList(leftNullableNumber, vSrcToOutputSMap.getLhs().size()));

      250          }
      251          // Condition1: the right child is null-side
      252          // Condition2: the right child is a inline view

      307      public void projectOutputTuple() {
      308          if (outputTupleDesc == null) {
      309              return;
      310          }
      311          if (outputTupleDesc.getSlots().size() == outputSlotIds.size()) {
      312              return;
      313          }
      314          Iterator<SlotDescriptor> iterator = outputTupleDesc.getSlots().iterator();
      315          while (iterator.hasNext()) {
      316              SlotDescriptor slotDescriptor = iterator.next();
      317              boolean keep = false;
      318              for (SlotId outputSlotId : outputSlotIds) {
      319                  if (slotDescriptor.getId().equals(outputSlotId)) {
      320                      keep = true;
      321                      break;
      322                  }
      323              }
      324              if (!keep) {
      325                  iterator.remove();
      326                  SlotRef slotRef = new SlotRef(slotDescriptor);
      327                  vSrcToOutputSMap.removeByRhsExpr(slotRef);
      328              }
      329          }
      330          outputTupleDesc.computeStatAndMemLayout();
