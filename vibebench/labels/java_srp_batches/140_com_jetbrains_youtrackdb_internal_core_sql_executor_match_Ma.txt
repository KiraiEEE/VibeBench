       64    /**
       65     * Executes the multi-step traversal pipeline. Each sub-item in the
       66     * {@link SQLMultiMatchPathItem} is applied sequentially, with each stage's output
       67     * feeding the next stage's input.
       68     *
       69     * @param startingPoint    the record to start the multi-step traversal from
       70     * @param iCommandContext  the command execution context
       71     * @return a stream of records produced by the final stage of the pipeline
       72     */
       73    @Override
       74    protected ExecutionStream traversePatternEdge(
       75        Result startingPoint, CommandContext iCommandContext) {
       76      assert MatchAssertions.checkNotNull(startingPoint, "startingPoint");
       77      assert MatchAssertions.checkNotNull(iCommandContext, "command context");
       78  
       79      Iterable<Identifiable> possibleResults = null;
       80  
       81      var item = (SQLMultiMatchPathItem) this.item;
       82      List<ResultInternal> result = new ArrayList<>();
       83  
       84      // The "pipeline" — starts with the single starting point
       85      List<Result> nextStep = new ArrayList<>();
       86      nextStep.add(startingPoint);
       87  
       88      var db = iCommandContext.getDatabaseSession();
       89      var oldCurrent = iCommandContext.getVariable("$current");
       90  
       91      // Process each sub-item in sequence
       92      for (var sub : item.getItems()) {
       93        List<ResultInternal> rightSide = new ArrayList<>();
       94        for (var o : nextStep) {
       95          var whileCond =
       96              sub.getFilter() == null ? null : sub.getFilter().getWhileCondition();
       97  
       98          var method = sub.getMethod();
       99          // SQLMatchPathItemFirst uses a function instead of a method
      100          if (sub instanceof SQLMatchPathItemFirst sqlMatchPathItemFirst) {
      101            method = sqlMatchPathItemFirst.getFunction().toMethod();
      102          }
      103  
      104          if (whileCond != null) {
      105            // Delegate to a standard MatchEdgeTraverser for recursive WHILE expansion
      106            var subtraverser = new MatchEdgeTraverser(null, sub);
      107            var rightStream =
      108                subtraverser.executeTraversal(iCommandContext, sub, o, 0,
      109                    null);
      110            while (rightStream.hasNext(iCommandContext)) {
      111              rightSide.add((ResultInternal) rightStream.next(iCommandContext));
      112            }
      113  
      114          } else {
      115            // Simple single-hop expansion via the sub-item's method
      116            iCommandContext.setVariable("$current", o);
      117            var nextSteps = method.execute(o, possibleResults, iCommandContext);
      118  
      119            // Normalize the heterogeneous return types into ResultInternal, applying
      120            // the sub-item's WHERE filter to each candidate
      121            dispatchTraversalResult(

      123          }
      124        }
      125        // The right side of this stage becomes the left side of the next stage
      126        //noinspection unchecked,rawtypes
      127        nextStep = (List) rightSide;

      129      }

      135    /**
      136     * Dispatches a raw traversal result into the appropriate conversion path based on its
      137     * runtime type, applies the sub-item's WHERE filter, and adds passing records to the
      138     * right side of the pipeline.
      139     *
      140     * <p>The method handles six possible return types from a traversal method:
      141     * {@link Collection}, {@link Identifiable}, {@link Relation}, {@link ResultInternal},
      142     * {@link Iterable} (non-Collection), and {@link Iterator}.
      143     *
      144     * @param nextSteps        the raw result from method.execute()
      145     * @param db               the database session for wrapping records
      146     * @param filter           the sub-item's match filter (may be null)
      147     * @param iCommandContext  the command execution context
      148     * @param rightSide        accumulator for matching results
      149     */
      150    static void dispatchTraversalResult(
      151        Object nextSteps,
      152        DatabaseSessionEmbedded db,
      153        SQLMatchFilter filter,
      154        CommandContext iCommandContext,
      155        List<ResultInternal> rightSide) {
      156      assert MatchAssertions.checkNotNull(db, "database session");
      157      assert MatchAssertions.checkNotNull(rightSide, "right side accumulator");
      158  
      159      if (nextSteps instanceof Collection) {
      160        ((Collection<?>) nextSteps)
      161            .stream()
      162            .map(obj -> toOResultInternal(db, obj))
      163            .filter(
      164                x ->
      165                    matchesCondition(x, filter, iCommandContext))
      166            .forEach(rightSide::add);
      167      } else if (nextSteps instanceof Identifiable identifiable) {
      168        var res = new ResultInternal(db, identifiable);
      169        if (matchesCondition(res, filter, iCommandContext)) {
      170          rightSide.add(res);
      171        }
      172      } else if (nextSteps instanceof Relation<?> bidirectionalLink) {
      173        var res = new ResultInternal(db, bidirectionalLink);
      174        if (matchesCondition(res, filter, iCommandContext)) {
      175          rightSide.add(res);
      176        }
      177      } else if (nextSteps instanceof ResultInternal resultInternal) {
      178        if (matchesCondition(resultInternal, filter, iCommandContext)) {
      179          rightSide.add(resultInternal);
      180        }
      181      } else if (nextSteps instanceof Iterable) {
      182        for (var step : (Iterable<?>) nextSteps) {
      183          var converted = toOResultInternal(db, step);
      184          if (matchesCondition(converted, filter, iCommandContext)) {
      185            rightSide.add(converted);
      186          }
      187        }
      188      } else if (nextSteps instanceof Iterator<?> iterator) {
      189        while (iterator.hasNext()) {
      190          var converted = toOResultInternal(db, iterator.next());
      191          if (matchesCondition(converted, filter, iCommandContext)) {
      192            rightSide.add(converted);
      193          }
      194        }
      195      }
      196    }

      198    /**
      199     * Evaluates the sub-item's WHERE filter against a candidate record.
      200     *
      201     * @return `true` if no filter is defined or the record passes the filter
      202     */
      203    static boolean matchesCondition(ResultInternal x, SQLMatchFilter filter,
      204        CommandContext ctx) {
      205      assert MatchAssertions.checkNotNull(x, "candidate record");
      206  
      207      if (filter == null) {
      208        return true;
      209      }
      210      var where = filter.getFilter();
      211      if (where == null) {
      212        return true;
      213      }
      214      return where.matchesFilters(x, ctx);
      215    }

      217    /**
      218     * Converts a raw traversal result object into a {@link ResultInternal}.
      219     *
      220     * @throws CommandExecutionException if the object type is unrecognized
      221     */
      222    static ResultInternal toOResultInternal(DatabaseSessionEmbedded session, Object x) {
      223      assert MatchAssertions.checkNotNull(session, "database session");
      224      assert MatchAssertions.checkNotNull(x, "traversal result");
      225  
      226      if (x instanceof ResultInternal resultInternal) {
      227        return resultInternal;
      228      } else if (x instanceof Identifiable identifiable) {
      229        return new ResultInternal(session, identifiable);
      230      } else if (x instanceof Relation<?> bidirectionalLink) {
      231        return new ResultInternal(session, bidirectionalLink);
      232      }
      233      throw new CommandExecutionException(session, "Cannot execute traversal on " + x);
      234    }
