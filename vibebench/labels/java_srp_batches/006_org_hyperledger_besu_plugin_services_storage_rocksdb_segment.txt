      184     * Create a Column Family Descriptor for a given segment It defines basically the different
      185     * options to apply to the corresponding Column Family
      186     *
      187     * @param segment the segment identifier
      188     * @param configuration RocksDB configuration

      191    private ColumnFamilyDescriptor createColumnDescriptor(
      192        final SegmentIdentifier segment, final RocksDBConfiguration configuration) {
      193      boolean dynamicLevelBytes = true;
      194      try {
      195        ConfigOptions configOptions = new ConfigOptions();
      196        DBOptions dbOptions = new DBOptions();
      197        List<ColumnFamilyDescriptor> cfDescriptors = new ArrayList<>();
      198  
      199        String latestOptionsFileName =
      200            OptionsUtil.getLatestOptionsFileName(
      201                configuration.getDatabaseDir().toString(), Env.getDefault());
      202        LOG.trace("Latest OPTIONS file detected: " + latestOptionsFileName);
      203  
      204        String optionsFilePath =
      205            configuration.getDatabaseDir().toString() + "/" + latestOptionsFileName;
      206        OptionsUtil.loadOptionsFromFile(configOptions, optionsFilePath, dbOptions, cfDescriptors);
      207  
      208        LOG.trace("RocksDB options loaded successfully from: " + optionsFilePath);
      209  
      210        if (!cfDescriptors.isEmpty()) {
      211          Optional<ColumnFamilyOptions> matchedCfOptions = Optional.empty();
      212          for (ColumnFamilyDescriptor descriptor : cfDescriptors) {
      213            if (Arrays.equals(descriptor.getName(), segment.getId())) {
      214              matchedCfOptions = Optional.of(descriptor.getOptions());
      215              break;
      216            }
      217          }
      218          if (matchedCfOptions.isPresent()) {
      219            dynamicLevelBytes = matchedCfOptions.get().levelCompactionDynamicLevelBytes();
      220            LOG.trace("dynamicLevelBytes is set to an existing value : " + dynamicLevelBytes);
      221          }
      222        }
      223      } catch (RocksDBException ex) {
      224        // Options file is not found in the database
      225      }
      226      BlockBasedTableConfig basedTableConfig = createBlockBasedTableConfig(segment, configuration);
      227  
      228      final var options =
      229          new ColumnFamilyOptions()
      230              .setTtl(0)
      231              .setCompressionType(CompressionType.LZ4_COMPRESSION)
      232              .setTableFormatConfig(basedTableConfig)
      233              .setLevelCompactionDynamicLevelBytes(dynamicLevelBytes);
      234      if (segment.containsStaticData()) {
      235        configureBlobDBForSegment(segment, configuration, options);
      236      }
      237  
      238      return new ColumnFamilyDescriptor(segment.getId(), options);
      239    }

      241    private static void configureBlobDBForSegment(
      242        final SegmentIdentifier segment,
      243        final RocksDBConfiguration configuration,
      244        final ColumnFamilyOptions options) {
      245      options
      246          .setEnableBlobFiles(true)
      247          .setEnableBlobGarbageCollection(
      248              isStaticDataGarbageCollectionEnabled(segment, configuration))
      249          .setMinBlobSize(100)
      250          .setBlobCompressionType(CompressionType.LZ4_COMPRESSION);
      251      if (configuration.getBlobGarbageCollectionAgeCutoff().isPresent()) {
      252        // fraction of file age to be considered eligible for GC;
      253        // 0.25 = oldest 25% of files eligible;
      254        // 1 = all files eligible
      255        options.setBlobGarbageCollectionAgeCutoff(
      256            configuration.getBlobGarbageCollectionAgeCutoff().get());
      257      }
      258      if (configuration.getBlobGarbageCollectionForceThreshold().isPresent()) {
      259        // fraction of garbage in eligible blob files to trigger GC;
      260        // 1 = trigger when eligible file contains 100% garbage;
      261        // 0 = trigger for all eligible files;
      262        options.setBlobGarbageCollectionForceThreshold(
      263            configuration.getBlobGarbageCollectionForceThreshold().get());
      264      }
      265    }

      322    /**
      323     * Parse RocksDBException and wrap in StorageException
      324     *
      325     * @param ex RocksDBException
      326     * @param defaultSegments segments requested to open
      327     * @param ignorableSegments segments which are ignorable if not present
      328     * @return StorageException wrapping the RocksDB Exception

      330    protected static StorageException parseRocksDBException(
      331        final RocksDBException ex,
      332        final List<SegmentIdentifier> defaultSegments,
      333        final List<SegmentIdentifier> ignorableSegments) {
      334      String message = ex.getMessage();
      335      List<SegmentIdentifier> knownSegments =
      336          Streams.concat(defaultSegments.stream(), ignorableSegments.stream()).distinct().toList();
      337  
      338      // parse out unprintable segment names for a more useful exception:
      339      String columnExceptionMessagePrefix = "Column families not opened: ";
      340      if (message.contains(columnExceptionMessagePrefix)) {
      341        String substring = message.substring(message.indexOf(": ") + 2);
      342  
      343        List<String> unHandledSegments = new ArrayList<>();
      344        Splitter.on(", ")
      345            .splitToStream(substring)
      346            .forEach(
      347                part -> {
      348                  byte[] bytes = part.getBytes(StandardCharsets.UTF_8);
      349                  unHandledSegments.add(
      350                      knownSegments.stream()
      351                          .filter(seg -> Arrays.equals(seg.getId(), bytes))
      352                          .findFirst()
      353                          .map(seg -> new SegmentRecord(seg.getName(), seg.getId()))
      354                          .orElse(new SegmentRecord(part, bytes))
      355                          .forDisplay());
      356                });
      357  
      358        return new StorageException(
      359            "RocksDBException: Unhandled column families: ["
      360                + unHandledSegments.stream().collect(Collectors.joining(", "))
      361                + "]");
      362      } else {
      363        return new StorageException(ex);
      364      }
      365    }

      371    void initColumnHandles() throws RocksDBException {
      372      // will not include the DEFAULT columnHandle, we do not use it:
      373      columnHandlesBySegmentIdentifier =
      374          trimmedSegments.stream()
      375              .collect(
      376                  Collectors.toMap(
      377                      segmentId -> segmentId,
      378                      segment -> {
      379                        var columnHandle =
      380                            columnHandles.stream()
      381                                .filter(
      382                                    ch -> {
      383                                      try {
      384                                        return Arrays.equals(ch.getName(), segment.getId());
      385                                      } catch (RocksDBException e) {
      386                                        throw new RuntimeException(e);
      387                                      }
      388                                    })
      389                                .findFirst()
      390                                .orElseThrow(
      391                                    () ->
      392                                        new RuntimeException(
      393                                            "Column handle not found for segment "
      394                                                + segment.getName()));
      395                        return new RocksDbSegmentIdentifier(getDB(), columnHandle);
      396                      }));
      397    }

      430      try (final OperationTimer.TimingContext ignored = metrics.getMultiReadLatency().startTimer()) {
      431        List<byte[]> rawResult =
      432            getDB().multiGetAsList(readOptions, Collections.nCopies(keys.size(), columnHandle), keys);
      433        if (rawResult == null) {
      434          return Collections.nCopies(keys.size(), Optional.empty());
      435        }
      436        List<Optional<byte[]>> result = new ArrayList<>(rawResult.size());
      437        for (byte[] value : rawResult) {
      438          result.add(Optional.ofNullable(value));
      439        }
      440        return result;
      441      } catch (final RocksDBException e) {
      442        throw new StorageException(e);
