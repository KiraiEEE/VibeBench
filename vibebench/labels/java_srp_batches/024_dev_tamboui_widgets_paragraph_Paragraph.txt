      141          if (maxWidth <= 0) {

      145          switch (overflow) {

      162      private List<Line> clipLines(List<Line> lines, int maxWidth) {
      163          List<Line> result = new ArrayList<>();
      164  
      165          for (Line line : lines) {
      166              if (line.width() <= maxWidth) {
      167                  result.add(line);
      168                  continue;
      169              }
      170  
      171              // Need to clip - preserve spans up to maxWidth
      172              List<Span> clippedSpans = new ArrayList<>();
      173              int remainingWidth = maxWidth;
      174  
      175              for (Span span : line.spans()) {
      176                  if (remainingWidth <= 0) {
      177                      break;
      178                  }
      179  
      180                  String content = span.content();
      181                  int spanWidth = CharWidth.of(content);
      182                  if (spanWidth <= remainingWidth) {
      183                      clippedSpans.add(span);
      184                      remainingWidth -= spanWidth;
      185                  } else {
      186                      // Partial span - truncate content by display width
      187                      String clipped = CharWidth.substringByWidth(content, remainingWidth);
      188                      if (!clipped.isEmpty()) {

      190                      }
      191                      break;
      192                  }
      193              }
      194  
      195              result.add(Line.from(clippedSpans));
      196          }
