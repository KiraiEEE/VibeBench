       30      public static String simplePrint(GraphQLType type) {
       31          Assert.assertNotNull(type, "type can't be null");
       32          if (isNonNull(type)) {
       33              return simplePrint(unwrapOne(type)) + "!";
       34          } else if (isList(type)) {
       35              return "[" + simplePrint(unwrapOne(type)) + "]";
       36          }
       37          return ((GraphQLNamedType) type).getName();
       38      }

      213      private static GraphQLType unwrapAllImpl(GraphQLType type) {
      214          while (true) {
      215              if (isNotWrapped(type)) {
      216                  return type;
      217              }
      218              type = unwrapOne(type);
      219          }

      224       * Unwraps a single non-nullable layer of the type if its present.  Note there can
      225       * only ever be one non-nullable wrapping of a type and this is enforced by {@link GraphQLNonNull}

      231      public static GraphQLType unwrapNonNull(GraphQLType type) {
      232          // its illegal to have a type that is a non-null wrapping a non-null type
      233          // and GraphQLNonNull has code that prevents it so we can just check once during the unwrapping
      234          if (isNonNull(type)) {
      235              // is cheaper doing this direct rather than calling #unwrapOne
      236              type = ((GraphQLNonNull) type).getWrappedType();
      237          }
      238          return type;

      255      /**
      256       * graphql types can be wrapped in {@link GraphQLNonNull} and {@link GraphQLList} type wrappers
      257       * so this method will unwrap the type down to the raw unwrapped type and return that wrapping
      258       * as a stack, with the top of the stack being the raw underling type.
      259       *
      260       * @param type the type to unwrap
      261       *
      262       * @return a stack of the type wrapping which will be at least 1 later deep
      263       */
      264      public static Stack<GraphQLType> unwrapType(GraphQLType type) {
      265          assertNotNull(type);
      266          Stack<GraphQLType> decoration = new Stack<>();
      267          while (true) {
      268              decoration.push(type);
      269              if (isNotWrapped(type)) {
      270                  break;
      271              }
      272              type = unwrapOne(type);
      273          }
      274          return decoration;
      275      }
