     1929          protected void selectAllColumnsFromJoinPaths(Collection<JoinPath> allPaths,
     1930                                                       @Nullable
     1931                                                       Map<JoinPath, String> joinAliasOverride) {
     1932              if (CollectionUtils.isEmpty(allPaths)) {
     1933                  return;
     1934              }
     1935  
     1936              List<JoinPath> joinPaths = allPaths.stream().filter(jp -> jp.getJoinType().isFetch()).collect(Collectors.toList());
     1937              Collections.reverse(joinPaths);
     1938  
     1939              if (CollectionUtils.isEmpty(joinPaths)) {
     1940                  return;
     1941              }
     1942              for (JoinPath joinPath : joinPaths) {
     1943                  Association association = joinPath.getAssociation();
     1944                  if (association.isEmbedded()) {
     1945                      // joins on embedded don't make sense
     1946                      continue;
     1947                  }
     1948  
     1949                  PersistentEntity associatedEntity = association.getAssociatedEntity();
     1950                  NamingStrategy namingStrategy = getNamingStrategy(associatedEntity);
     1951  
     1952                  String joinAlias = joinAliasOverride == null ? getAliasName(joinPath) : joinAliasOverride.get(joinPath);
     1953                  Objects.requireNonNull(joinAlias);
     1954                  String joinPathAlias = getPathOnlyAliasName(joinPath);
     1955  
     1956                  query.append(COMMA);
     1957  
     1958                  boolean includeIdentity = association.isForeignKey();
     1959                  // in the case of a foreign key association the ID is not in the table,
     1960                  // so we need to retrieve it
     1961                  PersistentEntityUtils.traversePersistentProperties(associatedEntity, includeIdentity, true, (propertyAssociations, prop) -> {
     1962  
     1963                      String transformed = getDataTransformerReadValue(joinAlias, prop).orElse(null);
     1964                      String columnAlias = getColumnAlias(prop);
     1965                      String columnName;
     1966                      if (computePropertyPaths()) {
     1967                          columnName = getMappedName(namingStrategy, propertyAssociations, prop);
     1968                      } else {
     1969                          columnName = asPath(propertyAssociations, prop);
     1970                      }
     1971                      if (transformed != null) {
     1972                          query.append(transformed).append(AS_CLAUSE);
     1973                      } else {
     1974                          query
     1975                              .append(joinAlias)
     1976                              .append(DOT)
     1977                              .append(queryState.shouldEscape() ? quote(columnName) : columnName)
     1978                              .append(AS_CLAUSE);
     1979                      }
     1980                      if (StringUtils.isNotEmpty(columnAlias)) {
     1981                          query.append(columnAlias);
     1982                      } else {
     1983                          query.append(joinPathAlias).append(columnName);
     1984                      }
     1985                      query.append(COMMA);
     1986                  });
     1987                  query.setLength(query.length() - 1);

     1999          public void selectAllColumns(AnnotationMetadata annotationMetadata, PersistentEntity entity, @Nullable String alias) {
     2000              if (canUseWildcardForSelect(annotationMetadata, entity)) {
     2001                  selectAllColumns(query, alias);
     2002                  return;
     2003              }
     2004              boolean escape = shouldEscape(entity);
     2005              NamingStrategy namingStrategy = getNamingStrategy(entity);
     2006              int length = query.length();
     2007              PersistentEntityUtils.traversePersistentProperties(entity, (associations, property)
     2008                  -> appendProperty(query, associations, property, namingStrategy, alias, escape));
     2009              int newLength = query.length();
     2010              if (newLength == length) {
     2011                  selectAllColumns(query, alias);
     2012              } else {
     2013                  query.setLength(newLength - 1);
