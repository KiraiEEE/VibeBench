      189  
      190      final var functions = session.getMetadata().getFunctionLibrary().getFunctionNames();
      191      for (var fName : functions) {
      192        final var f = session.getMetadata().getFunctionLibrary().getFunction(session, fName);
      193  
      194        if (f.getLanguage() == null) {
      195          throw new ConfigurationException(session.getDatabaseName(),
      196              "Database function '" + fName + "' has no language");
      197        }
      198  
      199        if (f.getLanguage().equalsIgnoreCase(iLanguage)) {
      200          final var def = getFunctionDefinition(session, f);
      201          if (def != null) {
      202            code.append(def);
      203            code.append("\n");
      204          }
      205        }
      206      }
      207  
      208      return code.length() == 0 ? null : code.toString();

      251      if (dbManager == null) {
      252        // CREATE A NEW DATABASE SCRIPT MANAGER
      253        dbManager = new DatabaseScriptManager(this, db.getDatabaseName());
      254        final var prev = dbManagers.putIfAbsent(db.getDatabaseName(), dbManager);
      255        if (prev != null) {
      256          dbManager.close();
      257          // GET PREVIOUS ONE
      258          dbManager = prev;
      259        }
      260      }
      261

      373    public String throwErrorMessage(String dbName, final ScriptException e, final String lib) {
      374      var errorLineNumber = e.getLineNumber();
      375  
      376      if (errorLineNumber <= 0) {
      377        // FIX TO RHINO: SOMETIMES HAS THE LINE NUMBER INSIDE THE TEXT :-(
      378        final var excMessage = e.toString();
      379        final var pos = excMessage.indexOf("<Unknown Source>#");
      380        if (pos > -1) {
      381          final var end = excMessage.indexOf(')', pos + "<Unknown Source>#".length());
      382          var lineNumberAsString = excMessage.substring(pos + "<Unknown Source>#".length(), end);
      383          errorLineNumber = Integer.parseInt(lineNumberAsString);
      384        }
      385      }
      386  
      387      if (errorLineNumber <= 0) {
      388        throw new CommandScriptException(dbName,
      389            "Error on evaluation of the script library. Error: "
      390                + e.getMessage()
      391                + "\nScript library was:\n"
      392                + lib);
      393      } else {
      394        final var code = new StringBuilder();
      395        final var scanner = new Scanner(lib);
      396        try {
      397          scanner.useDelimiter("\n");
      398          String currentLine = null;
      399          var lastFunctionName = "unknown";
      400  
      401          for (var currentLineNumber = 1; scanner.hasNext(); currentLineNumber++) {
      402            currentLine = scanner.next();
      403            var pos = currentLine.indexOf("function");
      404            if (pos > -1) {
      405              final var words =
      406                  StringParser.getWords(
      407                      currentLine.substring(
      408                          Math.min(pos + "function".length() + 1, currentLine.length())),
      409                      " \r\n\t");
      410              if (words.length > 0 && words[0] != "(") {
      411                lastFunctionName = words[0];
      412              }
      413            }
      414  
      415            if (currentLineNumber == errorLineNumber)
      416            // APPEND X LINES BEFORE
      417            {
      418              code.append(String.format("%4d: >>> %s\n", currentLineNumber, currentLine));
      419            } else if (Math.abs(currentLineNumber - errorLineNumber) <= LINES_AROUND_ERROR)
      420            // AROUND: APPEND IT
      421            {
      422              code.append(String.format("%4d: %s\n", currentLineNumber, currentLine));
      423            }
