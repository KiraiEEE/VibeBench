      392       * Create a new OAuth consumer secret for the given client Id.
      393       *
      394       * @param consumerSecretDTO {@link OAuthConsumerSecretDTO} containing client Id, provided clientSecret,
      395       *                                                        description and expiry time
      396       * @return {@link OAuthConsumerSecretDTO} containing the created consumer secret information

      398       * store or if the operation is not supported.

      400      public OAuthConsumerSecretDTO createOAuthConsumerSecret(OAuthConsumerSecretDTO consumerSecretDTO)
      401              throws IdentityOAuthAdminException {
      402  
      403          if (!OAuth2Util.isMultipleClientSecretsEnabled()) {
      404              throw handleClientError(INVALID_REQUEST,
      405                      OAuthConstants.OPERATION_NOT_SUPPORTED_FOR_SINGLE_CLIENT_SECRET_MODE);
      406          }
      407          if (LOG.isDebugEnabled()) {
      408              LOG.debug("Creating a new OAuth consumer secret for consumer key: " + consumerSecretDTO.getClientId());
      409          }
      410          String consumerKey = consumerSecretDTO.getClientId();
      411          OAuthAppDO oAuthAppDO = validateOAuthAppExistence(consumerKey);
      412          // Validate the provided expiry time to check whether it's in the past.
      413          if (OAuth2Util.isClientSecretExpired(consumerSecretDTO.getExpiresAt())) {
      414              throw handleClientError(INVALID_REQUEST,
      415                      "The provided expiry time for the new client secret is in the past.");
      416          }
      417  
      418          // Build and persist the new consumer secret
      419          OAuthConsumerSecretDO consumerSecret = new OAuthConsumerSecretDO();
      420          consumerSecret.setSecretId(UUID.randomUUID().toString());
      421          consumerSecret.setDescription(consumerSecretDTO.getDescription());
      422          consumerSecret.setClientId(consumerKey);
      423          String providedSecret = consumerSecretDTO.getClientSecret();
      424          consumerSecret.setSecretValue(StringUtils.isNotBlank(providedSecret)
      425                  ? providedSecret
      426                  : OAuthUtil.getRandomNumberSecure());
      427          consumerSecret.setExpiresAt(consumerSecretDTO.getExpiresAt());
      428          OAuthAppDAO oAuthAppDAO = new OAuthAppDAO();
      429          // Persist the new secret and update the IDN_OAUTH_CONSUMER_APPS table with the new secret.
      430          oAuthAppDAO.rotateOAuthConsumerSecret(consumerSecret, oAuthAppDO.getOauthConsumerSecret());
      431          // Clear the cache after adding a new secret. This is because the latest secret is added to the
      432          // IDN_OAUTH_CONSUMER_APPS table as the oauthConsumerSecret column value and the cache is populated from
      433          // this column. Therefore, to avoid having a stale value in the cache, we need to clear the cache entry
      434          // here.
      435          AppInfoCache.getInstance().clearCacheEntry(consumerKey);
      436          // A new secret was added, so clear the cache for this consumer key.
      437          // This ensures that the next authentication request will rebuild the cache
      438          // including the new secret, keeping the cache consistent with DB state.
      439          OAuthClientSecretsCache.getInstance().clearCacheEntry(consumerKey);
      440          return OAuthUtil.buildConsumerSecretDTO(consumerSecret);
      441      }
