      317      public void convertSuperEntityColumns(Class<?> clazz) {
      318          List<Field> fields = TableInfoHelper.getAllFields(clazz, annotationHandler);
      319          this.superEntityColumns.addAll(fields.stream().map(field -> {
      320              TableId tableId = annotationHandler.getAnnotation(field, TableId.class);
      321              if (tableId != null && StringUtils.isNotBlank(tableId.value())) {
      322                  return tableId.value();
      323              }
      324              TableField tableField = annotationHandler.getAnnotation(field, TableField.class);
      325              if (tableField != null && StringUtils.isNotBlank(tableField.value())) {
      326                  return tableField.value();
      327              }
      328              if (null == columnNaming || columnNaming == NamingStrategy.no_change) {
      329                  return field.getName();
      330              }
      331              return StringUtils.camelToUnderline(field.getName());
      332          }).collect(Collectors.toSet()));
      333      }
