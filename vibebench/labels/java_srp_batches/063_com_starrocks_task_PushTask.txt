      145      public TPushReq toThrift() {
      146          TPushReq request = new TPushReq(tabletId, schemaHash, version, 0, timeoutSecond, pushType);
      147          if (taskType == TTaskType.REALTIME_PUSH) {
      148              request.setPartition_id(partitionId);
      149              request.setTransaction_id(transactionId);
      150              request.setTimezone(timezone);
      151          }
      152          request.setIs_schema_changing(isSchemaChanging);
      153          switch (pushType) {
      154              case LOAD:
      155              case LOAD_DELETE:
      156                  Preconditions.checkArgument(false, "LOAD and LOAD_DELETE not supported");
      157                  break;
      158              case DELETE:
      159                  List<TCondition> tConditions = new ArrayList<TCondition>();
      160                  for (Predicate condition : conditions) {
      161                      TCondition tCondition = new TCondition();
      162                      ArrayList<String> conditionValues = new ArrayList<String>();
      163                      if (condition instanceof BinaryPredicate) {
      164                          BinaryPredicate binaryPredicate = (BinaryPredicate) condition;
      165                          String columnName = ((SlotRef) binaryPredicate.getChild(0)).getColumnName();
      166                          String value = ((LiteralExpr) binaryPredicate.getChild(1)).getStringValue();
      167                          BinaryType op = binaryPredicate.getOp();
      168                          tCondition.setColumn_name(MetaUtils.getColumnByColumnName(dbId, tableId, columnName)
      169                                  .getColumnId().getId());
      170                          tCondition.setCondition_op(op.toString());
      171                          conditionValues.add(value);
      172                      } else if (condition instanceof IsNullPredicate) {
      173                          IsNullPredicate isNullPredicate = (IsNullPredicate) condition;
      174                          String columnName = ((SlotRef) isNullPredicate.getChild(0)).getColumnName();
      175                          String op = "IS";
      176                          String value = "NULL";
      177                          if (isNullPredicate.isNotNull()) {
      178                              value = "NOT NULL";
      179                          }
      180                          tCondition.setColumn_name(MetaUtils.getColumnByColumnName(dbId, tableId, columnName)
      181                                  .getColumnId().getId());
      182                          tCondition.setCondition_op(op);
      183                          conditionValues.add(value);
      184                      } else if (condition instanceof InPredicate) {
      185                          InPredicate inPredicate = (InPredicate) condition;
      186                          String columnName = ((SlotRef) inPredicate.getChild(0)).getColumnName();
      187                          String op = inPredicate.isNotIn() ? "!*=" : "*=";
      188                          tCondition.setColumn_name(MetaUtils.getColumnByColumnName(dbId, tableId, columnName)
      189                                  .getColumnId().getId());
      190                          tCondition.setCondition_op(op);
      191                          for (int i = 1; i <= inPredicate.getInElementNum(); i++) {
      192                              conditionValues.add(((LiteralExpr) inPredicate.getChild(i)).getStringValue());
      193                          }
      194                      }
      195  
      196                      tCondition.setCondition_values(conditionValues);
      197  
      198                      tConditions.add(tCondition);
      199                  }
      200                  request.setDelete_conditions(tConditions);
      201                  break;
      202              case LOAD_V2:
      203                  request.setBroker_scan_range(tBrokerScanRange);
      204                  request.setDesc_tbl(tDescriptorTable);
      205                  request.setTablet_type(tabletType);
      206                  break;
      207              case CANCEL_DELETE:
      208                  request.setTransaction_id(transactionId);
      209                  break;
      210              default:
      211                  LOG.warn("unknown push type. type: " + pushType.name());
      212                  break;
      213          }
      214          request.setColumns_desc(columnsDesc);
      215  
      216          return request;
      217      }
