      128          if (catalog == null) {

      130          }

      133          }

      135          DatabaseIf db = catalog.getDbNullable(dbName);

      157      public void replayRefreshTable(ExternalObjectLog log) {
      158          ExternalCatalog catalog = (ExternalCatalog) Env.getCurrentEnv().getCatalogMgr().getCatalog(log.getCatalogId());
      159          if (catalog == null) {
      160              LOG.warn("failed to find catalog when replaying refresh table: {}", log.debugForRefreshTable());
      161              return;
      162          }
      163          Optional<ExternalDatabase<? extends ExternalTable>> db;
      164          if (!Strings.isNullOrEmpty(log.getDbName())) {
      165              db = catalog.getDbForReplay(log.getDbName());
      166          } else {
      167              db = catalog.getDbForReplay(log.getDbId());
      168          }
      169          // See comment in refreshDbInternal for why db and table may be null.
      170          if (!db.isPresent()) {
      171              LOG.warn("failed to find db when replaying refresh table: {}", log.debugForRefreshTable());
      172              return;
      173          }
      174          Optional<? extends ExternalTable> table;
      175          if (!Strings.isNullOrEmpty(log.getTableName())) {
      176              table = db.get().getTableForReplay(log.getTableName());
      177          } else {
      178              table = db.get().getTableForReplay(log.getTableId());
      179          }
      180          if (!table.isPresent()) {
      181              LOG.warn("failed to find table when replaying refresh table: {}", log.debugForRefreshTable());
      182              return;
      183          }
      184          if (!Strings.isNullOrEmpty(log.getNewTableName())) {
      185              // this is a rename table op
      186              db.get().unregisterTable(log.getTableName());
      187              db.get().resetMetaCacheNames();
      188          } else {
      189              List<String> modifiedPartNames = log.getPartitionNames();
      190              List<String> newPartNames = log.getNewPartitionNames();
      191              if (catalog instanceof HMSExternalCatalog
      192                      && ((modifiedPartNames != null && !modifiedPartNames.isEmpty())
      193                      || (newPartNames != null && !newPartNames.isEmpty()))) {
      194                  // Partition-level cache invalidation, only for hive catalog
      195                  HiveExternalMetaCache cache = Env.getCurrentEnv().getExtMetaCacheMgr()
      196                          .hive(catalog.getId());
      197                  cache.refreshAffectedPartitionsCache((HMSExternalTable) table.get(), modifiedPartNames, newPartNames);
      198                  if (table.get() instanceof HMSExternalTable && log.getLastUpdateTime() > 0) {
      199                      ((HMSExternalTable) table.get()).setUpdateTime(log.getLastUpdateTime());
      200                  }
      201                  LOG.info("replay refresh partitions for table {}, "
      202                                  + "modified partitions count: {}, "
      203                                  + "new partitions count: {}",
      204                          table.get().getName(), modifiedPartNames == null ? 0 : modifiedPartNames.size(),
      205                          newPartNames == null ? 0 : newPartNames.size());
      206              } else {
      207                  // Full table cache invalidation
      208                  refreshTableInternal(db.get(), table.get(), log.getLastUpdateTime());
      209              }
      210          }
      211      }

      234      public void refreshTableInternal(ExternalDatabase db, ExternalTable table, long updateTime) {
      235          table.unsetObjectCreated();
      236          // Iceberg partition evolution can change partition specs across FEs.
      237          // Clear related-table validation cache to avoid stale partitioned/unpartitioned judgment.
      238          if (table instanceof IcebergExternalTable) {
      239              ((IcebergExternalTable) table).setIsValidRelatedTableCached(false);
      240          }
      241          if (updateTime > 0) {
      242              table.setUpdateTime(updateTime);
      243          }
      244          Env.getCurrentEnv().getExtMetaCacheMgr().invalidateTableCache(table);
      245          LOG.info("refresh table {}, id {} from db {} in catalog {}, update time: {}",
      246                  table.getName(), table.getId(), db.getFullName(), db.getCatalog().getName(), updateTime);
      247      }

      250      public void refreshPartitions(String catalogName, String dbName, String tableName,
      251              List<String> partitionNames, long updateTime, boolean ignoreIfNotExists)
      252              throws DdlException {
      253          CatalogIf catalog = Env.getCurrentEnv().getCatalogMgr().getCatalog(catalogName);
      254          if (catalog == null) {
      255              if (!ignoreIfNotExists) {
      256                  throw new DdlException("No catalog found with name: " + catalogName);
      257              }
      258              return;
      259          }
      260          if (!(catalog instanceof ExternalCatalog)) {
      261              throw new DdlException("Only support ExternalCatalog");
      262          }
      263          DatabaseIf db = catalog.getDbNullable(dbName);
      264          if (db == null) {
      265              if (!ignoreIfNotExists) {
      266                  throw new DdlException("Database " + dbName + " does not exist in catalog " + catalog.getName());
      267              }
      268              return;
      269          }
      270  
      271          TableIf table = db.getTableNullable(tableName);
      272          if (table == null) {
      273              if (!ignoreIfNotExists) {
      274                  throw new DdlException("Table " + tableName + " does not exist in db " + dbName);
      275              }
      276              return;
      277          }
      278  
      279          ExternalTable externalTable = (ExternalTable) table;
      280          HiveExternalMetaCache cache = Env.getCurrentEnv().getExtMetaCacheMgr().hive(externalTable.getCatalog().getId());
      281          for (String partitionName : partitionNames) {
      282              cache.invalidatePartitionCache(externalTable, partitionName);
      283          }
      284          ((HMSExternalTable) table).setUpdateTime(updateTime);
      285      }

      297      public void start() {
      298          RefreshTask refreshTask = new RefreshTask();
      299          this.refreshScheduler.scheduleAtFixedRate(refreshTask, 0, REFRESH_TIME_SEC,
      300                  TimeUnit.SECONDS);

      303      private class RefreshTask implements Runnable {
      304          @Override
      305          public void run() {
      306              for (Map.Entry<Long, Integer[]> entry : refreshMap.entrySet()) {
      307                  Long catalogId = entry.getKey();
      308                  Integer[] timeGroup = entry.getValue();
      309                  Integer original = timeGroup[0];
      310                  Integer current = timeGroup[1];
      311                  if (current - REFRESH_TIME_SEC > 0) {
      312                      timeGroup[1] = current - REFRESH_TIME_SEC;
      313                      refreshMap.put(catalogId, timeGroup);
      314                  } else {
      315                      CatalogIf catalog = Env.getCurrentEnv().getCatalogMgr().getCatalog(catalogId);
      316                      if (catalog != null) {
      317                          String catalogName = catalog.getName();
      318                          LOG.info("Scheduled refresh triggered for catalog {} (id={}), interval={}s, invalidCache=true",
      319                                  catalogName, catalogId, original);
      320                          /**
      321                           * Now do not invoke
      322                           * {@link org.apache.doris.analysis.RefreshCatalogStmt#analyze(Analyzer)} is ok,
      323                           * because the default value of invalidCache is true.
      324                           * */
      325                          try {
      326                              Env.getCurrentEnv().getRefreshManager().handleRefreshCatalog(catalogName, true);
      327                              LOG.info("Scheduled refresh completed for catalog {} (id={}), next refresh in {}s",
      328                                      catalogName, catalogId, original);
      329                          } catch (Exception e) {
      330                              LOG.warn("Failed to execute scheduled refresh for catalog {} (id={})",
      331                                      catalogName, catalogId, e);
      332                          }
      333  
      334                          // reset
      335                          timeGroup[1] = original;
      336                          refreshMap.put(catalogId, timeGroup);
      337                      } else {
      338                          LOG.warn("Scheduled refresh skipped: catalog id={} not found, removing from refresh map",
      339                                  catalogId);
      340                          refreshMap.remove(catalogId);
      341                      }
      342                  }
      343              }
      344          }
      345      }
