       92      @Scheduled(concurrentExecution = SKIP, every = "{apicurio.gitops.refresh.every}")
       93      void refresh() {
       94          if (registryStorageType.equals("gitops")) {
       95              log.debug("Running GitOps refresh. Active database is {} and state is {}.",
       96                      active == green ? "green" : "blue", state);
       97              switch (state) {
       98                  case READY_TO_SWITCH: {
       99                      try {
      100                          if (switchLock.writeLock().tryLock(5, TimeUnit.SECONDS)) {
      101                              var previous = active;
      102                              try {
      103                                  active = inactive;
      104                                  inactive = previous;
      105                              } finally {
      106                                  state = State.READY_TO_WRITE;
      107                                  switchLock.writeLock().unlock();
      108                                  log.info("GitOps update published");
      109                              }
      110                          }
      111                      } catch (InterruptedException e) {
      112                          throw new RuntimeException(e);
      113                      }
      114                  }
      115                      break;
      116                  case READY_TO_WRITE: {
      117                      try {
      118                          var updatedCommit = gitManager.poll();
      119                          if (!updatedCommit.equals(gitManager.getPreviousCommit())) {
      120                              // TODO Delete *all* data
      121                              // TODO Improve performance by remembering when the storage is clean
      122                              inactive.deleteAllUserData();
      123  
      124                              var processingState = new ProcessingState(inactive);
      125                              gitManager.run(processingState, updatedCommit);
      126  
      127                              if (processingState.isSuccessful()) {
      128                                  log.info("GitOps update loaded successfully");
      129                                  gitManager.updateCurrentCommit(updatedCommit);
      130                                  state = State.READY_TO_SWITCH;
      131                              } else {
      132                                  log.error("GitOps update failed to load");
      133                                  processingState.getErrors().forEach(e -> {
      134                                      log.error("Error: {}", e);
      135                                  });
      136                              }
      137                          }
      138                      } catch (Exception e) {
      139                          throw new RuntimeException(e);
      140                      }
      141                  }
      142                      break;
      143              }
      144              log.debug("GitOps refresh finished. Active database is {} and state is {}.",
      145                      active == green ? "green" : "blue", state);
      146          }
      147      }

      149      public <T> T proxy(Function<RegistryStorage, T> operation) {
      150          try {
      151              if (switchLock.readLock().tryLock(5, TimeUnit.SECONDS)) {
      152                  try {
      153                      return operation.apply(active);
      154                  } finally {
      155                      switchLock.readLock().unlock();
      156                  }
      157              } else {
      158                  throw new RegistryStorageException(
      159                          "Could not acquire read lock to get the active storage within 5 seconds");
      160              }
      161          } catch (InterruptedException ex) {
      162              throw new RegistryStorageException("Could not acquire read lock to get the active storage", ex);
      163          }
      164      }
