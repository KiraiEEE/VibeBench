      128    public static IntegerEncodingResult encodeMortonCodes(
      129        List<Integer> values, PhysicalLevelTechnique physicalLevelTechnique) {
      130      var previousValue = 0;
      131      var deltaValues = new ArrayList<Integer>();
      132      for (var i = 0; i < values.size(); i++) {
      133        var value = values.get(i);
      134        var delta = value - previousValue;
      135        deltaValues.add(delta);
      136        previousValue = value;
      137      }
      138  
      139      var encodedValues =
      140          physicalLevelTechnique == PhysicalLevelTechnique.FAST_PFOR
      141              ? encodeFastPfor(deltaValues, false)
      142              : encodeVarint(
      143                  deltaValues.stream().mapToLong(i -> i).boxed().collect(Collectors.toList()), false);
      144  
      145      var result = new IntegerEncodingResult();
      146      result.logicalLevelTechnique1 = LogicalLevelTechnique.MORTON;
      147      result.logicalLevelTechnique2 = LogicalLevelTechnique.DELTA;
      148      result.physicalLevelEncodedValuesLength = values.size();
      149      result.numRuns = 0;
      150      result.encodedValues = encodedValues;
      151      return result;
      152    }

      158    public static IntegerEncodingResult encodeInt(
      159        List<Integer> values, PhysicalLevelTechnique physicalLevelTechnique, boolean isSigned) {
      160      var previousValue = 0;
      161      var previousDelta = 0;
      162      var runs = 1;
      163      var deltaRuns = 1;
      164      var deltaValues = new ArrayList<Integer>();
      165      for (var i = 0; i < values.size(); i++) {
      166        var value = values.get(i);
      167        var delta = value - previousValue;
      168        deltaValues.add(delta);
      169  
      170        if (value != previousValue && i != 0) {
      171          runs++;
      172        }
      173  
      174        if (delta != previousDelta && i != 0) {
      175          deltaRuns++;
      176        }
      177  
      178        previousValue = value;
      179        previousDelta = delta;
      180      }
      181  
      182      BiFunction<List<Integer>, Boolean, byte[]> encoder =
      183          physicalLevelTechnique == PhysicalLevelTechnique.FAST_PFOR
      184              ? IntegerEncoder::encodeFastPfor
      185              : (v, s) ->
      186                  encodeVarint(v.stream().mapToLong(i -> i).boxed().collect(Collectors.toList()), s);
      187  
      188      var plainEncodedValues = encoder.apply(values, isSigned);
      189      var deltaEncodedValues = encoder.apply(deltaValues, true);
      190      var encodedValues = Lists.newArrayList(plainEncodedValues, deltaEncodedValues);
      191      byte[] rleEncodedValues = null;
      192      byte[] deltaRleEncodedValues = null;
      193      var rlePhysicalLevelEncodedValuesLength = 0;
      194      var deltaRlePhysicalLevelEncodedValuesLength = 0;
      195  
      196      /* Use selection logic from BTR Blocks -> https://github.com/maxi-k/btrblocks/blob/c954ffd31f0873003dbc26bf1676ac460d7a3b05/btrblocks/scheme/double/RLE.cpp#L17 */
      197      /*
      198       * if there are ony a view values (e.g. 4 times 1) rle only produces the same size then other
      199       * encodings. Since we want to force that all const streams use RLE encoding we use this current
      200       * workaround
      201       */
      202      var isConstStream = false;
      203      if (values.size() / runs >= 2) {
      204        // TODO: get rid of conversion
      205        var rleValues = EncodingUtils.encodeRle(values.stream().mapToInt(i -> i).toArray());
      206        rlePhysicalLevelEncodedValuesLength =
      207            rleValues.getLeft().size() + rleValues.getRight().size();
      208        rleEncodedValues =
      209            encoder.apply(
      210                Stream.concat(
      211                        rleValues.getLeft().stream(),
      212                        isSigned
      213                            ? Arrays.stream(
      214                                    EncodingUtils.encodeZigZag(
      215                                        rleValues.getRight().stream().mapToInt(i -> i).toArray()))
      216                                .boxed()
      217                            : rleValues.getRight().stream())
      218                    .toList(),
      219                false);
      220        isConstStream = rleValues.getLeft().size() == 1;
      221      }
      222  
      223      if (deltaValues.size() / deltaRuns >= 2) {
      224        // TODO: get rid of conversion
      225        var deltaRleValues = EncodingUtils.encodeRle(deltaValues.stream().mapToInt(i -> i).toArray());
      226        deltaRlePhysicalLevelEncodedValuesLength =
      227            deltaRleValues.getLeft().size() + deltaRleValues.getRight().size();
      228        var zigZagDelta =
      229            EncodingUtils.encodeZigZag(deltaRleValues.getRight().stream().mapToInt(i -> i).toArray());
      230        // TODO: encode runs and length separate?
      231        deltaRleEncodedValues =
      232            encoder.apply(
      233                Stream.concat(deltaRleValues.getLeft().stream(), Arrays.stream(zigZagDelta).boxed())
      234                    .toList(),
      235                false);
      236      }
      237  
      238      encodedValues.add(rleEncodedValues);
      239      encodedValues.add(deltaRleEncodedValues);
      240  
      241      // TODO: refactor -> find proper solution
      242      var encodedValuesSizes =
      243          encodedValues.stream()
      244              .map(v -> v == null ? Integer.MAX_VALUE : v.length)
      245              .collect(Collectors.toList());
      246      var index =
      247          isConstStream
      248              ? LogicalLevelIntegerTechnique.RLE.ordinal()
      249              : encodedValuesSizes.indexOf(Collections.min(encodedValuesSizes));
      250      var encoding = LogicalLevelIntegerTechnique.values()[index];
      251  
      252      var result = new IntegerEncodingResult();
      253      result.encodedValues = encodedValues.get(index);
      254      result.physicalLevelEncodedValuesLength = values.size();
      255      if (encoding == LogicalLevelIntegerTechnique.RLE || isConstStream) {
      256        result.numRuns = runs;
      257        result.physicalLevelEncodedValuesLength = rlePhysicalLevelEncodedValuesLength;
      258        result.logicalLevelTechnique1 = LogicalLevelTechnique.RLE;
      259        result.logicalLevelTechnique2 = LogicalLevelTechnique.NONE;
      260      } else if (encoding == LogicalLevelIntegerTechnique.DELTA_RLE) {
      261        result.numRuns = deltaRuns;
      262        result.physicalLevelEncodedValuesLength = deltaRlePhysicalLevelEncodedValuesLength;
      263        result.logicalLevelTechnique1 = LogicalLevelTechnique.DELTA;
      264        result.logicalLevelTechnique2 = LogicalLevelTechnique.RLE;
      265      } else if (encoding == LogicalLevelIntegerTechnique.DELTA) {
      266        result.logicalLevelTechnique1 = LogicalLevelTechnique.DELTA;
      267        result.logicalLevelTechnique2 = LogicalLevelTechnique.NONE;
      268      } else {
      269        result.logicalLevelTechnique1 = LogicalLevelTechnique.NONE;
      270        result.logicalLevelTechnique2 = LogicalLevelTechnique.NONE;
      271      }
      272  
      273      return result;
      274    }
