      382      private void finishCreateReplica(AgentTask task, TFinishTaskRequest request) {
      383          // if we get here, this task will be removed from AgentTaskQueue for certain.
      384          // because in this function, the only problem that cause failure is meta missing.
      385          // and if meta is missing, we no longer need to resend this task

      388              if (request.getTask_status().getStatus_code() != TStatusCode.OK) {
      389                  createReplicaTask.countDownToZero(
      390                          task.getBackendId() + ": " + request.getTask_status().getError_msgs().toString());
      391              } else {
      392                  long tabletId = createReplicaTask.getTabletId();
      393                  if (request.isSetFinish_tablet_infos()) {
      394                      Replica replica = GlobalStateMgr.getCurrentState().getTabletInvertedIndex()
      395                              .getReplica(tabletId, createReplicaTask.getBackendId());
      396                      if (replica != null) {
      397                          replica.setPathHash(request.getFinish_tablet_infos().get(0).getPath_hash());
      398                          if (createReplicaTask.getRecoverySource() == RecoverySource.REPORT) {
      399                              /*
      400                               * This creates replica task is created by ReportHanlder,
      401                               * (See comment of Config.recover_with_empty_tablet)
      402                               * So we set replica back to normal state.
      403                               */
      404                              replica.setBad(false);
      405                              LOG.info(
      406                                      "finish recover create replica task. set replica to good. tablet {}, replica {}, backend {}",
      407                                      tabletId, task.getBackendId(), replica.getId());
      408                          }
      409                      }
      410                  } else if (!RunMode.isSharedDataMode()) { // finish_tablet_infos will not be set in shared data mode
      411                      LOG.warn("tablet_info is not set in finishTaskRequest");
      412                  }
      413  
      414                  // this should be called before 'countDownLatch()'
      415                  GlobalStateMgr.getCurrentState().getNodeMgr().getClusterInfo()
      416                          .updateBackendReportVersion(task.getBackendId(), request.getReport_version(), task.getDbId());
      417  
      418                  createReplicaTask.countDownLatch(task.getBackendId(), task.getSignature());
      419                  LOG.debug("finish create replica. tablet id: {}, be: {}, report version: {}, tablet type: {}",
      420                          tabletId, task.getBackendId(), request.getReport_version(), createReplicaTask.getTabletType());
      421              }
      422  
      423              if (createReplicaTask.getRecoverySource() == RecoverySource.SCHEDULER) {
      424                  GlobalStateMgr.getCurrentState().getTabletScheduler()
      425                          .finishCreateReplicaTask(createReplicaTask, request);
      426              }
      427          } finally {

      429          }
