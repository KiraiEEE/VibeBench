       51    // Pattern to match mention tag format: <#E.type.name|[@DisplayName](url)>
       52    // Supports both . and :: as separators
       53    private static final Pattern MENTION_TAG_PATTERN =
       54        Pattern.compile("<#E[.:][^|]+\\|\\[@([^\\]]+)\\]\\(([^)]+)\\)>");
       55  
       56    // Pattern to match consecutive/nested anchor tags (case 1)

       58    private static final Pattern CONSECUTIVE_ANCHORS_PATTERN =
       59        Pattern.compile("<a[^>]*href=\"([^\"]+)\"[^>]*></a>\\s*<a[^>]*href=\"\\1\"[^>]*>([^<]+)</a>");

       94    /**
       95     * Processes mention tags in the given text and converts them to clean HTML links.
       96     *
       97     * <p>Handles two mention formats from the UI:
       98     * <ol>
       99     *   <li>Nested/consecutive anchors: {@code <a href="url"></a><a href="url">@Name</a>}
      100     *   <li>Raw mention tags: {@code <#E.type.name|[@Name](url)>}
      101     * </ol>
      102     *
      103     * <p>All other HTML content (including regular links) is preserved as-is.
      104     *
      105     * @param text The text containing mention tags (may contain HTML entities)
      106     * @return Text with mentions converted to clean HTML links
      107     */
      108    private String processMentions(String text) {
      109      // Decode HTML entities first
      110      String result = StringEscapeUtils.unescapeHtml4(text);
      111  
      112      // Step 1: Remove data-type="mention" wrapper anchors
      113      // Example: <a data-type="mention" ...>CONTENT</a> -> CONTENT
      114      Matcher wrapperMatcher = MENTION_WRAPPER_PATTERN.matcher(result);
      115      StringBuilder unwrapped = new StringBuilder();
      116      while (wrapperMatcher.find()) {
      117        String content = wrapperMatcher.group(1);
      118        wrapperMatcher.appendReplacement(unwrapped, Matcher.quoteReplacement(content));
      119      }
      120      wrapperMatcher.appendTail(unwrapped);
      121      result = unwrapped.toString();
      122  
      123      // Step 2: Merge consecutive duplicate anchors (case 1)
      124      // Example: <a href="url"></a><a href="url">@Name</a> -> <a href="url">@Name</a>
      125      Matcher consecutiveMatcher = CONSECUTIVE_ANCHORS_PATTERN.matcher(result);
      126      StringBuilder merged = new StringBuilder();
      127      while (consecutiveMatcher.find()) {
      128        String url = consecutiveMatcher.group(1);
      129        String displayName = consecutiveMatcher.group(2);
      130        String replacement = String.format("<a href=\"%s\">%s</a>", url, displayName);
      131        consecutiveMatcher.appendReplacement(merged, Matcher.quoteReplacement(replacement));
      132      }
      133      consecutiveMatcher.appendTail(merged);
      134      result = merged.toString();
      135  
      136      // Step 3: Convert mention tags to clean HTML links (case 2)
      137      // Example: <#E.team.Legal Admin|[@Legal Admin](url)> -> <a href="url">@Legal Admin</a>
      138      Matcher mentionMatcher = MENTION_TAG_PATTERN.matcher(result);
      139      StringBuilder final_result = new StringBuilder();
      140      while (mentionMatcher.find()) {
      141        String displayName = mentionMatcher.group(1);
      142        String url = mentionMatcher.group(2);
      143        String replacement = String.format("<a href=\"%s\">@%s</a>", url, displayName);
      144        mentionMatcher.appendReplacement(final_result, Matcher.quoteReplacement(replacement));
      145      }
      146      mentionMatcher.appendTail(final_result);
      147  
      148      return final_result.toString();
      149    }
