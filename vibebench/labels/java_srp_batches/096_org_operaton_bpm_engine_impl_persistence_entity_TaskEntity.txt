      316    @Override
      317    public void complete() {
      318  
      319      if (TaskState.STATE_COMPLETED.equals(this.lifecycleState)
      320          || TaskListener.EVENTNAME_COMPLETE.equals(this.eventName)
      321          || TaskListener.EVENTNAME_DELETE.equals(this.eventName)) {
      322  
      323        throw LOG.invokeTaskListenerException(new IllegalStateException("invalid task state"));
      324      }
      325      // if the task is associated with a case
      326      // execution then call complete on the
      327      // associated case execution. The case
      328      // execution handles the completion of
      329      // the task.
      330      if (caseExecutionId != null) {
      331        getCaseExecution().manualComplete();
      332        return;
      333      }
      334  
      335      // in the other case:
      336  
      337      // ensure the the Task is not suspended
      338      ensureTaskActive();
      339  
      340      // trigger TaskListener.complete event
      341      final boolean shouldDeleteTask = transitionTo(TaskState.STATE_COMPLETED);
      342  
      343      // shouldn't attempt to delete the task if the COMPLETE Task listener failed,
      344      // or managed to cancel the Process or Task Instance
      345      if (shouldDeleteTask)
      346      {
      347        // delete the task
      348        // this method call doesn't invoke additional task listeners
      349        Context
      350        .getCommandContext()
      351        .getTaskManager()
      352        .deleteTask(this, TaskEntity.DELETE_REASON_COMPLETED, false, skipCustomListeners);
      353  
      354        // if the task is associated with a
      355        // execution (and not a case execution)
      356        // and it's still in the same activity
      357        // then call signal an the associated
      358        // execution.
      359        if (executionId !=null) {
      360          ExecutionEntity exec = getExecution();
      361          exec.removeTask(this);
      362          exec.signal(null, null);
      363        }
      364      }
      365    }
