       63  class BaseLatentInfectionProcess(RandomVariable):
       64      """
       65      Base class for latent infection processes with subpopulation structure.
       66  
       67      Provides common functionality for hierarchical and partitioned infection models:
       68      - Population fraction validation and parsing (at sample time)
       69      - Standard output structure via LatentSample
       70  
       71      All subclasses return infections as a ``LatentSample`` named tuple with fields:
       72      (aggregate, all_subpops). Observation processes are responsible for selecting
       73      which subpopulations they observe via indexing.
       74  
       75      The constructor specifies model structure (generation interval, priors,
       76      temporal processes). Population structure (subpop_fractions) is provided at
       77      sample time, allowing a single model to be fit to multiple jurisdictions.
       78  
       79      Parameters
       80      ----------
       81      gen_int_rv
       82          Generation interval PMF
       83      n_initialization_points
       84          Number of initialization days before day 0. Must be at least
       85          ``len(gen_int_rv())`` to provide enough history for the renewal
       86          equation convolution.
       87  
       88      Notes
       89      -----
       90      Population structure (subpop_fractions) is passed to the sample() method, not
       91      the constructor. This allows a single model instance to be fit to multiple
       92      datasets with different jurisdiction structures.
       93  
       94      When using PyrenewBuilder (recommended), n_initialization_points is computed
       95      automatically from all observation processes. When constructing latent processes
       96      directly, you must specify n_initialization_points explicitly.
       97      """

      155              Parsed and validated population structure

      157          Raises
      158          ------
      159          ValueError
      160              If fractions are invalid or don't sum to 1.0
      161          """
      162          if subpop_fractions is None:
      163              raise ValueError("subpop_fractions must be provided")
      164  
      165          fractions = jnp.asarray(subpop_fractions)
      166  
      167          if fractions.ndim != 1:
      168              raise ValueError("subpop_fractions must be a 1D array")
      169  
      170          if len(fractions) == 0:
      171              raise ValueError("Must have at least one subpopulation")
      172  
      173          # Only validate when results are concrete (not during JAX tracing)
      174          # Check the result of the comparison, not just the input, because
      175          # JIT compilation can trace operations on concrete arrays
      176          neg_check = jnp.any(fractions < 0)
      177          if not_jax_tracer(neg_check) and neg_check:
      178              raise ValueError("All population fractions must be non-negative")
      179  
      180          total = jnp.sum(fractions)
      181          sum_check = jnp.isclose(total, 1.0, atol=1e-6)
      182          if not_jax_tracer(sum_check) and not sum_check:
      183              raise ValueError(
      184                  f"Population fractions must sum to 1.0, got {float(total):.6f}"
      185              )
      186  
      187          return PopulationStructure(
