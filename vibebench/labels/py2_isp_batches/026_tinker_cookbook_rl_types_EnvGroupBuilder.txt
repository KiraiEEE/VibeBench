      211  class EnvGroupBuilder(ABC):
      212      """
      213      Builds a group of environments. The group will be used in the following way:
      214  
      215      - Algorithms like GRPO will center rewards across the group.
      216      - The reward function (compute_group_rewards) has access to the trajectories from the
      217        whole group, even though many reward functions will evaluate each one independently.
      218  
      219        - For example, this enables us to use pairwise reward models that look at a pair of
      220          trajectories at a time. With such a reward model, we effectively have a multi-agent
      221          environment, where the agents are playing a zero-sum game.
      222  
      223      Groups can be used in two ways, in practice:
      224  
      225      - To define a multi-agent environment
      226      - As a part of the *algorithm* (e.g. GRPO), when dealing with single-agent tasks.
      227  
      228      **Picklability:** Implementations must be pickleable (via standard ``pickle``) to support
      229      distributed rollout execution where builders are serialized and sent to remote workers.
      230      Avoid storing live network connections, file handles, or other unpickleable objects as
      231      fields. Use ``get_renderer()`` to create Renderers (which are automatically pickle-safe).
      232      Store configuration strings (model names, connection params) and construct heavy objects
      233      in ``make_envs()`` when possible. See ``HarborEnvGroupBuilder`` for a reference
      234      implementation of the lazy-construction pattern.
      235      """
      236  
      237      @abstractmethod
      238      async def make_envs(self) -> Sequence[Env]:
      239          """Create the environments for this group.
      240  
      241          Returns:
      242              Sequence[Env]: The environments to run rollouts in.
      243          """
      244          pass
      245  
      246      async def compute_group_rewards(
      247          self, trajectory_group: list[Trajectory], env_group: Sequence[Env]
      248      ) -> list[tuple[float, Metrics]]:
      249          """Compute a final reward for each trajectory that depends on the whole group.
      250  
      251          This is called after all rollouts in the group complete.  The total
      252          reward for each trajectory is the sum of the per-timestep rewards
      253          (from :meth:`Env.step`) plus the final group reward returned here.
      254  
      255          Override this when the reward depends on comparing trajectories within
      256          the group (e.g., pairwise reward models).  The default implementation
      257          returns ``(0.0, {})`` for every trajectory, so only per-timestep
      258          rewards are used.
      259  
      260          Args:
      261              trajectory_group (list[Trajectory]): The completed trajectories,
      262                  one per environment in the group.
      263              env_group (Sequence[Env]): The corresponding environments (same
      264                  order as ``trajectory_group``).
      265  
      266          Returns:
      267              list[tuple[float, Metrics]]: A list of ``(reward, metrics)`` pairs,
      268                  one per trajectory.  The reward is added to the per-timestep
      269                  total; the metrics dict is merged into training logs.
      270          """
      271          return [(0.0, {}) for _ in trajectory_group]
      272  
      273      async def cleanup(self) -> None:
      274          """Clean up resources created by make_envs().
      275  
      276          Called after rollouts and reward computation complete, regardless
      277          of success or failure. Override this to release expensive resources
      278          like cloud sandboxes, remote browsers, etc.
      279  
      280          Default is a no-op. Implementations should be idempotent (safe to
      281          call multiple times) and handle exceptions internally, as `do_group_rollout`
      282          does not catch exceptions from this method.
      283          """
      284          pass
      285  
      286      def logging_tags(self) -> list[str]:
      287          """Return tags used to aggregate metrics in training logs.
      288  
      289          Tags let the training loop group metrics (rewards, episode lengths,
      290          etc.) by environment type.  Return a short list of names at different
      291          levels of granularity.
      292  
      293          Returns:
      294              list[str]: Tag strings for this environment group.  Default is
      295                  an empty list.
      296  
      297          Example::
      298  
      299              def logging_tags(self) -> list[str]:
      300                  return ["gsm", "math", "rlvr"]
      301          """
      302          return []
