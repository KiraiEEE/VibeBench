       41          value: str,

       44      ) -> str:
       45          """Wrap value in parentheses if inside a nested expression."""
       46          # do not wrap into parentheses if
       47          #   1. already wrapped or

       57                  return value
       58          elif not isinstance(parent, ExpressionRule):
       59              return value
