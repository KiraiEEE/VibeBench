      164  class NoiseGenerator(SignalGenerator):
      165      """
      166      Abstract base class for noise signal generators.
      167  
      168      The :class:`NoiseGenerator` class defines the common interface for all :class:`SignalGenerator`
      169      classes with noise signals. This class may be used as a base for class handling noise signals
      170      that can be characterized by their RMS amplitude.
      171  
      172      It should not be used directly as it contains no real functionality.
      173  
      174      See Also
      175      --------
      176      :class:`~acoular.signals.PNoiseGenerator` : For pink noise generation.
      177      :class:`~acoular.signals.WNoiseGenerator` : For pink white generation.
      178      :class:`~acoular.sources.UncorrelatedNoiseSource` : For per-channel noise generation.
      179      """
      180  
      181      #: Root mean square (RMS) amplitude of the signal. For a point source,
      182      #: this corresponds to the RMS amplitude at a distance of 1 meter. Default is ``1.0``.
      183      rms = Float(1.0)
      184  
      185      #: Seed for random number generator. Default is ``0``.
      186      #: This parameter should be set differently for different instances
      187      #: to guarantee statistically independent (non-correlated) outputs.
      188      seed = Int(0)
      189  
      190      #: Internal identifier based on generator properties. (read-only)
      191      digest = Property(depends_on=['rms', 'seed', 'sample_freq', 'num_samples'])
      192  
      193      @abstractmethod
      194      def _get_digest(self):
      195          """Return the internal identifier."""
      196  
      197      @abstractmethod
      198      def signal(self):
      199          """Generate and deliver the periodic signal."""

      306      def signal(self):
      307          """
      308          Generate and deliver the pink noise signal.
      309  
      310          The signal is computed using the Voss-McCartney algorithm, which generates noise
      311          with a :math:`1/f` power spectral density. The method ensures that the output has the
      312          desired :attr:`RMS<SignalGenerator.rms>` amplitude and spectrum.
      313  
      314          Returns
      315          -------
      316          :class:`numpy.ndarray`
      317              A 1D array of floats containing the generated pink noise signal. The length
      318              of the array is equal to :attr:`~SignalGenerator.num_samples`.
      319  
      320          Notes
      321          -----
      322          - The "depth" parameter controls the number of octaves included in the pink noise
      323            simulation. If the specified depth exceeds the maximum possible value based on
      324            the number of samples, it is automatically adjusted, and a warning is printed.
      325          - The output signal is scaled to have the same overall level as white noise by dividing
      326            the result by ``np.sqrt(depth + 1.5)``.
      327          """
      328          nums = self.num_samples
      329          depth = self.depth
      330          # maximum depth depending on number of samples
      331          max_depth = int(np.log(nums) / np.log(2))
      332  
      333          if depth > max_depth:
      334              depth = max_depth
      335              print(f'Pink noise filter depth set to maximum possible value of {max_depth:d}.')
      336  
      337          rnd_gen = np.random.RandomState(self.seed)
      338          s = rnd_gen.standard_normal(nums)
      339          for _ in range(depth):
      340              ind = 2**_ - 1
      341              lind = nums - ind
      342              dind = 2 ** (_ + 1)
      343              s[ind:] += np.repeat(rnd_gen.standard_normal(nums // dind + 1), dind)[:lind]
      344          # divide by np.sqrt(depth+1.5) to get same overall level as white noise
      345          return self.rms / np.sqrt(depth + 1.5) * s
