      606          """List namespaces from the given namespace. If not given, list top-level namespaces from the catalog.
      607  
      608          Args:

      610  
      611          Returns:

      619  
      620          table_stmt = select(IcebergTables.table_namespace).where(IcebergTables.catalog_name == self.name)
      621          namespace_stmt = select(IcebergNamespaceProperties.namespace).where(IcebergNamespaceProperties.catalog_name == self.name)
      622          if namespace:

      624              table_stmt = table_stmt.where(IcebergTables.table_namespace.like(namespace_like))

      627              table_stmt,
      628              namespace_stmt,
      629          )
