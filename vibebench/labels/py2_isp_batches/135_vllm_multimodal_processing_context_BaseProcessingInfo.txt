      429          is invalid.
      430          """

      432          allowed_limit = self.allowed_mm_limits.get(modality, 0)
      433  
      434          if supported_limit is None:

      440              msg = f"At most {limit} {modality}(s) may be provided in one prompt."
      441

      443                  msg += " Set `--limit-mm-per-prompt` to increase this limit."
      444
