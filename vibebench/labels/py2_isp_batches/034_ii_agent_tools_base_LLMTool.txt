       33      """A tool that fits into the standard LLM tool-calling paradigm.
       34  
       35      An LLM tool can be called by supplying the parameters specified in its
       36      input_schema, and returns a string that is then shown to the model.
       37      """
       38  
       39      name: str
       40      description: str
       41      input_schema: ToolInputSchema
       42  
       43      @property
       44      def should_stop(self) -> bool:
       45          """Whether the tool wants to stop the current agentic run."""
       46          return False
       47  
       48      # Final is here to indicate that subclasses should override run_impl(), not
       49      # run(). There may be a reason in the future to override run() itself, and
       50      # if such a reason comes up, this @final decorator can be removed.
       51      @final
       52      def run(
       53          self,
       54          tool_input: dict[str, Any],
       55          message_history: Optional[MessageHistory] = None,
       56      ) -> str | list[dict[str, Any]]:
       57          """Run the tool.
       58  
       59          Args:
       60              tool_input: The input to the tool.
       61              message_history: The dialog messages so far, if available. The tool
       62                  is allowed to modify this object, so the caller should make a copy
       63                  if that's not desired. The dialog messages should not contain
       64                  pending tool calls. They should end where it's the user's turn.
       65          """
       66          if message_history:
       67              assert message_history.is_next_turn_user()
       68  
       69          try:
       70              self._validate_tool_input(tool_input)
       71              result = self.run_impl(tool_input, message_history)
       72              tool_output = result.tool_output
       73          except jsonschema.ValidationError as exc:
       74              tool_output = "Invalid tool input: " + exc.message
       75          except BadRequestError as exc:
       76              raise RuntimeError("Bad request: " + exc.message)
       77  
       78          return tool_output
       79  
       80      def get_tool_start_message(self, tool_input: ToolInputSchema) -> str:
       81          """Return a user-friendly message to be shown to the model when the tool is called."""
       82          return f"Calling tool '{self.name}'"
