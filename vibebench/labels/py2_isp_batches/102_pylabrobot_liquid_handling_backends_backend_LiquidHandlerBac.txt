     2974      # new child absolute rotation. The moved resource's rotation will be the original child
     2975      # rotation plus the rotation applied by the movement.

     2977      # The new resource absolute location is
     2978      # drop.resource.get_absolute_rotation().z + drop.rotation
     2979      center_in_absolute_space = drop.resource.center().rotated(
     2980        Rotation(z=drop.resource.get_absolute_rotation().z + drop.rotation)
     2981      )
     2982      x, y, z = drop.destination + center_in_absolute_space + drop.offset
     2983

     2985        traversal_height_start = (
     2986          minimum_traverse_height_at_beginning_of_a_command or self._iswap_traversal_height
     2987        )
     2988        z_position_at_the_command_end = z_position_at_the_command_end or self._iswap_traversal_height
     2989        assert (
     2990          drop.resource.get_absolute_rotation().x == 0
     2991          and drop.resource.get_absolute_rotation().y == 0
     2992        )
     2993        assert drop.resource.get_absolute_rotation().z % 90 == 0
     2994  
     2995        # Use the pickup direction to determine how wide the plate is gripped.
     2996        # Note that the plate is still in the original orientation at this point,
     2997        # so get_absolute_size_{x,y}() will return the size of the plate in the original orientation.
     2998        if (
     2999          drop.pickup_direction == GripDirection.FRONT or drop.pickup_direction == GripDirection.BACK
     3000        ):
     3001          plate_width = drop.resource.get_absolute_size_x()
     3002        elif (
     3003          drop.pickup_direction == GripDirection.RIGHT or drop.pickup_direction == GripDirection.LEFT
     3004        ):
