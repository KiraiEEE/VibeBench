       90          self, from_stage: str, to_stage: str, get_key: str, metadata: dict[str, Any] | None = None
       91      ) -> tuple[Any, int] | None:
       92          if not self.client:
       93              logger.error("Datasystem client not initialized")
       94              return None
       95  
       96          key = self._make_key(get_key, from_stage, to_stage)
       97          try:
       98              raw_list = self.client.get([key], False, self.get_sub_timeout_ms)
       99              raw_data = raw_list[0] if raw_list else None
      100              if raw_data is not None:
      101                  data = self.deserialize_obj(raw_data)
      102                  self._metrics["gets"] += 1
      103                  payload_size = len(raw_data)
      104                  logger.debug(
      105                      "YuanrongConnector: retrieved %s (%s -> %s) %d bytes",
      106                      key,
      107                      from_stage,
      108                      to_stage,
      109                      payload_size,
      110                  )
      111                  return data, payload_size
      112  
      113          except Exception as exc:
      114              self._metrics["timeouts"] += 1
      115              logger.error("YuanrongConnector get failed: %s", exc)
      116              return None
      118      def cleanup(self, request_id: str) -> None:
      119          if not self.client:
      120              return
      121  
      122          # Note: Datasystem doesn't have explicit delete, data will be garbage collected
      123          logger.debug("YuanrongConnector: cleanup requested for %s (no-op)", request_id)
      125      def health(self) -> dict[str, Any]:
      126          if not self.client:
      127              return {"status": "unhealthy", "error": "Datasystem client not initialized"}
      128  
      129          return {"status": "healthy", "host": self.host, "port": self.port, **self._metrics}
      131      def close(self) -> None:
      132          if not self.client:
      133              return
      134  
      135          self.client = None
      136          logger.info("YuanrongConnector closed")
