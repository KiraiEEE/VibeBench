      761      def moe_problem_size(
      762          self,
      763          a1: torch.Tensor,
      764          w1: torch.Tensor,
      765          w2: torch.Tensor,
      766          topk_ids: torch.Tensor,
      767      ) -> tuple[int, int, int, int, int]:
      768          """
      769          Extract the MoE problem size from the given tensor arguments:
      770          - a: The hidden states, input to the MoE layer.
      771          - w1: The first set of expert weights.
      772          - w2: The second set of expert weights.
      773          - topk_ids: The topk ids.
      774  
      775          Note: extracting the problem shape from the weight and activation
      776          tensors is not obvious.  It needs to be done this way specifically
      777          due to subtle issues with particular kernels, e.g. the int4 kernels
      778          divide the trailing dimension by two, so it's not "correct" to
      779          extract N or K from the trailing dimension of w1 or w2.  Similarly,
      780          some kernels transpose the weights, so this needs to be kept in mind.
      781  
      782          Note: This implementation covers most cases. However, if experts
      783          require a specialized implementation, like MarlinExperts, they are free
      784          to override this function.
      785          """
      786          assert w1.dim() == 3 and w2.dim() == 3
      787          E, N, _ = w1.size()
      788          K = a1.size(-1)
      789  
      790          if a1.dim() == 2:
      791              # Make sure we are using the correct a1 (pre-permute).
      792              assert topk_ids.size(0) == a1.size(0), f"{topk_ids.size(0)} != {a1.size(0)}"
      793              M = a1.size(0)
      794          else:
      795              assert a1.dim() == 3
      796              assert a1.size(0) == E, f"{a1.size(0)} == {E}"
      797              M = a1.size(1)  # This is max_num_tokens
      798  
      799          assert topk_ids.dim() == 2
      800          topk = topk_ids.size(1)
      801  
      802          return E, M, N, K, topk

      851      @staticmethod
      852      def adjust_N_for_activation(N: int, activation: MoEActivation) -> int:
      853          """
      854          Calculate the output dimension for the activation function.
      855  
      856          For *_no_mul activations (e.g. relu2_no_mul),
      857          there's no gate/up split, so output size equals input size (N).
      858  
      859          For regular gated activations (e.g., silu, gelu, swigluoai),
      860          output size is N // 2 due to gate × activation(up) multiplication.
      861  
      862          Args:
      863              N: The intermediate size (width of w1/w3 weights).
      864              activation: The activation function enum.
      865  
      866          Returns:
      867              The output dimension after activation.
      868          """
      869          return N if not activation.is_gated else N // 2
