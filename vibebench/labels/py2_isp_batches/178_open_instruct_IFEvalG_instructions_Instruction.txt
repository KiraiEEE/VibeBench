      200              if 'less than', the actual number of sentences < the threshold;
      201              if 'at least', the actual number of sentences >= the threshold.
      202  
      203          Returns:
      204            A string representing the instruction description.

      206          # The number of sentences as a threshold for comparison.

      213          elif relation not in _COMPARISON_RELATION:
      214              raise ValueError(
      215                  f"The supported relation for comparison must be in {_COMPARISON_RELATION}, but {relation} is given."
      216              )
      217          else:
      218              self._comparison_relation = relation
      219  
      220          self._description_pattern = "Your response should contain {relation} {num_sentences} sentences."
      221          return self._description_pattern.format(

      223          )

      235

      237            value: A string representing the response.

      241
