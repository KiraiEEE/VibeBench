  1831    private void setConsumeEndTime(SegmentZKMetadata segmentZKMetadata, long now) {
  1832      long maxConsumeTimeMillis = _streamConfig.getFlushThresholdTimeMillis();
  1833      _consumeEndTime = segmentZKMetadata.getCreationTime() + maxConsumeTimeMillis;
  1834  
  1835      // When we restart a server, the consuming segments retain their creationTime (derived from segment
  1836      // metadata), but a couple of corner cases can happen:
  1837      // (1) The server was down for a very long time, and the consuming segment is not yet completed.
  1838      // (2) The consuming segment was just about to be completed, but the server went down.
  1839      // In either of these two cases, if a different replica could not complete the segment, it is possible
  1840      // that we get a value for _consumeEndTime that is in the very near future, or even in the past. In such
  1841      // cases, we let some minimum consumption happen before we attempt to complete the segment (unless, of course
  1842      // the max consumption time has been configured to be less than the minimum time we use in this class).
  1843      long minConsumeTimeMillis =
  1844          Math.min(maxConsumeTimeMillis, TimeUnit.MILLISECONDS.convert(MINIMUM_CONSUME_TIME_MINUTES, TimeUnit.MINUTES));
  1845      if (_consumeEndTime - now < minConsumeTimeMillis) {
  1846        _consumeEndTime = now + minConsumeTimeMillis;
  1847      }
