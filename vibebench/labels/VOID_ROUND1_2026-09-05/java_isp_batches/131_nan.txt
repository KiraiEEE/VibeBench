      101                  .orElseThrow(() -> new StarRocksConnectorException("invalid argument type for %s, expected positive BIGINT",
      102                          BATCH_SIZE))
      103                  : DEFAULT_BATCH_SIZE;
      104          long batchParallelism = args.get(BATCH_PARALLELISM) != null ? args.get(BATCH_PARALLELISM)
      105                  .castTo(IntegerType.BIGINT)
      106                  .map(ConstantOperator::getBigint)
      107                  .filter(v -> v > 0)
      108                  .orElseThrow(() -> new StarRocksConnectorException("invalid argument type for %s, expected positive BIGINT",
      109                          BATCH_PARALLELISM))
      110                  : DEFAULT_BATCH_PARALLELISM;
      111  
      112          AlterTableStmt stmt = context.stmt();
      113          Expr partitionFilter = context.clause().getWhere();
      114          String catalogName = context.stmt().getCatalogName();
      115          String dbName = context.stmt().getDbName();
      116          String tableName = context.stmt().getTableName();
      117          long startMs = System.currentTimeMillis();
      118          boolean success = false;
      119          String failureReason = IcebergMetricsMgr.classifyFailReason((String) null);
      120          IcebergRewriteDataJob.RewriteMetrics metrics = IcebergRewriteDataJob.RewriteMetrics.EMPTY;
      121  
      122          VelocityContext velCtx = new VelocityContext();
      123          velCtx.put("catalogName", catalogName);
      124          velCtx.put("dbName", dbName);
      125          velCtx.put("tableName", tableName);
      126  
      127          String partitionFilterSql = null;
      128          if (partitionFilter != null) {
      129              List<SlotRef> slots = new ArrayList<>();
      130              partitionFilter.collect(SlotRef.class, slots);
      131              for (SlotRef slot : slots) {
      132                  slot.setTblName(new TableName(dbName, tableName));
      133              }
      134              partitionFilterSql = ExprToSql.toSql(partitionFilter);
      135          }
      136          velCtx.put("partitionFilterSql", partitionFilterSql);
      137          VelocityEngine defaultVelocityEngine = new VelocityEngine();
      138          defaultVelocityEngine.setProperty(VelocityEngine.RUNTIME_LOG_REFERENCE_LOG_INVALID, false);
      139          StringWriter writer = new StringWriter();
      140          defaultVelocityEngine.evaluate(velCtx, writer, "InsertSelectTemplate",
      141                  "INSERT INTO $catalogName.$dbName.$tableName" +
      142                          " SELECT * FROM $catalogName.$dbName.$tableName" +
      143                          " #if ($partitionFilterSql)" +
      144                          " WHERE $partitionFilterSql" +
      145                          " #end"
      146          );
      147          String executeStmt = writer.toString();
      148          IcebergRewriteDataJob job = new IcebergRewriteDataJob(executeStmt, rewriteAll,
      149                  minFileSizeBytes, batchSize, batchParallelism, context.context(), stmt);
      150          long durationMs = 0L;
      151          try {
      152              job.prepare();
      153              metrics = job.execute();
      154              success = true;
      155          } catch (Exception e) {
      156              failureReason = IcebergMetricsMgr.classifyFailReason(e);
      157              LOGGER.error("failed to rewrite data files for iceberg table {}.{}",
      158                      stmt.getDbName(), stmt.getTableName(), e);
      159              throw new StarRocksConnectorException("execute rewrite data files for iceberg table %s.%s failed: %s",
      160                      stmt.getDbName(), stmt.getTableName(), e.getMessage(), e);
      161          } finally {
      162              durationMs = Math.max(0, System.currentTimeMillis() - startMs);
      163              recordManualCompactionMetrics(success, durationMs, metrics, failureReason);
      164          }
      165          return getRewriteResult(context, metrics, durationMs);
      166      }

      168      private ShowResultSet getRewriteResult(IcebergTableProcedureContext context, 
      169                                                              IcebergRewriteDataJob.RewriteMetrics metrics,
      170                                                              long durationMs) {
      171          ShowResultSetMetaData metaData = ShowResultSetMetaData.builder()
      172                  .column("db_name", TypeFactory.createVarcharType(128))
      173                  .column("table_name", TypeFactory.createVarcharType(128))
      174                  .column("input_data_files", TypeFactory.createType(PrimitiveType.BIGINT))
      175                  .column("output_data_files", TypeFactory.createType(PrimitiveType.BIGINT))
      176                  .column("removed_delete_files", TypeFactory.createType(PrimitiveType.BIGINT))
      177                  .column("duration_ms", TypeFactory.createType(PrimitiveType.BIGINT))
      178                  .column("status", TypeFactory.createVarcharType(128))
      179                  .build();
      180          List<List<String>> rows = new ArrayList<>();
      181          rows.add(Arrays.asList(
      182                  context.stmt().getDbName(),
      183                  context.stmt().getTableName(),
      184                  String.valueOf(metrics.getRewrittenDataFilesCount()),
      185                  String.valueOf(metrics.getAddedDataFilesCount()),
      186                  String.valueOf(metrics.getRewrittenDeleteFilesCount()),
      187                  String.valueOf(durationMs),
      188                  "success"
      189          ));
      190          String message = String.format(
      191                  "Iceberg compaction finished [%s.%s]: input_data_files=%d, output_data_files=%d, " +
      192                          "removed_delete_files=%d, duration_ms=%d",
      193                  context.stmt().getDbName(),
      194                  context.stmt().getTableName(),
      195                  metrics.getRewrittenDataFilesCount(),
      196                  metrics.getAddedDataFilesCount(),
      197                  metrics.getRewrittenDeleteFilesCount(),
      198                  durationMs);
      199          Tracers.record(Tracers.Module.EXTERNAL,
      200                  String.format("ICEBERG.RewriteDataFiles.%s.%s", context.stmt().getDbName(), context.stmt().getTableName()),
      201                  message);
      202          context.context().getState().setOk(0, 0, message);
      203          return new ShowResultSet(metaData, rows);
