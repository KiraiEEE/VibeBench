       94      public long selectBackupWorker(long workerId) {
       95          if (availableID2ComputeNode.isEmpty() || !id2ComputeNode.containsKey(workerId)) {
       96              return -1;
       97          }
       98          if (allComputeNodeIds == null) {
       99              createAvailableIdList();

      102          Preconditions.checkState(allComputeNodeIds.contains(workerId));
      103  
      104          int startPos = allComputeNodeIds.indexOf(workerId);
      105          int attempts = allComputeNodeIds.size();
      106          while (attempts-- > 0) {
      107              startPos = (startPos + 1) % allComputeNodeIds.size();
      108              long buddyId = allComputeNodeIds.get(startPos);
      109              // Skip SimpleScheduler.isInBlocklist(buddyId) check - only verify buddyId != workerId and is in availableID2ComputeNode
      110              if (buddyId != workerId && availableID2ComputeNode.containsKey(buddyId)) {
      111                  return buddyId;
      112              }
      113          }
