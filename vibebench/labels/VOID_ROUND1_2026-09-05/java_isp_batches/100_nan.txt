      141    @Override
      142    public boolean isAllowed(
      143        final DatabaseSessionEmbedded session,
      144        final Set<Identifiable> iAllowAll,
      145        final Set<Identifiable> iAllowOperation) {
      146      if ((iAllowAll == null || iAllowAll.isEmpty())
      147          && (iAllowOperation == null || iAllowOperation.isEmpty()))
      148      // NO AUTHORIZATION: CAN'T ACCESS
      149      {
      150        return false;
      151      }
      152  
      153      final var currentUser = session.getCurrentUser();
      154      if (currentUser != null) {
      155        // CHECK IF CURRENT USER IS ENLISTED
      156        if (iAllowAll == null || !iAllowAll.contains(currentUser.getIdentity())) {
      157          // CHECK AGAINST SPECIFIC _ALLOW OPERATION
      158          if (iAllowOperation != null && iAllowOperation.contains(currentUser.getIdentity())) {
      159            return true;
      160          }
      161  
      162          // CHECK IF AT LEAST ONE OF THE USER'S ROLES IS ENLISTED
      163          for (var r : currentUser.getRoles()) {
      164            // CHECK AGAINST GENERIC _ALLOW
      165            if (iAllowAll != null && iAllowAll.contains(r.getIdentity())) {
      166              return true;
      167            }
      168            // CHECK AGAINST SPECIFIC _ALLOW OPERATION
      169            if (iAllowOperation != null && iAllowOperation.contains(r.getIdentity())) {
      170              return true;
      171            }
      172            // CHECK inherited permissions from parent roles, fixes #1980: Record Level Security:
      173            // permissions don't follow role's
      174            // inheritance
      175            var parentRole = r.getParentRole();
      176            while (parentRole != null) {
      177              if (iAllowAll != null && iAllowAll.contains(parentRole.getIdentity())) {
      178                return true;
      179              }
      180              if (iAllowOperation != null && iAllowOperation.contains(
      181                  parentRole.getIdentity())) {
      182                return true;
      183              }
      184              parentRole = parentRole.getParentRole();
      185            }
      186          }
      187          return false;
      188        }
      189      }
      190      return true;
      191    }
