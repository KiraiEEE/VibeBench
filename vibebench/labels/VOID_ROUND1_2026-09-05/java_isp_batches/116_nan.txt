      268          Matcher match = pat.matcher(text);

      271              if (data.contains(key)) text = text.replace(match.group(), data.get(key));
      272              else if (key.equals("player")) text = text.replace(match.group(), caster.getName());

      278                  snipped.put(uuid.toString(), key);
