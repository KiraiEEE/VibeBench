       96              protected void onFirstWrite() throws IOException {
       97                  dispatchToClient(false);
       98              }

      100              protected void dispatchToClient(boolean empty) throws IOException {
      101                  final MessageImpl m = new MessageImpl();
      102                  localDestinationFactory.copy(message, m);
      103                  if (!empty) {
      104                      final PipedInputStream stream = new PipedInputStream();
      105                      wrappedStream = new PipedOutputStream(stream);
      106                      m.setContent(InputStream.class, stream);
      107                  }
      108  
      109                  final Runnable receiver = new Runnable() {
      110                      public void run() {
      111                          if (exchange != null) {
      112                              exchange.setInMessage(m);
      113                          }
      114                          conduit.getMessageObserver().onMessage(m);
      115                      }
      116                  };
      117                  Executor ex = message.getExchange() != null
      118                      ? message.getExchange().get(Executor.class) : null;
      119                  // Need to avoid to get the SynchronousExecutor
      120                  if (ex == null || SynchronousExecutor.isA(ex)) {
      121                      if (exchange == null) {
      122                          ex = localDestinationFactory.getExecutor(bus);
      123                      } else {
      124                          ex = localDestinationFactory.getExecutor(exchange.getBus());
      125                      }
      126                      if (ex != null) {
      127                          ex.execute(receiver);
      128                      } else {
      129                          new Thread(receiver).start();
      130                      }
      131                  } else {
      132                      ex.execute(receiver);
      133                  }
      134              }
      135          }

      144          public void prepare(final Message message) throws IOException {
      145              if (!Boolean.TRUE.equals(message.getExchange().get(LocalConduit.DIRECT_DISPATCH))) {
      146                  final Exchange exchange = (Exchange)message.getExchange().get(LocalConduit.IN_EXCHANGE);
      147  
      148                  AbstractWrappedOutputStream cout
      149                      = new LocalDestinationOutputStream(exchange, message);
      150  
      151                  message.setContent(OutputStream.class, cout);
      152  
      153              } else {
      154                  CachedOutputStream stream = new CachedOutputStream();
      155                  message.setContent(OutputStream.class, stream);
      156                  message.setContent(CachedOutputStream.class, stream);
      157                  stream.holdTempFile();
      158              }
      159          }

      161          @Override
      162          public void close(Message message) throws IOException {
      163              // set the pseudo status code if not set (REVISIT add this method in MessageUtils to be reused elsewhere?)
      164              Integer i = (Integer)message.get(Message.RESPONSE_CODE);
      165              if (i == null) {
      166                  int code = ((message.getExchange().isOneWay() && !MessageUtils.isPartialResponse(message))
      167                      || MessageUtils.isEmptyPartialResponse(message)) ? 202 : 200;
      168                  message.put(Message.RESPONSE_CODE, code);
      169              }
      170              if (Boolean.TRUE.equals(message.getExchange().get(LocalConduit.DIRECT_DISPATCH))) {
      171                  final Exchange exchange = (Exchange)message.getExchange().get(LocalConduit.IN_EXCHANGE);
      172  
      173                  MessageImpl copy = new MessageImpl();
      174                  copy.putAll(message);
      175                  message.getContent(OutputStream.class).close();
      176                  CachedOutputStream stream = message.getContent(CachedOutputStream.class);
      177                  message.setContent(OutputStream.class, stream);
      178                  MessageImpl.copyContent(message, copy);
      179                  copy.setContent(InputStream.class, stream.getInputStream());
      180                  stream.releaseTempFileHold();
      181                  if (exchange != null && exchange.getInMessage() == null) {
      182                      exchange.setInMessage(copy);
      183                  }
      184                  conduit.getMessageObserver().onMessage(copy);
      185                  return;
      186              }
      187  
      188              super.close(message);
      189          }
