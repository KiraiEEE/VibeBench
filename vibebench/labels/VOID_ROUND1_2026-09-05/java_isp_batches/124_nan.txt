       92      public long getNowTruncSubSec(MTMVPartitionSyncTimeUnit timeUnit, int syncLimit)
       93              throws AnalysisException {
       94          if (syncLimit < 1) {
       95              throw new AnalysisException("Unexpected syncLimit, syncLimit: " + syncLimit);
       96          }
       97          // get current time
       98          Expression now = DateTimeAcquire.now();
       99          if (!(now instanceof DateTimeLiteral)) {
      100              throw new AnalysisException("now() should return DateTimeLiteral, now: " + now);
      101          }
      102          DateTimeLiteral nowLiteral = (DateTimeLiteral) now;
      103          // date trunc
      104          now = DateTimeExtractAndTransform
      105                  .dateTrunc(nowLiteral, new VarcharLiteral(timeUnit.name()));
      106          if (!(now instanceof DateTimeLiteral)) {
      107              throw new AnalysisException("dateTrunc() should return DateTimeLiteral, now: " + now);
      108          }
      109          nowLiteral = (DateTimeLiteral) now;
      110          // date sub
      111          if (syncLimit > 1) {
      112              nowLiteral = dateSub(nowLiteral, timeUnit, syncLimit - 1);
      113          }
      114          return ((IntegerLiteral) DateTimeExtractAndTransform.unixTimestamp(nowLiteral)).getValue();
      115      }

      120              int num)
