      122      }
      123  
      124      // Get the top X valid platforms (ones with logo) with the most number of documents
      125      return getTopKValues(opContext, urnCounts).stream()
      126          .map(entry -> buildRecommendationContent(entry.getKey(), entry.getValue()))
      127          .collect(Collectors.toList());
      128    }

      130    protected List<String> getEntityNames(EntityRegistry entityRegistry) {
      131      // By default, no list is applied which means searching across entities.
      132      return QueryUtils.getQueryByDefaultEntitySpecs(entityRegistry).stream()
      133          .map(EntitySpec::getName)
      134          .collect(Collectors.toList());
      135    }

      137    // Get top K entries with the most count
      138    private <T> List<Map.Entry<T, Long>> getTopKValues(
      139        @Nonnull OperationContext opContext, Map<T, Long> countMap) {
      140      final PriorityQueue<Map.Entry<T, Long>> queue =
      141          new PriorityQueue<>(getMaxContent(), Map.Entry.comparingByValue(Comparator.naturalOrder()));
      142      for (Map.Entry<T, Long> entry : countMap.entrySet()) {
      143        if (queue.size() < getMaxContent() && isValidCandidate(opContext, entry.getKey())) {
      144          queue.add(entry);
      145        } else if (queue.size() > 0
      146            && queue.peek().getValue() < entry.getValue()
      147            && isValidCandidate(opContext, entry.getKey())) {
      148          queue.poll();
      149          queue.add(entry);
      150        }
      151      }
      152  
      153      // Since priority queue polls in reverse order (nature of heaps), need to reverse order before
      154      // returning
      155      final LinkedList<Map.Entry<T, Long>> topK = new LinkedList<>();
      156      while (!queue.isEmpty()) {
      157        topK.addFirst(queue.poll());
      158      }
      159      return topK;
      160    }

      162    private Map<String, Long> mergeAggregation(Map<String, Long> first, Map<String, Long> second) {
      163      return Stream.concat(first.entrySet().stream(), second.entrySet().stream())
      164          .collect(Collectors.toMap(Map.Entry::getKey, Map.Entry::getValue, Long::sum));
      165    }

      167    private <T> RecommendationContent buildRecommendationContent(T candidate, long count) {
      168      // Set filters for platform
      169      SearchParams searchParams =
      170          new SearchParams()
      171              .setQuery("")
      172              .setFilters(
      173                  new CriterionArray(
      174                      ImmutableList.of(
      175                          buildCriterion(
      176                              getSearchFieldName(), Condition.EQUAL, candidate.toString()))));
      177      ContentParams contentParams = new ContentParams().setCount(count);
      178      RecommendationContent content = new RecommendationContent();
      179      if (candidate instanceof Urn) {
      180        content.setEntity((Urn) candidate);
      181      }
      182      return content
      183          .setValue(candidate.toString())
      184          .setParams(
      185              new RecommendationParams()
      186                  .setSearchParams(searchParams)
      187                  .setContentParams(contentParams));
      188    }
