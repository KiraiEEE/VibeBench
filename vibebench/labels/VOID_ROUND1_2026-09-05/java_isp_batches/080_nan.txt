       45    /* All-or-nothing semantics:
       46     * If ByteBuffer allocation fails, cleanupByteBuffers(bbs) cleans up and dest[] remains untouched.
       47     * Only after all allocations succeed does it update dest[].
       48     */
       49    // MemoryBuffer in dest[] is uninitialized, i.e., with no backing ByteBuffer.
       50    @Override
       51    public void allocateMultiple(MemoryBuffer[] dest, int size, BufferObjectFactory factory) {
       52      int len = dest.length;
       53  
       54      // Pre-allocate the ByteBuffers for those slots
       55      ByteBuffer[] bbs = new ByteBuffer[len];
       56      try {
       57        for (int k = 0; k < len; k++) {
       58          bbs[k] = isDirect ? ByteBuffer.allocateDirect(size) : ByteBuffer.allocate(size);
       59        }
       60      } catch (OutOfMemoryError | RuntimeException e) {
       61        cleanupByteBuffers(bbs); // frees only the ones actually created
       62        throw e;
       63      }
       64  
       65      // Success → initialize and publish
       66      for (int k = 0; k < len; k++) {
       67        LlapAllocatorBuffer buf = (LlapAllocatorBuffer)dest[k];
       68        buf.initialize(bbs[k], 0, size);
       69      }
       70    }

       72    /**
       73     * Helper method to cleanup partially allocated ByteBuffers
       74     */
       75    private void cleanupByteBuffers(ByteBuffer[] byteBuffers) {
       76      for (int i = 0; i < byteBuffers.length; ++i) {
       77        ByteBuffer bb = byteBuffers[i];
       78        if (bb == null) continue; // This and subsequent buffers were not allocated
       79  
       80        // Only cleanup direct buffers as heap buffers will be GC'd
       81        if (bb.isDirect() && CleanerUtil.UNMAP_SUPPORTED) {
       82          try {
       83            CleanerUtil.getCleaner().freeBuffer(bb);
       84          } catch (Throwable t) {
       85            LlapIoImpl.LOG.warn("Error cleaning up DirectByteBuffer during allocation failure", t);
       86          }
       87        }
       88        byteBuffers[i] = null; // Help GC
       89      }
       90    }
