      141    /**
      142     * Spawn new entities, depending on the {@code SpawnMode}, spawnAmount, spawnDelay, and spawnInterval of an {@code EntitySpawner}.
      143     *
      144     * @see SpawnMode
      145     */
      146    protected void spawnNewEntities() {
      147      if (this.getSpawnMode() != SpawnMode.CUSTOMSPAWNPOINTS && this.getSpawnPoints().isEmpty()) {
      148        return;
      149      }
      150  
      151      switch (this.getSpawnMode()) {
      152        case ALLSPAWNPOINTS:
      153          for (int i = 0; i < this.getSpawnPoints().size(); i++) {
      154            final int index = i;
      155            Game.loop().perform(this.getSpawnDelay() + this.getSpawnDelay() * i,
      156              () -> this.spawn(this.getSpawnPoints().get(index), this.getSpawnAmount()));
      157          }
      158          break;
      159        case ONERANDOMSPAWNPOINT:
      160          this.spawn(Game.random().choose(this.getSpawnPoints()), this.getSpawnAmount());
      161          break;
      162        case RANDOMSPAWNPOINTS:
      163          for (int i = 0; i < this.getSpawnAmount(); i++) {
      164            Game.loop().perform(this.getSpawnDelay() + this.getSpawnDelay() * i, () -> this.spawn(Game.random().choose(this.getSpawnPoints()), 1));
      165          }
      166          break;
      167        case CUSTOMSPAWNPOINTS:
      168          List<Spawnpoint> spawnPoints = this.customSpawnpoints != null ? this.customSpawnpoints.apply(this) : this.getCustomSpawnpoints();
      169  
      170          int index = 0;
      171          for (Spawnpoint spawn : spawnPoints) {
      172            Game.loop().perform(this.getSpawnDelay() + this.getSpawnDelay() * index, () -> this.spawn(spawn, 1));
      173            index++;
      174          }
      175          break;
      176        default:
      177          break;
      178      }
