      267      private Pattern compilePattern(String authorityName) {
      268          // default authority two character name
      269          String er = "((\\s|,|\\.|^|\\-)";
      270          er += authorityName + ")";
      271  
      272          if (authorityName.length() == 2) {
      273              // authority name with dots
      274              er += "|(" + authorityName.charAt(0) + "\\.(\\s)?" + authorityName.charAt(1) + ")";
      275          }
      276  
      277          // using language resources for authority patterns
      278          for (String language : languages) {
      279              List<String> expressions = languageResources.get(language + authorityName);
      280              if (expressions != null) {
      281                  for (String expression : expressions) {
      282                      if ((expression != null) && (expression.trim().length() > 1)) {
      283                          expression = expression.trim();
      284  
      285                          if (!expression.contains("-") && !expression.contains(".")) {
      286                              if (TextUtilities.isAllLowerCase(expression)) {
      287                                  expression = "("
      288                                          + expression.charAt(0)
      289                                          + "|"
      290                                          + Character.toUpperCase(expression.charAt(0))
      291                                          + ")"
      292                                          + expression.substring(1, expression.length());
      293                              }
      294                          } else {
      295                              if (expression.contains("-")) {
      296                                  String[] parts = expression.split("-");
      297                                  expression = "";
      298                                  for (int j = 0; j < parts.length; j++) {
      299                                      String part = parts[j];
      300                                      if (j > 0) {
      301                                          expression += "(\\s|-)*";
      302                                      }
      303  
      304                                      if (TextUtilities.isAllLowerCase(part)) {
      305                                          expression += "("
      306                                                  + part.charAt(0)
      307                                                  + "|"
      308                                                  + Character.toUpperCase(part.charAt(0))
      309                                                  + ")"
      310                                                  + part.substring(1, part.length());
      311                                      }
      312                                  }
      313                              }
      314  
      315                              if (expression.contains(".")) {
      316                                  String[] parts = expression.split(".");
      317                                  expression = "";
      318                                  for (int j = 0; j < parts.length; j++) {
      319                                      String part = parts[j];
      320                                      if (j > 0) {
      321                                          expression += "(\\s)?\\.(\\s)?";
      322                                      }
      323  
      324                                      if (TextUtilities.isAllLowerCase(part)) {
      325                                          expression += "("
      326                                                  + part.charAt(0)
      327                                                  + "|"
      328                                                  + Character.toUpperCase(part.charAt(0))
      329                                                  + ")"
      330                                                  + part.substring(1, part.length());
      331                                      }
      332                                  }
      333                              }
      334                          }
      335                          er += "|(" + expression + ")";
      336                      }
      337                  }
      338              }
      339          }
      340  
      341          return Pattern.compile(er);
