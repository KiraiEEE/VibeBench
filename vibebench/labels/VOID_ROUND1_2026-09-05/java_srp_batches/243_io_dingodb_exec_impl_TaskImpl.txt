      230      @Override
      231      public void run(Object @Nullable [] paras) {
      232          if (status.get() == Status.BORN) {
      233              LogUtils.error(log, "jobId:{}, taskId:{}, Run task but check task has init failed: {}",
      234                  jobId.toString(), id.toString(), taskInitStatus);
      235              Vertex vertex = vertexes.get(runList.get(0));
      236              // Try to propagate error by any operator, may not succeed because task init failed.
      237              // operator.fin(0, FinWithException.of(taskInitStatus));
      238              Operator operator = OperatorFactory.getInstance(vertex.getOp());
      239              operator.fin(0, FinWithException.of(taskInitStatus), vertex);
      240              return;
      241          }
      242          DingoTimeZoneProcessor processor = DingoTimeZoneContext.getProcessor();
      243          // This method should not be blocked, so schedule a running thread.
      244          Executors.execute("task-" + jobId + "-" + id, () -> internalRun(paras, processor));
      245      }

      247      // Synchronize to make sure there are only one thread run this.
      248      private synchronized void internalRun(Object @Nullable [] paras, DingoTimeZoneProcessor processor) {
      249          MdcUtils.setTxnId(txnId.toString());
      250          if (!status.compareAndSet(Status.READY, Status.RUNNING)) {
      251              throw new RuntimeException("Status should be READY.");
      252          }
      253          LogUtils.debug(log, "Task is starting at {}...", location);
      254          activeThreads = new CountDownLatch(runList.size());
      255          setParas(paras);
      256          if (bachTask) {
      257              setStartTs(txnId.seq);
      258          }
      259          for (CommonId operatorId : runList) {
      260              Vertex vertex = vertexes.get(operatorId);
      261              Operator operator = OperatorFactory.getInstance(vertex.getOp());
      262              assert operator instanceof SourceOperator
      263                  : "Operators in run list must be source operator.";
      264              Executors.execute("operator-" + jobId + "-" + id + "-" + operatorId, () -> {
      265                  DingoTimeZoneContext.setProcessor(processor);
      266                  MdcUtils.setTxnId(txnId.toString());
      267                  final long startTime = System.currentTimeMillis();
      268                  activeTaskCount.incrementAndGet();
      269                  try {
      270                      while (operator.push(getContext().copy(), null, vertex)) {
      271                          LogUtils.info(log, "Operator {} need another pushing.", vertex.getId());
      272                      }
      273                      operator.fin(0, null, vertex);
      274                  } catch (DingoTypeRangeException e) {
      275                      LogUtils.error(log, "Run Task:" + getId().toString()
      276                          + ",catch operator:" + vertex.getId() + " run Exception: ", e);
      277                      status.compareAndSet(Status.RUNNING, Status.STOPPED);
      278                      TaskStatus taskStatus = new TaskStatus();
      279                      taskStatus.setStatus(false);
      280                      taskStatus.setTaskId(vertex.getTask().getId().toString());
      281                      taskStatus.setErrorMsg(e.getMessage());
      282                      taskStatus.setErrorType(ErrorType.OutOfValueRange);
      283  
      284                      try {
      285                          operator.fin(0, FinWithException.of(taskStatus), vertex);
      286                      } catch (RuntimeException exception) {
      287                          LogUtils.error(log, "Run Task Fin:" + getId() + " catch operator:" + vertex.getId()
      288                              + " run Exception:", exception);
      289                          throw exception;
      290                      }
      291                  } catch (RuntimeException e) {
      292                      LogUtils.error(log, "Run Task:" + getId().toString()
      293                              + ",catch operator:" + vertex.getId() + " run Exception: ", e);
      294                      TaskStatus taskStatus = new TaskStatus();
      295                      taskStatus.setStatus(false);
      296                      taskStatus.setTaskId(vertex.getTask().getId().toString());
      297                      taskStatus.setErrorMsg(e.toString());
      298                      if (e instanceof WriteConflictException) {
      299                          taskStatus.setErrorType(ErrorType.WriteConflict);
      300                      } else if (e instanceof DuplicateEntryException) {
      301                          taskStatus.setErrorType(ErrorType.DuplicateEntry);
      302                      } else if (e instanceof LockWaitException) {
      303                          taskStatus.setErrorType(ErrorType.LockWait);
      304                          this.isLockWait.compareAndSet(false, true);
      305                      } else if (e instanceof TaskCancelException) {
      306                          taskStatus.setErrorType(ErrorType.TaskCancel);
      307                      } else {
      308                          taskStatus.setErrorType(ErrorType.TaskFin);
      309                      }
      310                      status.compareAndSet(Status.RUNNING, Status.STOPPED);
      311                      try {
      312                          operator.fin(0, FinWithException.of(taskStatus), vertex);
      313                      } catch (RuntimeException exception) {
      314                          LogUtils.error(log, "Run Task Fin:" + getId() + " catch operator:" + vertex.getId()
      315                              + " run Exception:", exception);
      316                          throw exception;
      317                      }
      318                  } finally {
      319                      activeTaskCount.decrementAndGet();
      320                      LogUtils.debug(log, "TaskImpl run cost: {}ms.", System.currentTimeMillis() - startTime);
      321                      MdcUtils.removeTxnId();
      322                      activeThreads.countDown();
      323                      // DingoTimeZoneContext.clear();
      324                  }
      325              });
      326          }
      327          while (true) {
      328              try {
      329                  activeThreads.await();
      330                  break;
      331              } catch (InterruptedException ignored) {
      332              }
      333          }
      334          status.compareAndSet(Status.RUNNING, Status.READY);
      335          status.compareAndSet(Status.STOPPED, Status.READY);
      336      }
