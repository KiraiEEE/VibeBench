       69      /**
       70       * Assign gold standard labels to each ReCiterArticle based on accepted/rejected PMID lists.
       71       *
       72       * @param reCiterArticles articles to label
       73       * @param acceptedPmids   PMIDs with ACCEPTED curator decision (goldStandard = 1)
       74       * @param rejectedPmids   PMIDs with REJECTED curator decision (goldStandard = -1)
       75       */
       76      public static void assignGoldStandard(List<ReCiterArticle> reCiterArticles,
       77                                            List<Long> acceptedPmids,
       78                                            List<Long> rejectedPmids) {
       79          Set<Long> acceptedSet = new HashSet<>();
       80          if (acceptedPmids != null) {
       81              acceptedSet.addAll(acceptedPmids);
       82          }
       83  
       84          for (ReCiterArticle article : reCiterArticles) {
       85              if (acceptedSet.contains(article.getArticleId())) {
       86                  article.setGoldStandard(1);
       87              }
       88          }
       89  
       90          if (rejectedPmids != null) {
       91              Set<Long> rejectedSet = new HashSet<>(rejectedPmids);
       92              for (ReCiterArticle article : reCiterArticles) {
       93                  if (rejectedSet.contains(article.getArticleId())) {
       94                      article.setGoldStandard(-1);
       95                  }
       96              }
       97          }
       98      }

      100      /**
      101       * Compute precision/recall/accuracy over a set of articles.
      102       *
      103       * <p>Called by {@code ReCiterController} (7 filter branches) and
      104       * {@code ReCiterFeatureGenerator}.</p>
      105       *
      106       * @param articles          articles to evaluate; each must have
      107       *                          {@code goldStandard} set (1 = ACCEPTED,
      108       *                          -1 = REJECTED, 0 = PENDING/NULL)
      109       * @param goldStandardPmids PMIDs of accepted (ground-truth positive)
      110       *                          articles for this person
      111       * @return populated Analysis object
      112       */
      113      public static Analysis performAnalysis(List<ReCiterArticle> articles,
      114                                             List<Long> goldStandardPmids) {
      115          Analysis analysis = new Analysis();
      116  
      117          if (goldStandardPmids == null || goldStandardPmids.isEmpty()) {
      118              return analysis;
      119          }
      120  
      121          Set<Long> goldSet = new HashSet<>(goldStandardPmids);
      122          Set<Long> articlePmids = new HashSet<>();
      123          int pendingSkipped = 0;
      124  
      125          analysis.setGoldStandardSize(goldStandardPmids.size());
      126          analysis.setSelectedClusterSize(articles.size());
      127  
      128          for (ReCiterArticle article : articles) {
      129              long pmid = article.getArticleId();
      130              articlePmids.add(pmid);
      131  
      132              // --- FIX: skip PENDING articles (goldStandard == 0) ---
      133              // These have no curator decision and must not inflate FP count.
      134              if (article.getGoldStandard() == 0) {
      135                  pendingSkipped++;
      136                  continue;
      137              }
      138  
      139              if (goldSet.contains(pmid)) {
      140                  // Ground-truth positive and present in article set → TP
      141                  analysis.getTruePositiveList().add(pmid);
      142              } else if (article.getGoldStandard() == -1) {
      143                  // Confirmed rejected and present in article set → FP
      144                  analysis.getFalsePositiveList().add(pmid);
      145              } else {
      146                  // goldStandard == 1 but not in goldSet — data inconsistency;
      147                  // treat conservatively as TN
      148                  analysis.getTrueNegativeList().add(pmid);
      149              }
      150          }
      151  
      152          // False negatives: accepted articles NOT present in the article set
      153          for (Long pmid : goldStandardPmids) {
      154              if (!articlePmids.contains(pmid)) {
      155                  analysis.getFalseNegativeList().add(pmid);
      156              }
      157          }
      158  
      159          analysis.setTruePos(analysis.getTruePositiveList().size());
      160          analysis.setTrueNeg(analysis.getTrueNegativeList().size());
      161          analysis.setFalsePos(analysis.getFalsePositiveList().size());
      162          analysis.setFalseNeg(analysis.getFalseNegativeList().size());
      163          analysis.setPendingSkippedCount(pendingSkipped);
      164  
      165          // Cache computed values
      166          analysis.setPrecision(analysis.getPrecision());
      167          analysis.setRecall(analysis.getRecall());
      168          analysis.setAccuracy(analysis.getAccuracy());
      169  
      170          return analysis;
      171      }
