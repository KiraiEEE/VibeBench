      118      protected Optional<String> generateQuerySql(Expr where) {
      119          String sql = QUERY_SQL;
      120          if (where != null) {
      121              ColumnRefFactory factory = new ColumnRefFactory();
      122              ScalarOperator p = SqlToScalarOperatorTranslator.translateWithSlotRef(where, slotRef -> {
      123                  if ("global".equalsIgnoreCase(slotRef.getColumnName())) {
      124                      return ConstantOperator.TRUE;
      125                  }
      126                  return factory.create(BASELINE_FIELD_TO_COLUMN_MAPPING.get(slotRef.getColumnName().toLowerCase()),
      127                          slotRef.getType(), true);
      128              });
      129              ScalarOperatorRewriter re = new ScalarOperatorRewriter();
      130              ScalarOperator result = re.rewrite(p, ScalarOperatorRewriter.DEFAULT_REWRITE_RULES);
      131              if (!result.isConstantRef()) {
      132                  Expr2SQLPrinter<Void> print = new Expr2SQLPrinter<>();
      133                  sql += " WHERE " + print.print(result);
      134              } else if (result.isConstantNullOrFalse()) {
      135                  return Optional.empty();
      136              }
      137          }
      138          return Optional.of(sql);
      139      }

      237      private List<BaselinePlan> castToBaselinePlan(List<TResultBatch> datas) {
      238          List<BaselinePlan> result = Lists.newArrayList();
      239          for (TResultBatch batch : ListUtils.emptyIfNull(datas)) {
      240              for (ByteBuffer buffer : batch.getRows()) {
      241                  ByteBuf copied = Unpooled.copiedBuffer(buffer);
      242                  String jsonString = copied.toString(Charset.defaultCharset());
      243                  /*
      244                   * SPM baselines table
      245                   * 0.id         | bigint
      246                   * 1.enable     | boolean
      247                   * 2.bind_sql   | varchar(65530)
      248                   * 3.bind_sql_digest | varchar(65530)
      249                   * 4.bind_sql_hash | bigint
      250                   * 5.plan_sql   | varchar(65530)
      251                   * 6.costs      | double
      252                   * 7.query_ms   | double
      253                   * 8.source     | varchar(100)
      254                   * 9.update_time| datetime
      255                   */
      256                  JsonElement obj = JsonParser.parseString(jsonString);
      257                  JsonArray data = obj.getAsJsonObject().get("data").getAsJsonArray();
      258                  BaselinePlan bp =
      259                          new BaselinePlan(data.get(2).getAsString(), data.get(3).getAsString(), data.get(4).getAsLong(),
      260                                  data.get(5).getAsString(), data.get(6).getAsDouble());
      261                  bp.setId(data.get(0).getAsLong());
      262                  bp.setGlobal(true);
      263                  bp.setEnable(data.get(1).getAsInt() == 1);
      264                  bp.setQueryMs(data.get(7).getAsDouble());
      265                  bp.setSource(data.get(8).getAsString());
      266                  bp.setUpdateTime(DateUtils.parseUnixDateTime(data.get(9).getAsString()));
      267                  result.add(bp);
      268              }
      269          }
      270          return result;
