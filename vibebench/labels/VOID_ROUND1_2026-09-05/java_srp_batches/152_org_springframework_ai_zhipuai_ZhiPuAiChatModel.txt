      323  				.execute(ctx -> this.zhiPuAiApi.chatCompletionStream(request));
      324

      326  			// The rest of the chunks with same ID share the same role.
      327  			ConcurrentHashMap<String, String> roleMap = new ConcurrentHashMap<>();
      328  
      329  			final ChatModelObservationContext observationContext = ChatModelObservationContext.builder()
      330  				.prompt(requestPrompt)
      331  				.provider(ZhiPuApiConstants.PROVIDER_NAME)

      334  			Observation observation = ChatModelObservationDocumentation.CHAT_MODEL_OPERATION.observation(
      335  					this.observationConvention, DEFAULT_OBSERVATION_CONVENTION, () -> observationContext,
      336  					this.observationRegistry);

      338  			observation.parentObservation(contextView.getOrDefault(ObservationThreadLocalAccessor.KEY, null)).start();
      339  
      340  			Flux<ChatResponse> chatResponse = completionChunks.map(this::chunkToChatCompletion)
      341  				.switchMap(chatCompletion -> Mono.just(chatCompletion).map(chatCompletion2 -> {
      342  					try {
      343  						String id = chatCompletion2.id();

      346  						List<Generation> generations = chatCompletion2.choices().stream().map(choice -> {
      347  							if (choice.message().role() != null) {
      348  								roleMap.putIfAbsent(id, choice.message().role().name());
      349  							}
      350  							Map<String, Object> metadata = Map.of(
      351  								"id", chatCompletion2.id(),
      352  								"role", roleMap.getOrDefault(id, ""),
      353  								"finishReason", choice.finishReason() != null ? choice.finishReason().name() : ""
      354  							);

      356  						}).toList();

      360  					}

      363  						return new ChatResponse(List.of());

      365  
      366  				}));
      367  
      368  			// @formatter:off

      371  							// FIXME: bounded elastic needs to be used since tool calling

      378  								}

      384  									return Flux.just(ChatResponse.builder().from(response)

      389  									// Send the tool execution result back to the model.
      390  									return this.stream(new Prompt(toolExecutionResult.conversationHistory(), requestPrompt.getOptions()));
      391  								}
      392  							}).subscribeOn(Schedulers.boundedElastic());
      393  						}
      394  						return Flux.just(response);
      395  			})
      396  			.doOnError(observation::error)
      397  			.doFinally(s -> observation.stop())
      398  			.contextWrite(ctx -> ctx.put(ObservationThreadLocalAccessor.KEY, observation));
      399  			// @formatter:on

      401  			return new MessageAggregator().aggregate(flux, observationContext::setResponse);

      446  	Prompt buildRequestPrompt(Prompt prompt) {
      447  		// Process runtime options
      448  		ZhiPuAiChatOptions runtimeOptions = null;
      449  		if (prompt.getOptions() != null) {
      450  			if (prompt.getOptions() instanceof ToolCallingChatOptions toolCallingChatOptions) {
      451  				runtimeOptions = ModelOptionsUtils.copyToTarget(toolCallingChatOptions, ToolCallingChatOptions.class,
      452  						ZhiPuAiChatOptions.class);
      453  			}
      454  			else {
      455  				runtimeOptions = ModelOptionsUtils.copyToTarget(prompt.getOptions(), ChatOptions.class,
      456  						ZhiPuAiChatOptions.class);
      457  			}
      458  		}
      459  
      460  		// Define request options by merging runtime options and default options
      461  		ZhiPuAiChatOptions requestOptions = ModelOptionsUtils.merge(runtimeOptions, this.defaultOptions,
      462  				ZhiPuAiChatOptions.class);
      463  
      464  		// Merge @JsonIgnore-annotated options explicitly since they are ignored by
      465  		// Jackson, used by ModelOptionsUtils.
      466  		if (runtimeOptions != null) {
      467  			requestOptions.setInternalToolExecutionEnabled(
      468  					ModelOptionsUtils.mergeOption(runtimeOptions.getInternalToolExecutionEnabled(),
      469  							this.defaultOptions.getInternalToolExecutionEnabled()));
      470  			requestOptions.setToolNames(ToolCallingChatOptions.mergeToolNames(runtimeOptions.getToolNames(),
      471  					this.defaultOptions.getToolNames()));
      472  			requestOptions.setToolCallbacks(ToolCallingChatOptions.mergeToolCallbacks(runtimeOptions.getToolCallbacks(),
      473  					this.defaultOptions.getToolCallbacks()));
      474  			requestOptions.setToolContext(ToolCallingChatOptions.mergeToolContext(runtimeOptions.getToolContext(),
      475  					this.defaultOptions.getToolContext()));
      476  		}
      477  		else {
      478  			requestOptions.setInternalToolExecutionEnabled(this.defaultOptions.getInternalToolExecutionEnabled());
      479  			requestOptions.setToolNames(this.defaultOptions.getToolNames());
      480  			requestOptions.setToolCallbacks(this.defaultOptions.getToolCallbacks());
      481  			requestOptions.setToolContext(this.defaultOptions.getToolContext());
      482  		}
      483  
      484  		ToolCallingChatOptions.validateToolCallbacks(requestOptions.getToolCallbacks());
      485  
      486  		return new Prompt(prompt.getInstructions(), requestOptions);
      487  	}
