      290      private void indexHeatSinks() {
      291          validHeatSinks.clear();
      292          coolantPerTick.clear();
      293          ArrayList<BlockPos> sortedHeatSinks = new ArrayList<>();
      294          for (String toCheck: FissionReactorRegistration.hsSchedule) {
      295              if (indexedHeatSinks.containsKey(toCheck)) {
      296                  HashSet<BlockPos> posSet = indexedHeatSinks.get(toCheck);
      297                  sortedHeatSinks.addAll(posSet);
      298                  for (BlockPos hsPos: posSet) {
      299                      validHeatSinks.put(hsPos.asLong(), (HeatSinkBlock) getBlockState(hsPos).getBlock());
      300                  }
      301              }
      302          }
      303          for (BlockPos hsPos: sortedHeatSinks) {
      304              long pos = hsPos.asLong();
      305              if (isHeatSinkValid(hsPos)) {
      306                  HeatSinkBlock hb = validHeatSinks.get(pos);
      307                  addSecondConnectionsToFuelCell(hsPos);
      308                  if(hb.isActive()) {
      309                      if (!activeHeatSinks.contains(pos)) {
      310                          addActiveCoolant(hb.def.name.replace("active_", ""));
      311                      }
      312                      addIfNotExists(pos, activeHeatSinks);
      313                  }
      314              } else {
      315                  if (activeHeatSinks.contains(pos)) {
      316                      activeHeatSinks.remove(pos);
      317                      removeActiveCoolant(validHeatSinks.get(pos).def.name.replace("active_", ""));
      318                  }
      319                  validHeatSinks.remove(pos);
      320                  removeSecondConnectionsToFuelCell(hsPos);
      321              }
      322          }
      323      }

      330      private void indexFuelCellAttachments() {
      331          cellsHeatMult = 0D;
      332          moderatorsHeatMult = 0D;
      333          cellsEnergyMult = 0D;
      334          moderatorsEnergyMult = 0D;
      335          for(long pos: fuelCells) {
      336              extraFuelCells = 0;
      337              extraFuelCells += countAdjacentFuelCells(BlockPos.of(pos));
      338              cellsHeatMult += (extraFuelCells + 1D)*(extraFuelCells + 2D)/2D;
      339              cellsEnergyMult += extraFuelCells + 1D;
      340              int moderators = getFuelCellModerators(pos);
      341              moderatorsHeatMult += moderators * (extraFuelCells+1D)*(FISSION_CONFIG.MODERATOR_HEAT_MULTIPLIER.get() / 100);
      342              moderatorsEnergyMult += moderators * (extraFuelCells+1D)*(FISSION_CONFIG.MODERATOR_FE_MULTIPLIER.get() / 100);
      343          }
      344      }
