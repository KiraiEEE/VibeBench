      303                      "Can't update non-dynamic configuration"));

      397      private void checkDeadlockedThreads() {
      398          ThreadMXBean threadBean = ManagementFactory.getThreadMXBean();
      399          long[] threadIds = threadBean.findDeadlockedThreads();
      400          if (threadIds != null && threadIds.length > 0) {
      401              ThreadInfo[] threadInfos = threadBean.getThreadInfo(threadIds, false, false);
      402              String threadNames = Arrays.stream(threadInfos)
      403                      .map(threadInfo -> threadInfo.getThreadName() + "(tid=" + threadInfo.getThreadId() + ")").collect(
      404                              Collectors.joining(", "));
      405              if (System.currentTimeMillis() - threadDumpLoggedTimestamp
      406                      > LOG_THREADDUMP_INTERVAL_WHEN_DEADLOCK_DETECTED) {
      407                  threadDumpLoggedTimestamp = System.currentTimeMillis();
      408                  LOG.error("Deadlocked threads detected. {}\n{}", threadNames,
      409                          ThreadDumpUtil.buildThreadDiagnosticString());
      410              } else {
      411                  LOG.error("Deadlocked threads detected. {}", threadNames);
      412              }
      413              throw new IllegalStateException("Deadlocked threads detected. " + threadNames);
      414          }
      415      }
