      170      public static Map<CharSequence, Object> toValueMap(Map<CharSequence, Object> values, VisitorContext context) {
      171          var newValues = new HashMap<CharSequence, Object>(values.size());
      172          for (Map.Entry<CharSequence, Object> entry : values.entrySet()) {
      173              CharSequence key = entry.getKey();
      174              Object value = entry.getValue();
      175  
      176              if (value instanceof AnnotationValue<?> av) {
      177                  final Map<CharSequence, Object> valueMap = toValueMap(av.getValues(), context);
      178                  newValues.put(key, valueMap);
      179              } else if (value instanceof AnnotationClassValue<?> acv) {
      180                  acv.getType().ifPresent(aClass -> newValues.put(key, aClass));
      181              } else if (value != null) {
      182                  if (value.getClass().isArray()) {
      183                      Object[] a = (Object[]) value;
      184                      if (ArrayUtils.isNotEmpty(a)) {
      185                          Object first = a[0];
      186                          boolean areAnnotationValues = first instanceof AnnotationValue;
      187                          boolean areClassValues = first instanceof AnnotationClassValue;
      188  
      189                          if (areClassValues) {
      190                              var classes = new ArrayList<Class<?>>(a.length);
      191                              for (Object o : a) {
      192                                  var acv = (AnnotationClassValue<?>) o;
      193                                  acv.getType().ifPresent(classes::add);
      194                              }
      195                              newValues.put(key, classes);
      196                          } else if (areAnnotationValues) {
      197                              String annotationName = ((AnnotationValue<?>) first).getAnnotationName();
      198                              if (io.swagger.v3.oas.annotations.security.SecurityRequirement.class.getName().equals(annotationName)) {
      199                                  var securityRequirements = new ArrayList<SecurityRequirement>(a.length);
      200                                  for (Object o : a) {
      201                                      securityRequirements.add(ConvertUtils.mapToSecurityRequirement((AnnotationValue<io.swagger.v3.oas.annotations.security.SecurityRequirement>) o));
      202                                  }
      203                                  newValues.put(key, securityRequirements);
      204                              } else if (Extension.class.getName().equals(annotationName)) {
      205                                  var extensions = new HashMap<String, Object>();
      206                                  for (Object o : a) {
      207                                      processExtensions(extensions, (AnnotationValue<Extension>) o);
      208                                  }
      209                                  newValues.put(PROP_EXTENSIONS, extensions);
      210                              } else if (Server.class.getName().equals(annotationName)) {
      211                                  var servers = new ArrayList<Map<CharSequence, Object>>();
      212                                  for (Object o : a) {
      213                                      var sv = (AnnotationValue<ServerVariable>) o;
      214                                      Map<CharSequence, Object> variables = new LinkedHashMap<>(toValueMap(sv.getValues(), context));
      215                                      servers.add(variables);
      216                                  }
      217                                  newValues.put(key, servers);
      218                              } else if (OAuthScope.class.getName().equals(annotationName)) {
      219                                  Map<String, String> params = toTupleSubMap(a, PROP_NAME, PROP_DESCRIPTION);
      220                                  newValues.put(key, params);
      221                              } else if (ServerVariable.class.getName().equals(annotationName)) {
      222                                  var variables = new LinkedHashMap<String, Map<CharSequence, Object>>();
      223                                  for (Object o : a) {
      224                                      var sv = (AnnotationValue<ServerVariable>) o;
      225                                      sv.stringValue(PROP_NAME).ifPresent(name -> {
      226                                          Map<CharSequence, Object> map = toValueMap(sv.getValues(), context);
      227                                          Object dv = map.get(PROP_DEFAULT_VALUE);
      228                                          if (dv != null) {
      229                                              map.put(PROP_DEFAULT, dv);
      230                                          }
      231                                          if (map.containsKey(PROP_ALLOWABLE_VALUES)) {
      232                                              // The key in the generated openapi needs to be "enum"
      233                                              map.put(PROP_ENUM, map.remove(PROP_ALLOWABLE_VALUES));
      234                                          }
      235                                          variables.put(name, map);
      236                                      });
      237                                  }
      238                                  newValues.put(key, variables);
      239                              } else {
      240                                  if (a.length == 1) {
      241                                      var av = (AnnotationValue<?>) a[0];
      242                                      final Map<CharSequence, Object> valueMap = toValueMap(av.getValues(), context);
      243                                      newValues.put(key, toValueMap(valueMap, context));
      244                                  } else {
      245  
      246                                      var list = new ArrayList<>();
      247                                      for (Object o : a) {
      248                                          if (o instanceof AnnotationValue<?> av) {
      249                                              final Map<CharSequence, Object> valueMap = toValueMap(av.getValues(), context);
      250                                              list.add(valueMap);
      251                                          } else {
      252                                              list.add(o);
      253                                          }
      254                                      }
      255                                      newValues.put(key, list);
      256                                  }
      257                              }
      258                          } else {
      259                              newValues.put(key, value);
      260                          }
      261                      } else {
      262                          newValues.put(key, a);
      263                      }
      264                  } else {
      265                      var parsedJsonValue = parseJsonString(value);
      266                      newValues.put(key, parsedJsonValue != null ? parsedJsonValue : value);
      267                  }
      268              }
      269          }
      270          return newValues;
      271      }

      296      public static <T> T treeToValue(JsonNode jn, Class<T> clazz, VisitorContext context) throws JsonProcessingException {
      297  
      298          var fixed = false;
      299          T value;
      300          try {
      301              value = CONVERT_JSON_MAPPER.treeToValue(jn, clazz);
      302          } catch (Exception e) {
      303              // maybe exception with groovy
      304              if (context.getLanguage() == VisitorContext.Language.GROOVY) {
      305                  value = fixForGroovy(jn, clazz, null);
      306                  fixed = true;
      307              } else {
      308                  throw e;
      309              }
      310          }
      311          if (!fixed && context.getLanguage() == VisitorContext.Language.GROOVY) {
      312              value = fixForGroovy(jn, clazz, null);
      313          }
      314  
      315          if (value == null) {
      316              return null;
      317          }
      318  
      319          var finalValue = value;
      320  
      321          var exts = resolveExtensions(jn);
      322          if (exts != null) {
      323              BeanMap.of(finalValue).put(PROP_EXTENSIONS, exts);
      324          }
      325          String elType = jn.has(PROP_TYPE) ? jn.get(PROP_TYPE).textValue() : null;
      326          String elFormat = jn.has(PROP_ONE_FORMAT) ? jn.get(PROP_ONE_FORMAT).textValue() : null;
      327          JsonNode defaultValueNode = jn.get(PROP_DEFAULT_VALUE);
      328          // fix for default value
      329          Object defaultValue;
      330          try {
      331              defaultValue = ConvertUtils.normalizeValue(defaultValueNode != null ? defaultValueNode.textValue() : null, elType, elFormat, context);
      332          } catch (JsonProcessingException e) {
      333              defaultValue = defaultValueNode != null ? defaultValueNode.textValue() : null;
      334          }
      335  
      336          BeanMap<T> beanMap = BeanMap.of(value);
      337          if (defaultValue != null) {
      338              beanMap.put(PROP_DEFAULT, defaultValue);
      339          }
      340  
      341          JsonNode allowableValuesNode = jn.get(PROP_ALLOWABLE_VALUES);
      342          if (allowableValuesNode != null && allowableValuesNode.isArray()) {
      343              var allowableValues = new ArrayList<>(allowableValuesNode.size());
      344              for (JsonNode allowableValueNode : allowableValuesNode) {
      345                  if (allowableValueNode == null) {
      346                      continue;
      347                  }
      348                  try {
      349                      allowableValues.add(ConvertUtils.normalizeValue(allowableValueNode.textValue(), elType, elFormat, context));
      350                  } catch (IOException e) {
      351                      allowableValues.add(allowableValueNode.textValue());
      352                  }
      353              }
      354              beanMap.put(PROP_ALLOWABLE_VALUES, allowableValues);
      355          }
      356  
      357          return value;
      358      }

      360      private static <T> Map<String, T> deserMap(String name, JsonNode jn, Class<T> clazz) throws JsonProcessingException {
      361  
      362          var mapNode = jn.get(name);
      363          if (mapNode == null) {
      364              return null;
      365          }
      366          ((ObjectNode) jn).remove(name);
      367  
      368          var iter = mapNode.fieldNames();
      369          var result = new HashMap<String, T>();
      370          while (iter.hasNext()) {
      371              var entryKey = iter.next();
      372              var objectNode = mapNode.get(entryKey);
      373              var object = CONVERT_JSON_MAPPER.treeToValue(objectNode, clazz);
      374              result.put(entryKey, object);
      375          }
      376          return !result.isEmpty() ? result : null;
      377      }

      379      private static <T> T fixForGroovy(JsonNode jn, Class<T> clazz, Exception e) throws JsonProcessingException {
      380  
      381          // fix for problem with groovy. Jackson throw exception with Operation class with content and mediaType
      382          // see https://github.com/micronaut-projects/micronaut-openapi/issues/1418
      383          if (clazz == Operation.class) {
      384  
      385              var requestBodyNode = jn.get("requestBody");
      386              ((ObjectNode) jn).remove("requestBody");
      387              T value = CONVERT_JSON_MAPPER.treeToValue(jn, clazz);
      388              var requestBody = fixContentForGroovy(requestBodyNode, RequestBody.class);
      389              ((Operation) value).setRequestBody(requestBody);
      390  
      391              var responsesNode = jn.get(PROP_RESPONSES);
      392              ((ObjectNode) jn).remove(PROP_RESPONSES);
      393              ApiResponses responses = null;
      394              if (responsesNode != null && !responsesNode.isEmpty()) {
      395                  responses = new ApiResponses();
      396                  for (var entry : responsesNode.properties()) {
      397                      responses.put(entry.getKey(), fixContentForGroovy(entry.getValue(), ApiResponse.class));
      398                  }
      399              }
      400              ((Operation) value).setResponses(responses);
      401              return value;
      402          } else if (clazz == ApiResponse.class
      403              || clazz == Header.class
      404              || clazz == Parameter.class
      405              || clazz == RequestBody.class) {
      406              return fixContentForGroovy(jn, clazz);
      407          } else {
      408              return CONVERT_JSON_MAPPER.treeToValue(jn, clazz);
      409          }
      410      }

      412      private static <T> T fixContentForGroovy(JsonNode parentNode, Class<T> clazz) throws JsonProcessingException {
      413          if (parentNode == null) {
      414              return null;
      415          }
      416          Map<String, Example> examples = null;
      417          Map<String, Encoding> encoding = null;
      418          Map<String, Object> extensions = null;
      419          Schema<?> schema = null;
      420          JsonNode mediaTypeNode = null;
      421  
      422          var contentNode = parentNode.get(PROP_CONTENT);
      423          if (contentNode != null) {
      424              examples = deserMap(PROP_EXAMPLES, contentNode, Example.class);
      425              encoding = deserMap("encoding", contentNode, Encoding.class);
      426              extensions = deserMap(PROP_EXTENSIONS, contentNode, Object.class);
      427              var schemaNode = contentNode.get(PROP_SCHEMA);
      428              if (schemaNode != null) {
      429                  schema = CONVERT_JSON_MAPPER.treeToValue(schemaNode, Schema.class);
      430              }
      431  
      432              mediaTypeNode = contentNode.get(PROP_MEDIA_TYPE);
      433              ((ObjectNode) contentNode).remove(PROP_MEDIA_TYPE);
      434          }
      435          var value = CONVERT_JSON_MAPPER.treeToValue(parentNode, clazz);
      436          Content content = null;
      437          if (value instanceof ApiResponse apiResponse) {
      438              content = apiResponse.getContent();
      439          } else if (value instanceof Header header) {
      440              content = header.getContent();
      441          } else if (value instanceof Parameter parameter) {
      442              content = parameter.getContent();
      443          } else if (value instanceof RequestBody requestBody) {
      444              content = requestBody.getContent();
      445          }
      446  
      447          if (content != null) {
      448              var mediaType = content.get(PROP_SCHEMA);
      449              content.remove(PROP_SCHEMA);
      450              if (mediaType == null) {
      451                  mediaType = new MediaType();
      452              }
      453              var contentType = mediaTypeNode != null ? mediaTypeNode.textValue() : io.micronaut.http.MediaType.APPLICATION_JSON;
      454              content.put(contentType, mediaType);
      455              mediaType.setExamples(examples);
      456              mediaType.setEncoding(encoding);
      457              if (extensions != null) {
      458                  extensions.forEach(mediaType::addExtension);
      459              }
      460              mediaType.setSchema(schema);
      461          }
      462  
      463          return value;
      464      }
