      164      public static List<Geocache> searchByFilter(@NonNull final GeocacheFilter filter, @NonNull final SuConnector connector) throws SuApiException {
      165  
      166          //for now we have to assume that SUConnector supports only SINGLE criteria search
      167  
      168          final List<BaseGeocacheFilter> filters = filter.getAndChainIfPossible();
      169          final OriginGeocacheFilter of = GeocacheFilter.findInChain(filters, OriginGeocacheFilter.class);
      170          if (of != null && !of.allowsCachesOf(connector)) {
      171              return new ArrayList<>();
      172          }
      173          final DistanceGeocacheFilter df = GeocacheFilter.findInChain(filters, DistanceGeocacheFilter.class);
      174          if (df != null) {
      175              return searchByCenter(df.getEffectiveCoordinate(), df.getMaxRangeValue() == null ? 20f : df.getMaxRangeValue(), connector);
      176          }
      177          final NameGeocacheFilter nf = GeocacheFilter.findInChain(filters, NameGeocacheFilter.class);
      178          if (nf != null && !StringUtils.isEmpty(nf.getStringFilter().getTextValue())) {
      179              return searchByKeyword(nf.getStringFilter().getTextValue(), connector);
      180          }
      181          final OwnerGeocacheFilter ownf = GeocacheFilter.findInChain(filters, OwnerGeocacheFilter.class);
      182          if (ownf != null && !StringUtils.isEmpty(ownf.getStringFilter().getTextValue())) {
      183              return searchByOwner(ownf.getStringFilter().getTextValue(), connector);
      184          }
      185          final SelfOwnedGeocacheFilter sof = GeocacheFilter.findInChain(filters, SelfOwnedGeocacheFilter.class);
      186          if (sof != null) {
      187              final String ownName = SelfOwnedGeocacheFilter.getOwnerNameForConnector(connector);
      188              if (StringUtils.isNotEmpty(ownName)) {
      189                  return searchByOwner(ownName, connector);
      190              }
      191          }
      192  
      193          //by default, search around current position
      194          return searchByCenter(LocationDataProvider.getInstance().currentGeo().getCoords(), 20f, connector);
      195      }

      403      @NonNull
      404      @WorkerThread
      405      private static JSONResult request(@NonNull final SuConnector connector, @NonNull final SuApiEndpoint endpoint, @NonNull final String method, @NonNull final Parameters params) throws SuApiException {
      406          final String host = connector.getHost();
      407          if (StringUtils.isBlank(host)) {
      408              return new JSONResult("unknown connector host");
      409          }
      410  
      411          final OAuthTokens tokens = new OAuthTokens(connector);
      412          if (!tokens.isValid()) {
      413              throw new NotAuthorizedException();
      414          }
      415          OAuth.signOAuth(host, endpoint.methodName, method, connector.isHttps(), params, tokens, connector.getConsumerKey(), connector.getConsumerSecret());
      416  
      417          final String uri = connector.getHostUrl() + endpoint.methodName;
      418          final JSONResult result;
      419          try {
      420              if ("GET".equals(method)) {
      421                  result = new JSONResult(Network.getRequest(uri, params).blockingGet());
      422              } else {
      423                  result = new JSONResult(Network.postRequest(uri, params).blockingGet());
      424              }
      425          } catch (final RuntimeException e) {
      426              if (e.getCause() instanceof InterruptedException) {
      427                  Log.w("SuApi.JSONResult Interrupted");
      428                  return new JSONResult("interrupted");
      429              }
      430              Log.e("SuApi.JSONResult unknown error", e);
      431              throw new ConnectionErrorException();
      432          } catch (final Exception e) {
      433              Log.e("SuApi.JSONResult unknown error", e);
      434              throw new ConnectionErrorException();
      435          }
      436  
      437          if (!result.isSuccess) {
      438              Log.w("JSONResult exception, failed request");
      439              throw new ConnectionErrorException();
      440          }
      441          if (result.data.get("status").get("code").toString().contains("ERROR") && result.data.get("status").get("type").toString().contains("AuthorisationRequired")) {
      442              throw new NotAuthorizedException();
      443          }
      444          return result;
      445      }
