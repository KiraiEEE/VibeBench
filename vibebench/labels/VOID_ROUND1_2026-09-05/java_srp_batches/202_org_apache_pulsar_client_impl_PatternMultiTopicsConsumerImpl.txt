      128      CompletableFuture<Void> recheckTopicsChange() {
      129          String pattern = topicsPattern.inputPattern();
      130          final int epoch = recheckPatternEpoch.incrementAndGet();
      131          return client.getLookup().getTopicsUnderNamespace(namespaceName, subscriptionMode, pattern, topicsHash)
      132              .thenCompose(getTopicsResult -> {
      133                  // If "recheckTopicsChange" has been called more than one times, only make the last one take affects.
      134                  // Use "synchronized (recheckPatternTaskBackoff)" instead of
      135                  // `synchronized(PatternMultiTopicsConsumerImpl.this)` to avoid locking in a wider range.
      136                  synchronized (PatternMultiTopicsConsumerImpl.this) {
      137                      if (recheckPatternEpoch.get() > epoch) {
      138                          return CompletableFuture.completedFuture(null);
      139                      }
      140                      if (log.isDebugEnabled()) {
      141                          log.debug("Pattern consumer [{}] get topics under namespace {}, topics.size: {},"
      142                                          + " topicsHash: {}, filtered: {}",
      143                                  PatternMultiTopicsConsumerImpl.this.getSubscription(),
      144                                  namespaceName, getTopicsResult.getTopics().size(), getTopicsResult.getTopicsHash(),
      145                                  getTopicsResult.isFiltered());
      146                          getTopicsResult.getTopics().forEach(topicName ->
      147                                  log.debug("Get topics under namespace {}, topic: {}", namespaceName, topicName));
      148                      }
      149  
      150                      final List<String> oldTopics = new ArrayList<>(getPartitions());
      151                      return updateSubscriptions(topicsPattern, this::setTopicsHash, getTopicsResult,
      152                              topicsChangeListener, oldTopics, subscription);
      153                  }
      154              }).thenAccept(__ -> {
      155                  if (recheckPatternTimeout != null) {
      156                      this.recheckPatternTimeout = client.timer().newTimeout(
      157                              this, Math.max(1, conf.getPatternAutoDiscoveryPeriod()), TimeUnit.SECONDS);
      158                  }
      159              });
      160      }

      162      static CompletableFuture<Void> updateSubscriptions(TopicsPattern topicsPattern,
      163                                                         java.util.function.Consumer<String> topicsHashSetter,
      164                                                         GetTopicsResult getTopicsResult,
      165                                                         TopicsChangedListener topicsChangedListener,
      166                                                         List<String> oldTopics,
      167                                                         String subscriptionForLog) {
      168          topicsHashSetter.accept(getTopicsResult.getTopicsHash());
      169          if (!getTopicsResult.isChanged()) {
      170              return CompletableFuture.completedFuture(null);
      171          }
      172  
      173          List<String> newTopics;
      174          if (getTopicsResult.isFiltered()) {
      175              newTopics = getTopicsResult.getNonPartitionedOrPartitionTopics();
      176          } else {
      177              newTopics = getTopicsResult.filterTopics(topicsPattern).getNonPartitionedOrPartitionTopics();
      178          }
      179  
      180          final List<CompletableFuture<?>> listenersCallback = new ArrayList<>(2);
      181          Set<String> topicsAdded = TopicList.minus(newTopics, oldTopics);
      182          Set<String> topicsRemoved = TopicList.minus(oldTopics, newTopics);
      183          if (log.isDebugEnabled()) {
      184              log.debug("Pattern consumer [{}] Recheck pattern consumer's topics. topicsAdded: {}, topicsRemoved: {}",
      185                      subscriptionForLog, topicsAdded, topicsRemoved);
      186          }
      187          listenersCallback.add(topicsChangedListener.onTopicsAdded(topicsAdded));
      188          listenersCallback.add(topicsChangedListener.onTopicsRemoved(topicsRemoved));
      189          return FutureUtil.waitForAll(Collections.unmodifiableList(listenersCallback));
      190      }

      222          @Override
      223          public CompletableFuture<Void> onTopicsRemoved(Collection<String> removedTopics) {
      224              if (removedTopics.isEmpty()) {
      225                  return CompletableFuture.completedFuture(null);
      226              }
      227  
      228              // Unsubscribe and remove consumers in memory.
      229              List<CompletableFuture<Void>> unsubscribeList = new ArrayList<>(removedTopics.size());
      230              Set<String> partialRemoved = new HashSet<>(removedTopics.size());
      231              Set<String> partialRemovedForLog = new HashSet<>(removedTopics.size());
      232              for (String tp : removedTopics) {
      233                  TopicName topicName = TopicName.get(tp);
      234                  ConsumerImpl<T> consumer = consumers.get(topicName.toString());
      235                  if (consumer != null) {
      236                      CompletableFuture<Void> unsubscribeFuture = new CompletableFuture<>();
      237                      consumer.closeAsync().whenComplete((__, ex) -> {
      238                          if (ex != null) {
      239                              log.error("Pattern consumer [{}] failed to unsubscribe from topics: {}",
      240                                      PatternMultiTopicsConsumerImpl.this.getSubscription(), topicName.toString(), ex);
      241                              unsubscribeFuture.completeExceptionally(ex);
      242                          } else {
      243                              consumers.remove(topicName.toString(), consumer);
      244                              unsubscribeFuture.complete(null);
      245                          }
      246                      });
      247                      unsubscribeList.add(unsubscribeFuture);
      248                      partialRemoved.add(topicName.getPartitionedTopicName());
      249                      partialRemovedForLog.add(topicName.toString());
      250                  }
      251              }
      252              if (log.isDebugEnabled()) {
      253                  log.debug("Pattern consumer [{}] remove topics. {}",
      254                          PatternMultiTopicsConsumerImpl.this.getSubscription(),
      255                          partialRemovedForLog);
      256              }
      257  
      258              // Remove partitioned topics in memory.
      259              return FutureUtil.waitForAll(unsubscribeList).handle((__, ex) -> {
      260                  List<String> removedPartitionedTopicsForLog = new ArrayList<>();
      261                  for (String groupedTopicRemoved : partialRemoved) {
      262                      Integer partitions = partitionedTopics.get(groupedTopicRemoved);
      263                      if (partitions != null) {
      264                          boolean allPartitionsHasBeenRemoved = true;
      265                          for (int i = 0; i < partitions; i++) {
      266                              if (consumers.containsKey(
      267                                      TopicName.get(groupedTopicRemoved).getPartition(i).toString())) {
      268                                  allPartitionsHasBeenRemoved = false;
      269                                  break;
      270                              }
      271                          }
      272                          if (allPartitionsHasBeenRemoved) {
      273                              removedPartitionedTopicsForLog.add(String.format("%s with %s partitions",
      274                                      groupedTopicRemoved, partitions));
      275                              partitionedTopics.remove(groupedTopicRemoved, partitions);
      276                          }
      277                      }
      278                  }
      279                  if (log.isDebugEnabled()) {
      280                      log.debug("Pattern consumer [{}] remove partitioned topics because all partitions have been"
      281                                      + " removed. {}", PatternMultiTopicsConsumerImpl.this.getSubscription(),
      282                              removedPartitionedTopicsForLog);
      283                  }
      284                  return null;
      285              });

      288          /**
      289           * {@inheritDoc}
      290           */
      291          @Override
      292          public CompletableFuture<Void> onTopicsAdded(Collection<String> addedTopics) {
      293              if (addedTopics.isEmpty()) {
      294                  return CompletableFuture.completedFuture(null);
      295              }
      296              List<CompletableFuture<Void>> futures = Lists.newArrayListWithExpectedSize(addedTopics.size());
      297              /**
      298               * Three normal cases:
      299               *  1. Expand partitions.
      300               *  2. Non-partitioned topic, but has been subscribing.
      301               *  3. Non-partitioned topic or Partitioned topic, but has not been subscribing.
      302               * Two unexpected cases:
      303               *   Error-1: Received adding non-partitioned topic event, but has subscribed a partitioned topic with the
      304               *     same name.
      305               *   Error-2: Received adding partitioned topic event, but has subscribed a non-partitioned topic with the
      306               *     same name.
      307               *
      308               * Note: The events that triggered by {@link TopicsPartitionChangedListener} after expanding partitions has
      309               *    been disabled through "conf.setAutoUpdatePartitions(false)" when creating
      310               *    {@link PatternMultiTopicsConsumerImpl}.
      311               */
      312              Set<String> groupedTopics = new HashSet<>();
      313              List<String> expendPartitionsForLog = new ArrayList<>();
      314              for (String tp : addedTopics) {
      315                  TopicName topicName = TopicName.get(tp);
      316                  groupedTopics.add(topicName.getPartitionedTopicName());
      317              }
      318              for (String tp : addedTopics) {
      319                  TopicName topicName = TopicName.get(tp);
      320                  // Case 1: Expand partitions.
      321                  if (partitionedTopics.containsKey(topicName.getPartitionedTopicName())) {
      322                      if (consumers.containsKey(topicName.toString())) {
      323                          // Already subscribed.
      324                      } else if (topicName.getPartitionIndex() < 0) {
      325                          // Error-1: Received adding non-partitioned topic event, but has subscribed a partitioned topic
      326                          // with the same name.
      327                          log.error("Pattern consumer [{}] skip to subscribe to the non-partitioned topic {}, because has"
      328                                  + "subscribed a partitioned topic with the same name",
      329                                  PatternMultiTopicsConsumerImpl.this.getSubscription(), topicName.toString());
      330                      } else {
      331                          if (topicName.getPartitionIndex() + 1
      332                                  > partitionedTopics.get(topicName.getPartitionedTopicName())) {
      333                              partitionedTopics.put(topicName.getPartitionedTopicName(),
      334                                      topicName.getPartitionIndex() + 1);
      335                          }
      336                          expendPartitionsForLog.add(topicName.toString());
      337                          CompletableFuture consumerFuture = subscribeAsync(topicName.toString(),
      338                                  PartitionedTopicMetadata.NON_PARTITIONED);
      339                          consumerFuture.whenComplete((__, ex) -> {
      340                              if (ex != null) {
      341                                  log.warn("Pattern consumer [{}] Failed to subscribe to topics: {}",
      342                                          PatternMultiTopicsConsumerImpl.this.getSubscription(), topicName, ex);
      343                              }
      344                          });
      345                          futures.add(consumerFuture);
      346                      }
      347                      groupedTopics.remove(topicName.getPartitionedTopicName());
      348                  } else if (consumers.containsKey(topicName.toString())) {
      349                      // Case-2: Non-partitioned topic, but has been subscribing.
      350                      groupedTopics.remove(topicName.getPartitionedTopicName());
      351                  } else if (consumers.containsKey(topicName.getPartitionedTopicName())
      352                          && topicName.getPartitionIndex() >= 0) {
      353                      // Error-2: Received adding partitioned topic event, but has subscribed a non-partitioned topic
      354                      // with the same name.
      355                      log.error("Pattern consumer [{}] skip to subscribe to the partitioned topic {}, because has"
      356                                      + "subscribed a non-partitioned topic with the same name",
      357                              PatternMultiTopicsConsumerImpl.this.getSubscription(), topicName);
      358                      groupedTopics.remove(topicName.getPartitionedTopicName());
      359                  }
      360              }
      361              // Case 3: Non-partitioned topic or Partitioned topic, which has not been subscribed.
      362              for (String partitionedTopic : groupedTopics) {
      363                  CompletableFuture consumerFuture = subscribeAsync(partitionedTopic, false);
      364                  consumerFuture.whenComplete((__, ex) -> {
      365                      if (ex != null) {
      366                          log.warn("Pattern consumer [{}] Failed to subscribe to topics: {}",
      367                                  PatternMultiTopicsConsumerImpl.this.getSubscription(), partitionedTopic, ex);
      368                      }
      369                  });
      370                  futures.add(consumerFuture);
      371              }
      372              if (log.isDebugEnabled()) {
      373                  log.debug("Pattern consumer [{}] add topics. expend partitions {}, new subscribing {}",
      374                          PatternMultiTopicsConsumerImpl.this.getSubscription(), expendPartitionsForLog, groupedTopics);
      375              }
      376              return FutureUtil.waitForAll(futures);
      377          }
      378      }

      380      @Override
      381      @SuppressFBWarnings
      382      public CompletableFuture<Void> closeAsync() {
      383          Timeout timeout = recheckPatternTimeout;
      384          if (timeout != null) {
      385              timeout.cancel();
      386              recheckPatternTimeout = null;
      387          }
      388          List<CompletableFuture<?>> closeFutures = new ArrayList<>(2);
      389          if (watcherFuture.isDone() && !watcherFuture.isCompletedExceptionally()) {
      390              TopicListWatcher watcher = watcherFuture.getNow(null);
      391              // watcher can be null when subscription mode is not persistent
      392              if (watcher != null) {
      393                  closeFutures.add(watcher.closeAsync());
      394              }
      395          }
      396          closeFutures.add(updateTaskQueue.cancelAllAndWaitForTheRunningTask().thenCompose(__ -> super.closeAsync()));
      397          return FutureUtil.waitForAll(closeFutures);
      398      }
