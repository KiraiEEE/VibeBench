      276      protected void extractFile(
      277              final File srcF,
      278              final File dir,
      279              final InputStream compressedInputStream,
      280              String entryName,
      281              final Date entryDate,
      282              final boolean isDirectory,
      283              final Integer mode,
      284              String symlinkDestination,
      285              final FileMapper[] fileMappers)
      286              throws IOException, ArchiverException {
      287          if (fileMappers != null) {
      288              for (final FileMapper fileMapper : fileMappers) {
      289                  entryName = fileMapper.getMappedFileName(entryName);
      290              }
      291          }
      292  
      293          // For symlinks, we need to get the file path without following symlinks.
      294          // FileUtils.resolveFile calls getCanonicalFile() which follows symlinks,
      295          // causing the symlink to resolve to its target instead of the symlink itself.
      296          final File targetFileName;
      297          if (!StringUtils.isEmpty(symlinkDestination)) {
      298              // For symlinks, use simple path resolution without canonicalization
      299              targetFileName = resolveFileWithoutFollowingSymlinks(dir, entryName);
      300          } else {
      301              // For regular files and directories, use the existing logic
      302              targetFileName = FileUtils.resolveFile(dir, entryName);
      303          }
      304  
      305          // Make sure that the resolved path of the extracted file doesn't escape the destination directory
      306          // getCanonicalFile().toPath() is used instead of getCanonicalPath() (returns String),
      307          // because "/opt/directory".startsWith("/opt/dir") would return false negative.
      308          Path canonicalDirPath = dir.getCanonicalFile().toPath();
      309  
      310          // For symlinks, we need to check the symlink path itself, not the target it points to
      311          Path canonicalDestPath;
      312          if (!StringUtils.isEmpty(symlinkDestination)) {
      313              // For symlinks, normalize without following the link
      314              canonicalDestPath = targetFileName.toPath().toAbsolutePath().normalize();
      315          } else {
      316              canonicalDestPath = targetFileName.getCanonicalFile().toPath();
      317          }
      318  
      319          if (!canonicalDestPath.startsWith(canonicalDirPath)) {
      320              throw new ArchiverException("Entry is outside of the target directory (" + entryName + ")");
      321          }
      322  
      323          // don't allow override target symlink by standard file
      324          if (StringUtils.isEmpty(symlinkDestination) && Files.isSymbolicLink(canonicalDestPath)) {
      325              throw new ArchiverException("Entry is outside of the target directory (" + entryName + ")");
      326          }
      327  
      328          try {
      329              if (!shouldExtractEntry(dir, targetFileName, entryName, entryDate)) {
      330                  return;
      331              }
      332  
      333              // create intermediary directories - sometimes zip don't add them
      334              final File dirF = targetFileName.getParentFile();
      335              if (dirF != null) {
      336                  dirF.mkdirs();
      337              }
      338  
      339              if (!StringUtils.isEmpty(symlinkDestination)) {
      340                  SymlinkUtils.createSymbolicLink(targetFileName, new File(symlinkDestination));
      341              } else if (isDirectory) {
      342                  targetFileName.mkdirs();
      343              } else {
      344                  Files.copy(compressedInputStream, targetFileName.toPath(), REPLACE_EXISTING);
      345              }
      346  
      347              targetFileName.setLastModified(entryDate.getTime());
      348  
      349              if (!isIgnorePermissions() && mode != null && !isDirectory) {
      350                  ArchiveEntryUtils.chmod(targetFileName, mode);
      351              }
      352          } catch (final FileNotFoundException ex) {
      353              getLogger().warn("Unable to expand to file " + targetFileName.getPath());
      354          }
      355      }

      362      // Visible for testing
      363      protected boolean shouldExtractEntry(File targetDirectory, File targetFileName, String entryName, Date entryDate)
      364              throws IOException {
      365          //     entryname  | entrydate | filename   | filedate | behavior
      366          // (1) readme.txt | 1970      | -          | -        | always extract if the file does not exist
      367          // (2) readme.txt | 1970      | readme.txt | 2020     | do not overwrite unless isOverwrite() is true
      368          // (3) readme.txt | 2020      | readme.txt | 1970     | always override when the file is older than the archive
      369          // entry
      370          // (4) README.txt | 1970      | readme.txt | 2020     | case-insensitive filesystem: warn + do not overwrite
      371          // unless isOverwrite()
      372          //                                                      case-sensitive filesystem: extract without warning
      373          // (5) README.txt | 2020      | readme.txt | 1970     | case-insensitive filesystem: warn + overwrite because
      374          // entry is newer
      375          //                                                      case-sensitive filesystem: extract without warning
      376  
      377          // The canonical file name follows the name of the archive entry, but takes into account the case-
      378          // sensitivity of the filesystem. So on a case-sensitive file system, file.exists() returns false for
      379          // scenario (4) and (5).
      380          // No matter the case sensitivity of the file system, file.exists() returns false when there is no file with the
      381          // same name (1).
      382          if (!targetFileName.exists()) {
      383              return true;
      384          }
      385  
      386          boolean entryIsDirectory =
      387                  entryName.endsWith("/"); // directory entries always end with '/', regardless of the OS.
      388          String canonicalDestPath = targetFileName.getCanonicalPath();
      389          String suffix = (entryIsDirectory ? "/" : "");
      390          String relativeCanonicalDestPath =
      391                  canonicalDestPath.replace(targetDirectory.getCanonicalPath() + File.separatorChar, "") + suffix;
      392          boolean fileOnDiskIsOlderThanEntry = targetFileName.lastModified() < entryDate.getTime();
      393          boolean differentCasing =
      394                  !normalizedFileSeparator(entryName).equals(normalizedFileSeparator(relativeCanonicalDestPath));
      395  
      396          // Warn for case (4) and (5) if the file system is case-insensitive
      397          if (differentCasing) {
      398              String casingMessage = String.format(
      399                      Locale.ENGLISH,
      400                      "Archive entry '%s' and existing file '%s' names differ only by case."
      401                              + " This may lead to an unexpected outcome on case-insensitive filesystems.",
      402                      entryName,
      403                      canonicalDestPath);
      404              getLogger().warn(casingMessage);
      405              casingMessageEmitted.incrementAndGet();
      406          }
      407  
      408          // Override the existing file if isOverwrite() is true or if the file on disk is older than the one in the
      409          // archive
      410          return isOverwrite() || fileOnDiskIsOlderThanEntry;
      411      }

      413      private String normalizedFileSeparator(String pathOrEntry) {
      414          return pathOrEntry.replace("/", File.separator);
      415      }

      417      /**
      418       * Resolves a file path relative to a base directory without following symlinks.
      419       * This is similar to FileUtils.resolveFile but doesn't call getCanonicalFile(),
      420       * which would follow symlinks and resolve them to their targets.
      421       *
      422       * @param baseDir the base directory
      423       * @param filename the filename to resolve
      424       * @return the resolved file
      425       */
      426      private File resolveFileWithoutFollowingSymlinks(File baseDir, String filename) {
      427          String filenm = filename;
      428          if ('/' != File.separatorChar) {
      429              filenm = filename.replace('/', File.separatorChar);
      430          }
      431  
      432          if ('\\' != File.separatorChar) {
      433              filenm = filenm.replace('\\', File.separatorChar);
      434          }
      435  
      436          // For absolute paths, just return a File object without canonicalization
      437          if (filenm.startsWith(File.separator)) {
      438              return new File(filenm);
      439          }
      440  
      441          // For relative paths, combine with base directory and get absolute path
      442          // but don't call getCanonicalFile() which would follow symlinks
      443          return new File(baseDir, filenm).getAbsoluteFile();
      444      }
