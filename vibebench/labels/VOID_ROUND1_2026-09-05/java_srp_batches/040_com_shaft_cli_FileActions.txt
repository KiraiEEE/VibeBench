      235                      terminalSession.getSshPortNumber(), terminalSession.getSshUsername(),
      236                      terminalSession.getSshKeyFileFolderName(), terminalSession.getSshKeyFileName());
      237              // copy from docker to machine
      238              terminalSessionForRemoteMachine.performTerminalCommand("rm -r " + pathToTempDirectoryOnRemoteMachine[0]);
      239              terminalSessionForRemoteMachine
      240                      .performTerminalCommand("mkdir -p " + pathToTempDirectoryOnRemoteMachine[0] + targetFileFolderPath);
      241              terminalSessionForRemoteMachine.performTerminalCommand("docker cp " + terminalSession.getDockerName() + ":"
      242                      + targetFilePath + " " + pathToTempDirectoryOnRemoteMachine[0] + targetFilePath);
      243              targetFilePath = pathToTempDirectoryOnRemoteMachine[0] + targetFilePath;
      244          }
      245          // fetch file from terminal session to local machine
      246          if (terminalSession.isRemoteTerminal()) {
      247              // remote regular

      249                      terminalSession.getSshKeyFileName()) + " -P " + terminalSession.getSshPortNumber();

      251              String pathToRemoteFileThatWillBeCopied = targetFilePath;

      253                      + pathToRemoteFileThatWillBeCopied;
      254  
      255              // creating local temp directory
      256              String pathToLocalParentFolder = FileActions.getInstance(true).getAbsolutePath("target/temp");

      399                  }

      402          }
