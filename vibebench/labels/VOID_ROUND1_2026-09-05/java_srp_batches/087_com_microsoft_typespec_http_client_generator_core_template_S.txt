      396          public void writeStreamStyleSerialization(JavaClass classBlock) {
      397              if (model.getXmlName() != null) {
      398                  writeToXml(classBlock);
      399                  if (isSuperTypeWithDiscriminator(model)) {
      400                      writeSuperTypeFromXml(classBlock);
      401                  } else {
      402                      writeTerminalTypeFromXml(classBlock);
      403                  }
      404              } else {
      405                  if (isJsonMergePatchModel) {
      406                      writeToJson(classBlock, true);
      407                      addGeneratedAnnotation.accept(classBlock);
      408                      classBlock.privateMethod("JsonWriter toJsonMergePatch(JsonWriter jsonWriter) throws IOException",
      409                          methodBlock -> serializeJsonProperties(methodBlock, true, false, false));
      410                  } else {
      411                      writeToJson(classBlock, false);
      412                  }
      413  
      414                  // All classes will create a public fromJson(JsonReader) method that initiates reading.
      415                  // How the implementation looks depends on whether the type is a super type, subtype, both, or is a
      416                  // stand-alone type.
      417                  //
      418                  // Intermediate types, those that are both a super type and subtype, will pass null as the type
      419                  // discriminator
      420                  // value. This is done as super types are written to only support themselves or their subtypes, passing
      421                  // a
      422                  // discriminator into its own super type would confuse this scenario. For example, one of the test
      423                  // Swaggers
      424                  // generates the following hierarchy
      425                  //
      426                  // Fish
      427                  // Salmon Shark
      428                  // SmartSalmon Sawshark GoblinShark Cookiecuttershark
      429                  //
      430                  // If Salmon called into Fish with its discriminator and an error occurred it would mention the Shark
      431                  // subtypes
      432                  // as potential legal values for deserialization, confusing the Salmon deserializer. So, calling into
      433                  // Salmon
      434                  // will only attempt to handle Salmon and SmartSalmon, and same goes for Shark with Shark, Sawshark,
      435                  // GoblinShark, and Cookiecuttershark. It's only Fish that will handle all subtypes, as this is the most
      436                  // generic
      437                  // super type. This also creates a well-defined bounds for code generation in regard to type
      438                  // hierarchies.
      439                  //
      440                  // In a real scenario someone deserializing to Salmon should have an exception about discriminator types
      441                  // if the JSON payload is a Shark and not get a ClassCastException as it'd be more confusing on why a
      442                  // Shark
      443                  // was trying to be converted to a Salmon.
      444                  if (isSuperTypeWithDiscriminator(model)) {
      445                      writeSuperTypeFromJson(classBlock);
      446                  } else {
      447                      readJsonObject(classBlock, false, this::writeFromJsonDeserialization);
      448                  }
      449  
      450                  if (isManagementErrorSubclass.test(model)) {
      451                      addGeneratedAnnotation.accept(classBlock);
      452                      classBlock.staticMethod(JavaVisibility.Private,
      453                          model.getName() + " " + READ_MANAGEMENT_ERROR_METHOD_NAME
      454                              + "(JsonReader jsonReader) throws IOException",
      455                          methodBlock -> readJsonObjectMethodBody(methodBlock, this::writeFromJsonDeserialization0));
      456                  }
      457              }

      461           * write toJson() method.
      462           * <p>
      463           * If it is a JsonMergePatch model, toJson() should first check jsonMergePatch flag value and then do
      464           * serialization accordingly.

      466           * @param classBlock The class block to write the toJson method to.

      469          private void writeToJson(JavaClass classBlock, boolean isJsonMergePatch) {
      470              boolean callToJsonSharedForParentProperties = settings.isShareJsonSerializableCode()
      471                  && !isJsonMergePatch
      472                  && model.isAllPolymorphicModelsInSamePackage()
      473                  && !CoreUtils.isNullOrEmpty(model.getParentModelName());
      474              boolean callToJsonSharedForThisProperties = settings.isShareJsonSerializableCode()
      475                  && !isJsonMergePatch
      476                  && model.isAllPolymorphicModelsInSamePackage()
      477                  && model.isPolymorphicParent();
      478  
      479              classBlock.javadocComment(JavaJavadocComment::inheritDoc);
      480              addGeneratedAnnotation.accept(classBlock);
      481              classBlock.annotation("Override");
      482              classBlock.publicMethod("JsonWriter toJson(JsonWriter jsonWriter) throws IOException", methodBlock -> {
      483                  if (isJsonMergePatch) {
      484                      // If the model is the root parent use the JSON merge patch serialization tracking property
      485                      // directly,
      486                      // otherwise use the access helper to determine whether to use JSON merge patch serialization.
      487                      ClientModel rootParent = ClientModelUtil.getRootParent(model);
      488                      String ifStatement = (rootParent == model)
      489                          ? "jsonMergePatch"
      490                          : JSON_MERGE_PATCH_HELPER_CLASS_NAME + ".get" + rootParent.getName()
      491                              + "Accessor().isJsonMergePatch(this)";
      492  
      493                      methodBlock.ifBlock(ifStatement, ifBlock -> ifBlock.methodReturn("toJsonMergePatch(jsonWriter)"))
      494                          .elseBlock(elseBlock -> serializeJsonProperties(methodBlock, false, false, false));
      495                  } else {
      496                      // For polymorphic models, when all models are contained in the same package, a package-private
      497                      // 'toJsonShared'
      498                      // method will be generated by parent models that handles serialization of properties defined by the
      499                      // parent
      500                      // model.
      501                      // So, if we have that case, we skip serializing the properties defined by the parent model(s) and
      502                      // instead call
      503                      // the 'toJsonShared' method to serialize those properties. And, for properties defined by this
      504                      // model, if the
      505                      // model is a parent model generate a 'toJsonShared' method to serialize those properties.
      506                      //
      507                      // At this time, due to implementation complexity and the low value add, JSON merge patch
      508                      // serialization won't
      509                      // support 'toJsonMergePatchShared'. JSON merge patch serialization has low usage, and combined with
      510                      // the low
      511                      // usage of polymorphism as well, there isn't a large expectation for this feature.
      512                      serializeJsonProperties(methodBlock, false, callToJsonSharedForParentProperties,
      513                          callToJsonSharedForThisProperties);
      514                  }
      515              });
      516  
      517              // If this model is defining a 'toJsonShared' method, write it.
      518              if (callToJsonSharedForThisProperties) {
      519                  classBlock.packagePrivateMethod("void toJsonShared(JsonWriter jsonWriter) throws IOException",
      520                      methodBlock -> {
      521                          if (callToJsonSharedForParentProperties) {
      522                              methodBlock.line("super.toJsonShared(jsonWriter);");
      523                          }
      524  
      525                          serializeThisJsonProperties(
      526                              (property, fromSuper) -> serializeJsonProperty(methodBlock, property,
      527                                  property.getSerializedName(), fromSuper, true, false),
      528                              methodBlock, callToJsonSharedForParentProperties);
      529                      });
      530              }
      531          }

      540              boolean callToJsonSharedForParentProperties, boolean callToJsonSharedForThisProperties) {
      541              methodBlock.line("jsonWriter.writeStartObject();");
      542  
      543              // If we're calling toJsonShared for this model and for parent properties, this is an early out.
      544              if (callToJsonSharedForParentProperties && callToJsonSharedForThisProperties) {
      545                  methodBlock.line("toJsonShared(jsonWriter);");
      546                  methodBlock.methodReturn("jsonWriter.writeEndObject()");
      547                  return;
      548              }
      549  
      550              BiConsumer<ClientModelProperty, Boolean> serializeJsonProperty
      551                  = (property, fromSuper) -> serializeJsonProperty(methodBlock, property, property.getSerializedName(),
      552                      fromSuper, true, isJsonMergePatch);
      553

      555                  methodBlock.line("toJsonShared(jsonWriter);");
      556              } else {
      557                  serializeParentJsonProperties(serializeJsonProperty);
      558              }

      562              } else {

      564              }

      685           * <p>
      686           * If the JSON property needs to be flattened or is additional properties this is a no-op as those require
      687           * special handling that will occur later.

      689           * @param methodBlock The method handling serialization.
      690           * @param property The property being serialized.
      691           * @param serializedName The serialized JSON property name. Generally, this is just the
      692           * {@code property property's} serialized name but if a flattened property is being serialized it'll be the last
      693           * segment of the flattened JSON name.
      694           * @param fromSuperType Whether the property is defined by a super type of the model. If the property is
      695           * declared by a super type a getter method will be used to retrieve the value instead of accessing the field
      696           * directly.
      697           * @param isJsonMergePatch Whether the serialization is for a JSON Merge Patch model.
      698           */
      699          private void serializeJsonProperty(JavaBlock methodBlock, ClientModelProperty property, String serializedName,
      700              boolean fromSuperType, boolean isJsonMergePatch) {
      701              IType clientType = property.getClientType();
      702              IType wireType = property.getWireType();
      703              String propertyValueGetter = getPropertyGetterStatement(property, fromSuperType);
      704  
      705              // Attempt to determine whether the wire type is simple serialization.
      706              // This is primitives, boxed primitives, a small set of string based models, and other ClientModels.
      707              String fieldSerializationMethod = wireType.jsonSerializationMethodCall("jsonWriter", serializedName,
      708                  propertyValueGetter, isJsonMergePatch);
      709              if (wireType == ClassType.BINARY_DATA) {
      710                  // Special handling for BinaryData (instead of using "serializationMethodBase" and
      711                  // "serializationValueGetterModifier")
      712                  // The reason is that some backend would fail the request on "null" value (e.g. OpenAI)
      713                  String writeBinaryDataExpr = propertyValueGetter + ".writeTo(jsonWriter);";
      714                  if (!property.isRequired()) {
      715                      methodBlock.ifBlock(propertyValueGetter + " != null", ifAction -> {
      716                          ifAction.line("jsonWriter.writeFieldName(\"" + serializedName + "\");");
      717                          ifAction.line(writeBinaryDataExpr);
      718                      });
      719                  } else {
      720                      methodBlock.line("jsonWriter.writeFieldName(\"" + serializedName + "\");");
      721                      methodBlock.line(writeBinaryDataExpr);
      722                  }
      723              } else if (fieldSerializationMethod != null) {
      724                  if (isJsonMergePatch && wireType instanceof ClassType && ((ClassType) wireType).isSwaggerType()) {
      725                      methodBlock.line("JsonMergePatchHelper.get" + clientType.toString()
      726                          + "Accessor().prepareModelForJsonMergePatch(" + propertyValueGetter + ", true);");
      727                  }
      728                  writeSerializeJsonPropertyViaFieldSerializationMethod(methodBlock, property, model, serializedName,
      729                      fieldSerializationMethod, fromSuperType);
      730                  if (isJsonMergePatch && wireType instanceof ClassType && ((ClassType) wireType).isSwaggerType()) {
      731                      methodBlock.line("JsonMergePatchHelper.get" + clientType.toString()
      732                          + "Accessor().prepareModelForJsonMergePatch(" + propertyValueGetter + ", false);");
      733                  }
      734              } else if (wireType == ClassType.OBJECT) {
      735                  if (!property.isRequired() && !property.isRequiredForCreate()
      736                  // In json-merge-patch, we don't need to check whether the optional untyped property is null again.
      737                      && !isJsonMergePatch) {
      738                      methodBlock.ifBlock(propertyValueGetter + " != null", ifBlock -> ifBlock.line(
      739                          "jsonWriter.writeUntypedField(\"" + serializedName + "\", " + propertyValueGetter + ");"));
      740                  } else {
      741                      methodBlock
      742                          .line("jsonWriter.writeUntypedField(\"" + serializedName + "\", " + propertyValueGetter + ");");
      743                  }
      744              } else if (wireType instanceof IterableType) {
      745                  serializeJsonContainerProperty(methodBlock, "writeArrayField", wireType,
      746                      ((IterableType) wireType).getElementType(), serializedName, propertyValueGetter, 0,
      747                      isJsonMergePatch);
      748              } else if (wireType instanceof MapType) {
      749                  // Assumption is that the key type for the Map is a String. This may not always hold true and when that
      750                  // becomes reality this will need to be reworked to handle that case.
      751                  serializeJsonContainerProperty(methodBlock, "writeMapField", wireType,
      752                      ((MapType) wireType).getValueType(), serializedName, propertyValueGetter, 0, isJsonMergePatch);
      753              } else {
      754                  // TODO (alzimmer): Resolve this as deserialization logic generation needs to handle all cases.
      755                  throw new RuntimeException(
      756                      "Unknown wire type " + wireType + " in serialization. Need to add support for it.");
      757              }
      758          }
