       99      private void advanceVarStoreSlot() throws NetworkFuzzerException {
      100          varStoreSlot++;
      101  
      102          if (varStoreSlot >= BlockEntityVariablestore.INVENTORY_SIZE) {
      103              // Current store is full, create a new one on top
      104              BlockEntityVariablestore currentStore = getCurrentVarStore();
      105              BlockPos currentStorePos = currentStore.getBlockPos();
      106  
      107              // Place new variable store on top of the current one
      108              BlockPos newStorePos = currentStorePos.above();
      109              if (!level.isEmptyBlock(newStorePos)) {
      110                  throw new NetworkFuzzerException("Cannot place new variable store at " + newStorePos + " - block already exists");
      111              }
      112  
      113              level.setBlock(newStorePos, RegistryEntries.BLOCK_VARIABLE_STORE.get().defaultBlockState(), 2);
      114  
      115              // Get the new variable store block entity
      116              if (!(level.getBlockEntity(newStorePos) instanceof BlockEntityVariablestore newStore)) {
      117                  throw new NetworkFuzzerException("Failed to get new variable store block entity at " + newStorePos);
      118              }
      119  
      120              // Add it to our list and update indices
      121              varStores.add(newStore);
      122              currentVarStoreIndex++;
      123              varStoreSlot = 0;
      124          }

      127      /**
      128       * Generate a fuzzed network with three phases:
      129       * 1. Select a random writer part and aspect
      130       * 2. Build an operator chain that produces the required input type
      131       * 3. Select random readers to provide inputs to the operator chain
      132       *
      133       * @throws NetworkFuzzerException if an error occurs during network generation
      134       * @return If the generation succeeded (false if it failed due to lack of space or parts, true otherwise)
      135       */
      136      public boolean generate() throws NetworkFuzzerException {
      137          // Phase 1: Select writer
      138          Pair<IPartTypeWriter<?, ?>, IAspectWrite<?, ?>> writerAspect = selectRandomWriter();
      139  
      140          IPartTypeWriter<?, ?> writerType = writerAspect.getLeft();
      141          IAspectWrite<?, ?> writeAspect = writerAspect.getRight();
      142          IValueType<?> writerInputType = writeAspect.getValueType();
      143  
      144          // Place writer part
      145          Pair<BlockPos, Direction> writerPos = selectRandomOuterFace();
      146          if (writerPos == null) {
      147              return false; // No space to place writer
      148          }
      149  
      150          PartHelpers.addPart(level, writerPos.getLeft(), writerPos.getRight(),
      151                  writerType, new ItemStack(writerType.getItem()));
      152  
      153          // Set up context blocks for the writer
      154          setupWriterContextBlocks(writerType, writerPos.getLeft(), writerPos.getRight());
      155  
      156          // Phase 2: Build operator chain
      157          List<IOperator> operatorChain = buildOperatorChain(writerInputType);
      158  
      159          // Phase 3: Select readers for unmapped inputs
      160          List<OperatorInputNeeded> inputsNeeded = determineInputsNeeded(operatorChain);
      161          Map<IOperator, OperatorVariableAssignment> operatorAssignments = new HashMap<>();
      162          for (IOperator op : operatorChain) {
      163              operatorAssignments.put(op, new OperatorVariableAssignment(op, op.getRequiredInputLength()));
      164          }
      165          fulfillInputs(inputsNeeded, operatorChain, operatorAssignments, writerPos.getLeft(), writerPos.getRight(), writeAspect);
      166  
      167          return true;
      168      }

      174          List<IPartTypeWriter<?, ?>> writers = new ArrayList<>();
      175  
      176          // Collect all registered writer parts by checking if they implement IPartTypeWriter
      177          for (Object partType : PartTypes.REGISTRY.getPartTypes()) {
      178              if (partType instanceof IPartTypeWriter<?, ?> writer && !writer.getWriteAspects().isEmpty()) {
      179                  writers.add(writer);
      180              }
      181          }
      182  
      183          if (writers.isEmpty()) {
      184              throw new NetworkFuzzerException("No writer parts are available");
      185          }
      186  
      187          IPartTypeWriter<?, ?> writerType = writers.get(random.nextInt(writers.size()));
      188          List<?> writeAspects = writerType.getWriteAspects();
      189  
      190          IAspectWrite<?, ?> aspect = (IAspectWrite<?, ?>) writeAspects.get(random.nextInt(writeAspects.size()));
      191          return Pair.of(writerType, aspect);

      199          Set<BlockPos> cableSet = new HashSet<>(cables);
      200          List<Pair<BlockPos, Direction>> outerFaces = new ArrayList<>();
      201  
      202          for (BlockPos cable : cables) {
      203              for (Direction dir : Direction.values()) {
      204                  BlockPos adjacent = cable.relative(dir);
      205                  if (!cableSet.contains(adjacent) && PartHelpers.getPart(PartPos.of(DimPos.of(level, cable), dir)) == null) {
      206                      outerFaces.add(Pair.of(cable, dir));

      208              }
      209          }

      211          if (outerFaces.isEmpty()) {
      212              return null;
      213          }
      214  
      215          return outerFaces.get(random.nextInt(outerFaces.size()));

      218      /**
      219       * Build an operator chain that produces the required value type.
      220       * Returns a list of operators in order from input to output.
      221       *
      222       * @throws NetworkFuzzerException if building the chain fails
      223       */
      224      private List<IOperator> buildOperatorChain(IValueType<?> requiredType) throws NetworkFuzzerException {
      225          List<IOperator> chain = new ArrayList<>();
      226          IValueType<?> currentType = requiredType;
      227          int depth = random.nextInt(maxOperatorDepth + 1);
      228  
      229          for (int i = 0; i < depth; i++) {
      230              // Find an operator that produces currentType
      231              IOperator op = findRandomOperatorProducing(currentType);
      232              if (op == null) {
      233                  break;
      234              }
      235              chain.addFirst(op); // Add to front since we're building backwards
      236  
      237              // Get the first input type of this operator
      238              IValueType<?>[] inputTypes = op.getInputTypes();
      239              if (inputTypes.length > 0) {
      240                  currentType = inputTypes[0];
      241              } else {
      242                  break;
      243              }
      244          }
      245  
      246          return chain;
      247      }

      271          // Narrow down category types

      274              List<IValueType> valueTypes = ValueTypes.REGISTRY.getValueTypes().stream()
      275                      .filter(valueTypeCategory::correspondsTo)
      276                      .toList();
      277              valueType = valueTypes.get(random.nextInt(valueTypes.size()));
      278          }
      279  
      280          List<IOperator> result = new ArrayList<>();
      281  
      282          try {
      283              // Use the registry's built-in method to get operators with the specified output type
      284              IValueType<?> finalValueType = valueType;
      285              result.addAll(Operators.REGISTRY.getOperatorsWithOutputType(valueType)
      286                      // Only keep operators that require entity inputs at a 10% chance.
      287                      // Otherwise, the TARGETENTITY operator is overwhelmingly common since it is the only one that produces an entity type.
      288                      .stream().filter(operator -> random.nextInt(10) == 9 || finalValueType == ValueTypes.OBJECT_ENTITY || Arrays.stream(operator.getInputTypes()).noneMatch(vt -> vt == ValueTypes.OBJECT_ENTITY)).toList());
      289          } catch (Exception e) {
      290              // If the registry query fails, throw an exception
      291              throw new NetworkFuzzerException("Failed to find operators for type " + valueType, e);
      292          }
