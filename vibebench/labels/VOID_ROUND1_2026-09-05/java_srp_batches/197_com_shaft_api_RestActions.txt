      174          try {
      175              return parseJsonBody(body);
      176          } catch (Exception e) {
      177              // response is not parsable to JSON
      178              ByteArrayOutputStream byteArrayOutputStream = new ByteArrayOutputStream();
      179              ObjectOutputStream oos;
      180              try {
      181                  oos = new ObjectOutputStream(byteArrayOutputStream);
      182                  oos.writeObject(body);
      183                  oos.flush();
      184                  oos.close();
      185                  return new ByteArrayInputStream(byteArrayOutputStream.toByteArray());
      186              } catch (IOException ioe) {
      187                  if (body.getClass().getName().toLowerCase().contains("restassured")) {
      188                      // if it's a string response body
      189                      return ((ResponseBody<?>) body).asInputStream();
      190                  } else {
      191                      return new ByteArrayInputStream((body.toString()).getBytes());

      193              }
