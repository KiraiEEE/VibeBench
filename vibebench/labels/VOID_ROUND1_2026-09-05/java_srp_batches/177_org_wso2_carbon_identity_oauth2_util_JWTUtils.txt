      143       * Retrieves the signing tenant domain from the given JWT claims set and AccessTokenDO, considering various sources
      144       * such as the 'realm' claim, the 'signing_tenant' claim, and the OAuth application associated with the AccessToken.
      145       *
      146       * @param claimsSet     The JWTClaimsSet containing the claims of the JWT, including the 'realm' claim.
      147       * @param accessTokenDO The AccessTokenDO associated with the OAuth token.
      148       * @return The signing tenant domain based on the provided claims and AccessTokenDO.
      149       * @throws ParseException          If an error occurs while parsing the JWT claims.
      150       * @throws IdentityOAuth2Exception If an error occurs in OAuth2-related functionality.

      152      @SuppressWarnings({"unchecked", "rawtypes"})
      153      public static String getSigningTenantDomain(JWTClaimsSet claimsSet, AccessTokenDO accessTokenDO)
      154              throws ParseException, IdentityOAuth2Exception {
      155  
      156          Map<String, String> realm = (HashMap) claimsSet.getClaim(OAuthConstants.OIDCClaims.REALM);
      157          if (MapUtils.isNotEmpty(realm)) {
      158              if (realm.get(OAuthConstants.OIDCClaims.SIGNING_TENANT) != null) {
      159                  if (log.isDebugEnabled()) {
      160                      log.debug("Getting signing tenant domain from JWT's 'signing_tenant' claim.");
      161                  }
      162                  return realm.get(OAuthConstants.OIDCClaims.SIGNING_TENANT);
      163              } else if (realm.get(OAuthConstants.OIDCClaims.TENANT) != null) {
      164                  if (log.isDebugEnabled()) {
      165                      log.debug("Getting signing tenant domain from JWT's 'tenant' claim.");
      166                  }
      167                  return realm.get(OAuthConstants.OIDCClaims.TENANT);
      168              }
      169          }
      170          if (accessTokenDO == null) {
      171              return getTenantDomain();
      172          }
      173          boolean isJWTSignedWithSPKey = OAuthServerConfiguration.getInstance().isJWTSignedWithSPKey();
      174          if (isJWTSignedWithSPKey) {
      175              try {
      176                  if (log.isDebugEnabled()) {
      177                      log.debug("Getting signing tenant domain from OAuth app.");
      178                  }
      179                  return OAuth2Util.getTenantDomainOfOauthApp(accessTokenDO.getConsumerKey());
      180              } catch (InvalidOAuthClientException e) {
      181                  throw new IdentityOAuth2Exception("Error while getting tenant domain from OAuth app with consumer key: "
      182                          + accessTokenDO.getConsumerKey());
      183              }
      184          } else {
      185              if (log.isDebugEnabled()) {
      186                  log.debug("Getting signing tenant domain from authenticated user.");
      187              }
      188              return accessTokenDO.getAuthzUser().getTenantDomain();
      189          }

      212       * @param claimsSet    The JWTClaimsSet containing the claims of the JWT, including the issuer

      217      public static IdentityProvider getResidentIDPForIssuer(JWTClaimsSet claimsSet, String tenantDomain)
      218              throws IdentityOAuth2Exception {
      219  
      220          String issuer = StringUtils.EMPTY;
      221          IdentityProvider residentIdentityProvider;
      222          try {
      223              residentIdentityProvider = IdentityProviderManager.getInstance().getResidentIdP(tenantDomain);
      224          } catch (IdentityProviderManagementException e) {

      226                      String.format("Error while getting Resident Identity Provider of '%s' tenant.", tenantDomain);
      227              throw new IdentityOAuth2Exception(errorMsg, e);
      228          }
      229          FederatedAuthenticatorConfig[] fedAuthnConfigs = residentIdentityProvider.getFederatedAuthenticatorConfigs();
      230          FederatedAuthenticatorConfig oauthAuthenticatorConfig =
      231                  IdentityApplicationManagementUtil.getFederatedAuthenticator(fedAuthnConfigs,
      232                          IdentityApplicationConstants.Authenticator.OIDC.NAME);
      233          if (oauthAuthenticatorConfig != null) {
      234              issuer = IdentityApplicationManagementUtil.getProperty(oauthAuthenticatorConfig.getProperties(),

      326      /**
      327       * Verifies the signature of the given SignedJWT using the provided X.509 certificate and signature algorithm.
      328       *
      329       * @param signedJWT       The SignedJWT to verify.
      330       * @param x509Certificate The X.509 certificate used for signature verification.
      331       * @param alg             The signature algorithm.
      332       * @return True if the signature is valid, false otherwise.
      333       * @throws IdentityOAuth2Exception If an error occurs during signature verification.
      334       * @throws JOSEException           If an error occurs in the JOSE library.
      335       */
      336      public static boolean verifySignature(SignedJWT signedJWT, X509Certificate x509Certificate, String alg)
      337              throws IdentityOAuth2Exception, JOSEException {
      338  
      339          JWSVerifier verifier = null;
      340          if (alg.indexOf(ALGO_PREFIX) == 0 || alg.indexOf(ALGO_PREFIX_PS) == 0) {
      341              // At this point 'x509Certificate' will never be null.
      342              PublicKey publicKey = x509Certificate.getPublicKey();
      343              if (publicKey instanceof RSAPublicKey) {
      344                  verifier = new RSASSAVerifier((RSAPublicKey) publicKey);
      345              } else {
      346                  throw new IdentityOAuth2Exception("Public key is not an RSA public key.");
      347              }
      348          } else {
      349              if (log.isDebugEnabled()) {
      350                  log.debug("Signature Algorithm not supported yet: " + alg);
      351              }
      352          }
      353          if (verifier == null) {
      354              throw new IdentityOAuth2Exception("Could not create a signature verifier for algorithm type: " + alg);
      355          }
      356          boolean isValid;
      357          isValid = signedJWT.verify(verifier);
      358          if (log.isDebugEnabled()) {
      359              log.debug("Signature verified: " + isValid);
      360          }
      361          return isValid;
      362      }

      365       * Retrieves an X.509 certificate from the JWT claims, specifically from the 'realm' claim,
      366       * considering the signing tenant information. If available, the certificate is obtained from the tenant's keystore.
      367       *
      368       * @param jwtClaimsSet The JWTClaimsSet containing the claims, including the 'realm' claim with signing tenant
      369       *                     information.
      370       * @return An Optional containing the X.509 certificate if found, or an empty Optional if not available.
      371       * @throws IdentityOAuth2Exception If an error occurs during the retrieval of the X.509 certificate.
      372       */
      373      @SuppressWarnings({"unchecked", "rawtypes"})
      374      public static Optional<X509Certificate> getCertificateFromClaims(JWTClaimsSet jwtClaimsSet)
      375              throws IdentityOAuth2Exception {
      376  
      377          X509Certificate x509Certificate = null;
      378          Map<String, String> realm = (HashMap) jwtClaimsSet.getClaim(OAuthConstants.OIDCClaims.REALM);
      379          // Get certificate from tenant if available in claims.
      380          if (MapUtils.isNotEmpty(realm)) {
      381              String tenantDomain = null;
      382              // Get signed key tenant from JWT token or ID token based on claim key.
      383              if (realm.get(OAuthConstants.OIDCClaims.SIGNING_TENANT) != null) {
      384                  tenantDomain = realm.get(OAuthConstants.OIDCClaims.SIGNING_TENANT);
      385              } else if (realm.get(OAuthConstants.OIDCClaims.TENANT) != null) {
      386                  tenantDomain = realm.get(OAuthConstants.OIDCClaims.TENANT);
      387              }
      388              if (tenantDomain != null) {
      389                  int tenantId = IdentityTenantUtil.getTenantId(tenantDomain);
      390                  // Retrieve the X.509 certificate from the tenant's keystore.
      391                  x509Certificate = (X509Certificate) OAuth2Util.getCertificate(tenantDomain, tenantId);
      392              }
      393          }
      394          return Optional.ofNullable(x509Certificate);
      395      }
