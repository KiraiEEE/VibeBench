      110    public int find(final K key, final BinarySerializer<K> keySerializer,
      111        BinarySerializerFactory serializerFactory) {
      112      var low = 0;
      113      var high = size() - 1;
      114  
      115      while (low <= high) {
      116        var mid = (low + high) >>> 1;
      117        var midVal = getKey(mid, keySerializer, serializerFactory);
      118        var cmp = DefaultComparator.INSTANCE.compare(midVal, key);
      119  
      120        if (cmp < 0) {
      121          low = mid + 1;
      122        } else if (cmp > 0) {
      123          high = mid - 1;
      124        } else {
      125          return mid; // key found
      126        }
      127      }
      128      return -(low + 1); // key not found.
