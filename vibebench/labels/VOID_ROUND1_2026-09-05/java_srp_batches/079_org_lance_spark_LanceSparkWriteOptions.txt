      383       * @param options the options map

      386      public Builder fromOptions(Map<String, String> options) {
      387        this.storageOptions = new HashMap<>(options);
      388        if (options.containsKey(CONFIG_WRITE_MODE)) {
      389          this.writeMode = WriteMode.valueOf(options.get(CONFIG_WRITE_MODE).toUpperCase());
      390        }
      391        if (options.containsKey(CONFIG_MAX_ROWS_PER_FILE)) {
      392          this.maxRowsPerFile = Integer.parseInt(options.get(CONFIG_MAX_ROWS_PER_FILE));
      393        }
      394        if (options.containsKey(CONFIG_MAX_ROWS_PER_GROUP)) {
      395          this.maxRowsPerGroup = Integer.parseInt(options.get(CONFIG_MAX_ROWS_PER_GROUP));
      396        }
      397        if (options.containsKey(CONFIG_MAX_BYTES_PER_FILE)) {
      398          this.maxBytesPerFile = Long.parseLong(options.get(CONFIG_MAX_BYTES_PER_FILE));
      399        }
      400        if (options.containsKey(CONFIG_DATA_STORAGE_VERSION)) {
      401          this.dataStorageVersion = options.get(CONFIG_DATA_STORAGE_VERSION);
      402        }
      403        if (options.containsKey(CONFIG_USE_QUEUED_WRITE_BUFFER)) {
      404          this.useQueuedWriteBuffer =
      405              Boolean.parseBoolean(options.get(CONFIG_USE_QUEUED_WRITE_BUFFER));
      406        }
      407        if (options.containsKey(CONFIG_QUEUE_DEPTH)) {
      408          this.queueDepth = Integer.parseInt(options.get(CONFIG_QUEUE_DEPTH));
      409        }
      410        if (options.containsKey(CONFIG_BATCH_SIZE)) {
      411          int parsedBatchSize = Integer.parseInt(options.get(CONFIG_BATCH_SIZE));
      412          Preconditions.checkArgument(parsedBatchSize > 0, "batch_size must be positive");
      413          this.batchSize = parsedBatchSize;
      414        }
      415        return this;
      416      }
