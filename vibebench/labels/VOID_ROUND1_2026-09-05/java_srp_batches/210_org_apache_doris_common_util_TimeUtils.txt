      200      public static Date parseDate(String dateStr, PrimitiveType type) throws AnalysisException {
      201          Date date = null;
      202          Matcher matcher = DATETIME_FORMAT_REG.matcher(dateStr);
      203          if (!matcher.matches()) {
      204              throw new AnalysisException("Invalid date string: " + dateStr);
      205          }
      206          dateStr = formatDateStr(dateStr);
      207          if (type == PrimitiveType.DATE) {
      208              ParsePosition pos = new ParsePosition(0);
      209              date = Date.from(
      210                      LocalDate.from(getDateFormatWithTimeZone().parse(dateStr, pos)).atStartOfDay()
      211                              .atZone(getDorisZoneId()).toInstant());
      212              if (pos.getIndex() != dateStr.length() || date == null) {
      213                  throw new AnalysisException("Invalid date string: " + dateStr);
      214              }
      215          } else if (type == PrimitiveType.DATETIME) {
      216              try {
      217                  date = Date.from(LocalDateTime.parse(dateStr, getDatetimeFormatWithTimeZone())
      218                          .atZone(getDorisZoneId()).toInstant());
      219              } catch (DateTimeParseException e) {
      220                  throw new AnalysisException("Invalid date string: " + dateStr);
      221              }
      222          } else {
      223              Preconditions.checkState(false, "error type: " + type);
      224          }
      225  
      226          return date;
      227      }

      272      // Check if the time zone_value is valid
      273      public static String checkTimeZoneValidAndStandardize(String value) throws DdlException {
      274          Function<String, String> standardizeValue = s -> {
      275              boolean positive = s.charAt(0) != '-';
      276              String[] parts = s.replaceAll("[+-]", "").split(":");
      277              return (positive ? "+" : "-") + String.format("%02d:%02d",
      278                      Integer.parseInt(parts[0]), Integer.parseInt(parts[1]));
      279          };
      280  
      281          try {
      282              if (value == null) {
      283                  ErrorReport.reportDdlException(ErrorCode.ERR_UNKNOWN_TIME_ZONE, "null");
      284              }
      285              // match offset type, such as +08:00, -07:00
      286              Matcher matcher = TIMEZONE_OFFSET_FORMAT_REG.matcher(value);
      287              // it supports offset and region timezone type, and short zone ids
      288              boolean match = matcher.matches();
      289              if (!value.contains("/") && !timeZoneAliasMap.containsKey(value) && !match) {
      290                  if ((value.startsWith("GMT") || value.startsWith("UTC"))
      291                          && TIMEZONE_OFFSET_FORMAT_REG.matcher(value.substring(3)).matches()) {
      292                      value = value.substring(0, 3) + standardizeValue.apply(value.substring(3));
      293                  } else {
      294                      ErrorReport.reportDdlException(ErrorCode.ERR_UNKNOWN_TIME_ZONE, value);
      295                  }
      296              }
      297              if (match) {
      298                  value = standardizeValue.apply(value);
      299  
      300                  // timezone offsets around the world extended from -12:00 to +14:00
      301                  int tz = Integer.parseInt(value.substring(1, 3)) * 100 + Integer.parseInt(value.substring(4, 6));
      302                  if (value.charAt(0) == '-' && tz > 1200) {
      303                      ErrorReport.reportDdlException(ErrorCode.ERR_UNKNOWN_TIME_ZONE, value);
      304                  } else if (value.charAt(0) == '+' && tz > 1400) {
      305                      ErrorReport.reportDdlException(ErrorCode.ERR_UNKNOWN_TIME_ZONE, value);
      306                  }
      307              }
      308              ZoneId.of(value, timeZoneAliasMap);
      309              return value;
      310          } catch (DateTimeException ex) {
      311              ErrorReport.reportDdlException(ErrorCode.ERR_UNKNOWN_TIME_ZONE, value);
      312          }
      313          throw new DdlException("Parse time zone " + value + " error");
      314      }

      317      public static LocalDateTime formatDateTimeAndFullZero(String datetime, DateTimeFormatter formatter) {
      318          TemporalAccessor temporal = formatter.parse(datetime);
      319          int year = temporal.isSupported(ChronoField.YEAR)
      320                  ? temporal.get(ChronoField.YEAR) : 0;
      321          int month = temporal.isSupported(ChronoField.MONTH_OF_YEAR)
      322                  ? temporal.get(ChronoField.MONTH_OF_YEAR) : 1;
      323          int day = temporal.isSupported(ChronoField.DAY_OF_MONTH)
      324                  ? temporal.get(ChronoField.DAY_OF_MONTH) : 1;
      325          int hour = temporal.isSupported(ChronoField.HOUR_OF_DAY)
      326                  ? temporal.get(ChronoField.HOUR_OF_DAY) : 0;
      327          int minute = temporal.isSupported(ChronoField.MINUTE_OF_HOUR)
      328                  ? temporal.get(ChronoField.MINUTE_OF_HOUR) : 0;
      329          int second = temporal.isSupported(ChronoField.SECOND_OF_MINUTE)
      330                  ? temporal.get(ChronoField.SECOND_OF_MINUTE) : 0;
      331          int milliSecond = temporal.isSupported(ChronoField.MILLI_OF_SECOND)
      332                  ? temporal.get(ChronoField.MILLI_OF_SECOND) : 0;
      333          return LocalDateTime.of(LocalDate.of(year, month, day),
      334                  LocalTime.of(hour, minute, second, milliSecond * 1000000));
      335      }

      341                          Integer.parseInt(parts[4]), Integer.parseInt(parts[5])) : "");

      358      public static long convertStringToDateTimeV2(String dateTimeStr, int scale) {
      359          String format = "yyyy-MM-dd HH:mm:ss";
      360          if (scale > 0) {
      361              format += ".";
      362              for (int i = 0; i < scale; i++) {

      369                  dateTime.getHour(), dateTime.getMinute(), dateTime.getSecond(), dateTime.getNano() / 1000);
