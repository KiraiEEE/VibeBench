      103              }

      146          if (formComponent.action != null) {

      151              ActionInvocation ai = formComponent.getStack().getActionContext().getActionInvocation();
      152              if (ai != null) {
      153                  action = ai.getProxy().getActionName();
      154                  namespace = ai.getProxy().getNamespace();

      166              action = action.substring(0, action.indexOf('?'));

      254              urlComponent.setValue(urlComponent.findString(urlComponent.getValue()));
      255          }

      257          // no explicit url set so attach params from current url, do
      258          // this at start so body params can override any of these they wish.
      259          try {
      260              // ww-1266
      261              String includeParams = (urlComponent.getUrlIncludeParams() != null ? urlComponent.getUrlIncludeParams().toLowerCase() : UrlProvider.GET);
      262  
      263              if (urlComponent.getIncludeParams() != null) {
      264                  includeParams = urlComponent.findString(urlComponent.getIncludeParams());
      265              }
      266  
      267              if (UrlProvider.NONE.equalsIgnoreCase(includeParams)) {
      268                  mergeRequestParameters(urlComponent.getValue(), urlComponent.getParameters(), Collections.emptyMap());
      269              } else if (UrlProvider.ALL.equalsIgnoreCase(includeParams)) {
      270                  mergeRequestParameters(urlComponent.getValue(), urlComponent.getParameters(), urlComponent.getHttpServletRequest().getParameterMap());
      271  
      272                  // for ALL also include GET parameters
      273                  includeGetParameters(urlComponent);
      274                  includeExtraParameters(urlComponent);

      276                  includeGetParameters(urlComponent);
      277                  includeExtraParameters(urlComponent);
      278              } else if (includeParams != null) {
      279                  LOG.warn("Unknown value for includeParams parameter to URL tag: {}", includeParams);
      280              }
      281          } catch (Exception e) {
      282              LOG.warn("Unable to put request parameters ({}) into parameter map.", urlComponent.getHttpServletRequest().getQueryString(), e);
      283          }

      301      private String extractQueryString(UrlProvider urlComponent) {
      302          // Parse the query string to make sure that the parameters come from the query, and not some posted data
      303          String query = urlComponent.getHttpServletRequest().getQueryString();
      304          if (query == null) {
      305              query = (String) urlComponent.getHttpServletRequest().getAttribute("javax.servlet.forward.query_string");
      306          }
      307  
      308          if (query != null) {
      309              // Remove possible #foobar suffix
      310              int idx = query.lastIndexOf('#');
      311  
      312              if (idx != -1) {
      313                  query = query.substring(0, idx);
      314              }
      315          }
      316          return query;
      317      }

      319      /**
      320       * Merge request parameters into current parameters. If a parameter is
      321       * already present, than the request parameter in the current request and value attribute
      322       * will not override its value.
      323       * <p>
      324       * The priority is as follows:-
      325       * <ul>
      326       *  <li>parameter from the current request (least priority)</li>
      327       *  <li>parameter form the value attribute (more priority)</li>
      328       *  <li>parameter from the param tag (most priority)</li>
      329       * </ul>
      330       *
      331       * @param value             the value attribute (URL to be generated by this component)
      332       * @param parameters        component parameters
      333       * @param contextParameters request parameters
      334       * @return {@link QueryStringParser.Result} of value's ?query-string or empty()
      335       */
      336      protected QueryStringParser.Result mergeRequestParameters(String value, Map<String, Object> parameters, Map<String, ?> contextParameters) {
      337          Map<String, Object> mergedParams = new LinkedHashMap<>(contextParameters);
      338          QueryStringParser.Result result = queryStringParser.empty();
      339  
      340          // Merge contextParameters (from current request) with parameters specified in value attribute
      341          // eg. value="someAction.action?id=someId&venue=someVenue"
      342          // where the parameters specified in value attribute takes priority.
      343  
      344          if (StringUtils.contains(value, "?")) {
      345              String queryString = value.substring(value.indexOf('?') + 1);
      346  
      347              result = queryStringParser.parse(queryString);
      348              mergedParams = new LinkedHashMap<>(result.getQueryParams());
      349              for (Map.Entry<String, ?> entry : contextParameters.entrySet()) {
      350                  if (!mergedParams.containsKey(entry.getKey())) {
      351                      mergedParams.put(entry.getKey(), entry.getValue());
      352                  }
      353              }
      354          }
      355  
      356          // Merge parameters specified in value attribute
      357          // eg. value="someAction.action?id=someId&venue=someVenue"
      358          // with parameters specified though param tag
      359          // eg. <param name="id" value="%{'someId'}" />
      360          // where parameters specified through param tag takes priority.
      361  
      362          for (Map.Entry<String, Object> entry : mergedParams.entrySet()) {
      363              if (!parameters.containsKey(entry.getKey())) {
      364                  parameters.put(entry.getKey(), entry.getValue());
      365              }
      366          }
      367  
      368          return result;
      369      }
