      155      /**
      156       * handle Frontend's heartbeat response. Because the replayed journal id is very likely to be
      157       * changed at each heartbeat response, so we simple return true if the heartbeat status is OK.
      158       * But if heartbeat status is BAD, only return true if it is the first time to transfer from
      159       * alive to dead.
      160       */
      161      public boolean handleHbResponse(FrontendHbResponse hbResponse, boolean isReplay) {
      162          boolean isChanged = false;
      163          if (hbResponse.getStatus() == HbStatus.OK) {
      164              if (!isAlive && !isReplay && Config.edit_log_type.equalsIgnoreCase("bdb")) {
      165                  BDBHA bdbha = (BDBHA) Env.getCurrentEnv().getHaProtocol();
      166                  bdbha.removeUnReadyElectableNode(nodeName, Env.getCurrentEnv().getFollowerCount());
      167              }
      168              isAlive = true;
      169              version = hbResponse.getVersion();
      170              queryPort = hbResponse.getQueryPort();
      171              rpcPort = hbResponse.getRpcPort();
      172              arrowFlightSqlPort = hbResponse.getArrowFlightSqlPort();
      173              replayedJournalId = hbResponse.getReplayedJournalId();
      174              lastUpdateTime = hbResponse.getHbTime();
      175              heartbeatErrMsg = "";
      176              lastStartupTime = hbResponse.getProcessUUID();
      177              diskInfos = hbResponse.getDiskInfos();
      178              isChanged = true;
      179              processUUID = hbResponse.getProcessUUID();
      180          } else {
      181              // A non-master node disconnected.
      182              // Set startUUID to zero, and be's heartbeat mgr will ignore this hb,
      183              // so that its cancel worker will not cancel queries from this fe immediately
      184              // until it receives a valid start UUID.
      185              processUUID = 0;
      186              if (isAlive) {
      187                  isAlive = false;
      188                  isChanged = true;
      189              }
      190              heartbeatErrMsg = hbResponse.getMsg() == null ? "Unknown error" : hbResponse.getMsg();
      191          }
      192          return isChanged;
      193      }
