      145    public void run() {
      146  
      147      setStatus(TaskStatus.PROCESSING);
      148      logger.info("Started parsing file " + file);
      149  
      150      try {
      151  
      152        // decision: reject DTDs and external entities in user-supplied XML files.
      153        final SAXParser saxParser = XMLUtils.newSAXParser();
      154        saxParser.parse(file, handler);
      155  
      156        newMZmineFile.getAppliedMethods()
      157            .add(new SimpleFeatureListAppliedMethod(module, parameters, getModuleCallDate()));
      158        project.addFile(newMZmineFile);
      159  
      160      } catch (Throwable e) {
      161        e.printStackTrace();
      162        /* we may already have set the status to CANCELED */
      163        if (getStatus() == TaskStatus.PROCESSING) {
      164          setStatus(TaskStatus.ERROR);
      165          setErrorMessage(ExceptionUtils.exceptionToString(e));
      166        }
      167        return;
      168      }
      169  
      170      if (parsedScans == 0) {
      171        setStatus(TaskStatus.ERROR);
      172        setErrorMessage("No scans found");
      173        return;
      174      }
      175  
      176      logger.info(
      177          "Finished parsing " + file + ", parsed " + parsedScans + " of " + totalScans + " scans");
      178      setStatus(TaskStatus.FINISHED);
      179  
      180    }

      194      @Override
      195      public void startElement(String namespaceURI, String lName, // local
      196          // name
      197          String qName, // qualified name
      198          Attributes attrs) throws SAXException {
      199  
      200        if (isCanceled()) {
      201          throw new SAXException("Parsing Cancelled");
      202        }
      203  
      204        // <spectrumList>
      205        if (qName.equals("spectrumList")) {
      206          String s = attrs.getValue("count");
      207          if (s != null) {
      208            totalScans = Integer.parseInt(s);
      209          }
      210        }
      211  
      212        // <spectrum>
      213        if (qName.equalsIgnoreCase("spectrum")) {
      214          msLevel = 0;
      215          retentionTime = 0f;
      216          // parentScan = -1;
      217          polarity = PolarityType.UNKNOWN;
      218          precursorMz = 0f;
      219          precursorCharge = 0;
      220          scanNumber = Integer.parseInt(attrs.getValue("id"));
      221        }
      222  
      223        // <spectrumInstrument> 1.05 version, <acqInstrument> 1.04 version
      224        if ((qName.equalsIgnoreCase("spectrumInstrument")) || (qName.equalsIgnoreCase(
      225            "acqInstrument"))) {
      226          msLevel = Integer.parseInt(attrs.getValue("msLevel"));
      227          spectrumInstrumentFlag = true;
      228        }
      229  
      230        // <cvParam>
      231        /*
      232         * The terms time.min, time.sec & mz belongs to mzData 1.04 standard.
      233         */
      234        if (qName.equalsIgnoreCase("cvParam")) {
      235          if (spectrumInstrumentFlag) {
      236            if ((attrs.getValue("accession").equals("PSI:1000037")) || (attrs.getValue("name")
      237                .equals("Polarity"))) {
      238              if (attrs.getValue("value").toLowerCase().equals("positive")) {
      239                polarity = PolarityType.POSITIVE;
      240              } else if (attrs.getValue("value").toLowerCase().equals("negative")) {
      241                polarity = PolarityType.NEGATIVE;
      242              } else {
      243                polarity = PolarityType.UNKNOWN;
      244              }
      245            }
      246            if ((attrs.getValue("accession").equals("PSI:1000038")) || (attrs.getValue("name")
      247                .equals("time.min"))) {
      248              retentionTime = (float) Double.parseDouble(attrs.getValue("value"));
      249            }
      250  
      251            if ((attrs.getValue("accession").equals("PSI:1000039")) || (attrs.getValue("name")
      252                .equals("time.sec"))) {
      253              retentionTime = (float) (Double.parseDouble(attrs.getValue("value")) / 60d);
      254            }
      255          }
      256          if (precursorFlag) {
      257            if ((attrs.getValue("accession").equals("PSI:1000040")) || (attrs.getValue("name")
      258                .equals("mz"))) {
      259              precursorMz = Double.parseDouble(attrs.getValue("value"));
      260            }
      261            if (attrs.getValue("accession").equals("PSI:1000041")) {
      262              precursorCharge = Integer.parseInt(attrs.getValue("value"));
      263            }
      264          }
      265        }
      266  
      267        // <mzArrayBinary>
      268        if (qName.equalsIgnoreCase("mzArrayBinary")) {
      269          // clean the current char buffer for the new element
      270          mzArrayBinaryFlag = true;
      271        }
      272  
      273        // <intenArrayBinary>
      274        if (qName.equalsIgnoreCase("intenArrayBinary")) {
      275          // clean the current char buffer for the new element
      276          intenArrayBinaryFlag = true;
      277        }
      278  
      279        // <data>
      280        if (qName.equalsIgnoreCase("data")) {
      281          // clean the current char buffer for the new element
      282          charBuffer.setLength(0);
      283          if (mzArrayBinaryFlag) {
      284            endian = attrs.getValue("endian");
      285            precision = attrs.getValue("precision");
      286            String len = attrs.getValue("length");
      287            if (len != null) {
      288              peaksCount = Integer.parseInt(len);
      289            }
      290          }
      291          if (intenArrayBinaryFlag) {
      292            endian = attrs.getValue("endian");
      293            precision = attrs.getValue("precision");
      294            String len = attrs.getValue("length");
      295            if (len != null) {
      296              peaksCount = Integer.parseInt(len);
      297            }
      298          }
      299        }
      300  
      301        // <precursor>
      302        if (qName.equalsIgnoreCase("precursor")) {
      303          /*
      304           * String parent = attrs.getValue("spectrumRef"); if (parent != null) parentScan =
      305           * Integer.parseInt(parent); else parentScan = -1;
      306           */
      307          precursorFlag = true;
      308        }
      309      }

      316      @Override
      317      public void endElement(String namespaceURI, String sName, // simple name
      318          String qName // qualified name
      319      ) throws SAXException {
      320  
      321        // <spectrumInstrument>
      322        if (qName.equalsIgnoreCase("spectrumInstrument")) {
      323          spectrumInstrumentFlag = false;
      324        }
      325  
      326        // <precursor>
      327        if (qName.equalsIgnoreCase("precursor")) {
      328          precursorFlag = false;
      329        }
      330  
      331        // <spectrum>
      332        if (qName.equalsIgnoreCase("spectrum")) {
      333  
      334          spectrumInstrumentFlag = false;
      335  
      336          // Auto-detect whether this scan is centroided
      337          MassSpectrumType spectrumType = ScanUtils.detectSpectrumType(mzDataPoints,
      338              intensityDataPoints);
      339  
      340          final DDAMsMsInfo info =
      341              msLevel != 1 && Double.compare(precursorMz, 0d) != 0 ? new DDAMsMsInfoImpl(precursorMz,
      342                  precursorCharge, null, null, null, msLevel, ActivationMethod.UNKNOWN, null) : null;
      343  
      344          buildingScan = new SimpleScan(newMZmineFile, scanNumber, msLevel, retentionTime, info,
      345              mzDataPoints, intensityDataPoints, spectrumType, polarity, "", null);
      346  
      347          /*
      348           * Verify the size of parentStack. The actual size of the window to cover possible
      349           * candidates for fragmentScanNumber update is 10 elements.
      350           */
      351          if (parentStack.size() > 10) {
      352            SimpleScan scan = parentStack.removeLast();
      353            newMZmineFile.addScan(scan);
      354            parsedScans++;
      355          }
      356  
      357          parentStack.addFirst(buildingScan);
      358          buildingScan = null;
      359  
      360        }
      361  
      362        // <mzArrayBinary>
      363        if (qName.equalsIgnoreCase("mzArrayBinary")) {
      364  
      365          mzArrayBinaryFlag = false;
      366          mzDataPoints = new double[peaksCount];
      367  
      368          byte[] peakBytes = Base64.getDecoder().decode(charBuffer.toString().trim());
      369  
      370          ByteBuffer currentMzBytes = ByteBuffer.wrap(peakBytes);
      371  
      372          if (endian.equals("big")) {
      373            currentMzBytes = currentMzBytes.order(ByteOrder.BIG_ENDIAN);
      374          } else {
      375            currentMzBytes = currentMzBytes.order(ByteOrder.LITTLE_ENDIAN);
      376          }
      377  
      378          for (int i = 0; i < mzDataPoints.length; i++) {
      379            if (precision == null || precision.equals("32")) {
      380              mzDataPoints[i] = currentMzBytes.getFloat();
      381            } else {
      382              mzDataPoints[i] = currentMzBytes.getDouble();
      383            }
      384          }
      385        }
      386  
      387        // <intenArrayBinary>
      388        if (qName.equalsIgnoreCase("intenArrayBinary")) {
      389  
      390          intenArrayBinaryFlag = false;
      391          intensityDataPoints = new double[peaksCount];
      392  
      393          byte[] peakBytes = Base64.getDecoder().decode(charBuffer.toString().trim());
      394  
      395          ByteBuffer currentIntensityBytes = ByteBuffer.wrap(peakBytes);
      396  
      397          if (endian.equals("big")) {
      398            currentIntensityBytes = currentIntensityBytes.order(ByteOrder.BIG_ENDIAN);
      399          } else {
      400            currentIntensityBytes = currentIntensityBytes.order(ByteOrder.LITTLE_ENDIAN);
      401          }
      402  
      403          for (int i = 0; i < intensityDataPoints.length; i++) {
      404            if (precision == null || precision.equals("32")) {
      405              intensityDataPoints[i] = currentIntensityBytes.getFloat();
      406            } else {
      407              intensityDataPoints[i] = currentIntensityBytes.getDouble();
      408            }
      409          }
      410        }
      411      }

      425        while (!parentStack.isEmpty()) {
