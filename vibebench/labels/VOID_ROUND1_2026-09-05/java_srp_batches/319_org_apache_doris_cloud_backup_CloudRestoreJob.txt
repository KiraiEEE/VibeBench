      191          try {
      192              handleMetaObject(MetaSeriviceOperation.PREPARE);
      193              // send create tablets requests
      194              boolean needSetStorageVault = ((CloudEnv) Env.getCurrentEnv()).getEnableStorageVault();
      195              for (Map.Entry<Pair<OlapTable, Partition>, List<Cloud.CreateTabletsRequest.Builder>> entry
      196                      : tabletRequests.entrySet()) {
      197                  Pair<OlapTable, Partition> tableToPartition = entry.getKey();
      198                  OlapTable table = tableToPartition.first;
      199                  List<Cloud.CreateTabletsRequest.Builder> requestBuilders = entry.getValue();
      200                  for (Cloud.CreateTabletsRequest.Builder requestBuilder : requestBuilders) {
      201                      Cloud.CreateTabletsResponse resp = sendCreateTabletsRequests(requestBuilder, table,
      202                              needSetStorageVault);
      203                      if (resp.hasStorageVaultId()) {
      204                          storageVaultId = resp.getStorageVaultId();
      205                          needSetStorageVault = false;
      206                      }
      207                  }
      208              }
      209              // set storage vault for new restoring table
      210              if (((CloudEnv) Env.getCurrentEnv()).getEnableStorageVault()) {
      211                  for (Table table : restoredTbls) {
      212                      if (table.getType() == TableIf.TableType.OLAP) {
      213                          OlapTable olapTable = (OlapTable) table;
      214                          if (olapTable.getStorageVaultId().isEmpty() && storageVaultId != null) {
      215                              olapTable.setStorageVaultId(storageVaultId);
      216                          }
      217                      }
      218                  }
      219              }
      220          } catch (Exception e) {
      221              status = new Status(Status.ErrCode.COMMON_ERROR, e.getMessage());
