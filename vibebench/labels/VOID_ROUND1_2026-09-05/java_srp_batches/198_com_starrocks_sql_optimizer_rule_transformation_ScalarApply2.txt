      321          private boolean checkReferencedTables() {
      322              List<Table> outerTables = getAllTables(leftOps);
      323              List<Table> subqueryTables = getAllTables(rightOps);
      324  
      325              Set<Long> outerTableIds = outerTables.stream()
      326                      .map(Table::getId)

      329                      .map(Table::getId)
      330                      .collect(Collectors.toSet());
      331              if (outerTables.size() != outerTableIds.size()
      332                      || subqueryTables.size() != subqueryTableIds.size()) {
      333                  return false;
      334              }
      335  
      336              if (outerTables.size() != subqueryTables.size() + 1) {
      337                  return false;

      341                      .map(context.getColumnRefFactory()::getColumn)
      342                      .collect(Collectors.toList());

      344              if (outerTables.size() != 1) {
      345                  return false;
      346              }

      348              correlatedOuterTable = outerTables.get(0);
      349  
      350              for (Column correlatedOuterColumn : correlatedOuterColumns) {
      351                  if (!Objects.equals(correlatedOuterColumn,

      359          /**
      360           * E.g.
      361           * select * from t0, t1
      362           * where t0.v1 = t1.v4
      363           * and t0.v2 < 5
      364           * and t1.v5 > 10
      365           * and t0.v3 < (select max(v6) from t1 where t0.v1 = t1.v4 and t1.v5 >10 )

      367           * Technically, one of the sufficient condition in predicate is that we must guarantee that the subquery and
      368           * the outer block yields the same row set, so all the predicates in the outer block should exist in subquery,
      369           * except those not related to the subquery's table, so here comes the detailed requirements that should be met:
      370           * 1. correlated conjunct must exist in the outer block, e.g. t0.v1 = t1.v4
      371           * 2. conjuncts of outer block except those not related to the subquery must exist in the subquery, for example,
      372           * t0.v2 < 5 is not related to the subquery's table, t1.v5 > 10 is related to the subquery's table
      373           */
      374          private boolean checkPredicate() {
      375              List<ScalarOperator> outerConjuncts = Utils.extractConjuncts(filterOp.getPredicate());
      376  
      377              // First, check correlated predicate exists in outerConjuncts, and remove it if found
      378              // E.g. t0.v1 = t1.v4 in the above case
      379              List<ScalarOperator> subqueryCorrelationConjuncts =
      380                      Utils.extractConjuncts(applyOp.getCorrelationConjuncts());
      381              {
      382                  Iterator<ScalarOperator> outerIt = outerConjuncts.iterator();
      383                  Iterator<ScalarOperator> subIt = subqueryCorrelationConjuncts.iterator();
      384                  while (outerIt.hasNext()) {
      385                      ScalarOperator outerConjunct = outerIt.next();
      386  
      387                      while (subIt.hasNext()) {
      388                          ScalarOperator correlationConjunct = subIt.next();
      389                          if (PredicateComparator.isIdentical(correlationConjunct, outerConjunct,
      390                                  context.getColumnRefFactory())) {
      391                              outerCorrelatedConjuncts.add(outerConjunct);
      392                              outerIt.remove();
      393                              subIt.remove();
      394                              break;
      395                          }
      396                      }
      397                  }
      398              }
      399              if (outerCorrelatedConjuncts.isEmpty() || !subqueryCorrelationConjuncts.isEmpty()) {
      400                  return false;
      401              }
      402  
      403              // Second, remove the conjunct that contains the subquery
      404              // E.g. t0.v3 < (<subquery>) in the above case
      405              {
      406                  Iterator<ScalarOperator> outerIt = outerConjuncts.iterator();
      407                  while (outerIt.hasNext()) {
      408                      ScalarOperator outerConjunct = outerIt.next();
      409                      if (Utils.collect(outerConjunct, ColumnRefOperator.class).stream()
      410                              .anyMatch(columnRefOperator -> Objects.equals(applyOp.getOutput(), columnRefOperator))) {
      411                          outerSubqueryConjunct = outerConjunct;
      412                          outerIt.remove();
      413                          break;
      414                      }
      415                  }
      416              }
      417              if (outerSubqueryConjunct == null) {
      418                  return false;
      419              }
      420  
      421              // Third, remove all the conjuncts which only related to the correlated outer table from the outerConjuncts
      422              // E.g. t0.v2 < 5 in the above case
      423              {
      424                  Iterator<ScalarOperator> outerIt = outerConjuncts.iterator();
      425                  while (outerIt.hasNext()) {
      426                      ScalarOperator outerConjunct = outerIt.next();
      427                      List<ColumnRefOperator> colRefs = Utils.collect(outerConjunct, ColumnRefOperator.class);
      428                      boolean relatedToOtherTable = false;
      429                      for (ColumnRefOperator colRef : colRefs) {
      430                          Column column = context.getColumnRefFactory().getColumn(colRef);
      431                          if (column == null) {
      432                              continue;
      433                          }
      434                          if (!Objects.equals(column, correlatedOuterTable.getColumn(column.getName()))) {
      435                              relatedToOtherTable = true;
      436                              break;
      437                          }
      438                      }
      439                      if (!relatedToOtherTable) {
      440                          outerIt.remove();
      441                      }
      442                  }
      443              }
      444  
      445              // We assume that the right subtree only contains one FilterOperator, for other complex cases, we may
      446              // support later
      447              if (rightOps.stream().filter(LogicalFilterOperator.class::isInstance).count() > 1) {
      448                  return false;
      449              }
      450  
      451              // Forth, check all the conjuncts from the remaining outerConjuncts and subConjuncts are exactly the same
      452              // E.g. t1.v5 >10 in the above case
      453              LogicalFilterOperator subFilterOp = rightOps.stream()
      454                      .filter(LogicalFilterOperator.class::isInstance)
      455                      .map(Operator::<LogicalFilterOperator>cast)
      456                      .findFirst()
      457                      .orElse(null);
      458              if (subFilterOp == null) {
      459                  return outerConjuncts.isEmpty();
      460              }
      461              List<ScalarOperator> subConjuncts = Utils.extractConjuncts(subFilterOp.getPredicate());
      462              if (outerConjuncts.size() != subConjuncts.size()) {
      463                  return false;
      464              }
      465              {
      466                  Iterator<ScalarOperator> outerIt = outerConjuncts.iterator();
      467                  while (outerIt.hasNext()) {
      468                      ScalarOperator outerConjunct = outerIt.next();
      469                      Iterator<ScalarOperator> subIt = subConjuncts.iterator();
      470                      boolean found = false;
      471                      while (subIt.hasNext()) {
      472                          ScalarOperator subConjunct = subIt.next();
      473                          if (PredicateComparator.isIdentical(outerConjunct, subConjunct,
      474                                  context.getColumnRefFactory())) {
      475                              found = true;
      476                              outerIt.remove();
      477                              subIt.remove();
      478                              break;
      479                          }
      480                      }
      481                      if (!found) {
      482                          return false;
      483                      }
      484                  }
      485              }
      486              Preconditions.checkState(outerConjuncts.isEmpty());
      487              Preconditions.checkState(subConjuncts.isEmpty());
      488  
      489              return true;
      490          }

      492          private void createWindowOp() {
      493              outerColumnRefOperators = getAllColumnRefOperators(leftOps);
      494  
      495              Map<ColumnRefOperator, CallOperator> windows = Maps.newHashMap();
      496              Map<ColumnRefOperator, CallOperator> aggregations = subAggOp.getAggregations();
      497  
      498              for (Map.Entry<ColumnRefOperator, CallOperator> entry : aggregations.entrySet()) {
      499                  ColumnRefOperator columnRefOperator = entry.getKey();
      500                  CallOperator callOp = entry.getValue();
      501  
      502                  CallOperator newCallOp = cloneCallOperator(callOp);
      503                  ColumnRefOperator newColumnRefOperator = context.getColumnRefFactory()
      504                          .create(newCallOp, newCallOp.getType(), newCallOp.isNullable());
      505                  windows.put(newColumnRefOperator, newCallOp);
      506                  peerColumnRefMap.put(columnRefOperator, newColumnRefOperator);
      507              }
      508  
      509              List<ScalarOperator> partitionByColumns = Lists.newArrayList();
      510              for (ScalarOperator outerCorrelatedConjunct : outerCorrelatedConjuncts) {
      511                  Preconditions.checkState(outerCorrelatedConjunct instanceof BinaryPredicateOperator);
      512                  // We can take either first or second child of outerCorrelatedConjunct as the partition by expr
      513                  partitionByColumns.add(outerCorrelatedConjunct.getChild(0));
      514              }
      515  
      516              windowOp = new LogicalWindowOperator.Builder()
      517                      .setWindowCall(windows)
      518                      .setPartitionExpressions(partitionByColumns)
      519                      .setEnforceSortColumns(
      520                              partitionByColumns.stream()
      521                                      .map(ScalarOperator::<ColumnRefOperator>cast)
      522                                      .map(columnRefOperator -> new Ordering(columnRefOperator, true, true))
      523                                      .collect(Collectors.toList()))
      524                      .build();
      525          }
