      339  	private static void appendSignature(StringBuilder signature, JsonNode node, SignatureAspect aspect) {
      340  		if (node == null || node.isNull()) {
      341  			signature.append("null");
      342  			return;
      343  		}
      344  
      345  		String type = readText(node, "type");
      346  		signature.append('(').append(type);
      347  		switch (aspect) {
      348  		case JOIN_ALGORITHMS:
      349  			if (isJoinType(type)) {
      350  				signature.append("|algorithm=").append(readText(node, "algorithm"));
      351  			}
      352  			break;
      353  		case ACTUAL_RESULT_SIZES:
      354  			signature.append("|resultSizeActual=").append(readNumberToken(node, "resultSizeActual"));
      355  			break;
      356  		case ESTIMATES:
      357  			signature.append("|costEstimate=").append(readNumberToken(node, "costEstimate"));
      358  			signature.append("|resultSizeEstimate=").append(readNumberToken(node, "resultSizeEstimate"));
      359  			break;
      360  		case STRUCTURE:
      361  			break;
      362  		default:
      363  			throw new IllegalStateException("Unhandled signature aspect: " + aspect);
      364  		}
      365  
      366  		for (JsonNode child : readChildren(node)) {
      367  			appendSignature(signature, child, aspect);
      368  		}
      369  		signature.append(')');
      370  	}
