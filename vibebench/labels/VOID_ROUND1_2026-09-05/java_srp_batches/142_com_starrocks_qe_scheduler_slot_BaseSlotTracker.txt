      153       * Add a slot requirement.
      154       * @param slot The required slot.
      155       * @return True if the slot is required successfully or already required , false if the query queue is full.
      156       */
      157      public boolean requireSlot(LogicalSlot slot) {
      158          if (!isResourceCapacityEnough(slot)) {
      159              return false;
      160          }
      161  
      162          if (slots.containsKey(slot.getSlotId())) {
      163              return true;
      164          }
      165  
      166          slots.put(slot.getSlotId(), slot);
      167          slotsOrderByExpiredTime.add(slot);
      168          pendingSlots.put(slot.getSlotId(), slot);
      169  
      170          MetricRepo.COUNTER_QUERY_QUEUE_SLOT_PENDING.increase((long) slot.getNumPhysicalSlots());
      171  
      172          listeners.forEach(listener -> listener.onRequireSlot(slot));
      173          slot.onRequire();
      174  
      175          // try to register the slot to the connected context
      176          tryRegisterConnectContext(slot);
      177  
      178          return true;
      179      }

      182       * Allocate a slot which has already been required.

      186      public void allocateSlot(LogicalSlot slot) {
      187          TUniqueId slotId = slot.getSlotId();
      188          if (!slots.containsKey(slotId)) {
      189              return;
      190          }
      191  
      192          if (pendingSlots.remove(slotId) == null) {
      193              return;
      194          }
      195          MetricRepo.COUNTER_QUERY_QUEUE_SLOT_PENDING.increase((long) -slot.getNumPhysicalSlots());
      196  
      197          if (allocatedSlots.put(slotId, slot) != null) {
      198              return;
      199          }
      200  
      201          MetricRepo.COUNTER_QUERY_QUEUE_SLOT_RUNNING.increase((long) slot.getNumPhysicalSlots());
      202          numAllocatedSlots += slot.getNumPhysicalSlots();
      203  
      204          listeners.forEach(listener -> listener.onAllocateSlot(slot));
      205          slot.onAllocate();

      221      protected void tryAddIntoHistorySlots(LogicalSlot slot) {
      222          if (Config.max_query_queue_history_slots_number <= 0) {
      223              return;
      224          }
      225          while (historySlots.size() > Config.max_query_queue_history_slots_number) {
      226              historySlots.poll();
      227          }
      228          historySlots.add(slot);
      229      }

      231      /**
      232       * Try to register the slot to the connected context for more observing information.
      233       * @param slot: The slot to be registered.
      234       */
      235      protected void tryRegisterConnectContext(LogicalSlot slot) {
      236          // ignore the slot is null or slot's extra message has been set
      237          if (Config.max_query_queue_history_slots_number <= 0 ||
      238                  slot == null || (slot.getExtraMessage() != null && slot.getExtraMessage().isPresent())) {
      239              return;
      240          }
      241          // find the connected context and register the logical slot
      242          try {
      243              TUniqueId slotId = slot.getSlotId();
      244              UUID queryId = UUIDUtil.fromTUniqueid(slotId);
      245              ConnectContext ctx = ExecuteEnv.getInstance().getScheduler().findContextByQueryId(queryId.toString());
      246              if (ctx == null) {
      247                  LOG.debug("Failed to find the context for queryId: {}", queryId);
      248                  return;
      249              }
      250              LOG.debug("Registering the slot {} to context {}", slot, ctx);
      251          } catch (Exception e) {
      252              LOG.warn("Failed to register the slot to context", e);
      253          }
      254      }

      256      protected LogicalSlot releaseSlotImpl(TUniqueId slotId) {
      257          LogicalSlot slot = slots.remove(slotId);
      258          if (slot == null) {
      259              return null;
      260          }
      261  
      262          slotsOrderByExpiredTime.remove(slot);
      263  
      264          if (allocatedSlots.remove(slotId) != null) {
      265              numAllocatedSlots -= slot.getNumPhysicalSlots();
      266              MetricRepo.COUNTER_QUERY_QUEUE_SLOT_RUNNING.increase((long) -slot.getNumPhysicalSlots());
      267          } else {
      268              if (pendingSlots.remove(slotId) != null) {
      269                  MetricRepo.COUNTER_QUERY_QUEUE_SLOT_PENDING.increase((long) -slot.getNumPhysicalSlots());
      270              }
      271          }
      272  
      273          listeners.forEach(listener -> listener.onReleaseSlot(slot));
      274          slot.onRelease();
      275  
      276          return slot;
      277      }

      287       * Peak all the expired slots.
      288       *
      289       * <p> Note that this method does not remove the expired slots from the tracker,
      290       * and {@link #releaseSlot} should be called to release these slots after peaking.
      291       * @return The expired slots.

      293      public List<LogicalSlot> peakExpiredSlots() {
      294          final long nowMs = System.currentTimeMillis();
      295          List<LogicalSlot> expiredSlots = new ArrayList<>();
      296          for (LogicalSlot slot : slotsOrderByExpiredTime) {
      297              if (!slot.isAllocatedExpired(nowMs)) {
      298                  break;
      299              }
      300              expiredSlots.add(slot);
      301          }
      302          return expiredSlots;
      303      }

      320      public Collection<LogicalSlot> getSlots() {
      321          if (Config.max_query_queue_history_slots_number > 0) {
      322              List<LogicalSlot> result = Lists.newArrayList();
      323              // the newest slots are in the front
      324              result.addAll(slots.values());
      325              List<LogicalSlot> histories = Lists.newArrayList(historySlots);
      326              if (!histories.isEmpty()) {
      327                  Collections.reverse(histories);
      328                  result.addAll(histories);
