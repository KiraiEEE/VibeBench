      173      private ThemeFile createThemeFor(@NonNull final ThemeData theme) {
      174          final String[] themeIdTokens = theme.id.split(ZIP_THEME_SEPARATOR);
      175          final boolean isZipTheme = themeIdTokens.length == 2;
      176  
      177          if (theme.fileInfo == null || theme.fileInfo.uri == null || theme.containingFolder == null) {
      178              return null;
      179          }
      180  
      181          ThemeFile xmlRenderTheme = null;
      182          try {
      183              if (!isZipTheme) {
      184                  if (UriUtils.isFileUri(theme.fileInfo.uri)) {
      185                      xmlRenderTheme = new ExternalRenderTheme(theme.fileInfo.uri.toString(), this);
      186                  } else {
      187                      //this is the SLOW THEME path. Show OneTimeDialog to warn user about this
      188                      Dialogs.basicOneTimeMessage(CgeoApplication.getInstance(), OneTimeDialogs.DialogType.MAP_THEME_FIX_SLOWNESS);
      189                      xmlRenderTheme = new ContentRenderTheme(getContentResolver(), theme.fileInfo.uri, this);
      190                      xmlRenderTheme.setResourceProvider(new ContentResolverResourceProvider(getContentResolver(), ContentStorage.get().getUriForFolder(theme.containingFolder), true));
      191                  }
      192              } else {
      193                  //always cache the last used ZipResourceProvider. Check if current one can be reused, if not then reload
      194                  synchronized (cachedZipMutex) {
      195                      if (cachedZipProvider == null || !themeIdTokens[0].equals(cachedZipProviderFilename)) {
      196                          if (theme.fileInfo.size > ZIP_FILE_SIZE_LIMIT) {
      197                              cachedZipProvider = null;
      198                              cachedZipProviderFilename = null;
      199                          } else {
      200                              cachedZipProvider = new ZipXmlThemeResourceProvider(new ZipInputStream(ContentStorage.get().openForRead(theme.fileInfo.uri)), ZIP_RESOURCE_READ_LIMIT);
      201                              cachedZipProviderFilename = themeIdTokens[0];
      202                          }
      203                      }
      204                      xmlRenderTheme = cachedZipProvider == null ? null : new ZipRenderTheme(themeIdTokens[1], cachedZipProvider, this);
      205                  }
      206              }
      207          } catch (Exception ex) {
      208              Log.w("Problem loading Theme [" + theme.id + "]'" + theme.fileInfo.uri + "'", ex);
      209              xmlRenderTheme = null;
      210              cachedZipProvider = null;
      211              cachedZipProviderFilename = null;
      212          }
      213          return xmlRenderTheme;
      214      }

      217          if (!tileProvider.supportsThemes()) {

      240              reapplyMapTheme(map, tileProvider);

      242          });

      292       * Set a new map theme. The theme is evaluated against available themes and possibly corrected.
      293       * Next time a map viewer is opened, the theme will be evaluated and used if possible
      294       */
      295      private static ThemeData setSelectedMapThemeInternal(final String themeIdCandidate) {
      296          //try to apply stored value
      297          ThemeData selectedTheme = null;
      298          final List<ThemeData> avThemes = getAvailableThemes();
      299          //search for exact match first
      300          for (ThemeData avTheme : avThemes) {
      301              if (avTheme.id.equals(themeIdCandidate)) {
      302                  selectedTheme = avTheme;
      303                  break;
      304              }
      305          }
      306  
      307          //if no exact match found, check special cases
      308          if (selectedTheme == null) {
      309              for (ThemeData avTheme : avThemes) {
      310                  final String avThemeId = avTheme.id;
      311  
      312                  //might be a legacy value. Try to find a matching theme ending with stored value
      313                  if (themeIdCandidate.endsWith("/" + avThemeId)) {
      314                      selectedTheme = avTheme;
      315                      break;
      316                  }
      317                  //might be an incomplete ZIP value (only Zip name without a selected inside theme). Then use first found value
      318                  if (avThemeId.startsWith(themeIdCandidate + ZIP_THEME_SEPARATOR)) {
      319                      selectedTheme = avTheme;
      320                      break;
      321                  }
      322              }
      323          }
      324          setSelectedTheme(selectedTheme);
      325          return selectedTheme;
      326      }

      335      private static void recalculateAvailableThemes() {
      336          final List<ThemeData> newAvailableThemes = new ArrayList<>();
      337          addAvailableThemes(isThemeSynchronizationActive() ? Folder.fromFile(MAP_THEMES_INTERNAL_FOLDER) : MAP_THEMES_FOLDER.getFolder(), newAvailableThemes, "", 0);
      338  
      339          Collections.sort(newAvailableThemes, (t1, t2) -> TextUtils.COLLATOR.compare(t1.userDisplayableName, t2.userDisplayableName));
      340  
      341          synchronized (availableThemesMutex) {
      342              availableThemes.clear();
      343              availableThemes.addAll(newAvailableThemes);
      344              availableThemesInitialized = true;
      345          }

      347          synchronized (cachedZipMutex) {
      348              cachedZipProvider = null;
      349              cachedZipProviderFilename = null;
      350          }
      351      }

      353      private static List<ThemeData> getAvailableThemes() {
      354          synchronized (availableThemesMutex) {
      355              if (!availableThemesInitialized) {

      364      private static void addAvailableThemes(@NonNull final Folder dir, final List<ThemeData> themes, final String prefix, final int level) {
      365          for (FileInformation candidate : Objects.requireNonNull(ContentStorage.get().list(dir))) {
      366              if (candidate.isDirectory && (AVAILABLE_THEMES_SCAN_MAXDEPTH < 0 || level < AVAILABLE_THEMES_SCAN_MAXDEPTH)) {
      367                  addAvailableThemes(candidate.dirLocation, themes, prefix + candidate.name + "/", level + 1);
      368              } else if (candidate.name.endsWith(".xml")) {
      369                  final String themeId = prefix + candidate.name;
      370                  themes.add(new ThemeData(themeId, toUserDisplayableName(candidate, null), candidate, dir));
      371              } else if (candidate.name.endsWith(".zip") && candidate.size <= ZIP_FILE_SIZE_LIMIT) {
      372                  try (InputStream is = ContentStorage.get().openForRead(candidate.uri)) {
      373                      for (String zipXmlTheme : ZipXmlThemeResourceProvider.scanXmlThemes(new ZipInputStream(is))) {
      374                          final String themeId = prefix + candidate.name + ZIP_THEME_SEPARATOR + zipXmlTheme;
      375                          themes.add(new ThemeData(themeId, toUserDisplayableName(candidate, zipXmlTheme), candidate, dir));
      376                      }
      377                  } catch (IOException ioe) {
      378                      Log.w("Map Theme ZIP '" + candidate + "' could not be read", ioe);
      379                  } catch (Exception e) {
      380                      Log.w("Problem opening map Theme ZIP '" + candidate + "'", e);
      381                  }
      382              }
      383          }
      384      }

      387       * Calculates a xml theme name fit for user display (in dropdown etc)
      388       *
      389       * @param file    theme file name
      390       * @param zipPath if theme file is a ZIP, then this contains the zip-internal path to the xml. Otherwise null

      393      private static String toUserDisplayableName(final FileInformation file, final String zipPath) {
      394          String userDisplay = StringUtils.removeEnd(file.name, ".xml");
      395          if (zipPath != null) {
      396              final int idx = zipPath.lastIndexOf("/");
      397              userDisplay = userDisplay + "/" + StringUtils.removeEnd(idx < 0 ? zipPath : zipPath.substring(idx + 1), ".xml");
      398          }
      399          return userDisplay;
