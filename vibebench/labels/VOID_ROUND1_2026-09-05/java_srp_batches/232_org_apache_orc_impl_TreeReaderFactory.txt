      245      public TypeReader.ReaderCategory getReaderCategory(int columnId) {
      246        TypeReader.ReaderCategory result;
      247        if (getColumnFilterIds().contains(columnId)) {
      248          // parent filter columns that might include non-filter children. These are classified as
      249          // FILTER_PARENT. This is used during the reposition for non-filter read. Only Struct and
      250          // Union Readers are supported currently
      251          TypeDescription col = columnId == -1 ? null : getSchemaEvolution()
      252              .getFileSchema()
      253              .findSubtype(columnId);
      254          if (col == null || col.getChildren() == null || col.getChildren().isEmpty()) {
      255            result = TypeReader.ReaderCategory.FILTER_CHILD;
      256          } else {
      257            result = TypeReader.ReaderCategory.FILTER_PARENT;
      258          }
      259        } else {
      260          result = TypeReader.ReaderCategory.NON_FILTER;
      261        }
      262        return result;
      263      }

      390      /**
      391       * Populates the isNull vector array in the previousVector object based on
      392       * the present stream values. This function is called from all the child
      393       * readers, and they all set the values based on isNull field value.
      394       *
      395       * @param previous The columnVector object whose isNull value is populated
      396       * @param isNull Whether the each value was null at a higher level. If
      397       *               isNull is null, all values are non-null.
      398       * @param batchSize      Size of the column vector
      399       * @param filterContext the information about the rows that were selected
      400       *                      by the filter.

      404      public void nextVector(ColumnVector previous,
      405                             boolean[] isNull,
      406                             final int batchSize,
      407                             FilterContext filterContext,
      408                             ReadPhase readPhase) throws IOException {
      409        if (present != null || isNull != null) {
      410          // Set noNulls and isNull vector of the ColumnVector based on
      411          // present stream
      412          previous.noNulls = true;
      413          boolean allNull = true;
      414          for (int i = 0; i < batchSize; i++) {
      415            if (isNull == null || !isNull[i]) {
      416              if (present != null && present.next() != 1) {
      417                previous.noNulls = false;
      418                previous.isNull[i] = true;
      419              } else {
      420                previous.isNull[i] = false;
      421                allNull = false;
      422              }
      423            } else {
      424              previous.noNulls = false;
      425              previous.isNull[i] = true;
      426            }
      427          }
      428          previous.isRepeating = !previous.noNulls && allNull;
      429        } else {
      430          // There is no present stream, this means that all the values are
      431          // present.
      432          previous.noNulls = true;
      433          for (int i = 0; i < batchSize; i++) {
      434            previous.isNull[i] = false;
      435          }
      436        }
      437      }
