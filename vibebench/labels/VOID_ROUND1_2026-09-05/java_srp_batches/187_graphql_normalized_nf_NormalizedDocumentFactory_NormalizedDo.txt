      278  
      279                  assumedSkipIncludeVariableValues = null;

      285                  } else {
      286                      int combinations = (int) Math.pow(2, skipIncludeVariableNames.size());
      287                      for (int i = 0; i < combinations; i++) {
      288                          assumedSkipIncludeVariableValues = new LinkedHashMap<>();
      289                          int variableIndex = 0;

      292                          }
      293                          NormalizedOperation operationWithAssumedVariables = createNormalizedOperation(operationDefinition);
      294                          normalizedOperations.add(new NormalizedDocument.NormalizedOperationWithAssumedSkipIncludeVariables(assumedSkipIncludeVariableValues, operationWithAssumedVariables));
      295                      }
      296                  }
      297              }
      298  
      299              return new NormalizedDocument(
      300                      normalizedOperations
      301              );

      343          private void buildNormalizedFieldsRecursively(@Nullable NormalizedField normalizedField,
      344                                                        @Nullable OperationDefinition operationDefinition,
      345                                                        @Nullable ImmutableList<CollectedField> fieldAndAstParents,
      346                                                        int curLevel) {
      347              if (this.maxDepthSeen < curLevel) {
      348                  this.maxDepthSeen = curLevel;
      349                  checkMaxDepthExceeded(curLevel);
      350              }
      351              Set<GraphQLObjectType> possibleObjects;
      352              List<CollectedField> collectedFields;
      353  
      354              // special handling for the root selection Set
      355              if (normalizedField == null) {
      356                  GraphQLObjectType rootType = SchemaUtil.getOperationRootType(graphQLSchema, Assert.assertNotNull(operationDefinition));
      357                  possibleObjects = ImmutableSet.of(rootType);
      358                  collectedFields = new ArrayList<>();
      359                  collectFromSelectionSet(operationDefinition.getSelectionSet(), collectedFields, rootType, possibleObjects);
      360              } else {
      361                  List<GraphQLFieldDefinition> fieldDefs = normalizedField.getFieldDefinitions(graphQLSchema);
      362                  possibleObjects = resolvePossibleObjects(fieldDefs);
      363                  if (possibleObjects.isEmpty()) {
      364                      return;
      365                  }
      366                  collectedFields = new ArrayList<>();
      367                  for (CollectedField fieldAndAstParent : Assert.assertNotNull(fieldAndAstParents)) {
      368                      if (fieldAndAstParent.field.getSelectionSet() == null) {
      369                          continue;
      370                      }
      371                      // the AST parent comes from the previous collect from selection set call
      372                      // and is the type to which the field belongs (the container type of the field) and output type
      373                      // of the field needs to be determined based on the field name
      374                      GraphQLFieldDefinition fieldDefinition = Introspection.getFieldDef(graphQLSchema, fieldAndAstParent.astTypeCondition, fieldAndAstParent.field.getName());
      375                      // it must a composite type, because the field has a selection set
      376                      GraphQLCompositeType selectionSetType = (GraphQLCompositeType) unwrapAll(fieldDefinition.getType());
      377                      this.collectFromSelectionSet(fieldAndAstParent.field.getSelectionSet(),
      378                              collectedFields,
      379                              selectionSetType,
      380                              possibleObjects
      381                      );
      382                  }
      383              }
      384  
      385              Map<String, List<CollectedField>> fieldsByName = fieldsByResultKey(collectedFields);
      386              ImmutableList.Builder<NormalizedField> resultNFs = ImmutableList.builder();
      387              ImmutableListMultimap.Builder<NormalizedField, CollectedField> normalizedFieldToAstFields = ImmutableListMultimap.builder();
      388              createNFs(resultNFs, fieldsByName, normalizedFieldToAstFields, curLevel + 1, normalizedField);
      389  
      390              ImmutableList<NormalizedField> nextLevelChildren = resultNFs.build();
      391              ImmutableListMultimap<NormalizedField, CollectedField> nextLevelNormalizedFieldToAstFields = normalizedFieldToAstFields.build();
      392  
      393              for (NormalizedField childENF : nextLevelChildren) {
      394                  if (normalizedField == null) {
      395                      // all root ENFs don't have a parent, but are collected in the rootEnfs list
      396                      rootEnfs.add(childENF);
      397                  } else {
      398                      normalizedField.addChild(childENF);
      399                  }
      400                  ImmutableList<CollectedField> childFieldAndAstParents = nextLevelNormalizedFieldToAstFields.get(childENF);
      401  
      402                  MergedField mergedField = newMergedField(childFieldAndAstParents);
      403                  captureMergedField(childENF, mergedField);
      404  
      405                  updateFieldToNFMap(childENF, childFieldAndAstParents);
      406                  updateCoordinatedToNFMap(childENF);
      407  
      408                  // recursive call
      409                  buildNormalizedFieldsRecursively(childENF,
      410                          null,
      411                          childFieldAndAstParents,
      412                          curLevel + 1);
      413              }
      414          }

      476          private @Nullable NormalizedField createNF(CollectedFieldGroup collectedFieldGroup,
      477                                           int level,
      478                                           @Nullable NormalizedField parent) {
      479  
      480              this.fieldCount++;
      481              if (this.fieldCount > this.options.getMaxFieldsCount()) {

      483              }

      485              Set<GraphQLObjectType> objectTypes = collectedFieldGroup.objectTypes;
      486              field = collectedFieldGroup.fields.iterator().next().field;
      487              List<Directive> directives = collectedFieldGroup.fields.stream().flatMap(f -> f.field.getDirectives().stream()).collect(Collectors.toList());
      488              String fieldName = field.getName();
      489              ImmutableList<String> objectTypeNames = map(objectTypes, GraphQLObjectType::getName);
      490              return NormalizedField.newNormalizedField()
      491                      .alias(field.getAlias())
      492                      .astArguments(field.getArguments())
      493                      .astDirectives(directives)
      494                      .objectTypeNames(objectTypeNames)

      502          private List<CollectedFieldGroup> groupByCommonParents(Collection<CollectedField> fields) {
      503              ImmutableSet.Builder<GraphQLObjectType> objectTypes = ImmutableSet.builder();
      504              for (CollectedField collectedField : fields) {
      505                  objectTypes.addAll(collectedField.objectTypes);
      506              }
      507              Set<GraphQLObjectType> allRelevantObjects = objectTypes.build();
      508              Map<GraphQLType, ImmutableList<CollectedField>> groupByAstParent = groupingBy(fields, fieldAndType -> fieldAndType.astTypeCondition);
      509              if (groupByAstParent.size() == 1) {
      510                  return singletonList(new CollectedFieldGroup(ImmutableSet.copyOf(fields), allRelevantObjects));
      511              }
      512              ImmutableList.Builder<CollectedFieldGroup> result = ImmutableList.builder();
      513              for (GraphQLObjectType objectType : allRelevantObjects) {
      514                  Set<CollectedField> relevantFields = filterSet(fields, field -> field.objectTypes.contains(objectType));
      515                  result.add(new CollectedFieldGroup(relevantFields, singleton(objectType)));
      516              }
      517              return result.build();
      518          }

      581          private void collectField(List<CollectedField> result,
      582                                    Field field,
      583                                    Set<GraphQLObjectType> possibleObjectTypes,
      584                                    GraphQLCompositeType astTypeCondition
      585          ) {
      586              Boolean shouldInclude;
      587              if (assumedSkipIncludeVariableValues == null) {
      588                  if ((shouldInclude = conditionalNodes.shouldIncludeWithoutVariables(field)) == null) {
      589  
      590                      String skipVariableName = conditionalNodes.getSkipVariableName(field);
      591                      String includeVariableName = conditionalNodes.getIncludeVariableName(field);
      592                      if (skipVariableName != null) {
      593                          skipIncludeVariableNames.add(skipVariableName);
      594                      }
      595                      if (includeVariableName != null) {
      596                          skipIncludeVariableNames.add(includeVariableName);
      597                      }
      598                  }
      599                  if (shouldInclude != null && !shouldInclude) {
      600                      return;
      601                  }
      602              } else {
      603                  if (!conditionalNodes.shouldInclude(field, (Map) assumedSkipIncludeVariableValues, graphQLSchema, null)) {
      604                      return;
      605                  }
      606              }
      607              // this means there is actually no possible type for this field, and we are done
      608              if (possibleObjectTypes.isEmpty()) {
      609                  return;
      610              }
      611              result.add(new CollectedField(field, possibleObjectTypes, astTypeCondition));
      612          }
