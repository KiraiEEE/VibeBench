      191                  int totalSize = 0;
      192                  long totalMemSize = 0L;
      193                  long startTime = System.currentTimeMillis();

      201                      }

      204                          if (totalSize >= MIN_COMMIT_EVENT_SIZE || totalMemSize >= MIN_COMMIT_MEM_SIZE) {
      205                              break;
      206                          }
      207                          try {
      208                              Thread.sleep(sleepTimeMs);
      209                          } catch (InterruptedException e) {
      210                              // do nothing
      211                          }
      212                      } else {
      213                          if (debug) {
      214                              // print summary of the batch
      215                              CanalUtils.printSummary(dataEvents);
      216                          }
      217                          List<CanalEntry.Entry> entries = dataEvents.getDatas();
      218                          int size = entries.size();
      219                          ackBatches.add(dataEvents.getId());
      220                          positionMeta.addBatch(dataEvents.getId(), dataEvents.getPositionRange());
      221                          executeOneBatch(dataEvents);
      222                          totalSize += size;
      223                          totalMemSize += dataEvents.getMemSize();
      224                          // size of bytes received so far is larger than max commit memory size.
      225                          if (totalMemSize >= MAX_COMMIT_MEM_SIZE) {
      226                              break;
      227                          }
      228                      }
      229  
      230                      if (System.currentTimeMillis() - startTime >= commitIntervalSecond * 1000) {
      231                          break;
      232                      }
      233                  }
      234  
      235                  // wait all channels done
      236                  Status st = waitForTxn();
      237                  if (!running) {
      238                      abortForTxn("stopping client");
      239                      continue;
      240                  }
      241                  if (st.ok()) {
      242                      commitForTxn();
      243                  } else {
      244                      abortForTxn(st.getErrorMsg());
      245                      syncJob.cancel(SyncFailMsg.MsgType.RUN_FAIL, st.getErrorMsg());
      246                  }
      247              } catch (Exception e) {
      248                  logger.error("Executor is error!", e);
      249                  abortForTxn(e.getMessage());
      250                  syncJob.cancel(SyncFailMsg.MsgType.SUBMIT_FAIL, e.getMessage());
      251              } finally {
      252                  cleanForTxn();
      253              }

      257      public void put(Message message, int size) {
      258          List<CanalEntry.Entry> entries;
      259          if (message.isRaw()) {
      260              entries = new ArrayList<>(message.getRawEntries().size());
      261              for (ByteString rawEntry : message.getRawEntries()) {
      262                  try {
      263                      entries.add(CanalEntry.Entry.parseFrom(rawEntry));
      264                  } catch (InvalidProtocolBufferException e) {
      265                      throw new CanalException(e);
      266                  }
      267              }
      268          } else {
      269              entries = message.getEntries();
      270          }
      271  
      272          int startIndex = 0;
      273          // if last ack position is null, it is the first time to consume batch
      274          EntryPosition lastAckPosition = positionMeta.getAckPosition();
      275          if (lastAckPosition != null) {
      276              EntryPosition firstPosition = EntryPosition.createPosition(entries.get(0));
      277              // only get data after the last ack position
      278              if (EntryPosition.min(firstPosition, lastAckPosition).equals(firstPosition)) {
      279                  for (int i = 0; i <= entries.size() - 1; i++) {
      280                      startIndex++;
      281                      if (EntryPosition.checkPosition(entries.get(i), lastAckPosition)) {
      282                          break;
      283                      }
      284                  }
      285              }
      286          }
      287  
      288          if (startIndex <= size - 1) {
      289              Events<CanalEntry.Entry, EntryPosition> dataEvents = new Events<>(message.getId());
      290              PositionRange<EntryPosition> range = new PositionRange<>();
      291              dataEvents.setPositionRange(range);
      292              range.setStart(EntryPosition.createPosition(entries.get(startIndex)));
      293              range.setEnd(EntryPosition.createPosition(entries.get(size - 1)));
      294              dataEvents.setDatas(entries);
      295              long memsize = 0L;
      296              for (CanalEntry.Entry entry : entries) {
      297                  memsize += entry.getHeader().getEventLength();
      298              }
      299              dataEvents.setMemSize(memsize);
      300              try {
      301                  dataBlockingQueue.put(dataEvents);
      302              } catch (InterruptedException e) {
      303                  logger.error("put message to executor error:", e);
      304                  throw new CanalException(e);
      305              }

      309      private void executeOneBatch(Events<CanalEntry.Entry, EntryPosition> dataEvents) throws UserException {
      310          final long batchId = dataEvents.getId();
      311          Map<String, CanalSyncChannel> preferChannels = Maps.newHashMap();
      312          Map<String, CanalSyncChannel> secondaryChannels = Maps.newHashMap();
      313          EntryPosition startPosition = dataEvents.getPositionRange().getStart();
      314          EntryPosition endPosition = dataEvents.getPositionRange().getEnd();
      315          for (CanalSyncChannel channel : idToChannels.values()) {
      316              String key = CanalUtils.getFullName(channel.getSrcDataBase(), channel.getSrcTable());
      317              // if last commit position is null, it is the first time to execute batch
      318              EntryPosition commitPosition = positionMeta.getCommitPosition(channel.getId());
      319              if (commitPosition != null) {
      320                  if (commitPosition.compareTo(startPosition) < 0) {
      321                      preferChannels.put(key, channel);
      322                  } else if (commitPosition.compareTo(endPosition) < 0) {
      323                      secondaryChannels.put(key, channel);
      324                  }
      325              } else {
      326                  preferChannels.put(key, channel);
      327              }
      328          }
      329  
      330          // distribute data to channels
      331          for (CanalEntry.Entry entry : dataEvents.getDatas()) {
      332              CanalEntry.EntryType entryType = entry.getEntryType();
      333              try {
      334                  if (entryType == CanalEntry.EntryType.ROWDATA) {
      335                      CanalEntry.RowChange rowChange;
      336                      try {
      337                          rowChange = CanalEntry.RowChange.parseFrom(entry.getStoreValue());
      338                      } catch (InvalidProtocolBufferException e) {
      339                          throw new CanalException("parse event has an error , data:" + entry.toString(), e);
      340                      }
      341                      final CanalEntry.Header header = entry.getHeader();
      342                      final CanalEntry.EventType eventType = rowChange.getEventType();
      343                      if (!CanalUtils.isDML(eventType) || rowChange.getIsDdl()) {
      344                          String sql = rowChange.getSql();
      345                          processDDL(header, eventType, sql);
      346                          return;
      347                      }
      348                      String schemaTableName = CanalUtils.getFullName(header.getSchemaName(), header.getTableName());
      349                      if (preferChannels.containsKey(schemaTableName)) {
      350                          CanalSyncChannel channel = preferChannels.get(schemaTableName);
      351                          channel.submit(batchId, eventType, rowChange);
      352                      } else if (secondaryChannels.containsKey(schemaTableName)) {
      353                          CanalSyncChannel channel = secondaryChannels.get(schemaTableName);
      354                          EntryPosition position = EntryPosition.createPosition(entry);
      355                          EntryPosition commitPosition = positionMeta.getCommitPosition(channel.getId());
      356                          if (position.compareTo(commitPosition) > 0) {
      357                              channel.submit(batchId, eventType, rowChange);
      358                          }
      359                      }
      360                      // print row
      361                      if (debug) {
      362                          CanalUtils.printRow(rowChange, header);
      363                      }
      364                  }
      365              } catch (Exception e) {
      366                  logger.error("execute event has an error, data: {}, msg: {}", entry.toString(), e);
      367                  throw new UserException("execute batch failed, batchId: " + batchId + " ,msg: " + e.getMessage());
      368              }
      369          }

      404      private void ack() {
      405          if (!ackBatches.isEmpty()) {
      406              logger.info("client ack batches: {}", ackBatches);
      407              while (!ackBatches.isEmpty()) {
      408                  // ack the oldest batch
      409                  long minBatchId = Collections.min(ackBatches);
      410                  connector.ack(minBatchId);
      411                  ackBatches.remove(minBatchId);
      412                  PositionRange<EntryPosition> positionRange = positionMeta.removeBatch(minBatchId);
      413                  positionMeta.setAckPosition(positionRange.getEnd());
      414                  positionMeta.setAckTime(System.currentTimeMillis());
      415              }
      416          }
