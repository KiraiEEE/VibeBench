      191          Objects.requireNonNull(candidateArgument, "Candidate type cannot be null");
      192          if (!isAssignableFrom(candidateArgument.getType())) {
      193              return false;
      194          }
      195          Argument[] typeParameters = getTypeParameters();
      196          Argument[] candidateArgumentTypeParameters = candidateArgument.getTypeParameters();
      197          if (typeParameters.length == 0) {
      198              // Wildcard or no type parameters
      199              return candidateArgumentTypeParameters.length >= 0;
      200          }
      201          if (candidateArgumentTypeParameters.length == 0) {
      202              for (Argument typeParameter : typeParameters) {
      203                  if (typeParameter.getType() != Object.class) {
      204                      return false;
      205                  }
      206              }
      207              // Wildcard
      208              return true;
      209          }
      210          for (int i = 0; i < typeParameters.length; i++) {
      211              Argument typeParameter = typeParameters[i];
      212              Argument candidateArgumentTypeParameter = candidateArgumentTypeParameters[i];
      213              if (!typeParameter.isAssignableFrom(candidateArgumentTypeParameter)) {
      214                  return false;

      216          }
      217          return true;
