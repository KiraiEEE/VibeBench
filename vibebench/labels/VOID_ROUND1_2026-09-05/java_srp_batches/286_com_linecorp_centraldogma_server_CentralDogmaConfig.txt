      145       * starts with a prefix followed by a colon {@code ':'}.

      147      @Nullable
      148      public static String convertValue(@Nullable String value, String propertyName) {
      149          if (value == null) {
      150              return null;
      151          }
      152  
      153          final int index = value.indexOf(':');
      154          if (index <= 0) {
      155              // no prefix or starts with ':'.
      156              return value;
      157          }
      158  
      159          final String prefix = value.substring(0, index);
      160          if (!PREFIX_PATTERN.matcher(prefix).matches()) {
      161              // Not a prefix.
      162              return value;
      163          }
      164  
      165          final String rest = value.substring(index + 1);
      166  
      167          final ConfigValueConverter converter = CONFIG_VALUE_CONVERTERS.get(prefix);
      168          if (converter != null) {
      169              return converter.convert(prefix, rest);
      170          }
      171          logger.warn("No {} found for {}. prefix: {}",
      172                      ConfigValueConverter.class.getSimpleName(), propertyName, prefix);
      173          return value;
      174      }
