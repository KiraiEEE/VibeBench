      104      public synchronized CompletableFuture<Void> addConsumer(Consumer consumer) {
      105          if (IS_FENCED_UPDATER.get(this) == TRUE) {
      106              log.warn("Attempting to add consumer {} on a fenced subscription", consumer);
      107              return FutureUtil.failedFuture(new SubscriptionFencedException("Subscription is fenced"));
      108          }
      109  
      110          if (dispatcher == null || !dispatcher.isConsumerConnected()) {
      111              Dispatcher previousDispatcher = null;
      112  
      113              switch (consumer.subType()) {
      114              case Exclusive:
      115                  if (dispatcher == null || dispatcher.getType() != SubType.Exclusive) {
      116                      previousDispatcher = dispatcher;
      117                      dispatcher = new NonPersistentDispatcherSingleActiveConsumer(SubType.Exclusive, 0, topic, this);
      118                  }
      119                  break;
      120              case Shared:
      121                  if (dispatcher == null || dispatcher.getType() != SubType.Shared) {
      122                      previousDispatcher = dispatcher;
      123                      dispatcher = new NonPersistentDispatcherMultipleConsumers(topic, this);
      124                  }
      125                  break;
      126              case Failover:
      127                  int partitionIndex = TopicName.getPartitionIndex(topicName);
      128                  if (partitionIndex < 0) {
      129                      // For non partition topics, assume index 0 to pick a predictable consumer
      130                      partitionIndex = 0;
      131                  }
      132  
      133                  if (dispatcher == null || dispatcher.getType() != SubType.Failover) {
      134                      previousDispatcher = dispatcher;
      135                      dispatcher = new NonPersistentDispatcherSingleActiveConsumer(SubType.Failover, partitionIndex,
      136                              topic, this);
      137                  }
      138                  break;
      139              case Key_Shared:
      140                  KeySharedMeta ksm = consumer.getKeySharedMeta();
      141                  if (dispatcher == null || dispatcher.getType() != SubType.Key_Shared
      142                          || !((NonPersistentStickyKeyDispatcherMultipleConsumers) dispatcher)
      143                                  .hasSameKeySharedPolicy(ksm)) {
      144                      previousDispatcher = dispatcher;
      145                      this.dispatcher = new NonPersistentStickyKeyDispatcherMultipleConsumers(topic, this, ksm);
      146                  }
      147                  break;
      148              default:
      149                  return FutureUtil.failedFuture(new ServerMetadataException("Unsupported subscription type"));
      150              }
      151  
      152              if (previousDispatcher != null) {
      153                  previousDispatcher.close().thenRun(() -> {
      154                      log.info("[{}][{}] Successfully closed previous dispatcher", topicName, subName);
      155                  }).exceptionally(ex -> {
      156                      log.error("[{}][{}] Failed to close previous dispatcher", topicName, subName, ex);
      157                      return null;
      158                  });
      159              }
      160          } else {
      161              Optional<CompletableFuture<Void>> compatibilityError =
      162                      checkForConsumerCompatibilityErrorWithDispatcher(dispatcher, consumer);
      163              if (compatibilityError.isPresent()) {
      164                  return compatibilityError.get();
      165              }
      166          }
      167  
      168          return dispatcher.addConsumer(consumer);
      169      }

      369       * @return CompletableFuture indicating the completion of delete operation
      370       */
      371      private CompletableFuture<Void> delete(boolean closeIfConsumersConnected) {
      372          CompletableFuture<Void> deleteFuture = new CompletableFuture<>();
      373  
      374          log.info("[{}][{}] Unsubscribing", topicName, subName);
      375  
      376          CompletableFuture<Void> closeSubscriptionFuture = new CompletableFuture<>();
      377  
      378          if (closeIfConsumersConnected) {
      379              this.close(true, Optional.empty()).thenRun(() -> {
      380                  closeSubscriptionFuture.complete(null);
      381              }).exceptionally(ex -> {
      382                  log.error("[{}][{}] Error disconnecting and closing subscription", topicName, subName, ex);
      383                  closeSubscriptionFuture.completeExceptionally(ex);
      384                  return null;
      385              });
      386          } else {
      387              this.fence().thenRun(() -> {
      388                  closeSubscriptionFuture.complete(null);
      389              }).exceptionally(exception -> {
      390                  log.error("[{}][{}] Error closing subscription", topicName, subName, exception);
      391                  closeSubscriptionFuture.completeExceptionally(exception);
      392                  return null;
      393              });
      394          }
      395  
      396          // cursor close handles pending delete (ack) operations
      397          closeSubscriptionFuture.thenCompose(v -> topic.unsubscribe(subName)).thenAccept(v -> {
      398              synchronized (this) {
      399                  (dispatcher != null ? dispatcher.close() : CompletableFuture.completedFuture(null)).thenRun(() -> {
      400                      log.info("[{}][{}] Successfully deleted subscription", topicName, subName);
      401                      deleteFuture.complete(null);
      402                  }).exceptionally(ex -> {
      403                      IS_FENCED_UPDATER.set(this, FALSE);
      404                      if (dispatcher != null) {
      405                          dispatcher.reset();
      406                      }
      407                      log.error("[{}][{}] Error deleting subscription", topicName, subName, ex);
      408                      deleteFuture.completeExceptionally(ex);
      409                      return null;

      411              }
      412          }).exceptionally(exception -> {
      413              IS_FENCED_UPDATER.set(this, FALSE);
      414              log.error("[{}][{}] Error deleting subscription", topicName, subName, exception);
      415              deleteFuture.completeExceptionally(exception);
