      178    private boolean setupRunnerParametersForFolderIfApplicable(final Module module,
      179      @NotNull final KarateRunConfiguration configuration,
      180      @NotNull final VirtualFile dir) {
      181      if (module == null) {
      182        return false;
      183      }
      184      if (Arrays.stream(dir.getChildren()).map(VirtualFile::getName).allMatch(s -> s.endsWith(".feature"))) {
      185        configuration.setAllInFolderAreFeature(true);
      186      }
      187      configuration.setPath(getRelativePathFromModule(module, dir.getPath(), dir.getPath()));
      188      configuration.setPreferredTest(PreferredTest.ALL_IN_FOLDER);
      189      configuration.setName("Karate tests in '" + dir.getName() + "'");
      190      return true;

      193    private static String getRelativePathFromModule(Module contextModule, String path, String name) {
      194      if (contextModule == null) {

      198          VirtualFile::getPath)

      200        .map(s -> Optional.ofNullable(FileUtil.getRelativePath(s, path, '/')))
