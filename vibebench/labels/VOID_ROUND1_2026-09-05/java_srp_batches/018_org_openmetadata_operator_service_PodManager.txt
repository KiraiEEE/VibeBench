       69  
       70      String podName = generateMainPodName(omJob);
       71      String namespace = omJob.getMetadata().getNamespace();
       72  
       73      // First check if pod already exists with the expected name (idempotent operation)
       74      try {
       75        Pod existingPod = client.pods().inNamespace(namespace).withName(podName).get();
       76        if (existingPod != null) {
       77          LOG.info(
       78              "Main pod already exists: {} in namespace: {}, returning existing pod",
       79              podName,
       80              namespace);
       81          return existingPod;
       82        }
       83      } catch (Exception e) {
       84        LOG.debug("Pod {} does not exist yet, proceeding with creation", podName);
       85      }
       86  
       87      // Build the pod specification
       88      OMJobSpec.OMJobPodSpec podSpec = omJob.getSpec().getMainPodSpec();
       89      Pod pod = buildPod(omJob, podName, podSpec, LabelBuilder.buildMainPodLabels(omJob), null);
       90  
       91      try {
       92        Pod createdPod = client.pods().inNamespace(namespace).resource(pod).create();
       93  
       94        LOG.info(
       95            "Created main pod: {} in namespace: {}",
       96            createdPod.getMetadata().getName(),
       97            createdPod.getMetadata().getNamespace());
       98  
       99        return createdPod;
      100  
      101      } catch (Exception e) {
      102        // If creation fails due to "already exists", try to fetch the existing pod
      103        if (e.getMessage() != null && e.getMessage().toLowerCase().contains("already exists")) {
      104          LOG.info("Pod {} already exists (race condition), fetching existing pod", podName);
      105          try {
      106            Pod existingPod = client.pods().inNamespace(namespace).withName(podName).get();
      107            if (existingPod != null) {
      108              return existingPod;
      109            }
      110          } catch (Exception fetchException) {
      111            LOG.warn("Failed to fetch existing pod after creation conflict", fetchException);
      112          }
      113        }
      114        LOG.error("Failed to create main pod for OMJob: {}", omJob.getMetadata().getName(), e);
      115        throw new RuntimeException("Failed to create main pod", e);

      192     * Find main pod for an OMJob
      193     */
      194    public Optional<Pod> findMainPod(OMJobResource omJob) {
      195      // First try with operator-created pod selector
      196      Optional<Pod> pod = findPod(omJob, LabelBuilder.buildMainPodSelector(omJob));
      197  
      198      if (pod.isPresent()) {
      199        return pod;
      200      }
      201  
      202      // If not found, try with server-created pod selector (fallback for compatibility)
      203      Map<String, String> serverSelector = new HashMap<>();
      204      serverSelector.put(LabelBuilder.LABEL_OMJOB_NAME, omJob.getMetadata().getName());
      205      serverSelector.put(LabelBuilder.LABEL_POD_TYPE, LabelBuilder.POD_TYPE_MAIN);
      206      // Note: server-created pods have app.kubernetes.io/managed-by = openmetadata
      207  
      208      Optional<Pod> serverPod = findPod(omJob, serverSelector);
      209  
      210      if (serverPod.isPresent()) {
      211        LOG.info("Found server-created main pod for OMJob: {}", omJob.getMetadata().getName());
      212        return serverPod;
      213      }
      214  
      215      // Last resort: try to find by pod name if it was recorded in status
      216      String recordedPodName = omJob.getStatus() != null ? omJob.getStatus().getMainPodName() : null;
      217      if (recordedPodName != null && !recordedPodName.isEmpty()) {
      218        try {
      219          Pod namedPod =
      220              client
      221                  .pods()
      222                  .inNamespace(omJob.getMetadata().getNamespace())
      223                  .withName(recordedPodName)
      224                  .get();
      225          if (namedPod != null) {
      226            LOG.info("Found main pod by name for OMJob: {}", omJob.getMetadata().getName());
      227            return Optional.of(namedPod);
      228          }
      229        } catch (Exception e) {
      230          LOG.debug("Could not find pod by name {}: {}", recordedPodName, e.getMessage());
      231        }
      232      }
      233  
      234      return Optional.empty();
      235    }

      270    public Optional<Integer> getPodExitCode(Pod pod) {
      271      PodStatus status = getPodStatus(pod);
      272      if (status == null || status.getContainerStatuses() == null) {
      273        return Optional.empty();
      274      }
      275  
      276      return status.getContainerStatuses().stream()
      277          .filter(cs -> cs.getState() != null && cs.getState().getTerminated() != null)
      278          .map(cs -> cs.getState().getTerminated().getExitCode())
      279          .findFirst();

      346     * Build container security context from pod spec.
      347     * Uses settings from the CR's pod security context, with secure defaults for
      348     * container-specific settings not specified at the pod level.
      349     */
      350    private SecurityContext buildContainerSecurityContext(OMJobSpec.OMJobPodSpec podSpec) {
      351      SecurityContextBuilder builder = new SecurityContextBuilder();
      352  
      353      // Use runAsNonRoot and runAsUser from pod security context if available
      354      if (podSpec.getSecurityContext() != null) {
      355        PodSecurityContext podSecurity = podSpec.getSecurityContext();
      356        if (podSecurity.getRunAsNonRoot() != null) {
      357          builder.withRunAsNonRoot(podSecurity.getRunAsNonRoot());
      358        }
      359        if (podSecurity.getRunAsUser() != null) {
      360          builder.withRunAsUser(podSecurity.getRunAsUser());
      361        }
      362        if (podSecurity.getRunAsGroup() != null) {
      363          builder.withRunAsGroup(podSecurity.getRunAsGroup());
      364        }
      365      }
      366  
      367      // Always apply secure container-specific defaults
      368      builder
      369          .withAllowPrivilegeEscalation(false)
      370          .withReadOnlyRootFilesystem(false) // Ingestion may need to write temp files
      371          .withCapabilities(new CapabilitiesBuilder().withDrop("ALL").build());
      372  
      373      return builder.build();
      374    }

      418    private String determinePipelineStatus(OMJobResource omJob) {
      419      OMJobStatus status = omJob.getStatus();
      420      if (status == null || status.getMainPodExitCode() == null) {
      421        return PipelineStatusType.FAILED.value();
      422      }
      423      return status.getMainPodExitCode() == 0
      424          ? PipelineStatusType.SUCCESS.value()
      425          : PipelineStatusType.FAILED.value();
