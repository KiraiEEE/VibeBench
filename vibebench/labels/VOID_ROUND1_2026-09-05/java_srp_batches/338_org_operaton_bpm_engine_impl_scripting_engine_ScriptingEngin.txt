      104      if (language != null) {

      107  
      108      ProcessApplicationReference pa = Context.getCurrentProcessApplication();
      109      ProcessEngineConfigurationImpl config = Context.getProcessEngineConfiguration();
      110  
      111      ScriptEngine engine = null;
      112      if (config.isEnableFetchScriptEngineFromProcessApplication() && pa != null) {
      113        engine = getPaScriptEngine(language, pa);

      116      if(engine == null) {
