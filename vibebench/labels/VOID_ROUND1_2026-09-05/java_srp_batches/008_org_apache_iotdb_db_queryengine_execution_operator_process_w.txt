      182    private void processRecognizerStates() {

      184        PartitionState state = partitionRecognizer.nextState();

      228            && !tsBlockBuilder.isFull()
      229            && partitionExecutor.hasNext()) {
      230          partitionExecutor.processNextRow(tsBlockBuilder);

      233        if (!partitionExecutor.hasNext()) {
      234          cachedPartitionExecutors.removeFirst();

      238          return getTsBlockFromTsBlockBuilder();
