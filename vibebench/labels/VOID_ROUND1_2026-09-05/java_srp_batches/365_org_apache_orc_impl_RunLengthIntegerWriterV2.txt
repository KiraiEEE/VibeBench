      189    private void writeDeltaValues() throws IOException {
      190      int len = 0;
      191      int fb = bitsDeltaMax;
      192      int efb = 0;
      193  
      194      if (alignedBitpacking) {
      195        fb = utils.getClosestAlignedFixedBits(fb);
      196      }
      197  
      198      if (isFixedDelta) {
      199        // if fixed run length is greater than threshold then it will be fixed
      200        // delta sequence with delta value 0 else fixed delta sequence with
      201        // non-zero delta value
      202        if (fixedRunLength > MIN_REPEAT) {
      203          // ex. sequence: 2 2 2 2 2 2 2 2
      204          len = fixedRunLength - 1;
      205          fixedRunLength = 0;
      206        } else {
      207          // ex. sequence: 4 6 8 10 12 14 16
      208          len = variableRunLength - 1;
      209          variableRunLength = 0;
      210        }
      211      } else {
      212        // fixed width 0 is used for long repeating values.
      213        // sequences that require only 1 bit to encode will have an additional bit
      214        if (fb == 1) {
      215          fb = 2;
      216        }
      217        efb = utils.encodeBitWidth(fb);
      218        efb = efb << 1;
      219        len = variableRunLength - 1;
      220        variableRunLength = 0;
      221      }
      222  
      223      // extract the 9th bit of run length
      224      final int tailBits = (len & 0x100) >>> 8;
      225  
      226      // create first byte of the header
      227      final int headerFirstByte = getOpcode() | efb | tailBits;
      228  
      229      // second byte of the header stores the remaining 8 bits of runlength
      230      final int headerSecondByte = len & 0xff;
      231  
      232      // write header
      233      output.write(headerFirstByte);
      234      output.write(headerSecondByte);
      235  
      236      // store the first value from zigzag literal array
      237      if (signed) {
      238        utils.writeVslong(output, literals[0]);
      239      } else {
      240        utils.writeVulong(output, literals[0]);
      241      }
      242  
      243      if (isFixedDelta) {
      244        // if delta is fixed then we don't need to store delta blob
      245        utils.writeVslong(output, fixedDelta);
      246      } else {
      247        // store the first value as delta value using zigzag encoding
      248        utils.writeVslong(output, adjDeltas[0]);
      249  
      250        // adjacent delta values are bit packed. The length of adjDeltas array is
      251        // always one less than the number of literals (delta difference for n
      252        // elements is n-1). We have already written one element, write the
      253        // remaining numLiterals - 2 elements here
      254        utils.writeInts(adjDeltas, 1, numLiterals - 2, fb, output);
      255      }
      256    }

      258    private void writePatchedBaseValues() throws IOException {
      259  
      260      // NOTE: Aligned bit packing cannot be applied for PATCHED_BASE encoding
      261      // because patch is applied to MSB bits. For example: If fixed bit width of
      262      // base value is 7 bits and if patch is 3 bits, the actual value is
      263      // constructed by shifting the patch to left by 7 positions.
      264      // actual_value = patch << 7 | base_value
      265      // So, if we align base_value then actual_value can not be reconstructed.
      266  
      267      // write the number of fixed bits required in next 5 bits
      268      final int fb = brBits95p;
      269      final int efb = utils.encodeBitWidth(fb) << 1;
      270  
      271      // adjust variable run length, they are one off
      272      variableRunLength -= 1;
      273  
      274      // extract the 9th bit of run length
      275      final int tailBits = (variableRunLength & 0x100) >>> 8;
      276  
      277      // create first byte of the header
      278      final int headerFirstByte = getOpcode() | efb | tailBits;
      279  
      280      // second byte of the header stores the remaining 8 bits of runlength
      281      final int headerSecondByte = variableRunLength & 0xff;
      282  
      283      // if the min value is negative toggle the sign
      284      final boolean isNegative = min < 0;
      285      if (isNegative) {
      286        min = -min;
      287      }
      288  
      289      // find the number of bytes required for base and shift it by 5 bits
      290      // to accommodate patch width. The additional bit is used to store the sign
      291      // of the base value.
      292      final int baseWidth = utils.findClosestNumBits(min) + 1;
      293      final int baseBytes = baseWidth % 8 == 0 ? baseWidth / 8 : (baseWidth / 8) + 1;
      294      final int bb = (baseBytes - 1) << 5;
      295  
      296      // if the base value is negative then set MSB to 1
      297      if (isNegative) {
      298        min |= (1L << ((baseBytes * 8) - 1));
      299      }
      300  
      301      // third byte contains 3 bits for number of bytes occupied by base
      302      // and 5 bits for patchWidth
      303      final int headerThirdByte = bb | utils.encodeBitWidth(patchWidth);
      304  
      305      // fourth byte contains 3 bits for page gap width and 5 bits for
      306      // patch length
      307      final int headerFourthByte = (patchGapWidth - 1) << 5 | patchLength;
      308  
      309      // write header
      310      output.write(headerFirstByte);
      311      output.write(headerSecondByte);
      312      output.write(headerThirdByte);
      313      output.write(headerFourthByte);
      314  
      315      // write the base value using fixed bytes in big endian order
      316      for(int i = baseBytes - 1; i >= 0; i--) {
      317        byte b = (byte) ((min >>> (i * 8)) & 0xff);
      318        output.write(b);
      319      }
      320  
      321      // base reduced literals are bit packed
      322      int closestFixedBits = utils.getClosestFixedBits(fb);
      323  
      324      utils.writeInts(baseRedLiterals, 0, numLiterals, closestFixedBits,
      325          output);
      326  
      327      // write patch list
      328      closestFixedBits = utils.getClosestFixedBits(patchGapWidth + patchWidth);
      329  
      330      utils.writeInts(gapVsPatchList, 0, gapVsPatchList.length, closestFixedBits,
      331          output);
      332  
      333      // reset run length
      334      variableRunLength = 0;
      335    }

      346  
      347      // write the number of fixed bits required in next 5 bits
      348      int fb = zzBits100p;

      351        fb = utils.getClosestAlignedFixedBits(fb);

      355

      380    private void writeShortRepeatValues() throws IOException {
      381      // get the value that is repeating, compute the bits and bytes required
      382      long repeatVal = 0;
      383      if (signed) {
      384        repeatVal = utils.zigzagEncode(literals[0]);
      385      } else {
      386        repeatVal = literals[0];
      387      }
      388  
      389      final int numBitsRepeatVal = utils.findClosestNumBits(repeatVal);
      390      final int numBytesRepeatVal = numBitsRepeatVal % 8 == 0 ? numBitsRepeatVal >>> 3
      391          : (numBitsRepeatVal >>> 3) + 1;
      392  
      393      // write encoding type in top 2 bits
      394      int header = getOpcode();
      395  
      396      // write the number of bytes required for the value
      397      header |= ((numBytesRepeatVal - 1) << 3);
      398  
      399      // write the run length
      400      fixedRunLength -= MIN_REPEAT;
      401      header |= fixedRunLength;
      402  
      403      // write the header
      404      output.write(header);
      405  
      406      // write the repeating value in big endian byte order
      407      for(int i = numBytesRepeatVal - 1; i >= 0; i--) {
      408        int b = (int) ((repeatVal >>> (i * 8)) & 0xff);
      409        output.write(b);
      410      }
      411  
      412      fixedRunLength = 0;
      413    }

      430    private void determineEncoding() {
      431      // we need to compute zigzag values for DIRECT encoding if we decide to
      432      // break early for delta overflows or for shorter runs
      433  
      434      // not a big win for shorter runs to determine encoding
      435      if (numLiterals <= MIN_REPEAT) {
      436        prepareForDirectOrPatchedBase();
      437        encoding = EncodingType.DIRECT;
      438        return;
      439      }
      440  
      441      // DELTA encoding check
      442  
      443      // for identifying monotonic sequences
      444      boolean isIncreasing = true;
      445      boolean isDecreasing = true;
      446      this.isFixedDelta = true;
      447  
      448      this.min = literals[0];
      449      long max = literals[0];
      450      final long initialDelta = literals[1] - literals[0];
      451      long currDelta = 0;
      452      long deltaMax = 0;
      453      this.adjDeltas[0] = initialDelta;
      454  
      455      for (int i = 1; i < numLiterals; i++) {
      456        final long l1 = literals[i];
      457        final long l0 = literals[i - 1];
      458        currDelta = l1 - l0;
      459        min = Math.min(min, l1);
      460        max = Math.max(max, l1);
      461  
      462        isIncreasing &= (l0 <= l1);
      463        isDecreasing &= (l0 >= l1);
      464  
      465        isFixedDelta &= (currDelta == initialDelta);
      466        if (i > 1) {
      467          adjDeltas[i - 1] = Math.abs(currDelta);
      468          deltaMax = Math.max(deltaMax, adjDeltas[i - 1]);
      469        }
      470      }
      471  
      472      // its faster to exit under delta overflow condition without checking for
      473      // PATCHED_BASE condition as encoding using DIRECT is faster and has less
      474      // overhead than PATCHED_BASE
      475      if (!utils.isSafeSubtract(max, min)) {
      476        prepareForDirectOrPatchedBase();
      477        encoding = EncodingType.DIRECT;
      478        return;
      479      }
      480  
      481      // invariant - subtracting any number from any other in the literals after
      482      // this point won't overflow
      483  
      484      // if min is equal to max then the delta is 0, this condition happens for
      485      // fixed values run >10 which cannot be encoded with SHORT_REPEAT

      487        assert isFixedDelta : min + "==" + max +
      488            ", isFixedDelta cannot be false";
      489        assert currDelta == 0 : min + "==" + max + ", currDelta should be zero";
      490        fixedDelta = 0;
      491        encoding = EncodingType.DELTA;
      492        return;
      493      }
      494  
      495      if (isFixedDelta) {
      496        assert currDelta == initialDelta
      497            : "currDelta should be equal to initialDelta for fixed delta encoding";
      498        encoding = EncodingType.DELTA;
      499        fixedDelta = currDelta;
      500        return;
      501      }
      502  
      503      // if initialDelta is 0 then we cannot delta encode as we cannot identify
      504      // the sign of deltas (increasing or decreasing)
      505      if (initialDelta != 0) {

      507        // delta encoding

      512          encoding = EncodingType.DELTA;
      513          return;
      514        }
      515      }
