      112    private void runPipeline(ReindexingConfiguration config, Set<String> entities)
      113        throws InterruptedException {
      114      int numConsumers =
      115          config.consumerThreads() > 0 ? Math.min(config.consumerThreads(), MAX_CONSUMER_THREADS) : 2;
      116      int queueSize = config.queueSize() > 0 ? config.queueSize() : DEFAULT_QUEUE_SIZE;
      117      int batchSize = config.batchSize();
      118  
      119      taskQueue = new LinkedBlockingQueue<>(queueSize);
      120      String jobIdTag = MDC.get("reindexJobId");
      121      String threadPrefix = "reindex-" + (jobIdTag != null ? jobIdTag + "-" : "");
      122      consumerExecutor =
      123          Executors.newFixedThreadPool(
      124              numConsumers,
      125              Thread.ofPlatform().name(threadPrefix + "pipeline-consumer-", 0).factory());
      126      producerExecutor =
      127          Executors.newFixedThreadPool(
      128              config.producerThreads() > 0 ? config.producerThreads() : 2,
      129              Thread.ofPlatform().name(threadPrefix + "pipeline-producer-", 0).factory());
      130      jobExecutor =
      131          Executors.newFixedThreadPool(
      132              Math.min(entities.size(), MAX_JOB_THREADS),
      133              Thread.ofPlatform().name(threadPrefix + "pipeline-job-", 0).factory());
      134  
      135      entityReader = new EntityReader(producerExecutor, stopped);
      136  
      137      CountDownLatch consumerLatch = new CountDownLatch(numConsumers);
      138      Map<String, String> mdc = MDC.getCopyOfContextMap();
      139      for (int i = 0; i < numConsumers; i++) {
      140        final int id = i;
      141        consumerExecutor.submit(
      142            () -> {
      143              if (mdc != null) MDC.setContextMap(mdc);
      144              try {
      145                runConsumer(id, consumerLatch);
      146              } finally {
      147                MDC.clear();
      148              }
      149            });
      150      }
      151  
      152      try {
      153        readAllEntities(config, entities, batchSize);
      154        signalConsumersToStop(numConsumers);
      155        consumerLatch.await();
      156      } catch (InterruptedException e) {
      157        stopped.set(true);
      158        Thread.currentThread().interrupt();
      159        throw e;
      160      } finally {
      161        shutdownExecutors();
      162      }
      163    }

      165    private void readAllEntities(ReindexingConfiguration config, Set<String> entities, int batchSize)
      166        throws InterruptedException {
      167      List<String> ordered = EntityPriority.sortByPriority(entities);
      168      Phaser producerPhaser = new Phaser(entities.size());
      169      Map<String, String> mdc = MDC.getCopyOfContextMap();
      170  
      171      for (String entityType : ordered) {
      172        jobExecutor.submit(
      173            () -> {
      174              if (mdc != null) MDC.setContextMap(mdc);
      175              try {
      176                int totalRecords = getTotalEntityRecords(entityType);
      177                listeners.onEntityTypeStarted(entityType, totalRecords);
      178  
      179                int effectiveBatchSize =
      180                    EntityBatchSizeEstimator.estimateBatchSize(entityType, batchSize);
      181                Long filterStartTs = null;
      182                Long filterEndTs = null;
      183                long startTs = config.getTimeSeriesStartTs(entityType);
      184                if (startTs > 0) {
      185                  filterStartTs = startTs;
      186                  filterEndTs = System.currentTimeMillis();
      187                }
      188                entityReader.readEntity(
      189                    entityType,
      190                    totalRecords,
      191                    effectiveBatchSize,
      192                    producerPhaser,
      193                    (type, batch, offset) -> {
      194                      if (!stopped.get()) {
      195                        taskQueue.put(new IndexingTask<>(type, batch, offset));
      196                      }
      197                    },
      198                    filterStartTs,
      199                    filterEndTs);
      200              } catch (Exception e) {
      201                LOG.error("Error reading entity type {}", entityType, e);
      202              } finally {
      203                producerPhaser.arriveAndDeregister();
      204                MDC.clear();
      205              }
      206            });
      207      }
      208  
      209      int phase = 0;
      210      while (!producerPhaser.isTerminated()) {
      211        if (stopped.get() || Thread.currentThread().isInterrupted()) {
      212          break;
      213        }
      214        try {
      215          producerPhaser.awaitAdvanceInterruptibly(phase, 1, TimeUnit.SECONDS);
      216          break;
      217        } catch (TimeoutException e) {
      218          // Continue
      219        }
      220      }

      224    private void runConsumer(int consumerId, CountDownLatch consumerLatch) {
      225      try {
      226        while (!stopped.get()) {
      227          IndexingTask<?> task = taskQueue.poll(200, TimeUnit.MILLISECONDS);
      228          if (task == null) continue;
      229          if (POISON_PILL.equals(task.entityType())) break;
      230  
      231          String entityType = task.entityType();
      232          ResultList<?> entities = task.entities();
      233          Map<String, Object> contextData = createContextData(entityType);
      234  
      235          int readerSuccess = listOrEmpty(entities.getData()).size();
      236          int readerFailed = listOrEmpty(entities.getErrors()).size();
      237          int readerWarnings = entities.getWarningsCount() != null ? entities.getWarningsCount() : 0;
      238          updateReaderStats(readerSuccess, readerFailed, readerWarnings);
      239  
      240          try {
      241            if (!EntityReader.TIME_SERIES_ENTITIES.contains(entityType)) {
      242              searchIndexSink.write(
      243                  (java.util.List<EntityInterface>) entities.getData(), contextData);
      244            } else {
      245              searchIndexSink.write(
      246                  (java.util.List<EntityTimeSeriesInterface>) entities.getData(), contextData);
      247            }
      248  
      249            StepStats entityStats = new StepStats();
      250            entityStats.setSuccessRecords(readerSuccess);
      251            entityStats.setFailedRecords(readerFailed);
      252            updateEntityAndJobStats(entityType, entityStats);
      253  
      254            if (Entity.TABLE.equals(entityType)) {
      255              updateColumnStatsFromSink();
      256            }
      257  
      258            listeners.onProgressUpdate(stats.get(), null);
      259          } catch (Exception e) {

      262                new IndexingError()
      263                    .withErrorSource(IndexingError.ErrorSource.SINK)
      264                    .withMessage(e.getMessage());

      267        }
      268      } catch (InterruptedException e) {
      269        Thread.currentThread().interrupt();
      270      } finally {
      271        consumerLatch.countDown();

      311    private void finalizeReindex() {
      312      if (recreateIndexHandler == null || recreateContext == null) return;
      313  
      314      try {
      315        recreateContext
      316            .getEntities()
      317            .forEach(
      318                entityType -> {
      319                  if (promotedEntities.contains(entityType)) return;
      320                  try {
      321                    EntityReindexContext ctx = buildEntityReindexContext(entityType);
      322                    recreateIndexHandler.finalizeReindex(ctx, !stopped.get());
      323                  } catch (Exception ex) {
      324                    LOG.error("Failed to finalize reindex for {}", entityType, ex);
      325                  }
      326                });
      327      } finally {
      328        recreateContext = null;
      329        promotedEntities.clear();
      330      }

      350      syncSinkStats();
      351      updateColumnStatsFromSink();

      353      if (currentStats != null) {
      354        StatsReconciler.reconcile(currentStats);
      355      }
      356  
      357      ExecutionResult.Status status;
      358      if (stopped.get()) {
      359        status = ExecutionResult.Status.STOPPED;
      360        listeners.onJobStopped(currentStats);
      361      } else if (hasFailures()) {
      362        status = ExecutionResult.Status.COMPLETED_WITH_ERRORS;
      363        listeners.onJobCompletedWithErrors(currentStats, System.currentTimeMillis() - startTime);
      364      } else {
      365        status = ExecutionResult.Status.COMPLETED;
      366        listeners.onJobCompleted(currentStats, System.currentTimeMillis() - startTime);
      367      }
      368  
      369      return ExecutionResult.fromStats(currentStats, status, startTime);

      372    private boolean hasFailures() {
      373      Stats s = stats.get();
      374      if (s == null || s.getJobStats() == null) return false;
      375      StepStats js = s.getJobStats();
      376      long failed = js.getFailedRecords() != null ? js.getFailedRecords() : 0;
      377      long success = js.getSuccessRecords() != null ? js.getSuccessRecords() : 0;
      378      long total = js.getTotalRecords() != null ? js.getTotalRecords() : 0;
      379      return failed > 0 || (total > 0 && success < total);
      380    }
