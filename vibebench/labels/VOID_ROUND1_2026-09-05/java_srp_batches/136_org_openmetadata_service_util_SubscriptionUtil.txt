       81      Set<String> data = new HashSet<>();
       82      UserRepository userEntityRepository = (UserRepository) Entity.getEntityRepository(USER);
       83      ResultList<User> result;
       84      ListFilter listFilter = new ListFilter(Include.ALL);
       85      listFilter.addQueryParam("isAdmin", "true");
       86      String after = null;
       87      try {
       88        do {
       89          result =
       90              userEntityRepository.listAfter(
       91                  null, userEntityRepository.getFields("email,profile"), listFilter, 50, after);

       93          after = result.getPaging().getAfter();
       94        } while (after != null);
       95      } catch (Exception ex) {

       97      }
       98      return data;

      101    public static Set<String> getEmailOrWebhookEndpointForUsers(
      102        List<User> users, SubscriptionDestination.SubscriptionType type) {
      103      if (type == SubscriptionDestination.SubscriptionType.EMAIL) {
      104        return users.stream().map(User::getEmail).collect(Collectors.toSet());
      105      } else {
      106        return users.stream()
      107            .map(user -> getWebhookUrlFromProfile(user.getProfile(), user.getId(), USER, type))
      108            .filter(Optional::isPresent)
      109            .map(Optional::get)
      110            .collect(Collectors.toSet());
      111      }
      112    }

      133    public static Set<String> getOwnerOrFollowers(
      134        SubscriptionDestination.SubscriptionType type,
      135        CollectionDAO daoCollection,
      136        UUID entityId,
      137        String entityType,
      138        Relationship relationship) {
      139      Set<String> data = new HashSet<>();
      140      try {
      141        List<CollectionDAO.EntityRelationshipRecord> ownerOrFollowers =
      142            daoCollection.relationshipDAO().findFrom(entityId, entityType, relationship.ordinal());
      143        // Users
      144        List<User> users =
      145            ownerOrFollowers.stream()
      146                .filter(e -> USER.equals(e.getType()))
      147                .map(user -> (User) Entity.getEntity(USER, user.getId(), "", Include.NON_DELETED))
      148                .toList();
      149        data.addAll(getEmailOrWebhookEndpointForUsers(users, type));
      150  
      151        // Teams
      152        List<Team> teams =
      153            ownerOrFollowers.stream()
      154                .filter(e -> TEAM.equals(e.getType()))
      155                .map(
      156                    team ->
      157                        (Team)
      158                            Entity.getEntity(
      159                                TEAM, team.getId(), "id,profile,email", Include.NON_DELETED))
      160                .toList();
      161        data.addAll(getEmailOrWebhookEndpointForTeams(teams, type));
      162      } catch (Exception ex) {
      163        LOG.error("Failed in listing all Owners/Followers, Reason : ", ex);
      164      }
      165      return data;
      166    }

      301    private static Optional<String> getWebhookUrlFromProfile(
      302        Profile profile, UUID id, String entityType, SubscriptionDestination.SubscriptionType type) {
      303      if (profile != null) {
      304        SubscriptionConfig subscriptionConfig = profile.getSubscription();
      305        if (subscriptionConfig != null) {
      306          Webhook webhookConfig =
      307              switch (type) {
      308                case SLACK -> profile.getSubscription().getSlack();
      309                case MS_TEAMS -> profile.getSubscription().getMsTeams();
      310                case G_CHAT -> profile.getSubscription().getgChat();
      311                case WEBHOOK -> profile.getSubscription().getGeneric();
      312                default -> null;
      313              };
      314          if (webhookConfig != null && !CommonUtil.nullOrEmpty(webhookConfig.getEndpoint())) {
      315            String endpointStr = webhookConfig.getEndpoint().toString();
      316            // Validate URL before returning
      317            try {
      318              URLValidator.validateURL(endpointStr);
      319              return Optional.of(endpointStr);
      320            } catch (Exception e) {
      321              LOG.warn(
      322                  "[GetWebhookUrlsFromProfile] Invalid webhook URL for owner with id {} type {}: {}",
      323                  id,
      324                  entityType,
      325                  e.getMessage());
      326              return Optional.empty();
      327            }
      328          } else {
      329            LOG.debug(
      330                "[GetWebhookUrlsFromProfile] Owner with id {} type {}, will not get any Notification as not webhook config is missing for type {}, webhookConfig {} ",
      331                id,
      332                entityType,
      333                type.value(),
      334                webhookConfig);
      335          }
      336        }
      337      }
      338      LOG.debug(
      339          "[GetWebhookUrlsFromProfile] Failed to Get Profile for Owner with ID : {} and type {} ",
      340          id,
      341          type);
      342      return Optional.empty();
      343    }

      345    public static Set<String> buildReceiversListFromActions(
      346        SubscriptionAction action,
      347        SubscriptionDestination.SubscriptionCategory category,
      348        SubscriptionDestination.SubscriptionType type,
      349        CollectionDAO daoCollection,
      350        UUID entityId,
      351        String entityType) {
      352      Set<String> receiverList = new HashSet<>();
      353  
      354      if (category.equals(SubscriptionDestination.SubscriptionCategory.USERS)) {
      355        if (nullOrEmpty(action.getReceivers())) {
      356          throw new IllegalArgumentException(
      357              "Email Alert Invoked with Illegal Type and Settings. Emtpy or Null Users Recipients List");
      358        }
      359        List<User> users =
      360            action.getReceivers().stream()
      361                .map(user -> (User) Entity.getEntityByName(USER, user, "", Include.NON_DELETED))
      362                .toList();
      363        receiverList.addAll(getEmailOrWebhookEndpointForUsers(users, type));
      364      } else if (category.equals(SubscriptionDestination.SubscriptionCategory.TEAMS)) {
      365        if (nullOrEmpty(action.getReceivers())) {
      366          throw new IllegalArgumentException(
      367              "Email Alert Invoked with Illegal Type and Settings. Emtpy or Null Teams Recipients List");
      368        }
      369        List<Team> teams =
      370            action.getReceivers().stream()
      371                .map(team -> (Team) Entity.getEntityByName(TEAM, team, "", Include.NON_DELETED))
      372                .toList();
      373        receiverList.addAll(getEmailOrWebhookEndpointForTeams(teams, type));
      374      } else {
      375        receiverList =
      376            action.getReceivers() == null ? receiverList : new HashSet<>(action.getReceivers());
      377      }
      378  
      379      // Send to Admins
      380      if (Boolean.TRUE.equals(action.getSendToAdmins())) {
      381        receiverList.addAll(getAdminsData(type));
      382      }
      383  
      384      // Send To Owners
      385      if (Boolean.TRUE.equals(action.getSendToOwners())) {
      386        receiverList.addAll(
      387            getOwnerOrFollowers(type, daoCollection, entityId, entityType, Relationship.OWNS));
      388      }
      389  
      390      // Send To Followers
      391      if (Boolean.TRUE.equals(action.getSendToFollowers())) {
      392        receiverList.addAll(
      393            getOwnerOrFollowers(type, daoCollection, entityId, entityType, Relationship.FOLLOWS));
      394      }
      395  
      396      return receiverList;
      397    }

      399    public static Set<String> getTargetsForAlert(
      400        SubscriptionAction action, SubscriptionDestination destination, ChangeEvent event) {
      401      SubscriptionDestination.SubscriptionCategory category = destination.getCategory();
      402      SubscriptionDestination.SubscriptionType type = destination.getType();
      403  
      404      Set<String> receiverUrls = new HashSet<>();
      405      if (event.getEntityType().equals(THREAD)) {
      406        Thread thread = AlertsRuleEvaluator.getThread(event);
      407        switch (thread.getType()) {
      408          case Task -> receiverUrls.addAll(getTaskAssignees(action, category, type, event));
      409          case Conversation -> receiverUrls.addAll(
      410              handleConversationNotification(action, category, type, event));
      411            // TODO: For Announcement, Immediate Consumer needs to be Notified (find information from
      412            // Lineage)
      413          case Announcement -> {
      414            receiverUrls.addAll(
      415                buildReceivers(
      416                    action,
      417                    category,
      418                    type,
      419                    thread.getEntityRef().getType(),
      420                    thread.getEntityRef().getId()));
      421          }
      422        }
      423      } else {
      424        EntityInterface entityInterface = getEntity(event);
      425        receiverUrls.addAll(
      426            buildReceivers(action, category, type, event.getEntityType(), entityInterface.getId()));
      427  
      428        // Add lineage downstream receivers if enabled
      429        if (Boolean.TRUE.equals(destination.getNotifyDownstream())
      430            && category != SubscriptionDestination.SubscriptionCategory.EXTERNAL) {
      431          LineageGraphExplorer lineageExplorer = new LineageGraphExplorer(Entity.getCollectionDAO());
      432  
      433          Set<EntityReference> downstreamEntities =
      434              lineageExplorer.findUniqueEntitiesDownstream(
      435                  entityInterface.getId(), event.getEntityType(), destination.getDownstreamDepth());
      436  
      437          for (EntityReference downstream : downstreamEntities) {
      438            receiverUrls.addAll(
      439                buildReceivers(action, category, type, downstream.getType(), downstream.getId()));
      440          }
      441        }
      442      }
      443  
      444      return receiverUrls;
      445    }
