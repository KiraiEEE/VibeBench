      179          public FieldInjector(ContainerImpl container, Field field, String name)
      180              throws MissingDependencyException {
      181              this.field = field;
      182              if (!isPublicForReflection(field) && !field.isAccessible()) {
      183                  SecurityManager sm = System.getSecurityManager();
      184                  try {
      185                      if (sm != null) {
      186                          sm.checkPermission(new ReflectPermission("suppressAccessChecks"));
      187                      }
      188                      field.setAccessible(true);
      189                  } catch (AccessControlException e) {
      190                      throw new DependencyException("Security manager in use, could not access field: "
      191                          + field.getDeclaringClass().getName() + "(" + field.getName() + ")", e);
      192                  }
      193              }
      194  
      195              final Key<?> key = Key.newInstance(field.getType(), name);
      196              factory = container.getFactory(key);
      197              if (factory == null) {
      198                  throw new MissingDependencyException("No mapping found for dependency " + key + " in " + field + ".");
      199              }
      200  
      201              this.externalContext = ExternalContext.newInstance(field, key, container);
      202          }

      204          @Override
      205          public void inject(InternalContext context, Object o) {
      206              final ExternalContext<?> previous = context.getExternalContext();
      207              context.setExternalContext(externalContext);
      208              try {
      209                  field.set(o, factory.create(context));
      210              } catch (IllegalAccessException e) {
      211                  throw new AssertionError(e);
      212              } finally {
      213                  context.setExternalContext(previous);
