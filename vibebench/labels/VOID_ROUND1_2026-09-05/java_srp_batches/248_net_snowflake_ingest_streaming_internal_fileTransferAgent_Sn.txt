      308      }

      310      final String[] srcLocations;

      313        srcLocations = mapper.readValue(locationsNode.toString(), String[].class);
      314      } catch (Exception ex) {
      315        throw new SnowflakeSQLException(
      316            queryId,
      317            ErrorCode.INTERNAL_ERROR,
      318            "Failed to parse the locations due to: " + ex.getMessage());
      319      }

      321      try {
      322        encryptionMaterial = getEncryptionMaterial(commandType, jsonNode);
      323      } catch (Exception ex) {
      324        throw new SnowflakeSQLException(

      326            ErrorCode.INTERNAL_ERROR,
      327            "Failed to parse encryptionMaterial due to: " + ex.getMessage());
      328      }
      329  
      330      // For UPLOAD we expect encryptionMaterial to have length 1
      331      if (encryptionMaterial.size() != 1) {
      332        throw new SnowflakeSQLException(

      335            "Encryption material for UPLOAD should have size 1 but have "
