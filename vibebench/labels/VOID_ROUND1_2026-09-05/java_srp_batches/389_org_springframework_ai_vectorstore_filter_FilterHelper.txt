       44  	/**
       45  	 * Transforms the input expression into a semantically equivalent one with negation
       46  	 * operators propagated thought the expression tree by following the negation rules:
       47  	 *
       48  	 * <pre>
       49  	 * 	NOT(NOT(a)) = a
       50  	 *
       51  	 * 	NOT(a AND b) = NOT(a) OR NOT(b)
       52  	 * 	NOT(a OR b) = NOT(a) AND NOT(b)
       53  	 *
       54  	 * 	NOT(a EQ b) = a NE b
       55  	 * 	NOT(a NE b) = a EQ b
       56  	 *
       57  	 * 	NOT(a GT b) = a LTE b
       58  	 * 	NOT(a GTE b) = a LT b
       59  	 *
       60  	 * 	NOT(a LT b) = a GTE b
       61  	 * 	NOT(a LTE b) = a GT b
       62  	 *
       63  	 * 	NOT(a IN [...]) = a NIN [...]
       64  	 * 	NOT(a NIN [...]) = a IN [...]
       65  	 * </pre>
       66  	 * @param operand Filter expression to negate.
       67  	 * @return Returns an negation of the input expression.
       68  	 */
       69  	public static Filter.Operand negate(Filter.Operand operand) {
       70  
       71  		if (operand instanceof Filter.Group group) {
       72  			Operand inEx = negate(group.content());
       73  			if (inEx instanceof Filter.Group inEx2) {
       74  				inEx = inEx2.content();
       75  			}
       76  			return new Filter.Group((Expression) inEx);
       77  		}
       78  		else if (operand instanceof Filter.Expression exp) {
       79  			switch (exp.type()) {
       80  				case NOT: // NOT(NOT(a)) = a
       81  					return negate(exp.left());
       82  				case AND: // NOT(a AND b) = NOT(a) OR NOT(b)
       83  				case OR: // NOT(a OR b) = NOT(a) AND NOT(b)
       84  					return new Filter.Expression(TYPE_NEGATION_MAP.get(exp.type()), negate(exp.left()),
       85  							negate(exp.right()));
       86  				case EQ: // NOT(e EQ b) = e NE b
       87  				case NE: // NOT(e NE b) = e EQ b
       88  				case GT: // NOT(e GT b) = e LTE b
       89  				case GTE: // NOT(e GTE b) = e LT b
       90  				case LT: // NOT(e LT b) = e GTE b
       91  				case LTE: // NOT(e LTE b) = e GT b
       92  					return new Filter.Expression(TYPE_NEGATION_MAP.get(exp.type()), exp.left(), exp.right());
       93  				case IN: // NOT(e IN [...]) = e NIN [...]
       94  				case NIN: // NOT(e NIN [...]) = e IN [...]
       95  					return new Filter.Expression(TYPE_NEGATION_MAP.get(exp.type()), exp.left(), exp.right());
       96  				default:
       97  					throw new IllegalArgumentException("Unknown expression type: " + exp.type());
       98  			}
       99  		}
      100  		else {
      101  			throw new IllegalArgumentException("Can not negate operand of type: " + operand.getClass());
      102  		}
      103  	}

      106  	 * Expands the IN into a semantically equivalent boolean expressions of ORs of EQs.
      107  	 * Useful for providers that don't provide native IN support.
      108  	 *
      109  	 * For example the <pre>
      110  	 * foo IN ["bar1", "bar2", "bar3"]
      111  	 * </pre>
      112  	 *
      113  	 * expression is equivalent to
      114  	 *
      115  	 * <pre>
      116  	 * {@code foo == "bar1" || foo == "bar2" || foo == "bar3" (e.g. OR(foo EQ "bar1" OR(foo EQ "bar2" OR(foo EQ "bar3")))}
      117  	 * </pre>
      118  	 * @param exp input IN expression.
      119  	 * @param context Output native expression.
      120  	 * @param filterExpressionConverter {@link FilterExpressionConverter} used to compose
      121  	 * the OR and EQ expanded expressions.
      122  	 */
      123  	public static void expandIn(Expression exp, StringBuilder context,
      124  			FilterExpressionConverter filterExpressionConverter) {
      125  		Assert.isTrue(exp.type() == ExpressionType.IN, "Expected IN expressions but was: " + exp.type());
      126  		expandInNinExpressions(ExpressionType.OR, ExpressionType.EQ, exp, context, filterExpressionConverter);
      127  	}

      130  	 *
      131  	 * Expands the NIN (e.g. NOT IN) into a semantically equivalent boolean expressions of
      132  	 * ANDs of NEs. Useful for providers that don't provide native NIN support.<br/>
      133  	 *
      134  	 * For example the
      135  	 *
      136  	 * <pre>
      137  	 * foo NIN ["bar1", "bar2", "bar3"] (or foo NOT IN ["bar1", "bar2", "bar3"])
      138  	 * </pre>
      139  	 *
      140  	 * express is equivalent to
      141  	 *
      142  	 * <pre>
      143  	 * {@code foo != "bar1" && foo != "bar2" && foo != "bar3" (e.g. AND(foo NE "bar1" AND( foo NE "bar2" OR(foo NE "bar3"))) )}
      144  	 * </pre>
      145  	 * @param exp input NIN expression.
      146  	 * @param context Output native expression.
      147  	 * @param filterExpressionConverter {@link FilterExpressionConverter} used to compose
      148  	 * the AND and NE expanded expressions.
      149  	 */
      150  	public static void expandNin(Expression exp, StringBuilder context,
      151  			FilterExpressionConverter filterExpressionConverter) {
      152  		Assert.isTrue(exp.type() == ExpressionType.NIN, "Expected NIN expressions but was: " + exp.type());
      153  		expandInNinExpressions(ExpressionType.AND, ExpressionType.NE, exp, context, filterExpressionConverter);
      154  	}

      156  	private static void expandInNinExpressions(Filter.ExpressionType outerExpressionType,
      157  			Filter.ExpressionType innerExpressionType, Expression exp, StringBuilder context,
      158  			FilterExpressionConverter expressionConverter) {
      159  		if (exp.right() instanceof Filter.Value value) {
      160  			if (value.value() instanceof List list) {
      161  				// 1. foo IN ["bar1", "bar2", "bar3"] is equivalent to foo == "bar1" ||
      162  				// foo == "bar2" || foo == "bar3"
      163  				// or equivalent to OR(foo == "bar1" OR( foo == "bar2" OR(foo == "bar3")))
      164  				// 2. foo IN ["bar1", "bar2", "bar3"] is equivalent to foo != "bar1" &&
      165  				// foo != "bar2" && foo != "bar3"
      166  				// or equivalent to AND(foo != "bar1" AND( foo != "bar2" OR(foo !=
      167  				// "bar3")))
      168  				List<Filter.Expression> eqExprs = new ArrayList<>();
      169  				for (Object o : list) {
      170  					eqExprs.add(new Filter.Expression(innerExpressionType, exp.left(), new Filter.Value(o)));
      171  				}
      172  				context.append(expressionConverter.convertExpression(aggregate(outerExpressionType, eqExprs)));
      173  			}
      174  			else {
      175  				// 1. foo IN ["bar"] is equivalent to foo == "BAR"
      176  				// 2. foo NIN ["bar"] is equivalent to foo != "BAR"
      177  				context.append(expressionConverter
      178  					.convertExpression(new Filter.Expression(innerExpressionType, exp.left(), exp.right())));
      179  			}
      180  		}
      181  		else {
      182  			throw new IllegalStateException(
      183  					"Filter IN right expression should be of Filter.Value type but was " + exp.right().getClass());
      184  		}
      185  	}

      187  	/**
      188  	 * Recursively aggregates a list of expression into a binary tree with 'aggregateType'
      189  	 * join nodes.
      190  	 * @param aggregateType type all tree splits.
      191  	 * @param expressions list of expressions to aggregate.
      192  	 * @return Returns a binary tree expression.
      193  	 */
      194  	private static Filter.Expression aggregate(Filter.ExpressionType aggregateType,
      195  			List<Filter.Expression> expressions) {
      196  
      197  		if (expressions.size() == 1) {
      198  			return expressions.get(0);
      199  		}
      200  		return new Filter.Expression(aggregateType, expressions.get(0),
      201  				aggregate(aggregateType, expressions.subList(1, expressions.size())));
      202  	}
