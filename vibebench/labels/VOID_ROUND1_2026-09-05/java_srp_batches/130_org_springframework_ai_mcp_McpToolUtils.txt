       88

       90  			throw new IllegalArgumentException("Prefix or toolName cannot be null or empty");

      105  		return input;

      112  	public static String format(String input) {
      113  		// Replace any character that isn't alphanumeric, underscore, or hyphen with
      114  		// concatenation. Support Han script + CJK blocks for complete Chinese character
      115  		// coverage
      116  		String formatted = input
      117  			.replaceAll("[^\\p{IsHan}\\p{InCJK_Unified_Ideographs}\\p{InCJK_Compatibility_Ideographs}a-zA-Z0-9_-]", "");
      118  
      119  		return formatted.replaceAll("-", "_");
      120  	}

      122  	/**
      123  	 * Shortens a string by taking the first letter of each word separated by underscores
      124  	 * @param input String in format "Word1_Word2_Word3_server"
      125  	 * @return Shortened string with first letters in lowercase "w_w_w_s"
      126  	 */
      127  	private static String shorten(String input) {
      128  		if (input == null || input.isEmpty()) {
      129  			return "";
      130  		}
      131  
      132  		return Stream.of(input.toLowerCase().split("_"))
      133  			.filter(word -> !word.isEmpty())
      134  			.map(word -> String.valueOf(word.charAt(0)))
      135  			.collect(java.util.stream.Collectors.joining("_"));
      136  	}

      252  	private static SharedSyncToolSpecification toSharedSyncToolSpecification(ToolCallback toolCallback,
      253  			@Nullable MimeType mimeType) {
      254  
      255  		var toolBuilder = McpSchema.Tool.builder()
      256  			.name(toolCallback.getToolDefinition().name())
      257  			.description(toolCallback.getToolDefinition().description())
      258  			.inputSchema(ModelOptionsUtils.jsonToObject(toolCallback.getToolDefinition().inputSchema(),
      259  					McpSchema.JsonSchema.class));
      260  
      261  		// Propagate returnDirect from ToolMetadata as MCP ToolAnnotations (MCP
      262  		// 2025-03-26)
      263  		if (toolCallback.getToolMetadata().returnDirect()) {
      264  			toolBuilder.annotations(McpSchema.ToolAnnotations.builder().returnDirect(true).build());
      265  		}
      266  
      267  		var tool = toolBuilder.build();
      268  
      269  		return new SharedSyncToolSpecification(tool, (exchangeOrContext, request) -> {
      270  			try {
      271  				String callResult = toolCallback.call(ModelOptionsUtils.toJsonString(request.arguments()),
      272  						new ToolContext(Map.of(TOOL_CONTEXT_MCP_EXCHANGE_KEY, exchangeOrContext)));
      273  				if (mimeType != null) {
      274  					McpSchema.Annotations annotations = new McpSchema.Annotations(List.of(Role.ASSISTANT), null);
      275  					String mimeTypeStr = mimeType.toString();
      276  					if (mimeTypeStr.startsWith("image")) {
      277  						return McpSchema.CallToolResult.builder()
      278  							.content(
      279  									List.of(new McpSchema.ImageContent(annotations, callResult, mimeTypeStr)))
      280  							.isError(false)
      281  							.build();
      282  					}
      283  					if (mimeTypeStr.startsWith("audio")) {
      284  						return McpSchema.CallToolResult.builder()
      285  							.content(
      286  									List.of(new McpSchema.AudioContent(annotations, callResult, mimeTypeStr)))
      287  							.isError(false)
      288  							.build();
      289  					}
      290  				}
      291  				return McpSchema.CallToolResult.builder()
      292  					.content(List.of(new McpSchema.TextContent(callResult)))
      293  					.isError(false)
      294  					.build();
      295  			}
      296  			catch (Exception e) {
      297  				return McpSchema.CallToolResult.builder()
      298  					.content(List.of(new McpSchema.TextContent(e.getMessage())))
      299  					.isError(true)
      300  					.build();
      301  			}
      302  		});
      303  	}
