      691          /**
      692           * Generic helper to check varargs parameters with a custom type checking strategy.
      693           * For varargs, the last declared type is the element type of the varargs array.
      694           * All fixed parameters are checked normally, and the varargs parameter is validated
      695           * to be an array of the expected element type.
      696           */
      697          private void checkVarargsParametersGeneric(Method method, Type[] declaredArgTypes, 
      698                                                     int paramOffset, TypeChecker typeChecker) {
      699              Parameter[] params = method.getParameters();
      700              
      701              // All parameters except the last should match the declared types
      702              for (int i = 0; i < declaredArgTypes.length - 1; i++) {
      703                  Parameter param = params[i + paramOffset];
      704                  typeChecker.check(method, declaredArgTypes[i], param.getType(), param.getName());
      705              }
      706              
      707              // The last parameter should be a varargs array
      708              if (declaredArgTypes.length > 0) {
      709                  Type varargsElementType = declaredArgTypes[declaredArgTypes.length - 1];
      710                  Parameter varargsParam = params[paramOffset + declaredArgTypes.length - 1];
      711                  
      712                  // Java varargs are represented as arrays
      713                  if (!varargsParam.getType().isArray()) {
      714                      ErrorReport.reportSemanticException(ErrorCode.ERR_COMMON_ERROR,
      715                              String.format("UDF class '%s' method '%s' varargs parameter '%s' must be an array",
      716                                      clazz.getCanonicalName(), method.getName(), varargsParam.getName()));
      717                  }
      718                  
      719                  // Check that the array component type matches the declared varargs element type
      720                  Class<?> arrayComponentType = varargsParam.getType().getComponentType();
      721                  typeChecker.check(method, varargsElementType, arrayComponentType, varargsParam.getName());
      722              }
      723          }
