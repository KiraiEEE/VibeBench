      132      public void buildJavafuncMap (LuaTable environment) {
      133          LuaTable[] packages = new LuaTable[] {
      134              environment,
      135              (LuaTable)environment.rawget("string"),

      141          for (int i = 0; i < packages.length; i++) {

      180      private void storeObject (Object obj, DataOutputStream out)
      181      throws IOException {
      182          if (obj == null) {
      183              out.writeByte(LUA_NIL);
      184              return;
      185          }
      186          Integer i = (Integer) objectStore.get(obj);
      187          if (i != null) {
      188              out.writeByte(LUA_REFERENCE);
      189               if (debug) debug("reference " + i + " (" + obj.toString() + ")");
      190              out.writeInt(i);
      191          } else {
      192              i = Integer.valueOf(currentId++);
      193              objectStore.put(obj, i);
      194              if (debug) debug("(ref" + i + ")");
      195              if (obj instanceof Serializable) {
      196                  out.writeByte(LUA_OBJECT);
      197                  out.writeUTF(obj.getClass().getName());
      198                  if (debug) debug(obj.getClass().getName() + " (" + obj.toString()+")");
      199                  ((Serializable)obj).serialize(out);
      200              } else if (obj instanceof LuaTable) {
      201                  out.writeByte(LUA_TABLE);
      202                  if (debug) debug("table("+obj.toString()+"):\n");
      203                  serializeLuaTable((LuaTable)obj, out);
      204              } else if (obj instanceof LuaClosure) {
      205                  out.writeByte(LUA_CLOSURE);
      206                  if (debug) debug("closure("+obj.toString()+")");
      207                  serializeLuaClosure((LuaClosure)obj, out);
      208              } else {
      209                  // we're busted
      210                  out.writeByte(LUA_NIL);
      211                  if (debug) debug("UFO");
      212                  Engine.log("STOR: unable to store object of type "+obj.getClass().getName(), Engine.LOG_WARN);
      213              }
      214          }
      215      }

      217      public void storeValue (Object obj, DataOutputStream out)
      218      throws IOException {
      219          if (obj == null) {
      220              if (debug) debug("nil");
      221              out.writeByte(LUA_NIL);
      222          } else if (obj instanceof String) {
      223              out.writeByte(LUA_STRING);
      224              if (debug) debug("\""+obj.toString()+"\"");
      225              out.writeUTF((String)obj);
      226          } else if (obj instanceof Boolean) {
      227              if (debug) debug(obj.toString());
      228              out.writeByte(LUA_BOOLEAN);
      229              out.writeBoolean(((Boolean)obj).booleanValue());
      230          } else if (obj instanceof Double) {
      231              out.writeByte(LUA_DOUBLE);
      232              if (debug) debug(obj.toString());
      233              out.writeDouble(((Double)obj).doubleValue());
      234          } else if (obj instanceof JavaFunction) {
      235              int i = findJavafuncId((JavaFunction)obj);
      236              if (debug) debug("javafunc("+i+")-"+obj.toString());
      237              out.writeByte(LUA_JAVAFUNC);
      238              out.writeInt(i);
      239          } else {
      240              storeObject(obj, out);
      241          }
      242      }

      245      throws IOException {
      246          level++;
      247          Object next = null;
      248          while ((next = table.next(next)) != null) {
      249              Object value = table.rawget(next);
      250              out.writeByte(LUATABLE_PAIR);
      251              if (debug) for (int i = 0; i < level; i++) debug("  ");
      252

      254              if (debug) debug(" : ");

      256              if (debug) debug("\n");
      257          }

      262      public Object restoreValue (DataInputStream in, Object target)
      263      throws IOException {
      264          byte type = in.readByte();
      265          switch (type) {
      266              case LUA_NIL:
      267                  if (debug) debug("nil");
      268                  return null;
      269              case LUA_DOUBLE:
      270                  double d = in.readDouble();
      271                  if (debug) debug(String.valueOf(d));
      272                  return LuaState.toDouble(d);
      273              case LUA_STRING:
      274                  String s = in.readUTF();
      275                  if (debug) debug("\"" + s + "\"");
      276                  return s;
      277              case LUA_BOOLEAN:
      278                  boolean b = in.readBoolean();
      279                  if (debug) debug(String.valueOf(b));
      280                  return LuaState.toBoolean(b);
      281              case LUA_JAVAFUNC:
      282                  int i = in.readInt();
      283                  JavaFunction jf = findJavafuncObject(i);
      284                  if (debug) debug("javafunc("+i+")-"+jf);
      285                  return jf;
      286              default:
      287                  return restoreObject(in, type, target);
      288          }
      289      }

      297      private Object restoreObject (DataInputStream in, byte type, Object target)
      298      throws IOException {
      299          switch (type) {
      300              case LUA_TABLE:
      301                  LuaTable lti;
      302                  if (target instanceof LuaTable)
      303                      lti = (LuaTable)target;
      304                  else
      305                      lti = new LuaTableImpl();
      306                  restCache(lti);
      307                  if (debug) debug("table:\n");
      308                  return deserializeLuaTable(in, lti);
      309              case LUA_CLOSURE:
      310                  if (debug) debug("closure: ");
      311                  LuaClosure lc = deserializeLuaClosure(in);
      312                  if (debug) debug(lc.toString());
      313                  return lc;
      314              case LUA_OBJECT:
      315                  String cls = in.readUTF();
      316                  Serializable s = null;
      317                  try {
      318                      if (debug) debug("object of type "+cls+"...\n");
      319                      Class c = classForName(cls);
      320                      if (Serializable.class.isAssignableFrom(c)) {
      321                          s = (Serializable)c.newInstance();
      322                      }
      323                  } catch (Throwable e) {
      324                      if (debug) debug("(failed to deserialize "+cls+")\n");
      325                      Engine.log("REST: while trying to deserialize "+cls+":\n"+e.toString(), Engine.LOG_ERROR);
      326                      BaseLib.fail("Could not deserialize object of class '" + cls + "': " + e);
      327                  }
      328                  if (s != null) {
      329                      restCache(s);
      330                      s.deserialize(in);
      331                  }
      332                  return s;
      333              case LUA_REFERENCE:
      334                  final Integer what = Integer.valueOf(in.readInt());
      335                  if (debug) debug("reference " + what);
      336                  final Object result = objectStore.get(what);
      337                  if (result == null) {
      338                      Engine.log("REST: not found reference " + what + " in object store", Engine.LOG_WARN);
      339                      if (debug) debug(" (which happens to be null?)");
      340                      return target;
      341                  } else {
      342                      if (debug) debug(" : " + result);
      343                  }
      344                  return result;
      345              default:
      346                  Engine.log("REST: found unknown type "+type, Engine.LOG_WARN);
      347                  if (debug) debug("UFO");
      348                  return null;
      349          }
      350      }

      354      public LuaTable deserializeLuaTable (DataInputStream in, LuaTable table)
      355      throws IOException {
      356          level++;
      357          while (true) {
      358              byte next = in.readByte();
      359              if (next == LUATABLE_END) break;
      360              if (debug) for (int i = 0; i < level; i++) debug("  ");
      361              Object key = restoreValue(in, null);
      362              if (debug) debug(" : ");
      363              Object value = restoreValue(in, table.rawget(key));
      364              if (debug) debug("\n");
      365              table.rawset(key, value);
      366          }
      367          level--;
      368          return table;
      369      }

      371      private void serializeLuaClosure (LuaClosure closure, DataOutputStream out)
      372      throws IOException {
      373          closure.prototype.dump(out);
      374          for (int i = 0; i < closure.upvalues.length; i++) {
      375              UpValue u = closure.upvalues[i];
      376              if (u.value == null) {
      377                  Engine.log("STOR: unclosed upvalue in "+closure.toString(), Engine.LOG_WARN);
      378                  u.value = u.thread.objectStack[u.index];
      379              }
      380              storeValue(u.value, out);
      381          }
      382      }

      384      private LuaClosure deserializeLuaClosure (DataInputStream in)
      385      throws IOException {
      386          LuaClosure closure = LuaPrototype.loadByteCode(in, Engine.state.getEnvironment());
      387          restCache(closure);
      388          for (int i = 0; i < closure.upvalues.length; i++) {
      389              UpValue u = new UpValue();
      390              u.value = restoreValue(in, null);
      391              closure.upvalues[i] = u;
      392          }
      393          return closure;
      394      }
