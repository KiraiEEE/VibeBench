      286       * This is the key fix that solves the mixed dependency problem.

      291      private void establishOperationDependencies(Map<String, ExecutionNode> operationNodes) {
      292          for (ExecutionNode node : operationNodes.values()) {
      293              Set<String> dependencies = new HashSet<>();
      294              
      295              // For each input variable, find the operation that produces it
      296              for (String inputVar : node.getInputVariables()) {
      297                  String producerOpName = findProducerOperation(inputVar);
      298                  
      299                  if (producerOpName != null && operationNodes.containsKey(producerOpName)) {
      300                      dependencies.add(producerOpName);
      301                      log.trace("Operation {} depends on operation {} (via variable {})", 
      302                              node.getOperationName(), producerOpName, inputVar);
      303                  }
      304              }
      305              
      306              // Add control dependencies
      307              SameDiffOp sameDiffOp = sameDiff.getOps().get(node.getOperationName());
      308              if (sameDiffOp != null && sameDiffOp.getControlDeps() != null) {
      309                  for (String controlDep : sameDiffOp.getControlDeps()) {
      310                      if (operationNodes.containsKey(controlDep)) {
      311                          dependencies.add(controlDep);
      312                          log.trace("Operation {} has control dependency on operation {}", 
      313                                  node.getOperationName(), controlDep);
      314                      }
      315                  }
      316              }
      317              
      318              // Handle special cases for control flow operations
      319              addControlFlowDependencies(node, dependencies, operationNodes);
      320              
      321              node.getDependsOnOperations().addAll(dependencies);
      322          }
      323      }

      342              // Enter creates new frame context

      346              // Exit returns to parent frame

      348          } else if (op instanceof NextIteration) {
      349              // NextIteration advances iteration counter

      351          } else if (op instanceof LoopCond) {

      361      private List<ExecutionNode> createTopologicalOrder(Map<String, ExecutionNode> operationNodes) {
      362          List<ExecutionNode> result = new ArrayList<>();
      363          Set<String> visited = new HashSet<>();
      364          Set<String> visiting = new HashSet<>();
      365          
      366          // Start with nodes that have no dependencies (constants, variables, placeholders)
      367          for (ExecutionNode node : operationNodes.values()) {
      368              if (node.getDependsOnOperations().isEmpty()) {
      369                  topologicalSort(node, operationNodes, visited, visiting, result);
      370              }
      371          }
      372          
      373          // Then process remaining nodes
      374          for (ExecutionNode node : operationNodes.values()) {
      375              if (!visited.contains(node.getOperationName())) {
      376                  topologicalSort(node, operationNodes, visited, visiting, result);
      377              }
      378          }
      379          
      380          log.info("Created topological execution order with {} nodes", result.size());
      381          return result;
      382      }

      385       * Recursive topological sort implementation
      386       */
      387      private void topologicalSort(ExecutionNode node, Map<String, ExecutionNode> allNodes,
      388                                 Set<String> visited, Set<String> visiting, List<ExecutionNode> result) {
      389          
      390          String nodeName = node.getOperationName();
      391          
      392          if (visiting.contains(nodeName)) {
      393              log.warn("Cycle detected involving node: {}", nodeName);
      394              return;
      395          }
      396          
      397          if (visited.contains(nodeName)) {
      398              return;
      399          }
      400          
      401          visiting.add(nodeName);
      402          
      403          // Visit all dependencies first
      404          for (String depName : node.getDependsOnOperations()) {
      405              ExecutionNode depNode = allNodes.get(depName);
      406              if (depNode != null) {
      407                  topologicalSort(depNode, allNodes, visited, visiting, result);
      408              }
      409          }
      410          
      411          visiting.remove(nodeName);
      412          visited.add(nodeName);
      413          result.add(node);
      414      }
