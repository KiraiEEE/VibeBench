      347       *            is to be excluded

      353      public static <T> String join(CharSequence delimiter, Iterable<T> values, Function<T, String> conversion) {
      354          Objects.requireNonNull(delimiter);
      355          Objects.requireNonNull(values);
      356          Iterator<T> iter = values.iterator();
      357          if (!iter.hasNext()) {
      358              return "";
      359          }
      360          StringBuilder sb = new StringBuilder();
      361          String first = conversion.apply(iter.next());
      362          boolean delimit = false;
      363          if (first != null) {
      364              sb.append(first);
      365              delimit = true;
      366          }
      367          while (iter.hasNext()) {
      368              String next = conversion.apply(iter.next());
      369              if (next != null) {
      370                  if (delimit) {
      371                      sb.append(delimiter);
      372                  }
      373                  sb.append(next);
      374                  delimit = true;
      375              }
      376          }
      377          return sb.toString();
      378      }

      396       * @param str the string to be trimmed; may not be null

      398       * @return the trimmed string; never null
      399       * @see #trim(String)

      401      public static String trim(String str, CharacterPredicate predicate) {
      402          int len = str.length();
      403          if (len == 0) {
      404              return str;
      405          }
      406          int st = 0;
      407          while ((st < len) && predicate.test(str.charAt(st))) {
      408              st++;
      409          }
      410          while ((st < len) && predicate.test(str.charAt(len - 1))) {
      411              len--;
      412          }
      413          return ((st > 0) || (len < str.length())) ? str.substring(st, len) : str;
      414      }
