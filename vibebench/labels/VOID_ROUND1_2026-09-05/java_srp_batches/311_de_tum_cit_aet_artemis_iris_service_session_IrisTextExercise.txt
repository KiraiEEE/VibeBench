       92      public void requestAndHandleResponse(IrisTextExerciseChatSession irisSession) {
       93          var session = (IrisTextExerciseChatSession) irisSessionRepository.findByIdWithMessagesAndContents(irisSession.getId());
       94          var exercise = textRepositoryApi.orElseThrow(() -> new TextApiNotPresentException(TextApi.class)).findByIdElseThrow(session.getExerciseId());
       95          if (exercise.isExamExercise()) {
       96              throw new ConflictException("Iris is not supported for exam exercises", "Iris", "irisExamExercise");
       97          }
       98          var course = exercise.getCourseViaExerciseGroupOrCourseMember();
       99          var settings = irisSettingsService.getSettingsForCourse(course);
      100          if (!settings.enabled()) {
      101              throw new ConflictException("Iris is not enabled for this exercise", "Iris", "irisDisabled");
      102          }
      103          var user = userRepository.findByIdElseThrow(session.getUserId());
      104          // TODO: Once we can receive client form data through the IrisMessageResource, we should use that instead of fetching the latest submission to get the text
      105          var participation = studentParticipationRepository.findWithEagerSubmissionsByExerciseIdAndStudentLogin(exercise.getId(), user.getLogin());
      106          var latestSubmission = participation.flatMap(p -> p.getSubmissions().stream().max(Comparator.comparingLong(Submission::getId))).orElse(null);
      107          String latestSubmissionText;
      108          if (latestSubmission instanceof TextSubmission textSubmission) {
      109              latestSubmissionText = textSubmission.getText();
      110          }
      111          else {
      112              latestSubmissionText = null;
      113          }
      114          var chatHistory = session.getMessages().stream().map(PyrisMessageDTO::of).toList();
      115          // @formatter:off
      116          pyrisPipelineService.executePipeline(
      117                  "text-exercise-chat",
      118                  user.getSelectedLLMUsage(),
      119                  settings.variant().jsonValue(),
      120                  Optional.empty(),
      121                  pyrisJobService.createTokenForJob(token -> new TextExerciseChatJob(token, course.getId(), exercise.getId(), session.getId())),
      122                  dto -> new PyrisTextExerciseChatPipelineExecutionDTO(PyrisTextExerciseDTO.of(exercise), session.getTitle(), chatHistory, new PyrisUserDTO(user), latestSubmissionText, dto.settings(), dto.initialStages(), settings.customInstructions()),
      123                  stages -> irisChatWebsocketService.sendMessage(session, null, stages)
      124          );

      134      public TextExerciseChatJob handleStatusUpdate(TextExerciseChatJob job, PyrisTextExerciseChatStatusUpdateDTO statusUpdate) {
      135          // TODO: LLM Token Tracking - or better, make this class a subclass of AbstractIrisChatSessionService
      136          var session = (IrisTextExerciseChatSession) irisSessionRepository.findByIdElseThrow(job.sessionId());
      137          String sessionTitle = AbstractIrisChatSessionService.setSessionTitle(session, statusUpdate.sessionTitle(), irisSessionRepository);
      138          if (statusUpdate.result() != null) {
      139              var message = session.newMessage();
      140              message.addContent(new IrisTextMessageContent(statusUpdate.result()));
      141              IrisMessage savedMessage = irisMessageService.saveMessage(message, session, IrisMessageSender.LLM);
      142              irisChatWebsocketService.sendMessage(session, savedMessage, statusUpdate.stages(), sessionTitle);
      143          }
      144          else {
      145              irisChatWebsocketService.sendMessage(session, null, statusUpdate.stages(), sessionTitle);
      146          }
      147  
      148          return job;
      149      }

      159      public void checkHasAccessTo(User user, IrisTextExerciseChatSession session) {
      160          // TODO: This check is probably unnecessary since we are fetching the sessions from the database with the user ID already
      161          if (session.getUserId() != user.getId()) {
      162              throw new AccessForbiddenException("Iris Text Exercise Chat Session", session.getId());
      163          }
      164          var exercise = textRepositoryApi.orElseThrow(() -> new TextApiNotPresentException(TextApi.class)).findByIdElseThrow(session.getExerciseId());
      165          // TODO: This check is probably unnecessary as the endpoint already checks it via the @EnforceAtLeastStudentInExercise annotation
      166          authCheckService.checkHasAtLeastRoleForExerciseElseThrow(Role.STUDENT, exercise, user);
      167      }
