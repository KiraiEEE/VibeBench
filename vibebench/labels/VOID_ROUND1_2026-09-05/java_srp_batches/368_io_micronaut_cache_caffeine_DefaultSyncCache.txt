      190      protected com.github.benmanes.caffeine.cache.Cache buildCache(CacheConfiguration cacheConfiguration) {
      191          Caffeine<Object, Object> builder = Caffeine.newBuilder();
      192          cacheConfiguration.getExpireAfterAccess().ifPresent(duration -> builder.expireAfterAccess(duration.toMillis(), TimeUnit.MILLISECONDS));
      193          cacheConfiguration.getExpireAfterWrite().ifPresent(duration -> builder.expireAfterWrite(duration.toMillis(), TimeUnit.MILLISECONDS));
      194          cacheConfiguration.getInitialCapacity().ifPresent(builder::initialCapacity);
      195          cacheConfiguration.getMaximumSize().ifPresent(builder::maximumSize);
      196          cacheConfiguration.getMaximumWeight().ifPresent(weight -> {
      197              builder.maximumWeight(weight);
      198              builder.weigher(findWeigher());
      199          });
      200          CaffeineCacheConfiguration caffeineCacheConfiguration = cacheConfiguration instanceof CaffeineCacheConfiguration ? (CaffeineCacheConfiguration) cacheConfiguration : null;
      201          if (caffeineCacheConfiguration != null) {
      202              RemovalListener removalListener = findRemovalListener();
      203              if (removalListener != null) {
      204                  if (caffeineCacheConfiguration.isListenToRemovals()) {
      205                      builder.removalListener((key, value, cause) -> removalListener.onRemoval(key, value, cause));
      206                  }
      207                  if (caffeineCacheConfiguration.isListenToEvictions()) {
      208                      builder.evictionListener((key, value, cause) -> removalListener.onRemoval(key, value, cause));
      209                  }
      210              }
      211          }
      212          if (cacheConfiguration.isRecordStats()) {
      213              builder.recordStats();
      214          }
      215          if (cacheConfiguration.isTestMode()) {
      216              // run commands on same thread
      217              builder.executor(Runnable::run);
      218          }
      219          return builder.build();
      220      }

      236      private Map<String, Object> getCaffeineCacheData(Cache caffeineCache) {
      237  
      238          Policy policy = caffeineCache.policy();
      239          Optional<Policy.Eviction> eviction = policy.eviction();
      240          Optional<Policy.FixedExpiration> expireAfterAccess = policy.expireAfterAccess();
      241          Optional<Policy.FixedExpiration> expireAfterWrite = policy.expireAfterWrite();
      242  
      243          Long maximumSize = eviction.filter(e -> !e.isWeighted()).map(e -> e.getMaximum()).orElse(null);
      244          Long maximumWeight = eviction.filter(e -> e.isWeighted()).map(e -> e.getMaximum()).orElse(null);
      245          Long weightedSize = eviction.flatMap(e -> e.weightedSize().isPresent() ? Optional.of(e.weightedSize().getAsLong()) : Optional.empty()).orElse(null);
      246          boolean isRecordingStats = policy.isRecordingStats();
      247  
      248          Map<String, Object> values = new LinkedHashMap<>(8);
      249  
      250          values.put("estimatedSize", caffeineCache.estimatedSize());
      251          values.put("maximumSize", maximumSize);
      252          values.put("maximumWeight", maximumWeight);
      253          values.put("weightedSize", weightedSize);
      254          values.put("expireAfterAccess", getExpiresAfter(expireAfterAccess));
      255          values.put("expireAfterWrite", getExpiresAfter(expireAfterWrite));
      256          values.put("recordingStats", isRecordingStats);
      257  
      258          if (isRecordingStats) {
      259              values.put("stats", getStatsData(caffeineCache.stats()));
      260          }
      261  
      262          return values;
      263      }
