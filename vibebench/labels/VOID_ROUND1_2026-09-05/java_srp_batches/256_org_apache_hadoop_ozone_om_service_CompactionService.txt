       77    private List<String> validateTables(List<String> tables) {
       78      if (tables == null || tables.isEmpty()) {
       79        return Collections.emptyList();
       80      }
       81      List<String> validTables = new ArrayList<>();
       82      Set<String> allTableNames = new HashSet<>(omMetadataManager.listTableNames());
       83      for (String table : tables) {
       84        if (allTableNames.contains(table)) {
       85          validTables.add(table);
       86        } else {
       87          LOG.warn("CompactionService: Table \"{}\" not found in OM metadata. Skipping this table.", table);
       88        }
       89      }
       90      if (validTables.isEmpty()) {
       91        LOG.error("CompactionService: No valid compaction tables found. Failing initialization.");
       92        throw new IllegalArgumentException("CompactionService: None of the provided tables are valid.");
       93      }
       94      return Collections.unmodifiableList(validTables);
       95    }

      141    protected void compactFully(String tableName) throws IOException {
      142      long startTime = Time.monotonicNow();
      143      LOG.info("Compacting column family: {}", tableName);
      144      try (ManagedCompactRangeOptions options = new ManagedCompactRangeOptions()) {
      145        options.setBottommostLevelCompaction(ManagedCompactRangeOptions.BottommostLevelCompaction.kForce);
      146        options.setExclusiveManualCompaction(true);
      147        options.setMaxSubcompactions(manualCompactionMaxSubCompactions);
      148        RocksDatabase rocksDatabase = ((RDBStore) omMetadataManager.getStore()).getDb();
      149  
      150        try {
      151          // Find CF Handler
      152          RocksDatabase.ColumnFamily columnFamily = rocksDatabase.getColumnFamily(tableName);
      153          rocksDatabase.compactRange(columnFamily, null, null, options);
      154          LOG.info("Compaction of column family: {} completed in {} ms",
      155              tableName, Time.monotonicNow() - startTime);
      156        } catch (NullPointerException ex) {
      157          LOG.error("Unable to trigger compaction for \"{}\". Column family not found ", tableName);
      158          throw new IOException("Column family \"" + tableName + "\" not found.");
      159        }
      160      }
      161    }
