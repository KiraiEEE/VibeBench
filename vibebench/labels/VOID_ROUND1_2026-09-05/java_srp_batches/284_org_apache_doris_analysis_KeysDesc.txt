       68      public void analyze(List<ColumnDef> cols) throws AnalysisException {
       69          if (type == null) {
       70              throw new AnalysisException("Keys type is null.");
       71          }
       72  
       73          if ((keysColumnNames == null || keysColumnNames.size() == 0) && type != KeysType.DUP_KEYS) {
       74              throw new AnalysisException("The number of key columns is 0.");
       75          }
       76  
       77          if (keysColumnNames.size() > cols.size()) {

       79          }
       80  
       81          for (int i = 0; i < keysColumnNames.size(); ++i) {
       82              String name = cols.get(i).getName();

      119      private void analyzeClusterKeys(List<ColumnDef> cols) throws AnalysisException {
      120          if (type != KeysType.UNIQUE_KEYS) {
      121              throw new AnalysisException("Cluster keys only support unique keys table");
      122          }
      123          // check that cluster keys is not duplicated
      124          for (int i = 0; i < clusterKeysColumnNames.size(); i++) {
      125              String name = clusterKeysColumnNames.get(i);
      126              for (int j = 0; j < i; j++) {
      127                  if (clusterKeysColumnNames.get(j).equalsIgnoreCase(name)) {
      128                      throw new AnalysisException("Duplicate cluster key column[" + name + "].");
      129                  }
      130              }
      131          }
      132          // check that cluster keys is not equal to primary keys
      133          int minKeySize = Math.min(keysColumnNames.size(), clusterKeysColumnNames.size());
      134          boolean sameKey = true;
      135          for (int i = 0; i < minKeySize; i++) {
      136              if (!keysColumnNames.get(i).equalsIgnoreCase(clusterKeysColumnNames.get(i))) {
      137                  sameKey = false;
      138                  break;
      139              }
      140          }
      141          if (sameKey) {
      142              throw new AnalysisException("Unique keys and cluster keys should be different.");
      143          }
      144          // check that cluster key column exists
      145          for (int i = 0; i < clusterKeysColumnNames.size(); i++) {
      146              String name = clusterKeysColumnNames.get(i);
      147              for (int j = 0; j < cols.size(); j++) {
      148                  if (cols.get(j).getName().equalsIgnoreCase(name)) {
      149                      cols.get(j).setClusterKeyId(i);
      150                      break;
      151                  }
      152                  if (j == cols.size() - 1) {
      153                      throw new AnalysisException("Cluster key column[" + name + "] doesn't exist.");
      154                  }
      155              }
      156          }
      157      }
