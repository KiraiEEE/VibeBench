      131      if (ProviderType.SYSTEM.equals(template.getProvider())) {
      132        throw new IllegalArgumentException(
      133            String.format(

      135                template.getName()));
      136      }

      138      // Nullify stale references in EventSubscription JSON blobs before deletion
      139      // This ensures ListFilter queries on JSON don't return stale data
      140      List<EntityReference> subscriptions =
      141          findFrom(
      142              template.getId(),
      143              Entity.NOTIFICATION_TEMPLATE,
      144              Relationship.USES,
      145              Entity.EVENT_SUBSCRIPTION);
      146  
      147      EventSubscriptionRepository subscriptionRepository =
      148          (EventSubscriptionRepository) Entity.getEntityRepository(Entity.EVENT_SUBSCRIPTION);
      149  
      150      for (EntityReference subRef : subscriptions) {
      151        try {
      152          EventSubscription original =
      153              Entity.getEntity(Entity.EVENT_SUBSCRIPTION, subRef.getId(), "*", Include.ALL);
      154  
      155          EventSubscription updated = JsonUtils.deepCopy(original, EventSubscription.class);
      156          updated.setNotificationTemplate(null);
      157  
      158          EntityRepository<EventSubscription>.EntityUpdater updater =
      159              subscriptionRepository.getUpdater(original, updated, Operation.PUT, null);
      160          updater.update();
      161  
      162          LOG.debug(
      163              "Nullified template reference in subscription {} before deleting template {}",
      164              original.getId(),
      165              template.getId());
      166        } catch (Exception e) {
      167          LOG.warn(
      168              "Failed to nullify template reference in subscription {}: {}",

      222    private boolean shouldUpdateSystemTemplate(
      223        NotificationTemplate existing, NotificationTemplate seedTemplate) {
      224      // Update only if system template has not been update and seed content differs from current
      225      // content
      226      return ProviderType.SYSTEM.equals(existing.getProvider())
      227          && Boolean.FALSE.equals(existing.getIsModifiedFromDefault())
      228          && !Objects.equals(
      229              calculateTemplateChecksum(seedTemplate), calculateTemplateChecksum(existing));
      230    }

      247    public void resetToDefault(NotificationTemplate original) {
      248      try {
      249        if (!ProviderType.SYSTEM.equals(original.getProvider())) {
      250          return;
      251        }
      252  
      253        NotificationTemplate defaultTemplate = getDefaultTemplateFromSeed(original.getName());
      254        NotificationTemplate updated = JsonUtils.deepCopy(original, NotificationTemplate.class);
      255        updated
      256            .withTemplateBody(defaultTemplate.getTemplateBody())
      257            .withTemplateSubject(defaultTemplate.getTemplateSubject())
      258            .withDescription(defaultTemplate.getDescription())
      259            .withDisplayName(defaultTemplate.getDisplayName());
      260  
      261        EntityUpdater entityUpdater = getUpdater(original, updated, Operation.PUT, null);
      262        entityUpdater.update();
      263  
      264        LOG.info("Reset NotificationTemplate {} to default", original.getName());
      265      } catch (IllegalArgumentException e) {
      266        LOG.error("Failed to reset template: {}", e.getMessage(), e);
      267        throw e;
      268      } catch (IOException e) {
      269        LOG.error("Failed to load seed data for reset operation", e);
      270        throw new RuntimeException("Failed to reset template due to seed data loading error", e);
      271      }
      272    }

      425    private String calculateTemplateChecksum(NotificationTemplate template) {
      426      // Only include functional fields in checksum
      427      // This allows users to customize displayName and description without triggering "modified"
      428      // status
      429      String content =
      430          String.join(
      431              "|",
      432              StringUtils.defaultString(template.getTemplateBody()),
      433              StringUtils.defaultString(template.getTemplateSubject()));
      434      return EntityUtil.hash(content);
      435    }

      450        NotificationTemplate template) {

      452      NotificationMessage message =
      453          messageEngine.generateMessageWithTemplate(event, subscription, destination, template);
      454  
      455      switch (destination.getType()) {
      456        case EMAIL -> sendEmailNotification((EmailMessage) message, destination);
      457        case SLACK, MS_TEAMS, G_CHAT, WEBHOOK -> sendWebhookNotification(message, destination);
