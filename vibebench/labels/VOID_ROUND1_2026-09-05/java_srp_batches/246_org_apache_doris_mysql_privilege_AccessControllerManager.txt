       79      private CatalogAccessController loadAccessControllerOrThrow(String accessControllerName) {
       80          if (accessControllerName.equalsIgnoreCase("default")) {
       81              return new InternalAccessController(auth);
       82          }
       83          if (accessControllerFactoriesCache.containsKey(accessControllerName)) {
       84              Map<String, String> prop;
       85              try {
       86                  prop = PropertiesUtils.loadAccessControllerPropertiesOrNull();
       87              } catch (IOException e) {
       88                  throw new RuntimeException("Failed to load authorization properties."
       89                          + "Please check the configuration file, authorization name is " + accessControllerName, e);
       90              }
       91              return accessControllerFactoriesCache.get(accessControllerName).createAccessController(prop);
       92          }
       93          throw new RuntimeException("No authorization plugin factory found for " + accessControllerName
       94                  + ". Please confirm that your plugin is placed in the correct location.");
       95      }

       97      private void loadAccessControllerPlugins() {
       98          ServiceLoader<AccessControllerFactory> loaderFromClasspath = ServiceLoader.load(AccessControllerFactory.class);
       99          for (AccessControllerFactory factory : loaderFromClasspath) {
      100              LOG.info("Found Authentication Plugin Factories: {} from class path.", factory.factoryIdentifier());
      101              accessControllerFactoriesCache.put(factory.factoryIdentifier(), factory);
      102              accessControllerClassNameMapping.put(factory.getClass().getName(), factory.factoryIdentifier());
      103          }
      104          List<AccessControllerFactory> loader = null;
      105          try {
      106              loader = ClassLoaderUtils.loadServicesFromDirectory(AccessControllerFactory.class);
      107          } catch (IOException e) {
      108              throw new RuntimeException("Failed to load Authentication Plugin Factories", e);
      109          }
      110          for (AccessControllerFactory factory : loader) {
      111              LOG.info("Found Access Controller Plugin Factory: {} from directory.", factory.factoryIdentifier());
      112              accessControllerFactoriesCache.put(factory.factoryIdentifier(), factory);
      113              accessControllerClassNameMapping.put(factory.getClass().getName(), factory.factoryIdentifier());
      114          }
      115      }

      117      public CatalogAccessController getAccessControllerOrDefault(String ctl) {
      118          CatalogAccessController catalogAccessController = ctlToCtlAccessController.get(ctl);
      119          if (catalogAccessController != null) {
      120              return catalogAccessController;
      121          }
      122          CatalogIf catalog = Env.getCurrentEnv().getCatalogMgr().getCatalog(ctl);
      123          if (catalog != null && catalog instanceof ExternalCatalog) {
      124              lazyLoadCtlAccessController((ExternalCatalog) catalog);
      125              return ctlToCtlAccessController.get(ctl);
      126          }
      127  
      128          return defaultAccessController;
      129      }

      131      private synchronized void lazyLoadCtlAccessController(ExternalCatalog catalog) {
      132          if (ctlToCtlAccessController.containsKey(catalog.getName())) {
      133              return;
      134          }
      135          catalog.initAccessController(false);
      136          if (!ctlToCtlAccessController.containsKey(catalog.getName())) {
      137              ctlToCtlAccessController.put(catalog.getName(), defaultAccessController);
      138          }
      139      }

      145      public void createAccessController(String ctl, String acFactoryClassName, Map<String, String> prop,
      146                                         boolean isDryRun) {
      147          String pluginIdentifier = getPluginIdentifierForAccessController(acFactoryClassName);
      148          CatalogAccessController accessController = accessControllerFactoriesCache.get(pluginIdentifier)
      149                  .createAccessController(prop);
      150          if (!isDryRun) {
      151              ctlToCtlAccessController.put(ctl, accessController);
      152              LOG.info("create access controller {} for catalog {}", acFactoryClassName, ctl);
      153          }
      154      }

      156      private String getPluginIdentifierForAccessController(String acClassName) {
      157          String pluginIdentifier = null;
      158          if (accessControllerClassNameMapping.containsKey(acClassName)) {
      159              pluginIdentifier = accessControllerClassNameMapping.get(acClassName);
      160          }
      161          if (accessControllerFactoriesCache.containsKey(acClassName)) {
      162              pluginIdentifier = acClassName;
      163          }
      164          if (null == pluginIdentifier || !accessControllerFactoriesCache.containsKey(pluginIdentifier)) {
      165              throw new RuntimeException("Access Controller Plugin Factory not found for " + acClassName);
      166          }
      167          return pluginIdentifier;
      168      }

      299      public boolean checkPrivByAuthInfo(ConnectContext ctx, AuthorizationInfo authInfo, PrivPredicate wanted) {
      300          if (authInfo == null) {
      301              return false;
      302          }
      303          if (authInfo.getDbName() == null) {
      304              return false;
      305          }
      306          if (authInfo.getTableNameList() == null || authInfo.getTableNameList().isEmpty()) {
      307              return checkDbPriv(ctx, InternalCatalog.INTERNAL_CATALOG_NAME, authInfo.getDbName(), wanted);
      308          }
      309          for (String tblName : authInfo.getTableNameList()) {
      310              if (!checkTblPriv(ConnectContext.get(), InternalCatalog.INTERNAL_CATALOG_NAME, authInfo.getDbName(),
      311                      tblName, wanted)) {
      312                  return false;
      313              }
      314          }
      315          return true;
      316      }
