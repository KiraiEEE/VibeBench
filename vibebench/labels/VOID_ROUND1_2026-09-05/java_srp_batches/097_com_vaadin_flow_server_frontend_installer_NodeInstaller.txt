      228      private void installNode(InstallData data) throws InstallationException {
      229          try {
      230  
      231              downloadFileIfMissing(data.getDownloadUrl(), data.getArchive(),
      232                      userName, password);
      233  
      234              extractFile(data.getArchive(), data.getTmpDirectory());
      235          } catch (DownloadException e) {
      236              throw new InstallationException(
      237                      "Node.js download failed. This may be due to loss of internet connection.\n"
      238                              + "If you are behind a proxy server you should configure your proxy settings.\n"
      239                              + "Verify connection and proxy settings or follow the https://nodejs.org/en/download/ guide to install Node.js globally.",
      240                      e);
      241          } catch (ArchiveExtractionException e) {
      242              throw new InstallationException(
      243                      "Could not extract the Node archive", e);
      244          }
      245  
      246          try {
      247              if (platform.isWindows()) {
      248                  installNodeWindows(data);
      249              } else {
      250                  installNodeUnix(data);
      251              }
      252          } catch (IOException e) {
      253              throw new InstallationException("Could not install Node", e);
      254          }
      255  
      256          getLogger().info("Local node installation successful.");
      257      }

      259      private void installNodeUnix(InstallData data)
      260              throws InstallationException, IOException {
      261  
      262          File extractedNodeDir = new File(data.getTmpDirectory(),
      263                  data.getNodeFilename());
      264  
      265          // Verify the extraction created the expected directory
      266          if (!extractedNodeDir.exists() || !extractedNodeDir.isDirectory()) {
      267              throw new FileNotFoundException(
      268                      "Could not find the extracted Node.js directory at "
      269                              + extractedNodeDir);
      270          }
      271  
      272          // Verify the node binary exists in the extracted archive
      273          File nodeBinary = new File(extractedNodeDir,
      274                  "bin" + File.separator + data.getNodeExecutable());
      275  
      276          if (!nodeBinary.exists()) {
      277              throw new FileNotFoundException(
      278                      "Could not find the downloaded Node.js binary in "
      279                              + nodeBinary);
      280          }
      281  
      282          File destinationDirectory = getNodeInstallDirectory();
      283  
      284          // Copy the entire Node.js distribution as-is, overwriting any existing
      285          // files
      286          getLogger().info(
      287                  "Installing complete Node.js distribution from {} to {}",
      288                  extractedNodeDir, destinationDirectory);
      289          FileIOUtils.copyDirectory(extractedNodeDir, destinationDirectory);
      290  
      291          // Make the node binary and npm scripts executable
      292          File installedNodeBinary = new File(destinationDirectory,
      293                  "bin" + File.separator + data.getNodeExecutable());
      294          if (!installedNodeBinary.setExecutable(true, false)) {
      295              throw new InstallationException(
      296                      "Could not install Node: Was not allowed to make "
      297                              + installedNodeBinary + " executable.");
      298          }
      299  
      300          // Make npm and npx scripts executable
      301          makeScriptExecutable(destinationDirectory, "bin/npm");
      302          makeScriptExecutable(destinationDirectory, "bin/npx");
      303          makeScriptExecutable(destinationDirectory, "bin/corepack");
      304  
      305          deleteTempDirectory(data.getTmpDirectory());
      306      }

      380      private void extractFile(File archive, File destinationDirectory)
      381              throws ArchiveExtractionException {
      382          long size;
      383          try {
      384              size = Files.size(archive.toPath());
      385          } catch (IOException e) {
      386              throw new ArchiveExtractionException(
      387                      "Error determining archive size", e);
      388          }
      389          try {
      390              getLogger().info("Unpacking {} ({} bytes) into {}", archive, size,
      391                      destinationDirectory);
      392              archiveExtractor.extract(archive, destinationDirectory);
      393          } catch (ArchiveExtractionException e) {
      394              if (e.getCause() instanceof EOFException) {
      395                  // https://github.com/eirslett/frontend-maven-plugin/issues/794
      396                  // The downloading was probably interrupted and archive file is
      397                  // incomplete:
      398                  // delete it to retry from scratch
      399                  getLogger().error(
      400                          "The archive file {} is corrupted and will be deleted. "
      401                                  + "Please run the application again.",
      402                          archive.getPath());
      403                  removeArchiveFile(archive);
      404                  try {
      405                      FileIOUtils.delete(destinationDirectory);
      406                  } catch (IOException ioe) {
      407                      getLogger().error("Failed to remove target directory '{}'",
      408                              destinationDirectory, ioe);
      409                  }
      410              } else {
      411                  removeArchiveFile(archive);
      412              }
      413  
      414              throw e;
      415          }
      416      }
