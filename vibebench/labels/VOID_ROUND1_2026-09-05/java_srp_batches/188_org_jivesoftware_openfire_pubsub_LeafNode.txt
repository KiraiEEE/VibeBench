      183      public synchronized void setLastPublishedItem(PublishedItem item)
      184      {
      185          // Always update when:
      186          // 1. There is no last-published item yet (initial state).
      187          // 2. The incoming item overwrites the current last-published item (same unique identifier).
      188          //    XEP-0060 §7.1.2 requires the server to replace an existing item with the same ID.
      189          //    Even if both items share the same creation-date (e.g. published in the same millisecond),
      190          //    the in-memory cache must reflect the new payload so that getPublishedItem() does not
      191          //    serve the stale first item.
      192          // 3. The incoming item is strictly newer than the current last-published item.
      193          if (item != null && (lastPublished == null
      194                  || lastPublished.getUniqueIdentifier().equals(item.getUniqueIdentifier())
      195                  || item.getCreationDate().after(lastPublished.getCreationDate()))) {
      196              Log.trace("Set last published item to: {}", item.getID());
      197              lastPublished = item;
      198          }
      199      }

      229      /**
      230       * Publishes the list of items to the node. Event notifications will be sent to subscribers
      231       * for the new published event, as well as all connected resources of the owner, in case the node
      232       * is a node in a PEP service.<p>
      233       *
      234       * The published event may or may not include an item. When the
      235       * node is not persistent and does not require payloads then an item is not going to be created
      236       * or included in the event notification.<p>
      237       *
      238       * When an affiliate has many subscriptions to the node, the affiliate will get a
      239       * notification for each set of items that affected the same list of subscriptions.<p>
      240       *
      241       * When an item is included in the published event then a new {@link PublishedItem} is
      242       * going to be created and added to the list of published item. Each published item will
      243       * have a unique ID in the node scope. The new published item will be added to the end
      244       * of the published list to keep the chronological order. When the max number of published
      245       * items is exceeded then the oldest published items will be removed.<p>
      246       *
      247       * For performance reasons the newly added published items and the deleted items (if any)
      248       * are saved to the database using a background thread. Sending event notifications to
      249       * node subscribers may also use another thread to ensure good performance.<p>
      250       *
      251       * @param publisher the full JID of the user that sent the new published event.
      252       * @param itemElements list of dom4j elements that contain info about the published items.
      253       */
      254      public void publishItems(JID publisher, List<Element> itemElements) {
      255          Log.trace("Publisher '{}' is publishing {} item(s)", publisher, itemElements.size());
      256          List<PublishedItem> newPublishedItems = new ArrayList<>();
      257          if (isItemRequired()) {
      258              String itemID;
      259              Element payload;
      260              PublishedItem newItem;
      261              for (Element item : itemElements) {
      262                  itemID = item.attributeValue("id");
      263                  List<Element> entries = item.elements();
      264                  payload = entries.isEmpty() ? null : entries.get(0);
      265                  
      266                  // Make sure that the published item has a unique ID if NOT assigned by publisher
      267                  if (itemID == null) {
      268                      itemID = UUID.randomUUID().toString();
      269                  }
      270  
      271                  // Create a new published item
      272                  newItem = new PublishedItem(this, publisher, itemID, new Date(CacheFactory.getClusterTime()));
      273                  newItem.setPayload(payload);
      274                  Log.trace("Created new PublishedItem instance with itemID {}", itemID);
      275  
      276                  // Add the new item to the list of published items
      277                  newPublishedItems.add(newItem);
      278                  setLastPublishedItem(newItem);
      279                  // Add the new published item to the queue of items to add to the database. The
      280                  // queue is going to be processed by another thread
      281                  if (isPersistPublishedItems()) {
      282                      Log.trace("Adding PublishedItem in persistence provider as node is configured to persist published items");
      283                      XMPPServer.getInstance().getPubSubModule().getPersistenceProvider().savePublishedItem(newItem);
      284                  }
      285              }
      286          }
      287  
      288          // Build event notification packet to broadcast to subscribers
      289          Message message = new Message();
      290          Element event = message.addChildElement("event", "http://jabber.org/protocol/pubsub#event");
      291          // Broadcast event notification to subscribers and parent node subscribers
      292          Set<NodeAffiliate> affiliatesToNotify = getAffiliatesToNotify();
      293          Log.trace("Built event notification stanza to broadcast notification to {} affiliate(s)", affiliatesToNotify.size());
      294  
      295          // TODO Use another thread for this (if # of subscribers is > X)????
      296          for (NodeAffiliate affiliate : affiliatesToNotify) {
      297              affiliate.sendPublishedNotifications(message, event, this, newPublishedItems);
      298          }
      299  
      300          // Invoke event listeners.
      301          PubSubEventDispatcher.dispatchItemsPublished(this.getUniqueIdentifier(), newPublishedItems);
      302      }

      307       * @return A list of node affiliates. Possibly empty.

      309      public Set<NodeAffiliate> getAffiliatesToNotify() {
      310          Log.trace("Getting affiliates to notify...");
      311  
      312          Set<NodeAffiliate> affiliatesToNotify = new HashSet<>(affiliates);
      313          Log.trace("... found {} direct affiliate(s)", affiliates.size());
      314  
      315          // Get affiliates that are subscribed to a parent in the hierarchy of parent nodes
      316          for (CollectionNode parentNode : getParents()) {
      317              Set<NodeAffiliate> parentAffiliates = new HashSet<>();
      318              for (NodeSubscription subscription : parentNode.getSubscriptions()) {
      319                  // OF-2365: Prevent sending notifications to subscribers that are not allowed to access this node.
      320                  if (parentNode.getAccessModel().canAccessItems(this, subscription.getOwner(), subscription.getJID() )
      321                      && accessModel.canAccessItems(this, subscription.getOwner(), subscription.getJID()))
      322                  {
      323                      parentAffiliates.add(subscription.getAffiliate());
      324                  }
      325              }
      326              Log.trace("... found {} applicable affiliate(s) in parent node {}", parentAffiliates.size(), parentNode.getNodeID());
      327              affiliatesToNotify.addAll(parentAffiliates);
      328          }
      329  
      330          // XEP-0136 specifies that all connected resources of the owner of the PEP service should also get a notification (pending filtering)
      331          // To ensure that happens, the affiliate that represents the owner of the PEP server is added here, if it's not already present.
      332          if (getService() instanceof PEPService && affiliatesToNotify.stream().noneMatch( a -> a.getAffiliation().equals(NodeAffiliate.Affiliation.owner))) {
      333              final NodeAffiliate owner = getService().getRootCollectionNode().getAffiliate( getService().getAddress() );
      334              Log.trace("... added owner of the PEP service");
      335              affiliatesToNotify.add(owner);
      336          }
      337          Log.trace("In total {} unique affiliate(s) were identified.", affiliatesToNotify.size());
      338          return affiliatesToNotify;

      341      /**
      342       * Deletes the list of published items from the node. Event notifications may be sent to
      343       * subscribers for the deleted items, as well as all connected resources of the service owner,
      344       * if the service is a PEP service.<p>
      345       *
      346       * When an affiliate has many subscriptions to the node, the affiliate will get a notification
      347       * for each set of items that affected the same list of subscriptions.<p>
      348       *
      349       * @param toDelete list of items that were deleted from the node.
      350       */
      351      public void deleteItems(List<PublishedItem> toDelete) {
      352          Log.trace("Deleting {} item(s)", toDelete.size());
      353          // Remove deleted items from the database
      354          for (PublishedItem item : toDelete) {
      355              Log.trace("Removing PublishedItem from persistence provider");
      356              XMPPServer.getInstance().getPubSubModule().getPersistenceProvider().removePublishedItem(item);
      357              if (lastPublished != null && lastPublished.getID().equals(item.getID())) {
      358                  Log.trace("Removed item was previously the last published item. Setting last published item to null.");
      359                  lastPublished = null;
      360              }
      361          }
      362          if (isNotifiedOfRetract()) {
      363              // Build packet to broadcast to subscribers
      364              Message message = new Message();
      365              Element event = message.addChildElement("event", "http://jabber.org/protocol/pubsub#event");
      366              // Send notification that items have been deleted to subscribers and parent node subscribers
      367              Set<NodeAffiliate> affiliatesToNotify = getAffiliatesToNotify();
      368              Log.trace("Built event notification stanza to broadcast notification to {} affiliate(s)", affiliatesToNotify.size());
      369  
      370              // TODO Use another thread for this (if # of subscribers is > X)????
      371              for (NodeAffiliate affiliate : affiliatesToNotify) {
      372                  affiliate.sendDeletionNotifications(message, event, this, toDelete);
      373              }
      374  
      375              // XEP-0136 specifies that all connected resources of the owner of the PEP service should also get a notification.
      376              if ( getService() instanceof PEPService )
      377              {
      378                  final PEPService service = (PEPService) getService();
      379                  Element items = event.addElement("items");
      380                  items.addAttribute("node", getUniqueIdentifier().getNodeId());
      381                  for (PublishedItem publishedItem : toDelete) {
      382                      // Add retract information to the event notification
      383                      Element item = items.addElement("retract");
      384                      if (isItemRequired()) {
      385                          item.addAttribute("id", publishedItem.getID());
      386                      }
      387  
      388                      // Send the notification
      389                      final Collection<ClientSession> sessions = SessionManager.getInstance().getSessions(service.getAddress().getNode());
      390                      Log.trace("Also notifying {} connected resource(s) of the owner of the PEP service: {}", sessions.size(), service.getAddress());
      391  
      392                      for (final ClientSession session : sessions) {
      393                          service.sendNotification( this, message, session.getAddress() );
      394                      }
      395  
      396                      // Remove the added items information
      397                      event.remove(items);
      398                  }
      399              }
      400          }
      401  
      402          // Invoke event listeners.
      403          PubSubEventDispatcher.dispatchItemsDeleted(this.getUniqueIdentifier(), toDelete);
      404      }

      406      /**
      407       * Sends an IQ result with the list of items published to the node. Item ID and payload are always included.
      408       * Should only be used for use-cases described in section "6.5 Retrieve Items from a Node" from XEP-0060 (as
      409       * opposed to processing of notifications or service discovery, which are allowed to discard payloads).
      410       *
      411       * @param originalRequest the IQ packet sent by a subscriber (or anyone) to get the node items.
      412       * @param publishedItems the list of published items to send to the subscriber.
      413       */
      414      void sendPublishedItems(IQ originalRequest, List<PublishedItem> publishedItems) {
      415          Log.trace("Sending {} published item(s) in response to request from '{}'", publishedItems.size(), originalRequest.getFrom());
      416          IQ result = IQ.createResultIQ(originalRequest);
      417          Element pubsubElem = result.setChildElement("pubsub", "http://jabber.org/protocol/pubsub");
      418          Element items = pubsubElem.addElement("items");
      419          items.addAttribute("node", nodeID);
      420          
      421          for (PublishedItem publishedItem : publishedItems) {
      422              Element item = items.addElement("item");
      423              if (isItemRequired()) {
      424                  item.addAttribute("id", publishedItem.getID());
      425              }
      426              item.add(publishedItem.getPayload().createCopy());
      427          }
      428          // Send the result
      429          getService().send(result);
      430      }

      433      public PublishedItem getPublishedItem(String itemID) {
      434          if (!isItemRequired()) {
      435              Log.trace("Get published item {}, but returning null, as node is configured to not require item.", itemID);
      436              return null;
      437          }
      438  
      439          final PublishedItem.UniqueIdentifier itemIdentifier = new PublishedItem.UniqueIdentifier( getUniqueIdentifier(), itemID );
      440          synchronized (this) {
      441              if (lastPublished != null && lastPublished.getUniqueIdentifier().equals( itemIdentifier )) {
      442                  Log.trace("Get published item {} that is the last published item", itemID);
      443                  return lastPublished;
      444              }
      445          }
      446          final PublishedItem publishedItem = XMPPServer.getInstance().getPubSubModule().getPersistenceProvider().getPublishedItem(this, itemIdentifier);
      447          Log.trace("Get published item {} from persistence provider{}", itemID, publishedItem == null ? ", but it did not exist there." : ".");
      448          return publishedItem;
      449      }
