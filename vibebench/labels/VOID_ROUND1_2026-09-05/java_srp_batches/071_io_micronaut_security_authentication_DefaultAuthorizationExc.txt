       99      @Override
      100      public MutableHttpResponse<?> handle(HttpRequest request, AuthorizationException exception) {
      101          if (shouldRedirect(request, exception)) {
      102              try {
      103                  URI location = new URI(getRedirectUri(request, exception));
      104                  //prevent redirect loop
      105                  if (!request.getUri().equals(location)) {
      106                      MutableHttpResponse<?> response = httpResponseWithStatus(location);
      107                      if (priorToLoginPersistence != null && !exception.isForbidden()) {
      108                          priorToLoginPersistence.onUnauthorized(request, response);
      109                      }
      110                      return response;
      111                  }
      112                  return httpResponseWithStatus(request, exception);
      113              } catch (URISyntaxException e) {
      114                  if (LOG.isErrorEnabled()) {
      115                      LOG.error("Rejection redirect URL is invalid", e);
      116                  }
      117                  return HttpResponse.serverError();
      118              }
      119          }
      120          return httpResponseWithStatus(request, exception);
      121      }

      123      /**

      126       * @return The response to be used when a redirect is not appropriate

      128      protected MutableHttpResponse<?> httpResponseWithStatus(HttpRequest<?> request, AuthorizationException exception) {
      129          HttpStatus status = exception.isForbidden() ? HttpStatus.FORBIDDEN : HttpStatus.UNAUTHORIZED;
      130          MutableHttpResponse<?> response = HttpResponse.status(status);
      131          if (CollectionUtils.isNotEmpty(wwwAuthenticateChallengeProviders)) {
      132              response.header(HttpHeaders.WWW_AUTHENTICATE,
      133                  String.join(COMMA_SPACE, wwwAuthenticateChallengeProviders.stream()
      134                      .map(provider -> provider.getWwwAuthenticateChallenge(request))
      135                      .toList()));
      136          }
      137          return errorResponseProcessor.processResponse(ErrorContext.builder(request)
      138              .cause(exception)
      139              .errorMessage(status.getReason())
      140              .build(), response);
      141      }

      143      /**
      144       * Decides whether the request should be handled with a redirect.

      146       * @param request The HTTP Request

      150      protected boolean shouldRedirect(HttpRequest<?> request, AuthorizationException exception) {
      151          if (redirectConfiguration != null && redirectConfiguration.isEnabled()) {
      152              return (
      153                      (exception.isForbidden() && redirectConfiguration.getForbidden().isEnabled()) ||
      154                              (!exception.isForbidden() && redirectConfiguration.getUnauthorized().isEnabled())
      155              ) && request.getHeaders()
      156                      .accept()
      157                      .stream()
      158                      .anyMatch(mediaType -> mediaType.equals(MediaType.TEXT_HTML_TYPE));
      159          } else {
      160              return false;
      161          }
      162      }
