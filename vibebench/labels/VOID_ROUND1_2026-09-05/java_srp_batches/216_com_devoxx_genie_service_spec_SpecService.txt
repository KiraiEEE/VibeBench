      142                      .sorted(Map.Entry.<TaskSpec, Double>comparingByValue().reversed())
      143                      .toList();
      144  
      145              Stream<TaskSpec> resultStream = scored.stream().map(Map.Entry::getKey);
      146              if (limit > 0) {

      163      public @NotNull List<TaskSpec> searchSpecs(@NotNull String query,
      164                                                  @Nullable String status,
      165                                                  @Nullable String priority,
      166                                                  int limit) {
      167          java.util.Map<String, TaskSpec> candidates = new java.util.LinkedHashMap<>();
      168          specCache.values().forEach(spec -> candidates.put(spec.getId().toUpperCase(java.util.Locale.ROOT), spec));
      169          getArchivedTasks().forEach(spec -> candidates.putIfAbsent(spec.getId().toUpperCase(java.util.Locale.ROOT), spec));
      170  
      171          Stream<TaskSpec> stream = candidates.values().stream();
      172  
      173          if (status != null && !status.isEmpty()) {
      174              stream = stream.filter(s -> status.equalsIgnoreCase(s.getStatus()));
      175          }
      176          if (priority != null && !priority.isEmpty()) {
      177              stream = stream.filter(s -> priority.equalsIgnoreCase(s.getPriority()));
      178          }
      179  
      180          // Score each spec and filter out non-matches, then sort by relevance
      181          List<Map.Entry<TaskSpec, Double>> scored = stream
      182                  .map(s -> Map.entry(s, FuzzySearchHelper.scoreMultiField(query, s.getTitle(), s.getDescription())))
      183                  .filter(e -> e.getValue() >= 0.3)
      184                  .sorted(Map.Entry.<TaskSpec, Double>comparingByValue().reversed())
      185                  .toList();
      186  
      187          Stream<TaskSpec> resultStream = scored.stream().map(Map.Entry::getKey);
      188          if (limit > 0) {
      189              resultStream = resultStream.limit(limit);
      190          }
      191  
      192          return resultStream.toList();

      222              }

      224              // Apply Definition of Done defaults from config if the task has none

      373      /**
      374       * Returns all archived tasks by scanning the archive/tasks/ directory.
      375       * These are NOT part of the normal specCache.
      376       */
      377      public @NotNull List<TaskSpec> getArchivedTasks() {
      378          BacklogConfigService configService = BacklogConfigService.getInstance(project);
      379          Path archiveDir = configService.getArchiveTasksDir();
      380          if (archiveDir == null || !Files.isDirectory(archiveDir)) {
      381              return List.of();
      382          }
      383  
      384          List<TaskSpec> archived = new ArrayList<>();
      385          try (Stream<Path> files = Files.walk(archiveDir)) {
      386              files.filter(p -> p.toString().endsWith(".md"))
      387                      .filter(Files::isRegularFile)
      388                      .forEach(file -> {
      389                          try {
      390                              String content = Files.readString(file, StandardCharsets.UTF_8);
      391                              TaskSpec spec = SpecFrontmatterParser.parse(content, file.toAbsolutePath().toString());
      392                              if (spec != null && spec.getId() != null) {
      393                                  spec.setLastModified(Files.getLastModifiedTime(file).toMillis());
      394                                  archived.add(spec);
      395                              }
      396                          } catch (IOException e) {
      397                              log.warn("Failed to read archived task file: {}", file, e);
      398                          }
      399                      });
      400          } catch (IOException e) {
      401              log.warn("Failed to scan archive directory: {}", archiveDir, e);
      402          }
      403          return archived;
      404      }
