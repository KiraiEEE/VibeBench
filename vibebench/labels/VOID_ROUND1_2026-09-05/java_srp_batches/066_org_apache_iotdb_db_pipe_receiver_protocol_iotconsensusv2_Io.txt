      165    public TIoTConsensusV2TransferResp receive(final TIoTConsensusV2TransferReq req) {
      166      long startNanos = System.nanoTime();
      167      // PreCheck: if there are these cases: read-only; null impl; inactive impl, etc. The receiver
      168      // will reject synchronization.
      169      TIoTConsensusV2TransferResp resp = preCheckForReceiver(req);
      170      if (resp != null) {
      171        // release tsFileWriter if pre-check failed as leader will resend the whole tsFileEvent
      172        releaseTsFileWriter(
      173            iotConsensusV2TsFileWriterPool.tryToFindCorrespondingWriter(req.getCommitId()), false);
      174        return resp;
      175      }
      176  
      177      final short rawRequestType = req.getType();
      178      if (IoTConsensusV2RequestType.isValidatedRequestType(rawRequestType)) {
      179        switch (IoTConsensusV2RequestType.valueOf(rawRequestType)) {
      180          case TRANSFER_TS_FILE_PIECE:
      181          case TRANSFER_TS_FILE_PIECE_WITH_MOD:
      182            // Just take a place in requestExecutor's buffer, the further seal request will remove
      183            // its place from buffer.
      184            requestExecutor.onRequest(req, true, false);
      185            resp = loadEvent(req);
      186            break;
      187          case TRANSFER_TS_FILE_SEAL:
      188          case TRANSFER_TS_FILE_SEAL_WITH_MOD:
      189            resp = requestExecutor.onRequest(req, false, true);
      190            break;
      191          case TRANSFER_DELETION:
      192          case TRANSFER_TABLET_BINARY:
      193          case TRANSFER_TABLET_INSERT_NODE:
      194          case TRANSFER_TABLET_BATCH:
      195          default:
      196            resp = requestExecutor.onRequest(req, false, false);
      197            break;
      198        }
      199        // update receive an event's duration
      200        long durationNanos = System.nanoTime() - startNanos;
      201        iotConsensusV2ReceiverMetrics.recordReceiveEventTimer(durationNanos);
      202        return resp;
      203      }
      204      // Unknown request type, which means the request can not be handled by this receiver,
      205      // maybe the version of the receiver is not compatible with the sender
      206      final TSStatus status =
      207          RpcUtils.getStatus(
      208              TSStatusCode.PIPE_TYPE_ERROR,
      209              String.format("IoTConsensusV2 Unknown PipeRequestType %s.", rawRequestType));
      210      if (LOGGER.isWarnEnabled()) {
      211        LOGGER.warn("IoTConsensusV2 Unknown PipeRequestType, response status = {}.", status);
      212      }
      213      return new TIoTConsensusV2TransferResp(status);

      247    private TIoTConsensusV2TransferResp loadEvent(final TIoTConsensusV2TransferReq req) {
      248      if (isClosed.get()) {
      249        return IoTConsensusV2ReceiverAgent.closedResp(
      250            consensusPipeName.toString(), req.getCommitId());
      251      }
      252      // synchronized load event, ensured by upper caller's lock.
      253      try {
      254        final short rawRequestType = req.getType();
      255        if (IoTConsensusV2RequestType.isValidatedRequestType(rawRequestType)) {
      256          switch (IoTConsensusV2RequestType.valueOf(rawRequestType)) {
      257            case TRANSFER_TABLET_INSERT_NODE:
      258              return handleTransferTabletInsertNode(
      259                  IoTConsensusV2TabletInsertNodeReq.fromTIoTConsensusV2TransferReq(req));
      260            case TRANSFER_TABLET_BINARY:
      261              return handleTransferTabletBinary(
      262                  IoTConsensusV2TabletBinaryReq.fromTIoTConsensusV2TransferReq(req));
      263            case TRANSFER_DELETION:
      264              return handleTransferDeletion(
      265                  IoTConsensusV2DeleteNodeReq.fromTIoTConsensusV2TransferReq(req));
      266            case TRANSFER_TS_FILE_PIECE:
      267              return handleTransferFilePiece(
      268                  IoTConsensusV2TsFilePieceReq.fromTIoTConsensusV2TransferReq(req), true);
      269            case TRANSFER_TS_FILE_SEAL:
      270              return handleTransferFileSeal(
      271                  IoTConsensusV2TsFileSealReq.fromTIoTConsensusV2TransferReq(req));
      272            case TRANSFER_TS_FILE_PIECE_WITH_MOD:
      273              return handleTransferFilePiece(
      274                  IoTConsensusV2TsFilePieceReq.fromTIoTConsensusV2TransferReq(req), false);
      275            case TRANSFER_TS_FILE_SEAL_WITH_MOD:
      276              return handleTransferFileSealWithMods(
      277                  IoTConsensusV2TsFileSealWithModReq.fromTIoTConsensusV2TransferReq(req));
      278            case TRANSFER_TABLET_BATCH:
      279              LOGGER.info("IoTConsensusV2 transfer batch hasn't been implemented yet.");
      280            default:
      281              break;
      282          }
      283        }
      284        // Unknown request type, which means the request can not be handled by this receiver,
      285        // maybe the version of the receiver is not compatible with the sender
      286        final TSStatus status =
      287            RpcUtils.getStatus(
      288                TSStatusCode.IOT_CONSENSUS_V2_TYPE_ERROR,
      289                String.format("Unknown IoTConsensusV2RequestType %s.", rawRequestType));
      290        LOGGER.warn(
      291            "IoTConsensusV2-PipeName-{}: Unknown PipeRequestType, response status = {}.",
      292            consensusPipeName,
      293            status);
      294        return new TIoTConsensusV2TransferResp(status);
      295      } catch (Exception e) {
      296        final String error = String.format("Serialization error during pipe receiving, %s", e);
      297        LOGGER.warn("IoTConsensusV2-PipeName-{}: {}", consensusPipeName, error, e);
      298        return new TIoTConsensusV2TransferResp(RpcUtils.getStatus(TSStatusCode.PIPE_ERROR, error));

      338    private TIoTConsensusV2TransferResp handleTransferFilePiece(
      339        final IoTConsensusV2TransferFilePieceReq req, final boolean isSingleFile) {
      340      tsFilePieceReadWriteLock.readLock().lock();

      342        if (LOGGER.isDebugEnabled()) {

      344              "IoTConsensusV2-PipeName-{}: starting to receive tsFile pieces", consensusPipeName);
      345        }
      346        long startBorrowTsFileWriterNanos = System.nanoTime();
      347        IoTConsensusV2TsFileWriter tsFileWriter =
      348            iotConsensusV2TsFileWriterPool.borrowCorrespondingWriter(req.getCommitId());
      349        long startPreCheckNanos = System.nanoTime();
      350        iotConsensusV2ReceiverMetrics.recordBorrowTsFileWriterTimer(
      351            startPreCheckNanos - startBorrowTsFileWriterNanos);
      352  
      353        try {
      354          updateWritingFileIfNeeded(tsFileWriter, req.getFileName(), isSingleFile);
      355          final File writingFile = tsFileWriter.getWritingFile();
      356          final RandomAccessFile writingFileWriter = tsFileWriter.getWritingFileWriter();
      357  
      358          if (isWritingFileOffsetNonCorrect(tsFileWriter, req.getStartWritingOffset())) {
      359            if (!writingFile.getName().endsWith(TsFileConstant.TSFILE_SUFFIX)) {
      360              // If the file is a tsFile, then the content will not be changed for a specific
      361              // filename. However, for other files (mod, snapshot, etc.) the content varies for the
      362              // same name in different times, then we must rewrite the file to apply the newest
      363              // version.
      364              writingFileWriter.setLength(0);
      365            }
      366  
      367            final TSStatus status =
      368                RpcUtils.getStatus(
      369                    TSStatusCode.IOT_CONSENSUS_V2_TRANSFER_FILE_OFFSET_RESET,
      370                    String.format(
      371                        "Request sender to reset file reader's offset from %s to %s.",
      372                        req.getStartWritingOffset(), writingFileWriter.length()));
      373            LOGGER.warn(
      374                "IoTConsensusV2-PipeName-{}: File offset reset requested by receiver, response status = {}.",
      375                consensusPipeName,
      376                status);
      377            return IoTConsensusV2TransferFilePieceResp.toTIoTConsensusV2TransferResp(
      378                status, writingFileWriter.length());
      379          }
      380  
      381          long endPreCheckNanos = System.nanoTime();
      382          iotConsensusV2ReceiverMetrics.recordTsFilePiecePreCheckTime(
      383              endPreCheckNanos - startPreCheckNanos);
      384          writingFileWriter.write(req.getFilePiece());
      385          iotConsensusV2ReceiverMetrics.recordTsFilePieceWriteTime(
      386              System.nanoTime() - endPreCheckNanos);
      387          return IoTConsensusV2TransferFilePieceResp.toTIoTConsensusV2TransferResp(
      388              RpcUtils.SUCCESS_STATUS, writingFileWriter.length());
      389        } catch (Exception e) {
      390          LOGGER.warn(
      391              "IoTConsensusV2-PipeName-{}: Failed to write file piece from req {}.",
      392              consensusPipeName,
      393              req,
      394              e);
      395          final TSStatus status =
      396              RpcUtils.getStatus(
      397                  TSStatusCode.IOT_CONSENSUS_V2_TRANSFER_FILE_ERROR,
      398                  String.format("Failed to write file piece, because %s", e.getMessage()));
      399          try {
      400            return IoTConsensusV2TransferFilePieceResp.toTIoTConsensusV2TransferResp(
      401                status, PipeTransferFilePieceResp.ERROR_END_OFFSET);
      402          } catch (IOException ex) {
      403            return IoTConsensusV2TransferFilePieceResp.toTIoTConsensusV2TransferResp(status);
      404          } finally {
      405            // Exception may occur when disk system go wrong. At this time, we may reset all resource
      406            // and receive this file from scratch when leader will try to resend this file from
      407            // scratch as well.
      408            releaseTsFileWriter(tsFileWriter, false);
      409          }
      410        }
      411      } finally {
      412        tsFilePieceReadWriteLock.readLock().unlock();
      413      }
      414    }
