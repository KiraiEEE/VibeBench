      160      /**
      161       * Connects to the Iggy server asynchronously.
      162       *
      163       * <p>This establishes the TCP connection using Netty's non-blocking I/O. After the
      164       * returned future completes, the sub-clients ({@link #messages()}, {@link #streams()},
      165       * etc.) become available. You must call this before performing any operations.
      166       *
      167       * @return a {@link CompletableFuture} that completes when the connection is established
      168       */
      169      public CompletableFuture<Void> connect() {
      170          TCPConnectionPoolConfig.Builder poolConfigBuilder = new TCPConnectionPoolConfig.Builder();
      171          connectionPoolSize.ifPresent(poolConfigBuilder::setMaxConnections);
      172          acquireTimeout.ifPresent(timeout -> poolConfigBuilder.setAcquireTimeoutMillis(timeout.toMillis()));
      173          TCPConnectionPoolConfig poolConfig = poolConfigBuilder.build();
      174          // TCPConnectionPoolConfig poolConfig = new TCPConnectionPoolConfig();
      175          connection = new AsyncTcpConnection(host, port, enableTls, tlsCertificate, poolConfig);
      176          return connection.connect().thenRun(() -> {
      177              messagesClient = new MessagesTcpClient(connection);
      178              consumerGroupsClient = new ConsumerGroupsTcpClient(connection);
      179              consumerOffsetsClient = new ConsumerOffsetsTcpClient(connection);
      180              streamsClient = new StreamsTcpClient(connection);
      181              topicsClient = new TopicsTcpClient(connection);
      182              usersClient = new UsersTcpClient(connection);
      183              systemClient = new SystemTcpClient(connection);
      184              personalAccessTokensClient = new PersonalAccessTokensTcpClient(connection);
      185              partitionsClient = new PartitionsTcpClient(connection);
      186          });
      187      }

      190       * Logs in using the credentials provided during client construction.
      191       *
      192       * <p>Credentials must have been set via
      193       * {@link AsyncIggyTcpClientBuilder#credentials(String, String)} when building

      202      public CompletableFuture<IdentityInfo> login() {
      203          if (usersClient == null) {
      204              throw new IggyNotConnectedException();
      205          }
      206          if (username.isEmpty() || password.isEmpty()) {
      207              throw new IggyMissingCredentialsException();
      208          }
      209          return usersClient.login(username.get(), password.get());
      210      }
