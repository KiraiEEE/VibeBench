       63      private AiService createAiService(String serviceType) {
       64          try {
       65              if ("openai".equalsIgnoreCase(serviceType)) {
       66                  // Check if OpenAI API key is configured
       67                  String apiKey = AiConfig.getProperty("openai.api.key", "");
       68                  String model = AiConfig.getProperty("openai.default.model", "");
       69                  if (apiKey != null && !apiKey.isEmpty() && !apiKey.equals("YOUR_API_KEY")
       70                          && model != null && !model.isEmpty()) {
       71                      return new OpenAiService();
       72                  }
       73              } else if ("anthropic".equalsIgnoreCase(serviceType)) {
       74                  // Check if Anthropic API key is configured
       75                  String apiKey = AiConfig.getProperty("anthropic.api.key", "");
       76                  String model = AiConfig.getProperty("anthropic.model", "");
       77                  if (apiKey != null && !apiKey.isEmpty() && !apiKey.equals("YOUR_API_KEY")
       78                          && model != null && !model.isEmpty()) {
       79                      return new ClaudeService();
       80                  }
       81              } else if ("bedrock".equalsIgnoreCase(serviceType)) {
       82                  // Check if AWS Bedrock credentials are configured
       83                  String accessKey = AiConfig.getProperty("bedrock.access.key", "");
       84                  String secretKey = AiConfig.getProperty("bedrock.secret.key", "");
       85                  String model = AiConfig.getProperty("bedrock.model", "");
       86                  if (accessKey != null && !accessKey.isEmpty() && !accessKey.equals("YOUR_AWS_ACCESS_KEY_ID")
       87                          && secretKey != null && !secretKey.isEmpty() && !secretKey.equals("YOUR_AWS_SECRET_ACCESS_KEY")
       88                          && model != null && !model.isEmpty()) {
       89                      return new BedrockService();
       90                  }
       91              }
       92          } catch (Exception e) {
       93              log.error("Error creating AI service", e);
       94          }
       95          return null;

      103      private void addToolbarIcon() {
      104          GuiPackage instance = GuiPackage.getInstance();
      105          if (instance != null) {
      106              final MainFrame mf = instance.getMainFrame();
      107              final ComponentFinder<JMeterToolBar> finder = new ComponentFinder<>(JMeterToolBar.class);
      108              SwingUtilities.invokeLater(new Runnable() {
      109                  @Override
      110                  public void run() {
      111                      JMeterToolBar toolbar = null;
      112                      while (toolbar == null) {
      113                          try {
      114                              Thread.sleep(1000);

      116                              log.debug("Couldn't add button to menubar", e);
      117                          }
      118                          log.debug("Searching for toolbar ... ");
      119                          toolbar = finder.findComponentIn(mf);

      121                      int pos = getPositionForIcon(toolbar.getComponents());
      122                      log.debug("validate rootPos: " + String.valueOf(pos));
      123                      Component toolbarButton = getToolbarButton();
      124                      toolbarButton.setSize(toolbar.getComponent(pos).getSize());
      125                      toolbar.add(toolbarButton, pos);

      139      private int getPositionForIcon(Component[] toolbarComponents) {
      140          int index = 0;
      141          for (Component item : toolbarComponents) {
      142              String itemClassName = item.getClass().getName();
      143              if (itemClassName.contains("javax.swing.JButton")) {
      144                  String actionCommandText = ((JButton) item).getModel().getActionCommand();
      145                  log.debug("Running for iteration: " + index + ", " + actionCommandText);
      146                  if (actionCommandText != null && actionCommandText.equals("start")) {
      147                      break;
      148                  }
      149              }

      166      private void openAiChatPanel() {
      167          GuiPackage guiPackage = GuiPackage.getInstance();
      168          if (guiPackage != null) {
      169              MainFrame mainFrame = guiPackage.getMainFrame();
      170  
      171              if (currentChatPanel != null && splitPane != null && splitPane.isShowing()) {
      172                  // Panel is currently shown, remove it
      173                  Container contentPane = mainFrame.getContentPane();
      174                  contentPane.remove(splitPane);
      175  
      176                  // Get the left component (main JMeter component) from the split pane
      177                  Component mainComponent = splitPane.getLeftComponent();
      178  
      179                  // Add it back to the content pane
      180                  contentPane.add(mainComponent, BorderLayout.CENTER);
      181  
      182                  // Clear references
      183                  splitPane = null;
      184                  log.info("AI Chat Panel hidden");
      185              } else {
      186                  // Panel is not shown, add it
      187                  if (currentChatPanel == null) {
      188                      // Only create a new panel if one doesn't exist
      189                      currentChatPanel = new AiChatPanel();
      190                      log.info("Created new AI Chat Panel");
      191                  }
      192  
      193                  // Get the current center component
      194                  Container contentPane = mainFrame.getContentPane();
      195                  Component centerComp = null;
      196                  for (Component comp : contentPane.getComponents()) {
      197                      if (contentPane.getLayout() instanceof BorderLayout &&
      198                              ((BorderLayout) contentPane.getLayout()).getConstraints(comp) == BorderLayout.CENTER) {
      199                          centerComp = comp;
      200                          break;
      201                      }
      202                  }
      203  
      204                  if (centerComp != null) {
      205                      // Remove the center component
      206                      contentPane.remove(centerComp);
      207  
      208                      // Create a split pane with the center component and chat panel
      209                      splitPane = new JSplitPane(JSplitPane.HORIZONTAL_SPLIT, centerComp, currentChatPanel);
      210                      splitPane.setResizeWeight(0.7); // Give more space to the main component
      211                      splitPane.setOneTouchExpandable(true);
      212                      splitPane.setContinuousLayout(true);
      213  
      214                      // Set divider location to give appropriate space to the chat panel
      215                      int preferredWidth = currentChatPanel.getPreferredSize().width;
      216                      int totalWidth = mainFrame.getWidth();
      217                      splitPane.setDividerLocation(totalWidth - preferredWidth - 10);
      218  
      219                      // Add the split pane to the content pane
      220                      contentPane.add(splitPane, BorderLayout.CENTER);
      221                      log.info("AI Chat Panel displayed");
      222                  }
      223              }
      224  
      225              mainFrame.revalidate();
      226              mainFrame.repaint();
      227          }
      228      }
