      248      public static boolean isValidEmailAddress(String emailAddr) {
      249          requireNonNull(emailAddr, "emailAddr");
      250          if (EMAIL_PATTERN.matcher(emailAddr).matches()) {
      251              return true;
      252          }
      253          // Try to check whether the domain part is IP address format.
      254          final Matcher m = GENERAL_EMAIL_PATTERN.matcher(emailAddr);
      255          if (m.matches()) {
      256              final String domainPart = m.group(1);
      257              return isValidIpV4Address(domainPart) ||
      258                     isValidIpV6Address(domainPart);
      259          }
      260          return false;
      261      }

      311      public static String simpleTypeName(Class<?> clazz, boolean decapitalize) {
      312          if (clazz == null) {
      313              return "null";
      314          }
      315  
      316          String className = clazz.getName();
      317          final int lastDotIdx = className.lastIndexOf('.');
      318          if (lastDotIdx >= 0) {
      319              className = className.substring(lastDotIdx + 1);
      320          }
      321  
      322          if (!decapitalize) {
      323              return className;
      324          }
      325  
      326          final StringBuilder buf = new StringBuilder(className.length());
      327          boolean lowercase = true;
      328          for (int i = 0; i < className.length(); i++) {
      329              final char c1 = className.charAt(i);
      330              final char c2;
      331              if (lowercase) {
      332                  c2 = Character.toLowerCase(c1);
      333                  if (c1 == c2) {
      334                      lowercase = false;
      335                  }
      336              } else {
      337                  c2 = c1;
      338              }
      339              buf.append(c2);
      340          }
      341  
      342          return buf.toString();
      343      }

      385      private static boolean isValidIpV4Word(CharSequence word, int from, int toExclusive) {
      386          final int len = toExclusive - from;
      387          final char c0;
      388          final char c1;
      389          final char c2;

      396                     (c0 <= '1' && c1 <= '9' && c2 <= '9' ||
      397                      c0 == '2' && c1 <= '5' && (c2 <= '5' || c1 < '5' && c2 <= '9'));
