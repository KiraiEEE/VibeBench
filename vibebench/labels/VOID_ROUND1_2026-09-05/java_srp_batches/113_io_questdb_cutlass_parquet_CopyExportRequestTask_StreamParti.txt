      550          public void setUp(RecordMetadata meta, PageFrameCursor pfc, IntList baseColumnMap) {
      551              metadata = meta;
      552              pageFrameCursor = pfc;
      553              columnNames.reopen();
      554              columnMetadata.reopen();

      556  
      557              for (int i = 0, n = meta.getColumnCount(); i < n; i++) {
      558                  CharSequence columnName = meta.getColumnName(i);
      559                  final int startSize = columnNames.size();
      560                  columnNames.put(columnName);
      561                  columnMetadata.add(columnNames.size() - startSize);
      562                  final int columnType = meta.getColumnType(i);
      563                  // GenericRecordMetadata (hybrid/cursor paths) returns i;
      564                  // table metadata returns the physical writer column index.
      565                  final int writerIdx = meta.getWriterIndex(i);
      566  
      567                  // A SYMBOL column needs symbol-table metadata when it is a
      568                  // real table column.  In the hybrid path (baseColumnMap != null)
      569                  // only pass-through columns (baseColumnMap >= 0) qualify;
      570                  // in the direct path (baseColumnMap == null) every column is real.
      571                  boolean isPassThroughSymbol = ColumnType.isSymbol(columnType)
      572                          && (baseColumnMap == null || baseColumnMap.getQuick(i) >= 0);
      573  
      574                  if (isPassThroughSymbol) {
      575                      assert pfc != null;
      576                      int symbolTableIdx = baseColumnMap != null ? baseColumnMap.getQuick(i) : i;
      577                      StaticSymbolTable symbolTable = pfc.getSymbolTable(symbolTableIdx);
      578                      assert symbolTable != null;
      579                      int symbolColumnType = columnType;
      580                      if (!symbolTable.containsNullValue()) {
      581                          symbolColumnType |= 1 << 31;
      582                      }
      583                      // Mask to 32 bits: symbolColumnType can have bit 31 set (no-null flag),
      584                      // making it negative. Without the mask, Java sign-extends it to 64 bits
      585                      // before the OR, clobbering writerIdx in the upper 32 bits.
      586                      columnMetadata.add((long) writerIdx << 32 | (symbolColumnType & 0xFFFFFFFFL));
      587                  } else {
      588                      columnMetadata.add((long) writerIdx << 32 | (columnType & 0xFFFFFFFFL));
      589                  }
      590              }
      591  
      592              long bloomFilterIndexesPtr = 0;
      593              int bloomFilterCount = 0;
      594              double fpp = Double.isNaN(bloomFilterFpp) ? DEFAULT_BLOOM_FILTER_FPP : bloomFilterFpp;
      595  
      596              if (bloomFilterColumns != null && !bloomFilterColumns.isEmpty()) {
      597                  bloomFilterColumnIndexes.reopen();
      598                  parseBloomFilterColumnIndexes(bloomFilterColumns, metadata, bloomFilterColumnIndexes, bloomFilterColumnsPosition);
      599                  if (bloomFilterColumnIndexes.size() > 0) {
      600                      bloomFilterIndexesPtr = bloomFilterColumnIndexes.getAddress();

      603              }
      604  
      605              streamWriter = createStreamingParquetWriter(
      606                      Unsafe.getNativeAllocator(MemoryTag.NATIVE_PARQUET_EXPORTER),
      607                      meta.getColumnCount(),
      608                      columnNames.ptr(),
      609                      columnNames.size(),
      610                      columnMetadata.getAddress(),
      611                      meta.getTimestampIndex(),
      612                      descending,
      613                      ParquetCompression.packCompressionCodecLevel(compressionCodec, compressionLevel),

      615                      rawArrayEncoding,
      616                      rowGroupSize,
      617                      dataPageSize,
      618                      parquetVersion,
      619                      bloomFilterIndexesPtr,
      620                      bloomFilterCount,
      621                      fpp
      622              );
      623          }

      742          private static int getRequiredAlignmentForSimd(int columnType) {
      743              switch (ColumnType.tagOf(columnType)) {
      744                  // Types using Simd<i64, 8> or Simd<f64, 8>
      745                  case ColumnType.LONG:
      746                  case ColumnType.DOUBLE:
      747                  case ColumnType.TIMESTAMP:
      748                  case ColumnType.DATE:
      749                      return 8;
      750                  // Types using Simd<i32, 16> or Simd<f32, 16>
      751                  case ColumnType.INT:
      752                  case ColumnType.FLOAT:
      753                  case ColumnType.SYMBOL:
      754                      return 4;
      755                  // All other types use scalar paths - no SIMD alignment required
      756                  default:
      757                      return 1;
      758              }
      759          }

      761          /**
      762           * Checks if the given address is properly aligned for SIMD operations.
      763           * Only types that use SIMD in Rust parquet encoder require alignment:
      764           * - LONG, DOUBLE, TIMESTAMP: 8-byte alignment (Simd<i64, 8>, Simd<f64, 8>)
      765           * - INT, FLOAT, SYMBOL: 4-byte alignment (Simd<i32, 16>, Simd<f32, 16>)
      766           * Other types (SHORT, BYTE, etc.) use scalar paths and don't require SIMD alignment.
      767           */
      768          private static boolean isAlignedForColumnType(long address, int columnType) {
      769              int alignment = getRequiredAlignmentForSimd(columnType);
      770              return alignment <= 1 || (address & (alignment - 1)) == 0;
      771          }
