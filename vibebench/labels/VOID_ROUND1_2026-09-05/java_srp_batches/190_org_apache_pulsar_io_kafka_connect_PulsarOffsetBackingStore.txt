       88      void readToEnd(CompletableFuture<Void> future) {
       89          synchronized (this) {
       90              if (outstandingReadToEnd != null) {
       91                  outstandingReadToEnd.whenComplete((result, cause) -> {
       92                      if (null != cause) {
       93                          future.completeExceptionally(cause);
       94                      } else {
       95                          future.complete(result);
       96                      }
       97                  });
       98                  // return if the outstanding read has been issued
       99                  return;
      100              } else {
      101                  outstandingReadToEnd = future;
      102                  future.whenComplete((result, cause) -> {
      103                      synchronized (PulsarOffsetBackingStore.this) {
      104                          outstandingReadToEnd = null;
      105                      }
      106                  });
      107              }
      108          }
      109          producer.flushAsync().whenComplete((ignored, cause) -> {
      110              if (null != cause) {
      111                  future.completeExceptionally(cause);
      112              } else {
      113                  checkAndReadNext(future);
      114              }
      115          });
      116      }

      118      private void checkAndReadNext(CompletableFuture<Void> endFuture) {
      119          reader.hasMessageAvailableAsync().whenComplete((hasMessageAvailable, cause) -> {
      120              if (null != cause) {
      121                  endFuture.completeExceptionally(cause);
      122              } else {
      123                  if (hasMessageAvailable) {
      124                      readNext(endFuture);

      143      void processMessage(Message<byte[]> message) {
      144          if (message.getKey() != null) {
      145              data.put(
      146                  ByteBuffer.wrap(message.getKey().getBytes(UTF_8)),
      147                  ByteBuffer.wrap(message.getValue()));
      148          } else {
      149              log.debug("Got message without key from the offset storage topic, skip it. message value: {}",
      150                      message.getValue());
      151          }

      226      @Override
      227      public Future<Void> set(Map<ByteBuffer, ByteBuffer> values, Callback<Void> callback) {
      228          values.forEach((key, value) -> {
      229              ByteBuf bb = Unpooled.wrappedBuffer(key);
      230              byte[] keyBytes = ByteBufUtil.getBytes(bb);
      231              byte[] valBytes = null;
      232              if (value != null) {
      233                  bb = Unpooled.wrappedBuffer(value);
      234                  valBytes = ByteBufUtil.getBytes(bb);
      235              } else {
      236                  // It does not actually matter if it is earliest or latest.
      237                  // The connector that provides null offsets works with the
      238                  // system that cannot seek to the offset anyway.
      239                  // Just need to store something to keep the offset store happy.
      240                  valBytes = MessageId.earliest.toByteArray();
      241              }
      242              producer.newMessage()
      243                  .key(new String(keyBytes, UTF_8))
      244                  .value(valBytes)
      245                  .sendAsync();
      246          });
      247          return producer.flushAsync().whenComplete((ignored, cause) -> {
      248              if (null != callback) {
      249                  callback.onCompletion(cause, ignored);
      250              }
      251              if (null == cause) {
      252                  readToEnd(new CompletableFuture<>());
      253              }
      254          });
      255      }
