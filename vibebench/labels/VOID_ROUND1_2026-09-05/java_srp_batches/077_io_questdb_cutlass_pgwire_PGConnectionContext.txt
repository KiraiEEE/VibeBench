      393      public void handleClientOperation(int operation) throws Exception {
      394          assert authenticator != null;
      395  
      396          try {
      397              handleTlsRequest();
      398              if (tlsSessionStarting) {
      399                  flushRemainingBuffer();
      400                  tlsSessionStarting = false;
      401                  try {
      402                      socket.startTlsSession(null);
      403                  } catch (TlsSessionInitFailedException e) {
      404                      LOG.error().$("failed to create new TLS session").$((Throwable) e).$();
      405                      throw PGMessageProcessingException.INSTANCE;
      406                  }
      407                  // Start listening for read.
      408                  throw PeerIsSlowToWriteException.INSTANCE;
      409              }
      410              handleAuthentication();
      411          } catch (PeerDisconnectedException | PeerIsSlowToReadException | PeerIsSlowToWriteException e) {
      412              // BAU, not error metric
      413              throw e;
      414          } catch (PGMessageProcessingException bpe) {
      415              shutdownSocketGracefully();
      416              throw bpe; // request disconnection
      417          } catch (Throwable th) {
      418              metrics.pgWireMetrics().getErrorCounter().inc();
      419              throw th;
      420          }
      421  
      422          flushRemainingBuffer();
      423          if (resumeCallback != null) {
      424              resumeCallback.resume();
      425          }
      426  
      427          long readOffsetBeforeParse = -1;
      428          // exit from this loop is via exception when either need wait to read / write from socket
      429          // or disconnection is detected / requested
      430          //noinspection InfiniteLoopStatement
      431          while (true) {
      432              // Read more from socket or throw when
      433              if (
      434                  // - parsing stalls, e.g. readOffsetBeforeParse == recvBufferReadOffset
      435                      readOffsetBeforeParse == recvBufferReadOffset
      436                              // - recv buffer is empty
      437                              || recvBufferReadOffset == recvBufferWriteOffset
      438                              // - socket is signalled ready to read at the first iteration of this loop
      439                              || (operation == IOOperation.READ && readOffsetBeforeParse == -1)) {
      440                  // free up recv buffer
      441                  if (!freezeRecvBuffer) {
      442                      if (recvBufferReadOffset == recvBufferWriteOffset) {
      443                          clearRecvBuffer();
      444                      } else if (recvBufferReadOffset > 0) {
      445                          // nothing changed?
      446                          // shift to start
      447                          shiftReceiveBuffer(recvBufferReadOffset);
      448                      }
      449                  }
      450                  recv();
      451              }
      452  
      453              // Parse will update the value of recvBufferOffset upon completion of
      454              // logical block. We cannot count on return value because 'parse' may try to
      455              // respond to client and fail with exception. When it does fail we would have
      456              // to retry 'send' but not parse the same input again
      457              readOffsetBeforeParse = recvBufferReadOffset;
      458              totalReceived += (recvBufferWriteOffset - recvBufferReadOffset);
      459              try {
      460                  parseMessage(recvBuffer + recvBufferReadOffset, (int) (recvBufferWriteOffset - recvBufferReadOffset));
      461              } catch (PGMessageProcessingException e) {
      462                  // Special handling for access control errors: to distinguish them from other parsing errors for better diagnostics
      463                  if (!Chars.startsWith(e.getFlyweightMessage(), "Access")) {
      464                      LOG.error().$safe(e.getFlyweightMessage()).I$();
      465                  } else {
      466                      LOG.error().$("failed to parse message [err: `").$safe(e.getFlyweightMessage()).$("`]").$();
      467                  }
      468                  // ignore, we are interrupting the current message processing, but have to continue processing other
      469                  // messages
      470              }
      471          }
      472      }
