      221      private static SSLContext sslContextFromHttpCaCrt(File file) {
      222          try(InputStream in = new FileInputStream(file)) {
      223              CertificateFactory cf = CertificateFactory.getInstance("X.509");
      224              Certificate certificate = cf.generateCertificate(in);
      225  
      226              final KeyStore keyStore = KeyStore.getInstance(KeyStore.getDefaultType());
      227              keyStore.load(null, null);
      228              keyStore.setCertificateEntry("elasticsearch-ca", certificate);
      229  
      230              TrustManagerFactory tmf = TrustManagerFactory.getInstance(TrustManagerFactory.getDefaultAlgorithm());
      231              tmf.init(keyStore);
      232  
      233              SSLContext sslContext = SSLContext.getInstance("TLS");
      234              sslContext.init(null, tmf.getTrustManagers(), null);
      235              return sslContext;
      236          } catch (CertificateException | NoSuchAlgorithmException | KeyManagementException | KeyStoreException | IOException e) {
      237              throw new RuntimeException(e);
      238          }
      239      }

      247          if (version != null) {
      248              return version;
      249          }
      250          logger.debug("get version");
      251          String response = httpGet(null);
      252          // We parse the response
      253          DocumentContext document = parseJsonAsDocumentContext(response);
      254          // Cache the version and the major version
      255          version = document.read("$.version.number");
      256          majorVersion = extractMajorVersion(version);
      257          minorVersion = extractMinorVersion(version);
      258          serverless = false;
      259  
      260          try {
      261              if ("serverless".equals(document.read("$.version.build_flavor"))) {
      262                  logger.debug("We are running on Elastic serverless cloud so we can not consider version number.");
      263                  serverless = true;
      264                  version = "serverless";
      265                  majorVersion = 99;
      266                  minorVersion = 999;
      267              }
      268          } catch (PathNotFoundException e) {
      269              // We are not running an official Elasticsearch version
      270              logger.info("FSCrawler only supports Elasticsearch official distribution. " +
      271                      "You are running another custom distribution. We will not be able to use some features like " +
      272                      "semantic search.");
      273          }
      274          logger.debug("get version returns {} and {} as the major version number", version, majorVersion);
      275          return version;

      279      public String getLicense() throws ElasticsearchClientException {
      280          if (license != null) {
      281              return license;
      282          }
      283  
      284          // License endpoint might not be ready in IT so we retry with exponential wait time up to 1 minute
      285          int retries = 0;
      286          int maxRetries = 5;
      287          int waitTime = 1000;
      288          while (retries < maxRetries) {
      289              try {
      290                  return getLicenseInternal();
      291              } catch (NotFoundException e) {
      292                  logger.warn("License endpoint is not ready yet. Retrying in {}ms", waitTime);
      293                  try {
      294                      Thread.sleep(waitTime);
      295                  } catch (InterruptedException ex) {
      296                      Thread.currentThread().interrupt();
      297                  }
      298                  waitTime *= 2;
      299                  retries++;
      300              }
      301          }
      302  
      303          throw new ElasticsearchClientException("License endpoint is not ready after " + maxRetries + " retries");
      304      }

      420          } catch (InterruptedException e) {

      422              throw new ElasticsearchClientException("Await interrupted while waiting for healthy index [" +

      427      private String catIndicesHealth(String index) {
      428          try {
      429              String response = httpGet("_cat/indices/" + index,
      430                      new AbstractMap.SimpleImmutableEntry<>("h", "health"));
      431              DocumentContext document = parseJsonAsDocumentContext(response);
      432              String health = document.read("$[0].health");
      433              logger.trace("index [{}] health: [{}]", index, health);
      434              return health;
      435          } catch (WebApplicationException e) {
      436              if (e.getResponse().getStatus() == Response.Status.NOT_FOUND.getStatusCode()) {
      437                  logger.debug("Index [{}] not found yet", index);
      438                  return null;
      439              }
      440              throw e;
      441          } catch (ElasticsearchClientException e) {
      442              throw new RuntimeException(e);
      443          }
      444      }
