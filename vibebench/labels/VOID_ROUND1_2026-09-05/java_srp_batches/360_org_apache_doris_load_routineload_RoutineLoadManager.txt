      206                      throws UserException {

      329      public void pauseRoutineLoadJob(PauseRoutineLoadCommand pauseRoutineLoadCommand)
      330              throws UserException {
      331          List<RoutineLoadJob> jobs = Lists.newArrayList();
      332          // it needs lock when getting routine load job,
      333          // otherwise, it may cause the editLog out of order in the following scenarios:
      334          // thread A: create job and record job meta
      335          // thread B: change job state and persist in editlog according to meta
      336          // thread A: persist in editlog
      337          // which will cause the null pointer exception when replaying editLog
      338          readLock();
      339          try {
      340              if (pauseRoutineLoadCommand.isAll()) {
      341                  jobs = checkPrivAndGetAllJobs(pauseRoutineLoadCommand.getDbFullName());
      342              } else {
      343                  RoutineLoadJob routineLoadJob = checkPrivAndGetJob(pauseRoutineLoadCommand.getDbFullName(),
      344                          pauseRoutineLoadCommand.getLabel());
      345                  jobs.add(routineLoadJob);
      346              }
      347          } finally {
      348              readUnlock();
      349          }
      350  
      351          for (RoutineLoadJob routineLoadJob : jobs) {
      352              try {
      353                  routineLoadJob.updateState(RoutineLoadJob.JobState.PAUSED,
      354                      new ErrorReason(InternalErrorCode.MANUAL_PAUSE_ERR,
      355                          "User " + ConnectContext.get().getQualifiedUser() + " pauses routine load job"),
      356                          false /* not replay */);
      357                  LOG.info(new LogBuilder(LogKey.ROUTINE_LOAD_JOB, routineLoadJob.getId()).add("current_state",
      358                          routineLoadJob.getState()).add("user", ConnectContext.get().getQualifiedUser()).add("msg",
      359                          "routine load job has been paused by user").build());
      360              } catch (UserException e) {
      361                  LOG.warn("failed to pause routine load job {}", routineLoadJob.getName(), e);
      362                  // if user want to pause a certain job and failed, return error.
      363                  // if user want to pause all possible jobs, skip error jobs.
      364                  if (!pauseRoutineLoadCommand.isAll()) {
      365                      throw e;
      366                  }
      367              }
      368          }
      369      }
