      186          final String retryMarker = " // ";
      187          final String currentStatus = Settings.getLastLoginErrorGC() == null || Settings.getLastLoginErrorGC().first == null ? "" : Settings.getLastLoginErrorGC().first;
      188          if (!retry && currentStatus.endsWith(retryMarker)) {
      189              Settings.setLastLoginErrorGC(currentStatus + status);
      190          } else {
      191              Settings.setLastLoginErrorGC(status + retryMarker);

      196      @WorkerThread
      197      private StatusCode loginInternal(final boolean retry, @NonNull final Credentials credentials) {
      198          final Context ctx = CgeoApplication.getInstance();
      199  
      200          if (credentials.isInvalid()) {
      201              clearLoginInfo();
      202              logLastLoginError(ctx.getString(R.string.err_auth_gc_missing_login), retry);
      203              return resetGcCustomDate(StatusCode.NO_LOGIN_INFO_STORED);
      204          }
      205  
      206          final String username = credentials.getUserName();
      207  
      208          setActualStatus(LocalizationUtils.getString(R.string.init_login_popup_working));
      209          try {
      210              final String tryLoggedInData = getLoginPage();
      211  
      212              if (StringUtils.isBlank(tryLoggedInData)) {
      213                  logLastLoginError(ctx.getString(R.string.err_auth_gc_loginpage1), retry);
      214                  return StatusCode.CONNECTION_FAILED_GC; // no login page
      215              }
      216  
      217              if (getLoginStatus(tryLoggedInData)) {
      218                  Log.i("Already logged in Geocaching.com as " + username + " (" + Settings.getGCMemberStatus() + ')');
      219                  return completeLoginProcess();
      220              }
      221  
      222              final String requestVerificationToken = extractRequestVerificationToken(tryLoggedInData);
      223              if (StringUtils.isEmpty(requestVerificationToken)) {
      224                  logLastLoginError(ctx.getString(R.string.err_auth_gc_verification_token), retry, tryLoggedInData);
      225                  return StatusCode.LOGIN_PARSE_ERROR;
      226              }
      227  
      228              final String loginData = postCredentials(credentials, requestVerificationToken);
      229              if (StringUtils.isBlank(loginData)) {
      230                  logLastLoginError(ctx.getString(R.string.err_auth_gc_loginpage2), retry, requestVerificationToken);
      231                  // FIXME: should it be CONNECTION_FAILED to match the first attempt?
      232                  return StatusCode.COMMUNICATION_ERROR; // no login page
      233              }
      234  
      235              if (getLoginStatus(loginData)) {
      236                  Log.i("Successfully logged in Geocaching.com as " + username + " (" + Settings.getGCMemberStatus() + ')');
      237                  return completeLoginProcess();
      238              }
      239  
      240              if (loginData.contains("<div class=\"g-recaptcha\" data-sitekey=\"")) {
      241                  logLastLoginError(ctx.getString(R.string.err_auth_gc_captcha), retry);
      242                  return resetGcCustomDate(StatusCode.LOGIN_CAPTCHA_ERROR);
      243              }
      244  
      245              if (loginData.contains("id=\"signup-validation-error\"")) {
      246                  logLastLoginError(ctx.getString(R.string.err_auth_gc_bad_login, username), retry);
      247                  return resetGcCustomDate(StatusCode.WRONG_LOGIN_DATA); // wrong login
      248              }
      249  
      250              if (loginData.contains("content=\"account/join/success\"")) {
      251                  logLastLoginError(ctx.getString(R.string.err_auth_gc_not_validated, username), retry);
      252                  return resetGcCustomDate(StatusCode.UNVALIDATED_ACCOUNT);
      253              }
      254  
      255              logLastLoginError(ctx.getString(R.string.err_auth_gc_unknown_error, username), retry, loginData);
      256              if (retry) {
      257                  getLoginStatus(loginData);
      258                  return login(false, credentials);
      259              }
      260  
      261              logLastLoginError(ctx.getString(R.string.err_auth_gc_unknown_error_generic), retry, loginData);
      262              return resetGcCustomDate(StatusCode.UNKNOWN_ERROR); // can't login
      263          } catch (final StatusException status) {
      264              return status.statusCode;
      265          } catch (final Exception ignored) {
      266              logLastLoginError(ctx.getString(R.string.err_auth_gc_communication_error), retry);
      267              return StatusCode.CONNECTION_FAILED_GC;
      268          }
      269      }

      285      private String getResponseBodyOrStatus(final Response response) {
      286          final String body;
      287          try {
      288              body = response.body().string();
      289          } catch (final IOException ignore) {
      290              throw new StatusException(StatusCode.COMMUNICATION_ERROR);
      291          }
      292          if (response.code() == 503 && TextUtils.matches(body, GCConstants.PATTERN_MAINTENANCE)) {
      293              throw new StatusException(StatusCode.MAINTENANCE);
      294          } else if (!response.isSuccessful()) {
      295              throw new StatusException(StatusCode.COMMUNICATION_ERROR);
      296          }
      297          return body;
      298      }

      427      private ServerParameters parseServerParams(final String javascriptBody) {
      428          try {
      429              final String jsonBody = javascriptBody.subSequence(javascriptBody.indexOf('{'), javascriptBody.lastIndexOf(';')).toString();
      430              serverParameters = MAPPER.readValue(jsonBody, ServerParameters.class);
      431  
      432              if (StringUtils.isNotBlank(serverParameters.userInfo.dateFormat)) {
      433                  Log.d("Setting GCCustomDate to " + serverParameters.userInfo.dateFormat);
      434                  Settings.setGcCustomDate(serverParameters.userInfo.dateFormat);
      435              }
      436  
      437              final GCMemberState memberState = GCMemberState.fromString(serverParameters.userInfo.userType);
      438              Log.d("Setting member status to " + memberState);
      439              Settings.setGCMemberStatus(memberState);
      440  
      441              if (StringUtils.isNotBlank(serverParameters.userInfo.avatarUrl)) {
      442                  final String avatarUrl = serverParameters.userInfo.avatarUrl.replace("/avatar/", "/user/large/");
      443                  Log.d("Setting avatar to " + avatarUrl);
      444                  AvatarUtils.changeAvatar(GCConnector.getInstance(), avatarUrl);
      445              }
      446  
      447              // check for race condition while logging in
      448              if (StringUtils.isBlank(serverParameters.userInfo.userType)) {
      449                  resetServerParameters(); // not yet logged in, thus try to read again on next call
      450              }
      451          } catch (final IOException e) {
      452              Settings.setGcCustomDate(GCConstants.DEFAULT_GC_DATE);
      453              Log.e("Error loading serverparameters", e);
      454              return null;
      455          }
      456          return serverParameters;
      457      }
