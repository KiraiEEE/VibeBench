      125      private static void getReferencedTypeNames(String typeName, Set<String> referencedTypeNames) {
      126          if (StringUtils.isNotBlank(typeName) && !referencedTypeNames.contains(typeName)) {
      127              if (typeName.startsWith(ATLAS_TYPE_ARRAY_PREFIX) && typeName.endsWith(ATLAS_TYPE_ARRAY_SUFFIX)) {
      128                  int    startIdx        = ATLAS_TYPE_ARRAY_PREFIX.length();
      129                  int    endIdx          = typeName.length() - ATLAS_TYPE_ARRAY_SUFFIX.length();
      130                  String elementTypeName = typeName.substring(startIdx, endIdx);
      131  
      132                  getReferencedTypeNames(elementTypeName, referencedTypeNames);
      133              } else if (typeName.startsWith(ATLAS_TYPE_MAP_PREFIX) && typeName.endsWith(ATLAS_TYPE_MAP_SUFFIX)) {
      134                  int      startIdx      = ATLAS_TYPE_MAP_PREFIX.length();
      135                  int      endIdx        = typeName.length() - ATLAS_TYPE_MAP_SUFFIX.length();
      136                  String[] keyValueTypes = typeName.substring(startIdx, endIdx).split(ATLAS_TYPE_MAP_KEY_VAL_SEP, 2);
      137                  String   keyTypeName   = keyValueTypes.length > 0 ? keyValueTypes[0] : null;
      138                  String   valueTypeName = keyValueTypes.length > 1 ? keyValueTypes[1] : null;
      139  
      140                  getReferencedTypeNames(keyTypeName, referencedTypeNames);
      141                  getReferencedTypeNames(valueTypeName, referencedTypeNames);
      142              } else {
      143                  referencedTypeNames.add(typeName);
      144              }
      145          }

      343      public static List<AtlasTypeDefHeader> toTypeDefHeader(AtlasTypesDef typesDef) {
      344          List<AtlasTypeDefHeader> headerList = new LinkedList<>();
      345          if (CollectionUtils.isNotEmpty(typesDef.getEnumDefs())) {
      346              for (AtlasEnumDef enumDef : typesDef.getEnumDefs()) {
      347                  headerList.add(new AtlasTypeDefHeader(enumDef));
      348              }
      349          }
      350          if (CollectionUtils.isNotEmpty(typesDef.getStructDefs())) {
      351              for (AtlasStructDef structDef : typesDef.getStructDefs()) {
      352                  headerList.add(new AtlasTypeDefHeader(structDef));
      353              }
      354          }
      355          if (CollectionUtils.isNotEmpty(typesDef.getClassificationDefs())) {
      356              for (AtlasClassificationDef classificationDef : typesDef.getClassificationDefs()) {
      357                  headerList.add(new AtlasTypeDefHeader(classificationDef));
      358              }
      359          }
      360          if (CollectionUtils.isNotEmpty(typesDef.getEntityDefs())) {
      361              for (AtlasEntityDef entityDef : typesDef.getEntityDefs()) {
      362                  headerList.add(new AtlasTypeDefHeader(entityDef));
      363              }
      364          }
      365          if (CollectionUtils.isNotEmpty(typesDef.getRelationshipDefs())) {
      366              for (AtlasRelationshipDef relationshipDef : typesDef.getRelationshipDefs()) {
      367                  headerList.add(new AtlasTypeDefHeader(relationshipDef));
      368              }
      369          }
      370          if (CollectionUtils.isNotEmpty(typesDef.getBusinessMetadataDefs())) {
      371              for (AtlasBusinessMetadataDef businessMetadataDef : typesDef.getBusinessMetadataDefs()) {
      372                  headerList.add(new AtlasTypeDefHeader(businessMetadataDef));
      373              }
      374          }
      375  
      376          return headerList;
      377      }
