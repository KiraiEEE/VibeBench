       38    private void initializeAllSchemas(final TopicConvention convention) {
       39      // Create topic-to-schema mapping using TopicConvention
       40      Map<String, String> topicToSchemaNameMap = createTopicToSchemaNameMap(convention);
       41  
       42      // Initialize schemas for each topic
       43      for (Map.Entry<String, String> entry : topicToSchemaNameMap.entrySet()) {
       44        String topicName = entry.getKey();
       45        String schemaName = entry.getValue();
       46  
       47        Map<Integer, Schema> topicSchemas = new HashMap<>();
       48  
       49        // Get all schema IDs for this schema name
       50        Optional<List<Integer>> schemaIdsOpt =
       51            EventSchemaConstants.getSchemaIdsForSchemaName(schemaName);
       52        if (schemaIdsOpt.isEmpty()) {
       53          continue;
       54        }
       55  
       56        List<Integer> schemaIds = schemaIdsOpt.get();
       57  
       58        // Add schemas for each schema ID
       59        // Each schema ID represents a different version of the schema (backwards incompatible)
       60        for (int i = 0; i < schemaIds.size(); i++) {
       61          int version = calculateVersionFromIndex(i);
       62          int schemaId = schemaIds.get(i);
       63  
       64          Schema schema = getSchemaForSchemaIdFromConstants(schemaId);
       65          if (schema != null) {
       66            topicSchemas.put(version, schema);
       67  
       68            // Build version to schema ID mapping
       69            if (!_versionToSchemaIdMap.containsKey(topicName)) {
       70              _versionToSchemaIdMap.put(topicName, new HashMap<>());
       71            }
       72            _versionToSchemaIdMap.get(topicName).put(version, schemaId);
       73  
       74            // Build schema ID to topic mapping
       75            _schemaIdToTopicMap.put(schemaId, topicName);
       76          }
       77        }
       78  
       79        if (!topicSchemas.isEmpty()) {
       80          _versionedSchemaMap.put(topicName, topicSchemas);
       81  
       82          // Set compatibility based on number of versions:
       83          // - Single version schemas: BACKWARD (can evolve backward-compatibly)
       84          // - Multi-version schemas: NONE (breaking changes between versions)
       85          String compatibility =
       86              (schemaIds.size() == 1)
       87                  ? EventSchemaConstants.SCHEMA_COMPATIBILITY_BACKWARD
       88                  : EventSchemaConstants.SCHEMA_COMPATIBILITY_NONE;
       89          _schemaCompatibilityMap.put(topicName, compatibility);
       90        }
       91      }
       92    }

      362    public boolean checkSchemaCompatibility(
      363        String topicName, Schema newSchema, Schema existingSchema) {
      364      // Get the compatibility level for this topic
      365      String compatibilityLevel = getSchemaCompatibility(topicName);
      366  
      367      switch (compatibilityLevel) {
      368        case EventSchemaConstants.SCHEMA_COMPATIBILITY_NONE:
      369          // No compatibility guarantees - any change breaks compatibility
      370          return newSchema.equals(existingSchema);
      371  
      372        case EventSchemaConstants.SCHEMA_COMPATIBILITY_BACKWARD:
      373          // New schema must be backward compatible with existing data
      374          // This means new fields can be added, but existing fields cannot be removed or changed
      375          return checkBackwardCompatibility(newSchema, existingSchema);
      376  
      377        case EventSchemaConstants.SCHEMA_COMPATIBILITY_FORWARD:
      378          // New schema must be forward compatible with existing consumers
      379          // This means existing fields cannot be removed, but new fields can be added
      380          return checkForwardCompatibility(newSchema, existingSchema);
      381  
      382        case EventSchemaConstants.SCHEMA_COMPATIBILITY_FULL:
      383          // Must be both backward and forward compatible
      384          return checkBackwardCompatibility(newSchema, existingSchema)
      385              && checkForwardCompatibility(newSchema, existingSchema);
      386  
      387        default:
      388          // Unknown compatibility level, default to no compatibility
      389          return newSchema.equals(existingSchema);
      390      }
      391    }
