       51          getPageHeader(request, response.getContent());
       52          appendHelpStyle(response.getContent());
       53  
       54          queryString = request.getSingleParameter("query");
       55          if (Strings.isNullOrEmpty(queryString)) {
       56              // ATTN: according to Mysql protocol, the default query should be "contents" 
       57              //       when you want to get server side help.
       58              queryString = "contents";
       59          } else {
       60              queryString = queryString.trim();
       61          }
       62          appendHelpInfo(response.getContent());
       63  
       64          getPageFooter(response.getContent());
       65          writeResponse(request, response);
       66      }
