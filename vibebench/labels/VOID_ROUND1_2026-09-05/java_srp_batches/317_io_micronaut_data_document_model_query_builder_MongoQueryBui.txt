      407      private Map<String, Object> lookup(String from,
      408                                         List<String> localFields,
      409                                         List<String> foreignFields,
      410                                         List<Map<String, Object>> pipeline,
      411                                         String as) {
      412          if (localFields.size() != foreignFields.size()) {
      413              throw new IllegalStateException("Un-matching join columns size: " + localFields.size() + " != " + foreignFields.size() + " " + localFields + ", " + foreignFields);
      414          }
      415          if (localFields.size() == 1) {
      416              return lookup(from, localFields.iterator().next(), foreignFields.iterator().next(), pipeline, as);
      417          }
      418          List<Map<String, Object>> matches = new ArrayList<>(localFields.size());
      419          Map<String, Object> let = new LinkedHashMap<>();
      420          int i = 1;
      421          Iterator<String> foreignIt = foreignFields.iterator();
      422          for (String localField : localFields) {
      423              String var = "v" + i++;
      424              let.put(var, "$" + localField);
      425              matches.add(Map.of("$eq", Arrays.asList("$$" + var, "$" + foreignIt.next())));
      426          }
      427  
      428          Map<String, Object> match;
      429          if (matches.size() > 1) {
      430              match = Map.of("$match", Map.of("$expr", Map.of("$and", matches)));
      431          } else {
      432              match = Map.of("$match", Map.of("$expr", matches.iterator().next()));
      433          }
      434  
      435          return lookup(from, let, match, pipeline, as);
      436      }
