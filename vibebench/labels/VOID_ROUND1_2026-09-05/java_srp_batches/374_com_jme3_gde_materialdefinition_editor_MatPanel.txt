      414      /**
      415       * Enable or disable animation rendering
      416       * @param enabled true to enable continuous rendering, false to disable
      417       */
      418      public void setAnimationEnabled(boolean enabled) {
      419          this.animationEnabled = enabled;
      420          if (enabled) {
      421              animationTimer.start();
      422          } else {
      423              animationTimer.stop();
      424          }
      425      }
