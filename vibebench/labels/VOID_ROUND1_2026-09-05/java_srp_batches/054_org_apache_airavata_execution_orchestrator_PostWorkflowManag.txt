      104      private boolean process(JobStatusResult jobStatusResult) {
      105  
      106          if (jobStatusResult == null) {
      107              logger.error("Job result is null");
      108              return false;
      109          }
      110  
      111          RegistryService.Client registryClient =
      112                  wfManager.getRegistryClientPool().getResource();
      113  
      114          var jobId = jobStatusResult.getJobId();
      115          var jobName = jobStatusResult.getJobName();
      116          var jobState = jobStatusResult.getState();
      117          var publisherId = jobStatusResult.getPublisherName();
      118          logger.info("processing JobStatusUpdate<{}> from {}: {}", jobId, publisherId, jobStatusResult);
      119  
      120          try {
      121              List<JobModel> jobs = registryClient.getJobs("jobId", jobId);
      122              logger.info("Found {} jobs in registry with id={}", jobs.size(), jobId);
      123              if (!jobs.isEmpty()) {
      124                  jobs = jobs.stream()
      125                          .filter(jm -> jm.getJobName().equals(jobName))
      126                          .toList();
      127                  logger.info("Found {} jobs in registry with id={} and name={}", jobs.size(), jobId, jobName);
      128              }
      129              if (jobs.size() != 1) {
      130                  logger.error("Found {} job(s) in registry with id={} and name={}", jobs.size(), jobId, jobName);
      131                  wfManager.getRegistryClientPool().returnResource(registryClient);
      132                  return false;
      133              }
      134              JobModel jobModel = jobs.get(0);
      135              ProcessModel processModel = registryClient.getProcess(jobModel.getProcessId());
      136              ExperimentModel experimentModel = registryClient.getExperiment(processModel.getExperimentId());
      137              ProcessStatus processStatus = registryClient.getProcessStatus(processModel.getProcessId());
      138  
      139              var processState = processStatus.getState();
      140              wfManager.getRegistryClientPool().returnResource(registryClient);
      141  
      142              if (experimentModel != null) {
      143                  jobModel.getJobStatuses()
      144                          .sort(Comparator.comparingLong(JobStatus::getTimeOfStateChange)
      145                                  .reversed());
      146                  JobState currentJobStatus = jobModel.getJobStatuses().get(0).getJobState();
      147                  logger.info("Last known state of job {} is {}", jobId, jobName);
      148  
      149                  if (!JobStateValidator.isValid(currentJobStatus, jobState)) {
      150                      logger.warn("JobStatusUpdate<{}> invalid. prev={} -> new={}", jobId, currentJobStatus, jobState);
      151                      return true;
      152                  }
      153  
      154                  String task = jobModel.getTaskId();
      155                  String processId = processModel.getProcessId();
      156                  String gateway = experimentModel.getGatewayId();
      157                  String experimentId = experimentModel.getExperimentId();
      158  
      159                  logger.info(
      160                          "saving JobStatusUpdate<{}>: pid={}, eid={}, gw={}, state={}",
      161                          jobId,
      162                          processId,
      163                          experimentId,
      164                          gateway,
      165                          jobState);
      166                  saveAndPublishJobStatus(jobId, task, processId, experimentId, gateway, jobState);
      167  
      168                  // TODO get cluster lock before that
      169                  if (ProcessState.CANCELLING.equals(processState) || ProcessState.CANCELED.equals(processState)) {
      170                      logger.info("Cancelled post workflow for process {} in experiment {}", processId, experimentId);
      171                      // This will mark a canceling Experiment with CANCELED status for a set of valid job statuses
      172                      // This is a safety check. Cancellation is originally handled in Job Cancellation Workflow
      173                      switch (jobState) {
      174                          case FAILED:
      175                          case SUSPENDED:
      176                          case CANCELED:
      177                          case COMPLETE:
      178                              logger.info("canceled job={}: eid={}, state={}", jobId, experimentId, jobState);
      179                              wfManager.publishProcessStatus(processId, experimentId, gateway, ProcessState.CANCELED);
      180                              break;
      181                          default:
      182                              logger.warn("skipping job={}: eid={}, state={}", jobId, experimentId, jobState);
      183                      }
      184                  } else {
      185                      logger.info("Job {} is in state={}", jobId, jobState);
      186                      if (jobState == JobState.COMPLETE || jobState == JobState.FAILED) {
      187                          // If Job has FAILED, still run output staging tasks to debug the reason for failure. And
      188                          // update the experiment status as COMPLETED as job failures are unrelated to Airavata scope.
      189                          logger.info("running PostWorkflow for process {} of experiment {}", processId, experimentId);
      190                          executePostWorkflow(processId, gateway, false);
      191  
      192                      } else if (jobStatusResult.getState() == JobState.CANCELED) {
      193                          logger.info("Setting process {} of experiment {} to state=CANCELED", processId, experimentId);
      194                          wfManager.publishProcessStatus(processId, experimentId, gateway, ProcessState.CANCELED);
      195                      }
      196                  }
      197                  return true;
      198              } else {
      199                  logger.warn("Could not find a monitoring register for job id {}", jobId);
      200                  return false;
      201              }
      202          } catch (Exception e) {
      203              logger.error(
      204                      "Failed to process job: {}, with status : {}",
      205                      jobStatusResult.getJobId(),
      206                      jobStatusResult.getState().name(),
      207                      e);
      208              wfManager.getRegistryClientPool().returnBrokenResource(registryClient);
      209              return false;
      210          }
      211      }

      346      private void startServer() throws Exception {
      347  
      348          init();
      349          final Consumer<String, JobStatusResult> consumer = createConsumer();
      350          new Thread(() -> {
      351                      while (true) {
      352                          final ConsumerRecords<String, JobStatusResult> consumerRecords = consumer.poll(Long.MAX_VALUE);
      353                          var executorCompletionService = new ExecutorCompletionService<>(processingPool);
      354                          var processingFutures = new ArrayList<>();
      355  
      356                          for (var topicPartition : consumerRecords.partitions()) {
      357                              var partitionRecords = consumerRecords.records(topicPartition);
      358                              logger.info("Received job records {}", partitionRecords.size());
      359  
      360                              for (var record : partitionRecords) {
      361                                  var topic = topicPartition.topic();
      362                                  var partition = topicPartition.partition();
      363                                  var key = record.key();
      364                                  var value = record.value();
      365                                  logger.info("received post on {}/{}: {}->{}", topic, partition, key, value);
      366                                  logger.info(
      367                                          "Submitting {} to process in thread pool",
      368                                          record.value().getJobId());
      369  
      370                                  // This avoids kafka read thread to wait until processing is completed before committing
      371                                  // There is a risk of missing 20 messages in case of a restart, but this improves the
      372                                  // robustness of the kafka read thread by avoiding wait timeouts
      373                                  processingFutures.add(executorCompletionService.submit(() -> {
      374                                      boolean success = process(record.value());
      375                                      logger.info(
      376                                              "Status of processing {} : {}",
      377                                              record.value().getJobId(),
      378                                              success);
      379                                      return success;
      380                                  }));
      381  
      382                                  consumer.commitSync(Collections.singletonMap(
      383                                          topicPartition, new OffsetAndMetadata(record.offset() + 1)));
      384                              }
      385                          }
      386  
      387                          for (var f : processingFutures) {
      388                              try {
      389                                  executorCompletionService.take().get();
      390                              } catch (Exception e) {
      391                                  logger.error("Failed processing job", e);
      392                              }
      393                          }
      394                          logger.info("All messages processed. Moving to next round");
      395                      }
      396                  })
      397                  .start();
      398      }

      400      private void saveAndPublishJobStatus(
      401              String jobId, String taskId, String processId, String experimentId, String gateway, JobState jobState)
      402              throws Exception {

      404  
      405              JobStatus jobStatus = new JobStatus();

      407              jobStatus.setTimeOfStateChange(AiravataUtils.getCurrentTimestamp().getTime());
      408              jobStatus.setJobState(jobState);
      409  
      410              if (jobStatus.getTimeOfStateChange() == 0 || jobStatus.getTimeOfStateChange() > 0) {
      411                  jobStatus.setTimeOfStateChange(
      412                          AiravataUtils.getCurrentTimestamp().getTime());
      413              } else {
      414                  jobStatus.setTimeOfStateChange(jobStatus.getTimeOfStateChange());
      415              }
      416  
      417              RegistryService.Client registryClient =
      418                      wfManager.getRegistryClientPool().getResource();
      419  
      420              try {
      421                  registryClient.addJobStatus(jobStatus, taskId, jobId);
      422                  wfManager.getRegistryClientPool().returnResource(registryClient);
      423  
      424              } catch (Exception e) {
      425                  logger.error("Failed to add job status " + jobId, e);
      426                  wfManager.getRegistryClientPool().returnBrokenResource(registryClient);
      427              }
      428  
      429              JobIdentifier identifier = new JobIdentifier(jobId, taskId, processId, experimentId, gateway);
      430  
      431              JobStatusChangeEvent jobStatusChangeEvent = new JobStatusChangeEvent(jobStatus.getJobState(), identifier);
      432              MessageContext msgCtx = new MessageContext(
      433                      jobStatusChangeEvent, MessageType.JOB, AiravataUtils.getId(MessageType.JOB.name()), gateway);
      434              msgCtx.setUpdatedTime(AiravataUtils.getCurrentTimestamp());

      437          } catch (Exception e) {
