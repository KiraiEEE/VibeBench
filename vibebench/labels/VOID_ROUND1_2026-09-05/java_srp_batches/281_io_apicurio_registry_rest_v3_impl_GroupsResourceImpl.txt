      189      /**
      190       * Builds a reference graph starting from the given artifact version.
      191       */
      192      private ReferenceGraph buildReferenceGraph(String groupId, String artifactId, String version,
      193              ArtifactVersionMetaDataDto rootMetadata, ReferenceGraphDirection direction, int maxDepth) {
      194  
      195          // Track visited nodes (version-level) to avoid processing the same node twice
      196          Set<String> visited = new HashSet<>();
      197          // Track current path of artifacts for cycle detection (artifact-level)
      198          // A cycle exists only if we encounter the same artifact in the current traversal path
      199          Set<String> currentPath = new HashSet<>();
      200          Set<String> cycleNodes = new HashSet<>();
      201  
      202          // Create the root node
      203          String rootNodeId = createNodeId(groupId, artifactId, version);
      204          String rootArtifactKey = createArtifactKey(groupId, artifactId);
      205          ReferenceGraphNode rootNode = ReferenceGraphNode.builder()
      206                  .id(rootNodeId)
      207                  .groupId(groupId != null ? groupId : "default")
      208                  .artifactId(artifactId)
      209                  .version(version)
      210                  .artifactType(rootMetadata.getArtifactType())
      211                  .name(rootMetadata.getName())
      212                  .isRoot(true)
      213                  .isCycleNode(false)
      214                  .build();
      215  
      216          // Collect all nodes and edges
      217          List<ReferenceGraphNode> nodes = new java.util.ArrayList<>();
      218          List<ReferenceGraphEdge> edges = new java.util.ArrayList<>();
      219          nodes.add(rootNode);
      220          visited.add(rootNodeId);
      221          currentPath.add(rootArtifactKey);
      222  
      223          // Track the actual max depth reached
      224          int[] actualMaxDepth = {0};
      225  
      226          // Traverse the graph
      227          if (direction == ReferenceGraphDirection.OUTBOUND || direction == ReferenceGraphDirection.BOTH) {
      228              traverseOutboundReferences(groupId, artifactId, version, rootNodeId, nodes, edges,
      229                      visited, currentPath, cycleNodes, 1, maxDepth, actualMaxDepth);
      230          }
      231          if (direction == ReferenceGraphDirection.INBOUND || direction == ReferenceGraphDirection.BOTH) {
      232              traverseInboundReferences(groupId, artifactId, version, rootNodeId, nodes, edges,
      233                      visited, currentPath, cycleNodes, 1, maxDepth, actualMaxDepth);
      234          }
      235  
      236          // Mark cycle nodes
      237          for (ReferenceGraphNode node : nodes) {
      238              if (cycleNodes.contains(node.getId())) {
      239                  node.setIsCycleNode(true);
      240              }
      241          }
      242  
      243          // Build metadata
      244          ReferenceGraphMetadata metadata = ReferenceGraphMetadata.builder()
      245                  .totalNodes(nodes.size())
      246                  .totalEdges(edges.size())
      247                  .maxDepth(actualMaxDepth[0])
      248                  .hasCycles(!cycleNodes.isEmpty())
      249                  .build();
      250  
      251          return ReferenceGraph.builder()
      252                  .root(rootNode)
      253                  .nodes(nodes)
      254                  .edges(edges)
      255                  .metadata(metadata)
      256                  .build();
      257      }

      259      /**
      260       * Traverses outbound references (artifacts that this version references).
      261       */
      262      private void traverseOutboundReferences(String groupId, String artifactId, String version,
      263              String sourceNodeId, List<ReferenceGraphNode> nodes, List<ReferenceGraphEdge> edges,
      264              Set<String> visited, Set<String> currentPath, Set<String> cycleNodes,
      265              int currentDepth, int maxDepth, int[] actualMaxDepth) {
      266  
      267          if (currentDepth > maxDepth) {
      268              return;
      269          }
      270  
      271          try {
      272              StoredArtifactVersionDto content = storage.getArtifactVersionContent(groupId, artifactId, version);
      273              List<ArtifactReferenceDto> references = content.getReferences();
      274  
      275              if (references == null || references.isEmpty()) {
      276                  return;
      277              }
      278  
      279              actualMaxDepth[0] = Math.max(actualMaxDepth[0], currentDepth);
      280  
      281              for (ArtifactReferenceDto ref : references) {
      282                  String refGroupId = ref.getGroupId();
      283                  String refArtifactId = ref.getArtifactId();
      284                  String refVersion = ref.getVersion();
      285                  String targetNodeId = createNodeId(refGroupId, refArtifactId, refVersion);
      286                  String targetArtifactKey = createArtifactKey(refGroupId, refArtifactId);
      287  
      288                  // Add edge
      289                  ReferenceGraphEdge edge = ReferenceGraphEdge.builder()
      290                          .sourceNodeId(sourceNodeId)
      291                          .targetNodeId(targetNodeId)
      292                          .name(ref.getName())
      293                          .build();
      294                  edges.add(edge);
      295  
      296                  // Check for cycle: same artifact appears in current path (not just visited globally)
      297                  boolean isArtifactCycle = currentPath.contains(targetArtifactKey);
      298                  if (isArtifactCycle) {
      299                      cycleNodes.add(targetNodeId);
      300                      cycleNodes.add(sourceNodeId);
      301                  }
      302  
      303                  // Skip processing if this exact version was already visited (avoid duplicate nodes)
      304                  if (visited.contains(targetNodeId)) {
      305                      continue;
      306                  }
      307  
      308                  // Add node
      309                  visited.add(targetNodeId);
      310                  try {
      311                      ArtifactVersionMetaDataDto refMetadata = storage.getArtifactVersionMetaData(
      312                              refGroupId, refArtifactId, refVersion);
      313  
      314                      ReferenceGraphNode node = ReferenceGraphNode.builder()
      315                              .id(targetNodeId)
      316                              .groupId(refGroupId != null ? refGroupId : "default")
      317                              .artifactId(refArtifactId)
      318                              .version(refVersion)
      319                              .artifactType(refMetadata.getArtifactType())
      320                              .name(refMetadata.getName())
      321                              .isRoot(false)
      322                              .isCycleNode(isArtifactCycle)
      323                              .build();
      324                      nodes.add(node);
      325  
      326                      // Recursively traverse (but don't continue if this is a cycle node)
      327                      if (!isArtifactCycle) {
      328                          // Add to current path before recursing, remove after (backtracking)
      329                          currentPath.add(targetArtifactKey);
      330                          traverseOutboundReferences(refGroupId, refArtifactId, refVersion, targetNodeId,
      331                                  nodes, edges, visited, currentPath, cycleNodes,
      332                                  currentDepth + 1, maxDepth, actualMaxDepth);
      333                          currentPath.remove(targetArtifactKey);
      334                      }
      335                  } catch (Exception e) {
      336                      // Reference might not exist, add a placeholder node
      337                      ReferenceGraphNode node = ReferenceGraphNode.builder()
      338                              .id(targetNodeId)
      339                              .groupId(refGroupId != null ? refGroupId : "default")
      340                              .artifactId(refArtifactId)
      341                              .version(refVersion)
      342                              .isRoot(false)
      343                              .isCycleNode(isArtifactCycle)
      344                              .build();
      345                      nodes.add(node);
      346                  }
      347              }
      348          } catch (Exception e) {
      349              // Artifact content might not be accessible
      350          }
      351      }
