      219    private int calculateMemoryAwareQueueSize(int requestedSize) {
      220      Runtime runtime = Runtime.getRuntime();
      221      long maxMemory = runtime.maxMemory();
      222      long estimatedEntitySize = 10 * 1024L;
      223      int batchSize = jobData.getBatchSize() != null ? jobData.getBatchSize() : DEFAULT_BATCH_SIZE;
      224      long maxQueueMemory = (long) (maxMemory * 0.15);
      225      int memoryBasedLimit = (int) (maxQueueMemory / (estimatedEntitySize * batchSize));
      226      return Math.min(requestedSize, Math.max(100, memoryBasedLimit));
      227    }

      239    private void reIndexFromStartToEnd() throws InterruptedException {
      240      long totalEntities = rdfIndexStats.get().getJobStats().getTotalRecords();
      241      int numProducers = Math.clamp((int) (totalEntities / 5000), 2, MAX_PRODUCER_THREADS);
      242      int numConsumers =
      243          jobData.getConsumerThreads() != null
      244              ? Math.min(jobData.getConsumerThreads(), MAX_CONSUMER_THREADS)
      245              : Math.min(3, MAX_CONSUMER_THREADS);
      246  
      247      LOG.info(
      248          "Starting RDF indexing with {} producer threads, {} consumer threads",
      249          numProducers,
      250          numConsumers);
      251  
      252      jobExecutor =
      253          Executors.newFixedThreadPool(
      254              jobData.getEntities().size(), Thread.ofPlatform().name("rdf-job-", 0).factory());
      255      producerExecutor =
      256          Executors.newFixedThreadPool(
      257              numProducers, Thread.ofPlatform().name("rdf-producer-", 0).factory());
      258      consumerExecutor =
      259          Executors.newFixedThreadPool(
      260              numConsumers, Thread.ofPlatform().name("rdf-consumer-", 0).factory());
      261  
      262      CountDownLatch consumerLatch = new CountDownLatch(numConsumers);
      263      for (int i = 0; i < numConsumers; i++) {
      264        final int consumerId = i;
      265        consumerExecutor.submit(() -> runConsumer(consumerId, consumerLatch));
      266      }
      267  
      268      try {
      269        // Clear entire RDF store before re-indexing to remove stale data
      270        LOG.info("Clearing RDF store before re-indexing");
      271        rdfRepository.clearAll();
      272  
      273        processEntityTypes();
      274        signalConsumersToStop(numConsumers);
      275        consumerLatch.await();
      276        LOG.info("All consumers have finished processing tasks");
      277      } catch (InterruptedException e) {
      278        LOG.info("Reindexing interrupted - stopping immediately");
      279        stopped = true;
      280        Thread.currentThread().interrupt();
      281        throw e;
      282      }
      283    }

      285    private void runConsumer(int consumerId, CountDownLatch consumerLatch) {
      286      LOG.info("Consumer {} started", consumerId);
      287      try {
      288        while (!stopped && (!producersDone.get() || !taskQueue.isEmpty())) {
      289          try {
      290            IndexingTask task = taskQueue.poll(100, TimeUnit.MILLISECONDS);
      291            if (task != null && !task.isPoisonPill()) {
      292              processTask(task);
      293            }
      294          } catch (InterruptedException e) {
      295            Thread.currentThread().interrupt();
      296            break;
      297          }
      298        }
      299      } finally {
      300        LOG.info("Consumer {} stopped", consumerId);
      301        consumerLatch.countDown();
      302      }
      303    }

      305    private void processTask(IndexingTask task) {
      306      String entityType = task.entityType();
      307      List<? extends EntityInterface> entities = task.entities();
      308  
      309      if (entities == null || entities.isEmpty()) {
      310        return;
      311      }
      312  
      313      int successCount = 0;
      314      int failedCount = 0;
      315  
      316      try {
      317        for (EntityInterface entity : entities) {
      318          if (stopped) {
      319            break;
      320          }
      321          try {
      322            rdfRepository.createOrUpdate(entity);
      323            successCount++;
      324          } catch (Exception e) {
      325            LOG.error("Failed to index entity {} to RDF", entity.getId(), e);
      326            failedCount++;
      327          }
      328        }
      329  
      330        processBatchRelationships(entityType, entities);
      331  
      332        // Process glossary term relations if this is a glossaryTerm batch
      333        if ("glossaryTerm".equals(entityType)) {
      334          processGlossaryTermRelations(entities);
      335        }
      336  
      337        StepStats currentStats =
      338            new StepStats().withSuccessRecords(successCount).withFailedRecords(failedCount);
      339        updateEntityStats(entityType, currentStats);
      340        sendUpdates(jobExecutionContext, false);
      341  
      342      } catch (Exception e) {
      343        LOG.error("Error processing batch for entity type {}", entityType, e);
      344        updateEntityStats(
      345            entityType,
      346            new StepStats()
      347                .withSuccessRecords(successCount)
      348                .withFailedRecords(entities.size() - successCount));
      349      }
      350    }

      367        List<EntityRelationshipObject> incomingLineage =

      371  
      372        List<org.openmetadata.schema.type.EntityRelationship> allRelationships = new ArrayList<>();
      373  
      374        for (EntityRelationshipObject rel : outgoingRelationships) {
      375          // Skip relationships to/from excluded entity types (changeEvent, auditLog, vote, etc.)
      376          // These don't provide meaningful semantic value in the knowledge graph
      377          if (EXCLUDED_RELATIONSHIP_ENTITY_TYPES.contains(rel.getToEntity())
      378              || EXCLUDED_RELATIONSHIP_ENTITY_TYPES.contains(rel.getFromEntity())) {
      379            LOG.debug(
      380                "Skipping relationship {} -> {} (excluded entity type: {} or {})",
      381                rel.getFromId(),
      382                rel.getToId(),
      383                rel.getFromEntity(),
      384                rel.getToEntity());
      385            continue;
      386          }
      387  
      388          // Skip excluded relationship types (VOTED, FOLLOWS, etc.)
      389          if (EXCLUDED_RELATIONSHIP_TYPES.contains(rel.getRelation())) {
      390            LOG.debug(
      391                "Skipping relationship {} -> {} (excluded relationship type: {})",
      392                rel.getFromId(),
      393                rel.getToId(),
      394                rel.getRelation());
      395            continue;
      396          }
      397  
      398          if (rel.getRelation() == Relationship.UPSTREAM.ordinal() && rel.getJson() != null) {
      399            processLineageRelationship(rel);

      402            // by processGlossaryTermRelations() with typed predicates
      403            if ("glossaryTerm".equals(entityType)
      404                && rel.getRelation() == Relationship.RELATED_TO.ordinal()

      406              LOG.debug(
      407                  "Skipping glossary term relation {} -> {} (handled by processGlossaryTermRelations)",
      408                  rel.getFromId(),
      409                  rel.getToId());
      410              continue;
      411            }
      412            allRelationships.add(convertToEntityRelationship(rel));
      413          }
      414        }
      415  
      416        for (EntityRelationshipObject rel : incomingLineage) {
      417          // Skip relationships to/from excluded entity types
      418          if (EXCLUDED_RELATIONSHIP_ENTITY_TYPES.contains(rel.getToEntity())
      419              || EXCLUDED_RELATIONSHIP_ENTITY_TYPES.contains(rel.getFromEntity())) {
      420            continue;
      421          }
      422  
      423          // Skip excluded relationship types
      424          if (EXCLUDED_RELATIONSHIP_TYPES.contains(rel.getRelation())) {
      425            continue;
      426          }
      427  
      428          if (rel.getJson() != null) {
      429            processLineageRelationship(rel);
      430          } else {
      431            allRelationships.add(convertToEntityRelationship(rel));
      432          }
      433        }

      435        if (!allRelationships.isEmpty()) {
      436          rdfRepository.bulkAddRelationships(allRelationships);
      437          LOG.debug(
      438              "Bulk added {} relationships for {} entities",
      439              allRelationships.size(),
      440              entities.size());
      441        }
      442  
      443      } catch (Exception e) {
      444        LOG.error("Failed to process batch relationships for entity type {}", entityType, e);
      445      }
      446    }
