      226    private Element getGroupE(Element map, String srcs, String tgts) {
      227      for (Element grp : map.getChildrenByName("group")) {
      228        if (grp.getChildValue("source").equals(srcs)) {
      229          Element tgt = grp.getNamedChild("target");
      230          if (tgt == null || tgts == null || tgts.equals(tgt.getValue())) {
      231            if (tgt == null && tgts != null)
      232              grp.makeElement("target").setValue(tgts);
      233            return grp;
      234          }
      235        }
      236      }
      237      Element grp = map.addElement("group");
      238      grp.makeElement("source").setValue(srcs);
      239      grp.makeElement("target").setValue(tgts);
      240      return grp;

      251    private ConceptMapRelationship readRelationship(FHIRLexer lexer) throws FHIRLexerException {
      252      String token = lexer.take();
      253      if (token.equals("-"))
      254        return ConceptMapRelationship.RELATEDTO;
      255      if (token.equals("=")) // temporary
      256        return ConceptMapRelationship.RELATEDTO;
      257      if (token.equals("=="))
      258        return ConceptMapRelationship.EQUIVALENT;
      259      if (token.equals("!="))
      260        return ConceptMapRelationship.NOTRELATEDTO;
      261      if (token.equals("<="))
      262        return ConceptMapRelationship.SOURCEISNARROWERTHANTARGET;
      263      if (token.equals(">="))
      264        return ConceptMapRelationship.SOURCEISBROADERTHANTARGET;
      265      throw lexer.error("Unknown relationship token '" + token + "'");
      266    }
