      318    public static void clearWork(Configuration conf) {
      319      Path mapPath = getPlanPath(conf, MAP_PLAN_NAME);
      320      Path reducePath = getPlanPath(conf, REDUCE_PLAN_NAME);
      321  
      322      // if the plan path hasn't been initialized just return, nothing to clean.
      323      if (mapPath == null && reducePath == null) {
      324        return;
      325      }
      326  
      327  
      328      try {
      329        if (!HiveConf.getBoolVar(conf, ConfVars.HIVE_RPC_QUERY_PLAN)) {
      330          FileSystem fs = mapPath.getFileSystem(conf);
      331          try {
      332            fs.delete(mapPath, true);
      333          } catch (FileNotFoundException e) {
      334            // delete if exists, don't panic if it doesn't
      335          }
      336          try {
      337            fs.delete(reducePath, true);
      338          } catch (FileNotFoundException e) {
      339            // delete if exists, don't panic if it doesn't
      340          }
      341        }
      342      } catch (Exception e) {
      343        LOG.warn("Failed to clean-up tmp directories.", e);
      344      } finally {
      345        // where a single process works with multiple plans - we must clear
      346        // the cache before working with the next plan.
      347        clearWorkMapForConf(conf);
      348      }
      349    }

      438     * Returns the Map or Reduce plan
      439     * Side effect: the BaseWork returned is also placed in the gWorkMap
      440     * @param conf
      441     * @param name
      442     * @return BaseWork based on the name supplied will return null if name is null
      443     * @throws RuntimeException if the configuration files are not proper or if plan can not be loaded
      444     */
      445    private static BaseWork getBaseWork(Configuration conf, String name) {
      446  
      447      Path path = getPlanPath(conf, name);
      448      LOG.debug("PLAN PATH = {}", path);
      449      if (path == null) { // Map/reduce plan may not be generated
      450        return null;
      451      }
      452  
      453      BaseWork gWork = gWorkMap.get(conf).get(path);
      454      if (gWork != null) {
      455        LOG.debug("Found plan in cache for name: {}", name);
      456        return gWork;
      457      }
      458  
      459      InputStream in = null;
      460      Kryo kryo = SerializationUtilities.borrowKryo();
      461      try {
      462        String engine = HiveConf.getVar(conf, ConfVars.HIVE_EXECUTION_ENGINE);
      463        Path localPath = path;
      464        LOG.debug("local path = {}", localPath);
      465        final long serializedSize;
      466        final String planMode;
      467        if (HiveConf.getBoolVar(conf, ConfVars.HIVE_RPC_QUERY_PLAN)) {
      468          String planStringPath = path.toUri().getPath();
      469          LOG.debug("Loading plan from string: {}", planStringPath);
      470          String planString = conf.getRaw(planStringPath);
      471          if (planString == null) {
      472            LOG.info("Could not find plan string in conf");
      473            return null;
      474          }
      475          serializedSize = planString.length();
      476          planMode = "RPC";
      477          byte[] planBytes = Base64.getDecoder().decode(planString);
      478          in = new ByteArrayInputStream(planBytes);
      479          in = new InflaterInputStream(in);
      480        } else {
      481          LOG.debug("Open file to read in plan: {}", localPath);
      482          FileSystem fs = localPath.getFileSystem(conf);
      483          in = fs.open(localPath);
      484          serializedSize = fs.getFileStatus(localPath).getLen();
      485          planMode = "FILE";
      486        }
      487  
      488        if(MAP_PLAN_NAME.equals(name)){
      489          if (ExecMapper.class.getName().equals(conf.get(MAPRED_MAPPER_CLASS))){
      490            gWork = SerializationUtilities.deserializePlan(kryo, in, MapWork.class);
      491          } else if(MergeFileMapper.class.getName().equals(conf.get(MAPRED_MAPPER_CLASS))) {
      492            gWork = SerializationUtilities.deserializePlan(kryo, in, MergeFileWork.class);
      493          } else if(ColumnTruncateMapper.class.getName().equals(conf.get(MAPRED_MAPPER_CLASS))) {
      494            gWork = SerializationUtilities.deserializePlan(kryo, in, ColumnTruncateWork.class);
      495          } else {
      496            throw new RuntimeException("unable to determine work from configuration ."
      497                + MAPRED_MAPPER_CLASS + " was "+ conf.get(MAPRED_MAPPER_CLASS));
      498          }
      499        } else if (REDUCE_PLAN_NAME.equals(name)) {
      500          if(ExecReducer.class.getName().equals(conf.get(MAPRED_REDUCER_CLASS))) {
      501            gWork = SerializationUtilities.deserializePlan(kryo, in, ReduceWork.class);
      502          } else {
      503            throw new RuntimeException("unable to determine work from configuration ."
      504                + MAPRED_REDUCER_CLASS +" was "+ conf.get(MAPRED_REDUCER_CLASS));
      505          }
      506        } else if (name.contains(MERGE_PLAN_NAME)) {
      507          if (name.startsWith(MAPNAME)) {
      508            gWork = SerializationUtilities.deserializePlan(kryo, in, MapWork.class);
      509          } else if (name.startsWith(REDUCENAME)) {
      510            gWork = SerializationUtilities.deserializePlan(kryo, in, ReduceWork.class);
      511          } else {
      512            throw new RuntimeException("Unknown work type: " + name);
      513          }
      514        }
      515        LOG.info("Deserialized plan (via {}) - name: {} size: {}", planMode,
      516            gWork.getName(), humanReadableByteCount(serializedSize));
      517        gWorkMap.get(conf).put(path, gWork);
      518        return gWork;
      519      } catch (FileNotFoundException fnf) {
      520        // happens. e.g.: no reduce work.
      521        LOG.debug("No plan file found: {}", path, fnf);
      522        return null;
      523      } catch (Exception e) {
      524        String msg = "Failed to load plan: " + path;
      525        LOG.error(msg, e);
      526        throw new RuntimeException(msg, e);
      527      } finally {
      528        SerializationUtilities.releaseKryo(kryo);
      529        IOUtils.closeStream(in);
      530      }
      531    }
