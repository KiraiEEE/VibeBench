       77    /**
       78     * This map tracks the storage engines for which some other component needs to latch onto their lifecycle.
       79     *
       80     * At the time of writing this JavaDoc, the only component which needs such lifecycle latch is the
       81     * {@link com.linkedin.davinci.kafka.consumer.StoreIngestionTask}. The latching is achieved via
       82     * {@link ReferenceCounted}, which the dependent component(s) will release when it no longer needs it.
       83     *
       84     * Note that it is possible for storage engines to exist within the {@link #storageEngineRepository}
       85     * without existing within this map, if no component has latched on to its lifecycle.
       86     *
       87     * Here is an example of the steps involved in the components discussed above:
       88     *
       89     * 1. A store-version gets assigned to a server for the first time.
       90     * 2. The storage engine gets created for the first time via {@link #openStore(VeniceStoreVersionConfig, Supplier)}.
       91     * 3. The {@link com.linkedin.davinci.kafka.consumer.StoreIngestionTask} then gets created for the first time, and it
       92     *    gets its own handle to the storage engine via {@link #getRefCountedStorageEngine(String)}. This is the step
       93     *    which establishes the latch from the SIT to the SE, via ref-counting.
       94     *
       95     * The above sequence is important, because in some cases, there can be a closing and re-opening of a store-version on
       96     * a server, and it can be the case that the SE gets closed, and a new one gets created, while the SIT does not have
       97     * time to get closed yet, and it instead gets reused. In that case, the value from this map can be retrieved such
       98     * that the storage engine reference can be updated.
       99     */
      100    private final Map<String, ReferenceCounted<DelegatingStorageEngine>> storageEngines = new VeniceConcurrentHashMap<>();

      247    static void deleteStorageEngineOnRocksDBError(
      248        String storageEngineName,
      249        ReadOnlyStoreRepository storeRepository,
      250        StorageEngineFactory factory) {
      251      String storeName = Version.parseStoreFromKafkaTopicName(storageEngineName);
      252      Store store;
      253      boolean doDeleteStorageEngine;
      254      try {
      255        int versionNumber = Version.parseVersionFromKafkaTopicName(storageEngineName);
      256        store = storeRepository.getStoreOrThrow(storeName);
      257        doDeleteStorageEngine = store.getVersion(versionNumber) == null || versionNumber < store.getCurrentVersion();
      258      } catch (VeniceNoStoreException e) {
      259        // The store does not exist in Venice anymore, so it will be deleted.
      260        doDeleteStorageEngine = true;
      261      }
      262  
      263      if (doDeleteStorageEngine) {
      264        LOGGER.info("Storage for {} does not exist, will delete it.", storageEngineName);
      265        factory.removeStorageEngine(storageEngineName);
      266      }
      267    }

      269    private void restoreAllStores(
      270        VeniceConfigLoader configLoader,
      271        boolean restoreDataPartitions,
      272        boolean restoreMetadataPartitions,
      273        Function<String, Boolean> checkWhetherStorageEngineShouldBeKeptOrNot) {
      274      LOGGER.info("Start restoring all the stores persisted previously");
      275      for (Map.Entry<PersistenceType, StorageEngineFactory> entry: persistenceTypeToStorageEngineFactoryMap.entrySet()) {
      276        PersistenceType pType = entry.getKey();
      277        StorageEngineFactory factory = entry.getValue();
      278        LOGGER.info("Start restoring all the stores with type: {}", pType);
      279        Set<String> storeNames = factory.getPersistedStoreNames();
      280        for (String storeName: storeNames) {
      281          LOGGER.info("Start restoring store: {} with type: {}", storeName, pType);
      282          /**
      283           * Setup store-level persistence type based on current database setup.
      284           */
      285          VeniceStoreVersionConfig storeConfig = configLoader.getStoreConfig(storeName, pType);
      286          // Load the metadata & data restore settings from config loader.
      287          storeConfig.setRestoreDataPartitions(restoreDataPartitions);
      288          storeConfig.setRestoreMetadataPartition(restoreMetadataPartitions);
      289          StorageEngine storageEngine;
      290  
      291          if (checkWhetherStorageEngineShouldBeKeptOrNot.apply(storeName)) {
      292            try {
      293              storageEngine = openStore(storeConfig, () -> null);
      294            } catch (Exception e) {
      295              if (ExceptionUtils.recursiveClassEquals(e, RocksDBException.class)) {
      296                LOGGER.warn("Encountered RocksDB error while opening store: {}", storeName, e);
      297                // if store version does not exist, clean up the resources.
      298                deleteStorageEngineOnRocksDBError(storeName, storeRepository, factory);
      299                continue;
      300              }
      301              LOGGER.error("Could not load the following store : " + storeName, e);
      302              aggVersionedStorageEngineStats.recordRocksDBOpenFailure(storeName);
      303              throw new VeniceException("Error caught during opening store " + storeName, e);
      304            }
      305  
      306            Set<Integer> partitionIds = storageEngine.getPartitionIds();
      307            LOGGER.info(
      308                "Loaded the following partitions: {}, for store: {}",
      309                Arrays.toString(partitionIds.toArray()),
      310                storeName);
      311            LOGGER.info("Done restoring store: {} with type: {}", storeName, pType);
      312          } else {
      313            LOGGER.info("Starting deleting local storage engine: {} with type: {}", storeName, pType);
      314            factory.removeStorageEngine(storeName);
      315            LOGGER.info("Done deleting local storage engine: {} with type: {}", storeName, pType);
      316          }
      317        }
      318        LOGGER.info("Done restoring all the stores with type: {}", pType);
      319      }
      320      LOGGER.info("Done restoring all the stores persisted previously");
      321    }

      369    /**
      370     * Creates a StorageEngineFactory for the persistence type if not already present.
      371     * Creates a new storage engine for the given store in the factory and registers the storage engine with the store repository.
      372     *
      373     * @param storeConfig   The store specific properties
      374     * @param initialStoreVersionStateSupplier invoked to initialize the SVS when a brand-new storage engine is created
      375     * @return StorageEngine that was created for the given store definition.
      376     */
      377    public synchronized StorageEngine openStore(
      378        VeniceStoreVersionConfig storeConfig,
      379        Supplier<StoreVersionState> initialStoreVersionStateSupplier) {
      380      String topicName = storeConfig.getStoreVersionName();
      381      StorageEngine engine = storageEngineRepository.getLocalStorageEngine(topicName);
      382      if (engine != null) {
      383        return engine;
      384      }
      385  
      386      long startTimeInBuildingNewEngine = System.nanoTime();
      387      /**
      388       * For new store, it will use the storage engine configured in host level if it is not known.
      389       */
      390      if (!storeConfig.isStorePersistenceTypeKnown()) {
      391        storeConfig.setStorePersistenceType(storeConfig.getPersistenceType());
      392      }
      393  
      394      LOGGER.info("Creating/Opening Storage Engine {} with type: {}", topicName, storeConfig.getStorePersistenceType());
      395      StorageEngineFactory factory = getInternalStorageEngineFactory(storeConfig);
      396      PersistenceType persistenceType = factory.getPersistenceType();
      397      boolean replicationMetadataEnabled = isReplicationMetadataEnabled(topicName, persistenceType);
      398      boolean mergedCfEnabled = isMergedValueRmdColumnFamilyEnabled(topicName, persistenceType);
      399      StorageEngine newEngine = factory.getStorageEngine(storeConfig, replicationMetadataEnabled, mergedCfEnabled);
      400      newEngine.updateStoreVersionStateCache(initialStoreVersionStateSupplier.get());
      401  
      402      // Let's check if a previous incarnation of the storage engine existed earlier
      403      ReferenceCounted<DelegatingStorageEngine> refCountedStorageEngine =
      404          this.storageEngines.compute(topicName, (k, v) -> {
      405            if (v != null) {
      406              v.get().setDelegate(newEngine);
      407              LOGGER.info(
      408                  "Injected newly created storage engine into existing ref-counted delegating storage engine for {}.",
      409                  k);
      410            }
      411            return v;
      412          });
      413      DelegatingStorageEngine delegatingStorageEngine =
      414          refCountedStorageEngine == null ? new DelegatingStorageEngine(newEngine) : refCountedStorageEngine.get();
      415  
      416      storageEngineRepository.addLocalStorageEngine(delegatingStorageEngine);
      417      // Setup storage engine stats
      418      aggVersionedStorageEngineStats.setStorageEngine(topicName, delegatingStorageEngine);
      419  
      420      LOGGER.info(
      421          "time spent on creating new storage Engine for store {}: {} ms",
      422          topicName,
      423          LatencyUtils.getElapsedTimeFromNSToMS(startTimeInBuildingNewEngine));
      424      return delegatingStorageEngine;
      425    }
