      364      private List<Base> executeSlice(FHIRPathEngine engine, Object appContext, List<Base> focus, List<List<Base>> parameters) {
      365        ValidationContext c = (ValidationContext) appContext;
      366        
      367        List<Base> res = new ArrayList<>();
      368        if (parameters.size() != 2 && !(appContext instanceof ValidationContext)) {
      369          return res;
      370        }
      371        
      372        StructureDefinition sd = null;
      373        // if present, first parameter must be a singleton that points to the current profile
      374        if (parameters.get(0).size() > 1) {
      375          return res;
      376        } else if (parameters.get(0).size() == 1) { // if it's there, we have to check it 
      377          Base b = parameters.get(0).get(0);
      378          if (b.isPrimitive()) {
      379            sd = profileUtilities.findProfile(b.primitiveValue(), null);
      380          } else if (b instanceof StructureDefinition) {
      381            sd = (StructureDefinition) b;
      382          }
      383        } else {
      384          sd = c.getProfile();
      385        }
      386  
      387        // second parameter must be present 
      388        if (parameters.get(1).size() != 1) {
      389          return res;
      390        }  
      391        String name = parameters.get(1).get(0).primitiveValue();
      392        if (!Utilities.noString(name)) {
      393          for (Base b : focus) {
      394            if (b instanceof Element) {
      395              Element e = (Element) b;
      396              if (e.hasSlice(sd, name)) {
      397                res.add(e);
      398              }
      399            }
      400          }
      401        }
      402        return res;
