      202      public static void copyMemory(Object src, long srcOffset, Object dst, long dstOffset, long length) {
      203          // Check if dstOffset is before or after srcOffset to determine if we should copy
      204          // forward or backwards. This is necessary in case src and dst overlap.
      205          if (dstOffset < srcOffset) {
      206              while (length > 0) {
      207                  long size = Math.min(length, UNSAFE_COPY_THRESHOLD);
      208                  UNSAFE.copyMemory(src, srcOffset, dst, dstOffset, size);
      209                  length -= size;
      210                  srcOffset += size;
      211                  dstOffset += size;
      212              }
      213          } else {
