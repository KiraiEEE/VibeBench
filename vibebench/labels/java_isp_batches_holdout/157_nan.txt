      189          if (customProperties.isEmpty()) {

      193          if (!rebuild && !convertedCustomProperties.isEmpty()) {

      201          SmallFileMgr smallFileMgr = Env.getCurrentEnv().getSmallFileMgr();
      202          for (Map.Entry<String, String> entry : customProperties.entrySet()) {
      203              if (entry.getValue().startsWith("FILE:")) {
      204                  // convert FILE:file_name -> FILE:file_id:md5
      205                  String file = entry.getValue().substring(entry.getValue().indexOf(":") + 1);
      206                  SmallFile smallFile = smallFileMgr.getSmallFile(dbId, KAFKA_FILE_CATALOG, file, true);
      207                  convertedCustomProperties.put(entry.getKey(), "FILE:" + smallFile.id + ":" + smallFile.md5);
