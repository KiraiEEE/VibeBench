      203     * N.B.: This function is called on the hot path, so it's important to make it as efficient as possible. The order of
      204     *       operations is carefully considered so that short-circuiting comes into play as much as possible. We also try
      205     *       to minimize the overhead of logging wherever possible (e.g., by minimizing expensive calls, such as the one
      206     *       to {@link IdentityParser#parseIdentityFromCert(X509Certificate)}).
      207     */
      208    protected AccessResult checkAccess(String uri, X509Certificate clientCert, String storeName, String method) {
      209      try {
      210        if (accessController.hasAccess(clientCert, storeName, method)) {
      211          return AccessResult.GRANTED;
      212        }
      213  
      214        if (VeniceSystemStoreUtils.isSystemStore(storeName)) {
      215          return AccessResult.GRANTED;
      216        }
      217  
      218        // Fact:
      219        // Request gets rejected.
      220        // Possible Reasons:
      221        // A. ACL not found. OR,
      222        // B. ACL exists but caller does not have permission.
      223  
      224        if (!accessController.isFailOpen() && !accessController.hasAcl(storeName)) { // short circuit, order matters
      225          // Case A
      226          // Conditions:
      227          // 0. (outside) Store exists and is being access controlled. AND,
      228          // 1. (left) The following policy is applied: if ACL not found, reject the request. AND,
      229          // 2. (right) ACL not found.
      230          // Result:
      231          // Request is rejected by DynamicAccessController#hasAccess()
      232          // Root cause:
      233          // Requested resource exists but does not have ACL.
      234          // Action:
      235          // return 401 Unauthorized
      236          if (LOGGER.isWarnEnabled()) {
      237            LOGGER.warn(
      238                "Requested store does not have ACL: {} requested {} {}",
      239                identityParser.parseIdentityFromCert(clientCert),
      240                method,
      241                uri);
      242            if (LOGGER.isDebugEnabled()) {
      243              LOGGER.debug(
      244                  "Existing stores: {}",
      245                  metadataRepository.getAllStores().stream().map(Store::getName).sorted().collect(Collectors.toList()));
      246              LOGGER.debug(
      247                  "Access-controlled stores: {}",
      248                  accessController.getAccessControlledResources().stream().sorted().collect(Collectors.toList()));
      249            }
      250          }
      251          return AccessResult.UNAUTHORIZED;
      252        } else {
      253          // Case B
      254          // Conditions:
      255          // 1. Fail closed, and ACL found. OR,
      256          // 2. Fail open, and ACL found. OR,
      257          // 3. Fail open, and ACL not found.
      258          // Analyses:
      259          // (1) ACL exists, therefore result is determined by ACL.
      260          // Since the request has been rejected, it must be due to lack of permission.
      261          // (2) ACL exists, therefore result is determined by ACL.
      262          // Since the request has been rejected, it must be due to lack of permission.
      263          // (3) In such case, request would NOT be rejected in the first place,
      264          // according to the definition of hasAccess() in DynamicAccessController interface.
      265          // Contradiction to the fact, therefore this case is impossible.
      266          // Root cause:
      267          // Caller does not have permission to access the resource.
      268          // Action:
      269          // return 403 Forbidden
      270          if (LOGGER.isDebugEnabled()) {
      271            LOGGER.debug(
      272                "Unauthorized access rejected: {} requested {} {}",
      273                identityParser.parseIdentityFromCert(clientCert),
      274                method,
      275                uri);
      276          }
      277          return AccessResult.FORBIDDEN;
      278        }
      279      } catch (AclException e) {
      280        if (accessController.isFailOpen()) {
      281          if (LOGGER.isWarnEnabled()) {
      282            LOGGER.warn(
      283                "Exception occurred! Access granted: {} requested {} {}",
      284                identityParser.parseIdentityFromCert(clientCert),
      285                method,
      286                uri,
      287                e);
      288          }
      289          return AccessResult.GRANTED;
      290        } else {
      291          if (LOGGER.isWarnEnabled()) {
      292            LOGGER.warn(
      293                "Exception occurred! Access rejected: {} requested {} {}",
      294                identityParser.parseIdentityFromCert(clientCert),
      295                method,
      296                uri,
      297                e);
      298          }
      299          return AccessResult.ERROR_FORBIDDEN;
      300        }
      301      }
      302    }

      106        ReqT message,
      107        ServerCall<ReqT, RespT> call,

      110      // (e.g. the ingestion monitor) use different message types that carry neither, so forward them rather than
      111      // mis-cast. Such diagnostic endpoints are gated by mutual TLS rather than store-level read ACL.
      112      if (!(message instanceof VeniceClientRequest)) {
      113        onAuthenticated.accept(message);
      114        return;
      115      }

      117      // For now, GRPC only supports STORAGE query
