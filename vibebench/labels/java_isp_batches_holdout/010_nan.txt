       50          List<SqlColumnMapping> columns = tableMapping.columns();

       84      protected final boolean matchingColumnType(SqlColumnMapping columnMapping, SqlColumnMetadata columnMetadata,
       85                                                 Dialect dialect) {
       86          if (matchingColumnTypes(columnMapping.getDbType(), columnMetadata.type())) {
       87              return true;
       88          }

       92          }

      148      private void validateColumn(SqlColumnMapping columnMapping, SqlColumnMetadata columnMetadata,
      149                                         Dialect dialect, String tableName) {
      150          if (StringUtils.isNotEmpty(columnMapping.getDefinition())) {
      151              // Don't compare columns with custom SQL definition
      152              // and let user be responsible for mapping of that field
      153              return;
      154          }
      155          if (matchingColumnType(columnMapping, columnMetadata, dialect)) {
      156              return;
      157          }
      158          throw new SchemaValidationException(String.format("Schema validation failed. Column [%s] in table [%s] of type [%s] is mapped to [%s].",
      159              columnMetadata.name(), tableName, columnMetadata.typeName(), columnMapping.getDbType()));
      160      }

      162      private static boolean matchingColumnTypes(SqlDbType dbType, int typeCode) {
      163          int mappedTypeCode = dbType.getType();
      164          return mappedTypeCode == typeCode
      165              || isCompatibleIntegralType(mappedTypeCode, typeCode)
      166              || isNumericOrDecimal(mappedTypeCode) && isNumericOrDecimal(typeCode)
      167              || isFloatOrRealOrDouble(mappedTypeCode) && isFloatOrRealOrDouble(typeCode)
      168              || isVarcharType(mappedTypeCode) && isVarcharType(typeCode)
      169              || isVarbinaryType(mappedTypeCode) && isVarbinaryType(typeCode)
      170              || isEnumType(mappedTypeCode) && isVarcharType(typeCode);
      171          // Add more checks/fallbacks during testing and/or reported issues
      172      }

      174      private static boolean isCompatibleIntegralType(int typeCode1, int typeCode2) {
      175          return switch (typeCode1) {
      176              case Types.TINYINT ->
      177                  typeCode2 == Types.TINYINT
      178                      || typeCode2 == Types.SMALLINT
      179                      || typeCode2 == Types.INTEGER
      180                      || typeCode2 == Types.BIGINT;
      181              case Types.SMALLINT ->
      182                  typeCode2 == Types.SMALLINT
      183                      || typeCode2 == Types.INTEGER
      184                      || typeCode2 == Types.BIGINT;
      185              case Types.INTEGER ->
      186                  typeCode2 == Types.INTEGER
      187                      || typeCode2 == Types.BIGINT;
      188              default -> false;
      189          };
      190      }

       57                  }

       59              }

       61          } else if (isOracleBinaryDoubleOrFloat(columnMetadata.typeName())) {
       62              return isFloatOrRealOrDouble(columnMapping.getDbType().getType());
