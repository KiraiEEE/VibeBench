       53      public RTMPClient() {
       54          ioHandler = new RTMPMinaIoHandler();
       55          ioHandler.setHandler(this);
       56      }
       57  
       58      /** {@inheritDoc} */
       59      @Override
       60      public Map<String, Object> makeDefaultConnectionParams(String server, int port, String application) {
       61          Map<String, Object> params = super.makeDefaultConnectionParams(server, port, application);
       62          if (!params.containsKey("tcUrl")) {
       63              params.put("tcUrl", String.format("%s://%s:%s/%s", protocol, server, port, application));
       64          }
       65          return params;
       66      }
       67  
       68      /** {@inheritDoc} */
       69      @Override
       70      protected void startConnector(String server, int port) {
       71          log.debug("startConnector - server: {} port: {}", server, port);
       72          socketConnector = new NioSocketConnector();
       73          socketConnector.setHandler(ioHandler);
       74          future = socketConnector.connect(new InetSocketAddress(server, port));
       75          future.addListener(new IoFutureListener<ConnectFuture>() {
       76              @Override
       77              public void operationComplete(ConnectFuture future) {
       78                  try {
       79                      // will throw RuntimeException after connection error
       80                      session = future.getSession();
       81                  } catch (Throwable e) {
       82                      log.warn("Exception in startConnector", e);
       83                      // disconnect this
       84                      disconnect();
       85                      // if there isn't an ClientExceptionHandler set, a RuntimeException may be thrown in handleException
       86                      handleException(e);
       87                  }
       88              }
       89          });
       90          // Now wait for the connect to be completed
       91          future.awaitUninterruptibly(CONNECTOR_WORKER_TIMEOUT);
       92          log.debug("startConnector {} done", Thread.currentThread().getName());
       93      }
       94  
       95      /** {@inheritDoc} */
       96      @Override
       97      public void disconnect() {
       98          if (future != null) {
       99              try {
      100                  // session will be null if connect failed
      101                  if (session != null) {
      102                      // close, now
      103                      CloseFuture closeFuture = session.closeNow();
      104                      // now wait for the close to be completed
      105                      if (closeFuture.await(1000, TimeUnit.MILLISECONDS)) {
      106                          if (!future.isCanceled()) {
      107                              if (future.cancel()) {
      108                                  log.debug("Connect future cancelled after close future");
      109                              }
      110                          }
      111                      }
      112                  } else if (future.cancel()) {
      113                      log.debug("Connect future cancelled");
      114                  }
      115              } catch (Exception e) {
      116                  log.warn("Exception during disconnect", e);
      117              } finally {
      118                  // we can now dispose the connector
      119                  socketConnector.dispose(false);
      120              }
      121          }
      122          super.disconnect();
      123      }

      125      /**
      126       * {@inheritDoc}
      127       *
      128       * Sets the RTMP protocol, the default is "rtmp". If "rtmps" or "rtmpt" are required, the appropriate client type should be selected.
      129       */
      130      @Override
      131      public void setProtocol(String protocol) throws Exception {
      132          this.protocol = protocol;
      133          if ("rtmps".equals(protocol) || "rtmpt".equals(protocol) || "rtmpte".equals(protocol) || "rtmfp".equals(protocol)) {
      134              throw new Exception("Unsupported protocol specified, please use the correct client for the intended protocol");
      135          }
      136      }
