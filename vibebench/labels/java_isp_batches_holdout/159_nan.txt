       37       * Checks if the given If-None-Match header value matches this ETag.
       38       * Properly handles:
       39       * <ul>
       40       *   <li>Weak ETags (W/"...")</li>
       41       *   <li>Multiple comma-separated ETags</li>
       42       *   <li>Wildcard "&#42;"</li>
       43       *   <li>Quote stripping</li>
       44       * </ul>
       45       * <p>
       46       * According to RFC 7232 section 2.3.2, for If-None-Match comparison,
       47       * weak comparison is used: ETags match if their opaque-tags match,
       48       * regardless of whether either or both are weak.
       49       *
       50       * @param ifNoneMatch the If-None-Match header value from the request
       51       * @return true if the ETag matches, false otherwise
       52       */
       53      public boolean matches(String ifNoneMatch) {
       54          if (isEmpty(ifNoneMatch)) {
       55              return false;
       56          }
       57          // Wildcard matches everything:
       58          if ("*".equals(ifNoneMatch.trim())) {
       59              return true;
       60          }
       61          // Weak comparison per RFC 7232 ignores the weak/strong distinction.
       62          return parseIfNoneMatch(ifNoneMatch).stream()
       63                  .anyMatch(requestETag -> requestETag.getValue().equals(etag.getValue()));
       64      }
       65  
       66      /**
       67       * Parses an If-None-Match header value into a list of ETags.
       68       * Handles multiple comma-separated ETags and weak ETags using JAX-RS's EntityTag delegate.
       69       *
       70       * @param ifNoneMatch the If-None-Match header value
       71       * @return list of parsed ETags
       72       */
       73      private List<EntityTag> parseIfNoneMatch(String ifNoneMatch) {
       74          var delegate = RuntimeDelegate.getInstance().createHeaderDelegate(EntityTag.class);
       75          return stream(ifNoneMatch.split(","))

       77                  .filter(s -> !s.isEmpty())
       78                  .map(delegate::fromString)
