      104          CompoundTag tag = (CompoundTag) compoundTag;
      105          RegistryAccess registry = (RegistryAccess) registryAccess;
      106          DynamicOps<Tag> ops = registry.createSerializationContext(NbtOps.INSTANCE);
      107          if (tag == null) {
      108              return ItemStack.EMPTY;
      109          }
      110          try {
      111              return ItemStack.CODEC.parse(ops, tag).getOrThrow();
      112          } catch (Exception e) {
      113              return ItemStack.EMPTY;
      114          }
      115      }

      158      @Override
      159      public int[] getImageResolution(Object resourceLocation) {
      160          Identifier rl = toIdentifier(resourceLocation);
      161          ResourceManager resourceManager = Minecraft.getInstance().getResourceManager();
      162          Optional<Resource> resource = resourceManager.getResource(rl);
      163          if (resource.isPresent()) {
      164              try (InputStream stream = resource.get().open()) {
      165                  BufferedImage image = ImageIO.read(stream);
      166                  return new int[]{image.getWidth(), image.getHeight()};
      167              } catch (IOException ignored) {
      168              }
      169          }
      170          return null;
      171      }

      173      @Override
      174      public boolean resourceExists(Object resourceLocation) {
      175          Identifier rl = toIdentifier(resourceLocation);
      176          return Minecraft.getInstance().getResourceManager().getResource(rl).isPresent();
      177      }

      179      private Identifier toIdentifier(Object resourceLocation) {
      180          if (resourceLocation instanceof Identifier id) {
      181              return id;
      182          } else if (resourceLocation instanceof NcId ncId) {
      183              return Identifier.fromNamespaceAndPath(ncId.namespace(), ncId.path());
      184          }
      185          throw new IllegalArgumentException("Expected Identifier or NcId, got: " + resourceLocation.getClass().getName());
      186      }
