      142      /**
      143       * if user trigger, return true
      144       * else, only can have 2 task. because every task can refresh all data.
      145       *
      146       * @param taskContext
      147       * @return
      148       */
      149      @Override
      150      public boolean isReadyForScheduling(MTMVTaskContext taskContext) {
      151          if (isManual(taskContext)) {
      152              return true;
      153          }
      154          List<MTMVTask> runningTasks = getRunningTasks();
      155          int runningNum = 0;
      156          for (MTMVTask task : runningTasks) {
      157              if (!isManual(task.getTaskContext())) {
      158                  runningNum++;
      159                  // Prerequisite: Each refresh will calculate which partitions to refresh
      160                  //
      161                  // For example, there is currently a running task that is refreshing partition p1.
      162                  // If the data of p2 changes at this time and triggers a refresh task t2,
      163                  // according to the logic (>=1), t2 will be lost
      164                  //
      165                  // If the logic is >=2, t2 will wait lock of MTMVJob.
      166                  // If the p3 data changes again and triggers the refresh task t3,
      167                  // then t3 will be discarded. However, when t2 runs, both p2 and p3 data will be refreshed.
      168                  if (runningNum >= 2) {
      169                      LOG.warn("isReadyForScheduling return false, because current taskContext is null, exist task: {}",
      170                              task);
      171                      if (MetricRepo.isInit) {
      172                          MetricRepo.COUNTER_ASYNC_MATERIALIZED_VIEW_TASK_SKIP_NUM.increase(1L);
      173                      }
      174                      return false;
      175                  }
      176              }
      177          }
      178          return true;
      179      }

      181      private boolean isManual(MTMVTaskContext taskContext) {
      182          return taskContext != null && taskContext.getTriggerMode() == MTMVTaskTriggerMode.MANUAL;
      183      }
