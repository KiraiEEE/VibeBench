      179    private long nextVersionResolution(
      180        @Nullable SystemMetadata currentSystemMetadata, @Nullable SystemMetadata newSystemMetadata) {
      181      // existing row 0 should have at least version 1
      182      long currentVersion =
      183          Optional.ofNullable(currentSystemMetadata)
      184              .map(SystemMetadata::getVersion)
      185              .map(Long::valueOf)
      186              .orElse(1L);
      187      // new row should have at least version 2 with the expected current version one behind
      188      long expectedCurrentVersion =
      189          Optional.ofNullable(newSystemMetadata)
      190                  .map(SystemMetadata::getVersion)
      191                  .map(Long::valueOf)
      192                  .orElse(2L)
      193              - 1;
      194  
      195      // reconcile the target version based on the max of the two values in case of previous
      196      // corruption
      197      long targetVersion = Math.max(1, Math.max(currentVersion, expectedCurrentVersion));
      198  
      199      // if mismatch, fix row 0 metadata
      200      if (currentVersion != targetVersion) {
      201        currentSystemMetadata.setVersion(String.valueOf(targetVersion), SetMode.DISALLOW_NULL);
      202      }
      203  
      204      return targetVersion;
      205    }

      188          selectFrom(CassandraAspect.TABLE_NAME)
      189              .selectors(

      191                  Selector.column(CassandraAspect.ASPECT_COLUMN),
      192                  Selector.function("min", Selector.column(CassandraAspect.VERSION_COLUMN))
      193                      .as("min_version"),
      194                  Selector.function("max", Selector.column(CassandraAspect.VERSION_COLUMN))
      195                      .as("max_version"))
      196              .whereColumn(CassandraAspect.URN_COLUMN)
      197              .isEqualTo(literal(urn))
      198              .whereColumn(CassandraAspect.ASPECT_COLUMN)
      199              .in(aspectNamesToLiterals(aspectNames))
      200              .groupBy(
      201                  ImmutableList.of(
      202                      Selector.column(CassandraAspect.URN_COLUMN),
      203                      Selector.column(CassandraAspect.ASPECT_COLUMN)))
      204              .build();
      205  
      206      ResultSet rs = _cqlSession.execute(ss);

      208          rs.all().stream()
      209              .collect(
      210                  Collectors.toMap(

      212                      row -> Pair.of(row.getLong("min_version"), row.getLong("max_version"))));
      213

      215      for (String aspect : aspectNames) {
      216        if (!aspectVersionRanges.containsKey(aspect)) {
      217          aspectVersionRanges.put(aspect, Pair.of(-1L, -1L));
