       66          terms.add(term);

       78       * Add a search term with qualifier.
       79       *
       80       * If {@code value} is empty or {@code null}, all terms with the current qualifier will be removed.
       81       *
       82       * @param qualifier
       83       *            the qualifier for this term
       84       * @param value
       85       *            the value for this term. If empty or null, all terms with the current qualifier will be removed.
       86       * @return the gh query builder
       87       */
       88      GHQueryBuilder<T> q(@Nonnull final String qualifier, @CheckForNull final String value) {
       89          if (StringUtils.isEmpty(qualifier)) {
       90              throw new IllegalArgumentException("qualifier cannot be null or empty");
       91          }
       92          if (StringUtils.isEmpty(value)) {
       93              final String removeQualifier = qualifier + ":";
       94              terms.removeIf(term -> term.startsWith(removeQualifier));
       95          } else {
       96              terms.add(qualifier + ":" + value);
