      155      static String translate(Locale locale, String key, Object... params) {
      156          VaadinService vaadinService = VaadinService.getCurrent();
      157          if (vaadinService == null) {
      158              throw new IllegalStateException(
      159                      "I18NProvider is not available as VaadinService is null");
      160          }
      161          Instantiator instantiator = vaadinService.getInstantiator();
      162          if (instantiator == null) {
      163              throw new IllegalStateException(
      164                      "I18NProvider is not available as Instantiator is null");
      165          }
      166  
      167          return LocaleUtil.getI18NProvider()
      168                  .map(i18n -> i18n.getTranslation(key, locale, params))
      169                  .orElseGet(() -> "!{" + key + "}!");
      170      }

       77      @Override
       78      public String getTranslation(String key, Locale locale, Object... params) {
       79          if (key == null) {
       80              getLogger().warn("Got lang request for key with null value!");
       81              return "";
       82          }
       83  
       84          final ResourceBundle bundle = getBundle(locale);
       85          if (bundle == null) {
       86              return key;
       87          }
       88  
       89          String value;
       90          try {
       91              value = bundle.getString(key);
       92          } catch (final MissingResourceException e) {
       93              getLogger().debug("Missing resource for key " + key, e);
       94              return "!" + locale.getLanguage() + ": " + key;
       95          }
       96          if (params.length > 0) {
       97              value = new MessageFormat(value, locale).format(params);
       98          }
       99          return value;
      100      }
      101  
      102      @Override
      103      public Map<String, String> getAllTranslations(Locale locale) {
      104          var bundle = getBundle(locale);
      105          return bundle == null ? Map.of()
      106                  : getTranslations(bundle.keySet(), locale);
      107      }
      108  
      109      ResourceBundle getBundle(Locale locale) {
      110          try {
      111              return ResourceBundle.getBundle(BUNDLE_PREFIX, locale, classLoader);
      112          } catch (final MissingResourceException e) {
      113              getLogger().warn("Missing resource bundle for " + BUNDLE_PREFIX
      114                      + " and locale " + locale.getDisplayName(), e);
      115          }
      116          return null;
      117      }
