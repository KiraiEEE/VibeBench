       58      // Based on the ZeroInWord algorithm: http://graphics.stanford.edu/~seander/bithacks.html#ZeroInWord
       59      assert buffer.length >= numBytesPerValue;

       61      boolean littleEndian = _dataBuffer.order() == ByteOrder.LITTLE_ENDIAN;
       62      ByteBuffer wrapper = ByteBuffer.wrap(buffer);
       63      if (littleEndian) {
       64        wrapper.order(ByteOrder.LITTLE_ENDIAN);
       65      }
       66      int endIndex = numBytesPerValue & 0xFFFFFFF8;
       67      int i = 0;
       68      for (; i < endIndex; i += Long.BYTES) {
       69        long word = _dataBuffer.getLong(startOffset + i);
       70        wrapper.putLong(i, word);
