       56       * Reads a class into the classWriter, filtering out any existing __read/__write synthetic
       57       * methods so that accessor generation is idempotent across repeated plugin runs.
       58       *
       59       * @param entity the class to read
       60       */
       61      protected void readClassFiltering(Class<?> entity) {
       62          String resourceName = "%s.class".formatted(entity.getName().replace('.', '/'));
       63          java.io.InputStream inputStream = entity.getClassLoader().getResourceAsStream(resourceName);
       64          if (inputStream == null) {
       65              throw new IllegalArgumentException("Could not find class file for %s".formatted(entity.getName()));
       66          }
       67          ClassVisitor filteringVisitor = new ClassVisitor(ASM9, classWriter) {
       68              @Override
       69              public MethodVisitor visitMethod(int access, String name, String descriptor,
       70                      String signature, String[] exceptions) {
       71                  if ((access & ACC_SYNTHETIC) != 0
       72                          && (name.startsWith("__read") || name.startsWith("__write"))) {
       73                      return null;
       74                  }
       75                  return super.visitMethod(access, name, descriptor, signature, exceptions);
       76              }
       77          };
       78          try {
       79              new ClassReader(inputStream).accept(filteringVisitor, 0);
       80          } catch (java.io.IOException e) {
       81              throw new RuntimeException("Failed to read class %s".formatted(entity.getName()), e);
       82          }
       83      }
