      133                  .getCharacterRuntimes()
      134                  .removeAll(characterStoryDataList.stream()
      135                          .map(CharacterStoryData::getCharacterRuntime)
      136                          .toList());
      137          if (environment == Environment.PRODUCTION) {
      138              for (InkAction inkAction : playerSession.getInkActions()) {
      139                  inkAction.stop();
      140              }
      141              playerSession.getInkActions().clear();
      142              playerSession.setCurrentCamera(null);
      143              return;
      144          }
      145          if (!keyframes.isEmpty()) {
      146              keyframes.getFirst().hideKeyframe(playerSession.getPlayer());
      147          }
      148          if (!keyframeTriggers.isEmpty()) {
      149              keyframeTriggers.getFirst().hideKeyframe(playerSession.getPlayer());
      150          }
      151          playerSession.getCharacterRuntimes().clear();
      152          if (save) {
      153              MainScreenKeyframe oldKeyframe = mainScreenData.getKeyframe();
      154              KeyframeTrigger oldKeyframeTrigger = mainScreenData.getKeyframeTrigger();
      155              MainScreenKeyframe keyframe = !keyframes.isEmpty() ? keyframes.getFirst() : null;
      156              KeyframeTrigger keyframeTrigger = !keyframeTriggers.isEmpty() ? keyframeTriggers.getFirst() : null;
      157              List<CharacterStoryData> oldCharacterStoryData = mainScreenData.getCharacterStoryDataList();
      158              mainScreenData.getCharacterStoryDataList().clear();
      159              try {
      160                  mainScreenData.setKeyframe(keyframe);
      161                  mainScreenData.setKeyframeTrigger(keyframeTrigger);
      162                  mainScreenData.getCharacterStoryDataList().addAll(characterStoryDataList);
      163                  NarrativeCraftFile.updateMainScreenBackground(mainScreenData, playerSession.getScene());
      164                  for (CharacterStoryData characterStoryData : mainScreenData.getCharacterStoryDataList()) {
      165                      characterStoryData.setSkinName(
      166                              characterStoryData.getCharacterRuntime().getSkinName());
      167                  }
      168                  playerSession.getPlayer().sendSystemMessage(Translation.message("controller.saved"));
      169              } catch (IOException e) {
      170                  mainScreenData.setKeyframe(oldKeyframe);
      171                  mainScreenData.setKeyframeTrigger(oldKeyframeTrigger);
      172                  mainScreenData.getCharacterStoryDataList().addAll(oldCharacterStoryData);
      173                  for (CharacterStoryData characterStoryData : mainScreenData.getCharacterStoryDataList()) {
      174                      characterStoryData.setSkinName(
      175                              characterStoryData.getCharacterRuntime().getOldSkinName());
      176                  }
      177                  Util.sendCrashMessage(playerSession.getPlayer(), e);
      178              }
      179          } else {
      180              for (CharacterStoryData characterStoryData : mainScreenData.getCharacterStoryDataList()) {
      181                  characterStoryData.setSkinName(
      182                          characterStoryData.getCharacterRuntime().getOldSkinName());
      183              }
      184          }
      185      }
