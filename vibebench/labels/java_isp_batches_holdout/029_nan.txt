       71        // skip files/folders that start with . such as .svn

       79    private void deleteFileOrDirectory(File file) {
       80      if (file.isDirectory()) {
       81        // clear it out (recursively) if needed...
       82        if (!Files.isSymbolicLink(file.toPath())) {
       83          Arrays.stream(file.listFiles()).forEach(this::deleteFileOrDirectory);
       84        }
       85      }
       86      if (!file.delete()) {
       87        throw new RuntimeException("Exception cleaning output directory, file [" + file + "]");
       88      }
