      126      protected void lockTablesForSchemaSnapshot(ChangeEventSourceContext sourceContext,
      127                                                 RelationalSnapshotContext<P, O> snapshotContext)
      128              throws SQLException {
      129          // Set the transaction isolation level to REPEATABLE READ. This is the default, but the default can be changed
      130          // which is why we explicitly set it here.
      131          //
      132          // With REPEATABLE READ, all SELECT queries within the scope of a transaction (which we don't yet have) will read
      133          // from the same MVCC snapshot. Thus each plain (non-locking) SELECT statements within the same transaction are
      134          // consistent also with respect to each other.
      135          //
      136          // See: https://dev.mysql.com/doc/refman/8.2/en/set-transaction.html
      137          // See: https://dev.mysql.com/doc/refman/8.2/en/innodb-transaction-isolation-levels.html
      138          // See: https://dev.mysql.com/doc/refman/8.2/en/innodb-consistent-read.html
      139          connection.connection().setTransactionIsolation(Connection.TRANSACTION_REPEATABLE_READ);
      140          connection.executeWithoutCommitting("SET SESSION lock_wait_timeout=" + connectorConfig.snapshotLockTimeout().getSeconds());
      141          try {
      142              connection.executeWithoutCommitting("SET SESSION innodb_lock_wait_timeout=" + connectorConfig.snapshotLockTimeout().getSeconds());
      143          }
      144          catch (SQLException e) {
      145              LOGGER.warn("Unable to set innodb_lock_wait_timeout", e);
      146          }
      147  
      148          // ------------------------------------
      149          // LOCK TABLES
      150          // ------------------------------------
      151          // Obtain read lock on all tables. This statement closes all open tables and locks all tables
      152          // for all databases with a global read lock, and it prevents ALL updates while we have this lock.
      153          // It also ensures that everything we do while we have this lock will be consistent.
      154          if (connectorConfig.getSnapshotLockingStrategy().isLockingEnabled() && connectorConfig.isGlobalLockUseRequested()) {
      155              try {
      156                  globalLock();
      157                  metrics.setGlobalLockAcquired();
      158              }
      159              catch (SQLException e) {
      160                  LOGGER.info("Unable to flush and acquire global read lock, will use table read locks after reading table names");
      161                  // Continue anyway, since RDS (among others) don't allow setting a global lock
      162                  assert !isGloballyLocked();
      163              }
      164              if (connectorConfig.getSnapshotLockingStrategy().isIsolationLevelResetOnFlush()) {
      165                  // FLUSH TABLES resets TX and isolation level
      166                  connection.executeWithoutCommitting("SET TRANSACTION ISOLATION LEVEL REPEATABLE READ");
      167              }
      168          }
