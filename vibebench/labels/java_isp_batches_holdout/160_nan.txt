      188      static long rateLimitStringToBytes(String sizeStr) {
      189        String baseStr = sizeStr;
      190        long multiplier = 1;
      191        if (baseStr.length() > 2) {
      192          String suffix = baseStr.substring(baseStr.length() - 2).toLowerCase();
      193          switch (suffix) {
      194            case "kb" -> {
      195              baseStr = baseStr.substring(0, baseStr.length() - 2);
      196              multiplier = 1024;
      197            }
      198            case "mb" -> {
      199              baseStr = baseStr.substring(0, baseStr.length() - 2);
      200              multiplier = 1024 * 1024;
      201            }
      202            case "gb" -> {
      203              baseStr = baseStr.substring(0, baseStr.length() - 2);
      204              multiplier = 1024 * 1024 * 1024;
      205            }
      206          }
      207        }
      208        if (baseStr.isEmpty()) {
      209          throw new IllegalArgumentException("Cannot convert rate limit string: " + sizeStr);
      210        }
      211        double baseValue = Double.parseDouble(baseStr);
      212        return (long) (baseValue * multiplier);
      213      }
