      378                      ThreadPoolResourceAllocator.getOrCreate(Platform.getThreadPoolResourceSupplier());
      379              MessageHandler messageHandler = buildMessageHandler(config);
      380              ChannelHandler channelHandler = buildChannelHandler(threadPoolResource, messageHandler);
      381              ConnectionManager connectionManager = buildConnectionManager(channelHandler);
      382              CommandManager commandManager = buildCommandManager(channelHandler);
      383              // TODO: Support exception throwing, including interrupted exceptions
      384              return connectionManager
      385                      .connectToValkey(config)
      386                      .thenApply(
      387                              ignored ->
      388                                      constructor.apply(
      389                                              new ClientBuilder(
      390                                                      connectionManager,
      391                                                      commandManager,
      392                                                      messageHandler,
      393                                                      Optional.ofNullable(config.getSubscriptionConfiguration()))));
      394          } catch (InterruptedException e) {
      395              // Something bad happened while we were establishing netty connection to UDS
      396              var future = new CompletableFuture<T>();
      397              future.completeExceptionally(e);
      398              return future;
      399          }
      400      }

      412       * Return a next pubsub message if it is present.
      413       *
      414       * @throws ConfigurationError If client is not subscribed to any channel or if client configured
      415       *     with a callback.
      416       * @return A message if any or <code>null</code> if there are no unread messages.
      417       */
      418      public PubSubMessage tryGetPubSubMessage() {
      419          if (subscriptionConfiguration.isEmpty()) {
      420              throw new ConfigurationError(
      421                      "The operation will never complete since there was no pubsub subscriptions applied to the"
      422                              + " client.");
      423          }
      424          if (subscriptionConfiguration.get().getCallback().isPresent()) {
      425              throw new ConfigurationError(
      426                      "The operation will never complete since messages will be passed to the configured"
      427                              + " callback.");
      428          }
      429          return messageHandler.getQueue().popSync();
      430      }
