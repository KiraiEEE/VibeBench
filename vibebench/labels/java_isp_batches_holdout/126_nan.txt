      149    private void pollForJobs() {
      150      if (!running.get()) {
      151        return;
      152      }
      153  
      154      long now = System.currentTimeMillis();
      155      long interval = participating.get() ? ACTIVE_POLL_INTERVAL_MS : IDLE_POLL_INTERVAL_MS;
      156  
      157      // Skip poll if not enough time has elapsed
      158      if (now - lastPollTime < interval) {
      159        return;
      160      }
      161      lastPollTime = now;
      162  
      163      try {
      164        // Fast, lightweight query for running jobs
      165        List<String> runningJobIds = collectionDAO.searchIndexJobDAO().getRunningJobIds();
      166  
      167        if (runningJobIds.isEmpty()) {
      168          // No jobs running - clear known jobs and stay in idle mode
      169          if (!knownJobs.isEmpty()) {
      170            LOG.debug("No running jobs found, clearing {} known jobs", knownJobs.size());
      171            knownJobs.clear();
      172          }
      173          return;
      174        }
      175  
      176        // Check for new jobs we haven't seen
      177        for (String jobIdStr : runningJobIds) {
      178          UUID jobId = UUID.fromString(jobIdStr);
      179          if (!knownJobs.contains(jobId)) {
      180            // New job discovered!
      181            LOG.info("Discovered new running job via polling: {}", jobId);
      182            knownJobs.add(jobId);
      183  
      184            if (jobStartedCallback != null) {
      185              jobStartedCallback.accept(jobId);
      186            }
      187          }
      188        }
      189  
      190        // Clean up jobs that are no longer running
      191        knownJobs.removeIf(
      192            jobId -> runningJobIds.stream().noneMatch(id -> id.equals(jobId.toString())));
      193  
      194      } catch (Exception e) {
      195        LOG.error("Error polling for jobs", e);
      196      }
      197    }
