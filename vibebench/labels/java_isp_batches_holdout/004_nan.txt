      132      protected List<Layer> collectLayers(T ref, String contentType, boolean includeAll) {
      133          List<Layer> layers = new LinkedList<>();
      134          if (isManifestMediaType(contentType)) {
      135              return getManifest(ref).getLayers();
      136          }
      137          Index index = getIndex(ref);
      138          for (ManifestDescriptor manifestDescriptor : index.getManifests()) {
      139              String manifestContentType = manifestDescriptor.getMediaType();
      140              // We just skip unknown media type descriptor
      141              if (!isManifestMediaType(manifestContentType) && !isIndexMediaType(manifestContentType)) {
      142                  LOG.info(
      143                          "Unrecognized content type {}, skipping descriptor {}",
      144                          manifestContentType,
      145                          manifestDescriptor.getDigest());
      146                  continue;
      147              }
      148              // Collect layer for each manifest
      149              List<Layer> manifestLayers =
      150                      getManifest(ref.withDigest(manifestDescriptor.getDigest())).getLayers();
      151              for (Layer manifestLayer : manifestLayers) {
      152                  if (manifestLayer.getAnnotations().isEmpty()
      153                          || !manifestLayer.getAnnotations().containsKey(Const.ANNOTATION_TITLE)) {
      154                      if (includeAll) {
      155                          LOG.debug("Including layer without title annotation: {}", manifestLayer.getDigest());
      156                          layers.add(manifestLayer);
      157                      }
      158                      LOG.debug("Skipping layer without title annotation: {}", manifestLayer.getDigest());
      159                      continue;
      160                  }
      161                  layers.add(manifestLayer);
      162              }
      163          }
      164          return layers;
      165      }

      193      private void setMeterRegistry(MeterRegistry meterRegistry) {
      194          this.meterRegistry = meterRegistry;
      195      }

      197      /**
      198       * Build the provider
      199       * @return The provider
      200       */
      201      private Registry build() {
      202          HttpClient.Builder clientBuilder = HttpClient.Builder.builder().withSkipTlsVerify(skipTlsVerify);
      203          if (meterRegistry != null) {
      204              clientBuilder = clientBuilder.withMeterRegistry(meterRegistry);
      205          }
      206          client = clientBuilder.build();
      207          if (executorService == null) {
      208              executorService = Executors.newFixedThreadPool(maxConcurrentDownloads, r -> {
      209                  Thread t = new Thread(r);
      210                  t.setName("layer-transfer-worker-%d".formatted(t.getId()));
      211                  return t;
      212              });
      213          }
      214          return this;
      215      }
