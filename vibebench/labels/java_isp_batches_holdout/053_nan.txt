       57    public FlushResult executeDbOperations(List<DbOperation> operations) {
       58      for (DbOperation operation : operations) {
       59        // stage operation
       60        executeDbOperation(operation);
       61      }
       62  
       63      List<BatchResult> batchResults;
       64      try {
       65        // applies all operations
       66        batchResults = flushBatchOperations();
       67      } catch (PersistenceException e) {
       68        return postProcessBatchFailure(operations, e);
       69      }

       74    protected FlushResult postProcessBatchSuccess(List<DbOperation> operations, List<BatchResult> batchResults) {
       75      Iterator<DbOperation> operationsIt = operations.iterator();
       76      List<DbOperation> failedOperations = new ArrayList<>();
       77      for (BatchResult successfulBatch : batchResults) {
       78        // even if all batches are successful, there can be concurrent modification failures
       79        // (e.g. 0 rows updated)
       80        postProcessJdbcBatchResult(operationsIt, successfulBatch.getUpdateCounts(), null, failedOperations);
       81      }
       82  
       83      // there should be no more operations remaining
       84      if (operationsIt.hasNext()) {
       85        throw LOG.wrongBatchResultsSizeException(operations);
       86      }
       87  
       88      return FlushResult.withFailures(failedOperations);
       89    }

       91    protected FlushResult postProcessBatchFailure(List<DbOperation> operations, PersistenceException exception) {
       92      BatchExecutorException batchExecutorException =
       93          ExceptionUtil.findBatchExecutorException(exception);
       94  
       95      if (batchExecutorException == null) {
       96        // Unexpected exception
       97        throw exception;
       98      }
       99  
      100      List<BatchResult> successfulBatches = batchExecutorException.getSuccessfulBatchResults();
      101      BatchUpdateException cause = batchExecutorException.getBatchUpdateException();
      102  
      103      Iterator<DbOperation> operationsIt = operations.iterator();
      104      List<DbOperation> failedOperations = new ArrayList<>();
      105  
      106      for (BatchResult successfulBatch : successfulBatches) {
      107        postProcessJdbcBatchResult(operationsIt, successfulBatch.getUpdateCounts(), null, failedOperations);
      108      }
      109  
      110      int[] failedBatchUpdateCounts = cause.getUpdateCounts();
      111      postProcessJdbcBatchResult(operationsIt, failedBatchUpdateCounts, exception, failedOperations);
      112  
      113      List<DbOperation> remainingOperations = CollectionUtil.collectInList(operationsIt);
      114      return FlushResult.withFailuresAndRemaining(failedOperations, remainingOperations);
      115    }

      117    /**
      118     * <p>This method can be called with three cases:
      119     *
      120     * <ul>
      121     * <li>Case 1: Success. statementResults contains the number of
      122     * affected rows for all operations.
      123     * <li>Case 2: Failure. statementResults contains the number of
      124     * affected rows for all successful operations that were executed
      125     * before the failed operation.
      126     * <li>Case 3: Failure. statementResults contains the number of
      127     * affected rows for all operations of the batch, i.e. further
      128     * statements were executed after the first failed statement.
      129     * </ul>
      130     *
      131     * <p>See {@link BatchUpdateException#getUpdateCounts()} for the specification
      132     * of cases 2 and 3.
      133     *
      134     * @return all failed operations
      135     */
      136    protected void postProcessJdbcBatchResult(
      137        Iterator<DbOperation> operationsIt,
      138        int[] statementResults,
      139        PersistenceException failure,
      140        List<DbOperation> failedOperations) {
      141      boolean failureHandled = false;
      142  
      143      for (int statementResult : statementResults) {
      144        EnsureUtil.ensureTrue("More batch results than scheduled operations detected. This indicates a bug",
      145            operationsIt.hasNext());
      146  
      147        DbOperation operation = operationsIt.next();
      148  
      149        if (statementResult == Statement.SUCCESS_NO_INFO) {
      150  
      151          if (requiresAffectedRows(operation.getOperationType())) {
      152            throw LOG.batchingNotSupported(operation);
      153          } else {
      154            postProcessOperationPerformed(operation, 1, null);
      155          }
      156  
      157        } else if (statementResult == Statement.EXECUTE_FAILED) {
      158  
      159          /*
      160           * All operations are marked with the root failure exception; this is not quite
      161           * correct and leads to the situation that we treat all failed operations in the
      162           * same way, whereas they might fail for different reasons.
      163           *
      164           * More precise would be to use BatchUpdateException#getNextException.
      165           * E.g. if we have three failed statements in a batch, #getNextException can be used to
      166           * access each operation's individual failure. However, this behavior is not
      167           * guaranteed by the java.sql javadocs (it doesn't specify that the number
      168           * and order of next exceptions matches the number of failures, unlike for row counts),
      169           * so we decided to not rely on it.
      170           */
      171          postProcessOperationPerformed(operation, 0, failure);
      172          failureHandled = true;
      173        } else { // it is the number of affected rows
      174          postProcessOperationPerformed(operation, statementResult, null);
      175        }
      176  
      177        if (operation.isFailed()) {
      178          failedOperations.add(operation); // the operation is added to the list only if it's marked as failed
      179        }
      180      }
      181  
      182      /*
      183       * case 2: The next operation is the one that failed
      184       */
      185      if (failure != null && !failureHandled) {
      186        EnsureUtil.ensureTrue("More batch results than scheduled operations detected. This indicates a bug",
      187            operationsIt.hasNext());
      188  
      189        DbOperation failedOperation = operationsIt.next();
      190        postProcessOperationPerformed(failedOperation, 0, failure);
      191        if (failedOperation.isFailed()) {
      192          failedOperations.add(failedOperation); // the operation is added to the list only if it's marked as failed
      193        }
      194      }
      195    }
