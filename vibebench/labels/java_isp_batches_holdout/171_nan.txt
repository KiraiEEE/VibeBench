       50    public void remove(final PendingTransaction invalidatedTx, final PoolRemovalReason reason) {
       51      nextLayer.remove(invalidatedTx, reason);
       52  
       53      final var senderTxs = txsBySender.get(invalidatedTx.getSender());
       54      final long invalidNonce = invalidatedTx.getNonce();
       55      if (senderTxs != null && Long.compareUnsigned(invalidNonce, senderTxs.lastKey()) <= 0) {
       56        // on sequential layers we need to push to next layer all the txs following the invalid one,
       57        // even if it belongs to a previous layer
       58  
       59        if (senderTxs.remove(invalidNonce) != null) {
       60          // invalid tx removed in this layer
       61          processRemove(senderTxs, invalidatedTx.getTransaction(), reason);
       62        }
       63  
       64        // push following to next layer
       65        pushDown(senderTxs, invalidNonce, 1);
       66  
       67        if (senderTxs.isEmpty()) {
       68          txsBySender.remove(invalidatedTx.getSender());
       69        }
       70      }
       71    }

       73    private void pushDown(

       77      senderTxs.tailMap(afterNonce, false).values().stream().toList().stream()
