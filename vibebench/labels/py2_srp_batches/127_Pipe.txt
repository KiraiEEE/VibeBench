      104  
      105          for message in reversed(messages):
      106              if message.get("role") == "user":
      107                  content = message.get("content", "")

      110                  if isinstance(content, list):
      111                      for item in content:
      112                          if item.get("type") == "text":
      113                              # Use text from the most recent user message
      114                              if not prompt:

      116                          elif item.get("type") == "image_url" and item.get(
      117                              "image_url", {}

      119                              image_urls.append(item["image_url"]["url"])
      120                  elif isinstance(content, str):
      121                      # Use text from the most recent user message
      122                      if not prompt:
      123                          prompt = content
      124  
      125                  # Extract base64 images (collect up to 3 images)
      126                  for image_url in image_urls:
      127                      if len(base64_images) >= 3:
      128                          break
      129                      try:
      130                          if "base64," in image_url:
      131                              parts = image_url.split("base64,", 1)
      132                              mime_part = parts[0]
      133                              if mime_part.startswith("data:"):
      134                                  mime_type = mime_part[5:]  # remove "data:"
      135                              else:
      136                                  mime_type = "image/jpeg"  # default
      137                              base64_data = parts[1]
      138                              base64_images.append(
      139                                  {"base64": base64_data, "mime_type": mime_type}
      140                              )
      141                          elif image_url.startswith("data:") and "," in image_url:
      142                              mime_part, base64_data = image_url.split(",", 1)
      143                              mime_type = mime_part[5:]  # remove "data:"
      144                              base64_images.append(
      145                                  {"base64": base64_data, "mime_type": mime_type}
      146                              )
      147                          else:
      148                              # assume it's base64 without prefix
      149                              base64_images.append(
      150                                  {"base64": image_url, "mime_type": "image/jpeg"}
      151                              )
      152                      except Exception as e:
      153                          logger.warning(
      154                              f"Unexpected error while extracting base64 image: {e}"
      155                          )
      156  
      157                  # Stop after processing the most recent user message
      158                  if prompt or base64_images:
      159                      break
