      336          if self.product is None:

      338  
      339          if self.is_free_product_price:
      340              return False
      341  
      342          # PWYW set to `0`, with an extra guard for metered prices
      343          # (those will still require a payment on next cycle)
      344          if self.amount == 0 and not self.has_metered_prices:
      345              return False
      346  
      347          # A 100% forever percentage discount makes the subscription entirely free
      348          # (including metered usage), so no payment setup is needed.
      349          if (
      350              self.discount is not None
      351              and self.discount.duration == DiscountDuration.forever
      352              and isinstance(self.discount, DiscountPercentage)
      353              and self.discount.basis_points == 10_000
      354          ):
      355              return False
      356  
      357          return self.product.is_recurring
