      230          """Evaluate tools asynchronously with early termination for high-scoring results.
      231          
      232          Args:
      233              query: The query to evaluate
      234              tools: List of tools to evaluate
      235              threshold: Score threshold for early termination (default 90)
      236              
      237          Returns:
      238              List of tool results with scores
      239          """
      240          # Create tasks for all tools
      241          tasks = [asyncio.create_task(self._evaluate_tool(query, tool)) for tool in tools]
      242          
      243          tool_results = []
      244          
      245          try:
      246              # Process results as they complete
      247              for completed_task in asyncio.as_completed(tasks):
      248                  try:
      249                      result = await completed_task
      250                      
      251                      if result and "score" in result:
      252                          score = int(result.get("score", 0))
      253                          tool_results.append(result)
      254                          
      255                          tool_name = result.get("tool", {}).name if result.get("tool") else "unknown"
      256                          # print(f"DEBUG: Tool {tool_name} completed with score {score}, threshold is {threshold}")

      259                          if score >= threshold:
      260                              # print(f"DEBUG: Score {score} >= threshold {threshold}, triggering early termination")
      261                              cancelled_count = 0

      292          # Cache key includes site_id

      308              types_to_check.extend(self.TYPE_HIERARCHY[schema_type])

      319              type_tools = [tool for tool in all_tools if tool.schema_type == type_name]
