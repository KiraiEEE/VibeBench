       37  class Transform(_vtk.DisableVtkSnakeCase, _vtk.vtkPyVistaOverride, _vtk.vtkTransform):
       38      """Describes linear transformations via a 4x4 matrix.
       39  
       40      A :class:`Transform` can be used to describe the full range of linear (also known
       41      as affine) coordinate transformations in three dimensions, which are internally
       42      represented as a 4x4 homogeneous transformation matrix.
       43  
       44      The transformation methods (e.g. :meth:`translate`, :meth:`rotate`,
       45      :meth:`compose`) can operate in either :meth:`pre_multiply` or
       46      :meth:`post_multiply` mode. In pre-multiply mode, any additional transformations
       47      will occur *before* any transformations represented by the current :attr:`matrix`.
       48      In post-multiply mode (the default), the additional transformation will occur
       49      *after* any transformations represented by the current matrix.
       50  
       51      .. note::
       52  
       53          This class performs all of its operations in a right-handed coordinate system
       54          with right-handed rotations. Some other graphics libraries use left-handed
       55          coordinate systems and rotations.
       56  
       57      .. versionadded:: 0.45
       58  
       59      Parameters
       60      ----------
       61      trans : TransformLike | Sequence[TransformLike], optional
       62          Initialize the transform with a transformation or sequence of transformations.
       63          By default, the transform is initialized as the identity matrix.
       64  
       65      point : VectorLike[float], optional
       66          Point to use when composing some transformations such as scale, rotation, etc.
       67          If set, two additional transformations are composed and added to
       68          the :attr:`matrix_list`:
       69  
       70              - :meth:`translate` to ``point`` before the transformation
       71              - :meth:`translate` away from ``point`` after the transformation
       72  
       73          By default, this value is ``None``, which means that the scale, rotation, etc.
       74          transformations are performed about the origin ``(0, 0, 0)``.
       75  
       76      multiply_mode : 'pre' | 'post', optional
       77          Multiplication mode to use when composing. Set this to ``'pre'`` for
       78          pre-multiplication or ``'post'`` for post-multiplication.
       79  
       80      See Also
       81      --------
       82      pyvista.DataObjectFilters.transform
       83          Apply a transformation to a mesh.
       84      pyvista.Prop3D.transform
       85          Transform an actor.
       86  
       87      Examples
       88      --------
       89      Create a transformation and use ``+`` to compose a translation.
       90  
       91      >>> import numpy as np
       92      >>> import pyvista as pv
       93      >>> position = (-0.6, -0.8, 2.1)
       94      >>> translation_T = pv.Transform() + position
       95      >>> translation_T.matrix
       96      array([[ 1. ,  0. ,  0. , -0.6],
       97             [ 0. ,  1. ,  0. , -0.8],
       98             [ 0. ,  0. ,  1. ,  2.1],
       99             [ 0. ,  0. ,  0. ,  1. ]])
      100  
      101      Using ``+`` performs the same concatenation as calling :meth:`translate`.
      102  
      103      >>> np.array_equal(
      104      ...     translation_T.matrix, pv.Transform().translate(position).matrix
      105      ... )
      106      True
      107  
      108      Create a transformation and use ``*`` to compose a scaling matrix.
      109  
      110      >>> scale_factor = 2.0
      111      >>> scaling_T = pv.Transform() * scale_factor
      112      >>> scaling_T.matrix
      113      array([[2., 0., 0., 0.],
      114             [0., 2., 0., 0.],
      115             [0., 0., 2., 0.],
      116             [0., 0., 0., 1.]])
      117  
      118      Using ``*`` performs the same concatenation as calling :meth:`scale`.
      119  
      120      >>> np.array_equal(scaling_T.matrix, pv.Transform().scale(scale_factor).matrix)
      121      True
      122  
      123      Compose the two transformations using ``*``. This will compose with
      124      post-multiplication such that the transformations are applied in order from left to
      125      right, i.e. translate first, then scale.
      126  
      127      >>> transform_post = translation_T * scaling_T
      128      >>> transform_post.matrix
      129      array([[ 2. ,  0. ,  0. , -1.2],
      130             [ 0. ,  2. ,  0. , -1.6],
      131             [ 0. ,  0. ,  2. ,  4.2],
      132             [ 0. ,  0. ,  0. ,  1. ]])
      133  
      134      Post-multiplication is equivalent to using matrix multiplication on the
      135      arrays directly but with the arguments reversed:
      136  
      137      >>> mat_mul = scaling_T.matrix @ translation_T.matrix
      138      >>> np.array_equal(transform_post.matrix, mat_mul)
      139      True
      140  
      141      Alternatively, compose the transformations by chaining the methods with a
      142      single :class:`Transform` instance. Note that post-multiply is used by default.
      143  
      144      >>> transform_post = pv.Transform()
      145      >>> transform_post.multiply_mode
      146      'post'
      147      >>> _ = transform_post.translate(position).scale(scale_factor)
      148      >>> transform_post.matrix
      149      array([[ 2. ,  0. ,  0. , -1.2],
      150             [ 0. ,  2. ,  0. , -1.6],
      151             [ 0. ,  0. ,  2. ,  4.2],
      152             [ 0. ,  0. ,  0. ,  1. ]])
      153  
      154      Use :attr:`n_transformations` to check that there are two transformations.
      155  
      156      >>> transform_post.n_transformations
      157      2
      158  
      159      Use :attr:`matrix_list` to get a list of the transformations. Since
      160      post-multiplication is used, the translation matrix is first in the list since
      161      it was applied first, and the scale matrix is second.
      162  
      163      >>> transform_post.matrix_list[0]  # translation
      164      array([[ 1. ,  0. ,  0. , -0.6],
      165             [ 0. ,  1. ,  0. , -0.8],
      166             [ 0. ,  0. ,  1. ,  2.1],
      167             [ 0. ,  0. ,  0. ,  1. ]])
      168  
      169      >>> transform_post.matrix_list[1]  # scaling
      170      array([[2., 0., 0., 0.],
      171             [0., 2., 0., 0.],
      172             [0., 0., 2., 0.],
      173             [0., 0., 0., 1.]])
      174  
      175      Create a similar transform but use pre-multiplication this time. Compose the
      176      transformations in the same order as before using :meth:`translate` and :meth:`scale`.
      177  
      178      >>> transform_pre = pv.Transform().pre_multiply()
      179      >>> _ = transform_pre.translate(position).scale(scale_factor)
      180  
      181      This is equivalent to using matrix multiplication directly on the arrays:
      182  
      183      >>> mat_mul = translation_T.matrix @ scaling_T.matrix
      184      >>> np.array_equal(transform_pre.matrix, mat_mul)
      185      True
      186  
      187      Show the matrix list again. Note how the order with pre-multiplication is the
      188      reverse of post-multiplication.
      189  
      190      >>> transform_pre.matrix_list[0]  # scaling
      191      array([[2., 0., 0., 0.],
      192             [0., 2., 0., 0.],
      193             [0., 0., 2., 0.],
      194             [0., 0., 0., 1.]])
      195  
      196      >>> transform_pre.matrix_list[1]  # translation
      197      array([[ 1. ,  0. ,  0. , -0.6],
      198             [ 0. ,  1. ,  0. , -0.8],
      199             [ 0. ,  0. ,  1. ,  2.1],
      200             [ 0. ,  0. ,  0. ,  1. ]])
      201  
      202      Apply the two post- and pre-multiplied transformations to a dataset and plot them.
      203      Note how the meshes have different positions since post- and pre-multiplication
      204      produce different transformations.
      205  
      206      >>> mesh_post = pv.Sphere().transform(transform_post, inplace=False)
      207      >>> mesh_pre = pv.Cone().transform(transform_pre, inplace=False)
      208      >>> pl = pv.Plotter()
      209      >>> _ = pl.add_mesh(mesh_post, color='goldenrod')
      210      >>> _ = pl.add_mesh(mesh_pre, color='teal')
      211      >>> _ = pl.add_axes_at_origin()
      212      >>> pl.show()
      213  
      214      Get the composed inverse transformation matrix of the pre-multiplication case.
      215  
      216      >>> inverse_matrix = transform_pre.inverse_matrix
      217      >>> inverse_matrix
      218      array([[ 0.5 ,  0.  ,  0.  ,  0.3 ],
      219             [ 0.  ,  0.5 ,  0.  ,  0.4 ],
      220             [ 0.  ,  0.  ,  0.5 , -1.05],
      221             [ 0.  ,  0.  ,  0.  ,  1.  ]])
      222  
      223      Similar to using :attr:`matrix_list`, we can inspect the individual transformation
      224      inverses with :attr:`inverse_matrix_list`.
      225  
      226      >>> transform_pre.inverse_matrix_list[0]  # inverse scaling
      227      array([[0.5, 0. , 0. , 0. ],
      228             [0. , 0.5, 0. , 0. ],
      229             [0. , 0. , 0.5, 0. ],
      230             [0. , 0. , 0. , 1. ]])
      231  
      232      >>> transform_pre.inverse_matrix_list[1]  # inverse translation
      233      array([[ 1. ,  0. ,  0. ,  0.6],
      234             [ 0. ,  1. ,  0. ,  0.8],
      235             [ 0. ,  0. ,  1. , -2.1],
      236             [ 0. ,  0. ,  0. ,  1. ]])
      237  
      238      Transform the mesh by its inverse to restore it to its original un-scaled state
      239      and positioning at the origin.
      240  
      241      >>> mesh_pre_inverted = mesh_pre.transform(inverse_matrix, inplace=False)
      242      >>> pl = pv.Plotter()
      243      >>> _ = pl.add_mesh(mesh_pre_inverted, color='teal')
      244      >>> _ = pl.add_axes_at_origin()
      245      >>> pl.show()
