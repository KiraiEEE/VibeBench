      223      @property
      224      def hvac_action(self) -> HVACAction | None:
      225          """Return current HVAC action based on actual device status."""
      226          hvac_mode = self.hvac_mode
      227          if hvac_mode == HVACMode.OFF:
      228              return HVACAction.OFF
      229  
      230          # Check both adapter (zone) and device data for most current status
      231          adapter = self.device.zone_data.get("adapter", {})
      232          device_data = self.device.device_data
      233  
      234          # Use device data if available (more current), otherwise use adapter data
      235          power = device_data.get("power", adapter.get("power", 0))
      236          operation_mode = device_data.get(
      237              "operationMode", adapter.get("operationMode", OPERATION_MODE_OFF)
      238          )
      239  
      240          if power == 0:
      241              return HVACAction.OFF
      242  
      243          # If device is on and has a valid operation mode, show it as active
      244          if operation_mode == OPERATION_MODE_HEAT:
      245              # For heating mode, show as heating if power is on
      246              return HVACAction.HEATING
      247          elif operation_mode == OPERATION_MODE_COOL:
      248              # For cooling mode, show as cooling if power is on
      249              return HVACAction.COOLING
      250          elif operation_mode == OPERATION_MODE_DRY:
      251              return HVACAction.DRYING
      252          elif operation_mode == OPERATION_MODE_VENT:
      253              return HVACAction.FAN
      254          elif operation_mode in (
      255              OPERATION_MODE_AUTO,
      256              OPERATION_MODE_AUTO_COOL,
      257              OPERATION_MODE_AUTO_HEAT,
      258          ):
      259              # For auto mode, determine action based on current vs target temperature
      260              current_temp = self.current_temperature
      261              target_temp = self.target_temperature
      262  
      263              if current_temp is not None and target_temp is not None:
      264                  temp_diff = current_temp - target_temp
      265                  if temp_diff > 1.0:  # More than 1 degree above target
      266                      return HVACAction.COOLING
      267                  elif temp_diff < -1.0:  # More than 1 degree below target
      268                      return HVACAction.HEATING
      269  
      270              # Default to idle for auto mode if we can't determine
      271              return HVACAction.IDLE
      272  
      273          # If power is on but we can't determine the action, show as idle
      274          return HVACAction.IDLE
