      318      async def update_confidence(
      319          self,
      320          pattern_id: str,
      321          success: bool,
      322          learning_rate: Optional[float] = None,
      323      ) -> PatternEmbedding:
      324          """
      325          Update pattern confidence based on usage outcome.
      326  
      327          Implements reinforcement learning-style confidence updates:
      328          - Success: Increase confidence by learning_rate
      329          - Failure: Decrease confidence by learning_rate
      330  
      331          Update Rule:
      332          confidence ← clamp(confidence + η·success_delta, 0, 1)
      333  
      334          Args:
      335              pattern_id: Pattern to update
      336              success: Whether the pattern was used successfully
      337              learning_rate: Learning rate (default: LEARNING_RATE)
      338  
      339          Returns:
      340              Updated PatternEmbedding
      341  
      342          Raises:
      343              ValueError: If pattern not found
      344          """
      345          lr = learning_rate or self.LEARNING_RATE
      346  
      347          # Fetch pattern
      348          result = await self.db.execute(
      349              select(PatternEmbedding).where(PatternEmbedding.pattern_id == pattern_id)
      350          )
      351          pattern = result.scalar_one_or_none()
      352  
      353          if not pattern:
      354              raise ValueError(f"Pattern not found: {pattern_id}")
      355  
      356          # Update confidence using pattern's built-in method
      357          old_confidence = pattern.confidence
      358          pattern.update_confidence(success=success, learning_rate=lr)
      359  
      360          logger.info(
      361              f"Updated confidence for {pattern_id}: "
      362              f"{old_confidence:.3f} -> {pattern.confidence:.3f} "
      363              f"(success={success}, lr={lr})"
      364          )
      365  
      366          await self.db.commit()
      367          await self.db.refresh(pattern)
      368  
      369          return pattern

      371      async def age_patterns(
      372          self,
      373          tenant_id: Optional[str] = None,
      374          half_life_days: Optional[int] = None,
      375          min_confidence: Optional[float] = None,
      376      ) -> int:
      377          """
      378          Age patterns based on time since last use (exponential decay).
      379  
      380          Implements exponential decay formula:
      381          confidence ← confidence × e^(-days_unused / half_life)
      382  
      383          Patterns that haven't been used recently have their confidence reduced.
      384          This encourages the system to prefer fresh, actively-used patterns.
      385  
      386          Args:
      387              tenant_id: Filter patterns by tenant
      388              half_life_days: Half-life for decay (default: AGING_HALF_LIFE_DAYS)
      389              min_confidence: Minimum confidence to maintain (default: MIN_CONFIDENCE)
      390  
      391          Returns:
      392              Number of patterns aged
      393          """
      394          hl = half_life_days or self.AGING_HALF_LIFE_DAYS
      395          min_conf = min_confidence or self.MIN_CONFIDENCE
      396  
      397          logger.info(f"Aging patterns with half_life={hl} days, min_confidence={min_conf}")
      398  
      399          # Fetch patterns that need aging
      400          query = select(PatternEmbedding)
      401          if tenant_id:
      402              query = query.where(PatternEmbedding.tenant_id == tenant_id)
      403  
      404          result = await self.db.execute(query)
      405          patterns = list(result.scalars().all())
      406  
      407          aged_count = 0
      408          now = datetime.utcnow()
      409  
      410          for pattern in patterns:
      411              # Calculate days since last use
      412              last_used = pattern.last_used_at or pattern.created_at
      413              days_unused = (now - last_used).days
      414  
      415              if days_unused == 0:
      416                  continue  # Pattern was used today, no aging
      417  
      418              # Apply exponential decay
      419              decay_factor = np.exp(-days_unused / hl)
      420              old_confidence = pattern.confidence
      421              new_confidence = max(min_conf, pattern.confidence * decay_factor)
      422  
      423              if new_confidence < old_confidence:
      424                  pattern.confidence = new_confidence
      425                  pattern.updated_at = now
      426                  aged_count += 1
