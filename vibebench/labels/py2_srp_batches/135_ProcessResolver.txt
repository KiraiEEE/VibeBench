      389      def _find_pid_for_port_sync(self, port: int) -> int | None:
      390          """Synchronous PID lookup using psutil.
      391  
      392          On Windows, uses system-wide net_connections() (doesn't require admin).
      393          On macOS/Linux, iterates user-owned processes since net_connections()
      394          requires root on those platforms.
      395          """
      396          import os
      397  
      398          # Fast path for Windows: system-wide net_connections() works without admin
      399          if self._is_windows:
      400              try:
      401                  for conn in psutil.net_connections(kind="tcp"):
      402                      if not conn.laddr or conn.laddr.port != port:
      403                          continue
      404                      if conn.raddr and conn.raddr.port == self._proxy_port and conn.pid:
      405                          return conn.pid
      406                      if conn.pid:
      407                          # Fallback: match just by local port
      408                          return conn.pid
      409              except (psutil.AccessDenied, OSError) as e:
      410                  logger.debug(f"[PSUTIL] System-wide net_connections failed: {e}")
      411                  # Fall through to per-process iteration
      412  
      413          uid = os.getuid() if hasattr(os, "getuid") else None
      414          fallback_pid: int | None = None
      415  
      416          for proc in psutil.process_iter(["pid", "uids"]):
      417              try:
      418                  # On Unix, skip processes owned by other users
      419                  if uid is not None:
      420                      proc_uids = proc.info.get("uids")
      421                      if proc_uids and proc_uids.real != uid:
      422                          continue
      423  
      424                  for conn in proc.net_connections(kind="tcp"):
      425                      if not conn.laddr or conn.laddr.port != port:
      426                          continue
      427                      # Primary: match client port connecting TO our proxy
      428                      if conn.raddr and conn.raddr.port == self._proxy_port:
      429                          logger.debug(f"[PSUTIL] Found PID {proc.pid} for port {port}")
      430                          return proc.pid
      431                      # Fallback: match just by local port (no raddr or different dest)
      432                      if fallback_pid is None:
      433                          fallback_pid = proc.pid
      434  
      435              except (psutil.AccessDenied, psutil.NoSuchProcess, psutil.ZombieProcess):
      436                  continue
      437  
      438          if fallback_pid is not None:
      439              logger.debug(f"[PSUTIL] Found PID {fallback_pid} for port {port} (fallback)")
      440          return fallback_pid

      476          """Extract process name from path."""
      477          if not path:

      479          if "\\" in path:

      482              name = path.rsplit("/", 1)[-1]
