       73      async def _read_modbus_parameters(self, transport: Any) -> dict[str, Any]:
       74          """Read configuration parameters using library's named parameter mapping.
       75  
       76          Uses pylxpweb's read_named_parameters() which maps Modbus registers
       77          to HTTP API-style parameter names automatically.
       78  
       79          Args:
       80              transport: ModbusTransport or DongleTransport instance
       81  
       82          Returns:
       83              Dictionary of parameter keys to values matching HTTP API format
       84          """
       85          params: dict[str, Any] = {}
       86  
       87          try:
       88              # Read all parameter ranges using library's register-to-name mapping
       89              # The library handles bit field extraction automatically
       90              register_ranges = [
       91                  (21, 1),  # Function enable register (bit fields)
       92                  (64, 16),  # Power settings + AC charge/discharge (64-79)
       93                  (101, 2),  # Charge/discharge current limits (101-102)
       94                  (105, 2),  # On-grid SOC cutoff (105-106)
       95                  (110, 1),  # System function register (bit fields)
       96                  (125, 1),  # Off-grid SOC cutoff (HOLD_SOC_LOW_LIMIT_EPS_DISCHG)
       97                  (151, 4),  # AC coupling start/end SOC + start/end voltage (12000XP)
       98                  (179, 1),  # Extended functions (FUNC_GRID_PEAK_SHAVING, etc.)
       99                  (227, 1),  # System charge SOC limit (HOLD_SYSTEM_CHARGE_SOC_LIMIT)
      100                  (231, 2),  # Grid peak shaving power (_12K_HOLD_GRID_PEAK_SHAVING_POWER)
      101                  (233, 1),  # Extended functions 2 (FUNC_BATTERY_BACKUP_CTRL, etc.)
      102              ]
      103  
      104              for start, count in register_ranges:
      105                  try:
      106                      named_params = await transport.read_named_parameters(start, count)
      107                      params.update(named_params)
      108                  except Exception as range_err:
      109                      _LOGGER.warning(
      110                          "Failed to read param registers %d-%d: %s",
      111                          start,
      112                          start + count - 1,
      113                          range_err,
      114                      )
      115  
      116              _LOGGER.debug("Read %d parameters from Modbus registers", len(params))
      117              # Debug: log key number entity parameters
      118              key_params = {
      119                  k: v
      120                  for k, v in params.items()
      121                  if k
      122                  in (
      123                      "HOLD_CHG_POWER_PERCENT_CMD",  # PV Charge Power (reg 64)
      124                      "HOLD_DISCHG_POWER_PERCENT_CMD",  # Discharge Power (reg 65)
      125                      "HOLD_AC_CHARGE_POWER_CMD",  # AC Charge Power (reg 66)
      126                      "HOLD_AC_CHARGE_SOC_LIMIT",  # AC Charge SOC Limit (reg 67)
      127                      "HOLD_LEAD_ACID_CHARGE_RATE",  # Charge Current (reg 101)
      128                      "HOLD_LEAD_ACID_DISCHARGE_RATE",  # Discharge Current (reg 102)
      129                      "HOLD_DISCHG_CUT_OFF_SOC_EOD",  # On-Grid SOC (reg 105)
      130                      "HOLD_SOC_LOW_LIMIT_EPS_DISCHG",  # Off-Grid SOC (reg 125)
      131                      "HOLD_AC_COUPLE_START_SOC",  # AC Coupling Start SOC (reg 151)
      132                      "HOLD_AC_COUPLE_END_SOC",  # AC Coupling End SOC (reg 152)
      133                      "HOLD_AC_COUPLE_START_VOLT",  # AC Coupling Start Voltage (reg 153)
      134                      "HOLD_AC_COUPLE_END_VOLT",  # AC Coupling End Voltage (reg 154)
      135                      "HOLD_SYSTEM_CHARGE_SOC_LIMIT",  # System Charge SOC (reg 227)
      136                      "_12K_HOLD_GRID_PEAK_SHAVING_POWER",  # Grid Peak Shaving Power (reg 231)
      137                  )
      138              }
      139              if key_params:
      140                  _LOGGER.debug("Number entity params: %s", key_params)
      141  
      142          except Exception as err:
      143              _LOGGER.warning("Failed to read parameters from Modbus: %s", err)
      144  
      145          return params

      147      def _check_grid_type_mismatch(
      148          self,
      149          serial: str,
      150          model: str,
      151          config: dict[str, Any],
      152          live_features: dict[str, Any],
      153      ) -> None:
      154          """Detect mismatch between config grid_type and live feature detection.
      155  
      156          When the user selected a wrong grid_type during setup (e.g., single_phase
      157          for a three-phase inverter), static entity creation skips phase-specific
      158          sensors. This creates a Repairs issue telling the user to reconfigure.
      159          """
      160          config_grid_type = config.get("grid_type")
      161          if not config_grid_type:
      162              return
      163  
      164          live_three = live_features.get("supports_three_phase", False)
      165          live_split = live_features.get("supports_split_phase", False)
      166  
      167          # Derive what grid_type the live detection implies
      168          if live_three:
      169              detected_type = GRID_TYPE_THREE_PHASE
      170          elif live_split:
      171              detected_type = GRID_TYPE_SPLIT_PHASE
      172          else:
      173              detected_type = GRID_TYPE_SINGLE_PHASE
      174  
      175          if config_grid_type == detected_type:
      176              return
      177  
      178          # Count phase-specific sensors that are missing due to the mismatch
      179          phase_sensor_counts = {
      180              GRID_TYPE_THREE_PHASE: len(THREE_PHASE_ONLY_SENSORS),
      181              GRID_TYPE_SPLIT_PHASE: len(SPLIT_PHASE_ONLY_SENSORS),
      182          }
      183          skipped = phase_sensor_counts.get(detected_type, 0)
      184  
      185          _LOGGER.warning(
      186              "Grid type mismatch for %s (%s): config=%s, detected=%s. "
      187              "%d phase-specific sensors are missing. "
      188              "Reconfigure the integration with the correct grid type.",
      189              serial,
      190              model,
      191              config_grid_type,
      192              detected_type,
      193              skipped,
      194          )
      195  
      196          self._grid_type_mismatch_notified.add(serial)
      197  
      198          ir.async_create_issue(
      199              self.hass,
      200              DOMAIN,
      201              f"grid_type_mismatch_{serial}",
      202              is_fixable=False,
      203              severity=ir.IssueSeverity.WARNING,
      204              translation_key="grid_type_mismatch",
      205              translation_placeholders={
      206                  "serial": serial,
      207                  "model": model,
      208                  "config_grid_type": config_grid_type,
      209                  "detected_grid_type": detected_type,
      210                  "skipped_count": str(skipped),
      211              },
      212          )

      214      def _check_missing_grid_type(
      215          self,
      216          serial: str,
      217          model: str,
      218          features: dict[str, Any],
      219      ) -> None:
      220          """Surface a Repairs issue when LXP device has no grid_type configured.
      221  
      222          LXP-EU defaults to three-phase but many residential users run
      223          single-phase. Without explicit grid_type, incorrect three-phase
      224          sensors are created.
      225          """
      226          if serial in self._grid_type_mismatch_notified:
      227              return
      228          family = features.get("inverter_family")
      229          if family != "LXP":
      230              return
      231          if not features.get("supports_three_phase", False):
      232              return
      233  
      234          self._grid_type_mismatch_notified.add(serial)
      235          ir.async_create_issue(
      236              self.hass,
      237              DOMAIN,
      238              f"grid_type_missing_{serial}",
      239              is_fixable=False,
      240              severity=ir.IssueSeverity.WARNING,
      241              translation_key="grid_type_missing",
      242              translation_placeholders={"serial": serial, "model": model},

      245      def _build_local_device_data(
      246          self,
      247          inverter: BaseInverter,
      248          serial: str,
      249          model: str,
      250          firmware_version: str,
      251          connection_type: str,
      252      ) -> dict[str, Any]:
      253          """Build device data structure for local transport (Modbus/Dongle).
      254  
      255          Args:
      256              inverter: BaseInverter with populated transport data
      257              serial: Device serial number
      258              model: Device model name
      259              firmware_version: Firmware version string
      260              connection_type: CONNECTION_TYPE_MODBUS or CONNECTION_TYPE_DONGLE
      261  
      262          Returns:
      263              Dictionary with device data, sensors, and features
      264          """
      265          device_data: dict[str, Any] = {
      266              "type": "inverter",
      267              "model": model,
      268              "serial": serial,
      269              "firmware_version": firmware_version,
      270              "sensors": _build_runtime_sensor_mapping(inverter._transport_runtime),
      271              "batteries": {},
      272          }
      273  
      274          if energy_data := inverter._transport_energy:
      275              device_data["sensors"].update(_build_energy_sensor_mapping(energy_data))
      276  
      277          if battery_data := inverter._transport_battery:
      278              device_data["sensors"].update(
      279                  _build_battery_bank_sensor_mapping(battery_data)
      280              )
      281              # Compute battery bank charge/discharge rate from merged sensor data
      282              compute_bank_charge_rates(device_data["sensors"])
      283  
      284          device_data["sensors"]["firmware_version"] = firmware_version
      285          device_data["sensors"]["connection_transport"] = _get_transport_label(
      286              connection_type
      287          )
      288  
      289          # Add computed sensors from inverter properties (for deprecated code path)
      290          if (val := inverter.consumption_power) is not None:
      291              device_data["sensors"]["consumption_power"] = val
      292          if (val := inverter.total_load_power) is not None:
      293              device_data["sensors"]["total_load_power"] = val
      294          if (val := inverter.battery_power) is not None:
      295              device_data["sensors"]["battery_power"] = val
      296          if (val := inverter.rectifier_power) is not None:
      297              device_data["sensors"]["rectifier_power"] = val
      298          if (val := inverter.power_to_user) is not None:
      299              device_data["sensors"]["grid_import_power"] = val
      300          # EPS per-leg power (computed from total EPS + voltage ratio)
      301          device_data["sensors"]["eps_power_l1"] = inverter.eps_power_l1
      302          device_data["sensors"]["eps_power_l2"] = inverter.eps_power_l2
      303  
      304          transport = getattr(inverter, "_transport", None)
      305          if transport and hasattr(transport, "host"):
      306              device_data["sensors"]["transport_host"] = transport.host
      307  
      308          # Add last_polled timestamp so users can see when data was last fetched
      309          # (not just when it last changed)
      310          device_data["sensors"]["last_polled"] = dt_util.utcnow()
      311  
      312          # Extract features for capability-based sensor filtering
      313          if features := self._extract_inverter_features(inverter):
      314              device_data["features"] = features
      315              _LOGGER.debug(
      316                  "%s: Features for %s: %s",
      317                  connection_type.upper(),
      318                  serial,
      319                  features,
      320              )
      321  
      322          return device_data

      324      async def _async_update_local_transport_data(
      325          self,
      326          transport: Any,
      327          serial: str,
      328          model: str,
      329          connection_type: str,
      330      ) -> dict[str, Any]:
      331          """Fetch data from a single local transport (Modbus or Dongle).
      332  
      333          DEPRECATED: Used by _async_update_modbus_data/_async_update_dongle_data.
      334          Remove in v4.0 — use CONNECTION_TYPE_LOCAL with local_transports instead.
      335          Does NOT support GridBOSS devices (only creates BaseInverter).
      336  
      337          Args:
      338              transport: The transport instance to use
      339              serial: Device serial number
      340              model: Device model name
      341              connection_type: CONNECTION_TYPE_MODBUS or CONNECTION_TYPE_DONGLE
      342  
      343          Returns:
      344              Dictionary containing device data from registers.
      345          """
      346          from pylxpweb.transports.exceptions import (
      347              TransportConnectionError,
      348              TransportError,
      349              TransportReadError,
      350              TransportTimeoutError,
      351          )
      352  
      353          transport_name = connection_type.capitalize()
      354  
      355          try:
      356              _LOGGER.debug("Fetching %s data for inverter %s", transport_name, serial)
      357  
      358              if not transport.is_connected:
      359                  await transport.connect()
      360  
      361              # Create or reuse BaseInverter via factory
      362              if serial not in self._inverter_cache:
      363                  _LOGGER.debug(
      364                      "Creating BaseInverter from %s transport for %s",
      365                      transport_name,
      366                      serial,
      367                  )
      368                  if connection_type == "dongle":
      369                      inverter = await BaseInverter.from_dongle_transport(
      370                          transport, model=model
      371                      )
      372                      self._align_inverter_cache_ttls(inverter, "wifi_dongle")
      373                  else:
      374                      inverter = await BaseInverter.from_modbus_transport(
      375                          transport, model=model
      376                      )
      377                      self._align_inverter_cache_ttls(inverter, "modbus_tcp")
      378                  self._inverter_cache[serial] = inverter
      379              else:
      380                  inverter = self._inverter_cache[serial]
      381  
      382              include_params = self._local_parameters_loaded
      383              await inverter.refresh(include_parameters=include_params)
      384  
      385              # Cache firmware version - only read once from transport, reuse on subsequent updates
      386              if serial not in self._firmware_cache:
      387                  self._firmware_cache[serial] = (
      388                      await transport.read_firmware_version() or "Unknown"
      389                  )
      390              firmware_version = self._firmware_cache[serial]
      391  
      392              if inverter._transport_runtime is None:
      393                  raise TransportReadError(
      394                      f"Failed to read runtime data from {transport_name}"
      395                  )
      396  
      397              device_data = self._build_local_device_data(
      398                  inverter=inverter,
      399                  serial=serial,
      400                  model=model,
      401                  firmware_version=firmware_version,
      402                  connection_type=connection_type,
      403              )
      404  
      405              processed: dict[str, Any] = {
      406                  "plant_id": None,
      407                  "devices": {serial: device_data},
      408                  "device_info": {},
      409                  "last_update": dt_util.utcnow(),
      410                  "connection_type": connection_type,
      411              }
      412  
      413              if include_params:
      414                  param_data = await self._read_modbus_parameters(transport)
      415              else:
      416                  param_data = {}
      417              processed["parameters"] = {serial: param_data}
      418  
      419              if not self._last_available_state:
      420                  _LOGGER.warning(
      421                      "EG4 %s connection restored for inverter %s", transport_name, serial
      422                  )
      423                  self._last_available_state = True
      424  
      425              runtime = inverter._transport_runtime
      426              _LOGGER.debug(
      427                  "%s update complete - FW: %s, PV: %.0fW, SOC: %d%%, Grid: %.0fW",
      428                  transport_name,
      429                  firmware_version,
      430                  runtime.pv_total_power,
      431                  runtime.battery_soc,
      432                  runtime.grid_power,
      433              )
      434  
      435              # Schedule deferred parameter load on first successful refresh
      436              if not self._local_parameters_loaded:
      437                  self._local_parameters_loaded = True
      438                  _LOGGER.info(
      439                      "%s: First refresh complete. Scheduling background parameter load.",
      440                      transport_name,
      441                  )
      442                  task = self.hass.async_create_task(
      443                      self._deferred_local_parameter_load()
      444                  )
      445                  self._background_tasks.add(task)
      446                  task.add_done_callback(self._remove_task_from_set)
      447                  task.add_done_callback(self._log_task_exception)
      448  
      449              return processed
      450  
      451          except TransportConnectionError as e:
      452              self._log_transport_error(f"{transport_name} connection lost", serial, e)
      453              raise UpdateFailed(f"{transport_name} connection failed: {e}") from e
      454  
      455          except TransportTimeoutError as e:
      456              self._log_transport_error(f"{transport_name} timeout", serial, e)
      457              raise UpdateFailed(f"{transport_name} timeout: {e}") from e
      458  
      459          except (TransportReadError, TransportError) as e:
      460              self._log_transport_error(f"{transport_name} read error", serial, e)
      461              raise UpdateFailed(f"{transport_name} read error: {e}") from e
      462  
      463          except Exception as e:
      464              self._log_transport_error(
      465                  f"Unexpected {transport_name} error", serial, e, log_exception=True
      466              )
      467              raise UpdateFailed(f"Unexpected error: {e}") from e
