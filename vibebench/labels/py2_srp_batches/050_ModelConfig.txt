      364      def _derive_context_length(self, context_length: int):
      365          is_draft_model = self.is_draft_model
      366          derived_context_len = get_context_length(self.hf_text_config)
      367  
      368          if context_length is not None:
      369              if context_length > derived_context_len:
      370                  reason = "Target model's" if is_draft_model else "User-specified"
      371                  msg = (
      372                      f"Warning: {reason} context_length ({context_length}) is greater than the derived context_length ({derived_context_len}). "
      373                      f"This may lead to incorrect model outputs or CUDA errors. Note that the derived context_length may differ from max_position_embeddings in the model's config."
      374                  )
      375                  if (
      376                      envs.SGLANG_ALLOW_OVERWRITE_LONGER_CONTEXT_LEN.get()
      377                      or is_in_ci()  # FIXME: fix this special case
      378                  ):
      379                      logger.warning(msg)
      380                      self.context_len = context_length
      381                      if is_draft_model:
      382                          self.hf_text_config.max_position_embeddings = context_length
      383                          logger.warning(
      384                              f"Overriding the draft model's max_position_embeddings to {context_length}."
      385                          )
      386                  else:
      387                      raise ValueError(
      388                          f"{msg} To allow overriding this maximum, set the env var SGLANG_ALLOW_OVERWRITE_LONGER_CONTEXT_LEN=1"
      389                      )
      390              else:
      391                  self.context_len = context_length
      392          else:
      393              self.context_len = derived_context_len
      394  
      395          # Transfer context_len to HuggingFace config so models can access it
      396          self.hf_config.context_len = self.context_len
