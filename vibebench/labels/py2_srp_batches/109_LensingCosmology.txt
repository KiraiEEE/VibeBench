      121      def cosmic_average_density_solar_mass_per_kpc3_from(
      122          self,
      123          redshift: float,
      124          xp=np,
      125      ):
      126          """
      127          Critical density of the Universe at redshift z in units of Msun / kpc^3.
      128  
      129          JAX / NumPy compatible via `xp`.
      130  
      131          Computes:
      132  
      133              rho_c(z) = 3 H(z)^2 / (8 pi G)
      134  
      135          Returns physical density (not scaled into arcsec units).
      136          """
      137  
      138          # -----------------------------
      139          # Physical constants
      140          # -----------------------------
      141  
      142          # Gravitational constant in kpc^3 / (Msun s^2)
      143          G = xp.asarray(4.30091e-6)  # kpc (km/s)^2 / Msun
      144          G = G / xp.asarray((3.085677581e16) ** 2)
      145  
      146          # H0 in km/s/Mpc → convert to 1/s
      147          H0_km_s_Mpc = xp.asarray(self.H0)
      148          H0_s = H0_km_s_Mpc / xp.asarray(3.085677581e19)
      149  
      150          # Dimensionless expansion factor
      151          Ez = self.E(redshift, xp=xp)
      152  
      153          # H(z) in 1/s
      154          Hz = H0_s * Ez
      155  
      156          # -----------------------------
      157          # Critical density in Msun/kpc^3
      158          # -----------------------------
      159  
      160          rho_crit = (3.0 * Hz**2) / (8.0 * xp.pi * G)
      161  
      162          return rho_crit

      164      def critical_density(self, z: float, xp=np):
      165          """
      166          Critical density of the Universe at redshift z, returned in Msun / kpc^3.
      167  
      168          This is an xp (NumPy / JAX) drop-in for the Astropy method:
      169              astropy.cosmology.FLRW.critical_density(z)
      170  
      171          Astropy returns g/cm^3, but in AutoLens you were immediately converting to:
      172              solMass / kpc^3
      173  
      174          So this function returns Msun / kpc^3 directly.
      175  
      176          Requires:
      177              self.H0  (km/s/Mpc)
      178              self.E(z, xp=xp)  where E(z) = H(z)/H0
      179          """
      180  
      181          z = xp.asarray(z)
      182  
      183          # H0 in km/s/Mpc -> 1/s
      184          # 1 Mpc = 3.085677581e19 km
      185          H0_s = xp.asarray(self.H0) / xp.asarray(3.085677581e19)
      186  
      187          # H(z) in 1/s
      188          Hz = H0_s * self.E(z, xp=xp)
      189  
      190          # Gravitational constant G in kpc^3 / (Msun s^2)
      191          # Start from: G = 4.30091e-6 kpc (km/s)^2 / Msun
      192          # Convert (km/s)^2 -> (kpc/s)^2 by dividing by (km per kpc)^2
      193          km_per_kpc = xp.asarray(3.085677581e16)
      194          G = xp.asarray(4.30091e-6) / (km_per_kpc**2)
      195  
      196          # rho_c = 3 H(z)^2 / (8 pi G)  [Msun / kpc^3]
      197          return (xp.asarray(3.0) * Hz**2) / (xp.asarray(8.0) * xp.pi * G)

      221      def critical_surface_density_between_redshifts_solar_mass_per_kpc2_from(
      222          self,
      223          redshift_0: float,
      224          redshift_1: float,
      225          xp=np,
      226      ):
      227          """
      228          Critical surface density in physical units (Msun / kpc^2):
      229  
      230              Sigma_crit = (c^2 / (4*pi*G)) * D_s / (D_l * D_ls)
      231  
      232          Distances must be angular diameter distances in kpc.
      233  
      234          JAX/NumPy compatible via `xp` (pass `jax.numpy` as xp).
      235          """
      236  
      237          # Distances in kpc
      238          D_l_kpc = self.angular_diameter_distance_to_earth_in_kpc_from(
      239              redshift=redshift_0, xp=xp
      240          )
      241          D_s_kpc = self.angular_diameter_distance_to_earth_in_kpc_from(
      242              redshift=redshift_1, xp=xp
      243          )
      244          D_ls_kpc = self.angular_diameter_distance_between_redshifts_in_kpc_from(
      245              redshift_0=redshift_0, redshift_1=redshift_1, xp=xp
      246          )
      247  
      248          # kpc -> m
      249          kpc_to_m = xp.asarray(3.085677581491367e19)
      250          D_l = D_l_kpc * kpc_to_m
      251          D_s = D_s_kpc * kpc_to_m
      252          D_ls = D_ls_kpc * kpc_to_m
      253  
      254          # SI constants
      255          c = xp.asarray(299792458.0)  # m/s
      256          G = xp.asarray(6.67430e-11)  # m^3/(kg s^2)
      257          Msun = xp.asarray(1.98847e30)  # kg
      258  
      259          # Sigma_crit in kg / m^2
      260          prefac = (c * c) / (xp.asarray(4.0) * xp.pi * G)
      261          sigma_SI = prefac * D_s / (D_l * D_ls)
      262  
      263          # kg/m^2 -> Msun/kpc^2
      264          sigma = sigma_SI * (kpc_to_m * kpc_to_m) / Msun
      265  
      266          return xp.where(D_ls_kpc == xp.asarray(0.0), xp.asarray(np.inf), sigma)

      268      def scaling_factor_between_redshifts_from(
      269          self,
      270          redshift_0: float,
      271          redshift_1: float,
      272          redshift_final: float,
      273          xp=np,
      274      ):
      275          """
      276          For strong lens systems with more than 2 planes, the deflection angles between different planes must be scaled
      277          by the angular diameter distances between the planes in order to properly perform multi-plane ray-tracing. This
      278          function computes the factor to scale deflections between `redshift_0` and `reshift_final`, to deflections between
      279          `redshift_0` and `redshift_1`.
      280  
      281          The second redshift should be strictly larger than the first. The scaling factor is unity when `redshift_1`
      282          is `redshift_final`, and 0 when `redshift_0` is equal to `redshift_1`.
      283  
      284          For a system with a first lens galaxy l0 at `redshift_0`, second lens galaxy l1 at `redshift_1` and final
      285          source galaxy at `redshift_final` this scaling factor is given by:
      286  
      287          (D_l0l1 * D_s) / (D_l1* D_l0s)
      288  
      289          The critical surface density for lensing, often written as $\\sigma_{cr}$, is given by:
      290  
      291          critical_surface_density = (c^2 * D_s) / (4 * pi * G * D_ls * D_l)
      292  
      293          D_l0l1 = Angular diameter distance of first lens redshift to second lens redshift.
      294          D_s = Angular diameter distance of source redshift to earth
      295          D_l1 = Angular diameter distance of second lens redshift to Earth.
      296          D_l0s = Angular diameter distance of first lens redshift to source redshift
      297  
      298          For systems with more planes this scaling factor is computed multiple times for the different redshift
      299          combinations and applied recursively when scaling the deflection angles.
      300  
      301          Parameters
      302          ----------
      303          redshift_0
      304              The redshift of the first strong lens galaxy (the first lens plane).
      305          redshift_1
      306              The redshift of the second strong lens galaxy (the second lens plane).
      307          redshift_final
      308              The redshift of the final source galaxy (the final source plane).
      309          """
      310  
      311          # D_l0l1 : between lens plane 0 and lens plane 1
      312          D_l0l1 = self.angular_diameter_distance_between_redshifts_in_kpc_from(
      313              redshift_0=redshift_0,
      314              redshift_1=redshift_1,
      315              xp=xp,
      316          )
      317  
      318          # D_s : observer to final source plane
      319          D_s = self.angular_diameter_distance_to_earth_in_kpc_from(
      320              redshift=redshift_final,
      321              xp=xp,
      322          )
      323  
      324          # D_l1 : observer to lens plane 1
      325          D_l1 = self.angular_diameter_distance_to_earth_in_kpc_from(
      326              redshift=redshift_1,
      327              xp=xp,
      328          )
      329  
      330          # D_l0s : between lens plane 0 and final source plane
      331          D_l0s = self.angular_diameter_distance_between_redshifts_in_kpc_from(
      332              redshift_0=redshift_0,
      333              redshift_1=redshift_final,
      334              xp=xp,
      335          )
      336  
      337          return (D_l0l1 * D_s) / (D_l1 * D_l0s)

      339      def velocity_dispersion_from(
      340          self, redshift_0: float, redshift_1: float, einstein_radius: float, xp=np
      341      ) -> float:
      342          """
      343          For a strong lens galaxy with an Einstein radius in arcseconds, the corresponding velocity dispersion of the
      344          lens galaxy can be computed (assuming an isothermal mass distribution).
      345  
      346          The velocity dispersion is given by:
      347  
      348          sigma_v = c * sqrt((R_Ein * D_s) / (4 * pi * D_l * D_ls))
      349  
      350          c = speed of light
      351          D_s = Angular diameter distance of source redshift to earth
      352          D_ls = Angular diameter distance of lens redshift to source redshift
      353          D_l = Angular diameter distance of lens redshift to earth
      354  
      355          Parameters
      356          ----------
      357          redshift_0
      358              The redshift of the first strong lens galaxy (the lens).
      359          redshift_1
      360              The redshift of the second strong lens galaxy (the source).
      361          """
      362  
      363          # Speed of light in km / s
      364          c_km_s = xp.asarray(299792.458)
      365  
      366          # Angular diameter distances in kpc
      367          D_l = self.angular_diameter_distance_to_earth_in_kpc_from(
      368              redshift=redshift_0, xp=xp
      369          )
      370  
      371          D_s = self.angular_diameter_distance_to_earth_in_kpc_from(
      372              redshift=redshift_1, xp=xp
      373          )
      374  
      375          D_ls = self.angular_diameter_distance_between_redshifts_in_kpc_from(
      376              redshift_0=redshift_0, redshift_1=redshift_1, xp=xp
      377          )
      378  
      379          # Convert Einstein radius to kpc
      380          kpc_per_arcsec = self.kpc_per_arcsec_from(redshift=redshift_0, xp=xp)
      381          R_ein_kpc = xp.asarray(einstein_radius) * kpc_per_arcsec
      382  
      383          # Velocity dispersion (km/s)
      384          sigma_v = c_km_s * xp.sqrt(
      385              (R_ein_kpc * D_s) / (xp.asarray(4.0) * xp.pi * D_l * D_ls)
      386          )
      387  
      388          return sigma_v
