      293      def get_optimizer_state(self, optimizer: Optimizer) -> dict[str, Tensor]:
      294          """Returns state of an optimizer.
      295  
      296          Allows for syncing/collating optimizer state from processes in custom plugins.
      297  
      298          """
      299          if hasattr(optimizer, "consolidate_state_dict"):
      300              # there are optimizers like PyTorch's ZeroRedundancyOptimizer that shard their
      301              # states, and to avoid OOM we consolidate the full state on rank 0 only
      302              optimizer.consolidate_state_dict()

      305          # for optimizers that are not sharded, we return the state dict on all ranks

      308      def load_checkpoint(
      309          self,
      310          path: _PATH,
      311          state: Optional[Union[Module, Optimizer, dict[str, Union[Module, Optimizer, Any]]]] = None,
      312          strict: bool = True,
      313          weights_only: Optional[bool] = None,
      314      ) -> dict[str, Any]:
      315          """Load the contents from a checkpoint and restore the state of the given objects.
      316  
      317          Args:
      318              path: A path to where the file is located
      319              state: Can be one of:
      320  
      321                  - A dictionary of objects whose state will be restored in-place from the checkpoint path.
      322                  - ``None`` or the empty dict: The loaded checkpoint will be returned in full.
      323                  - A :class:`~torch.nn.Module` instance, if the checkpoint file contains a raw module state dict.
      324                  - A :class:`~torch.optim.Optimizer` instance, if the checkpoint file contains a raw optimizer state.
      325  
      326              strict: Whether to enforce that the keys in `state` match the keys in the checkpoint.
      327              weights_only: Defaults to ``None``. If ``True``, restricts loading to ``state_dicts`` of plain
      328                  ``torch.Tensor`` and other primitive types. If loading a checkpoint from a trusted source that contains
      329                  an ``nn.Module``, use ``weights_only=False``. If loading checkpoint from an untrusted source, we
      330                  recommend using ``weights_only=True``. For more information, please refer to the
      331                  `PyTorch Developer Notes on Serialization Semantics <https://docs.pytorch.org/docs/main/notes/serialization.html#id3>`_.
      332  
      333          Returns:
      334              The remaining items that were not restored into the given state dictionary. If no state dictionary is
      335              given, the full checkpoint will be returned.
      336  
      337          """
      338          torch.cuda.empty_cache()
      339          checkpoint = self.checkpoint_io.load_checkpoint(path, weights_only=weights_only)
      340          if not state:
      341              return checkpoint
      342  
      343          if isinstance(state, Module):
      344              self.load_module_state_dict(module=state, state_dict=checkpoint, strict=strict)
      345              return {}
      346  
      347          if isinstance(state, Optimizer):
      348              state.load_state_dict(checkpoint)
      349              return {}
      350  
      351          _validate_keys_for_strict_loading(state.keys(), checkpoint.keys(), strict=strict)
      352          for name, obj in state.copy().items():
      353              if name not in checkpoint:
      354                  continue
      355              if isinstance(obj, _Stateful):
      356                  if isinstance(obj, Module):
      357                      self.load_module_state_dict(module=obj, state_dict=checkpoint.pop(name), strict=strict)
      358                  else:
      359                      obj.load_state_dict(checkpoint.pop(name))
      360              else:
      361                  state[name] = checkpoint.pop(name)
      362          return checkpoint

      406      def _convert_stateful_objects_in_state(
      407          self, state: dict[str, Union[Module, Optimizer, Any]], filter: dict[str, Callable[[str, Any], bool]]
      408      ) -> dict[str, Any]:
      409          converted_state: dict[str, Any] = {}
      410          for key, obj in state.items():
      411              # convert the state
      412              if isinstance(obj, Module):
      413                  converted = self.get_module_state_dict(module=obj)
      414              elif isinstance(obj, Optimizer):
      415                  converted = self.get_optimizer_state(optimizer=obj)
      416              elif isinstance(obj, _Stateful):
      417                  converted = obj.state_dict()
      418              else:
      419                  converted = obj
      420              _apply_filter(key, filter, converted, converted_state)
      421          return converted_state
