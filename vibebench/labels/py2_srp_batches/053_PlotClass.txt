       74      def __date_list__(self):
       75          """ Make dictionary of time differences between successive epochs. """
       76          dateList = []
       77          tbase = []
       78          # Get list of epochs
       79          for di in self.pairs:
       80              dates = os.path.basename(di).split('_')
       81              if not dates[1] in dateList:
       82                  dateList.append(dates[1])
       83              if not dates[0] in dateList:
       84                  dateList.append(dates[0])
       85  
       86          dateList.sort()
       87          d1 = datetime(*time.strptime(dateList[0], "%Y%m%d")[0:5])
       88          for ni in enumerate(dateList):
       89              d2 = datetime(*time.strptime(dateList[ni[0]], "%Y%m%d")[0:5])
       90              diff = d2 - d1
       91              tbase.append(diff.days)
       92          dateDict = {}
       93          for i in enumerate(dateList):
       94              dateDict[dateList[i[0]]] = tbase[i[0]]
       95          return dateDict

       97      def __design_matrix__(self):
       98          """ Make the design matrix for the inversion. """
       99  
      100          dateDict = self.__date_list__()
      101          numDates = len(dateDict)
      102          numIfgrams = len(self.pairs)
      103  
      104          # Initialize matrices for inversion
      105          A = np.zeros((numIfgrams, numDates))
      106          B = np.zeros(np.shape(A))
      107          L = np.zeros((numIfgrams, 1))
      108          daysList = []
      109          for day in dateDict.values():
      110              daysList.append(day)
      111  
      112          tbase = np.array(list(dateDict.values()))
      113          t = np.zeros((numIfgrams, 2))
      114          baseline_hist = []
      115  
      116          # Iterate through all IFGs
      117          for i in enumerate(self.product_dict[0]):
      118              date12 = self.product_dict[1][i[0]][0]
      119              date = date12.split('_')
      120              ndxt2 = daysList.index(dateDict[date[0]])
      121              ndxt1 = daysList.index(dateDict[date[1]])
      122              A[[i[0]], ndxt1] = -1
      123              A[[i[0]], ndxt2] = 1
      124              B[[i[0]], ndxt1:ndxt2] = tbase[ndxt1 +
      125                                             1:ndxt2 + 1] - tbase[ndxt1:ndxt2]
      126              t[[i[0]], :] = [dateDict[date[1]], dateDict[date[0]]]
      127  
      128              # Report mean
      129              pbaseline_nodata = gdal.Open(i[1][0])
      130              pbaseline_nodata = pbaseline_nodata.GetRasterBand(
      131                  1).GetNoDataValue()
      132              pbaseline_val = gdal.BuildVRT('', i[1]).ReadAsArray()
      133              pbaseline_val = np.ma.masked_where(
      134                  pbaseline_val == pbaseline_nodata, pbaseline_val)
      135              pbaseline_val = pbaseline_val.mean()
      136              # Record baseline val for histogram
      137              baseline_hist.append(pbaseline_val)
      138              L[i[0]] = float(pbaseline_val)
      139              if (np.isnan(L[i[0]])):
      140                  L[i[0]] = 0.0
      141              pbaseline_val = None
      142              pbaseline_nodata = None
      143  
      144          A = A[:, 1:]
      145          B = B[:, :-1]
      146  
      147          ind = ~np.isnan(L)
      148          return A[ind[:, 0], :], B[ind[:, 0], :], L[ind], baseline_hist
