      503      def _decode_action(self, action: int) -> tuple[str | None, str]:
      504          """Decode flat action index to (target_symbol, direction).
      505  
      506          Action layout: 0=flat, 1..S*psa=long, S*psa+1..2*S*psa=short
      507          where psa = per_symbol_actions (alloc_bins * level_bins).
      508          """
      509          if action == 0:
      510              return None, "flat"
      511          S = self.num_symbols
      512          psa = self.per_symbol_actions
      513          side_block = S * psa
      514          if action <= side_block:
      515              idx = action - 1
      516              sym_idx = idx // psa
      517              return self.symbols[sym_idx], "long"
      518          elif action <= 2 * side_block:
      519              idx = action - 1 - side_block
      520              sym_idx = idx // psa
      521              return self.symbols[sym_idx], "short"
      522          return None, "flat"

      529      def _mask_logits(self, logits: np.ndarray, tradable_symbols: list[str]) -> np.ndarray:
      530          """Mask logits for non-tradable symbols to -inf. Action 0 (FLAT) is never masked."""
      531          masked = logits.copy()
      532          tradable_set = set(tradable_symbols)
      533          for i in range(1, len(masked)):

      535              if sym is not None and sym not in tradable_set:
      536                  masked[i] = -np.inf

      548      @staticmethod
      549      def _softmax(logits: np.ndarray) -> np.ndarray:
      550          finite = np.isfinite(logits)
      551          if not finite.any():
      552              probs = np.zeros_like(logits, dtype=np.float64)
      553              if len(probs) > 0:
      554                  probs[0] = 1.0
      555              return probs
      556          shifted = np.full_like(logits, -np.inf, dtype=np.float64)
      557          shifted[finite] = logits[finite] - np.max(logits[finite])
      558          exp = np.exp(shifted)
      559          exp[~finite] = 0.0
      560          total = exp.sum()
      561          if not np.isfinite(total) or total <= 0.0:
      562              probs = np.zeros_like(logits, dtype=np.float64)
      563              if len(probs) > 0:
      564                  probs[0] = 1.0
      565              return probs
      566          return exp / total  # type: ignore[no-any-return]

      568      def _decode_action_metadata(self, action: int, direction: str) -> tuple[float, float]:
      569          if action <= 0 or direction == "flat":
      570              return 0.0, 0.0
      571          side_block = self.num_symbols * self.per_symbol_actions
      572          local_idx = action - 1
      573          if direction == "short":
      574              local_idx -= side_block
      575          if local_idx < 0:
      576              return 0.0, 0.0
      577          rem = local_idx % max(1, self.per_symbol_actions)
      578          alloc_idx = rem // max(1, self.action_level_bins)
      579          level_idx = rem % max(1, self.action_level_bins)
      580          allocation_pct = float(alloc_idx + 1) / float(max(1, self.action_allocation_bins))
      581          if self.action_level_bins <= 1:
      582              return min(1.0, allocation_pct), 0.0
      583          frac = float(level_idx) / float(max(1, self.action_level_bins - 1))
      584          return min(1.0, allocation_pct), 2.0 * frac - 1.0
