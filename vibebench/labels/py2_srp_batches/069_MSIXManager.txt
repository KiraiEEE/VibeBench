      234      def inject_data(self, result: Dict[str, Any], msix_data: MSIXData) -> None:
      235          """
      236          Inject preloaded MSI-X data into the generation result.
      237  
      238          Args:
      239              result: The generation result dictionary to update
      240              msix_data: The preloaded MSI-X data
      241          """
      242          if not self._should_inject(msix_data):
      243              return
      244  
      245          log_info_safe(
      246              self.logger, safe_format("Using preloaded MSI-X data"), prefix="MSIX"
      247          )
      248  
      249          # msix_info is guaranteed to be non-None by _should_inject
      250          if msix_data.msix_info is not None:
      251              if "msix_data" not in result or not result["msix_data"]:
      252                  result["msix_data"] = self._create_msix_result(msix_data.msix_info)
      253  
      254              # Update template context if present
      255              if (
      256                  "template_context" in result
      257                  and "msix_config" in result["template_context"]
      258              ):
      259                  result["template_context"]["msix_config"].update(
      260                      {
      261                          "is_supported": True,
      262                          "num_vectors": msix_data.msix_info["table_size"],
