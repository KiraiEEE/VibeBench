      249  @dataclass
      250  class RelevanceScorerConfig:
      251      """Configuration for relevance scoring in SmartCrusher.
      252  
      253      Relevance scoring determines which items to keep when compressing
      254      tool outputs. Uses the pattern: relevance(item, context) -> [0, 1].
      255  
      256      Available tiers:
      257      - "bm25": BM25 keyword matching (zero dependencies, fast)
      258      - "embedding": Semantic similarity via sentence-transformers
      259      - "hybrid": BM25 + embedding with adaptive fusion (RECOMMENDED)
      260  
      261      DEFAULT: "hybrid" - combines exact matching (UUIDs, IDs) with semantic
      262      understanding. Falls back to BM25 if sentence-transformers not installed.
      263  
      264      For full hybrid support, install: pip install headroom[relevance]
      265  
      266      WHY HYBRID IS DEFAULT:
      267      - Missing important items during compression is catastrophic
      268      - BM25 alone gives low scores for single-term matches (e.g., "Alice" = 0.07)
      269      - Semantic matching catches "errors" -> "failed", "issues", etc.
      270      - 5-10ms latency is acceptable vs. losing critical data
      271      """
      272  
      273      tier: Literal["bm25", "embedding", "hybrid"] = "hybrid"
      274  
      275      # BM25 parameters
      276      bm25_k1: float = 1.5  # Term frequency saturation
      277      bm25_b: float = 0.75  # Length normalization
      278  
      279      # Embedding parameters
      280      embedding_model: str = field(default_factory=lambda: ML_MODEL_DEFAULTS.sentence_transformer)
      281  
      282      # Hybrid parameters
      283      hybrid_alpha: float = 0.5  # BM25 weight (1-alpha = embedding weight)
      284      adaptive_alpha: bool = True  # Adjust alpha based on query type
      285  
      286      # Scoring thresholds
      287      # With hybrid/embedding: semantic scores are meaningful (0.3-0.5 for good matches)
      288      # With BM25 fallback: threshold is still reasonable for multi-term matches
      289      # Lower threshold = safer (keeps more items), higher = more aggressive compression
      290      relevance_threshold: float = 0.25  # Keep items above this score
