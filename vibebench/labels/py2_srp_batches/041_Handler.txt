       63          self, messages: List[Dict[str, Any]]
       64      ) -> tuple[str, List[Dict[str, Any]]]:
       65          """
       66          Convert Chat Completions API messages format to Responses API format.
       67          Returns (instructions, input_messages) tuple.
       68          """
       69          instructions = ""
       70          input_messages = []
       71  
       72          for msg in messages:
       73              role = msg.get("role")
       74              content = msg.get("content")
       75  
       76              if role == "system":
       77                  # System messages become instructions in Responses API
       78                  instructions = content
       79              elif role == "user":
       80                  # User messages map to "user" type
       81                  input_messages.append({"type": "user", "content": content})
       82              elif role == "assistant":
       83                  # Assistant messages map to "assistant" type
       84                  tool_calls = msg.get("tool_calls")
       85                  if tool_calls:
       86                      # Assistant message with tool calls
       87                      input_messages.append(
       88                          {"type": "assistant", "tool_calls": tool_calls}
       89                      )
       90                  else:
       91                      # Regular assistant message
       92                      input_messages.append({"type": "assistant", "content": content})
       93              elif role == "tool":
       94                  # Tool result messages map to "tool_result" type
       95                  input_messages.append(
       96                      {
       97                          "type": "tool_result",
       98                          "tool_call_id": msg.get("tool_call_id"),
       99                          "content": content,
      100                      }
      101                  )
      102  
      103          return instructions, input_messages

      170      def _get_completion_responses_api(
      171          self,
      172          model: str,
      173          temperature: float,
      174          top_p: float,
      175          messages: List[Dict[str, Any]],
      176          functions: Optional[List[Dict[str, str]]],
      177      ) -> Generator[str, None, None]:
      178          """Handle completion using the new Responses API."""
      179          tool_call_id = name = arguments = ""
      180          
      181          # Convert messages to Responses API format
      182          instructions, input_items = self._messages_to_responses_api_format(messages)
      183          
      184          # Build request parameters
      185          request_params: Dict[str, Any] = {
      186              "model": model,
      187              "temperature": temperature,
      188              "top_p": top_p,
      189              "input": input_items if input_items else "",
      190          }
      191          
      192          # Add instructions if present
      193          if instructions:
      194              request_params["instructions"] = instructions
      195          
      196          # Add tool/function parameters if present
      197          if functions:
      198              request_params["tool_choice"] = "auto"
      199              request_params["tools"] = functions
      200              request_params["parallel_tool_calls"] = False
      201          
      202          # Call the Responses API stream method
      203          try:
      204              response_stream = completion(**request_params, **additional_kwargs)
      205              
      206              # Iterate through the stream events
      207              for event in response_stream:
      208                  event_type = event.type
      209                  
      210                  # Handle text delta events (main content)
      211                  if event_type == "response.output_text.delta":
      212                      yield event.delta
      213                  
      214                  # Handle function call argument deltas
      215                  elif event_type == "response.function_call_arguments.delta":
      216                      arguments += event.delta
      217                      # Store the item_id as tool_call_id
      218                      tool_call_id = event.item_id
      219                  
      220                  # Handle function call arguments done (get function name)
      221                  elif event_type == "response.function_call_arguments.done":
      222                      # We need to get the function name from the output item
      223                      pass
      224                  
      225                  # Handle output item added (includes function name)
      226                  elif event_type == "response.output_item.added":
      227                      if hasattr(event, 'item') and event.item.get('type') == 'function_call':
      228                          name = event.item.get('name', '')
      229                          tool_call_id = event.item.get('id', '')
      230                  
      231                  # Handle output item done
      232                  elif event_type == "response.output_item.done":
      233                      # Check if this was a function call that completed
      234                      if hasattr(event, 'item'):
      235                          item = event.item
      236                          if item.get('type') == 'function_call' and arguments:
      237                              # Execute the function call
      238                              yield from self.handle_function_call(
      239                                  messages, tool_call_id, name, arguments
      240                              )
      241                              
      242                              # Recursive call for next completion
      243                              yield from self.get_completion(
      244                                  model=model,
      245                                  temperature=temperature,
      246                                  top_p=top_p,
      247                                  messages=messages,
      248                                  functions=functions,
      249                                  caching=False,
      250                              )
      251                              return
      252                  
      253          except KeyboardInterrupt:
      254              if hasattr(response_stream, 'close'):
      255                  response_stream.close()
