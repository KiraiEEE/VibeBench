      105          """Get current configuration merging data and options.
      106  
      107          Home Assistant OptionsFlow saves to entry.options, not entry.data.
      108          This method merges both, with options taking precedence.
      109          """
      110          entry = self._get_entry()
      111          # entry.options might be empty dict or not exist (in tests)
      112          options = getattr(entry, "options", {}) or {}
      113          # entry.data is a mappingproxy in real HA, dict or Mock in tests
      114          # Convert to dict for merging - check if it's dict-like first
      115          try:
      116              data = dict(entry.data) if entry.data else {}
      117          except (TypeError, AttributeError):
      118              data = entry.data if isinstance(entry.data, dict) else {}
      119          try:
      120              options = dict(options) if options else {}
      121          except (TypeError, AttributeError):
      122              options = options if isinstance(options, dict) else {}
      123          return {**data, **options}
