      234      def _build_mcp_servers_option(self) -> tuple[dict[str, Any] | str, Path | None]:
      235          """Return MCP servers value for ClaudeAgentOptions and optional temp file.
      236  
      237          On Windows the ``--mcp-config`` CLI argument is processed through
      238          three escaping layers — Python ``list2cmdline``, ``cmd.exe``
      239          (because ``claude.cmd`` is a batch wrapper), and Node.js arg
      240          parsing.  JSON with embedded backslash-escaped Windows paths and
      241          quotes is routinely corrupted by this chain, causing the CLI to
      242          receive a malformed MCP config.  Tool listing may still work but
      243          ``tools/call`` hangs because the server env is wrong.
      244  
      245          The fix: on Windows, write the MCP config JSON to a temp file and
      246          pass the **file path** to ``--mcp-config`` instead of inline JSON.
      247          This sidesteps all command-line escaping issues.
      248          """
      249          servers = self._build_mcp_servers()
      250  
      251          if sys.platform != "win32":
      252              return servers, None
      253  
      254          # Write MCP config to a temp file on Windows
      255          mcp_config = {"mcpServers": servers}
      256          fd, mcp_tmp_path = tempfile.mkstemp(suffix=".json", prefix="aw-mcpcfg-")
      257          try:
      258              os.write(fd, json.dumps(mcp_config, ensure_ascii=False).encode("utf-8"))
      259          finally:
      260              os.close(fd)
      261          logger.info("MCP config written to temp file: %s", mcp_tmp_path)
      262          return mcp_tmp_path, Path(mcp_tmp_path)
