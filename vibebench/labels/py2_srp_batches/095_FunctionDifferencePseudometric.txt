      171      def _generate_grid_points(self) -> None:
      172          """
      173          Generate a grid of sample points within the domain bounds.
      174  
      175          The generated points are stored in self._sample_points.
      176          """
      177          # Extract dimensions and bounds
      178          dimensions = list(self.domain_bounds.keys())
      179          bounds = [self.domain_bounds[dim] for dim in dimensions]
      180  
      181          # Calculate points per dimension
      182          # For n dimensions, we want approximately num_samples total points
      183          # So we need approximately num_samples^(1/n) points per dimension
      184          points_per_dim = max(2, int(self.num_samples ** (1 / len(dimensions))))
      185  
      186          # Generate grid points for each dimension
      187          grid_points = []
      188          for lower, upper in bounds:
      189              if points_per_dim > 1:
      190                  step = (upper - lower) / (points_per_dim - 1)
      191                  dim_points = [lower + i * step for i in range(points_per_dim)]
      192              else:
      193                  dim_points = [(lower + upper) / 2]  # Just the midpoint
      194              grid_points.append(dim_points)
      195  
      196          # Generate all combinations
      197          self._sample_points = []
      198  
      199          if len(dimensions) == 1:
      200              # 1D case - return scalars
      201              self._sample_points = grid_points[0]
      202          else:
      203              # Multi-dimensional case - return dicts
      204              # Use numpy's meshgrid to generate all combinations
      205              mesh = np.meshgrid(*grid_points, indexing="ij")
      206              grid_shape = mesh[0].shape
      207  
      208              for idx in np.ndindex(grid_shape):
      209                  point = {dim: mesh[i][idx] for i, dim in enumerate(dimensions)}
      210                  self._sample_points.append(point)
      211  
      212              # If we have too many points, randomly sample down to num_samples
      213              if len(self._sample_points) > self.num_samples:
      214                  self._sample_points = random.sample(
      215                      self._sample_points, self.num_samples

      281      def _calculate_difference(
      282          self, values1: List[float], values2: List[float]
      283      ) -> float:
      284          """
      285          Calculate the difference between two sets of function values.
      286  
      287          Parameters
      288          ----------
      289          values1 : List[float]
      290              The values of the first function at the evaluation points.
      291          values2 : List[float]
      292              The values of the second function at the evaluation points.
      293  
      294          Returns
      295          -------
      296          float
      297              The difference between the function values, based on the selected norm.
      298  
      299          Raises
      300          ------
      301          ValueError
      302              If the lists have different lengths.
      303          """
      304          if len(values1) != len(values2):
      305              raise ValueError("Function value lists must have the same length")
      306  
      307          # Calculate differences
      308          differences = [abs(v1 - v2) for v1, v2 in zip(values1, values2)]
      309  
      310          # Apply the selected norm
      311          if self.norm_type == "l1":
      312              # L1 norm (sum of absolute differences)
      313              return sum(differences)
      314          elif self.norm_type == "l2":
      315              # L2 norm (Euclidean distance)
      316              return np.sqrt(sum(d**2 for d in differences))
      317          elif self.norm_type == "max":
      318              # Maximum norm (maximum absolute difference)
      319              return max(differences) if differences else 0.0
      320          else:
      321              # This should never happen due to validation in __init__
      322              raise ValueError(f"Unsupported norm type: {self.norm_type}")
