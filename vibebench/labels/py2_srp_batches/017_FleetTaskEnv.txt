      307          try:
      308              await self._ensure_provisioned()

      310              # Emit rollout_completed so init failures are tracked in dashboards
      311              fleet_info(
      312                  "fleet_rollout_completed",
      313                  step_count=0,
      314                  reward=0.0,
      315                  verifier_success=False,
      316                  failure_reason="init_error",
      317              )
      318              self._rollout_completed_emitted = True
      319              raise

      321          # Reset episode state
      322          self._step_count = 0

      324          self._reward_computed = False
      325          self.final_reward = None
      326          self._submitted_answer = None
      327          self._tool_errors = []
      328          self._verifier_stdout = None
      329          self._verifier_error = None

      332          # reset() failure is non-fatal — env is up, just the manager API timed out
      333          reset_metadata = {}
      334          if self._orch:
      335              try:
      336                  saved_timeout = self._orch._timeout
      337                  self._orch._timeout = self.reset_timeout_s
      338                  try:
      339                      reset_result = await self._orch.reset_async()

      341                          reset_result.observation.metadata if reset_result else {}
      342                      )
      343                  finally:
      344                      self._orch._timeout = saved_timeout
      345              except Exception as e:
      346                  logger.warning(
      347                      f"[env={self.env_key}] Fleet env reset failed (timeout={self.reset_timeout_s}s), continuing with empty observation: {e}"
      348                  )
      349                  fleet_warning(
      350                      "fleet_env_reset_failed",
      351                      step_count=self._step_count,
      352                      timeout_s=self.reset_timeout_s,
      353                      error_type=type(e).__name__,
      354                      error_message=str(e)[:200],
      355                  )
      356  
      357          # Fetch tools — fatal if MCP call fails (no tools = dead rollout)
      358          try:
      359              if self._tools:

      362              if not self._tools_cache:
      363                  raise RuntimeError("list_tools returned no tools")
      364          except Exception as e:
      365              fleet_info(
      366                  "fleet_rollout_completed",
      367                  step_count=0,
      368                  reward=0.0,
      369                  verifier_success=False,
      370                  failure_reason="tools_error",
      371                  error_message=str(e)[:200],

      393                  or t.get("function", {}).get("name") == "computer"

      397                      t.get("name") or t.get("function", {}).get("name")
      398                      for t in self._tools_cache
      399                  ]

      403                      reward=0.0,
      404                      verifier_success=False,

      411                      f"Available tools: {available}. Check MCP image configuration."
      412                  )
      413              self._tools_cache = computer_tools

      415          if not self._tools_cache:

      417                  "fleet_rollout_completed",
      418                  step_count=0,
      419                  reward=0.0,
      420                  verifier_success=False,
      421                  failure_reason="tools_error",

      430          # they would in a Fleet harness session.

      432              self._tools_cache.append(SUBMIT_FINAL_ANSWER_TOOL)

      436              "prompt": self.prompt,
      437              "observation": reset_metadata,
      438              "step": 0,
      439              "task_key": self.task_key,
      440              "modality": self.modality,
      441              "tools": self._tools_cache,

      459                      step_count=self._step_count,
      460                  )
