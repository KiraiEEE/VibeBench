      241  class SampleMetadata(_MetadataBase, SkbioObject):
      242      """Store metadata associated with identifiers in a study.
      243  
      244      Metadata is tabular in nature, mapping study identifiers (e.g. sample or
      245      feature IDs) to columns of metadata associated with each ID.
      246  
      247      For more details about metadata in QIIME 2, including the TSV metadata file
      248      format, see the Metadata Tutorial at https://docs.qiime2.org.
      249  
      250      The following text focuses on design and considerations when working with
      251      ``Metadata`` objects at the API level.
      252  
      253      A ``Metadata`` object is composed of zero or more ``MetadataColumn``
      254      objects. A ``Metadata`` object always contains at least one ID, regardless
      255      of the number of columns. Each column in the ``Metadata`` object has an
      256      associated column type representing either *categorical* or *numeric*
      257      data. Each metadata column is represented by an object corresponding to the
      258      column's type: ``CategoricalMetadataColumn`` or ``NumericMetadataColumn``,
      259      respectively.
      260  
      261      A ``Metadata`` object is closely linked to its corresponding TSV metadata
      262      file format described at https://docs.qiime2.org. Therefore, certain
      263      requirements present in the file format are also enforced on the in-memory
      264      object in order to make serialized ``Metadata`` objects roundtrippable when
      265      loaded from disk again. For example, IDs cannot begin with a pound
      266      character (``#``) because those IDs would be interpreted as comment rows
      267      when written to disk as TSV. See the metadata file format spec for more
      268      details about data formatting requirements.
      269  
      270      In addition to being loaded from or saved to disk, a ``Metadata`` object
      271      can be constructed from a ``pandas.DataFrame`` object. See the *Parameters*
      272      section below for details on how to construct ``Metadata`` objects from
      273      dataframes.
      274  
      275      ``Metadata`` objects have various methods to access, filter, and merge
      276      data. A dataframe can be retrieved from the ``Metadata`` object for further
      277      data manipulation using the pandas API. Individual ``MetadataColumn``
      278      objects can be retrieved to gain access to APIs applicable to a single
      279      metadata column.
      280  
      281      Missing values may be encoded in one of the following schemes:
      282  
      283      'blank'
      284        The default, which treats `None`/`NaN` as the only valid missing values.
      285  
      286      'no-missing'
      287        Indicates there are no missing values in a column, any `None`/`NaN`
      288        values should be considered an error. If a scheme other than 'blank' is
      289        used by default, this scheme can be provided to preserve strings as
      290        categorical terms.
      291  
      292      'INSDC:missing'
      293        The INSDC vocabulary for missing values. The current implementation
      294        supports only lower-case terms which match exactly:
      295        'not applicable', 'missing', 'not provided', 'not collected', and
      296        'restricted access'.
      297  
      298      Parameters
      299      ----------
      300      dataframe : pandas.DataFrame
      301          Dataframe containing metadata. The dataframe's index defines the IDs,
      302          and the index name (``Index.name``) must match one of the required ID
      303          headers described in the metadata file format spec. Each column in the
      304          dataframe defines a metadata column, and the metadata column's type
      305          (i.e. *categorical* or *numeric*) is determined based on the column's
      306          dtype. If a column has ``dtype=object``, it may contain strings or
      307          pandas missing values (e.g. ``np.nan``, ``None``). Columns matching
      308          this requirement are assumed to be *categorical*. If a column in the
      309          dataframe has ``dtype=float`` or ``dtype=int``, it may contain floating
      310          point numbers or integers, as well as pandas missing values
      311          (e.g. ``np.nan``). Columns matching this requirement are assumed to be
      312          *numeric*. Regardless of column type (categorical vs numeric), the
      313          dataframe stored within the ``Metadata`` object will have any missing
      314          values normalized to ``np.nan``. Columns with ``dtype=int`` will be
      315          cast to ``dtype=float``. To obtain a dataframe from the ``Metadata``
      316          object containing these normalized data types and values, use
      317          ``Metadata.to_dataframe()``.
      318      column_missing_schemes : dict, optional
      319          Describe the metadata column handling for missing values described
      320          in the dataframe. This is a dict mapping column names (str) to
      321          missing-value schemes (str).  Valid values are 'blank',
      322          'no-missing', and 'INSDC:missing'. Column names may be omitted.
      323      default_missing_scheme : str, optional
      324          The missing scheme to use when none has been provided in the file
      325          or in `column_missing_schemes`.

      449      def _normalize_dataframe(
      450          self, dataframe, column_missing_schemes, default_missing_scheme
      451      ):
      452          norm_df = dataframe.copy()
      453  
      454          # Do not attempt to strip empty metadata
      455          if not norm_df.columns.empty:
      456              norm_df.columns = norm_df.columns.str.strip()
      457  
      458          norm_df.index = norm_df.index.str.strip()
      459  
      460          columns = OrderedDict()
      461          for column_name, series in norm_df.items():
      462              missing_scheme = column_missing_schemes.get(
      463                  column_name, default_missing_scheme
      464              )
      465              metadata_column = self._metadata_column_factory(series, missing_scheme)
      466              norm_df[column_name] = metadata_column.to_series()
      467              properties = ColumnProperties(
      468                  type=metadata_column.type, missing_scheme=missing_scheme
      469              )

      474      def _metadata_column_factory(self, series, missing_scheme):
      475          series = _missing.series_encode_missing(series, missing_scheme)
      476          # Collapse dtypes except for all NaN columns so that we can preserve
      477          # empty categorical columns. Empty numeric columns will already have
      478          # the expected dtype and values
      479          if not series.isna().all():
      480              series = series.infer_objects()
      481          dtype = series.dtype
      482          if NumericMetadataColumn._is_supported_dtype(dtype):
      483              column = NumericMetadataColumn(series, missing_scheme)
      484          elif CategoricalMetadataColumn._is_supported_dtype(dtype):
      485              column = CategoricalMetadataColumn(series, missing_scheme)
      486          else:
      487              raise TypeError(
      488                  "Metadata column %r has an unsupported pandas dtype of %s. "
      489                  "Supported dtypes: float, int, object" % (series.name, dtype)
      490              )
      491  
      492          return column

      585      def to_dataframe(self, encode_missing=False):
      586          """Create a pandas dataframe from the metadata.
      587  
      588          The dataframe's index name (``Index.name``) will match this metadata
      589          object's ``id_header``, and the index will contain this metadata
      590          object's IDs. The dataframe's column names will match the column names
      591          in this metadata. Categorical columns will be stored as
      592          ``dtype=object`` (containing strings), and numeric columns will be
      593          stored as ``dtype=float``.
      594  
      595          Parameters
      596          ----------
      597          encode_missing : bool, optional
      598              Whether to convert missing values (NaNs) back into their original
      599              vocabulary (strings) if a missing scheme was used.
      600  
      601          Returns
      602          -------
      603          pandas.DataFrame
      604              Dataframe constructed from the metadata.
      605  
      606          """
      607          df = self._dataframe.copy()
      608          if encode_missing:
      609  
      610              def replace_nan(series):
      611                  missing = _missing.series_extract_missing(series)
      612                  # avoid dtype changing if there's no missing values
      613                  if not missing.empty:
      614                      series = series.astype(object)
      615                      series[missing.index] = missing
      616                  return series
      617  
      618              df = df.apply(replace_nan)
      619  
      620          return df
