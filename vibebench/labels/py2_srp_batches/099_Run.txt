      347      def process_features(self, features):
      348          """Utility which handles postprocess_features."""
      349          if features is None:
      350              return []
      351          features = get_matching_features(features, FeatureType.RUN)
      352          for feature in features:
      353              assert feature.name in self.FEATURES, f"Incompatible feature: {feature.name}"
      354              feature.used = True
      355              tmp_run_config = {f"run.{key}": value for key, value in self.run_config.items()}
      356              feature.add_run_config(tmp_run_config)
      357              self.run_config = filter_config(tmp_run_config, "run", self.DEFAULTS, self.OPTIONAL, self.REQUIRED)
      358          return features

      448      def has_stage(self, stage):
      449          """Returns true if the given stage is available for this run."""
      450          if stage == RunStage.NOP:
      451              return True
      452          if stage == RunStage.LOAD:
      453              return self.model is not None and len(self.frontends) > 0
      454          if stage == RunStage.TUNE:
      455              return self.tune_enabled and self.backend is not None
      456          if stage == RunStage.BUILD:
      457              return self.backend is not None and self.framework is not None
      458          if stage == RunStage.COMPILE:
      459              return self.target is not None and len(self.platforms) > 0 and self.compile_platform is not None
      460          if stage == RunStage.RUN:
      461              return self.target is not None and len(self.platforms) > 0 and self.target_platform is not None
      462          if stage == RunStage.POSTPROCESS:
      463              return len(self.postprocesses) > 0
      464          if stage == RunStage.DONE:
      465              return False
      466          return False  # TODO: Throw error instead?
