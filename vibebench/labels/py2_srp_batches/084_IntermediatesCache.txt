       64      """
       65      Cache which stores intermediate values (activations) produced by batched, sequential
       66      execution of models. Values are offloaded to the `offload_device` when stored in
       67      the cache and onloaded to their original device when fetched from the cache. If
       68      `offload_device` is None, values will not be offloaded at all.

      101      @classmethod
      102      def from_dataloader(
      103          cls,
      104          dataloader: torch.utils.data.DataLoader,
      105          model_device: torch.device = torch.device("cpu"),
      106          offload_device: torch.device | None = torch.device("cpu"),
      107      ):
      108          """
      109          Initialize a cache with data from the provided dataloader
      110  
      111          This method iterates through all batches in the dataloader and offloads
      112          them to the specified device. For faster cache preparation, consider:
      113          - Increasing batch_size to reduce the number of iterations
      114          - Using num_workers > 0 in the DataLoader for parallel loading (e.g. the
      115            calibration DataLoader from format_calibration_data uses
      116            dataloader_num_workers; when > 0, pin_memory and prefetch_factor are
      117            also set where applicable, which speeds both cache build and calibration)
      118          - Ensuring data preprocessing is done before creating the dataloader
      119  
      120          :param dataloader: dataloader which generates values to be cached
      121          :param model_device: device which values will be onloaded to when fetched
      122          :param offload_device: device to offload values to
      123          """
      124          batch_intermediates = [
      125              {
      126                  key: cls._offload_value(value, offload_device, model_device)
      127                  for key, value in batch.items()
      128              }
      129              for batch in tqdm(dataloader, desc="Preparing cache")
      130          ]
      131  
      132          return cls(batch_intermediates, offload_device)

      190      def size(self) -> dict[torch.device, int]:
      191          """
      192          Returns the memory used by cached values, keyed by device, in bytes
      193  
      194          :return: dictionary mapping torch device to number of bytes in cache
      195          """
      196          sizes = defaultdict(lambda: 0)
      197          memo = set()
      198  
      199          def _size_helper(intermediate: IntermediateValue) -> int:
      200              value = intermediate.value
      201  
      202              match value:
      203                  case torch.Tensor():
      204                      if value not in memo:
      205                          sizes[value.device] += value.nbytes
      206                      memo.add(value)
      207                  case list() | tuple():
      208                      for v in value:
      209                          _size_helper(v)
      210                  case dict():
      211                      for v in value.values():
      212                          _size_helper(v)
      213                  case _ if is_dataclass(value):
      214                      for field in fields(value):
      215                          _size_helper(getattr(value, field.name))
      216                  case _:
      217                      # this handles primitive values that don't match any other cases
      218                      sizes[torch.device("cpu")] += sys.getsizeof(value, 0)
      219  
      220          for intermediates in self.batch_intermediates:
      221              for value in intermediates.values():
      222                  _size_helper(value)
      223  
      224          return dict(sizes)

      236      @classmethod
      237      def _onload_value(cls, intermediate: IntermediateValue) -> Any:
      238          """
      239          Onload a value's tensors to the onload device
      240  
      241          :param intermediate: intermediates value representation to onload
      242          :return: original value with tensors onloaded to the onload device
      243          """
      244          value = intermediate.value
      245          device = intermediate.device
      246  
      247          match value:
      248              case torch.Tensor():
      249                  return value.to(device=device)
      250              case list():
      251                  return [cls._onload_value(v) for v in value]
      252              case tuple():
      253                  return tuple(cls._onload_value(v) for v in value)
      254              case dict():
      255                  return {k: cls._onload_value(v) for k, v in value.items()}
      256              case _ if is_dataclass(value):
      257                  for field in fields(value):
      258                      v = getattr(value, field.name)
      259                      setattr(value, field.name, cls._onload_value(v))
      260                  return value
      261              case _:
      262                  # handles primitive values that should be returned as is.
      263                  # without this, a MatchError would be raised for unhandled types.
      264                  return value

      266      @classmethod
      267      def _offload_value(
      268          cls,
      269          value: Any,
      270          offload_device: torch.device | None,
      271          onload_device: torch.device | None = None,
      272      ) -> IntermediateValue:
      273          """
      274          Offload a value's tensors to the offload device
      275  
      276          :param value: value to offload
      277          :param offload_device: device to offload `torch.Tensor` values to
      278          :param onload_device: device used when onloading `torch.Tensor` values.
      279              If None is provided, use the tensor's current device
      280          :return: Instance of IntermediateValue representing the offloaded value
      281          """
      282          kwargs = {"offload_device": offload_device, "onload_device": onload_device}
      283          match value:
      284              case torch.Tensor():
      285                  with OverrideEqMode():
      286                      # check for cache hit between shared tensors
      287                      if value in cls.offload_values:
      288                          offloaded = cls.offload_values[value]
      289                      else:
      290                          # move to offload if no hit
      291                          offloaded = value.to(device=offload_device)
      292                          cls.offload_values[value] = offloaded
      293  
      294                  return IntermediateValue(
      295                      value=offloaded,
      296                      device=(onload_device if onload_device else value.device),
      297                  )
      298              case list():
      299                  return IntermediateValue(
      300                      value=[cls._offload_value(v, **kwargs) for v in value],
      301                      device=None,
      302                  )
      303              case tuple():
      304                  return IntermediateValue(
      305                      value=tuple(cls._offload_value(v, **kwargs) for v in value),
      306                      device=None,
      307                  )
      308              case dict():
      309                  return IntermediateValue(
      310                      value={
      311                          k: cls._offload_value(v, **kwargs) for k, v in value.items()
      312                      },
      313                      device=None,
      314                  )
      315              case _ if is_dataclass(value):
      316                  for field in fields(value):
      317                      v = getattr(value, field.name)
      318                      setattr(value, field.name, cls._offload_value(v, **kwargs))
      319                  return IntermediateValue(value=value, device=None)
      320              case _:
      321                  # handles primitive values and provides a warning for unsupported types.
      322                  # without this, values trigger a MatchError exception.
      323                  if not isinstance(
      324                      value,
      325                      (int, str, float, bool, torch.dtype, torch.device, type(None)),
      326                  ):
      327                      warnings.warn(f"Offloading not implemented for type {type(value)}.")
      328                  return IntermediateValue(value=value, device=None)
