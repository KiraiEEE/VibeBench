      301

      306                  return self._safe_numeric_compare(actual, expected, lambda a, e: a < e)
      307

      313              
      314              elif operator == Operator.IN_LIST:

      339      def _safe_regex_match(self, actual: Any, pattern: str) -> bool:
      340          """Safely perform regex matching with security controls."""
      341          if not isinstance(actual, str):
      342              return False
      343          
      344          # Security check: limit regex pattern length
      345          if len(pattern) > self.max_regex_length:
      346              logger.error(f"Regex pattern too long: {len(pattern)}")
      347              return False
      348          
      349          try:
      350              # Use cache to avoid recompiling patterns
      351              if pattern not in self._regex_cache:
      352                  if len(self._regex_cache) >= self._cache_size_limit:
      353                      # Clear oldest entries
      354                      self._regex_cache.clear()
      355                  
      356                  # Compile with timeout protection
      357                  compiled_pattern = re.compile(pattern, re.IGNORECASE)
      358                  self._regex_cache[pattern] = compiled_pattern
      359              
      360              regex = self._regex_cache[pattern]
      361              return bool(regex.search(actual))
      362              
      363          except re.error as e:
      364              logger.warning(f"Invalid regex pattern '{pattern}': {e}")

      398              elif len(condition.value) > self.max_regex_length:

      401                  try:
