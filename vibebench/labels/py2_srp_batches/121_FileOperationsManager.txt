      192          should_prompt = False

      200              # No file path means new/untitled graph

      203          if should_prompt:

      205              settings_key = f"graph_env_choice/{self.current_graph_name}"
      206              saved_choice = self.settings.value(settings_key, None)

      208              if saved_choice:

      214                  from ui.dialogs.environment_selection_dialog import EnvironmentSelectionDialog
      215                  dialog = EnvironmentSelectionDialog(self.current_graph_name, self.parent_window)
      216                  if dialog.exec():

      218                      # Save the choice for future loads
      219                      self.settings.setValue(settings_key, selected_option)
      220                      self._apply_environment_selection(selected_option)
      221                  else:
      222                      # User cancelled - default to default environment and save choice
