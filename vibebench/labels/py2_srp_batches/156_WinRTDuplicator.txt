      410      def _drain_to_latest_frame(self) -> tuple[Any | None, int, bool]:
      411          latest = None
      412          drained = 0
      413          while True:
      414              frame, failed = self._try_get_next_frame()
      415              if failed:
      416                  if latest is not None:
      417                      self._close_winrt_object(latest, "dropped frame")
      418                  return None, drained, True
