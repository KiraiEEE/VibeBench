      349      def find_most_divergent_representatives_from_genetree(self, genetree_subtree:TreeNode):
      350          """this method returns the N most divergent representatives from a rooted tree. within each cluster
      351          it selects the one representative sequence that corresponds to the the median distance from the root
      352          among the sequnces in the cluster.
      353  
      354          :param genetree: the rooted tree (with distances) that should be processed
      355          :return: the N most divergent representatives for that tree. N is taken from self.conf"""
      356          genetree = copy_tree(genetree_subtree)
      357  
      358          # we remove the non-enabled representatives from the tree
      359          keep = [n for n in genetree.iter_leaves() if self._rep_lookup[n.name].representative.enabled]
      360          if len(keep) < len(genetree):
      361              genetree.prune(keep, preserve_branch_length=True)
      362          if len(genetree) <= self.conf.number_of_samples_per_hog:
      363              return [Representative(self._rep_lookup[n.name].representative) for n in genetree.iter_leaves()]
      364  
      365          # first, annotate the distance from the root
      366          for n in genetree.traverse():
      367              n.add_feature('dist_from_root', n.dist + (n.up.dist_from_root if n.up is not None and hasattr(n.up, 'dist_from_root') else 0))
      368          # now, let's get a list of all the internal nodes ordered with the dist_from_root and find which distance
      369          # results in number_of_samples_per_hog subtrees that are most divergent
      370          ordered = sorted([n for n in genetree.traverse() if not n.is_leaf()], key=lambda x: x.dist_from_root)
      371          idx = min(len(ordered), self.conf.number_of_samples_per_hog) - 1  # in case of multi-fucations, len(ordered) can be shorter than number_of_samples_per_hog
      372          cutoff = ordered[idx].dist_from_root
      373          sub_clades = [sub for sub in genetree.iter_leaves(
      374              is_leaf_fn=lambda n: len(n.children) == 0 or n.dist_from_root >= cutoff
      375          )]
      376          # now, let's select for each sub_clade the one leave that is close to the median dist_from_root (no outliers)
      377          representatives = []
      378          for sub in sub_clades:
      379              order_by_dist = sorted((n for n in sub.iter_leaves()), key=lambda n: n.dist_from_root)
      380              median_elem = order_by_dist[len(order_by_dist)//2]
      381              new_rep = Representative(self._rep_lookup[median_elem.name].representative,
      382                                       [self._rep_lookup[z.name].representative for z in order_by_dist])
      383              representatives.append(new_rep)
      384          return representatives

      512      def infer_reconciliation(self, genetree:TreeNode, sos_threshold=0.0):
      513          """Annotate each internal node with a 'evoltype' and 'sos' attribute.
      514  
      515          The evoltype attribute will be either 'S' or 'D' (speciation or duplication),
      516          and the 'sos' attribute will contain the fraction of species that overlap among the
      517          subtrees. The parameter sos_threshold specifies the minimum fraction of species that
      518          must overlap among the subtrees in order to be considered a duplication.
      519  
      520          :param genetree: the rooted tree that should be processed
      521          :param sos_threshold: the minimum fraction of species that must overlap among the
      522                                subtrees in order to be considered a duplication.
      523          """
      524          logger.info("\n" + genetree.get_ascii(show_internal=True, attributes=['name', 'hogid']))
      525          cnt_D, cnt_S = 0, 0
      526          for n in genetree.traverse('postorder'):
      527              if n.is_leaf():
      528                  n.add_feature('species', self._rep_lookup[n.name].representative.get_species())
      529                  n.add_feature('hogid', self._rep_lookup[n.name].hog.hogid)
      530              else:
      531                  assert len(n.children) >= 2, f"{n.name} has {len(n.children)} children"
      532                  specs = tuple(c.species for c in n.children)
      533                  for c in n.children:
      534                      c.del_feature('species')  # cleanup to avoid excessive memory
      535                  sp_inter = set.intersection(*specs)
      536                  sp_union = set.union(*specs)
      537                  sos = len(sp_inter) / len(sp_union)
      538                  n.add_feature('species', sp_union)
      539                  n.add_feature('sos', sos)
      540                  n.add_feature('so_tuple', (len(sp_inter), len(sp_union)))
      541                  n.add_feature('evoltype', 'D' if sos > sos_threshold else 'S')
      542                  if sos > sos_threshold:
      543                      cnt_D += 1
      544                      n.name = f"D{cnt_D}_{len(sp_inter)}_{len(sp_union)}"
      545                  else:
      546                      cnt_S += 1
      547                      n.name = f"S{cnt_S}"
      548          genetree.del_feature('species')
      549          self.write_msa_or_tree_if_necessary(genetree, fn_suffix="_rec.nwk", features=['evoltype', 'sos', 'so_tuple', 'hogid'])

      551      def _compute_between_and_within_distances(self, genetree, subtrees_node, partitions):
      552          mrca = genetree.get_common_ancestor(*subtrees_node)
      553          id2part = {id_: i for i, p in enumerate(partitions) for id_ in p}
      554          N = len(partitions)
      555          dists = collections.defaultdict(list)
      556          for l1, l2 in itertools.combinations(mrca.iter_leaves(), 2):
      557              i1 = id2part.get(l1.name, N)

      563      def _resolve_conflicts(self, gene_tree: TreeNode, subtrees_node, partitions) -> bool:
      564          """resolves cases of conflicting subhog splittings where
      565          the representatives of a subhog are split among different subtrees
      566          and have been merged even before the previous level.
      567  
      568          the method returns a boolean value indicating whether a new reconciliation
      569          is necessary (e.g. in case the tree has been modified and hence the labeling
      570          of speciation/duplication might have changed).
      571          """
      572          updated_subtree_nodes = []
      573          for n in subtrees_node:
      574              # previous rounds of _resolve_conflicts might have left disconnected subtree_nodes
      575              # so we identify the first node which is either a leaf or a bifurcating node.
      576              while len(n.children) == 1:
      577                  n = n.children[0]
      578              updated_subtree_nodes.append(n)
      579          subtrees_node = updated_subtree_nodes
      580          mrca = gene_tree.get_common_ancestor(*subtrees_node)
      581          min_support = min(n.support for n in subtrees_node)
      582          min_branch_to_subtree = min(gene_tree.get_distance(mrca, n) for n in subtrees_node)
      583          partitions = sorted(partitions, key=lambda x: [-sum(len(self._rep_lookup[r].representative.get_subelements()) for r in x), -len(x),])
      584          if False and (min_support < 0.7 or min_branch_to_subtree < 0.01):
      585              # we collapse things, i.e. separate nodes in tree based on hogid
      586              # we mv all leaves from the non-biggest partitions to the mrca of the biggest partion
      587              if len(partitions[0]) > 1:
      588                  target_node = gene_tree.get_common_ancestor(partitions[0])
      589              else:
      590                  target_node = gene_tree.search_nodes(name=partitions[0][0])[0].up
      591              for small_partition in partitions[1:]:
      592                  for name in small_partition:
      593                      n = gene_tree.search_nodes(name=name)[0]
      594                      parent = n.up
      595                      n = n.detach()
      596                      target_node.add_child(n)
      597                      if len(parent.children) == 1 and parent.up is None:
      598                          parent.children[0].delete(prevent_nondicotomic=True, preserve_branch_length=True)
      599                      elif len(parent.children) == 1:
      600                          parent.delete(prevent_nondicotomic=True, preserve_branch_length=True)
      601                      # deactivate representative for next level
      602                      self._rep_lookup[n.name].representative.enabled = False
      603              return True
      604          else:
      605              # we remove the bogus representatives, but keep the labeling of the
      606              # speciation/duplication nodes. that way, the two subhogs will not get merged
      607              # at this level (since they are in different subtrees).
      608              stats = [{'len': len(p),
      609                        'rep_size': sum(len(self._rep_lookup[r].representative.get_subelements()) for r in p),
      610                       } for p in partitions]
      611              logger.info(f"Resolving conflict for {self._rep_lookup[partitions[0][0]].hog.hogid} by ignoring representatives {partitions[1:]}")
      612              logger.debug(f"{len(partitions)} partitions: {stats}")
      613              for small_partition in partitions[1:]:
      614                  for name in small_partition:
      615                      n = gene_tree.search_nodes(name=name)[0]
      616                      n.delete(prevent_nondicotomic=True, preserve_branch_length=True)
      617              return False

      619      def merge_subhogs(self, reconciled_genetree:TreeNode, msa:MultipleSeqAlignment):
      620          while True:
      621              hogids_in_subtrees = collections.defaultdict(list)
      622              for n in self._get_merge_candidates_with_hogids(reconciled_genetree):
      623                  for hogid in n.hogids:
      624                      hogids_in_subtrees[hogid].append(n)
      625              if any(len(z) > 1 for z in hogids_in_subtrees.values()):

      627                              + reconciled_genetree.get_ascii(show_internal=True, attributes=['evoltype', 'sos', 'hogids', 'hogid']))
      628              else:
      629                  break

      631              for hogid, subtrees in hogids_in_subtrees.items():
      632                  if len(subtrees) > 1:
      633                      logger.info(f"Representaives of {hogid} are split among {len(subtrees)} candidate subtrees.")
      634                      split_parts = [list(n.name for n in sub.iter_leaves() if n.hogid == hogid) for sub in subtrees]

      636                      if split_hogs and len(split_hogs) > 1:
      637                          # we could split the current hog.
      638                          self.subhogs.pop(hogid)
      639                          self.subhogs.update({h.hogid: h for h in split_hogs})
      640                      else:
      641                          redo_reconcilation |= self._resolve_conflicts(reconciled_genetree, subtrees, split_parts)
      642              self._rep_lookup = self._prepare_lookups()
      643              if redo_reconcilation:
      644                  if len(reconciled_genetree.children) == 1:
      645                      reconciled_genetree.children[0].delete(preserve_branch_length=True)
      646                  self.infer_reconciliation(reconciled_genetree, sos_threshold=self.conf.threshold_dubious_sd)
      647          new_hogs = []
      648          processed_nodes = set([])
      649          for subtrees in hogids_in_subtrees.values():
      650              for subtree in subtrees:
      651                  if subtree in processed_nodes:
      652                      continue
      653                  processed_nodes.add(subtree)
      654                  new_repr = self.find_most_divergent_representatives_from_genetree(subtree)
      655                  hog = HOG([self.subhogs[h] for h in subtree.hogids],
      656                            self.node_species_tree,
      657                            rhogid=self.rhogid,
      658                            msa=msa,
      659                            representatives=new_repr,
      660                            conf_infer_subhhogs=self.conf)
      661                  new_hogs.append(hog)
      662          return new_hogs

      664      def process(self) -> List[HOG]:
      665          """Process the given roothog at the given taxonomic level.
      666  
      667          This method does all the processessing of the subhogs from the
      668          previous taxonomic levels. It does
      669  
      670          1. Align the subhogs from the previous levels as a new MSA.
      671          2. Filter the initial MSA (remove gappy columns and rows)
      672          3. Infer a genetree for the new filtered MSA
      673          4. Root (mid-point) and reconcile (with species-overlap) the gene tree
      674          5. Merge the subhogs from the previous levels into new HOGs
      675          """
      676          msa = self.align_subhogs()
      677          filtered_msa = self.filter_msa(msa)
      678          if len(filtered_msa) == 0:
      679              return list(self.subhogs.values())
      680          gene_tree = self.infer_genetree_from_msa(filtered_msa)
      681          if len(gene_tree) > 1:
      682              rooted_gene_tree = self.infer_rooted_genetree(gene_tree)
      683              # TODO: dealing with fragments is not done yet.
      684              self.infer_reconciliation(rooted_gene_tree, sos_threshold=self.conf.threshold_dubious_sd)
      685          else:
      686              rooted_gene_tree = gene_tree
      687          new_hogs = self.merge_subhogs(rooted_gene_tree, msa=filtered_msa)
      688          new_hogs.extend(self._frozen_hogs_before)
      689          return new_hogs
