       97  
       98                  # Convert to list and extract relevant fields
       99                  results = []
      100                  for result in search_results:
      101                      results.append(
      102                          {

      104                              "url": result.get("href", ""),
      105                              "description": result.get("body", ""),

      107                      )

      109                  log_debug(f"Found {len(results)} search results")

      112              except Exception as e:
      113                  logger.warning(f"DuckDuckGo search attempt {attempt + 1} failed: {e}")
      114                  if "rate limit" in str(e).lower() or "429" in str(e):

      117                          self.rate_limit_delay * (2**attempt) if self.exponential_backoff else self.rate_limit_delay

      122                      # Other error - shorter wait
      123                      time.sleep(self.search_delay)

      213      def read(self, query: str) -> List[Document]:
      214          """Read content for a given query by performing web search and fetching content"""
      215          # Clear so URLs from previous queries aren't incorrectly skipped
      216          self._visited_urls.clear()
      217  
      218          if not query:
      219              raise ValueError("Query cannot be empty")
      220  
      221          log_debug(f"Starting web search reader for query: {query}")
      222  
      223          # Perform web search
      224          search_results = self._perform_web_search(query)
      225          if not search_results:
      226              logger.warning(f"No search results found for query: {query}")
      227              return []
      228  
      229          documents: List[Document] = []
      230  
      231          for result in search_results:
      232              url = result.get("url", "")
      233  
      234              # Skip if URL is invalid or already visited
      235              if not self._is_valid_url(url):
      236                  continue
      237  
      238              # Mark URL as visited
      239              self._visited_urls.add(url)
      240  
      241              # Add delay between requests to be respectful
      242              if len(documents) > 0:
      243                  time.sleep(self.delay_between_requests)
      244  
      245              # Fetch content from URL
      246              content = self._fetch_url_content(url)
      247              if content is None:
      248                  continue
      249  
      250              # Create document
      251              document = self._create_document_from_url(url, content, result)
      252  
      253              # Apply chunking if enabled
      254              if self.chunk:
      255                  chunked_docs = self.chunk_document(document)
      256                  documents.extend(chunked_docs)
      257              else:
      258                  documents.append(document)
      259  
      260              # Stop if we've reached max_results
      261              if len(documents) >= self.max_results:
      262                  break
      263  
      264          log_debug(f"Created {len(documents)} documents from web search")
      265          return documents
