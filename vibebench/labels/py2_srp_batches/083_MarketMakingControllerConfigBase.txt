      203      def get_spreads_and_amounts_in_quote(self, trade_type: TradeType) -> Tuple[List[float], List[float]]:
      204          buy_amounts_pct = getattr(self, 'buy_amounts_pct')
      205          sell_amounts_pct = getattr(self, 'sell_amounts_pct')
      206  
      207          # Calculate total percentages across buys and sells
      208          total_pct = sum(buy_amounts_pct) + sum(sell_amounts_pct)
      209  
      210          # Normalize amounts_pct based on total percentages
      211          if trade_type == TradeType.BUY:
      212              normalized_amounts_pct = [amt_pct / total_pct for amt_pct in buy_amounts_pct]
      213          else:  # TradeType.SELL
      214              normalized_amounts_pct = [amt_pct / total_pct for amt_pct in sell_amounts_pct]
      215  
      216          spreads = getattr(self, f'{trade_type.name.lower()}_spreads')
      217          return spreads, [amt_pct * self.total_amount_quote for amt_pct in normalized_amounts_pct]
