      117      def _grab_into(self, region: Region, dst: Frame) -> tuple[bool, int, int, int]:
      118          """Capture into ``dst`` and return ``(captured, frame_ticks, width, height)``.
      119  
      120          Contract:
      121          - ``captured`` is True only when ``dst`` has been fully written with a new frame.
      122          - ``captured`` is False with ``width == height == 0`` when no new frame is available.
      123          - ``captured`` is False with non-zero ``width``/``height`` when a new frame exists
      124            but ``dst`` shape does not match, so caller should rebuild and retry.
      125          """
      126          if not self._duplicator.update_frame():
      127              self._on_output_change()
      128              return False, 0, 0, 0
      129          if not self._duplicator.updated:
      130              return False, 0, 0, 0
      131  
      132          frame_width, frame_height = self._copy_region_to_stage(region)
      133          if frame_height != dst.shape[0] or frame_width != dst.shape[1]:
      134              return False, self._duplicator.latest_frame_ticks, frame_width, frame_height
      135          self._process_staging_frame_into(
      136              frame_width=frame_width,
      137              frame_height=frame_height,
      138              dst=dst,
      139          )
      140          return True, self._duplicator.latest_frame_ticks, frame_width, frame_height

      142      def _copy_region_to_stage(self, region: Region) -> tuple[int, int]:
      143          memory_region = self._region_to_memory_region(region)
      144          memory_width = memory_region[2] - memory_region[0]
      145          memory_height = memory_region[3] - memory_region[1]
      146          if (
      147              self._stagesurf.width != memory_width
      148              or self._stagesurf.height != memory_height

      150              self._stagesurf.release()
      151              self._stagesurf.rebuild(
