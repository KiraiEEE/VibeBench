      345          def decode_worker():
      346              while True:
      347                  try:
      348                      msg = self.server_socket.recv_multipart()
      349                      if msg and msg[0] == MoriKVManager.AUX_DATA_HEADER:
      350                          self._handle_aux_data(msg)
      351                          continue
      352  
      353                      if not msg or msg[0] != MORI_GUARD:
      354                          logger.warning(
      355                              "Received malformed status message on decode worker"
      356                          )
      357                          continue
      358                      payload = msg[1:]
      359                      if len(payload) < 3:
      360                          logger.warning("Incomplete status payload received")
      361                          continue
      362                      bootstrap_room = int(payload[0].decode("ascii"))
      363                      status_code = int(payload[1].decode("ascii"))
      364                      prefill_rank = int(payload[2].decode("ascii"))
      365                      failure_reason = (
      366                          payload[3].decode("utf-8")
      367                          if len(payload) > 3 and payload[3]
      368                          else None
      369                      )
      370  
      371                      if status_code == KVPoll.Success:
      372                          tracker = self.prefill_response_tracker[bootstrap_room]
      373                          tracker.add(prefill_rank)
      374                          expected = self.required_prefill_response_num_table.get(
      375                              bootstrap_room, 1
      376                          )
      377                          if len(tracker) >= expected:
      378                              self.prefill_response_tracker.pop(bootstrap_room, None)
      379                              self.update_status(bootstrap_room, KVPoll.Success)
      380                              self._cleanup_room_tracking(bootstrap_room)
      381                      elif status_code == KVPoll.Failed:
      382                          if failure_reason:
      383                              self.record_failure(bootstrap_room, failure_reason)
      384                          self.prefill_response_tracker.pop(bootstrap_room, None)
      385                          self.update_status(bootstrap_room, KVPoll.Failed)
      386                          self._cleanup_room_tracking(bootstrap_room)
      387                      else:
      388                          logger.warning(
      389                              "Unknown status code %s received for room %s",
      390                              status_code,
      391                              bootstrap_room,
      392                          )
      393                  except Exception:
      394                      logger.exception("Decode status worker failed")
      395  
      396          threading.Thread(target=decode_worker, daemon=True).start()
