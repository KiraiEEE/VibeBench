      322      async def _fetch_dates_chunk(
      323          self,
      324          imap: aioimaplib.IMAP4_SSL | aioimaplib.IMAP4,
      325          chunk: list[bytes],
      326          chunk_num: int,
      327          total_chunks: int,
      328      ) -> dict[str, datetime]:
      329          """Fetch INTERNALDATE for a single chunk of UIDs."""
      330          uid_list = ",".join(uid.decode() for uid in chunk)
      331          chunk_start = time.perf_counter()
      332          _, data = await imap.uid("fetch", uid_list, "(INTERNALDATE)")
      333          chunk_elapsed = time.perf_counter() - chunk_start
      334  
      335          chunk_dates: dict[str, datetime] = {}
      336          for item in data:
      337              if not isinstance(item, bytes) or b"INTERNALDATE" not in item:
      338                  continue
      339              uid_match = re.search(rb"UID (\d+)", item)
      340              date_match = re.search(rb'INTERNALDATE "([^"]+)"', item)
      341              if uid_match and date_match:
      342                  uid = uid_match.group(1).decode()
      343                  date_str = date_match.group(1).decode().strip()
      344                  chunk_dates[uid] = datetime.strptime(date_str, "%d-%b-%Y %H:%M:%S %z")
      345  
      346          if total_chunks > 1:
      347              logger.info(f"Fetched dates chunk {chunk_num}/{total_chunks}: {len(chunk)} UIDs in {chunk_elapsed:.2f}s")
      348  
      349          return chunk_dates

      378      async def _batch_fetch_headers(
      379          self,
      380          imap: aioimaplib.IMAP4_SSL | aioimaplib.IMAP4,
      381          email_ids: list[bytes] | list[str],
      382      ) -> dict[str, dict[str, Any]]:
      383          """Batch fetch headers for a list of UIDs."""
      384          if not email_ids:
      385              return {}
      386  
      387          # Normalize to list of strings
      388          str_ids = [uid.decode() if isinstance(uid, bytes) else uid for uid in email_ids]
      389          uid_list = ",".join(str_ids)
      390          _, data = await imap.uid("fetch", uid_list, "BODY.PEEK[HEADER]")
      391  
      392          results: dict[str, dict[str, Any]] = {}
      393          for i, item in enumerate(data):
      394              if not isinstance(item, bytes) or b"BODY[HEADER]" not in item:
      395                  continue
      396              # First try to find UID in the same line (standard format)
      397              uid_match = re.search(rb"UID (\d+)", item)
      398              if uid_match and i + 1 < len(data) and isinstance(data[i + 1], bytearray):
      399                  uid = uid_match.group(1).decode()
      400                  raw_headers = bytes(data[i + 1])
      401                  metadata = self._parse_headers(uid, raw_headers)
      402                  if metadata:
      403                      results[uid] = metadata
      404              # Proton Bridge format: UID comes AFTER header data in a separate item
      405              # Format: [i]=b'N FETCH (BODY[HEADER] {size}', [i+1]=bytearray(headers), [i+2]=b' UID xxx)'
      406              elif i + 2 < len(data) and isinstance(data[i + 1], bytearray):
      407                  uid_after_match = re.search(rb"UID (\d+)", data[i + 2]) if isinstance(data[i + 2], bytes) else None
      408                  if uid_after_match:
      409                      uid = uid_after_match.group(1).decode()
      410                      raw_headers = bytes(data[i + 1])
      411                      metadata = self._parse_headers(uid, raw_headers)
      412                      if metadata:
      413                          results[uid] = metadata
      414  
      415          return results
