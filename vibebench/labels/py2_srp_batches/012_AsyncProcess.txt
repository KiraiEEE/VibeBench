      205      async def read_stderr(self) -> bytes:
      206          """Read line from stderr."""
      207          if self.returncode is not None:
      208              return b""
      209          assert self.proc is not None  # for type checking
      210          assert self.proc.stderr is not None  # for type checking
      211          async with self._stderr_lock:
      212              try:
      213                  return await self.proc.stderr.readline()
      214              except ValueError as err:
      215                  # we're waiting for a line (separator found), but the line was too big
      216                  # this may happen with ffmpeg during a long (radio) stream where progress
      217                  # gets outputted to the stderr but no newline
      218                  # https://stackoverflow.com/questions/55457370/how-to-avoid-valueerror-separator-is-not-found-and-chunk-exceed-the-limit
      219                  # NOTE: this consumes the line that was too big
      220                  if "chunk exceed the limit" in str(err):
      221                      return await self.proc.stderr.readline()
      222                  # raise for all other (value) errors

      237      async def communicate(
      238          self,
      239          input: bytes | None = None,  # noqa: A002
      240          timeout: float | None = None,
      241      ) -> tuple[bytes, bytes]:
      242          """Communicate with the process and return stdout and stderr."""
      243          if self.closed:
      244              raise RuntimeError("communicate called while process already done")
      245          # abort existing readers on stderr/stdout first before we send communicate
      246          await self._stderr_lock.acquire()
      247          await self._stdout_lock.acquire()
      248          assert self.proc is not None  # for type checking
      249          stdout, stderr = await asyncio.wait_for(self.proc.communicate(input), timeout)
      250          return (stdout, stderr)

      252      async def close(self) -> None:
      253          """Close/terminate the process and wait for exit."""
      254          self._close_called = True
      255          if not self.proc:
      256              return
      257  
      258          # cancel existing stdin feeder task if any
      259          if self._stdin_feeder_task:
      260              if not self._stdin_feeder_task.done():
      261                  self._stdin_feeder_task.cancel()
      262              # Always await the task to consume any exception and prevent
      263              # "Task exception was never retrieved" errors.
      264              try:
      265                  await self._stdin_feeder_task
      266              except asyncio.CancelledError:
      267                  pass  # Expected when we cancel the task
      268              except Exception as err:
      269                  # Log unexpected exceptions from the stdin feeder before suppressing
      270                  LOGGER.warning(
      271                      "Process stdin feeder task ended with error: %s",
      272                      err,
      273                  )
      274  
      275          # close stdin to signal we're done sending data
      276          with suppress(TimeoutError, asyncio.CancelledError):
      277              await asyncio.wait_for(self._stdin_lock.acquire(), 5)
      278          if self.proc.stdin and not self.proc.stdin.is_closing():
      279              self.proc.stdin.close()
      280          elif not self.proc.stdin and self.proc.returncode is None:
      281              self.proc.send_signal(SIGINT)
      282  
      283          # ensure we have no more readers active and stdout is drained
      284          with suppress(TimeoutError, asyncio.CancelledError):
      285              await asyncio.wait_for(self._stdout_lock.acquire(), 5)
      286          if self.proc.stdout and not self.proc.stdout.at_eof():
      287              with suppress(Exception):
      288                  await self.proc.stdout.read(-1)
      289          # if we have a stderr task active, allow it to finish
      290          if self._stderr_reader_task:
      291              with suppress(TimeoutError, asyncio.CancelledError):
      292                  await asyncio.wait_for(self._stderr_reader_task, 5)
      293          elif self.proc.stderr and not self.proc.stderr.at_eof():
      294              with suppress(TimeoutError, asyncio.CancelledError):
      295                  await asyncio.wait_for(self._stderr_lock.acquire(), 5)
      296              # drain stderr
      297              with suppress(Exception):
      298                  await self.proc.stderr.read(-1)
      299  
      300          # make sure the process is really cleaned up.
      301          # especially with pipes this can cause deadlocks if not properly guarded
      302          # we need to ensure stdout and stderr are flushed and stdin closed
      303          pid = self.proc.pid
      304          terminate_attempts = 0
      305          while self.returncode is None:
      306              try:
      307                  # use communicate to flush all pipe buffers
      308                  await asyncio.wait_for(self.proc.communicate(), 2)
      309              except TimeoutError:
      310                  terminate_attempts += 1
      311                  self.logger.debug(
      312                      "Process %s with PID %s did not stop in time (attempt %d). Sending SIGKILL...",
      313                      self.name,
      314                      pid,
      315                      terminate_attempts,
      316                  )
      317                  # Use os.kill for more direct signal delivery
      318                  with suppress(ProcessLookupError, OSError):
      319                      os.kill(pid, 9)  # SIGKILL = 9
      320                  # Give up after 5 attempts - process may be zombie
      321                  if terminate_attempts >= 5:
      322                      self.logger.warning(
      323                          "Process %s (PID %s) did not terminate after %d SIGKILL attempts",
      324                          self.name,
      325                          pid,
      326                          terminate_attempts,
      327                      )
      328                      break
      329          self.logger.log(
      330              VERBOSE_LOG_LEVEL,
      331              "Process %s with PID %s stopped with returncode %s",
      332              self.name,
      333              self.proc.pid,
      334              self.returncode,
      335          )

      337      async def kill(self) -> None:
      338          """
      339          Immediately kill the process with SIGKILL.
      340  
      341          Use this for forceful termination when the process doesn't respond to
      342          normal termination signals. Unlike close(), this doesn't attempt graceful
      343          shutdown - it immediately sends SIGKILL.
      344          """
      345          self._close_called = True
      346          if not self.proc or self.returncode is not None:
      347              return
      348  
      349          pid = self.proc.pid
      350  
      351          # Cancel stdin feeder task if any
      352          if self._stdin_feeder_task and not self._stdin_feeder_task.done():
      353              self._stdin_feeder_task.cancel()
      354              with suppress(asyncio.CancelledError, Exception):
      355                  await self._stdin_feeder_task
      356  
      357          # Cancel stderr reader task if any
      358          if self._stderr_reader_task and not self._stderr_reader_task.done():
      359              self._stderr_reader_task.cancel()
      360              with suppress(asyncio.CancelledError, Exception):
      361                  await self._stderr_reader_task
      362  
      363          # Close stdin to signal we're done sending data
      364          # Note: Don't manually call feed_eof() on stdout/stderr - this causes
      365          # "feed_data after feed_eof" assertion errors when the subprocess transport
      366          # still has buffered data to deliver. Let the process termination naturally
      367          # close the streams.
      368          if self.proc.stdin and not self.proc.stdin.is_closing():
      369              self.proc.stdin.close()
      370  
      371          # Send SIGKILL immediately using os.kill for more direct signal delivery
      372          self.logger.debug("Killing process %s with PID %s", self.name, pid)
      373          with suppress(ProcessLookupError, OSError):
      374              os.kill(pid, 9)  # SIGKILL = 9
      375  
      376          # Wait for process to actually terminate
      377          try:
      378              await asyncio.wait_for(self.proc.wait(), 2)
      379          except TimeoutError:
      380              # Try one more time with os.kill
      381              with suppress(ProcessLookupError, OSError):
      382                  os.kill(pid, 9)
      383              try:
      384                  await asyncio.wait_for(self.proc.wait(), 2)
      385              except TimeoutError:
      386                  self.logger.warning(
      387                      "Process %s with PID %s did not terminate after SIGKILL - may be zombie",
      388                      self.name,
      389                      pid,
      390                  )
      391  
      392          self.logger.log(
      393              VERBOSE_LOG_LEVEL,
      394              "Process %s with PID %s killed with returncode %s",
      395              self.name,
      396              pid,
      397              self.returncode,
      398          )
