       61      def _extract_service_name(self, prefix: str) -> str:
       62          """
       63          Extract service name from the URL prefix.
       64  
       65          Example:
       66              "/google-ads" -> "google-ads"
       67              "/youtube-transcripts" -> "youtube-transcripts"
       68              "/api/v1/google-ads" -> "google-ads"
       69  
       70          Args:
       71              prefix: URL prefix string
       72  
       73          Returns:
       74              Extracted service name
       75          """
       76          # Remove leading slash if present
       77          if prefix.startswith("/"):
       78              prefix = prefix[1:]
       79  
       80          # Split on slashes
       81          parts = prefix.split("/")
       82          if not parts:
       83              raise ValueError(f"Cannot extract service name from prefix: {prefix}")
       84  
       85          # If the prefix is like "/api/v1/google-ads", take the last segment
       86          if len(parts) > 1 and parts[0] == "api" and parts[1].startswith("v"):
       87              service_name = parts[2] if len(parts) > 2 else parts[0]
       88          else:
       89              service_name = parts[0]
       90  
       91          # Validate that we have a non-empty service name
       92          if not service_name:
       93              raise ValueError(f"Cannot extract service name from prefix: {prefix}")
       94  
       95          return service_name

      180          self, status: int, title: str, detail: str, type: str, instance: Optional[str] = None, **kwargs
      181      ) -> dict:
      182          """
      183          Create an RFC7807 compliant error response.
      184  
      185          Args:
      186              status: HTTP status code
      187              title: Human-readable title of the error
      188              detail: Specific details about the error
      189              type: Error type identifier
      190              instance: URI of the specific occurrence of the error
      191              **kwargs: Additional fields to include in the error response
      192  
      193          Returns:
      194              Dictionary with RFC7807 compliant error details
      195          """
      196          # Ensure type is a proper URI
      197          if not type.startswith(("http://", "https://")):
      198              type = f"https://socialflood.com/problems/{type}"
      199  
      200          error = {"type": type, "title": title, "status": status, "detail": detail}
      201  
      202          if instance:
      203              error["instance"] = instance
      204  
      205          # Add any additional fields
      206          error.update(kwargs)

      210      def raise_http_exception(
      211          self,
      212          status_code: int,
      213          detail: str,
      214          title: Optional[str] = None,
      215          type: Optional[str] = None,
      216          instance: Optional[str] = None,
      217          headers: Optional[Dict[str, str]] = None,
      218          **kwargs,
      219      ) -> None:
      220          """
      221          Raise an HTTPException with RFC7807 compliant error details.
      222  
      223          Args:
      224              status_code: HTTP status code
      225              detail: Specific details about the error
      226              title: Human-readable title of the error (defaults to status code description)
      227              type: Error type identifier (defaults to generic type based on status code)
      228              instance: URI of the specific occurrence of the error
      229              headers: Optional HTTP headers to include in the response
      230              **kwargs: Additional fields to include in the error response
      231  
      232          Raises:
      233              HTTPException: With RFC7807 compliant error details
      234          """
      235          # Default title based on status code
      236          if not title:
      237              titles = {
      238                  400: "Bad Request",
      239                  401: "Unauthorized",
      240                  403: "Forbidden",
      241                  404: "Not Found",
      242                  422: "Unprocessable Entity",
      243                  500: "Internal Server Error",
      244              }
      245              title = titles.get(status_code, "Error")
      246  
      247          # Default type based on status code
      248          if not type:
      249              types = {
      250                  400: "validation_error",
      251                  401: "authentication_error",
      252                  403: "authorization_error",
      253                  404: "not_found",
      254                  422: "validation_error",
      255                  500: "server_error",
      256              }
      257              type = types.get(status_code, "error")
      258  
      259          # Create RFC7807 error detail
      260          error_detail = self._create_error_detail(
      261              status=status_code, title=title, detail=detail, type=type, instance=instance, **kwargs
      262          )
      263  
      264          # Set Content-Type header for RFC7807
      265          if not headers:
      266              headers = {}
      267          headers["Content-Type"] = "application/problem+json"
      268  
      269          raise HTTPException(status_code=status_code, detail=error_detail, headers=headers)
