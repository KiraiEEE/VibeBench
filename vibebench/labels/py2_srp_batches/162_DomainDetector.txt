      982      def _detect_mcq_format(self, query: str) -> tuple[bool, str, Optional[Domain]]:
      983          """
      984          Detect if query is a multiple-choice question and extract content.
      985  
      986          Args:
      987              query: Original query text
      988  
      989          Returns:
      990              Tuple of (is_mcq, extracted_content, domain_hint)
      991          """
      992          query_lower = query.lower()
      993  
      994          # Check MCQ patterns
      995          is_mcq = False
      996          for pattern in self.MCQ_PATTERNS:
      997              if re.search(pattern, query_lower, re.IGNORECASE | re.MULTILINE):
      998                  is_mcq = True
      999                  break
     1000  
     1001          if not is_mcq:
     1002              return False, query, None
     1003  
     1004          # Extract the actual question content (strip MCQ wrapper)
     1005          extracted = query
     1006  
     1007          # Remove common MCQ instruction prefixes
     1008          prefixes_to_remove = [
     1009              r"^answer the following multiple[- ]?choice question[.:]?\s*",
     1010              r"^provide your answer as a single letter[^.]*[.]\s*",
     1011              r"^choose the (?:best|correct) answer[.:]?\s*",
     1012              r"^select (?:one|the correct answer)[.:]?\s*",
     1013          ]
     1014  
     1015          for prefix in prefixes_to_remove:
     1016              extracted = re.sub(prefix, "", extracted, flags=re.IGNORECASE)
     1017  
     1018          # Extract content after "Question:" if present
     1019          question_match = re.search(
     1020              r"question[:\s]+(.+?)(?=\n[ABCD][\.\)]|\Z)", extracted, re.IGNORECASE | re.DOTALL
     1021          )
     1022          if question_match:
     1023              extracted = question_match.group(1).strip()
     1024  
     1025          # Remove answer choices (A. B. C. D. lines)
     1026          extracted = re.sub(r"\n[ABCD][\.\)]\s+[^\n]+", "", extracted)
     1027          # Remove trailing "Answer:" prompt
     1028          extracted = re.sub(r"\s*answer:\s*$", "", extracted, flags=re.IGNORECASE)
     1029  
     1030          # Try to detect domain from subject keywords in the question
     1031          domain_hint = self._detect_subject_domain(query_lower)
     1032  
     1033          return True, extracted.strip(), domain_hint
