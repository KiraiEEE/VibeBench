       85  class MLLMModelWrapper:
       86      """
       87      Wrapper for MLLM models to make them compatible with BatchGenerator.
       88  
       89      BatchGenerator expects model output to be subscriptable (logits array),
       90      but MLLM models return LanguageModelOutput objects. This wrapper extracts
       91      the logits from the output.
       92  
       93      Also handles Gemma 3's required pixel_values argument by injecting None
       94      for text-only requests.
       95      """
       96  
       97      def __init__(self, model):
       98          self._model = model
       99          # Detect if this is a Gemma 3 model (requires pixel_values as positional arg)
      100          self._is_gemma3 = (
      101              hasattr(model, "model_type")
      102              and "gemma3" in str(getattr(model, "model_type", "")).lower()
      103          )
      104  
      105      def __call__(self, *args, **kwargs):
      106          """Call the model and extract logits from LanguageModelOutput."""
      107          # Gemma 3 requires pixel_values as a positional argument, unlike Qwen
      108          # which makes it optional. Inject pixel_values=None for text-only requests.
      109          if self._is_gemma3 and "pixel_values" not in kwargs:
      110              kwargs["pixel_values"] = None
      111  
      112          output = self._model(*args, **kwargs)
      113          # If output has logits attribute, return just the logits
      114          if hasattr(output, "logits"):
      115              return output.logits
      116          return output
      117  
      118      def __getattr__(self, name):
      119          """Forward all other attributes to the wrapped model."""
      120          return getattr(self._model, name)
