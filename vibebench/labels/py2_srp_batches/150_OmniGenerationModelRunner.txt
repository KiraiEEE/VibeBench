      199          # sampled_token_ids (triggering scheduler's check_stop).
      200          # OmniModelConfig may not expose eos_token_id directly;

      203          if eos_id is None:

      206                  "eos_token_id",

      213          # kernel.  Without this the scheduler re-schedules the same
      214          # tokens every step (infinite loop).
      215          dummy_sampled = torch.full(
      216              (num_reqs, 1), eos_id, dtype=torch.long, device=self.device
      217          )

      220          )

      223          )
