      787      def _plan_certain(  # pylint: disable=too-many-positional-arguments
      788          self,
      789          model_name: str,
      790          region: str,
      791          desires: CapacityDesires,
      792          num_results: Optional[int] = None,
      793          num_regions: int = 3,
      794          lifecycles: Optional[Sequence[Lifecycle]] = None,
      795          instance_families: Optional[Sequence[str]] = None,
      796          drives: Optional[Sequence[str]] = None,
      797          extra_model_arguments: Optional[Dict[str, Any]] = None,
      798          planner_arguments: Optional[PlannerArguments] = None,
      799      ) -> _CertainResult:
      800          extra_model_arguments = extra_model_arguments or {}
      801          pargs = planner_arguments or PlannerArguments()
      802          model = self._models[model_name]
      803  
      804          plans: List[CapacityPlan] = []
      805          excuses: List[Excuse] = []
      806          pref_families = model.preferred_families()
      807          current_inst = current_instance_name(desires)
      808          for instance, drive, context in self.generate_scenarios(
      809              model, region, desires, num_regions, lifecycles, instance_families, drives
      810          ):
      811              match model.capacity_plan(
      812                  instance=instance,
      813                  drive=drive,
      814                  context=context,
      815                  desires=desires,
      816                  extra_model_arguments=extra_model_arguments,
      817              ):
      818                  case CapacityPlan() as plan:
      819                      if (
      820                          pref_families is not None
      821                          and instance.family not in pref_families
      822                          and pargs.preferred_family_penalty > 0
      823                      ):
      824                          service_cost = sum(
      825                              s.annual_cost for s in plan.candidate_clusters.services
      826                          )
      827                          compute_cost = (
      828                              plan.candidate_clusters.total_annual_cost - service_cost
      829                          )
      830                          plan = plan.model_copy(
      831                              update={
      832                                  "rank": plan.rank
      833                                  + compute_cost * pargs.preferred_family_penalty
      834                              }
      835                          )
      836                      plans.append(plan)
      837                  case Excuse() as excuse:
      838                      tags = compute_excuse_tags(excuse.instance, current_inst)
      839                      excuses.append(
      840                          excuse.model_copy(update={"tags": tags}) if tags else excuse
      841                      )
      842                  case None:
      843                      pass
      844  
      845          # lowest cost first
      846          plans.sort(key=lambda p: (p.rank, p.candidate_clusters.total_annual_cost))
      847  
      848          num_results = num_results or self._default_num_results
      849          return _CertainResult(
      850              plans=reduce_by_family(
      851                  plans, max_results_per_family=pargs.max_results_per_family
      852              )[:num_results],
      853              excuses=excuses,
      854          )
