      157      async def _propagate_to_group_members(self, command: str, **kwargs: Any) -> None:
      158          """Propagate command to group members when we are the leader."""
      159          # Skip if grouping is disabled at provider level
      160          provider = cast("MSXBridgeProvider", self.provider)
      161          if not provider.grouping_enabled:
      162              return
      163          # Prevent infinite recursion if member.play_media triggers propagation back
      164          if self._propagating:
      165              return
      166          self._propagating = True
      167          try:
      168              for member_id in self._get_group_member_ids():
      169                  member = self.mass.players.get(member_id)
      170                  if not member or not isinstance(member, MSXPlayer) or not member.available:
      171                      continue
      172                  try:
      173                      if command == "play_media":
      174                          media = kwargs.get("media")
      175                          if media:
      176                              # Call member.play_media directly — mass.players.play_media
      177                              # would redirect synced/grouped players back to the leader
      178                              await member.play_media(media)
      179                      elif command == "stop":
      180                          await member.stop()
      181                      elif command == "pause":
      182                          await member.pause()
      183                      elif command == "play":
      184                          await member.play()
      185                  except Exception:
      186                      self.logger.warning(
      187                          "Failed to propagate %s to member %s",
      188                          command,
      189                          member_id,
      190                          exc_info=True,
      191                      )
      192          finally:
      193              self._propagating = False
