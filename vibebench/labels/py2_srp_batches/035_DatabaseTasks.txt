      307      def insert_tasks(self, tasks_data: list[dict[str, Any]]) -> list[dict[str, Any]]:
      308          """
      309          Inserts multiple tasks into the Todoist API in parallel using threading.
      310          
      311          This method provides thread-safe parallel task insertion with retry logic,
      312          similar to how fetch_projects works in db_projects.py.
      313  
      314          Parameters:
      315          - tasks_data (list[dict]): List of dictionaries, where each dictionary contains
      316            the parameters for insert_task (content, description, project_id, etc.)
      317  
      318          Returns:
      319          - list[dict]: List of responses from the Todoist API in the same order as input.
      320            Failed insertions will have an empty dict.
      321            
      322          Example:
      323              tasks_data = [
      324                  {"content": "Buy milk", "project_id": "123", "priority": 2},
      325                  {"content": "Call dentist", "due_string": "tomorrow"},
      326              ]
      327              results = db.insert_tasks(tasks_data)
      328          """
      329          if not tasks_data:
      330              logger.info("No tasks to insert")
      331              return []

      333          def insert_single_task(task_data: dict[str, Any], index: int) -> dict[str, Any]:
      334              """Insert a single task with its data."""
      335              try:
      336                  return self.insert_task(**task_data)
      337              except (RuntimeError, ValueError, TypeError, KeyError) as e:
      338                  logger.error(f"Failed to insert task at index {index}: {e.__class__.__name__}: {e}")
      339                  return {}
      340  
      341          def insert_single_task_with_retry(task_data: dict[str, Any], index: int) -> dict[str, Any]:
      342              """Insert a single task with built-in retry logic."""
      343              return with_retry(
      344                  partial(insert_single_task, task_data, index),
      345                  operation_name=(
      346                      f"insert task {index} (content: {task_data.get(TaskField.CONTENT.value, 'N/A')})"
      347                  ),
      348                  max_attempts=RETRY_MAX_ATTEMPTS
      349              )
      350  
      351          logger.info(f"Inserting {len(tasks_data)} tasks with thread pool")
      352          max_workers = min(8, len(tasks_data))
      353          ordered_results: list[dict[str, Any] | None] = [None] * len(tasks_data)
      354          with ThreadPoolExecutor(max_workers=max_workers) as executor:
      355              future_to_index = {
      356                  executor.submit(insert_single_task_with_retry, task_data, idx): idx
      357                  for idx, task_data in enumerate(tasks_data)
      358              }
      359              for future in tqdm(
      360                  as_completed(future_to_index),
      361                  total=len(tasks_data),
      362                  desc='Inserting tasks',
      363                  unit='task',
      364                  position=0,
      365                  leave=True
      366              ):
      367                  idx = future_to_index[future]
      368                  try:
      369                      result = future.result(timeout=60)
      370                  except (MaxRetriesExceeded, RuntimeError, ValueError, TypeError, OSError) as e:  # pragma: no cover - defensive
      371                      logger.error(f"Failed inserting task at index {idx}: {e.__class__.__name__}: {e}")
      372                      result = {}
      373                  ordered_results[idx] = result
      374                  logger.debug(f"Inserted task {idx+1}/{len(tasks_data)}")
      375  
      376          # Replace any remaining None with empty dicts (should be rare)
      377          for i in range(len(ordered_results)):
      378              if ordered_results[i] is None:
      379                  ordered_results[i] = {}
      380  
      381          return [result or {} for result in ordered_results]
