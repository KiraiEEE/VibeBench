      196      async def _handle_ai_response(
      197          self,
      198          user_query: str,
      199          streaming_interface: "OpenAIChatCompletionsStreamingInterface",
      200          agent_state: AgentState,
      201          in_memory_message_history: List[Dict[str, Any]],
      202          letta_message_db_queue: List[Any],
      203      ) -> bool:
      204          """
      205          Now that streaming is done, handle the final AI response.
      206          This might yield additional SSE tokens if we do stalling.
      207          At the end, set self._continue_execution accordingly.
      208          """
      209          # 1. If we have any leftover content from partial stream, store it as an assistant message
      210          if streaming_interface.content_buffer:
      211              content = "".join(streaming_interface.content_buffer)
      212              in_memory_message_history.append({"role": "assistant", "content": content})
      213  
      214              assistant_msgs = create_assistant_messages_from_openai_response(
      215                  response_text=content,
      216                  agent_id=agent_state.id,
      217                  model=agent_state.llm_config.model,
      218                  timezone=agent_state.timezone,
      219              )
      220              letta_message_db_queue.extend(assistant_msgs)
      221  
      222          # 2. If a tool call was requested, handle it
      223          if streaming_interface.tool_call_happened:
      224              tool_call_name = streaming_interface.tool_call_name
      225              tool_call_args_str = streaming_interface.tool_call_args_str or "{}"
      226              try:
      227                  tool_args = json.loads(tool_call_args_str)
      228              except json.JSONDecodeError:
      229                  tool_args = {}
      230  
      231              tool_call_id = streaming_interface.tool_call_id or f"call_{uuid.uuid4().hex[:8]}"
      232              assistant_tool_call_msg = AssistantMessage(
      233                  content=None,
      234                  tool_calls=[
      235                      ToolCall(
      236                          id=tool_call_id,
      237                          function=ToolCallFunction(
      238                              name=tool_call_name,
      239                              arguments=tool_call_args_str,
      240                          ),
      241                      )
      242                  ],
      243              )
      244              in_memory_message_history.append(assistant_tool_call_msg.model_dump())
      245  
      246              tool_execution_result = await self._execute_tool(
      247                  user_query=user_query,
      248                  tool_name=tool_call_name,
      249                  tool_args=tool_args,
      250                  agent_state=agent_state,
      251              )
      252              tool_result = tool_execution_result.func_return
      253              success_flag = tool_execution_result.success_flag
      254  
      255              # 3. Provide function_call response back into the conversation
      256              # TODO: fix this tool format
      257              tool_message = ToolMessage(
      258                  content=json.dumps({"result": tool_result}),
      259                  tool_call_id=tool_call_id,
      260              )
      261              in_memory_message_history.append(tool_message.model_dump())
      262  
      263              # 4. Insert heartbeat message for follow-up
      264              heartbeat_user_message = UserMessage(
      265                  content=f"{NON_USER_MSG_PREFIX} Tool finished executing. Summarize the result for the user."
      266              )
      267              in_memory_message_history.append(heartbeat_user_message.model_dump())
      268  
      269              # 5. Also store in DB
      270              tool_call_messages = create_letta_messages_from_llm_response(
      271                  agent_id=agent_state.id,
      272                  model=agent_state.llm_config.model,
      273                  function_name=tool_call_name,
      274                  function_arguments=tool_args,
      275                  tool_call_id=tool_call_id,
      276                  function_response=tool_result,
      277                  tool_execution_result=tool_execution_result,
      278                  timezone=agent_state.timezone,
      279                  continue_stepping=True,
      280              )
      281              letta_message_db_queue.extend(tool_call_messages)
      282  
      283              # Because we have new data, we want to continue the while-loop in `step_stream`
      284              return True
      285          else:
      286              # If we got here, there's no tool call. If finish_reason_stop => done
      287              return not streaming_interface.finish_reason_stop
