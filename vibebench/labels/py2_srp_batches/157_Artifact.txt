      793      @classmethod
      794      def from_api_response(cls, data: list[Any]) -> "Artifact":
      795          """Parse artifact from API response.
      796  
      797          Structure: [id, title, type, ..., status, ..., metadata, ...]
      798          Position 9 contains options with variant code at [9][1][0]:
      799            - For type 4: 1=flashcards, 2=quiz
      800          """
      801          artifact_id = data[0] if len(data) > 0 else ""
      802          title = data[1] if len(data) > 1 else ""
      803          artifact_type = data[2] if len(data) > 2 else 0
      804          status = data[4] if len(data) > 4 else 0
      805  
      806          # Extract timestamp from data[15][0]
      807          created_at = None
      808          if len(data) > 15 and isinstance(data[15], list) and len(data[15]) > 0:
      809              try:
      810                  created_at = datetime.fromtimestamp(data[15][0])
      811              except (TypeError, ValueError):
      812                  pass
      813  
      814          # Extract variant code from data[9][1][0] for quiz/flashcard distinction
      815          variant = None
      816          if len(data) > 9 and isinstance(data[9], list) and len(data[9]) > 1:
      817              options = data[9][1]
      818              if isinstance(options, list) and len(options) > 0:
      819                  variant = options[0]
      820  
      821          return cls(
      822              id=str(artifact_id),
      823              title=str(title),
      824              _artifact_type=artifact_type,
      825              status=status,
      826              created_at=created_at,
      827              _variant=variant,
      828          )

      830      @classmethod
      831      def from_mind_map(cls, data: list[Any]) -> Optional["Artifact"]:
      832          """Parse artifact from mind map data (stored in notes system).
      833  
      834          Mind map structure:
      835          [
      836              "mind_map_id",
      837              [
      838                  "mind_map_id",           # [1][0]: ID
      839                  "JSON_content",          # [1][1]: Mind map JSON
      840                  [1, "user_id", [ts, ns]],  # [1][2]: Metadata
      841                  None,                    # [1][3]
      842                  "title"                  # [1][4]: Title
      843              ]
      844          ]
      845  
      846          Deleted/cleared mind map: ["id", None, 2]
      847  
      848          Returns:
      849              Artifact object, or None if deleted (status=2).
      850          """
      851          if not isinstance(data, list) or len(data) < 1:
      852              return None
      853  
      854          mind_map_id = data[0] if len(data) > 0 else ""
      855  
      856          # Check for deleted status (item[1] is None with status=2)
      857          if len(data) >= 3 and data[1] is None and data[2] == 2:
      858              return None  # Deleted, don't include
      859  
      860          # Extract title and timestamp from nested structure
      861          title = ""
      862          created_at = None
      863  
      864          if len(data) > 1 and isinstance(data[1], list):
      865              inner = data[1]
      866              # Title is at position [4]
      867              if len(inner) > 4 and isinstance(inner[4], str):
      868                  title = inner[4]
      869              # Timestamp is at [2][2][0]
      870              if len(inner) > 2 and isinstance(inner[2], list) and len(inner[2]) > 2:
      871                  ts_data = inner[2][2]
      872                  if isinstance(ts_data, list) and len(ts_data) > 0:
      873                      try:
      874                          created_at = datetime.fromtimestamp(ts_data[0])
      875                      except (TypeError, ValueError):
      876                          pass
      877  
      878          return cls(
      879              id=str(mind_map_id),
      880              title=title,
      881              _artifact_type=5,  # ArtifactTypeCode.MIND_MAP
      882              status=3,  # Mind maps are always "completed" once created
      883              created_at=created_at,
      884              _variant=None,
      885          )
