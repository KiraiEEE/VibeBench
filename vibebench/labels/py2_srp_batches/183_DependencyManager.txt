      321                  if name:

      327                  for dep in group_deps:

      355      def _parse_dependency_spec(self, spec: str) -> Tuple[Optional[str], str]:
      356          """Parse dependency specification."""
      357          # Remove comments
      358          spec = spec.split("#")[0].strip()
      359          if not spec:
      360              return None, ""
      361  
      362          # Remove extras specification
      363          clean_spec = re.sub(r"\[.*?\]", "", spec)
      364  
      365          # Extract package name (first part before version specifier)
      366          name_match = re.match(r"^([a-zA-Z0-9\-_.]+)", clean_spec)
      367          if not name_match:
      368              return None, ""
      369  
      370          name = name_match.group(1).lower().replace("_", "-")
      371          return name, spec
