       86      def evaluate(self, multi_rag_result: MultiRAGResult) -> MultiScoredRAGResult:
       87          """Evaluate RAG results against the golden answer.
       88  
       89          Args:
       90              multi_rag_result: Container with RAG results and expected_answer
       91  
       92          Returns:
       93              MultiScoredRAGResult with golden answer comparison scores
       94          """
       95          all_scored_results = []
       96          expected_answer = multi_rag_result.expected_answer
       97  
       98          # Skip if no golden answer provided
       99          if not expected_answer:
      100              logger.warning(
      101                  "No expected_answer for query %s. Skipping golden answer evaluation.",
      102                  multi_rag_result.query_id
      103              )
      104              return MultiScoredRAGResult(
      105                  query_id=multi_rag_result.query_id,
      106                  query=multi_rag_result.query,
      107                  scored_rag_results=[]
      108              )
      109  
      110          for i, rag_result in enumerate(multi_rag_result.rag_results):
      111              # Break after first if not running consistency
      112              if i > 0 and not self.run_consistency:
      113                  break
      114  
      115              try:
      116                  # Extract generated answer text
      117                  generated_answer = " ".join([
      118                      part.text for part in rag_result.generation_result.generated_answer
      119                  ])
      120                  query = rag_result.generation_result.query
      121  
      122                  # Compute all metrics
      123                  similarity_scores = self.semantic_similarity_metric.compute(
      124                      query, generated_answer, expected_answer
      125                  )
      126                  correctness_scores = self.factual_correctness_metric.compute(
      127                      query, generated_answer, expected_answer
      128                  )
      129  
      130                  # Extract and aggregate token usage
      131                  factual_tokens = correctness_scores.pop("token_usage", {})
      132                  total_input_tokens = factual_tokens.get("input_tokens", 0)
      133                  total_output_tokens = factual_tokens.get("output_tokens", 0)
      134  
      135                  aggregated_token_usage = {
      136                      "factual_correctness_tokens": factual_tokens,
      137                      "total_input_tokens": total_input_tokens,
      138                      "total_output_tokens": total_output_tokens,
      139                      "total_tokens": total_input_tokens + total_output_tokens
      140                  }
      141  
      142                  # Combine into scores structure
      143                  generation_scores = {
      144                      **similarity_scores,
      145                      **correctness_scores,
      146                      "expected_answer": expected_answer,
      147                      "token_usage": aggregated_token_usage
      148                  }
      149  
      150                  rag_scores = RAGScores(
      151                      RetrievalScores(scores={}),  # Not evaluating retrieval
      152                      AugmentedGenerationScores(scores=generation_scores)
      153                  )
      154  
      155                  scored_result = ScoredRAGResult(
      156                      rag_result=rag_result,
      157                      scores=rag_scores
      158                  )
      159                  all_scored_results.append(scored_result)
      160  
      161              except Exception as e:
      162                  logger.exception("Error evaluating golden answer result: %s", e)
      163                  rag_scores = RAGScores(
      164                      RetrievalScores(scores={}),
      165                      AugmentedGenerationScores(scores={})
      166                  )
      167                  scored_result = ScoredRAGResult(
      168                      rag_result=rag_result,
      169                      scores=rag_scores
      170                  )
      171                  all_scored_results.append(scored_result)
      172  
      173          return MultiScoredRAGResult(
      174              query_id=multi_rag_result.query_id,
      175              query=multi_rag_result.query,
      176              scored_rag_results=all_scored_results

      179      def collect_scores_for_consistency(
      180          self,
      181          scored_results: List[MultiScoredRAGResult],
      182          scores_for_consistency: Dict[str, Dict[str, List[float]]],
      183          max_workers: Optional[int] = 5
      184      ) -> Dict[str, Dict[str, List[float]]]:
      185          """Gather scores for consistency evaluation.
      186  
      187          Args:
      188              scored_results: List of evaluated results
      189              scores_for_consistency: Dict to update with scores
      190              max_workers: Thread pool size
      191  
      192          Returns:
      193              Updated scores_for_consistency dict
      194          """
      195          lock = threading.Lock()
      196  
      197          def _extract_scores(score_dict, target: Dict[str, List[float]], metrics: List[str]):
      198              for metric in metrics:
      199                  value = score_dict.get(metric)
      200                  if isinstance(value, (int, float)):
      201                      target.setdefault(metric, []).append(float(value))
      202  
      203          def process_single_result(multi_scored_result: MultiScoredRAGResult):
      204              local_scores = {}
      205              query_id = multi_scored_result.query_id
      206  
      207              if not multi_scored_result.scored_rag_results:
      208                  return
      209  
      210              for scored_result in multi_scored_result.scored_rag_results:
      211                  if scored_result.scores and scored_result.scores.generation_score:
      212                      _extract_scores(
      213                          scored_result.scores.generation_score.scores,
      214                          local_scores,
      215                          self.metrics_to_run_consistency
      216                      )
      217  
      218              with lock:
      219                  if query_id not in scores_for_consistency:
      220                      scores_for_consistency[query_id] = {}
      221                  for metric, vals in local_scores.items():
      222                      scores_for_consistency[query_id].setdefault(metric, []).extend(vals)
      223  
      224          with ThreadPoolExecutor(max_workers=max_workers) as executor:
      225              futures = [
      226                  executor.submit(process_single_result, msr)
      227                  for msr in scored_results
      228              ]
      229              for future in as_completed(futures):
      230                  future.result()
      231  
      232          return scores_for_consistency
