      518  class MultiRLModuleSpec:
      519      """A utility spec class to make it constructing MultiRLModules easier.
      520  
      521      Users can extend this class to modify the behavior of base class. For example to
      522      share neural networks across the modules, the build method can be overridden to
      523      create the shared module first and then pass it to custom module classes that would
      524      then use it as a shared module.
      525      """
      526  
      527      #: The class of the MultiRLModule to construct. By default,
      528      #: this is the base `MultiRLModule` class.
      529      multi_rl_module_class: Type[MultiRLModule] = MultiRLModule
      530      #: Optional global observation space for the MultiRLModule.
      531      #: Useful for shared network components that live only inside the MultiRLModule
      532      #: and don't have their own ModuleID and own RLModule within
      533      #: `self._rl_modules`.
      534      observation_space: Optional[gym.Space] = None
      535      #: Optional global action space for the MultiRLModule. Useful for
      536      #: shared network components that live only inside the MultiRLModule and don't
      537      #: have their own ModuleID and own RLModule within `self._rl_modules`.
      538      action_space: Optional[gym.Space] = None
      539      #: An optional global inference_only flag. If not set (None by
      540      #: default), considers the MultiRLModule to be inference_only=True, only if all
      541      #: submodules also have their own inference_only flags set to True.
      542      inference_only: Optional[bool] = None
      543      # TODO (sven): Once we support MultiRLModules inside other MultiRLModules, we would
      544      #  need this flag in here as well, but for now, we'll leave it out for simplicity.
      545      # learner_only: bool = False
      546      #: An optional global model_config dict. Useful to configure shared
      547      #: network components that only live inside the MultiRLModule and don't have
      548      #: their own ModuleID and own RLModule within `self._rl_modules`.
      549      model_config: Optional[dict] = None
      550      #: The module specs for each individual module. It can be either
      551      #: an RLModuleSpec used for all module_ids or a dictionary mapping from module
      552      #: IDs to RLModuleSpecs for each individual module.
      553      rl_module_specs: Union[RLModuleSpec, Dict[ModuleID, RLModuleSpec]] = None
      554  
      555      # TODO (sven): Deprecate these in favor of using the pure Checkpointable APIs for
      556      #  loading and saving state.
      557      load_state_path: Optional[str] = None
      558      modules_to_load: Optional[Set[ModuleID]] = None

      563      def __post_init__(self):
      564          if self.module_specs is not None:
      565              deprecation_warning(
      566                  old="MultiRLModuleSpec(module_specs=..)",
      567                  new="MultiRLModuleSpec(rl_module_specs=..)",
      568                  error=True,
      569              )
      570          if self.rl_module_specs is None:
      571              raise ValueError(
      572                  "Module_specs cannot be None. It should be either a "
      573                  "RLModuleSpec or a dictionary mapping from module IDs to "
      574                  "RLModuleSpecs for each individual module."
      575              )
      576          self.module_specs = self.rl_module_specs
      577          # Figure out global inference_only setting.
      578          # If not provided (None), only if all submodules are
      579          # inference_only, this MultiRLModule will be inference_only.
      580          self.inference_only = (
      581              self.inference_only
      582              if self.inference_only is not None
      583              else all(spec.inference_only for spec in self.rl_module_specs.values())
      584          )

      586      @OverrideToImplementCustomLogic
      587      def build(self, module_id: Optional[ModuleID] = None) -> RLModule:
      588          """Builds either the MultiRLModule or a (single) sub-RLModule under `module_id`.
      589  
      590          Args:
      591              module_id: Optional ModuleID of a single RLModule to be built. If None
      592                  (default), builds the MultiRLModule.
      593  
      594          Returns:
      595              The built RLModule if `module_id` is provided, otherwise the built
      596              MultiRLModule.
      597          """
      598          self._check_before_build()
      599  
      600          # ModuleID provided, return single-agent RLModule.
      601          if module_id:
      602              return self.rl_module_specs[module_id].build()
      603  
      604          # Return MultiRLModule.
      605          try:
      606              module = self.multi_rl_module_class(
      607                  observation_space=self.observation_space,
      608                  action_space=self.action_space,
      609                  inference_only=self.inference_only,
      610                  model_config=(
      611                      dataclasses.asdict(self.model_config)
      612                      if dataclasses.is_dataclass(self.model_config)
      613                      else self.model_config
      614                  ),
      615                  rl_module_specs=self.rl_module_specs,
      616              )
      617          # Older custom model might still require the old `MultiRLModuleConfig` under
      618          # the `config` arg.
      619          except AttributeError as e:
      620              if self.multi_rl_module_class is not MultiRLModule:
      621                  multi_rl_module_config = self.get_rl_module_config()
      622                  module = self.multi_rl_module_class(multi_rl_module_config)
      623              else:
      624                  raise e
      625  
      626          return module

      628      def add_modules(
      629          self,
      630          module_specs: Dict[ModuleID, RLModuleSpec],
      631          override: bool = True,
      632      ) -> None:
      633          """Add new module specs to the spec or updates existing ones.
      634  
      635          Args:
      636              module_specs: The mapping for the module_id to the single-agent module
      637                  specs to be added to this multi-agent module spec.
      638              override: Whether to override the existing module specs if they already
      639                  exist. If False, they are only updated.
      640          """
      641          if self.rl_module_specs is None:
      642              self.rl_module_specs = {}
      643          for module_id, module_spec in module_specs.items():
      644              if override or module_id not in self.rl_module_specs:
      645                  # Disable our `inference_only` as soon as any single-agent module has
      646                  # `inference_only=False`.
      647                  if not module_spec.inference_only:
      648                      self.inference_only = False
      649                  self.rl_module_specs[module_id] = module_spec
      650              else:
      651                  self.rl_module_specs[module_id].update(module_spec)

      653      def remove_modules(self, module_ids: Union[ModuleID, Collection[ModuleID]]) -> None:
      654          """Removes the provided ModuleIDs from this MultiRLModuleSpec.
      655  
      656          Args:
      657              module_ids: Collection of the ModuleIDs to remove from this spec.
      658          """
      659          for module_id in force_list(module_ids):
      660              self.rl_module_specs.pop(module_id, None)

      662      @classmethod
      663      def from_module(self, module: MultiRLModule) -> "MultiRLModuleSpec":
      664          """Creates a MultiRLModuleSpec from a MultiRLModule.
      665  
      666          Args:
      667              module: The MultiRLModule to create the spec from.
      668  
      669          Returns:
      670              The MultiRLModuleSpec.
      671          """
      672          # we want to get the spec of the underlying unwrapped module that way we can
      673          # easily reconstruct it. The only wrappers that we expect to support today are
      674          # wrappers that allow us to do distributed training. Those will be added back
      675          # by the learner if necessary.
      676          rl_module_specs = {
      677              module_id: RLModuleSpec.from_module(rl_module.unwrapped())
      678              for module_id, rl_module in module._rl_modules.items()
      679          }
      680          multi_rl_module_class = module.__class__
      681          return MultiRLModuleSpec(
      682              multi_rl_module_class=multi_rl_module_class,
      683              observation_space=module.observation_space,
      684              action_space=module.action_space,
      685              inference_only=module.inference_only,
      686              model_config=module.model_config,
      687              rl_module_specs=rl_module_specs,
      688          )

      690      def _check_before_build(self):
      691          if not isinstance(self.rl_module_specs, dict):
      692              raise ValueError(
      693                  f"When build() is called on {self.__class__}, the `rl_module_specs` "
      694                  "attribute should be a dictionary mapping ModuleIDs to "
      695                  "RLModuleSpecs for each individual RLModule."
      696              )

      729      def update(
      730          self,
      731          other: Union["MultiRLModuleSpec", RLModuleSpec],
      732          override: bool = False,
      733      ) -> None:
      734          """Updates this spec with the other spec.
      735  
      736          Traverses this MultiRLModuleSpec's module_specs and updates them with
      737          the module specs from the `other` (Multi)RLModuleSpec.
      738  
      739          Args:
      740              other: The other spec to update this spec with.
      741              override: Whether to override the existing module specs if they already
      742                  exist. If False, they are only updated.
      743          """
      744          if isinstance(other, RLModuleSpec):
      745              # Disable our `inference_only` as soon as any single-agent module has
      746              # `inference_only=False`.
      747              if not other.inference_only:
      748                  self.inference_only = False
      749              for mid, spec in self.rl_module_specs.items():
      750                  self.rl_module_specs[mid].update(other, override=False)
      751          elif isinstance(other.module_specs, dict):
      752              self.add_modules(other.rl_module_specs, override=override)
      753          else:
      754              assert isinstance(other, MultiRLModuleSpec)
      755              if not self.rl_module_specs:
      756                  self.inference_only = other.inference_only
      757                  self.rl_module_specs = other.rl_module_specs
      758              else:
      759                  if not other.inference_only:
      760                      self.inference_only = False
      761                  self.rl_module_specs.update(other.rl_module_specs)
