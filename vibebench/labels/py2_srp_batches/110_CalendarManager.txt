      378      def find_event_occurrence(self, event_id: str, occurrence_date: datetime) -> Event | None:
      379          """Find a specific occurrence of a recurring event.
      380  
      381          Searches for events near the occurrence_date and matches by comparing datetimes.
      382          If the datetime has no timezone, tries both as-provided and UTC interpretations.
      383  
      384          Args:
      385              event_id: The event identifier
      386              occurrence_date: Start time of the occurrence
      387  
      388          Returns:
      389              The matching occurrence, or None if not found
      390          """
      391          result = self._search_occurrence_by_datetime(event_id, occurrence_date)
      392          if result:
      393              return result
      394  
      395          # If naive datetime, try UTC interpretation (common when Claude constructs local times)
      396          if occurrence_date.tzinfo is None:
      397              logger.info(f"No match for {event_id}, trying UTC interpretation of {occurrence_date}")
      398              utc_datetime = occurrence_date.replace(tzinfo=timezone.utc)
      399              result = self._search_occurrence_by_datetime(event_id, utc_datetime)
      400              if result:
      401                  logger.info(f"Found match using UTC interpretation")
      402                  return result
      403  
      404          logger.info(f"No occurrence found for {event_id} at {occurrence_date}")
      405          return None

      407      def _search_occurrence_by_datetime(self, event_id: str, target_datetime: datetime) -> Event | None:
      408          """Search for occurrence by datetime with timezone-aware matching.
      409  
      410          Uses a minimal ±1 minute search window (required by EventKit's predicate API),
      411          then does timezone-normalized datetime matching.
      412  
      413          Args:
      414              event_id: The event identifier
      415              target_datetime: The datetime to match (can be timezone-aware or naive)
      416  
      417          Returns:
      418              Event if found, None otherwise

      420          # Convert search window to naive local time for EventKit
      421          search_start_naive = to_eventkit_datetime(target_datetime - timedelta(minutes=1))
      422          search_end_naive = to_eventkit_datetime(target_datetime + timedelta(minutes=1))
      423  
      424          predicate = self.event_store.predicateForEventsWithStartDate_endDate_calendars_(
      425              search_start_naive, search_end_naive, None
      426          )
      427          events = self.event_store.eventsMatchingPredicate_(predicate)
      428  
      429          # Convert target to naive local time for matching against EventKit's naive times
      430          target_naive = to_eventkit_datetime(target_datetime)
      431  
      432          for ekevent in events:
      433              # EventKit returns naive datetimes in local timezone
      434              if ekevent.eventIdentifier() == event_id and ekevent.startDate() == target_naive:
      435                  logger.debug(f"Found occurrence for {event_id} at {target_datetime}")
      436                  return Event.from_ekevent(ekevent)
