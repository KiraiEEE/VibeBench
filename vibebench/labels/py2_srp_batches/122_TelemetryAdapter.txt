      224      def _log_metric(self, event: MetricEvent) -> None:
      225          # Add metric as an event to the current span
      226          try:
      227              with self._lock:
      228                  # Only try to add to span if we have a valid span_id
      229                  if event.span_id:
      230                      try:
      231                          span_id = int(event.span_id, 16)
      232                          span = _GLOBAL_STORAGE["active_spans"].get(span_id)
      233  
      234                          if span:
      235                              timestamp_ns = int(event.timestamp.timestamp() * 1e9)
      236                              span.add_event(
      237                                  name=f"metric.{event.metric}",
      238                                  attributes={
      239                                      "value": event.value,
      240                                      "unit": event.unit,
      241                                      **(event.attributes or {}),
      242                                  },
      243                                  timestamp=timestamp_ns,
      244                              )
      245                      except (ValueError, KeyError):
      246                          # Invalid span_id or span not found, but we already logged to console above
      247                          pass
      248          except Exception:
      249              # Lock acquisition failed
      250              logger.debug("Failed to acquire lock to add metric to span")
      251  
      252          # Log to OpenTelemetry meter if available
      253          if self.meter is None:
      254              return
      255          if isinstance(event.value, int):
      256              counter = self._get_or_create_counter(event.metric, event.unit)
      257              counter.add(event.value, attributes=event.attributes)
      258          elif isinstance(event.value, float):
      259              up_down_counter = self._get_or_create_up_down_counter(event.metric, event.unit)
      260              up_down_counter.add(event.value, attributes=event.attributes)

      272      def _log_structured(self, event: StructuredLogEvent, ttl_seconds: int) -> None:
      273          with self._lock:
      274              span_id = int(event.span_id, 16)
      275              tracer = trace.get_tracer(__name__)
      276              if event.attributes is None:
      277                  event.attributes = {}
      278              event.attributes["__ttl__"] = ttl_seconds
      279  
      280              # Extract these W3C trace context attributes so they are not written to
      281              # underlying storage, as we just need them to propagate the trace context.
      282              traceparent = event.attributes.pop("traceparent", None)
      283              tracestate = event.attributes.pop("tracestate", None)
      284              if traceparent:
      285                  # If we have a traceparent header value, we're not the root span.
      286                  for root_attribute in ROOT_SPAN_MARKERS:
      287                      event.attributes.pop(root_attribute, None)
      288  
      289              if isinstance(event.payload, SpanStartPayload):
      290                  # Check if span already exists to prevent duplicates
      291                  if span_id in _GLOBAL_STORAGE["active_spans"]:
      292                      return
      293  
      294                  context = None
      295                  if event.payload.parent_span_id:
      296                      parent_span_id = int(event.payload.parent_span_id, 16)
      297                      parent_span = _GLOBAL_STORAGE["active_spans"].get(parent_span_id)
      298                      context = trace.set_span_in_context(parent_span)
      299                  elif traceparent:
      300                      carrier = {
      301                          "traceparent": traceparent,
      302                          "tracestate": tracestate,
      303                      }
      304                      context = TraceContextTextMapPropagator().extract(carrier=carrier)
      305  
      306                  span = tracer.start_span(
      307                      name=event.payload.name,
      308                      context=context,
      309                      attributes=event.attributes or {},
      310                  )
      311                  _GLOBAL_STORAGE["active_spans"][span_id] = span
      312  
      313              elif isinstance(event.payload, SpanEndPayload):
      314                  span = _GLOBAL_STORAGE["active_spans"].get(span_id)
      315                  if span:
      316                      if event.attributes:
      317                          span.set_attributes(event.attributes)
      318  
      319                      status = (
      320                          trace.Status(status_code=trace.StatusCode.OK)
      321                          if event.payload.status == SpanStatus.OK
      322                          else trace.Status(status_code=trace.StatusCode.ERROR)
      323                      )
      324                      span.set_status(status)
      325                      span.end()
      326                      _GLOBAL_STORAGE["active_spans"].pop(span_id, None)
      327              else:
      328                  raise ValueError(f"Unknown structured log event: {event}")
