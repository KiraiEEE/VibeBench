       27      """Abstract base class for managing a global state variable.
       28  
       29      Subclasses must:
       30  
       31      - Specify a `Literal` as the subclass' type argument. The literal's
       32        arguments must specify all allowable options for the state variable.
       33      - Define a getter and setter for the state. Input validation is not
       34        required - the input is automatically validated when setting the state.
       35  
       36      Examples
       37      --------
       38      >>> from pyvista.core.utilities.state_manager import _StateManager
       39      >>> from typing import Literal
       40  
       41      Define the available options as a ``Literal`` and initialize a global state variable.
       42  
       43      >>> _StateOptions = Literal['on', 'off']
       44      >>> _GLOBAL_STATE = ['off']  # Init global state. Use list to make it mutable.
       45  
       46      Define the class and its state property.
       47  
       48      >>> class MyState(_StateManager[_StateOptions]):
       49      ...     @property
       50      ...     def _state(self) -> _StateOptions:
       51      ...         return _GLOBAL_STATE[0]
       52      ...
       53      ...     @_state.setter
       54      ...     def _state(self, state: _StateOptions) -> None:
       55      ...         _GLOBAL_STATE[0] = state
       56  
       57      Finally, create an instance of the state manager.
       58  
       59      >>> my_state = MyState()
       60  
       61      Get the state.
       62  
       63      >>> my_state()
       64      'off'
       65  
       66      Set the state.
       67  
       68      >>> _ = my_state('on')
       69      >>> my_state()
       70      'on'
       71  
       72      Use it as a context manager to set the state temporarily:
       73  
       74      >>> with my_state('off'):
       75      ...     pass
