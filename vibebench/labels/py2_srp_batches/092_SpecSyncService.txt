      111      async def sync_spec(
      112          self,
      113          spec_id: str,
      114          session: Optional[AsyncSession] = None,
      115      ) -> SyncResult:
      116          """Sync all phase data for a spec to the database.
      117  
      118          This is the main entry point for the SYNC phase. It reads the
      119          phase_data from the spec and persists requirements, tasks, and
      120          criteria with deduplication.
      121  
      122          Args:
      123              spec_id: ID of the spec to sync.
      124              session: Optional async session.
      125  
      126          Returns:
      127              SyncResult with success status and statistics.
      128          """
      129          stats = SyncStats()
      130  
      131          async def _sync(sess: AsyncSession) -> SyncResult:
      132              # Load spec with phase data
      133              result = await sess.execute(
      134                  select(Spec)
      135                  .filter(Spec.id == spec_id)
      136                  .options(
      137                      selectinload(Spec.requirements).selectinload(
      138                          SpecRequirement.criteria
      139                      ),
      140                      selectinload(Spec.tasks),
      141                  )
      142              )
      143              spec = result.scalar_one_or_none()
      144  
      145              if not spec:
      146                  return SyncResult(
      147                      success=False,
      148                      spec_id=spec_id,
      149                      message="Spec not found",
      150                  )
      151  
      152              if not spec.phase_data:
      153                  return SyncResult(
      154                      success=True,
      155                      spec_id=spec_id,
      156                      stats=stats,
      157                      message="No phase data to sync",
      158                  )
      159  
      160              # Sync requirements from requirements phase
      161              requirements_data = spec.phase_data.get("requirements", {})
      162              if requirements_data:
      163                  await self._sync_requirements(
      164                      sess, spec, requirements_data, stats
      165                  )
      166  
      167              # Sync tasks from tasks phase
      168              tasks_data = spec.phase_data.get("tasks", {})
      169              if tasks_data:
      170                  await self._sync_tasks(sess, spec, tasks_data, stats)
      171  
      172              # Update linked_tickets count on spec
      173              if stats.tickets_created > 0:
      174                  spec.linked_tickets = (spec.linked_tickets or 0) + stats.tickets_created
      175  
      176              # Update spec embedding and hash for deduplication
      177              await self._update_spec_dedup_fields(sess, spec)
      178  
      179              # Commit all changes
      180              await sess.commit()
      181  
      182              logger.info(
      183                  "Spec sync complete",
      184                  extra={
      185                      "spec_id": spec_id,
      186                      "stats": stats.to_dict(),
      187                  },
      188              )
      189  
      190              return SyncResult(
      191                  success=True,
      192                  spec_id=spec_id,
      193                  stats=stats,
      194                  message=f"Synced {stats.requirements_created} requirements, "
      195                  f"{stats.tasks_created} tasks "
      196                  f"(skipped {stats.requirements_skipped + stats.tasks_skipped} duplicates)",
      197              )
      198  
      199          if session:
      200              return await _sync(session)
      201          else:
      202              async with self.db.get_async_session() as sess:
      203                  return await _sync(sess)
