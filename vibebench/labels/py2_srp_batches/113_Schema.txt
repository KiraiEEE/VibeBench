      173      @classmethod
      174      def sample(
      175          cls,
      176          num_rows: int | None = None,
      177          *,
      178          overrides: (
      179              Mapping[str, Sequence[Any] | Any] | Sequence[Mapping[str, Any]] | None
      180          ) = None,
      181          generator: Generator | None = None,
      182      ) -> DataFrame[Self]:
      183          """Create a random data frame with a predefined number of rows.
      184  
      185          Generally, **this method should only be used for testing**. Also, if you want
      186          to generate _realistic_ test data, it is inevitable to implement your custom
      187          sampling logic (by making use of the :class:`Generator` class).
      188  
      189          In order to allow for sampling random data frames in the presence of custom
      190          rules and primary key constraints, this method performs *fuzzy sampling*: it
      191          samples in a loop until it finds a data frame of length `num_rows` which
      192          adhere to the schema. The maximum number of sampling rounds is configured via
      193          `max_sampling_iterations` in the :class:`Config` class. By fixing this setting
      194          to 1, it is only possible to reliably sample from schemas without custom rules
      195          and without primary key constraints.
      196  
      197          Args:
      198              num_rows: The (optional) number of rows to sample for creating the random
      199                  data frame. Must be provided (only) if no `overrides` are provided. If
      200                  this is `None`, the number of rows in the data frame is determined by
      201                  the length of the values in `overrides`.
      202              overrides: Fixed values for a subset of the columns of the sampled data
      203                  frame. Just like when initializing a :mod:`polars.DataFrame`, overrides
      204                  may either be provided as "column-" or "row-layout", i.e. via a mapping
      205                  or a list of mappings, respectively. The number of rows in the result
      206                  data frame is equal to the length of the values in `overrides`. If both
      207                  `overrides` and `num_rows` are provided, the length of the values in
      208                  `overrides` must be equal to `num_rows`. The order of the items is
      209                  guaranteed to match the ordering in the returned data frame. When providing
      210                  values for a column, no sampling is performed for that column.
      211              generator: The (seeded) generator to use for sampling data. If `None`, a
      212                  generator with random seed is automatically created.
      213  
      214          Returns:
      215              A data frame valid under the current schema with a number of rows that matches
      216              the length of the values in `overrides` or `num_rows`.
      217  
      218          Raises:
      219              ValueError: If `num_rows` is not equal to the length of the values in
      220                  `overrides`.
      221              ValueError: If `overrides` are specified as a sequence of mappings and
      222                  the mappings do not provide the same keys.
      223              ValueError: If no valid data frame can be found in the configured maximum
      224                  number of iterations.
      225  
      226          Attention:
      227              Be aware that, due to sampling in a loop, the runtime of this method can be
      228              significant for complex schemas. Consider passing a seeded generator and
      229              evaluate whether the runtime impact in the tests is bearable. Alternatively,
      230              it can be beneficial to provide custom column overrides for columns
      231              associated with complex validation rules.
      232          """
      233          g = generator or Generator()
      234  
      235          # Precondition: valid overrides. We put them into a data frame to remember which
      236          # values have been used in the algorithm below. When the user passes a sequence
      237          # of mappings, they do not require to have the same keys. Hence, we have to
      238          # remember that the data frame has "holes".
      239          missing_override_indices: dict[str, pl.Series] = {}
      240          if overrides is not None:
      241              override_keys = (
      242                  set(overrides)
      243                  if isinstance(overrides, Mapping)
      244                  else (
      245                      set.union(*[set(o.keys()) for o in overrides])
      246                      if len(overrides) > 0
      247                      else set()
      248                  )
      249              )
      250  
      251              # Check that all override keys refer to valid columns
      252              column_names = set(cls.column_names())
      253              if not override_keys.issubset(column_names):
      254                  raise ValueError(
      255                      f"Values are provided for columns {override_keys - column_names} "
      256                      "which are not in the schema."
      257                  )
      258  
      259              # Remember the "holes" of the inputs if overrides are provided as a sequence
      260              if isinstance(overrides, Sequence):
      261                  for key in override_keys:
      262                      indices = [
      263                          i for i, override in enumerate(overrides) if key not in override
      264                      ]
      265                      if len(indices) > 0:
      266                          missing_override_indices[key] = pl.Series(indices)
      267  
      268              # NOTE: Even if the user-provided overrides have "holes", we can still just
      269              #  create the data frame. Polars will fill the missing values with nulls, we
      270              #  will replace them later during sampling. If we were to already replace
      271              #  them here, we would not be able to resample these values.
      272              values = pl.DataFrame(
      273                  overrides,
      274                  schema={
      275                      name: col.dtype
      276                      for name, col in cls.columns().items()
      277                      if name in override_keys
      278                  },
      279                  orient="col" if isinstance(overrides, Mapping) else "row",
      280              )
      281              if num_rows is not None and num_rows != values.height:
      282                  raise ValueError(
      283                      "`num_rows` is different from the length of the provided overrides."
      284                  )
      285              num_rows = values.height
      286          else:
      287              # In case that neither `num_rows` nor `overrides` are provided, fall back to `1`
      288              if num_rows is None:
      289                  num_rows = 1
      290  
      291              # NOTE: Code becomes rather ugly when allowing `values` to be `None`. Hence,
      292              #  we're using an empty data frame here and branch on the height of the data
      293              #  frame.
      294              values = pl.DataFrame()
      295  
      296          # Prepare expressions for columns that need to be preprocessed during sampling
      297          # iterations.
      298          sampling_overrides = cls._sampling_overrides()
      299          if superfluous_overrides := sampling_overrides.keys() - cls.columns():
      300              raise ValueError(
      301                  "The schema defines `_sampling_overrides` for columns that are not in the "
      302                  f"schema: {superfluous_overrides}."
      303              )
      304  
      305          override_expressions = [
      306              # Cast needed as column pre-processing might change the data types of a column
      307              expr.cast(cls.columns()[col].dtype).alias(col)
      308              for col, expr in sampling_overrides.items()
      309              # Only pre-process columns that are in the schema and were not provided
      310              # through `overrides`.
      311              if col not in values.columns
      312          ]
      313  
      314          # During sampling, we need to potentially sample many times if the schema has
      315          # (complex) rules.
      316          #
      317          # At the same time, we need to ensure that the overrides provided by the user
      318          # keep their order. To this end, we add a row index to the `values` data frame
      319          # and re-order the result accordingly.
      320          #
      321          # NOTE: One option to potentially run fewer loops would be to sample more than
      322          #  `n` elements. However, we cannot simply slice the result as that could
      323          #  potentially violate group rules. Hence, we're bound to calling `filter` on
      324          #  dataframes with length `n`.
      325          values = values.with_row_index(name="__row_index__")
      326  
      327          result, used_values, remaining_values = cls._sample_filter(
      328              num_rows,
      329              g,
      330              previous_result=cls.create_empty(),
      331              used_values=values.slice(0, 0),
      332              remaining_values=values,
      333              override_expressions=override_expressions,
      334              missing_value_indices=missing_override_indices,
      335          )
      336  
      337          sampling_rounds = 1
      338          while len(result) != num_rows:
      339              if sampling_rounds >= Config.options["max_sampling_iterations"]:
      340                  relevant_rows = pl.concat(
      341                      [
      342                          df
      343                          for df in [
      344                              result,
      345                              used_values.drop("__row_index__"),
      346                              remaining_values.drop("__row_index__"),
      347                          ]
      348                      ],
      349                      how="diagonal",  # `used_values` and `remaining_values` only contain columns in `overrides`
      350                  )
      351                  validation_error = None
      352                  try:
      353                      cls.validate(relevant_rows)
      354                  except ValidationError as e:
      355                      validation_error = str(e)
      356                  raise ValueError(
      357                      f"After sampling for {Config.options['max_sampling_iterations']} "
      358                      f"iterations, {validation_error or 'no valid data frame was found'}. "
      359                      f"Consider increasing the maximum number "
      360                      "of sampling iterations via `dy.Config` or implement your custom sampling "
      361                      "logic. Alternatively, passing predefined value to `overrides` "
      362                      "or implementing `_sampling_overrides` for your schema can also "
      363                      "help the sampling procedure find a valid data frame."
      364                  )
      365              result, used_values, remaining_values = cls._sample_filter(
      366                  num_rows - len(result),
      367                  g,
      368                  previous_result=result,
      369                  used_values=used_values,
      370                  remaining_values=remaining_values,
      371                  override_expressions=override_expressions,
      372                  missing_value_indices=missing_override_indices,
      373              )
      374              sampling_rounds += 1
      375  
      376          if len(used_values) > 0:
      377              # If we used any values, we want to re-order the result to adhere to the
      378              # input ordering
      379              result = (
      380                  pl.concat(
      381                      [result, used_values.select("__row_index__")], how="horizontal"
      382                  )
      383                  .sort(by="__row_index__")
      384                  .drop("__row_index__")
      385              )
      386  
      387          # NOTE: There's no need for an additional `validate` or `cast` here since this
      388          #  is just a re-ordered version of a data frame that was returned from `filter`.
      389          #  Row order does not affect the validity of a data frame.
      390          return result  # type: ignore

      392      @classmethod
      393      def _sample_filter(
      394          cls,
      395          num_rows: int,
      396          generator: Generator,
      397          previous_result: pl.DataFrame,
      398          used_values: pl.DataFrame,
      399          remaining_values: pl.DataFrame,
      400          override_expressions: list[pl.Expr],
      401          missing_value_indices: dict[str, pl.Series],
      402      ) -> tuple[pl.DataFrame, pl.DataFrame, pl.DataFrame]:
      403          """Private method to sample a data frame with the schema including subsequent
      404          filtering.
      405  
      406          Returns:
      407              The filtered data frame, the used values, and the remaining values.
      408          """
      409          sampled = pl.DataFrame(
      410              {
      411                  name: (
      412                      col.sample(generator, num_rows)
      413                      if name not in remaining_values.columns
      414                      else remaining_values.get_column(name)
      415                  )
      416                  for name, col in cls.columns().items()
      417              }
      418          )
      419  
      420          # If we have missing value indices, we need to sample new values for the
      421          # indices that overlap with indices in the remaining values and replace them
      422          # in the sampled data frame.
      423          for name, indices in missing_value_indices.items():
      424              remapped_indices = (
      425                  indices.to_frame("idx")
      426                  .join(
      427                      remaining_values.select("__row_index__").with_row_index(
      428                          "__row_index_loop__"
      429                      ),
      430                      left_on="idx",
      431                      right_on="__row_index__",
      432                  )
      433                  .select("__row_index_loop__")
      434                  .to_series()
      435              )
      436              if (num := len(remapped_indices)) > 0:
      437                  sampled_values = cls.columns()[name].sample(generator, num)
      438                  sampled = sampled.with_columns(
      439                      sampled[name]
      440                      # NOTE: We need to sort here as `scatter` requires sorted indices.
      441                      #  Due to concatenations in `remaining_values`, the indices can go
      442                      #  out of order.
      443                      .scatter(remapped_indices.sort(), sampled_values)
      444                      .alias(name)
      445                  )
      446  
      447          combined_dataframe = pl.concat([previous_result, sampled])
      448          # Pre-process columns before filtering.
      449          combined_dataframe = combined_dataframe.with_columns(override_expressions)
      450  
      451          # NOTE: We already know that all columns have the correct dtype
      452          rules = cls._validation_rules(with_cast=False)
      453          if not rules:
      454              # We surely included all items from `remaining_values`
      455              return combined_dataframe, remaining_values, remaining_values.head(0)
      456  
      457          evaluated = combined_dataframe.lazy().pipe(cls._with_evaluated_rules, rules)
      458          filtered = evaluated.filter(_COLUMN_VALID).select(cls.column_names())
      459  
      460          concat_values = pl.concat([used_values, remaining_values])
      461          if concat_values.height == 0:
      462              # If we didn't provide any values, we can simply return empty data frames
      463              # with the right schema for used and remaining values
      464              return filtered.collect(), concat_values, concat_values
