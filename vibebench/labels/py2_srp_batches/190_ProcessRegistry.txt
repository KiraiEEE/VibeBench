      218      def spawn_via_env(
      219          self,
      220          env: Any,
      221          command: str,
      222          cwd: str = None,
      223          task_id: str = "",
      224          session_key: str = "",
      225          timeout: int = 10,
      226      ) -> ProcessSession:
      227          """
      228          Spawn a background process through a non-local environment backend.
      229  
      230          For Docker/Singularity/Modal/SSH: runs the command inside the sandbox
      231          using the environment's execute() interface. We wrap the command to
      232          capture the in-sandbox PID and redirect output to a log file inside
      233          the sandbox, then poll the log via subsequent execute() calls.
      234  
      235          This is less capable than local spawn (no live stdout pipe, no stdin),
      236          but it ensures the command runs in the correct sandbox context.
      237          """
      238          session = ProcessSession(
      239              id=f"proc_{uuid.uuid4().hex[:12]}",
      240              command=command,
      241              task_id=task_id,
      242              session_key=session_key,
      243              cwd=cwd,
      244              started_at=time.time(),
      245              env_ref=env,
      246          )
      247  
      248          # Run the command in the sandbox with output capture
      249          log_path = f"/tmp/hermes_bg_{session.id}.log"
      250          pid_path = f"/tmp/hermes_bg_{session.id}.pid"
      251          quoted_command = shlex.quote(command)
      252          bg_command = (
      253              f"nohup bash -c {quoted_command} > {log_path} 2>&1 & "
      254              f"echo $! > {pid_path} && cat {pid_path}"
      255          )
      256  
      257          try:
      258              result = env.execute(bg_command, timeout=timeout)
      259              output = result.get("output", "").strip()
      260              # Try to extract the PID from the output
      261              for line in output.splitlines():
      262                  line = line.strip()
      263                  if line.isdigit():
      264                      session.pid = int(line)
      265                      break
      266          except Exception as e:
      267              session.exited = True
      268              session.exit_code = -1
      269              session.output_buffer = f"Failed to start: {e}"
      270  
      271          if not session.exited:
      272              # Start a poller thread that periodically reads the log file
      273              reader = threading.Thread(
      274                  target=self._env_poller_loop,
      275                  args=(session, env, log_path, pid_path),
      276                  daemon=True,
      277                  name=f"proc-poller-{session.id}",
      278              )
      279              session._reader_thread = reader
      280              reader.start()
      281  
      282          with self._lock:
      283              self._prune_if_needed()
      284              self._running[session.id] = session
      285  
      286          self._write_checkpoint()
      287          return session

      291      def _reader_loop(self, session: ProcessSession):
      292          """Background thread: read stdout from a local Popen process."""
      293          first_chunk = True
      294          try:
      295              while True:
      296                  chunk = session.process.stdout.read(4096)
      297                  if not chunk:
      298                      break
      299                  if first_chunk:
      300                      chunk = self._clean_shell_noise(chunk)
      301                      first_chunk = False
      302                  with session._lock:
      303                      session.output_buffer += chunk
      304                      if len(session.output_buffer) > session.max_output_chars:
      305                          session.output_buffer = session.output_buffer[-session.max_output_chars:]
      306          except Exception as e:
      307              logger.debug("Process stdout reader ended: %s", e)
      308  
      309          # Process exited
      310          try:
      311              session.process.wait(timeout=5)
      312          except Exception as e:
      313              logger.debug("Process wait timed out or failed: %s", e)
      314          session.exited = True
      315          session.exit_code = session.process.returncode
      316          self._move_to_finished(session)

      318      def _env_poller_loop(
      319          self, session: ProcessSession, env: Any, log_path: str, pid_path: str
      320      ):
      321          """Background thread: poll a sandbox log file for non-local backends."""
      322          while not session.exited:
      323              time.sleep(2)  # Poll every 2 seconds
      324              try:

      326                  result = env.execute(f"cat {log_path} 2>/dev/null", timeout=10)
      327                  new_output = result.get("output", "")
      328                  if new_output:
      329                      with session._lock:
      330                          session.output_buffer = new_output
      331                          if len(session.output_buffer) > session.max_output_chars:
      332                              session.output_buffer = session.output_buffer[-session.max_output_chars:]
      333  
      334                  # Check if process is still running
      335                  check = env.execute(
      336                      f"kill -0 $(cat {pid_path} 2>/dev/null) 2>/dev/null; echo $?",
      337                      timeout=5,

      339                  check_output = check.get("output", "").strip()
      340                  if check_output and check_output.splitlines()[-1].strip() != "0":
      341                      # Process has exited -- get exit code
      342                      exit_result = env.execute(

      348                          session.exit_code = int(exit_str.splitlines()[-1].strip())
      349                      except (ValueError, IndexError):
      350                          session.exit_code = -1

      355              except Exception:
      356                  # Environment might be gone (sandbox reaped, etc.)
      357                  session.exited = True

      362      def _pty_reader_loop(self, session: ProcessSession):
      363          """Background thread: read output from a PTY process."""
      364          pty = session._pty
      365          try:
      366              while pty.isalive():
      367                  try:
      368                      chunk = pty.read(4096)
      369                      if chunk:
      370                          # ptyprocess returns bytes
      371                          text = chunk if isinstance(chunk, str) else chunk.decode("utf-8", errors="replace")
      372                          with session._lock:
      373                              session.output_buffer += text
      374                              if len(session.output_buffer) > session.max_output_chars:
      375                                  session.output_buffer = session.output_buffer[-session.max_output_chars:]
      376                  except EOFError:
      377                      break
      378                  except Exception:
      379                      break
      380          except Exception as e:
      381              logger.debug("PTY stdout reader ended: %s", e)
      382  
      383          # Process exited
      384          try:
      385              pty.wait()
      386          except Exception as e:
      387              logger.debug("PTY wait timed out or failed: %s", e)
      388          session.exited = True
      389          session.exit_code = pty.exitstatus if hasattr(pty, 'exitstatus') else -1
      390          self._move_to_finished(session)

      392      def _move_to_finished(self, session: ProcessSession):
      393          """Move a session from running to finished."""
      394          with self._lock:
      395              self._running.pop(session.id, None)
      396              self._finished[session.id] = session

      406      def poll(self, session_id: str) -> dict:
      407          """Check status and get new output for a background process."""
      408          session = self.get(session_id)
      409          if session is None:
      410              return {"status": "not_found", "error": f"No process with ID {session_id}"}
      411  
      412          with session._lock:
      413              output_preview = session.output_buffer[-1000:] if session.output_buffer else ""
      414  
      415          result = {
      416              "session_id": session.id,
      417              "command": session.command,
      418              "status": "exited" if session.exited else "running",
      419              "pid": session.pid,
      420              "uptime_seconds": int(time.time() - session.started_at),
      421              "output_preview": output_preview,
      422          }
      423          if session.exited:
      424              result["exit_code"] = session.exit_code
      425          if session.detached:
      426              result["detached"] = True
      427              result["note"] = "Process recovered after restart -- output history unavailable"
