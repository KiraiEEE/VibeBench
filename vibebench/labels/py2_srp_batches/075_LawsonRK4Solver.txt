      296  class LawsonRK4Solver(AbstractSolver):
      297      """
      298      Lawson-RK4 exponential integrator for the Vlasov-Maxwell system.
      299  
      300      Solves dy/dt = L·y + N(y,t) where L is handled by exact exponentials
      301      and N is advanced with classical RK4 in the integrating factor frame.
      302  
      303      Algorithm per step (dt = t1 - t0):
      304  
      305          Eₕ(·) = exp(L·dt/2),  Ef(·) = exp(L·dt)
      306  
      307          N₁ = N(yₙ, tₙ)
      308          y*  = Eₕ(yₙ) + (dt/2)·Eₕ(N₁)
      309          N₂ = N(y*, tₙ + dt/2)
      310  
      311          y** = Eₕ(yₙ) + (dt/2)·N₂
      312          N₃ = N(y**, tₙ + dt/2)
      313  
      314          y*** = Ef(yₙ) + dt·Eₕ(N₃)
      315          N₄ = N(y***, tₙ + dt)
      316  
      317          yₙ₊₁ = Ef(yₙ) + (dt/6)·[Ef(N₁) + 2·Eₕ(N₂) + 2·Eₕ(N₃) + N₄]
      318  
      319      This is 4th-order accurate and treats all linear stiffness exactly.
      320      Uses fixed timestep (no adaptive error estimate).
      321  
      322      Args:
      323          combined_exp: CombinedLinearExponential handling exp(L·s) for all state components
      324      """

      340      def step(self, terms, t0, t1, y0, args, solver_state, made_jump):
      341          """
      342          Perform one Lawson-RK4 step.
      343  
      344          Args:
      345              terms: ODETerm wrapping the NonlinearVectorField
      346              t0, t1: Step interval
      347              y0: State dict (float64 views)
      348              args: Physical parameters
      349              solver_state: Unused (stateless)
      350              made_jump: Unused
      351  
      352          Returns:
      353              (y1, y_error, dense_info, solver_state, result)
      354          """
      355          dt = t1 - t0
      356          N = terms.vf  # NonlinearVectorField.__call__
      357          exp_L = self.combined_exp.apply  # exp_L(y, s) applies exp(L·s)
      358  
      359          # Cache repeated exponential applications
      360          Eh_yn = exp_L(y0, dt / 2)
      361          Ef_yn = exp_L(y0, dt)
      362  
      363          # --- Stage 1 ---
      364          N1 = N(t0, y0, args)
      365  
      366          # --- Stage 2 ---
      367          Eh_N1 = exp_L(N1, dt / 2)
      368          y_star = _tree_add(Eh_yn, _tree_scale(Eh_N1, dt / 2))
      369          N2 = N(t0 + dt / 2, y_star, args)
      370  
      371          # --- Stage 3 ---
      372          y_dstar = _tree_add(Eh_yn, _tree_scale(N2, dt / 2))
      373          N3 = N(t0 + dt / 2, y_dstar, args)
      374  
      375          # --- Stage 4 ---
      376          Eh_N3 = exp_L(N3, dt / 2)
      377          y_tstar = _tree_add(Ef_yn, _tree_scale(Eh_N3, dt))
      378          N4 = N(t1, y_tstar, args)
      379  
      380          # --- Combine ---
      381          Ef_N1 = exp_L(N1, dt)  # exp(L·dt) applied to N1
      382          Eh_N2 = exp_L(N2, dt / 2)
      383          Eh_N3_comb = Eh_N3  # Already computed above
      384  
      385          weighted = _tree_scale(
      386              _tree_add(
      387                  _tree_add(Ef_N1, _tree_scale(Eh_N2, 2.0)),
      388                  _tree_add(_tree_scale(Eh_N3_comb, 2.0), N4),
      389              ),
      390              dt / 6.0,
      391          )
      392          y1 = _tree_add(Ef_yn, weighted)
      393  
      394          # Embedded 2nd-order companion (Lawson-Heun) for adaptive step control.
      395          # Uses already-computed Ef_N1 and N4 — no extra nonlinear evaluations.
      396          # y1_2nd = Ef(y0) + (dt/2) * (Ef(N1) + N4)   [trapezoid rule in IF frame]
      397          # error = y1 - y1_2nd = O(dt^3), valid for PIDController with order=2.
      398          y1_2nd = _tree_add(Ef_yn, _tree_scale(_tree_add(Ef_N1, N4), dt / 2.0))
      399          y_error = jax.tree.map(lambda a, b: a - b, y1, y1_2nd)
      400  
      401          # Dense info for interpolation (linear between endpoints)
      402          dense_info = dict(y0=y0, y1=y1)
      403
