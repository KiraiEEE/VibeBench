      141      def _get_data_statistics(
      142          self, num_data: int | None, num_per_example_loss_terms: int | None
      143      ) -> tuple[int, int | None]:
      144          """Determine the number of data points and per-example loss terms.
      145  
      146          Traverses the data at most once, computing whichever statistics were not
      147          explicitly provided. The number of per-example loss terms is only
      148          inferred when ``NEEDS_NUM_PER_EXAMPLE_LOSS_TERMS`` is ``True`` and
      149          ``num_per_example_loss_terms`` is ``None``.
      150  
      151          Args:
      152              num_data: Number of data points, or ``None`` to infer.
      153              num_per_example_loss_terms: Per-example loss terms, or ``None`` to
      154                  infer when ``NEEDS_NUM_PER_EXAMPLE_LOSS_TERMS`` is ``True``.
      155  
      156          Returns:
      157              Tuple of ``(N_data, num_per_example_loss_terms)``. The second
      158              element is ``None`` when not required.
      159  
      160          Raises:
      161              ValueError: If the inferred number of loss terms is not divisible
      162                  by the number of data points.
      163          """
      164          need_N_data = num_data is None
      165          need_loss_terms = (
      166              self.NEEDS_NUM_PER_EXAMPLE_LOSS_TERMS
      167              and self._loss_func is not None
      168              and num_per_example_loss_terms is None
      169          )
      170          if not need_N_data and not need_loss_terms:
      171              return num_data, num_per_example_loss_terms
      172  
      173          # Accumulate number of data points and (maybe) loss terms
      174          N_data_acc, num_loss_terms_acc = 0, 0
      175          for X, y in self._loop_over_data(desc="data_statistics"):
      176              if need_N_data:
      177                  N_data_acc += self._batch_size_fn(X)
      178              if need_loss_terms:
      179                  num_loss_terms_acc += (
      180                      y.numel()
      181                      if isinstance(self._loss_func, CrossEntropyLoss)
      182                      else y.shape[:-1].numel()
      183                  )
      184  
      185          # Maybe update number of passed data points and loss terms
      186          N_data = N_data_acc if need_N_data else num_data
      187  
      188          if need_loss_terms:
      189              if num_loss_terms_acc % N_data != 0:
      190                  raise ValueError(
      191                      "The number of loss terms must be divisible by the number "
      192                      f"of data points; num_loss_terms="
      193                      f"{num_loss_terms_acc}, N_data={N_data}."
      194                  )
      195              num_per_example_loss_terms = num_loss_terms_acc // N_data
      196  
      197          return N_data, num_per_example_loss_terms

      199      def _check_deterministic(self, rtol: float = 5e-5, atol: float = 1e-6):
      200          """Check that the data and model are deterministic.
      201  
      202          Two independent passes over the data must yield identical total losses
      203          and total gradients. If ``FIXED_DATA_ORDER`` is ``True``, also checks
      204          that each mini-batch yields identical inputs, predictions, losses, and
      205          gradients.
      206  
      207          Subclasses can override this method to add additional checks (e.g.
      208          verifying ``vmap`` compatibility). They should call ``super()`` first.
      209  
      210          Args:
      211              rtol: Relative tolerance for comparison. Default: ``5e-5``.
      212              atol: Absolute tolerance for comparison. Default: ``1e-6``.
      213  
      214          Raises:
      215              RuntimeError: If non-deterministic behavior is detected.
      216          """
      217          has_loss = self._loss_func is not None
      218  
      219          if has_loss:
      220              total_grad1 = [zeros_like(p) for p in self._params.values()]
      221              total_grad2 = [zeros_like(p) for p in self._params.values()]
      222              total_loss1 = tensor(0.0, device=self.device, dtype=self.dtype)
      223              total_loss2 = tensor(0.0, device=self.device, dtype=self.dtype)
      224  
      225          for ((X1, y1), pred1, loss1, grad1), ((X2, y2), pred2, loss2, grad2) in zip(
      226              self._data_prediction_loss_gradient(), self._data_prediction_loss_gradient()
      227          ):
      228              if self.FIXED_DATA_ORDER:
      229                  self._check_deterministic_batch(
      230                      (X1, X2),
      231                      (y1, y2),
      232                      (pred1, pred2),
      233                      (loss1, loss2),
      234                      (grad1, grad2),
      235                      has_loss,
      236                      rtol=rtol,
      237                      atol=atol,
      238                  )
      239  
      240              if has_loss:
      241                  total_loss1.add_(loss1)
      242                  total_loss2.add_(loss2)
      243                  for tg1, g1 in zip(total_grad1, grad1):
      244                      tg1.add_(g1)
      245                  for tg2, g2 in zip(total_grad2, grad2):
      246                      tg2.add_(g2)
      247  
      248          if has_loss:
      249              if not allclose_report(total_loss1, total_loss2, rtol=rtol, atol=atol):
      250                  raise RuntimeError("Check for deterministic total loss failed.")
      251              if any(
      252                  not allclose_report(g1, g2, atol=atol, rtol=rtol)
      253                  for g1, g2 in zip(total_grad1, total_grad2)
      254              ):
      255                  raise RuntimeError("Check for deterministic total gradient failed.")

      258      def _check_deterministic_batch(
      259          Xs: tuple[Tensor | MutableMapping, Tensor | MutableMapping],
      260          ys: tuple[Tensor, Tensor],
      261          predictions: tuple[Tensor, Tensor],
      262          losses: tuple[Tensor | None, Tensor | None],
      263          gradients: tuple[list[Tensor] | None, list[Tensor] | None],
      264          has_loss_func: bool,
      265          rtol: float = 1e-5,
      266          atol: float = 1e-8,
      267      ):
      268          """Compare two batch outputs of a data-prediction-loss-gradient pass.
      269  
      270          Args:
      271              Xs: The two data inputs to compare.
      272              ys: The two data targets to compare.
      273              predictions: The two predictions to compare.
      274              losses: The two losses to compare.
      275              gradients: The two gradients to compare.
      276              has_loss_func: Whether a loss function is present.
      277              rtol: Relative tolerance for comparison. Default: ``1e-5``.
      278              atol: Absolute tolerance for comparison. Default: ``1e-8``.
      279  
      280          Raises:
      281              RuntimeError: If any of the pairs mismatch.
      282          """
      283          X1, X2 = Xs
      284          if isinstance(X1, MutableMapping) and isinstance(X2, MutableMapping):
      285              for k in X1:
      286                  v1, v2 = X1[k], X2[k]
      287                  if isinstance(v1, Tensor) and not allclose_report(
      288                      v1, v2, rtol=rtol, atol=atol
      289                  ):
      290                      raise RuntimeError("Check for deterministic X failed.")
      291          elif not allclose_report(X1, X2, rtol=rtol, atol=atol):
      292              raise RuntimeError("Check for deterministic X failed.")
      293  
      294          y1, y2 = ys
      295          if not allclose_report(y1, y2, rtol=rtol, atol=atol):
      296              raise RuntimeError("Check for deterministic y failed.")
      297  
      298          pred1, pred2 = predictions
      299          if not allclose_report(pred1, pred2, rtol=rtol, atol=atol):
      300              raise RuntimeError("Check for deterministic batch prediction failed.")
      301  
      302          loss1, loss2 = losses
      303          grad1, grad2 = gradients
      304          if has_loss_func:
      305              if not allclose_report(loss1, loss2, rtol=rtol, atol=atol):
      306                  raise RuntimeError("Check for deterministic batch loss failed.")
      307              if any(
      308                  not allclose_report(g1, g2, rtol=rtol, atol=atol)
      309                  for g1, g2 in zip(grad1, grad2)
      310              ):
      311                  raise RuntimeError("Check for deterministic batch gradient failed.")
