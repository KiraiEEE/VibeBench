      159      async def execute_with_timeout(self, task: str) -> ChatMessage:
      160          """Execute the task with a timeout constraint.
      161  
      162          Note: This is a legacy helper method used by some workflows.
      163          Ideally workflows should call run() directly.
      164          """
      165          start_time = time.time()
      166          success = False
      167          try:
      168              # Use run() which returns AgentRunResponse, then extract message
      169              coro = self.run(task)
      170              response_obj = await asyncio.wait_for(coro, timeout=self.timeout)
      171  
      172              # Handle direct ChatMessage return (e.g. from mocks or legacy agents)
      173              if isinstance(response_obj, ChatMessage):
      174                  success = True
      175                  return response_obj
      176  
      177              # Extract first message or text
      178              if response_obj.messages:
      179                  response = response_obj.messages[0]
      180              else:
      181                  response = ChatMessage(role=Role.ASSISTANT, text=response_obj.text)
      182  
      183              success = True
      184              return response
      185  
      186          except TimeoutError:
      187              logger.warning(f"Agent {self.name} timed out after {self.timeout}s")
      188              return self._create_timeout_response(self.timeout)
      189          except Exception as e:
      190              logger.error(f"Agent execution failed: {e}")
      191              return ChatMessage(
      192                  role=Role.ASSISTANT,
      193                  text=f"Error: {e}",
      194                  metadata={"status": "error", "error": str(e)},
      195              )
      196          finally:
      197              duration = time.time() - start_time
      198              self.tracker.record_execution(
      199                  agent_name=self.name or "unknown", duration=duration, success=success
      200              )

      202      def _normalize_input_to_text(
      203          self,
      204          messages: str | ChatMessage | list[str] | list[ChatMessage] | None,
      205          thread: AgentThread | None = None,
      206      ) -> str:
      207          """Extracts and formats the conversational history into a single string.
      208  
      209          The last message is treated as the primary prompt, and previous messages
      210          provide context.
      211          """
      212          history: list[str | ChatMessage] = []
      213          if thread:
      214              thread_messages = getattr(thread, "messages", None)
      215              if thread_messages:
      216                  history.extend(list(cast(list[str | ChatMessage], thread_messages)))
      217  
      218          if isinstance(messages, list):
      219              # If messages are provided, they are the source of truth.
      220              history = list(cast(list[str | ChatMessage], messages))
      221          elif messages and (not history or history[-1] != messages):
      222              # If there's a single new message, add it to the history if it's not already there.
      223              history.append(messages)
      224  
      225          if not history:
      226              return ""
      227  
      228          # Format the history into a string
      229          formatted_history = []
      230          for msg in history:
      231              if isinstance(msg, ChatMessage):
      232                  role_name = "User" if msg.role == Role.USER else "Assistant"
      233                  text = getattr(msg, "text", "") or ""
      234                  if text:  # Only include messages with content
      235                      formatted_history.append(f"{role_name}: {text}")
      236              else:
      237                  formatted_history.append(str(msg))
      238  
      239          return "\n\n".join(formatted_history)

      241      async def run(
      242          self,
      243          messages: str | ChatMessage | list[str] | list[ChatMessage] | None = None,
      244          *,
      245          thread: AgentThread | None = None,
      246          **kwargs: Any,
      247      ) -> AgentRunResponse:
      248          """Execute the agent's logic.
      249  
      250          Returns:
      251              AgentRunResponse containing the result.
      252          """
      253          prompt = self._normalize_input_to_text(messages, thread=thread)
      254          agent_kwargs = dict(kwargs)
      255          logger.info(f"Agent {self.name} running with strategy: {self.reasoning_strategy}")
      256  
      257          # Check if DSPy is enabled
      258          if self.enable_dspy:
      259              # Check cache
      260              cached_result = self.cache.get(prompt)
      261              if cached_result:
      262                  logger.info(f"Cache hit for agent {self.name}")
      263                  return AgentRunResponse(
      264                      messages=[ChatMessage(role=Role.ASSISTANT, text=cached_result)],
      265                      additional_properties={"cached": True, "strategy": self.reasoning_strategy},
      266                  )
      267  
      268              try:
      269                  response_text = ""
      270                  if self.reasoning_strategy == "react" and self.react_module:
      271                      # Use ReAct strategy
      272                      result = self.react_module(question=prompt)
      273                      response_text = getattr(result, "answer", str(result))
      274  
      275                  elif self.reasoning_strategy == "program_of_thought" and self.pot_module:
      276                      # Use Program of Thought strategy
      277                      try:
      278                          result = self.pot_module(question=prompt)
      279                      except RuntimeError as exc:
      280                          return await self._handle_pot_failure(
      281                              messages=messages,
      282                              thread=thread,
      283                              agent_kwargs=agent_kwargs,
      284                              error=exc,
      285                          )
      286                      response_text = getattr(result, "answer", str(result))
      287  
      288                  if response_text:
      289                      # Cache the result
      290                      self.cache.set(prompt, response_text)
      291  
      292                      return AgentRunResponse(
      293                          messages=[ChatMessage(role=Role.ASSISTANT, text=response_text)],
      294                          additional_properties={"strategy": self.reasoning_strategy},
      295                      )
      296  
      297                  # If strategy didn't produce specific output (or was CoT/Default), fallback might be better
      298                  # unless we implement explicit CoT module here. For now, if no specific module result,
      299                  # fall through to fallback.
      300  
      301              except Exception as e:
      302                  logger.error(f"DSPy strategy failed for {self.name}: {e}")
      303                  # Fall through to fallback
      304  
      305          # Fallback to standard ChatAgent execution
      306          return await super().run(messages=messages, thread=thread, **kwargs)

      308      async def run_stream(
      309          self,
      310          messages: str | ChatMessage | list[str] | list[ChatMessage] | None = None,
      311          *,
      312          thread: AgentThread | None = None,
      313          **kwargs: Any,
      314      ) -> AsyncIterable[AgentRunResponseUpdate]:
      315          """Run the agent as a stream.
      316  
      317          For DSPy strategies that do not support native streaming (ReAct, PoT),
      318          this method runs the agent to get the full response and then yields it
      319          as a single update. For other cases, it delegates to the parent's
      320          streaming implementation.
      321          """
      322          # DSPy modules (ReAct, PoT) are blocking and do not support streaming.
      323          if self.enable_dspy and self.reasoning_strategy in ("react", "program_of_thought"):
      324              logger.warning(
      325                  f"Agent {self.name} with reasoning strategy '{self.reasoning_strategy}' "
      326                  "does not support true streaming. The full response will be sent at once."
      327              )
      328              # Execute the non-streaming run method to get the complete response.
      329              response = await self.run(messages=messages, thread=thread, **kwargs)
      330  
      331              # Determine the role for the update.
      332              response_role = Role.ASSISTANT
      333              if response.messages:
      334                  response_role = response.messages[0].role
      335  
      336              # Yield a single update containing the full response.
      337              yield AgentRunResponseUpdate(
      338                  text=response.text,
      339                  messages=response.messages,
      340                  additional_properties=response.additional_properties,
      341                  role=response_role,
      342              )
      343              return
      344  
      345          # For streaming-compatible strategies, delegate to the parent implementation.
      346          async for update in super().run_stream(messages=messages, thread=thread, **kwargs):
      347              yield update

      349      async def _handle_pot_failure(

      351          messages: Any,
      352          thread: AgentThread | None,
      353          agent_kwargs: dict[str, Any],
      354          error: Exception,

      356          """
      357          Invoke the base ChatAgent as a fallback and attach a Program of Thought (PoT) failure note to the response.
      358  
      359          Parameters:
      360              messages (Any): Original input messages passed to the fallback run.
      361              thread (AgentThread | None): Optional thread context to pass to the fallback run.
      362              agent_kwargs (dict[str, Any]): Additional keyword arguments forwarded to the base agent's run method.
      363              error (Exception): The PoT failure that triggered the fallback; used to build the user-facing note.
      364  
      365          Returns:
      366              AgentRunResponse: The fallback response with the PoT note prepended to the first message text (or to the response text if no messages), and with `additional_properties` extended to include `strategy` and `pot_error`.
      367          """

      369          note = self._build_pot_error_note(error)
      370          logger.warning("Program of Thought failed for %s: %s", self.name, note)
      371  
      372          fallback_response = await super().run(messages=messages, thread=thread, **agent_kwargs)
      373          # Prepend note to the first message text
      374          # ChatMessage.text is read-only, so create a new ChatMessage instead of modifying
      375          if fallback_response.messages:
      376              first_msg = fallback_response.messages[0]
      377              original_text = getattr(first_msg, "text", "")
      378              updated_text = self._apply_note_to_text(original_text, note)
      379  
      380              # Create a new ChatMessage with the updated text while preserving all other fields.
      381              # Prefer Pydantic model cloning APIs if available.
      382              update_patch: dict[str, Any] = {"text": updated_text}
      383              model_fields = getattr(type(first_msg), "model_fields", None)
      384              v1_fields = getattr(type(first_msg), "__fields__", None)
      385              if (isinstance(model_fields, dict) and "content" in model_fields) or (
      386                  isinstance(v1_fields, dict) and "content" in v1_fields
      387              ):
      388                  update_patch["content"] = updated_text
      389  
      390              updated_msg: ChatMessage
      391              model_copy = getattr(first_msg, "model_copy", None)
      392              if callable(model_copy):
      393                  updated_msg = cast(ChatMessage, model_copy(update=update_patch))
      394              else:
      395                  v1_copy = getattr(first_msg, "copy", None)
      396                  if callable(v1_copy):
      397                      updated_msg = cast(ChatMessage, v1_copy(update=update_patch))
      398                  elif dataclasses.is_dataclass(first_msg):
      399                      # Only pass fields that exist on the dataclass.
      400                      patch = {k: v for k, v in update_patch.items() if hasattr(first_msg, k)}
      401                      updated_msg = cast(ChatMessage, dataclasses.replace(first_msg, **patch))
      402                  else:
      403                      # Last-resort fallback: manually reconstruct commonly used fields.
      404                      # (This may drop unknown fields on non-pydantic, non-dataclass message types.)
      405                      updated_msg = ChatMessage(
      406                          role=first_msg.role,
      407                          text=updated_text,
      408                          metadata=getattr(first_msg, "metadata", None),
      409                          additional_properties=getattr(first_msg, "additional_properties", None),
      410                      )
      411              # Replace the first message with the updated one
      412              # Note: We reassign messages assuming it's mutable. If AgentRunResponse.messages
      413              # becomes immutable, construct a new AgentRunResponse instead.
      414              fallback_response.messages = [updated_msg, *fallback_response.messages[1:]]
      415          else:
      416              fallback_response_text = getattr(fallback_response, "text", "")
      417              updated_text = self._apply_note_to_text(fallback_response_text, note)
      418              # Create a new message if there are no messages
      419              fallback_response.messages = [ChatMessage(role=Role.ASSISTANT, text=updated_text)]
      420  
      421          existing_props = dict(fallback_response.additional_properties or {})
      422          existing_props.update(
      423              {
      424                  "strategy": self.reasoning_strategy,
      425                  "pot_error": note,
      426              }
      427          )
      428          fallback_response.additional_properties = existing_props
      429          return fallback_response
