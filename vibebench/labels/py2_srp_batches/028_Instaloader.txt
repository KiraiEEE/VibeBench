      334      def download_pic(self, filename: str, url: str, mtime: datetime,
      335                       filename_suffix: Optional[str] = None, _attempt: int = 1) -> bool:
      336          """Downloads and saves picture with given url under given directory with given timestamp.
      337          Returns true, if file was actually downloaded, i.e. updated."""
      338          if filename_suffix is not None:
      339              filename += '_' + filename_suffix
      340          urlmatch = re.search('\\.[a-z0-9]*\\?', url)
      341          file_extension = url[-3:] if urlmatch is None else urlmatch.group(0)[1:-1]
      342          nominal_filename = filename + '.' + file_extension
      343          if os.path.isfile(nominal_filename):
      344              self.context.log(nominal_filename + ' exists', end=' ', flush=True)
      345              return False
      346          resp = self.context.get_raw(url)
      347          if 'Content-Type' in resp.headers and resp.headers['Content-Type']:
      348              header_extension = '.' + resp.headers['Content-Type'].split(';')[0].split('/')[-1]
      349              header_extension = header_extension.lower().replace('jpeg', 'jpg')
      350              filename += header_extension
      351          else:
      352              filename = nominal_filename
      353          if filename != nominal_filename and os.path.isfile(filename):
      354              self.context.log(filename + ' exists', end=' ', flush=True)
      355              return False
      356          self.context.write_raw(resp, filename)
      357          os.utime(filename, (datetime.now().timestamp(), mtime.timestamp()))
      358          return True
