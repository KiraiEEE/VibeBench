      150      /**
      151       * Returns the roster for the given username.
      152       *
      153       * @param username the username to search for.
      154       * @return the roster associated with the ID.
      155       * @throws org.jivesoftware.openfire.user.UserNotFoundException if the ID does not correspond
      156       *         to a known entity on the server.
      157       */
      158      public Roster getRoster(String username) throws UserNotFoundException {
      159          Roster roster = rosterCache.get(username);
      160          if (roster == null) {
      161              final Lock lock = rosterCache.getLock(username);
      162              lock.lock();
      163              try {
      164                  roster = rosterCache.get(username);
      165                  if (roster == null) {
      166                      // Not in cache so load a new one:
      167                      roster = new Roster(username);
      168                      rosterCache.put(username, roster);
      169                  }
      170              } finally {
      171                  lock.unlock();
      172              }
      173          }
      174          return roster;
      175      }

      177      /**
      178       * Removes the entire roster of a given user. This is necessary when a user
      179       * account is being deleted from the server.
      180       *
      181       * @param user the user.
      182       */
      183      public void deleteRoster(JID user) {
      184          if (!server.isLocal(user)) {
      185              // Ignore request if user is not a local user
      186              return;
      187          }
      188          try {
      189              String username = user.getNode();
      190              // Get the roster of the deleted user
      191              Roster roster = getRoster(username);
      192              // Remove each roster item from the user's roster
      193              for (RosterItem item : roster.getRosterItems()) {
      194                  try {
      195                      roster.deleteRosterItem(item.getJid(), false);
      196                  }
      197                  catch (SharedGroupException e) {
      198                      // Do nothing. We shouldn't have this exception since we disabled the checkings
      199                      Log.warn( "Unexpected exception while deleting roster of user '{}' .", user, e );
      200                  }
      201              }
      202              // Remove the cached roster from memory
      203              rosterCache.remove(username);
      204  
      205              // Get the rosters that have a reference to the deleted user
      206              Iterator<String> usernames = provider.getUsernames(user.toBareJID());
      207              while (usernames.hasNext()) {
      208                  username = usernames.next();
      209                  try {
      210                      // Get the roster that has a reference to the deleted user
      211                      roster = getRoster(username);
      212                      // Remove the deleted user reference from this roster
      213                      roster.deleteRosterItem(user, false);
      214                  }
      215                  catch (SharedGroupException e) {
      216                      // Do nothing. We shouldn't have this exception since we disabled the checkings
      217                      Log.warn( "Unexpected exception while deleting roster of user '{}' .", user, e );
      218                  }
      219                  catch (UserNotFoundException e) {
      220                      // Deleted user had user that no longer exists on their roster. Ignore and move on.
      221                  }
