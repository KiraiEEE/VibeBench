       84      }
       85      for (Map.Entry<String, String> e : users.entrySet()) {
       86        String v = e.getValue();
       87        if (v == null) {
       88          log.warn("user has no password {}", e.getKey());
       89          continue;
       90        }
       91        credentials.put(e.getKey(), v);
       92      }
       93    }

       95    @Override
       96    public boolean authenticate(String username, String password) {
       97      String cred = credentials.get(username);
       98      if (cred == null || cred.isEmpty()) return false;
       99      cred = cred.trim();
      100      String salt = null;
      101      if (cred.contains(" ")) {
      102        String[] ss = cred.split(" ");
      103        if (ss.length > 1 && !ss[1].isEmpty()) {
      104          salt = ss[1];
      105          cred = ss[0];
      106        }
      107      }
      108      return cred.equals(sha256(password, salt));
      109    }

      116    public static String sha256(String password, String saltKey) {
      117      MessageDigest digest;
      118      try {
      119        digest = MessageDigest.getInstance("SHA-256");
      120      } catch (NoSuchAlgorithmException e) {
      121        log.error("Cannot find algorithm ", e);
      122        return null; // should not happen
      123      }
      124      if (saltKey != null) {
      125        digest.reset();
      126        digest.update(Base64.getDecoder().decode(saltKey));
      127      }
      128  
      129      byte[] btPass = digest.digest(password.getBytes(StandardCharsets.UTF_8));
      130      digest.reset();
      131      btPass = digest.digest(btPass);
      132      return Base64.getEncoder().encodeToString(btPass);
      133    }

      135    @Override
      136    public Map<String, Object> edit(Map<String, Object> latestConf, List<CommandOperation> commands) {
      137      for (CommandOperation cmd : commands) {
      138        if (!supported_ops.contains(cmd.name)) {
      139          cmd.unknownOperation();
      140          return null;
      141        }
      142        if (cmd.hasError()) return null;
      143        if ("delete-user".equals(cmd.name)) {
      144          List<String> names = cmd.getStrs("");
      145          Map<?, ?> map = (Map<?, ?>) latestConf.get("credentials");
      146          if (map == null || !map.keySet().containsAll(names)) {
      147            cmd.addError("No such user(s) " + names);
      148            return null;
      149          }
      150          for (String name : names) {
      151            if (map.containsKey(name)) {
      152              if (map.size() == 1) {
      153                cmd.addError(CANNOT_DELETE_LAST_USER_ERROR);
      154                return null;
      155              }
      156            }
      157            map.remove(name);
      158          }
      159          return latestConf;
      160        }
      161        if ("set-user".equals(cmd.name)) {
      162          Map<String, Object> map = getMapValue(latestConf, "credentials");
      163          Map<String, Object> kv = cmd.getDataMap();
      164          for (Map.Entry<String, Object> e : kv.entrySet()) {
      165            if (e.getKey() == null || e.getValue() == null) {
      166              cmd.addError("name and password must be non-null");
      167              return null;
      168            }
      169            putUser(e.getKey(), String.valueOf(e.getValue()), map);
      170          }
      171        }
      172      }
      173      return latestConf;
      174    }
