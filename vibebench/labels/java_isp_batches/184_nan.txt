       38      static void findSeekEstimate(
       39              long timestamp,
       40              int frameCount,
       41              DirectIntList framePartitionIndexes,
       42              LongList partitionCeilings,
       43              LongList partitionTimestamps,
       44              TimeFrame timeFrame
       45      ) {
       46          if (frameCount == 0) {
       47              timeFrame.ofEstimate(-1, Long.MIN_VALUE, Long.MIN_VALUE);
       48              return;
       49          }
       50          int lo = 0, hi = frameCount - 1, result = -1;
       51          while (lo <= hi) {
       52              int mid = (lo + hi) >>> 1;
       53              int partIdx = framePartitionIndexes.get(mid);
       54              if (partitionCeilings.getQuick(partIdx) <= timestamp) {
       55                  result = mid;
       56                  lo = mid + 1;
       57              } else {
       58                  hi = mid - 1;
       59              }
       60          }
       61          if (result >= 0) {
       62              int partIdx = framePartitionIndexes.get(result);
       63              timeFrame.ofEstimate(result, partitionTimestamps.getQuick(partIdx), partitionCeilings.getQuick(partIdx));
       64          } else {
       65              timeFrame.ofEstimate(-1, Long.MIN_VALUE, Long.MIN_VALUE);
       66          }
       67      }
