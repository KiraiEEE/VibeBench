      195          AssertUtils.assertNotEmpty("requestId不能为空", requestId);
      196          // 从缓存中获取对应的SseEmitter
      197          SseEmitter emitter = AiragLocalCache.get(AiragConsts.CACHE_TYPE_SSE, requestId);
      198          if (emitter != null) {
      199              closeSSE(emitter, new EventData(requestId, null, EventData.EVENT_MESSAGE_END));
      200              return Result.ok("会话已成功终止");

      206      /**
      207       * 关闭sse
      208       *
      209       * @param emitter
      210       * @param eventData
      211       * @throws IOException
      212       * @author chenrui
      213       * @date 2025/2/27 15:56
      214       */
      215      private static void closeSSE(SseEmitter emitter, EventData eventData) {
      216          AssertUtils.assertNotEmpty("请求id不能为空", eventData);
      217          if (null == emitter) {
      218              log.warn("会话已关闭");
      219              return;
      220          }
      221          try {
      222              // 发送完成事件
      223              emitter.send(SseEmitter.event().data(eventData));
      224          } catch (Exception e) {
      225              if(!e.getMessage().contains("ResponseBodyEmitter has already completed")){
      226                  log.error("终止会话时发生错误", e);
      227              }
      228              try {
      229                  // 防止异常冒泡
      230                  emitter.completeWithError(e);
      231              } catch (Exception ignore) {}
      232          } finally {
      233              // 从缓存中移除emitter
      234              AiragLocalCache.remove(AiragConsts.CACHE_TYPE_SSE, eventData.getRequestId());
      235              // 关闭emitter
      236              try {
      237                  emitter.complete();
      238              } catch (Exception ignore) {}
      239          }
      240      }
