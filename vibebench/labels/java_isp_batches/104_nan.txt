       92      public void createKnowledgeSegmentBySplitContent(Long documentId, String content) {
       93          // 1. 校验
       94          AiKnowledgeDocumentDO documentDO = knowledgeDocumentService.validateKnowledgeDocumentExists(documentId);
       95          AiKnowledgeDO knowledgeDO = knowledgeService.validateKnowledgeExists(documentDO.getKnowledgeId());
       96          VectorStore vectorStore = getVectorStoreById(knowledgeDO);
       97  
       98          // 2. 文档切片
       99          List<Document> documentSegments = splitContentByToken(content, documentDO.getSegmentMaxTokens());
      100  
      101          // 3.1 存储切片
      102          List<AiKnowledgeSegmentDO> segmentDOs = convertList(documentSegments, segment -> {
      103              if (StrUtil.isEmpty(segment.getText())) {
      104                  return null;
      105              }
      106              return new AiKnowledgeSegmentDO().setKnowledgeId(documentDO.getKnowledgeId()).setDocumentId(documentId)
      107                      .setContent(segment.getText()).setContentLength(segment.getText().length())
      108                      .setVectorId(AiKnowledgeSegmentDO.VECTOR_ID_EMPTY)
      109                      .setTokens(tokenCountEstimator.estimate(segment.getText()))
      110                      .setStatus(CommonStatusEnum.ENABLE.getStatus());
      111          });
      112          segmentMapper.insertBatch(segmentDOs);
      113          // 3.2 切片向量化
      114          for (int i = 0; i < documentSegments.size(); i++) {
      115              Document segment = documentSegments.get(i);
      116              AiKnowledgeSegmentDO segmentDO = segmentDOs.get(i);
      117              writeVectorStore(vectorStore, segmentDO, segment);
      118          }
