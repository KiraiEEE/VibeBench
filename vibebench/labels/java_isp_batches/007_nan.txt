      128    /** {@inheritDoc} */
      129    @Override
      130    public double getHydrateEquilibriumTemperature() {
      131      if (!getFluid().getPhase(0).hasComponent("water")) {
      132        logger.info("no hydrate: no water in stream: " + name);
      133        return 0.0;
      134      }
      135      try {
      136        SystemInterface copySystem = getFluid().clone();
      137        copySystem.setHydrateCheck(true);
      138        ThermodynamicOperations thermoOps = new ThermodynamicOperations(copySystem);
      139        thermoOps.hydrateFormationTemperature();
      140        return copySystem.getTemperature();
      141      } catch (Exception ex) {
      142        logger.error(ex.getMessage(), ex);
      143      }
      144      return 0.0;
      145    }

      147    /**
      148     * <p>
      149     * getSolidFormationTemperature.
      150     * </p>
      151     *
      152     * @param solidName a {@link java.lang.String} object
      153     * @return a double
      154     */
      155    public double getSolidFormationTemperature(String solidName) {
      156      SystemInterface copySystem = getFluid().clone();
      157  
      158      try {
      159        if (solidName.equals("hydrate")) {
      160          copySystem.setHydrateCheck(true);
      161          ThermodynamicOperations thermoOps = new ThermodynamicOperations(copySystem);
      162          thermoOps.hydrateFormationTemperature();
      163        } else {
      164          copySystem.setSolidPhaseCheck(false);
      165          copySystem.setSolidPhaseCheck(solidName);
      166          ThermodynamicOperations thermoOps = new ThermodynamicOperations(copySystem);
      167          thermoOps.freezingPointTemperatureFlash();
      168        }
      169        return copySystem.getTemperature();
      170      } catch (Exception ex) {
      171        logger.error(ex.getMessage(), ex);
      172      }
      173      return 0.0;
      174    }
