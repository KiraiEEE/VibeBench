       33  @Extension("random")
       34  public class RandomLoadBalancer extends AbstractLoadBalancer {
       35  
       36      /**
       37       * 随机
       38       */
       39      private final Random random = new Random();
       40  
       41      /**
       42       * 构造函数
       43       *
       44       * @param consumerBootstrap 服务消费者配置
       45       */
       46      public RandomLoadBalancer(ConsumerBootstrap consumerBootstrap) {
       47          super(consumerBootstrap);
       48      }
       49  
       50      @Override
       51      public ProviderInfo doSelect(SofaRequest invocation, List<ProviderInfo> providerInfos) {
       52          ProviderInfo providerInfo = null;
       53          int size = providerInfos.size(); // 总个数
       54          int totalWeight = 0; // 总权重
       55          boolean isWeightSame = true; // 权重是否都一样
       56          for (int i = 0; i < size; i++) {
       57              int weight = getWeight(providerInfos.get(i));
       58              totalWeight += weight; // 累计总权重
       59              if (isWeightSame && i > 0 && weight != getWeight(providerInfos.get(i - 1))) {
       60                  isWeightSame = false; // 计算所有权重是否一样
       61              }
       62          }
       63          if (totalWeight > 0 && !isWeightSame) {
       64              // 如果权重不相同且权重大于0则按总权重数随机
       65              int offset = random.nextInt(totalWeight);
       66              // 并确定随机值落在哪个片断上
       67              for (int i = 0; i < size; i++) {
       68                  offset -= getWeight(providerInfos.get(i));
       69                  if (offset < 0) {
       70                      providerInfo = providerInfos.get(i);
       71                      break;
       72                  }
       73              }
       74          } else {
       75              // 如果权重相同或权重为0则均等随机
       76              providerInfo = providerInfos.get(random.nextInt(size));
       77          }
       78          return providerInfo;
       79      }
       80  }
