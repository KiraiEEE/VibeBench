      161      /**
      162       * Fixes an edge case where types might be missed during traversal due to replaced type references.
      163       * <p>
      164       * The problem: During schema construction or transformation, a field's type might initially
      165       * be a {@link GraphQLTypeReference} (a placeholder). Later, this reference gets replaced with the
      166       * actual type instance. However, the schema traverser uses {@code getChildrenWithTypeReferences()}
      167       * to discover children, which returns the original type references, not the replaced strong references.
      168       * <p>
      169       * The scenario:
      170       * <ol>
      171       *   <li>Field {@code foo} has type reference to {@code Bar}</li>
      172       *   <li>During schema editing, the reference is replaced with the actual {@code Bar} type</li>
      173       *   <li>Further edits remove all other paths to {@code Bar}</li>
      174       *   <li>Now the only way to reach {@code Bar} is via the replaced reference in {@code foo}</li>
      175       *   <li>But the traverser still sees the original type reference as the child, so {@code Bar}
      176       *       is never visited as a child node</li>
      177       * </ol>
      178       * <p>
      179       * The fix: During traversal, we also capture types directly from fields/arguments/inputs
      180       * (in {@link #indirectStrongReferences}). After traversal, we merge any types that were captured
      181       * this way but weren't found through normal traversal. Additionally, we traverse each newly
      182       * discovered indirect strong reference to collect any types it references, recursively handling
      183       * cases where indirect strong references are nested within other indirect strong references.
      184       * <p>
      185       * We reuse the same visitor instance to ensure duplicate type detection works correctly
      186       * across all traversals.
      187       *
      188       * @param visitedTypes the types collected through normal traversal
      189       *
      190       * @return the fixed map including any dangling replaced types
      191       */
      192      private Map<String, GraphQLNamedType> fixDanglingReplacedTypes(Map<String, GraphQLNamedType> visitedTypes) {
      193          // Collect indirect strong references that are not yet in the visited types
      194          List<GraphQLNamedType> newlyDiscoveredTypes = new ArrayList<>();
      195          for (GraphQLNamedType indirectStrongReference : indirectStrongReferences.values()) {
      196              String typeName = indirectStrongReference.getName();
      197              if (!visitedTypes.containsKey(typeName)) {
      198                  visitedTypes.put(typeName, indirectStrongReference);
      199                  newlyDiscoveredTypes.add(indirectStrongReference);
      200              }
      201          }
      202  
      203          // For each newly discovered type, traverse it to collect any types it references
      204          // We reuse this visitor instance to ensure duplicate type detection works correctly
      205          if (!newlyDiscoveredTypes.isEmpty()) {
      206              // Clear indirect strong references before traversing to capture new ones
      207              indirectStrongReferences.clear();
      208  
      209              SchemaTraverser traverser = new SchemaTraverser(
      210                      schemaElement -> schemaElement.getChildrenWithTypeReferences().getChildrenAsList());
      211              traverser.depthFirst(this, newlyDiscoveredTypes);
