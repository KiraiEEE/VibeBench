       80          notNull(factory, "The factory of bean to register cannot be null.");
       81          LockUtils.synchronize(this.monitor.writeLock(), () -> {
       82              Set<String> existing = this.factories.stream()
       83                      .map(current -> CollectionUtils.intersect(names(current), names(factory)))
       84                      .filter(CollectionUtils::isNotEmpty)
       85                      .flatMap(Collection::stream)
       86                      .collect(Collectors.toSet());
       87              if (existing.isEmpty()) {
       88                  this.factories.add(factory);
       89              } else {
       90                  throw new BeanDefinitionException(StringUtils.format(
       91                          "Beans with the same name already exists. [names={0}]",
       92                          existing));
       93              }
       94          });
       95          BeanMetadata metadata = factory.metadata();
       96          this.beanRegisteredObservers.forEach(observer -> observer.onBeanRegistered(metadata));
       97          return metadata;
