      154  	public static MusicClue forText(String text)
      155  	{
      156  		final Matcher m = SONG_PATTERN.matcher(text);
      157  		if (m.find())
      158  		{
      159  			final String song = m.group(1);
      160  
      161  			// Check for existing MusicClue, otherwise create new
      162  			MusicClue clue = forSong(song);
      163  			if (clue != null)
      164  			{
      165  				return clue;
      166  			}
      167  			else
      168  			{
      169  				return new MusicClue(-1, song);
      170  			}
      171  		}
      172  		return null;
      173  	}
