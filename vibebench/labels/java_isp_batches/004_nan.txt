      133          if (attachedNode == null) {
      134              return;
      135          }
      136          Node attachParent = attachedNode.getParent();
      137          if (attachParent == null || targetGeometry == null
      138                  || targetGeometry.getParent() == attachParent
      139                  && targetGeometry.getLocalTransform().isIdentity()) {
      140              /*
      141               * The animated meshes are in the same coordinate system as the
      142               * attachments node: no further transforms are needed.
      143               */
      144              attachedNode.setLocalTransform(getModelTransform());
      145  
      146          } else {
      147              Spatial loopSpatial = targetGeometry;
      148              Transform combined = getModelTransform().clone();
      149              /*
      150               * Climb the scene graph applying local transforms until the
      151               * attachments node's parent is reached.
      152               */
      153              while (loopSpatial != attachParent && loopSpatial != null) {
      154                  Transform localTransform = loopSpatial.getLocalTransform();
      155                  combined.combineWithParent(localTransform);
      156                  loopSpatial = loopSpatial.getParent();
      157              }
      158              attachedNode.setLocalTransform(combined);
      159          }
      160      }
