       99      /**
      100       * Constructs a new {@code AbstractTimestampLongConverter} with the specified
      101       * zone ID and time unit.
      102       *
      103       * @param zoneId   The string representation of the {@link ZoneId} (for
      104       *                 example "UTC" or "Europe/London") used when a parsed date
      105       *                 does not contain its own offset. All internal long values
      106       *                 are treated as UTC based.
      107       * @param timeUnit The {@link TimeUnit} defining the precision of the long
      108       *                 timestamp values handled by this converter.
      109       */
      110      protected AbstractTimestampLongConverter(String zoneId, TimeUnit timeUnit) {
      111          this.zoneId = ZoneId.of(zoneId);
      112          this.writingUtcDatesWithNoSuffix = this.zoneId.equals(UTC);
      113          this.amountPerSecond = timeUnit.convert(1, TimeUnit.SECONDS);
      114          this.nanosPerAmount = TimeUnit.NANOSECONDS.convert(1, timeUnit);
      115          this.formatterForParsing = createFormatter();
      116          if (writingUtcDatesWithNoSuffix) {
      117              this.formatterForFormatting = createFormatterWithNoZoneSuffix();
      118          } else {
      119              this.formatterForFormatting = formatterForParsing;
      120          }
      121      }

      123      /**
      124       * Parses the provided text and converts it into a long timestamp.
      125       * The input may be an ISO-8601 date, a date-time with or without a zone, or
      126       * a plain numeric timestamp. When no zone is present, the converter's
      127       * {@link ZoneId} is assumed.
      128       *
      129       * @param text The character sequence representing a date, timestamp or
      130       *             numeric value to be parsed into a UTC long in this
      131       *             converter's {@link TimeUnit}.
      132       * @return a long value representing the parsed timestamp
      133       */
      134      @Override
      135      public long parse(CharSequence text) {
      136          if (text == null || text.length() == 0)
      137              return 0;
      138          try {
      139              if (text.length() > 4 && text.charAt(4) == '/')
      140                  text = text.toString().replace('/', '-');
      141              final TemporalAccessor parse = formatterForParsing.parse(text);
      142              if (parse.query(TemporalQueries.zoneId()) != null) {
      143                  return parseFormattedDate(ZonedDateTime.from(parse).withZoneSameInstant(UTC));
      144              } else {
      145                  return parseFormattedDate(LocalDateTime.from(parse).atZone(zoneId).withZoneSameInstant(UTC));
      146              }
      147          } catch (DateTimeParseException dtpe) {
      148              try {
      149                  return parseTimestamp(Long.parseLong(text.toString()), text);
      150              } catch (NumberFormatException e) {
      151                  throw dtpe;
      152              }
      153          }
      154      }

      156      /**
      157       * Interpret a formatted date that has already been normalised to UTC.

      175      protected abstract long parseTimestamp(long value, CharSequence text);

      177      /**
      178       * Constructs a {@code DateTimeFormatter} for parsing timestamps. The formatter includes fraction parsing
      179       * and optional offset parsing.
      180       *
      181       * @return a newly constructed {@code DateTimeFormatter}
      182       */
      183      private DateTimeFormatter createFormatter() {
      184          final DateTimeFormatterBuilder builder = new DateTimeFormatterBuilder()
      185                  .appendPattern("yyyy-MM-dd'T'HH:mm:ss");
      186          appendFraction(builder);
      187          builder.optionalStart().appendOffsetId().optionalEnd();
      188          return builder.toFormatter();
      189      }
      190  
      191      /**
      192       * Constructs a {@code DateTimeFormatter} for parsing timestamps without a 'Z' suffix. The formatter includes fraction parsing.
      193       *
      194       * @return a newly constructed {@code DateTimeFormatter}
      195       */
      196      private DateTimeFormatter createFormatterWithNoZoneSuffix() {
      197          final DateTimeFormatterBuilder builder = new DateTimeFormatterBuilder()
      198                  .appendPattern("yyyy-MM-dd'T'HH:mm:ss");
      199          appendFraction(builder);
      200          return builder.toFormatter();
      201      }

      203      /**
      204       * Append the fraction-of-second pattern to the formatter builder.
      205       *
      206       * @param builder The {@link DateTimeFormatterBuilder} after the basic date
      207       *                and time pattern has been added.
      208       */
      209      protected abstract void appendFraction(DateTimeFormatterBuilder builder);
      210  
      211      /**
      212       * Format the timestamp value and append it to the supplied {@link Appendable}.
      213       *
      214       * @param text  The destination for the formatted date-time, such as a
      215       *              {@link StringBuilder} or {@link Bytes} instance.
      216       * @param value The UTC timestamp in this converter's {@link TimeUnit}.
      217       */
      218      public void append(Appendable text, long value) {
      219          if (value <= 0) {
      220              AppendableUtil.append(text, value);
      221              return;
      222          }
      223          LocalDateTime ldt = LocalDateTime.ofEpochSecond(
      224                  value / amountPerSecond,
      225                  (int) (value % amountPerSecond * nanosPerAmount),
      226                  ZoneOffset.UTC);
      227          if (writingUtcDatesWithNoSuffix) {
      228              formatterForFormatting.formatTo(ldt, text);
      229          } else {
      230              formatterForFormatting.formatTo(ZonedDateTime.of(ldt, UTC)
      231                      .withZoneSameInstant(zoneId), text);
      232          }
      233      }
      234  
      235      /**

      237       *
      238       * @param text  The {@link StringBuilder} to receive the formatted value.
      239       * @param value The UTC timestamp in this converter's {@link TimeUnit}.
      240       */
      241      @Override
      242      public void append(StringBuilder text, long value) {
      243          append((Appendable) text, value);
      244      }
      245  
      246      /**
      247       * Convenience overload for appending to a {@link Bytes} instance.
      248       *
      249       * @param bytes The {@link Bytes} sink for the formatted value.
      250       * @param value The UTC timestamp in this converter's {@link TimeUnit}.
      251       */
      252      @Override
      253      public void append(Bytes<?> bytes, long value) {
      254          append((Appendable) bytes, value);
      255      }
