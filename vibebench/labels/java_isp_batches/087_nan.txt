      204      @Override
      205      public boolean tryToCharge(@NonNull final AccountID accountId, final long amount) {
      206          requireNonNull(accountId);
      207          if (amount < 0) {
      208              throw new IllegalArgumentException("Cannot charge negative amount " + amount);
      209          }
      210          if (feeCharging.bypassForExtraHandlerCharges()) {
      211              return feeAccumulator.chargeFee(accountId, amount, null).networkFee() == amount;
      212          } else {
      213              return feeCharging
      214                              .charge(
      215                                      accountId,
      216                                      this,
      217                                      ValidationResult.newSuccess(creatorInfo.accountId()),
      218                                      new Fees(0, 0, amount))
      219                              .totalFee()
      220                      == amount;
      221          }
      222      }

      224      @Override
      225      public void refundBestEffort(@NonNull final AccountID accountId, final long amount) {
      226          requireNonNull(accountId);
      227          if (amount < 0) {
      228              throw new IllegalArgumentException("Cannot refund negative amount " + amount);
      229          }
      230          if (feeCharging.bypassForExtraHandlerCharges()) {
      231              feeAccumulator.refundFee(accountId, amount);
      232          } else {
      233              feeCharging.refund(accountId, this, new Fees(0, amount, 0));
      234          }
      235      }
