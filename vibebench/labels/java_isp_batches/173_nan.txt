      149    @Override
      150    public boolean isAccessTokenExpired() {
      151      try (Jedis jedis = this.jedisPool.getResource()) {
      152        String expiresTimeStr = jedis.get(ACCESS_TOKEN_EXPIRES_TIME_KEY);
      153  
      154        if (expiresTimeStr != null) {
      155          return System.currentTimeMillis() > Long.parseLong(expiresTimeStr);
      156        }
      157  
      158        return true;
      159  
      160      }
      161    }

      167      }
      168    }

      175    @Override
      176    public synchronized void updateAccessToken(String accessToken, int expiresInSeconds) {
      177      try (Jedis jedis = this.jedisPool.getResource()) {
      178        jedis.set(ACCESS_TOKEN_KEY, accessToken);
      179  
      180        jedis.set(ACCESS_TOKEN_EXPIRES_TIME_KEY,
      181          (System.currentTimeMillis() + (expiresInSeconds - 200) * 1000L) + "");
      182      }
