       51      /**
       52       * Constructs an unchecked view over {@code underlyingBytes}. The positions
       53       * and limits of the wrapped bytes are inherited.
       54       *
       55       * @param underlyingBytes the Bytes instance to wrap
       56       * @throws ClosedIllegalStateException    if the resource has been released or closed
       57       * @throws ThreadingIllegalStateException if accessed by multiple threads in an unsafe way
       58       */
       59      @SuppressWarnings("this-escape")
       60      public UncheckedBytes(@NotNull Bytes<?> underlyingBytes)
       61              throws IllegalStateException {
       62          super(uncheckedCast(requireNonNull(underlyingBytes.bytesStore())),
       63                  underlyingBytes.writePosition(),
       64                  Math.min(underlyingBytes.writeLimit(), underlyingBytes.realCapacity()));
       65          this.underlyingBytes = underlyingBytes;
       66          readPosition(underlyingBytes.readPosition());
       67          if (writeLimit > capacity())
       68              writeLimit(capacity());
       69      }

       71      /**
       72       * Replaces the wrapped {@link Bytes} instance. The current store is
       73       * released and the new one reserved.
       74       *
       75       * @param bytes the new underlying bytes
       76       * @throws ClosedIllegalStateException    if the resource has been released or closed
       77       * @throws ThreadingIllegalStateException if accessed by multiple threads in an unsafe way
       78       */
       79      public void setBytes(@NotNull Bytes<?> bytes)
       80              throws IllegalStateException {
       81          requireNonNull(bytes);
       82          final BytesStore<?, ?> underlying = bytes.bytesStore();
       83          if (bytesStore != underlying) {
       84              bytesStore.release(this);
       85              this.bytesStore(uncheckedCast(underlying));
       86              bytesStore.reserve(this);
       87          }
       88          readPosition(bytes.readPosition());
       89          this.uncheckedWritePosition(bytes.writePosition());
       90          this.writeLimit = bytes.writeLimit();
       91  
       92          this.underlyingBytes = bytes;
       93      }
