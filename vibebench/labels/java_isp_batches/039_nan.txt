      147          private Supplier<CompletableFuture<DeferredFragmentCall.FieldWithExecutionResult>> createResultSupplier(
      148                  MergedField currentField,

      153  
      154              ExecutionStrategyParameters executionStrategyParameters = parameters.transform(builder ->
      155                      {
      156                          MergedSelectionSet mergedSelectionSet = MergedSelectionSet.newMergedSelectionSet().subFields(fields).build();
      157                          ResultPath path = parameters.getPath().segment(currentField.getResultKey());
      158                          builder.deferredCallContext(alternativeCallContext)
      159                                  .field(currentField)

      179  
      180                  Instrumentation instrumentation = executionContext.getInstrumentation();

      182                  InstrumentationFieldParameters fieldParameters = new InstrumentationFieldParameters(executionContext, executionStepInfo);
      183                  InstrumentationContext<Object> deferredFieldCtx = nonNullCtx(instrumentation.beginDeferredField(fieldParameters, executionContext.getInstrumentationState()));
      184  
      185                  CompletableFuture<FieldValueInfo> fieldValueResult = resolveFieldWithInfoFn.apply(this.executionContext, executionStrategyParameters);
      186                  executionContext.getDataLoaderDispatcherStrategy().deferFieldFetched(executionStrategyParameters);
      187  
      188  
      189                  deferredFieldCtx.onDispatched();
      190  
      191                  fieldValueResult.whenComplete((fieldValueInfo, throwable) -> {
      192                      this.executionContext.getDataLoaderDispatcherStrategy().deferredOnFieldValue(currentField.getResultKey(), fieldValueInfo, throwable, executionStrategyParameters);

      194                  });
      195  
      196  
      197                  CompletableFuture<ExecutionResult> executionResultCF = fieldValueResult
      198                          .thenCompose(fvi -> fvi
      199                                  .getFieldValueFuture()
      200                                  .thenApply(fv -> ExecutionResultImpl.newExecutionResult().data(fv).build())
      201                          );
      202  
      203                  return executionResultCF
      204                          .thenApply(executionResult ->
      205                                  new DeferredFragmentCall.FieldWithExecutionResult(currentField.getResultKey(), executionResult)
      206                          );
