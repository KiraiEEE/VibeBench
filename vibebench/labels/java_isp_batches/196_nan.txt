      102     * This method is used during the object construction, but also to
      103     * reset/recreate the encapsulated CloudBlobClient object with new
      104     * credentials (after SAS token expiration)
      105     * @param stage   The stage information that the client will operate on
      106     * @param encMat  The encryption material
      107     *                required to decrypt/encrypt content in stage
      108     * @throws IllegalArgumentException when invalid credentials are used
      109     */
      110    private void setupAzureClient(
      111        StageInfo stage, RemoteStoreFileEncryptionMaterial encMat, SFBaseSession sfSession)
      112        throws IllegalArgumentException,
      113            SnowflakeSQLException,
      114            net.snowflake.client.jdbc.SnowflakeSQLException {
      115      // Save the client creation parameters so that we can reuse them,
      116      // to reset the Azure client.
      117      this.stageInfo = stage;
      118      this.encMat = encMat;
      119      this.session = sfSession;
      120  
      121      logger.debug("Setting up the Azure client ", false);
      122  
      123      try {
      124        URI storageEndpoint =
      125            buildAzureStorageEndpointURI(stage.getEndPoint(), stage.getStorageAccount());
      126  
      127        StorageCredentials azCreds;
      128        String sasToken = (String) stage.getCredentials().get("AZURE_SAS_TOKEN");
      129        if (sasToken != null) {
      130          // We are authenticated with a shared access token.
      131          azCreds = new StorageCredentialsSharedAccessSignature(sasToken);
      132        } else {
      133          // Use anonymous authentication.
      134          azCreds = StorageCredentialsAnonymous.ANONYMOUS;
      135        }
      136  
      137        if (stage.getIsClientSideEncrypted() && encMat != null) {
      138          byte[] decodedKey = Base64.getDecoder().decode(encMat.getQueryStageMasterKey());
      139          encryptionKeySize = decodedKey.length * 8;
      140  
      141          if (encryptionKeySize != 128 && encryptionKeySize != 192 && encryptionKeySize != 256) {
      142            throw new SnowflakeSQLLoggedException(
      143                QueryIdHelper.queryIdFromEncMatOr(encMat, null),
      144                session,
      145                ErrorCode.INTERNAL_ERROR.getMessageCode(),
      146                SqlState.INTERNAL_ERROR,
      147                "unsupported key size",
      148                encryptionKeySize);
      149          }
      150        }
      151        this.azStorageClient = new CloudBlobClient(storageEndpoint, azCreds);
      152        opContext = new OperationContext();
      153        if (session != null) {
      154          setProxyForAzure(session.getHttpClientKey(), opContext);
      155        } else {
      156          setSessionlessProxyForAzure(stage.getProxyProperties(), opContext);
      157        }
      158      } catch (URISyntaxException ex) {
      159        throw new IllegalArgumentException("invalid_azure_credentials");
      160      }
      161    }
