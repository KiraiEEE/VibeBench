      163      public Optional<ResultSet> computeResultInFe(
      164              CascadesContext cascadesContext, Optional<SqlCacheContext> sqlCacheContext, List<Slot> outputSlots) {
      165          List<Column> columns = Lists.newArrayList();
      166          List<String> data = Lists.newArrayList();
      167          for (Slot outputSlot : outputSlots) {
      168              for (int i = 0; i < projects.size(); i++) {
      169                  NamedExpression item = projects.get(i);
      170                  NamedExpression output = getOutput().get(i);
      171                  if (!outputSlot.getExprId().equals(output.getExprId())) {
      172                      continue;
      173                  }
      174                  Expression expr = item.child(0);
      175                  if (expr instanceof Literal) {
      176                      LiteralExpr legacyExpr = ((Literal) expr).toLegacyLiteral();
      177                      columns.add(new Column(output.getName(), output.getDataType().toCatalogDataType()));
      178                      if (output.getDataType().toCatalogDataType().isVarbinaryType()) {
      179                          // The FE (computeResultInFe) can currently only build a ResultSet<List<List<String>>>.
      180                          // If we materialize a VARBINARY literal via legacyExpr.getStringValueForQuery():
      181                          //   1) We first wrap the raw bytes in a Java String.
      182                          //   2) Later StmtExecutor.sendTextResultRow re-encodes that String as UTF-8 when
      183                          //      writing the MySQL wire protocol. This may expand bytes (e.g. 0xAB becomes
      184                          //      two bytes 0xC2 0xAB), so the client observes a different value than the BE path.
      185                          // The BE execution path is correct because it sends the raw bytes already serialized
      186                          // for the MySQL protocol without a lossy String round-trip.
      187                          // Todo: Refactor ResultSet (supports per-cell byte[])
      188                          // so we can VARBINARY safely and remove this early return.
      189                          return Optional.empty();
      190                      }
      191                      data.add(legacyExpr.getStringValueForQuery(
      192                              cascadesContext.getStatementContext().getFormatOptions()));
      193                  } else {
      194                      return Optional.empty();
      195                  }
      196              }
      197          }
      198  
      199          ResultSetMetaData metadata = new CommonResultSet.CommonResultSetMetaData(columns);
      200          ResultSet resultSet = new CommonResultSet(metadata, Collections.singletonList(data));
      201          StatementContext statementContext = cascadesContext.getStatementContext();
      202          boolean enableSqlCache
      203                  = CacheAnalyzer.canUseSqlCache(statementContext.getConnectContext().getSessionVariable());
      204          if (sqlCacheContext.isPresent() && enableSqlCache) {
      205              sqlCacheContext.get().setResultSetInFe(resultSet);
