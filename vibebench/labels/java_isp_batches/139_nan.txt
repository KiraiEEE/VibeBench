       73          String actualCookie = String.join(COOKIE_DELIMITER, this.headers.all(COOKIE));
       74          this.cookies = ConfigurableCookieCollection.create(HttpUtils.parseHeaderValue(actualCookie));
       75      }

      109      /**
      110       * 根据指定的数据类型设置消息头的 {@code 'Content-Type'}。
      111       *
      112       * @param headers 表示待设置的消息头的 {@link ConfigurableMessageHeaders}。
      113       * @param entity 表示指定的数据类型的 {@link Entity}。
      114       * @throws UnsupportedOperationException 当 {@code entity} 的类型为不支持的类型时。
      115       */
      116      protected void setContentTypeByEntity(ConfigurableMessageHeaders headers, Entity entity) {
      117          boolean isPresent = headers.first(CONTENT_TYPE).isPresent();
      118          if (isPresent) {
      119              return;
      120          }
      121          ParameterCollection mergedParameters = ParameterCollection.create();
      122          for (String key : this.parameters.keys()) {
      123              this.parameters.get(key).ifPresent(value -> mergedParameters.set(key, value));
      124          }
      125          entity.resolvedParameters().forEach(mergedParameters::set);
      126          ContentType contentType =
      127                  HeaderValue.create(entity.resolvedMimeType().value(), mergedParameters).toContentType();
      128          notNull(contentType,
      129                  () -> new UnsupportedOperationException(StringUtils.format(
      130                          "Not supported entity type. " + "[entityType={0}]", entity.getClass().getName())));
      131          headers.set(CONTENT_TYPE, contentType.toString());
      132      }

      169          MimeType mimeType = this.mimeTypeOrDefault();
      170          return this.getEntitySerializer(mimeType, type);
      171      }
      172  
      173      private EntitySerializer<?> getEntitySerializer(MimeType mimeType, Type type) {
      174          if (mimeType == MimeType.APPLICATION_JSON && type != null) {
      175              return this.httpResource()
      176                      .serializers()
      177                      .jsonEntity(type)
      178                      .orElseThrow(() -> new IllegalStateException("No json serializer."));
      179          }
      180          if (mimeType == MimeType.TEXT_EVENT_STREAM && type != null) {
      181              return this.httpResource()
      182                      .serializers()
      183                      .textEventStreamEntity(type)
      184                      .orElseThrow(() -> new IllegalStateException("No json serializer."));
      185          }
      186          EntitySerializer<?> entitySerializer = this.customEntitySerializers.get(mimeType);
      187          if (entitySerializer == null) {
      188              entitySerializer = this.httpResource().serializers().entities().get(mimeType);
      189          }
      190          return getIfNull(entitySerializer, EntitySerializer::readableBinarySerializer);
      191      }
