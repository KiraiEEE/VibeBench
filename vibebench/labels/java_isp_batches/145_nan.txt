      324    public void clearScope(String reason, boolean skipCustomListeners, boolean skipIoMappings, boolean externallyTerminated) {
      325      this.skipCustomListeners = skipCustomListeners;
      326      this.skipIoMapping = skipIoMappings;
      327  
      328      if (getSubProcessInstance() != null) {
      329        getSubProcessInstance().deleteCascade(reason, skipCustomListeners, skipIoMappings, externallyTerminated, false);
      330      }
      331  
      332      // remove all child executions and sub process instances:
      333      List<PvmExecutionImpl> executions = new ArrayList<>(getNonEventScopeExecutions());
      334      for (PvmExecutionImpl childExecution : executions) {
      335        if (childExecution.getSubProcessInstance() != null) {
      336          childExecution.getSubProcessInstance().deleteCascade(reason, skipCustomListeners, skipIoMappings, externallyTerminated, false);
      337        }
      338        childExecution.deleteCascade(reason, skipCustomListeners, skipIoMappings, externallyTerminated, false);
      339      }
      340  
      341      // fire activity end on active activity
      342      PvmActivity pvmActivity = getActivity();
      343      if ((isActive || externallyTerminated) && pvmActivity != null) {
      344        // set activity instance state to cancel
      345        if (activityInstanceState != ENDING.getStateCode() || activityInstanceEndListenersFailed) {
      346          setCanceled(true);
      347          performOperation(PvmAtomicOperation.FIRE_ACTIVITY_END);
      348        }
      349        // set activity instance state back to 'default'
      350        // -> execution will be reused for executing more activities and we want the state to
      351        // be default initially.
      352        activityInstanceState = ActivityInstanceState.DEFAULT.getStateCode();
      353      }
      354    }
