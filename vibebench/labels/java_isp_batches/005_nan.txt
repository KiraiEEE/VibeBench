       96      /**
       97       * The range may be across multi ledgers:
       98       *   - start: the latest ledger that closed before {@param targetTimestamp}.
       99       *     - only the latest entry is useful.
      100       *   - end: the earliest ledger that is larger than the target timestamp.
      101       */
      102      @VisibleForTesting
      103      public static Pair<Position, Position> getFindPositionRange(Iterable<LedgerInfo> ledgerInfos,
      104                                                                  Position lastConfirmedEntry, long targetTimestamp,
      105                                                                  int ledgerCloseTimestampMaxClockSkewMillis) {
      106          if (ledgerCloseTimestampMaxClockSkewMillis < 0) {
      107              // this feature is disabled when the value is negative
      108              return Pair.of(null, null);
      109          }
      110  
      111          long targetTimestampMin = targetTimestamp - ledgerCloseTimestampMaxClockSkewMillis;
      112          long targetTimestampMax = targetTimestamp + ledgerCloseTimestampMaxClockSkewMillis;
      113  
      114          Position start = null;
      115          Position end = null;
      116  
      117          // We do not use binary search here:
      118          // Since "managedLedger.ledgers" os a map, we can hardly use a binary search except to copy items to an array,
      119          // which causes frequently young GC. And "collection.toArray()" also loops the collection once, which does not
      120          // benefit performance anymore.
      121          for (LedgerInfo info : ledgerInfos) {
      122              if (!info.hasTimestamp()) {
      123                  // unexpected case, don't set start and end
      124                  return Pair.of(null, null);
      125              }
      126              long closeTimestamp = info.getTimestamp();
      127              // For an open ledger, closeTimestamp is 0
      128              if (closeTimestamp == 0) {
      129                  end = null;
      130                  break;
      131              }
      132              if (closeTimestamp <= targetTimestampMin) {
      133                  // Since we have "broker.conf -> managedLedgerCursorResetLedgerCloseTimestampMaxClockSkewMillis", which
      134                  // already expanded the scope for searching, the entries before the latest one is not useful.
      135                  start = PositionFactory.create(info.getLedgerId(), info.getEntries() - 1);
      136              } else if (closeTimestamp > targetTimestampMax) {
      137                  // If the close timestamp is greater than the timestamp
      138                  end = PositionFactory.create(info.getLedgerId(), info.getEntries() - 1);
      139                  break;
      140              }
      141          }
      142          return Pair.of(start, end);
      143      }
