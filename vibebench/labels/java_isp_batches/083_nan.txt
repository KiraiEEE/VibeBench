       93          providers.put(CryptoConstants.ECDH_ES_A192KW, new BCFIPSEcdhEsAlgorithmProvider());
       94          providers.put(CryptoConstants.ECDH_ES_A256KW, new BCFIPSEcdhEsAlgorithmProvider());
       95  
       96          if (existingBcFipsProvider == null) {
       97              checkSecureRandom(() -> Security.insertProviderAt(this.bcFipsProvider, 1));
       98              Provider bcJsseProvider = new BouncyCastleJsseProvider("fips:BCFIPS");
       99              Security.insertProviderAt(bcJsseProvider, 2);
      100              // force the key and trust manager factories if default values not present in BCJSSE
      101              modifyKeyTrustManagerSecurityProperties(bcJsseProvider);
      102              log.debugf("Inserted security providers: %s", Arrays.asList(this.bcFipsProvider.getName(),bcJsseProvider.getName()));
      103          } else {
      104              log.debugf("Security provider %s already loaded", existingBcFipsProvider.getName());
      105          }
      106  
      107          log.infof("FIPS1402Provider created: KC(%s%s, FIPS-JVM: %s)", bcFipsProvider,
      108                  CryptoServicesRegistrar.isInApprovedOnlyMode() ? " Approved Mode" : "",
      109                  isSystemFipsEnabled());
      110      }

      142      /* Create EC Params using BC FipS APIs.
      143       *
      144       * @see org.keycloak.common.crypto.CryptoProvider#createECParams(java.lang.String)
      145       */
      146      @Override
      147      public ECParameterSpec createECParams(String curveName) {
      148          X9ECParameters params = ECNamedCurveTable.getByName(curveName);
      149          ECField field ;
      150          ECCurve ecCurve = params.getCurve();
      151          if (ecCurve instanceof ECCurve.F2m) {
      152              ECCurve.F2m f2m = (ECCurve.F2m) ecCurve;
      153              field = new ECFieldF2m(f2m.getM(), new int[] { f2m.getK1(), f2m.getK2(), f2m.getK3()});
      154          }
      155          else
      156          if (ecCurve instanceof ECCurve.Fp) {
      157              ECCurve.Fp fp = (ECCurve.Fp) ecCurve;
      158              field = new ECFieldFp(fp.getQ());
      159          }
      160          else
      161              throw new RuntimeException("Unsupported curve");
      162  
      163  
      164          EllipticCurve c = new EllipticCurve(field,
      165                  ecCurve.getA().toBigInteger(),
      166                  ecCurve.getB().toBigInteger(),
      167                  params.getSeed());
      168          ECPoint point = new ECPoint( params.getG().getXCoord().toBigInteger(), params.getG().getYCoord().toBigInteger());
      169          return new ECParameterSpec( c,point, params.getN(), params.getH().intValue());
      170      }
