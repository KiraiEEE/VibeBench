       34    /**
       35     * Resolves an entity to its lineage-capable parent entities using ChangeEvent payload.
       36     *
       37     * This method is preferred when processing deletion events, as it uses the entity
       38     * snapshot captured in the ChangeEvent payload (before deletion). This allows parent
       39     * resolution to work even when the source entity has been deleted.
       40     *
       41     * Returns a set of parent entity references. Empty set means the entity itself
       42     * is lineage-capable and should be used for traversal.
       43     *
       44     * Examples:
       45     * - TestCase → {Table} (1:1 relationship)
       46     * - Thread → {TestCase} (1:1 relationship)
       47     * - TestSuite → {Table1, Table2, ...} (1:N from failed tests)
       48     * - Table → {} (empty set, is lineage entity)
       49     *
       50     * @param changeEvent the ChangeEvent containing the entity snapshot
       51     * @return Set of parent EntityReferences, or empty set if entity is lineage-capable
       52     */
       53    Set<EntityReference> resolveTraversalEntities(ChangeEvent changeEvent);
       54  
       55    /**
       56     * Resolves an entity to its lineage-capable parent entities by ID.
       57     *
       58     * This method is used when traversing lineage from parent entities onwards,

       60     *
       61     * Returns a set of parent entity references. Empty set means the entity itself
       62     * is lineage-capable and should be used for traversal.
       63     *
       64     * Examples:
       65     * - TestCase → {Table} (1:1 relationship)
       66     * - Thread → {TestCase} (1:1 relationship)
       67     * - TestSuite → {Table1, Table2, ...} (1:N from failed tests)
       68     * - Table → {} (empty set, is lineage entity)
       69     *
       70     * @param entityId the ID of the entity to resolve

       72     * @return Set of parent EntityReferences, or empty set if entity is lineage-capable

       74    Set<EntityReference> resolveTraversalEntities(UUID entityId, String entityType);
