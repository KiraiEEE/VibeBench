      138      public static long appendPlainValue(long appendAddress, ArrayView value) {
      139          long startAddress = appendAddress;
      140          if (value == null || value.isNull()) {
      141              Unsafe.getUnsafe().putLong(appendAddress, TableUtils.NULL_LEN);
      142              return Long.BYTES;
      143          }
      144          Unsafe.getUnsafe().putLong(appendAddress, value.getVanillaMemoryLayoutSize());
      145          appendAddress += Long.BYTES;
      146          Unsafe.getUnsafe().putInt(appendAddress, value.getType());
      147          appendAddress += Integer.BYTES;
      148          for (int nDims = value.getDimCount(), i = 0; i < nDims; i++) {
      149              Unsafe.getUnsafe().putInt(appendAddress, value.getDimLen(i));
      150              appendAddress += Integer.BYTES;
      151          }
      152          if (value.isVanilla()) {
      153              short elemType = value.getElemType();
      154              if (elemType == ColumnType.DOUBLE) {
      155                  appendAddress = value.flatView().appendPlainDoubleValue(appendAddress, value.getFlatViewOffset(), value.getFlatViewLength());
      156              } else {
      157                  throw new UnsupportedOperationException("Unsupported array element type: " + elemType);
      158              }
      159          } else {
      160              appendAddress = appendToMemRecursive(value, 0, 0, appendAddress);
      161          }
      162          long bytesWritten = appendAddress - startAddress;
      163          assert bytesWritten > 0;
      164          return bytesWritten;

      188          if (value == null || value.isNull()) {
      189              Unsafe.getUnsafe().putInt(addr, TableUtils.NULL_LEN);
      190              return;
      191          }
      192  
      193          int dataSize = value.getCardinality() * elemSize;
      194  
      195          Unsafe.getUnsafe().putInt(addr, dataSize);
      196          addr += Integer.BYTES;
      197

      199              Unsafe.getUnsafe().putInt(addr, value.getDimLen(i));
      200              addr += Integer.BYTES;
      201          }
