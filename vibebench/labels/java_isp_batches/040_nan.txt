       92        properties = new HashMap<>(propertiesMap.size());
       93  
       94      for (Map<String, Object> entry : propertiesMap) {
       95        final String propertyName = (String) entry.get("name");
       96  
       97        RemoteProperty p = properties.get(propertyName);
       98        if (p == null) {
       99          p = new RemoteProperty(this, entry);
      100          properties.put(propertyName, p);
      101        } else
      102          p.reload(entry);
      103      }
      104  
      105      if (record.hasProperty("custom"))
      106        custom = record.getProperty("custom");
      107    }
