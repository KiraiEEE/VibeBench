      126      private <K, V> LogMinerCache<K, V> createCache(String cacheName, OracleConnectorConfig connectorConfig, Field field) {
      127          Objects.requireNonNull(cacheName);
      128  
      129          final String cacheConfiguration = connectorConfig.getConfig().getString(field);
      130          Objects.requireNonNull(cacheConfiguration);
      131  
      132          // define the cache, parsing the supplied XML configuration
      133          cacheManager.defineConfiguration(cacheName, parseAndGetConfiguration(cacheName, cacheConfiguration));
      134          return new InfinispanLogMinerCache<>(cacheManager.getCache(cacheName));
      135      }

      137      private Configuration parseAndGetConfiguration(String cacheName, String configuration) {
      138          // We should only be parsing a single cache configuration; however since we have no control over the
      139          // contents of the provided XML, we validate there is at least 1 and only 1 valid configuration that
      140          // is provided, and we use the configuration as is.
      141          //
      142          // The purpose of these checks is to allow the user to supply a cache configuration with any valid
      143          // Infinispan name, and we automatically map that configuration to the cache names used by Debezium.
      144          // So for example, a user could supply a cache config named "processed_transactions" and we would
      145          // allow that being mapped to our internal cache name of "processed-transactions". Using iterators
      146          // avoids a NullPointerException in the event the user supplies a name that doesn't match to our
      147          // internal names exactly.
      148          final ConfigurationBuilderHolder builderHolder = new ParserRegistry().parse(configuration);
      149          final Map<String, ConfigurationBuilder> builders = builderHolder.getNamedConfigurationBuilders();
      150          if (builders.size() > 1) {
      151              throw new DebeziumException("Infinispan cache configuration for '" + cacheName +
      152                      "' contains multiple cache configurations and should only contain one.");
      153          }
      154          else if (builders.isEmpty()) {
      155              throw new DebeziumException("Infinispan cache configuration for '" + cacheName +
      156                      "' contained no valid cache configuration. Please check your connector configuration");
      157          }
      158          else {
      159              return builders.values().iterator().next().build();
      160          }
      161      }
      162  
      163      private GlobalConfiguration parseAndGetGlobalConfiguration(OracleConnectorConfig connectorConfig) {
      164          final String globalCacheConfiguration = connectorConfig.getLogMiningInifispanGlobalConfiguration();
      165          if (globalCacheConfiguration == null) {
      166              // if no configuration provided, use the default
      167              return new GlobalConfigurationBuilder().build();
      168          }
      169          final ConfigurationBuilderHolder builderHolder = new ParserRegistry().parse(globalCacheConfiguration);
      170          return builderHolder.getGlobalConfigurationBuilder().build();
      171      }
