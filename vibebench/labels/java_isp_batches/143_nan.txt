      240      static byte[] getBytes(ByteBuffer byteBuffer) {
      241          if (byteBuffer == null) {
      242              return null;
      243          }
      244          if (byteBuffer.hasArray() && byteBuffer.arrayOffset() == 0
      245                  && byteBuffer.array().length == byteBuffer.remaining()) {
      246              return byteBuffer.array();
      247          }
      248          // Direct buffer is not backed by array and it needs to be read from direct memory
      249          byte[] array = new byte[byteBuffer.remaining()];
      250          byteBuffer.get(array);
      251          return array;
      252      }
