       70       * @param capacity the capacity of the mapped file.
       71       * @param readOnly if the file is read-only.
       72       * @throws IORuntimeException if any I/O error occurs.
       73       */
       74      @SuppressWarnings("this-escape")
       75      public SingleMappedFile(@NotNull final File file,
       76                              @NotNull final RandomAccessFile raf,
       77                              @NonNegative final long capacity,
       78                              @Positive final int pageSize,
       79                              final boolean readOnly)
       80              throws IORuntimeException {
       81          super(file, readOnly);
       82  
       83          this.raf = raf;
       84          this.fileChannel = raf.getChannel();
       85          this.capacity = OS.mapAlign(capacity, PageUtil.getPageSize(file.getAbsolutePath()));
       86  
       87          final MapMode mode = readOnly() ? MapMode.READ_ONLY : MapMode.READ_WRITE;
       88  
       89          final long beginNs = System.nanoTime();
       90          boolean ok = false;
       91          try {
       92              Jvm.doNotCloseOnInterrupt(getClass(), this.fileChannel);
       93  
       94              resizeRafIfTooSmall(this.capacity);
       95              final long address = OS.map(fileChannel, mode, 0, this.capacity, pageSize);
       96              final MappedBytesStore mbs2 = MappedBytesStore.create(this, this, 0, address, this.capacity, this.capacity, pageSize);
       97              mbs2.syncMode(DEFAULT_SYNC_MODE);
       98  
       99              final long elapsedNs = System.nanoTime() - beginNs;
      100              if (newChunkListener != null)
      101                  newChunkListener.onNewChunk(file().getPath(), 0, elapsedNs / 1000);
      102              if (elapsedNs >= 2_000_000L)
      103                  Jvm.perf().on(getClass(), "Took " + elapsedNs / 1_000_000L + " ms to add mapping for " + file());
      104  
      105              store = mbs2;
      106  
      107              ok = true;
      108  
      109          } catch (IOException ioe) {
      110              throw new IORuntimeException(ioe);
      111  
      112          } finally {
      113              if (!ok)
      114                  close();
      115          }
      116      }

      161      @SuppressWarnings("try")
      162      private void resizeRafIfTooSmall(@NonNegative final long minSize)
      163              throws IOException {
      164          Jvm.safepoint();
      165  
      166          long size = fileChannel.size();
      167          Jvm.safepoint();
      168          if (size >= minSize || readOnly())
      169              return;
      170  
      171          // handle a possible race condition between processes.
      172          try {
      173              // A single JVM cannot lock a distinct canonical file more than once.
      174  
      175              // We might have several MappedFile objects that maps to
      176              // the same underlying file (possibly via hard or soft links)
      177              // so we use the canonical path as a lock key
      178  
      179              // Ensure exclusivity for any and all MappedFile objects handling
      180              // the same canonical file.
      181              synchronized (internalizedToken()) {
      182                  size = fileChannel.size();
      183                  if (size < minSize) {
      184                      final long beginNs = System.nanoTime();
      185                      try (FileLock ignore = ReentrantFileLock.lock(file(), fileChannel)) {
      186                          size = fileChannel.size();
      187                          if (size < minSize) {
      188                              Jvm.safepoint();
      189                              raf.setLength(minSize);
      190                              Jvm.safepoint();
      191                          }
      192                      }
      193                      final long elapsedNs = System.nanoTime() - beginNs;
      194                      if (elapsedNs >= 1_000_000L) {
      195                          Jvm.perf().on(getClass(), "Took " + elapsedNs / 1000L + " us to grow file " + file());
      196                      }
      197                  }
      198              }
      199          } catch (IOException ioe) {
      200              throw new IOException("Failed to resize to " + minSize, ioe);
      201          }
