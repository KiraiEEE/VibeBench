       65      public Collection<SnapshotSplit> generateSplits(TableId tableId) throws Exception {
       66          // For tables without primary keys, return empty split list instead of failing validation.
       67          // Note: ongoing splitting must continue to preserve ChunkSplitterState correctness.
       68          if (hasNextChunk()) {
       69              return super.generateSplits(tableId);
       70          }
       71  
       72          try {
       73              return super.generateSplits(tableId);
       74          } catch (RuntimeException e) {
       75              ValidationException validationException = findCause(e, ValidationException.class);
       76              if (validationException != null
       77                      && validationException.getMessage() != null
       78                      && NO_PRIMARY_KEY_MESSAGE.equals(validationException.getMessage())) {
       79                  return Collections.emptyList();
       80              }
       81              throw e;
       82          }
       83      }

      235      private static boolean isUUID(Column column) {
      236          // Debezium may return "uuid" or schema-qualified/cased names, normalize it.
      237          String typeName = column.typeName();
      238          if (typeName == null) {
      239              return false;
      240          }
      241          String normalized = typeName.trim();
      242          if ((normalized.startsWith("\"") && normalized.endsWith("\""))
      243                  || (normalized.startsWith("'") && normalized.endsWith("'"))) {
      244              normalized = normalized.substring(1, normalized.length() - 1).trim();
      245          }
      246          int lastDot = normalized.lastIndexOf('.');
      247          if (lastDot >= 0 && lastDot < normalized.length() - 1) {
      248              normalized = normalized.substring(lastDot + 1);
      249          }
      250          normalized = normalized.toLowerCase(Locale.ROOT);
      251          return "uuid".equals(normalized);
      252      }

      255      private static <T extends Throwable> T findCause(Throwable throwable, Class<T> causeType) {
      256          Throwable current = throwable;
      257          while (current != null) {
      258              if (causeType.isInstance(current)) {
      259                  return (T) current;
      260              }
      261              current = current.getCause();
      262          }
