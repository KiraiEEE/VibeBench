      478    /**
      479     * Configure URI compliance for Jetty 12. By default, Jetty 12 uses strict URI compliance which
      480     * rejects special characters that were allowed in Jetty 11. OpenMetadata allows special
      481     * characters in entity names (including encoded chars like %22 for double quotes), so we
      482     * default to UNSAFE compliance mode unless explicitly configured otherwise.
      483     * Note: For tests using DropwizardAppExtension, uriCompliance must be set in YAML config
      484     * since the server is initialized before run() is called.
      485     */
      486    private void configureUriCompliance(OpenMetadataApplicationConfig configuration) {
      487      if (configuration.getServerFactory() instanceof DefaultServerFactory serverFactory) {
      488        // Configure application connectors - always set to UNSAFE for backward compatibility
      489        for (ConnectorFactory connector : serverFactory.getApplicationConnectors()) {
      490          if (connector instanceof HttpConnectorFactory httpConnector) {
      491            httpConnector.setUriCompliance(UriCompliance.UNSAFE);
      492            LOG.info("Set URI compliance to UNSAFE for application connector");
      493          }
      494        }
      495        // Configure admin connectors - always set to UNSAFE for backward compatibility
      496        for (ConnectorFactory connector : serverFactory.getAdminConnectors()) {
      497          if (connector instanceof HttpConnectorFactory httpConnector) {
      498            httpConnector.setUriCompliance(UriCompliance.UNSAFE);
      499            LOG.info("Set URI compliance to UNSAFE for admin connector");
      500          }
      501        }
      502      }
      503    }

      505    /**
      506     * Configure the ServletHandler to allow ambiguous URIs in servlet API methods.
      507     * In Jetty 12 / Servlet 6, methods like getServletPath() and getPathInfo() throw
      508     * IllegalArgumentException for URIs containing ambiguous characters like %2F (encoded slash)
      509     * or %22 (encoded quote). Setting setDecodeAmbiguousURIs(true) allows these methods to work.
      510     *
      511     * This is required because OpenMetadata entity names can contain special characters
      512     * (e.g., "domain.name/with-slash") which get URL-encoded by clients.
      513     *
      514     * See: https://github.com/jetty/jetty.project/issues/12346
      515     * See: https://jetty.org/docs/jetty/12/programming-guide/server/compliance.html
      516     */
      517    private void configureServletHandler(Environment environment) {
      518      MutableServletContextHandler contextHandler = environment.getApplicationContext();
      519      org.eclipse.jetty.ee10.servlet.ServletHandler servletHandler =
      520          contextHandler.getServletHandler();
      521      if (servletHandler != null) {
      522        servletHandler.setDecodeAmbiguousURIs(true);
      523        LOG.info(
      524            "Configured ServletHandler to allow ambiguous URIs (required for %2F, %22 in paths)");
      525      }
      526    }
