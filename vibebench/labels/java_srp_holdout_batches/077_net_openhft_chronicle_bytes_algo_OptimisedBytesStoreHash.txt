      317      /**
      318       * Reads a long from the given address. If the length is less than 8,
      319       * it reads as many bytes as specified and converts them to a long.
      320       * Endianness is considered for reading bytes.
      321       *
      322       * @param address the starting address of the memory to read from
      323       * @param len     the number of bytes to read (must be non-negative). Reads up to eight bytes
      324       * @return the value read from memory converted to a {@code long}. Returns {@code 0} if {@code len} is 0 or negative
      325       */
      326      static long readIncompleteLong(long address, @NonNegative int len) {
      327          switch (len) {
      328              case 1:
      329                  return MEMORY.readByte(address);
      330              case 2:
      331                  return MEMORY.readShort(address);
      332              case 3:
      333                  return IS_LITTLE_ENDIAN
      334                          ? (MEMORY.readShort(address) & 0xFFFF) + ((MEMORY.readByte(address + 2) & 0xFF) << 16)
      335                          : ((MEMORY.readShort(address) & 0xFFFF) << 8) + (MEMORY.readByte(address + 2) & 0xFF);
      336              case 4:
      337                  return MEMORY.readInt(address);
      338              case 5:
      339                  return IS_LITTLE_ENDIAN
      340                          ? (MEMORY.readInt(address) & 0xFFFFFFFFL) + ((long) (MEMORY.readByte(address + 4) & 0xFF) << 32)
      341                          : ((MEMORY.readInt(address) & 0xFFFFFFFFL) << 8) + (MEMORY.readByte(address + 4) & 0xFF);
      342              case 6:

      345                          : ((MEMORY.readInt(address) & 0xFFFFFFFFL) << 16) + (MEMORY.readShort(address + 4) & 0xFFFF);
