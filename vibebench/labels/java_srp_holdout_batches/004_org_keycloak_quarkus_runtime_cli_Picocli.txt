      353      private void validateBuildtime() {
      354          final List<String> ignoredBuildTime = new ArrayList<>();
      355          // check for provider changes, or overrides of existing persisted options
      356          // we have to ignore things like the profile properties because the commands set them at runtime
      357          checkChangesInBuildOptions((key, oldValue, newValue) -> {
      358              if (key.startsWith(KC_PROVIDER_FILE_PREFIX)) {
      359                  boolean changed = false;
      360                  if (newValue == null || oldValue == null) {
      361                      changed = true;
      362                  } else if (!warnedTimestampChanged && timestampChanged(oldValue, newValue)) {
      363                      if (Configuration.getOptionalBooleanKcValue("run-in-container").orElse(false)) {
      364                          warnedTimestampChanged = true;
      365                          warn(PROVIDER_TIMESTAMP_WARNING);
      366                      } else {
      367                          changed = true;
      368                      }
      369                  }
      370                  if (changed) {
      371                      throw new PropertyException(PROVIDER_TIMESTAMP_ERROR);
      372                  }
      373              } else if (newValue != null && !isIgnoredPersistedOption(key)
      374                      && isUserModifiable(Configuration.getConfigValue(key))
      375                      // let quarkus handle this - it's unsupported for direct usage in keycloak
      376                      && !key.startsWith(MicroProfileConfigProvider.NS_QUARKUS_PREFIX)) {
      377                  ignoredBuildTime.add(key);
      378              }
      379          });
      380  
      381          if (!ignoredBuildTime.isEmpty()) {
      382              throw new PropertyException(format("The following build time options have values that differ from what is persisted - the new values will NOT be used until another build is run: %s\n",
      383                      String.join(", ", ignoredBuildTime)));
      384          }

      387      static boolean timestampChanged(String oldValue, String newValue) {
      388          long longNewValue = Long.valueOf(newValue);
      389          long longOldValue = Long.valueOf(oldValue);
      390          // docker commonly truncates to the second at runtime, so we'll allow that special case
      391          return ((longNewValue / 1000) * 1000) != longNewValue || ((longOldValue / 1000) * 1000) != longNewValue;
      392      }
