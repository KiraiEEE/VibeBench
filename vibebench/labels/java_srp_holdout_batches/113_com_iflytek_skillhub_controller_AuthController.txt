       71      @GetMapping("/me")
       72      public ApiResponse<AuthMeResponse> me(@AuthenticationPrincipal PlatformPrincipal principal,
       73                                            Authentication authentication,
       74                                            HttpServletRequest request) {
       75          if (principal == null || authentication == null || !authentication.isAuthenticated()) {
       76              throw new UnauthorizedException("error.auth.required");
       77          }
       78          UserAccount user = userAccountRepository.findById(principal.userId()).orElse(null);
       79          if (user == null || user.getStatus() == UserStatus.DISABLED) {
       80              request.getSession().invalidate();
       81              throw new UnauthorizedException("error.auth.required");
       82          }
       83          Set<String> freshRoles = PlatformRoleDefaults.withDefaultUserRole(
       84                  userRoleBindingRepository.findByUserId(principal.userId()).stream()
       85                          .map(binding -> binding.getRole().getCode())
       86                          .collect(Collectors.toSet()));
       87          if (!freshRoles.equals(principal.platformRoles())) {
       88              principal = new PlatformPrincipal(
       89                      principal.userId(), principal.displayName(), principal.email(),
       90                      principal.avatarUrl(), principal.oauthProvider(), freshRoles);
       91              platformSessionService.establishSession(principal, request, false);
       92          }
       93          return ok("response.success.read", AuthMeResponse.from(principal));
       94      }

      146      private String resolveClientIp(HttpServletRequest request) {
      147          String ip = request.getHeader("X-Forwarded-For");
      148          if (ip == null || ip.isEmpty() || "unknown".equalsIgnoreCase(ip)) {
      149              ip = request.getHeader("X-Real-IP");
      150          }
      151          if (ip == null || ip.isEmpty() || "unknown".equalsIgnoreCase(ip)) {
      152              ip = request.getRemoteAddr();
      153          }
      154          if (ip != null && ip.contains(",")) {
      155              ip = ip.split(",")[0].trim();
      156          }
      157          return ip;
      158      }
