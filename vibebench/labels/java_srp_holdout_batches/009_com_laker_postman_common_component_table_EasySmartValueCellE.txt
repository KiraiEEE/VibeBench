      158      private void switchToMultiLine(String text) {
      159          if (isMultiLine || currentTable == null) return;
      160          log.debug("[switchToMultiLine] text.length={}", text.length());
      161          switching = true;
      162          ignoreFocusLost = true;
      163          try {
      164              int caretPos = Math.min(textField.getCaretPosition(), text.length());
      165              isMultiLine = true;
      166              textArea.setText(text);
      167              textArea.setCaretPosition(caretPos);
      168  
      169              containerPanel.setFocusable(true);
      170              containerPanel.requestFocusInWindow();
      171              cardLayout.show(containerPanel, CARD_MULTI);
      172  
      173              if (!rowHeightExpanded) {
      174                  originalRowHeight = currentTable.getRowHeight(currentRow);
      175                  rowHeightExpanded = true;
      176              }
      177              int lines = Math.min(MAX_EDITOR_LINES, Math.max(MIN_EDITOR_LINES, countLines(text)));
      178              applyRowHeight(currentTable, currentRow,
      179                      textArea.getFontMetrics(textArea.getFont()).getHeight() * lines + 14);
      180  
      181              containerPanel.revalidate();
      182              containerPanel.repaint();
      183              SwingUtilities.invokeLater(() -> {
      184                  textArea.requestFocusInWindow();
      185                  SwingUtilities.invokeLater(() -> {
      186                      ignoreFocusLost = false;
      187                      containerPanel.setFocusable(false);
      188                  });
      189              });
      190          } finally {
      191              switching = false;
      192          }
      193      }

      237      private static void applyRowHeight(JTable table, int row, int height) {
      238          if (table == null) return;
      239          log.debug("[applyRowHeight] row={} height={} before={}", row, height, table.getRowHeight(row));
      240          table.setRowHeight(row, height);
      241          SwingUtilities.invokeLater(() -> {
      242              Container p = table.getParent();
      243              while (p != null) {
      244                  if (p instanceof JScrollPane) {
      245                      log.debug("[applyRowHeight] validate JScrollPane size={}", p.getSize());
      246                      p.validate();
      247                      break;
      248                  }
      249                  p = p.getParent();
      250              }
      251              Rectangle cellRect = table.getCellRect(row, 0, true);
      252              log.debug("[applyRowHeight] scrollRectToVisible={}", cellRect);
      253              table.scrollRectToVisible(cellRect);
      254          });
      255      }
