      167      public static Nd4jBackend load() throws NoAvailableBackendException {
      168          // First, try to return already loaded backend (fast path)
      169          if (isLoaded.get() && loadedBackend != null) {
      170              return loadedBackend;

      173          // Use write lock for loading operations
      174          loadLock.writeLock().lock();

      178                  return loadedBackend;

      181              return loadBackendInternal();
      182          } finally {
      183              loadLock.writeLock().unlock();

      190      private static Nd4jBackend loadBackendInternal() throws NoAvailableBackendException {
      191          String logInitProperty = System.getProperty(ND4JSystemProperties.LOG_INITIALIZATION, "true");
      192          boolean logInit = Boolean.parseBoolean(logInitProperty);
      193  
      194          List<Nd4jBackend> backends = new ArrayList<>();
      195          ServiceLoader<Nd4jBackend> loader = ND4JClassLoading.loadService(Nd4jBackend.class);
      196  
      197          try {
      198              for (Nd4jBackend nd4jBackend : loader) {
      199                  backends.add(nd4jBackend);
      200              }
      201          } catch (ServiceConfigurationError serviceError) {
      202              log.warn("Failed to process available backends", serviceError);
      203              // Don't return null, try dynamic loading instead
      204          }
      205  
      206          // Sort backends by priority (high-priority first)
      207          backends.sort((o1, o2) -> Integer.compare(o2.getPriority(), o1.getPriority()));
      208  
      209          // Try to load available backends
      210          for (Nd4jBackend backend : backends) {
      211              if (tryLoadBackend(backend, logInit)) {
      212                  loadedBackend = backend;
      213                  isLoaded.set(true);
      214                  IS_LOADED = true;
      215                  return backend;

      219          // If no backend found, try dynamic loading

      221              // Recursively try loading again after dynamic loading
      222              return loadBackendInternal();
      223          }
      224  
      225          throw new NoAvailableBackendException(
      226                  "Please ensure that you have an nd4j backend on your classpath. Please see: https://deeplearning4j.konduit.ai/nd4j/backend");

      268      private static boolean tryDynamicLoading() throws NoAvailableBackendException {
      269          // Use atomic boolean to prevent multiple attempts
      270          if (!triedDynamicLoad.compareAndSet(false, true)) {
      271              return false; // Already tried
      272          }
      273  
      274          String[] jarUris = getDynamicLoadPaths();
      275          if (jarUris == null || jarUris.length == 0) {
      276              return false;
      277          }
      278  
      279          // Load all the discoverable URIs
      280          for (String uri : jarUris) {
      281              if (uri != null && !uri.trim().isEmpty()) {
      282                  loadLibrary(new File(uri.trim()));
      283              }
      284          }

      310      /**
      311       * Adds the supplied Java Archive library to java.class.path. This is benign
      312       * if the library is already loaded.
      313       * @param jar the jar file to add
      314       * @throws NoAvailableBackendException
      315       */
      316      public static void loadLibrary(File jar) throws NoAvailableBackendException {
      317          if (jar == null || !jar.exists()) {
      318              log.warn("Jar file does not exist: {}", jar);
      319              return;
      320          }
      321  
      322          String jarPath = jar.getAbsolutePath();
      323  
      324          // Check if already loaded (fast path)
      325          if (loadedLibraries.contains(jarPath)) {
      326              return;
      327          }
      328  
      329          // Synchronize library loading
      330          synchronized (libraryLoadLock) {
      331              // Double-check after acquiring lock
      332              if (loadedLibraries.contains(jarPath)) {
      333                  return;
      334              }
      335  
      336              try {
      337                  URLClassLoader loader = (URLClassLoader) ND4JClassLoading.getNd4jClassloader();
      338                  java.net.URL url = jar.toURI().toURL();
      339  
      340                  // Check if URL is already in classpath
      341                  for (java.net.URL existingUrl : loader.getURLs()) {
      342                      if (existingUrl.equals(url)) {
      343                          loadedLibraries.add(jarPath);
      344                          return;
      345                      }
      346                  }
      347  
      348                  // Add URL to classpath
      349                  java.lang.reflect.Method method = URLClassLoader.class.getDeclaredMethod("addURL", java.net.URL.class);
      350                  method.setAccessible(true);
      351                  method.invoke(loader, url);
      352  
      353                  loadedLibraries.add(jarPath);
      354                  log.debug("Successfully loaded library: {}", jarPath);
      355  
      356              } catch (final NoSuchMethodException | IllegalAccessException |
      357                             java.net.MalformedURLException | java.lang.reflect.InvocationTargetException e) {
      358                  throw new NoAvailableBackendException("Failed to load library: " + jarPath, e);
      359              }
      360          }
      361      }
