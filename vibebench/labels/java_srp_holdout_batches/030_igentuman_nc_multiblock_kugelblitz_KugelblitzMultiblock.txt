      231      public void validateInner() {
      232          if (!outerValid) {
      233              clearStats();
      234              return;
      235          }
      236          if (centerBlockPos == null || centerBlockPos.equals(BlockPos.ZERO)) {
      237              validationResult = ValidationResult.INCOMPLETE;
      238              return;
      239          }
      240  
      241          final int radius = 4;
      242  
      243          for (int x = -radius; x <= radius; x++) {
      244              for (int y = -radius; y <= radius; y++) {
      245                  for (int z = -radius; z <= radius; z++) {
      246                      // Calculate squared distance from center
      247                      double distSquared = x*x + y*y + z*z;
      248  
      249                      // If within sphere (radius² = 16)
      250                      if (distSquared <= radius * radius) {
      251                          BlockPos pos = centerBlockPos.offset(x, y, z);
      252  
      253                          if (!isValidForInner(pos)) {
      254                              validationResult = ValidationResult.WRONG_INNER;
      255                              errorBlockPos = new BlockPos(pos);
      256                              return;
      257                          }
      258  
      259                          processInnerBlock(pos);
      260                      }
      261                  }
      262              }
      263          }
      264          controllerBE().controllerEnabled = isFormed();
      265          controllerBE().fluxRegulators = fluxRegulators();
      266          controllerBE().transformers = transformers();
      267          controllerBE().stabilizers = stabilizers();
      268          controllerBE().blackholePos = getCenter();
      269          controllerBE().setChanged();
      270          validationResult =  ValidationResult.VALID;
      271          errorBlockPos = BlockPos.ZERO;
      272          innerValid = true;

      295      private boolean isFrameValid(Direction.Axis axis, BlockPos center) {
      296          for (int i = -3; i <= 3; i++) {
      297              for (int j = -3; j <= 3; j++) {
      298                  if (Math.abs(i) != 3 && Math.abs(j) != 3) continue;
      299                  if ((i == -3 && j == -3) || (i == -3 && j == 3) ||
      300                          (i == 3 && j == -3) || (i == 3 && j == 3)) continue;
      301                  BlockPos posToCheck;
      302                  if (axis == Direction.Axis.Y) {
      303                      posToCheck = center.offset(i, 0, j);
      304                  } else if (axis == Direction.Axis.X) {
      305                      posToCheck = center.offset(0, i, j);
      306                  } else {
      307                      posToCheck = center.offset(i, j, 0);
      308                  }
      309  
      310                  if (!isValidCorner(posToCheck)) {
      311                      errorBlockPos = new BlockPos(posToCheck);
      312                      return false;
      313                  }
      314                  processOuterBlock(posToCheck);
      315              }
      316          }
      317  
      318          return true;
      319      }

      322      private boolean isWallValid(List<BlockState> initial, List<BlockState> toVerify) {
      323          if (initial.size() != toVerify.size()) {

      327              if (!initial.get(i).is(toVerify.get(i).getBlock())) {

      334      private List<BlockState> getWallBlocks(Direction.Axis axis, BlockPos center) {
      335          List<BlockState> blocks = new ArrayList<>();
      336          for (int i = -2; i <= 2; i++) {
      337              for (int j = -2; j <= 2; j++) {
      338                  BlockPos newPos;
      339                  if (axis == Direction.Axis.Y) {
      340                      newPos = center.offset(i, 0, j);
      341                  } else if (axis == Direction.Axis.X) {
      342                      newPos = center.offset(0, i, j);
      343                  } else {
      344                      newPos = center.offset(i, j, 0);
      345                  }
      346                  BlockState bs = getBlockState(newPos);
      347                  if (bs.isAir() || !isValidForOuter(newPos)) {
      348                      errorBlockPos = new BlockPos(newPos);
      349                      return blocks;
      350                  }
      351                  if (bs.is(KUGELBLITZ_BLOCKS.get("quantum_transformer").get())) {
      352                      transformers++;
      353                  }
      354                  if (bs.is(KUGELBLITZ_BLOCKS.get("quantum_flux_regulator").get())) {
      355                      fluxRegulators++;
      356                  }
      357                  if (bs.is(KUGELBLITZ_BLOCKS.get("event_horizon_stabilizer").get())) {
      358                      stabilizers++;
      359                  }
      360                  blocks.add(bs);
      361                  processOuterBlock(newPos);
      362              }
      363          }
