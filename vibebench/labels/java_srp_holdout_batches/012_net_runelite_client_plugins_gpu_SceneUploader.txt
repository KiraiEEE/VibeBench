      174  	private void uploadZoneLevelRoof(Scene scene, Zone zone, int mzx, int mzz, int level, int roofId, boolean visbelow, GpuIntBuffer vb, GpuIntBuffer ab)
      175  	{

      181  		this.basex = (mzx - offset) << 10;

      184  		for (int xoff = 0; xoff < 8; ++xoff)

      192  				int maplevel = level;
      193  				if (isbridge)

      195  					++maplevel;
      196  				}
      197  
      198  				boolean isvisbelow = maplevel <= 3 && (settings[maplevel][msx][msz] & Constants.TILE_FLAG_VIS_BELOW) != 0;

      200  				{
      201  					continue;
      202  				}

      204  				int rid;
