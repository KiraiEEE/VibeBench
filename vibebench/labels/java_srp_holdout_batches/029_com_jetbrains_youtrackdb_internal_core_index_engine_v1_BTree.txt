      303    @Override
      304    public Stream<RawPair<Object, RID>> iterateEntriesBetween(
      305        DatabaseSessionEmbedded db, Object rangeFrom,
      306        boolean fromInclusive,
      307        Object rangeTo,
      308        boolean toInclusive,
      309        boolean ascSortOrder,
      310        IndexEngineValuesTransformer transformer) {
      311      // "from", "to" are null, then scan whole tree as for infinite range
      312      if (rangeFrom == null && rangeTo == null) {
      313        return mapSVStream(svTree.allEntries());
      314      }
      315  
      316      // "from" could be null, then "to" is not (minor)
      317      final var toKey = convertToCompositeKey(rangeTo);
      318      if (rangeFrom == null) {
      319        return mapSVStream(svTree.iterateEntriesMinor(toKey, toInclusive, ascSortOrder));
      320      }
      321      final var fromKey = convertToCompositeKey(rangeFrom);
      322      // "to" could be null, then "from" is not (major)
      323      if (rangeTo == null) {
      324        return mapSVStream(svTree.iterateEntriesMajor(fromKey, fromInclusive, ascSortOrder));
      325      }
      326      return mapSVStream(
      327          svTree.iterateEntriesBetween(fromKey, fromInclusive, toKey, toInclusive, ascSortOrder));
      328    }

      406    private static Object extractKey(final CompositeKey compositeKey) {
      407      if (compositeKey == null) {
      408        return null;
      409      }
      410      final var keys = compositeKey.getKeys();
      411  
      412      final Object key;
      413      if (keys.size() == 2) {
      414        key = keys.getFirst();
      415      } else {
      416        key = new CompositeKey(keys.subList(0, keys.size() - 1));
      417      }
