      121      private static void showToast(String message, NotificationType type, int seconds,
      122                                    NotificationPosition position, String title) {
      123          SwingUtilities.invokeLater(() -> {
      124              Window mainFrame = getMainFrame();
      125              ToastWindow toast = new ToastWindow(mainFrame, message, title, type, seconds, position);
      126              synchronized (activeToasts) {
      127                  while (activeToasts.size() >= MAX_ACTIVE_TOASTS) {
      128                      activeToasts.get(0).closeQuietly();
      129                  }
      130                  activeToasts.add(toast);
      131                  updateToastPositions();
      132              }
      133              toast.startShow();
      134          });
      135      }

      138          synchronized (activeToasts) {
      139              int offset = 0;
      140              for (ToastWindow toast : activeToasts) {
      141                  toast.updateStackOffset(offset);
      142                  offset += toast.getHeight() + 6;
      143              }
      144          }
      145      }

      223          public void startShow() {
      224              Point startPos = getSlideStartPosition();
      225              setLocation(startPos);
      226              setVisible(true);
      227  
      228              slideStep = 0;
      229              slideTimer = new Timer(SLIDE_INTERVAL, e -> {
      230                  slideStep++;
      231                  double t = (double) slideStep / SLIDE_STEPS;
      232                  double eased = 1 - Math.pow(1 - t, 3);
      233                  int x = startPos.x + (int) ((targetPos.x - startPos.x) * eased);
      234                  int y = startPos.y + (int) ((targetPos.y - startPos.y) * eased);
      235                  setLocation(x, y);
      236                  if (slideStep >= SLIDE_STEPS) {
      237                      ((Timer) e.getSource()).stop();
      238                      setLocation(targetPos);
      239                      if (totalDuration > 0) startAutoCloseTimer();
      240                  }
      241              });
      242              slideTimer.start();
      243          }
