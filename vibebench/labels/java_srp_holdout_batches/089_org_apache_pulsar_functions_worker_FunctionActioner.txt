      150      private String getPackageFile(FunctionMetaData functionMetaData, FunctionDetails functionDetails, int instanceId,
      151                                    Function.PackageLocationMetaData pkgLocation,
      152                                    FunctionDetails.ComponentType componentType)
      153              throws URISyntaxException, IOException, ClassNotFoundException, PulsarAdminException {
      154          String packagePath = pkgLocation.getPackagePath();
      155          boolean isPkgUrlProvided = isFunctionPackageUrlSupported(packagePath);
      156          String packageFile;
      157          if (isPkgUrlProvided && packagePath.startsWith(FILE)) {
      158              if (!packageUrlValidator.isValidPackageUrl(componentType, packagePath)) {
      159                  throw new IllegalArgumentException("Package URL " + packagePath + " is not valid");
      160              }
      161              URL url = new URL(packagePath);
      162              File pkgFile = new File(url.toURI());
      163              packageFile = pkgFile.getAbsolutePath();
      164          } else if (FunctionCommon.isFunctionCodeBuiltin(functionDetails, componentType)) {
      165              File pkgFile = getBuiltinArchive(
      166                      componentType,
      167                      FunctionDetails.newBuilder(functionMetaData.getFunctionDetails()));
      168              packageFile = pkgFile.getAbsolutePath();
      169          } else {
      170              File pkgDir = new File(workerConfig.getDownloadDirectory(),
      171                      getDownloadPackagePath(functionMetaData, instanceId));
      172              pkgDir.mkdirs();
      173              File pkgFile = new File(
      174                      pkgDir,
      175                      new File(getDownloadFileName(functionMetaData.getFunctionDetails(),
      176                              pkgLocation)).getName());
      177              downloadFile(pkgFile, isPkgUrlProvided, functionMetaData, instanceId, pkgLocation, componentType);
      178              packageFile = pkgFile.getAbsolutePath();
      179          }
      180          return packageFile;
      181      }

      235      private void downloadFile(File pkgFile, boolean isPkgUrlProvided, FunctionMetaData functionMetaData,
      236                                int instanceId, Function.PackageLocationMetaData pkgLocation,
      237                                FunctionDetails.ComponentType componentType)
      238              throws IOException, PulsarAdminException {
      239  
      240          FunctionDetails details = functionMetaData.getFunctionDetails();
      241          File pkgDir = pkgFile.getParentFile();
      242  
      243          if (pkgFile.exists()) {
      244              log.warn("Function package exists already {} deleting it", pkgFile);
      245              pkgFile.delete();
      246          }
      247  
      248          File tempPkgFile;
      249          do {
      250              tempPkgFile = new File(
      251                      pkgDir,
      252                      pkgFile.getName() + "." + instanceId + "." + UUID.randomUUID());
      253          } while (tempPkgFile.exists() || !tempPkgFile.createNewFile());
      254          String pkgLocationPath = pkgLocation.getPackagePath();
      255          boolean downloadFromHttp = isPkgUrlProvided && pkgLocationPath.startsWith(HTTP);
      256          boolean downloadFromPackageManagementService = isPkgUrlProvided && hasPackageTypePrefix(pkgLocationPath);
      257          log.info("{}/{}/{} Function package file {} will be downloaded from {}", tempPkgFile, details.getTenant(),
      258                  details.getNamespace(), details.getName(),
      259                  downloadFromHttp ? pkgLocationPath : pkgLocation);
      260  
      261          if (downloadFromHttp) {
      262              if (!packageUrlValidator.isValidPackageUrl(componentType, pkgLocationPath)) {
      263                  throw new IllegalArgumentException("Package URL " + pkgLocationPath + " is not valid");
      264              }
      265              FunctionCommon.downloadFromHttpUrl(pkgLocationPath, tempPkgFile);
      266          } else if (downloadFromPackageManagementService) {
      267              getPulsarAdmin().packages().download(pkgLocationPath, tempPkgFile.getPath());
      268          } else {
      269              try (FileOutputStream tempPkgFos = new FileOutputStream(tempPkgFile)) {
      270                  WorkerUtils.downloadFromBookkeeper(
      271                          dlogNamespace,
      272                          tempPkgFos,
      273                          pkgLocationPath);
      274              }
      275          }
      276  
      277          try {
      278              // create a hardlink, if there are two concurrent createLink operations, one will fail.
      279              // this ensures one instance will successfully download the package.
      280              try {
      281                  Files.createLink(
      282                          Paths.get(pkgFile.toURI()),
      283                          Paths.get(tempPkgFile.toURI()));
      284                  log.info("Function package file is linked from {} to {}",
      285                          tempPkgFile, pkgFile);
      286              } catch (FileAlreadyExistsException faee) {
      287                  // file already exists
      288                  log.warn("Function package has been downloaded from {} and saved at {}",
      289                          pkgLocation, pkgFile);
      290              }
      291          } finally {
      292              tempPkgFile.delete();
      293          }
      294  
      295          if (details.getRuntime() == Function.FunctionDetails.Runtime.GO && !pkgFile.canExecute()) {
      296              pkgFile.setExecutable(true);
      297              log.info("Golang function package file {} is set to executable", pkgFile);
      298          }
      299      }

      415      private Supplier<Actions.ActionResult> getDeleteSubscriptionSupplier(
      416        String topic, boolean isRegex, String subscriptionName) {
      417          return () -> {
      418              try {
      419                  if (isRegex) {
      420                      pulsarAdmin.namespaces().unsubscribeNamespace(TopicName
      421                        .get(topic).getNamespace(), subscriptionName);
      422                  } else {
      423                      pulsarAdmin.topics().deleteSubscription(topic,
      424                        subscriptionName);
      425                  }
      426              } catch (PulsarAdminException e) {
      427                  if (e instanceof PulsarAdminException.NotFoundException) {
      428                      return Actions.ActionResult.builder()
      429                        .success(true)
      430                        .build();
      431                  } else {
      432                      // for debugging purposes
      433                      List<Map<String, String>> existingConsumers = Collections.emptyList();
      434                      SubscriptionStats sub = null;
      435                      try {
      436                          TopicStats stats = pulsarAdmin.topics().getStats(topic);
      437                          sub = stats.getSubscriptions().get(subscriptionName);
      438                          if (sub != null) {
      439                              existingConsumers = sub.getConsumers().stream()
      440                                .map(consumerStats -> consumerStats.getMetadata())
      441                                .collect(Collectors.toList());
      442                          }
      443                      } catch (PulsarAdminException e1) {
      444  
      445                      }
      446  
      447                      String errorMsg = e.getHttpError() != null ? e.getHttpError() : e.getMessage();
      448                      String finalErrorMsg;
      449                      if (sub != null) {
      450                          try {
      451                              finalErrorMsg = String.format("%s - existing consumers: %s",
      452                                errorMsg, ObjectMapperFactory.getMapper().writer().writeValueAsString(sub));
      453                          } catch (JsonProcessingException jsonProcessingException) {
      454                              finalErrorMsg = errorMsg;
      455                          }
      456                      } else {
      457                          finalErrorMsg = errorMsg;
      458                      }
      459                      return Actions.ActionResult.builder()
      460                        .success(false)
      461                        .errorMsg(finalErrorMsg)
      462                        .build();
      463                  }
      464              }
      465  
      466              return Actions.ActionResult.builder()
      467                .success(true)
      468                .build();
      469          };
      470      }
