      247      public static String writeValueAsPrettyString(Object value) throws JsonProcessingException {
      248          // XXX(trustin): prettyMapper.writeValueAsString() does not respect the custom pretty printer

      338      public static JsonNode mergeTree(Iterable<JsonNode> jsonNodes) {
      339          requireNonNull(jsonNodes, "jsonNodes");
      340          final int size = Iterables.size(jsonNodes);
      341          checkArgument(size > 0, "jsonNodes is empty.");
      342          final Iterator<JsonNode> it = jsonNodes.iterator();
      343          final JsonNode first = it.next();
      344          JsonNode merged = first.deepCopy();
      345  
      346          final StringBuilder fieldNameAppender = new StringBuilder("/");
      347          while (it.hasNext()) {
      348              final JsonNode addition = it.next();
      349              merged = traverse(merged, addition, fieldNameAppender, true, true);
      350          }
      351  
      352          if (size > 2) {
      353              // Traverse once more to find the mismatched value between the first and the merged node.
      354              traverse(first, merged, fieldNameAppender, false, true);
      355          }
      356          return merged;
      357      }

      359      private static JsonNode traverse(JsonNode base, JsonNode update, StringBuilder fieldNameAppender,
      360                                       boolean isMerging, boolean isRoot) {
      361          if (base.isObject() && update.isObject()) {
      362              final ObjectNode baseObject = (ObjectNode) base;
      363              final Iterator<String> fieldNames = update.fieldNames();
      364              while (fieldNames.hasNext()) {
      365                  final String fieldName = fieldNames.next();
      366                  final JsonNode baseValue = baseObject.get(fieldName);
      367                  final JsonNode updateValue = update.get(fieldName);
      368  
      369                  if (baseValue == null || baseValue.isNull() || updateValue.isNull()) {
      370                      if (isMerging) {
      371                          baseObject.set(fieldName, updateValue);
      372                      }
      373                      continue;
      374                  }
      375  
      376                  final int length = fieldNameAppender.length();
      377                  // Append the filed name and traverse the child.
      378                  fieldNameAppender.append(fieldName);
      379                  fieldNameAppender.append('/');
      380                  final JsonNode traversed =
      381                          traverse(baseValue, updateValue, fieldNameAppender, isMerging, false);
      382                  if (isMerging) {
      383                      baseObject.set(fieldName, traversed);
      384                  }
      385                  // Remove the appended filed name above.
      386                  fieldNameAppender.delete(length, fieldNameAppender.length());
      387              }
      388  
      389              return base;
      390          }
      391  
      392          if (isRoot || (base.getNodeType() != update.getNodeType() && (!base.isNull() || !update.isNull()))) {
      393              throw new QueryExecutionException("Failed to merge tree. " + fieldNameAppender +
      394                                                " type: " + update.getNodeType() +
      395                                                " (expected: " + (isRoot ? OBJECT : base.getNodeType()) + ')');
      396          }
      397  
      398          return update;
      399      }
