      124        CommandProcessorException cpe = null;

      175    public CommandProcessorResponse run() throws CommandProcessorException {
      176      executionIndex = 0;
      177      int maxExecutions = 1 + coreDriver.getConf().getIntVar(ConfVars.HIVE_QUERY_MAX_REEXECUTION_COUNT);
      178  
      179      while (true) {
      180        executionIndex++;
      181  
      182        for (IReExecutionPlugin p : plugins) {
      183          p.beforeExecute(executionIndex, explainReOptimization);
      184        }
      185        coreDriver.getContext().setExecutionIndex(executionIndex);
      186        LOG.info("Execution #{} of query", executionIndex);
      187        CommandProcessorResponse cpr = null;
      188        CommandProcessorException cpe = null;
      189        try {
      190          cpr = coreDriver.run();
      191        } catch (CommandProcessorException e) {
      192          cpe = e;
      193        }
      194  
      195        PlanMapper oldPlanMapper = coreDriver.getPlanMapper();
      196        boolean success = cpr != null;
      197        plugins.forEach(p -> p.afterExecute(oldPlanMapper, success));
      198  
      199        boolean shouldReExecute = explainReOptimization && executionIndex==1;
      200        shouldReExecute |= cpr == null && plugins.stream().anyMatch(p -> p.shouldReExecute(executionIndex));
      201  
      202        LOG.info("Re-execution decision is made according to: executionIndex: {}, maxExecutions: {}, shouldReExecute: {}",
      203            executionIndex, maxExecutions, shouldReExecute);
      204        if (executionIndex >= maxExecutions || !shouldReExecute) {
      205          if (cpr != null) {
      206            return cpr;
      207          } else {
      208            throw cpe;
      209          }
      210        }
      211        LOG.info("Preparing to re-execute query");
      212        plugins.forEach(IReExecutionPlugin::prepareToReExecute);
      213  
      214        try {
      215          coreDriver.compileAndRespond(currentQuery);
      216        } catch (CommandProcessorException e) {
      217          LOG.error("Recompilation of the query failed; this is unexpected.");
      218          // FIXME: somehow place pointers that re-execution compilation have failed; the query have been successfully compiled before?
      219          throw e;
      220        }
      221  
      222        PlanMapper newPlanMapper = coreDriver.getPlanMapper();
      223        if (!explainReOptimization &&
      224            !plugins.stream().anyMatch(p -> p.shouldReExecuteAfterCompile(executionIndex, oldPlanMapper, newPlanMapper))) {
      225          LOG.info("re-running the query would probably not yield better results; returning with last error");
      226          // FIXME: retain old error; or create a new one?
      227          return cpr;
      228        }
      229      }
