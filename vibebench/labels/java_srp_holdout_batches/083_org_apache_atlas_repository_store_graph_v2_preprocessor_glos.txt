       86      private void processCreateTerm(AtlasEntity entity, AtlasVertex vertex) throws AtlasBaseException {
       87          AtlasPerfMetrics.MetricRecorder metricRecorder = RequestContext.get().startMetricRecord("processCreateTerm");
       88          String termName = (String) entity.getAttribute(NAME);
       89          String termQName = vertex.getProperty(QUALIFIED_NAME, String.class);
       90  
       91          if (StringUtils.isEmpty(termName) || isNameInvalid(termName)) {
       92              throw new AtlasBaseException(AtlasErrorCode.INVALID_DISPLAY_NAME);
       93          }
       94  
       95          String glossaryQName = (String) anchor.getAttribute(QUALIFIED_NAME);
       96  
       97          termExists(termName, glossaryQName);
       98  
       99          String parentQname = validateAndGetCategory(entity);
      100  
      101          String lexicographicalSortOrder = (String) entity.getAttribute(LEXICOGRAPHICAL_SORT_ORDER);
      102          if(StringUtils.isEmpty(lexicographicalSortOrder)){
      103              assignNewLexicographicalSortOrder(entity, glossaryQName, parentQname, this.discovery);
      104          } else {
      105              isValidLexoRank(lexicographicalSortOrder, glossaryQName, parentQname, this.discovery);
      106          }
      107  
      108          entity.setAttribute(QUALIFIED_NAME, createQualifiedName());
      109          AtlasAuthorizationUtils.verifyAccess(new AtlasEntityAccessRequest(typeRegistry, AtlasPrivilege.ENTITY_CREATE, new AtlasEntityHeader(entity)),
      110                  "create entity: type=", entity.getTypeName());
      111  
      112          RequestContext.get().endMetricRecord(metricRecorder);
      113      }

      115      private void processUpdateTerm(AtlasEntity entity, AtlasVertex vertex) throws AtlasBaseException {
      116          AtlasPerfMetrics.MetricRecorder metricRecorder = RequestContext.get().startMetricRecord("processUpdateTerm");
      117          String termName = (String) entity.getAttribute(NAME);
      118          String vertexName = vertex.getProperty(NAME, String.class);
      119          String termGuid = entity.getGuid();
      120  
      121          if (StringUtils.isEmpty(termName) || isNameInvalid(termName)) {
      122              throw new AtlasBaseException(AtlasErrorCode.INVALID_DISPLAY_NAME);
      123          }
      124  
      125          String parentQname = validateAndGetCategory(entity);
      126  
      127          AtlasEntity storedTerm = entityRetriever.toAtlasEntity(vertex);
      128          AtlasRelatedObjectId currentGlossary = (AtlasRelatedObjectId) storedTerm.getRelationshipAttribute(ANCHOR);
      129          AtlasEntityHeader currentGlossaryHeader = entityRetriever.toAtlasEntityHeader(currentGlossary.getGuid());
      130          String currentGlossaryQualifiedName = (String) currentGlossaryHeader.getAttribute(QUALIFIED_NAME);
      131  
      132          String termQualifiedName = vertex.getProperty(QUALIFIED_NAME, String.class);
      133  
      134          String newGlossaryQualifiedName = (String) anchor.getAttribute(QUALIFIED_NAME);
      135  
      136          String lexicographicalSortOrder = (String) entity.getAttribute(LEXICOGRAPHICAL_SORT_ORDER);
      137          if(StringUtils.isNotEmpty(lexicographicalSortOrder)) {
      138              isValidLexoRank(lexicographicalSortOrder, newGlossaryQualifiedName, parentQname, this.discovery);
      139          } else {
      140              entity.removeAttribute(LEXICOGRAPHICAL_SORT_ORDER);
      141          }
      142  
      143          if (!currentGlossaryQualifiedName.equals(newGlossaryQualifiedName)){
      144              //Auth check
      145              isAuthorized(currentGlossaryHeader, anchor);
      146  
      147              String updatedTermQualifiedName = moveTermToAnotherGlossary(entity, vertex, currentGlossaryQualifiedName, newGlossaryQualifiedName, termQualifiedName);
      148  
      149              if (checkEntityTermAssociation(termQualifiedName)) {
      150                  if (taskManagement != null && DEFERRED_ACTION_ENABLED) {
      151                      createAndQueueTask(UPDATE_ENTITY_MEANINGS_ON_TERM_UPDATE, vertexName, termName, termQualifiedName, updatedTermQualifiedName, vertex);
      152                  } else {
      153                      updateMeaningsAttributesInEntitiesOnTermUpdate(vertexName, termName, termQualifiedName, updatedTermQualifiedName, termGuid);
      154                  }
      155              }
      156  
      157          } else {
      158  
      159              if (!vertexName.equals(termName)) {
      160                  termExists(termName, newGlossaryQualifiedName);
      161              }
      162  
      163              entity.setAttribute(QUALIFIED_NAME, termQualifiedName);
      164  
      165              if (!termName.equals(vertexName) && checkEntityTermAssociation(termQualifiedName)) {
      166                  if (taskManagement != null && DEFERRED_ACTION_ENABLED) {
      167                      createAndQueueTask(UPDATE_ENTITY_MEANINGS_ON_TERM_UPDATE, vertexName, termName, termQualifiedName, null, vertex);
      168                  } else {
      169                      updateMeaningsAttributesInEntitiesOnTermUpdate(vertexName, termName, termQualifiedName, null, termGuid);
      170                  }
      171              }
      172          }
      173  
      174          RequestContext.get().endMetricRecord(metricRecorder);
      175      }

      177      private String validateAndGetCategory(AtlasEntity entity) throws AtlasBaseException {
      178          String glossaryQualifiedName = (String) anchor.getAttribute(QUALIFIED_NAME);
      179          String categoryQualifiedName = null;
      180  
      181          if (entity.hasRelationshipAttribute(ATTR_CATEGORIES) && entity.getRelationshipAttribute(ATTR_CATEGORIES) != null) {
      182              List<AtlasObjectId> categories = (List<AtlasObjectId>) entity.getRelationshipAttribute(ATTR_CATEGORIES);
      183  
      184              if (CollectionUtils.isNotEmpty(categories)) {
      185                  AtlasObjectId category = categories.get(0);
      186  
      187                  if (category.getUniqueAttributes() != null && category.getUniqueAttributes().containsKey(QUALIFIED_NAME)) {
      188                      categoryQualifiedName = (String) category.getUniqueAttributes().get(QUALIFIED_NAME);
      189                  } else {
      190                      AtlasVertex categoryVertex = entityRetriever.getEntityVertex(category.getGuid());
      191                      categoryQualifiedName = categoryVertex.getProperty(QUALIFIED_NAME, String.class);
      192                  }
      193  
      194                  if (!categoryQualifiedName.endsWith(glossaryQualifiedName)) {
      195                      throw new AtlasBaseException(AtlasErrorCode.BAD_REQUEST, "Passed category doesn't belongs to Passed Glossary");
      196                  }
      197              }
      198          }
      199          return categoryQualifiedName;
      200      }

      202      public String moveTermToAnotherGlossary(AtlasEntity entity, AtlasVertex vertex,
      203                                             String sourceGlossaryQualifiedName,
      204                                             String targetGlossaryQualifiedName,
      205                                             String currentTermQualifiedName) throws AtlasBaseException {
      206  
      207          //check duplicate term name
      208          termExists((String) entity.getAttribute(NAME), targetGlossaryQualifiedName);
      209  
      210  
      211          String updatedQualifiedName = currentTermQualifiedName.replace(sourceGlossaryQualifiedName, targetGlossaryQualifiedName);
      212  
      213          //qualifiedName
      214          entity.setAttribute(QUALIFIED_NAME, updatedQualifiedName);
      215  
      216          // __categories
      217          /*  if category is not passed in relationshipAttributes, check
      218              whether category belongs to target glossary, if not throw an exception
      219           */
      220          if (!entity.hasRelationshipAttribute(ATTR_CATEGORIES)) {
      221              Iterator<AtlasVertex> categoriesItr = getActiveParentVertices(vertex, CATEGORY_TERMS_EDGE_LABEL);
      222  
      223              if (categoriesItr.hasNext()) {
      224                  AtlasVertex categoryVertex = categoriesItr.next();
      225  
      226                  String categoryQualifiedName = categoryVertex.getProperty(QUALIFIED_NAME, String.class);
      227  
      228                  if (!categoryQualifiedName.endsWith(targetGlossaryQualifiedName)) {
      229                      throw new AtlasBaseException(AtlasErrorCode.BAD_REQUEST, "Passed category doesn't belongs to Passed Glossary");
      230                  }
      231              }
      232          }
      233  
      234          return updatedQualifiedName;

      241      private void setAnchor(AtlasEntity entity, EntityMutationContext context) throws AtlasBaseException {
      242          AtlasPerfMetrics.MetricRecorder metricRecorder = RequestContext.get().startMetricRecord("TermPreProcessor.setAnchor");
      243          if (anchor == null) {
      244              AtlasObjectId objectId = (AtlasObjectId) entity.getRelationshipAttribute(ANCHOR);
      245  
      246              if (StringUtils.isNotEmpty(objectId.getGuid())) {
      247                  AtlasVertex vertex = context.getVertex(objectId.getGuid());
      248  
      249                  if (vertex == null) {
      250                      anchor = entityRetriever.toAtlasEntityHeader(objectId.getGuid());
      251                  } else {
      252                      anchor = entityRetriever.toAtlasEntityHeader(vertex);
      253                  }
      254  
      255              } else if (MapUtils.isNotEmpty(objectId.getUniqueAttributes()) &&
      256                      StringUtils.isNotEmpty( (String) objectId.getUniqueAttributes().get(QUALIFIED_NAME))) {
      257                  anchor = new AtlasEntityHeader(objectId.getTypeName(), objectId.getUniqueAttributes());
      258  
      259              }
      260          }
      261          RequestContext.get().endMetricRecord(metricRecorder);
      262      }
