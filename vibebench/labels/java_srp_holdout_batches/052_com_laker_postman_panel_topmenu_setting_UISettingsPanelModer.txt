      407      private void toggleSidebarTabVisibility(int index) {
      408          SidebarTabSettingItem item = sidebarTabListModel.get(index);
      409          if (item.visible && countVisibleSidebarTabs() == 1) {
      410              NotificationUtil.showError(I18nUtil.getMessage(MessageKeys.SETTINGS_GENERAL_SIDEBAR_TABS_AT_LEAST_ONE));
      411              return;
      412          }
      413          item.visible = !item.visible;
      414          sidebarTabList.repaint();
      415          syncSidebarTabsState();
      416      }
