      266      private void discoverPackages(final Map<String, String> packages,
      267              final Map<String, String> devPackages) {
      268          try {
      269              Class<? extends Annotation> loadedAnnotation = getFinder()
      270                      .loadClass(NpmPackage.class.getName());
      271              Set<Class<?>> annotatedClasses = getFinder()
      272                      .getAnnotatedClasses(loadedAnnotation);
      273              Set<String> logs = new HashSet<>();
      274              for (Class<?> clazz : annotatedClasses) {
      275                  classes.add(clazz.getName());
      276                  List<? extends Annotation> packageAnnotations = annotationFinder
      277                          .apply(clazz, loadedAnnotation);
      278                  packageAnnotations.forEach(annotation -> {
      279                      String value = getAnnotationValueAsString(annotation,
      280                              VALUE);
      281                      String version = getAnnotationValueAsString(annotation,
      282                              VERSION);
      283                      String[] npmAssets = (String[]) getAnnotationValue(
      284                              annotation, ASSETS);
      285  
      286                      boolean dev = getAnnotationValueAsBoolean(annotation, DEV);
      287                      logs.add(value + " " + version + " " + clazz.getName());
      288                      Map<String, String> result = dev ? devPackages : packages;
      289                      if (npmAssets.length > 0) {
      290                          List<String> assetsList = Arrays.asList(npmAssets);
      291                          if (!assetsList.isEmpty()) {
      292                              if (dev) {
      293                                  addValues(devAssets, value, assetsList);
      294                              } else {
      295                                  addValues(assets, value, assetsList);
      296                              }
      297                          }
      298                      }
      299                      if (result.containsKey(value)
      300                              && !result.get(value).equals(version)) {
      301                          String foundVersions = "[" + result.get(value) + ", "
      302                                  + version + "]";
      303                          getLogger().warn(
      304                                  "Multiple npm versions for {} found:  {}. First version found '{}' will be considered.",
      305                                  value, foundVersions, result.get(value));
      306                      } else {
      307                          result.put(value, version);
      308                      }
      309                  });
      310              }
      311              debug("npm dependencies", logs);
      312          } catch (ClassNotFoundException exception) {
      313              throw new IllegalStateException(
      314                      COULD_NOT_LOAD_ERROR_MSG + NpmPackage.class.getName(),
      315                      exception);
      316          }
      317      }

      319      private void discoverCss() {
      320          try {
      321              Class<? extends Annotation> loadedAnnotation = getFinder()
      322                      .loadClass(CssImport.class.getName());
      323              Set<Class<?>> annotatedClasses = getFinder()
      324                      .getAnnotatedClasses(loadedAnnotation);
      325              var themeCss = new LinkedHashSet<CssData>();
      326              var globalCss = new LinkedHashSet<CssData>();
      327              var webComponentCss = new LinkedHashSet<CssData>();
      328  
      329              for (Class<?> clazz : annotatedClasses) {
      330                  classes.add(clazz.getName());
      331                  var isAppShellClass = AppShellConfigurator.class
      332                          .isAssignableFrom(clazz);
      333                  var isWebcomponentExporter = WebComponentExporter.class
      334                          .isAssignableFrom(clazz);
      335                  var isThemeClass = AbstractTheme.class.isAssignableFrom(clazz);
      336                  if (isThemeClass && (themeDefinition == null
      337                          || !clazz.equals(themeDefinition.getTheme()))) {
      338                      // Do not add css from all found theme classes,
      339                      // only defined theme.
      340                      continue;
      341                  }
      342                  List<? extends Annotation> imports = annotationFinder
      343                          .apply(clazz, loadedAnnotation);
      344                  imports.stream()
      345                          .forEach(imp -> ((isAppShellClass || isThemeClass)
      346                                  ? themeCss
      347                                  : isWebcomponentExporter ? webComponentCss
      348                                          : globalCss)
      349                                  .add(createCssData(imp)));
      350              }
      351              themeCssData.addAll(themeCss);
      352              cssData.addAll(globalCss);
      353              webComponentCssData.addAll(webComponentCss);
      354          } catch (ClassNotFoundException exception) {
      355              throw new IllegalStateException(
      356                      COULD_NOT_LOAD_ERROR_MSG + CssData.class.getName(),
      357                      exception);
      358          }
      359      }

      361      private <T extends Annotation> void collectScripts(
      362              LinkedHashSet<String> target, LinkedHashSet<String> targetDevOnly,
      363              Class<T> annotationType) {
      364          try {
      365              Set<String> logs = new HashSet<>();
      366              Class<? extends Annotation> loadedAnnotation = getFinder()
      367                      .loadClass(annotationType.getName());
      368              Set<Class<?>> annotatedClasses = getFinder()
      369                      .getAnnotatedClasses(loadedAnnotation);
      370  
      371              annotatedClasses.stream()
      372                      .filter(c -> !isDisabledExperimentalClass(c.getName()))
      373                      .forEach(clazz -> annotationFinder
      374                              .apply(clazz, loadedAnnotation).forEach(ann -> {
      375                                  String value = getAnnotationValueAsString(ann,
      376                                          VALUE);
      377                                  Boolean developmentOnly = getAnnotationValueAsBoolean(
      378                                          ann, DEVELOPMENT_ONLY);
      379  
      380                                  classes.add(clazz.getName());
      381  
      382                                  if (isNotActiveThemeClass(clazz)) {
      383                                      // The scanner will discover all theme
      384                                      // classes (Lumo and Material)
      385                                      // but should include imports only from the
      386                                      // active one
      387                                      return;
      388                                  }
      389  
      390                                  if (developmentOnly) {
      391                                      targetDevOnly.add(value);
      392                                  } else {
      393                                      target.add(value);
      394                                  }
      395  
      396                                  logs.add(value + " " + clazz);
      397                              }));
      398  
      399              debug("@" + annotationType.getSimpleName(), logs);
      400          } catch (ClassNotFoundException exception) {
      401              throw new IllegalStateException(
      402                      COULD_NOT_LOAD_ERROR_MSG + annotationType.getName(),
      403                      exception);
      404          }
