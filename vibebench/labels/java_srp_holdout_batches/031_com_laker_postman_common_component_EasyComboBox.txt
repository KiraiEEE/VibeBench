      108      /**
      109       * 核心方法：重写 getPreferredSize()。
      110       * <p>
      111       * 每次布局管理器询问尺寸时，我们都在这里计算——
      112       * 此时组件已经在窗口中，super.getPreferredSize() 由 FlatComboBoxUI 计算，
      113       * 完全准确（含 HiDPI、cell insets、箭头按钮、边框）。
      114       * <p>
      115       * DYNAMIC 模式的关键：临时只放一个选项（当前选中项）让 UI 计算宽度，
      116       * 这样 BasicComboBoxUI 就不会遍历所有选项取最宽值。
      117       */
      118      @Override
      119      public Dimension getPreferredSize() {
      120          if (widthMode == WidthMode.FIXED_CUSTOM && customWidth > 0) {
      121              Dimension d = super.getPreferredSize();
      122              return new Dimension(customWidth, d.height);
      123          }
      124  
      125          if (widthMode == WidthMode.FIXED_MAX) {
      126              // 直接让 super 遍历所有选项取最宽值，这正是 BasicComboBoxUI 的默认行为
      127              return super.getPreferredSize();
      128          }
      129  
      130          // DYNAMIC：只用当前选中项计算宽度
      131          return calcDynamicSize();
      132      }

      140      @Override
      141      public Dimension getMinimumSize() {
      142          Dimension d = getPreferredSize();

      146      /**
      147       * 临时用只含当前选中项的 ComboBoxModel 替换，让 super.getPreferredSize()
      148       * 只量当前项的宽度，然后还原。全程在 EDT 上同步执行，无线程问题。
      149       */
      150      private Dimension calcDynamicSize() {
      151          Object selected = getSelectedItem();
      152  
      153          // 构造只有当前选中项的临时 model
      154          DefaultComboBoxModel<Object> tempModel = new DefaultComboBoxModel<>();
      155          if (selected != null) {
      156              tempModel.addElement(selected);
      157              tempModel.setSelectedItem(selected);
      158          }
      159  
      160          // 暂存真实 model，换成临时 model
      161          ComboBoxModel<E> realModel = getModel();
      162          @SuppressWarnings("unchecked")
      163          ComboBoxModel<E> castedTemp = (ComboBoxModel<E>) tempModel;
      164          // 直接操作父类，绕过子类可能的监听器
      165          super.setModel(castedTemp);
      166  
      167          Dimension d = super.getPreferredSize();
      168  
      169          // 还原真实 model
      170          super.setModel(realModel);
      171  
      172          return d;
      173      }
