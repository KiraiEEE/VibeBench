      105    /**
      106     * For every graph declared in the Gremlin settings' {@code graphs} section, reads the matching
      107     * {@code .properties} file, extracts {@value ArcadeGraph#CONFIG_DIRECTORY}, and makes sure the
      108     * database is registered with (and, if absent, created by) ArcadeDBServer before the Gremlin
      109     * server starts.
      110     */
      111    private void initPreConfiguredDatabases(final Settings settings) {
      112      if (settings.graphs == null || settings.graphs.isEmpty())
      113        return;
      114  
      115      for (final Map.Entry<String, String> entry : settings.graphs.entrySet()) {
      116        final String graphName = entry.getKey();
      117        final String propertiesPath = entry.getValue();
      118  
      119        try {
      120          final String resolvedPath = resolveConfigPath(server.getRootPath(), propertiesPath);
      121          final File propertiesFile = new File(resolvedPath);
      122          if (!propertiesFile.exists()) {
      123            LogManager.instance().log(this, Level.WARNING,
      124                "Gremlin graph '%s': properties file '%s' not found — skipping database init", graphName, resolvedPath);
      125            continue;
      126          }
      127  
      128          final Properties props = new Properties();
      129          try (final FileInputStream fis = new FileInputStream(propertiesFile)) {
      130            props.load(fis);
      131          }
      132  
      133          final String dbDirectory = props.getProperty(ArcadeGraph.CONFIG_DIRECTORY);
      134          if (dbDirectory == null) {
      135            LogManager.instance().log(this, Level.WARNING,
      136                "Gremlin graph '%s': property '%s' not found in '%s' — skipping database init",
      137                graphName, ArcadeGraph.CONFIG_DIRECTORY, resolvedPath);
      138            continue;
      139          }
      140  
      141          // Derive the database name from the last path component (e.g. "./databases/graph" → "graph")
      142          final String dbName = new File(dbDirectory.replaceAll("/+$", "")).getName();
      143          if (dbName.isEmpty()) {
      144            LogManager.instance().log(this, Level.WARNING,
      145                "Gremlin graph '%s': cannot derive database name from directory '%s' — skipping",
      146                graphName, dbDirectory);
      147            continue;
      148          }
      149  
      150          if (!server.existsDatabase(dbName)) {
      151            // createIfNotExists=true: creates the database if it doesn't exist yet
      152            server.getDatabase(dbName, true, true);
      153            LogManager.instance().log(this, Level.INFO,
      154                "Gremlin graph '%s': created/opened database '%s'", graphName, dbName);
      155          }
      156        } catch (final Exception e) {
      157          LogManager.instance().log(this, Level.WARNING, "Gremlin graph '%s': error initializing database", e, graphName);
      158        }
      159      }
      160    }

      162    private static String resolveConfigPath(final String rootPath, final String path) {
      163      final File f = new File(path);
      164      if (f.isAbsolute())
      165        return path;
      166      return new File(rootPath, path).getPath();
      167    }
