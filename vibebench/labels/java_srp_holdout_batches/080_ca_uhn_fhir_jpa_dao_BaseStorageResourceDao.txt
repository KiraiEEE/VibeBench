      252  	private IBaseResource applyPatchToResource(
      253  			PatchTypeEnum thePatchType,
      254  			String thePatchBody,
      255  			IBaseParameters theFhirPatchBody,
      256  			IBaseResource theResourceToUpdate) {
      257  		IBaseResource destination;
      258  		switch (thePatchType) {
      259  			case JSON_PATCH:
      260  				destination = JsonPatchUtils.apply(getContext(), theResourceToUpdate, thePatchBody);
      261  				break;
      262  			case XML_PATCH:
      263  				destination = XmlPatchUtils.apply(getContext(), theResourceToUpdate, thePatchBody);
      264  				break;
      265  			case FHIR_PATCH_XML:
      266  			case FHIR_PATCH_JSON:
      267  			default:
      268  				IBaseParameters fhirPatchJson = theFhirPatchBody;
      269  				new FhirPatch(getContext()).apply(theResourceToUpdate, fhirPatchJson);
      270  				destination = theResourceToUpdate;
      271  				break;
      272  		}
      273  		return destination;
      274  	}

      330  	protected DaoMethodOutcome doUpdateForUpdateOrPatch(UpdateParameters<T> theUpdateParameters) {
      331  
      332  		if (theUpdateParameters.getResourceIdToUpdate().hasVersionIdPart()) {
      333  			boolean versionMatches = theUpdateParameters.getResourceIdToUpdate().isVersionIdPartValidLong()
      334  					&& Long.parseLong(
      335  									theUpdateParameters.getResourceIdToUpdate().getVersionIdPart())
      336  							== theUpdateParameters.getEntity().getVersion();
      337  			if (!versionMatches) {
      338  				throw new ResourceVersionConflictException(Msg.code(989) + "Trying to update "
      339  						+ theUpdateParameters.getResourceIdToUpdate() + " but this is not the current version");
      340  			}
      341  		}
      342  
      343  		if (theUpdateParameters.getResourceIdToUpdate().hasResourceType()
      344  				&& !theUpdateParameters
      345  						.getResourceIdToUpdate()
      346  						.getResourceType()
      347  						.equals(getResourceName())) {
      348  			throw new UnprocessableEntityException(Msg.code(990) + "Invalid resource ID["
      349  					+ theUpdateParameters.getEntity().getIdDt().toUnqualifiedVersionless() + "] of type["
      350  					+ theUpdateParameters.getEntity().getResourceType()
      351  					+ "] - Does not match expected [" + getResourceName() + "]");
      352  		}
      353  
      354  		IBaseResource oldResource;
      355  		if (getStorageSettings().isMassIngestionMode()
      356  				&& !theUpdateParameters.shouldForcePopulateOldResourceForProcessing()) {
      357  			oldResource = null;
      358  		} else {
      359  			oldResource = getStorageResourceParser().toResource(theUpdateParameters.getEntity(), false);
      360  		}
      361  
      362  		/*
      363  		 * Mark the entity as not deleted - This is also done in the actual updateInternal()
      364  		 * method later on so it usually doesn't matter whether we do it here, but in the
      365  		 * case of a transaction with multiple PUTs we don't get there until later so
      366  		 * having this here means that a transaction can have a reference in one
      367  		 * resource to another resource in the same transaction that is being
      368  		 * un-deleted by the transaction. Wacky use case, sure. But it's real.
      369  		 *
      370  		 * See SystemProviderR4Test#testTransactionReSavesPreviouslyDeletedResources
      371  		 * for a test that needs this.
      372  		 */
      373  		boolean wasDeleted = theUpdateParameters.getEntity().isDeleted();
      374  		theUpdateParameters.getEntity().setNotDeleted();
      375  
      376  		/*
      377  		 * If we aren't indexing, that means we're doing this inside a transaction.
      378  		 * The transaction will do the actual storage to the database a bit later on,
      379  		 * after placeholder IDs have been replaced, by calling {@link #updateInternal}
      380  		 * directly. So we just bail now.
      381  		 */
      382  		if (!theUpdateParameters.shouldPerformIndexing()) {
      383  			theUpdateParameters
      384  					.getResource()
      385  					.setId(theUpdateParameters.getEntity().getIdDt().getValue());
      386  			DaoMethodOutcome outcome = toMethodOutcome(
      387  							theUpdateParameters.getRequest(),
      388  							theUpdateParameters.getEntity(),
      389  							theUpdateParameters.getResource(),
      390  							theUpdateParameters.getMatchUrl(),
      391  							theUpdateParameters.getOperationType())
      392  					.setCreated(wasDeleted);
      393  			outcome.setPreviousResource(oldResource);
      394  			if (!outcome.isNop()) {
      395  				// Technically this may not end up being right since we might not increment if the
      396  				// contents turn out to be the same
      397  				outcome.setId(outcome.getId()
      398  						.withVersion(Long.toString(outcome.getId().getVersionIdPartAsLong() + 1)));
      399  			}
      400  			return outcome;
      401  		}
      402  
      403  		/*
      404  		 * Otherwise, we're not in a transaction
      405  		 */
      406  		return updateInternal(
      407  				theUpdateParameters.getRequest(),
      408  				theUpdateParameters.getResource(),
      409  				theUpdateParameters.getMatchUrl(),
      410  				theUpdateParameters.shouldPerformIndexing(),
      411  				theUpdateParameters.shouldForceUpdateVersion(),
      412  				theUpdateParameters.getEntity(),
      413  				theUpdateParameters.getResourceIdToUpdate(),
      414  				oldResource,
      415  				theUpdateParameters.getOperationType(),
      416  				theUpdateParameters.getTransactionDetails());
      417  	}

      428  		if (!getStorageSettings().canDeleteExpunge()) {
      429  			throw new MethodNotAllowedException(Msg.code(963) + "_expunge is not enabled on this server: "
      430  					+ getStorageSettings().cannotDeleteExpungeReason());
      431  		}

      433  		RestfulServerUtils.DeleteCascadeDetails cascadeDelete =
      434  				RestfulServerUtils.extractDeleteCascadeParameter(theRequest);
      435  		boolean cascade = false;
      436  		Integer cascadeMaxRounds = null;
      437  		if (cascadeDelete.getMode() == DeleteCascadeModeEnum.DELETE) {
      438  			cascade = true;
      439  			cascadeMaxRounds = cascadeDelete.getMaxRounds();
      440  			if (cascadeMaxRounds == null) {
      441  				cascadeMaxRounds = myStorageSettings.getMaximumDeleteConflictQueryCount();
      442  			} else if (myStorageSettings.getMaximumDeleteConflictQueryCount() != null
      443  					&& myStorageSettings.getMaximumDeleteConflictQueryCount() < cascadeMaxRounds) {
      444  				cascadeMaxRounds = myStorageSettings.getMaximumDeleteConflictQueryCount();

      446  		}

      449  		try {

      452  							getStorageSettings().getExpungeBatchSize(),
      453  							urlsToDeleteExpunge,
