      109    private static IntBuffer decodeIntBuffer(
      110        int[] values, StreamMetadata streamMetadata, boolean isSigned) {
      111      /*
      112       * Currently the encoder uses only fixed combinations of encodings.
      113       * For performance reasons we also use the fixed combinations of the encodings and not a generic solution.
      114       * The following encodings and combinations are used:
      115       *   - Morton Delta -> always sorted so not ZigZag encoding needed
      116       *   - Delta -> currently always in combination with ZigZag encoding
      117       *   - Rle -> in combination with ZigZag encoding if data type is signed
      118       *   - Delta Rle
      119       *   - Componentwise Delta -> always ZigZag encoding is used
      120       * */
      121      switch (streamMetadata.logicalLevelTechnique1()) {
      122        case DELTA:
      123          if (streamMetadata.logicalLevelTechnique2().equals(LogicalLevelTechnique.RLE)) {
      124            var rleMetadata = (RleEncodedStreamMetadata) streamMetadata;
      125            values =
      126                VectorizedDecodingUtils.decodeUnsignedRLE(
      127                        values, rleMetadata.runs(), rleMetadata.numRleValues())
      128                    .array();
      129            /* Currently delta values are always ZigZag encoded */
      130            VectorizedDecodingUtils.decodeZigZagDelta(values);
      131            return IntBuffer.wrap(values);
      132          }
      133  
      134          // TODO: check if zigzag encoding is needed -> if values are sorted in ascending order no
      135          // need for zigzag
      136          // for only delta decoding without zigzag use Delta.fastinverseDelta(values) from Lemire
      137          VectorizedDecodingUtils.decodeZigZagDelta(values);
      138          return IntBuffer.wrap(values);
      139        case RLE:
      140          /* Currently no second logical level technique is used in combination with Rle */
      141          return VectorizedDecodingUtils.decodeRle(values, streamMetadata, isSigned);
      142        case MORTON:
      143          /*
      144           * Currently always used in combination with delta encoding and without ZigZag encoding
      145           * since the values are sorted in ascending order. The data are stored internally in
      146           * compressed form since they can be in parallel decompressed on the GPU.
      147           */
      148          Delta.fastinverseDelta(values);
      149          return IntBuffer.wrap(values);
      150        case COMPONENTWISE_DELTA:
      151          /* Currently only Vec2 is supported */
      152          VectorizedDecodingUtils.decodeComponentwiseDeltaVec2(values);
      153          return IntBuffer.wrap(values);
      154        case NONE:
      155          // TODO: merge with varint decoding
      156          if (isSigned) {
      157            DecodingUtils.decodeZigZag(values);
      158          }
      159          return IntBuffer.wrap(values);
      160      }
      161  
      162      throw new IllegalArgumentException(
      163          "The specified Logical level technique is not supported: "
      164              + streamMetadata.logicalLevelTechnique1());
      165    }

      211    private static IntBuffer decodeLengthToOffsetBuffer(int[] values, StreamMetadata streamMetadata) {
      212      if (streamMetadata.logicalLevelTechnique1().equals(LogicalLevelTechnique.DELTA)
      213          && streamMetadata.logicalLevelTechnique2().equals(LogicalLevelTechnique.NONE)) {
      214        var decodedValues = VectorizedDecodingUtils.zigZagDeltaOfDeltaDecoding(values);
      215        return IntBuffer.wrap(decodedValues);
      216      }
      217  
      218      if (streamMetadata.logicalLevelTechnique1().equals(LogicalLevelTechnique.RLE)
      219          && streamMetadata.logicalLevelTechnique2().equals(LogicalLevelTechnique.NONE)) {
      220        var rleMetadata = (RleEncodedStreamMetadata) streamMetadata;
      221        var decodedValues =
      222            VectorizedDecodingUtils.rleDeltaDecoding(
      223                values, rleMetadata.runs(), rleMetadata.numRleValues());
      224        return decodedValues;
      225      }
      226  
      227      if (streamMetadata.logicalLevelTechnique1().equals(LogicalLevelTechnique.NONE)
      228          && streamMetadata.logicalLevelTechnique2().equals(LogicalLevelTechnique.NONE)) {
      229        // TODO: optimize performance
      230        Delta.fastinverseDelta(values);
      231        var offsets = new int[streamMetadata.numValues() + 1];
      232        offsets[0] = 0;
      233        System.arraycopy(values, 0, offsets, 1, streamMetadata.numValues());
      234        return IntBuffer.wrap(offsets);
      235      }
      236  
      237      if (streamMetadata.logicalLevelTechnique1().equals(LogicalLevelTechnique.DELTA)
      238          && streamMetadata.logicalLevelTechnique2().equals(LogicalLevelTechnique.RLE)) {
      239        var rleMetadata = (RleEncodedStreamMetadata) streamMetadata;
      240        var decodedValues =
      241            VectorizedDecodingUtils.zigZagRleDeltaDecoding(
      242                values, rleMetadata.runs(), rleMetadata.numRleValues());
      243        Delta.fastinverseDelta(decodedValues.array());
      244        return decodedValues;
      245      }
      246  
      247      throw new IllegalArgumentException(
      248          "Only delta encoding is supported for transforming length to offset streams yet.");
      249    }
