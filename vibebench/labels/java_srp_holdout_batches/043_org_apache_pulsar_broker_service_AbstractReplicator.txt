       86      public enum State {
       87          /**
       88           * This enum has two mean meanings：
       89           *   Init: replicator is just created, has not been started now.
       90           *   Disconnected: the producer was closed after {@link PersistentTopic#checkGC} called {@link #disconnect}.
       91           */
       92          // The internal producer is disconnected.
       93          Disconnected,
       94          // Trying to create a new internal producer.

      103          // The replicator is in terminating.

      167      public void startProducer() {
      168          // Guarantee only one task call "producerBuilder.createAsync()".
      169          Pair<Boolean, State> setStartingRes = compareSetAndGetState(State.Disconnected, State.Starting);
      170          if (!setStartingRes.getLeft()) {
      171              if (setStartingRes.getRight() == State.Starting) {
      172                  log.info("[{}] Skip the producer creation since other thread is doing starting, state : {}",
      173                          replicatorId, state);
      174              } else if (setStartingRes.getRight() == State.Started) {
      175                  // Since the method "startProducer" will be called even if it is started, only print debug-level log.
      176                  if (log.isDebugEnabled()) {
      177                      log.debug("[{}] Replicator was already running. state: {}", replicatorId, state);
      178                  }
      179              } else if (setStartingRes.getRight() == State.Disconnecting) {
      180                  if (log.isDebugEnabled()) {
      181                      log.debug("[{}] Rep.producer is closing, delay to retry(wait the producer close success)."
      182                              + " state: {}", replicatorId, state);
      183                  }
      184                  delayStartProducerAfterDisconnected();
      185              } else {
      186                  /** {@link State.Terminating}, {@link State.Terminated}. **/
      187                  log.info("[{}] Skip the producer creation since the replicator state is : {}", replicatorId, state);
      188              }
      189              return;
      190          }
      191  
      192          log.info("[{}] Starting replicator", replicatorId);
      193  
      194          // Force only replicate messages to a non-partitioned topic, to avoid auto-create a partitioned topic on
      195          // the remote cluster.
      196          prepareCreateProducer().thenCompose(ignore -> {
      197              ProducerBuilderImpl builderImpl = (ProducerBuilderImpl) producerBuilder;
      198              builderImpl.getConf().setNonPartitionedTopicExpected(true);
      199              builderImpl.getConf().setReplProducer(true);
      200              return producerBuilder.createAsync().thenAccept(producer -> {
      201                  setProducerAndTriggerReadEntries(producer);
      202              });
      203          }).exceptionally(ex -> {
      204              Pair<Boolean, State> setDisconnectedRes = compareSetAndGetState(State.Starting, State.Disconnected);
      205              if (setDisconnectedRes.getLeft()) {
      206                  long waitTimeMs = backOff.next();
      207                  log.warn("[{}] Failed to create remote producer ({}), retrying in {} s",
      208                          replicatorId, ex.getMessage(), waitTimeMs / 1000.0);
      209                  // BackOff before retrying
      210                  scheduleCheckTopicActiveAndStartProducer(waitTimeMs);
      211              } else {
      212                  if (setDisconnectedRes.getRight() == State.Terminating
      213                          || setDisconnectedRes.getRight() == State.Terminated) {
      214                      log.info("[{}] Skip to create producer, because it has been terminated, state is : {}",
      215                              replicatorId, state);
      216                  } else {
      217                      /** {@link  State.Disconnected}, {@link  State.Starting}, {@link  State.Started} **/
      218                      // Since only one task can call "producerBuilder.createAsync()", this scenario is not expected.
      219                      // So print a warn log.
      220                      log.warn("[{}] Other thread will try to create the producer again. so skipped current one task."
      221                                      + " State is : {}",
      222                              replicatorId, state);
      223                  }
      224              }
      225              return null;
      226          });
      227  
      228      }

      244      protected void scheduleCheckTopicActiveAndStartProducer(final long waitTimeMs) {
      245          brokerService.executor().schedule(() -> {
      246              if (state == State.Terminating || state == State.Terminated) {
      247                  log.info("[{}] Skip scheduled to start the producer since the replicator state is : {}",
      248                          replicatorId, state);
      249                  return;
      250              }
      251              CompletableFuture<Optional<Topic>> topicFuture = brokerService.getTopics().get(localTopicName);
      252              if (topicFuture == null) {
      253                  // Topic closed.
      254                  log.info("[{}] Skip scheduled to start the producer since the topic was closed successfully."
      255                          + " And trigger a terminate.", replicatorId);
      256                  terminate();
      257                  return;
      258              }
      259              topicFuture.thenAccept(optional -> {
      260                  if (optional.isEmpty()) {
      261                      // Topic closed.
      262                      log.info("[{}] Skip scheduled to start the producer since the topic was closed. And trigger a"
      263                              + " terminate.", replicatorId);
      264                      terminate();
      265                      return;
      266                  }
      267                  if (optional.get() != localTopic) {
      268                      // Topic closed and created a new one, current replicator is outdated.
      269                      log.info("[{}] Skip scheduled to start the producer since the topic was closed. And trigger a"
      270                              + " terminate.", replicatorId);
      271                      terminate();
      272                      return;
      273                  }
      274                  Replicator replicator = localTopic.getReplicators().get(remoteCluster);
      275                  if (replicator != AbstractReplicator.this) {
      276                      // Current replicator has been closed, and created a new one.
      277                      log.info("[{}] Skip scheduled to start the producer since a new replicator has instead current"
      278                              + " one. And trigger a terminate.", replicatorId);
      279                      terminate();
      280                      return;
      281                  }
      282                  startProducer();
      283              }).exceptionally(ex -> {
      284                  log.error("[{}] [{}] Stop retry to create producer due to unknown error(topic create failed), and"
      285                                  + " trigger a terminate. Replicator state: {}",
      286                          localTopicName, replicatorId, STATE_UPDATER.get(this), ex);
      287                  terminate();
      288                  return null;
      289              });
      290          }, waitTimeMs, TimeUnit.MILLISECONDS);
      291      }

      310      public CompletableFuture<Void> disconnect() {
      311          long backlog = getNumberOfEntriesInBacklog();
      312          if (backlog > 0) {
      313              CompletableFuture<Void> disconnectFuture = new CompletableFuture<>();
      314              disconnectFuture.completeExceptionally(new TopicBusyException("Cannot close a replicator with backlog"));
      315              if (log.isDebugEnabled()) {
      316                  log.debug("[{}] Replicator disconnect failed since topic has backlog", replicatorId);
      317              }
      318              return disconnectFuture;
      319          }
      320          log.info("[{}] Disconnect replicator at position {} with backlog {}", replicatorId,
      321                  getReplicatorReadPosition(), backlog);
      322          return beforeDisconnect()
      323              .thenCompose(__ -> closeProducerAsync(true))
      324              .thenApply(__ -> {
      325                  afterDisconnected();
      326                  return null;
      327              });
      328      }

      349      protected CompletableFuture<Void> closeProducerAsync(boolean closeTheStartingProducer) {
      350          Pair<Boolean, State> setDisconnectingRes = compareSetAndGetState(State.Started, State.Disconnecting);
      351          if (!setDisconnectingRes.getLeft()) {
      352              if (setDisconnectingRes.getRight() == State.Starting) {
      353                  if (closeTheStartingProducer) {
      354                      /**
      355                       * Delay retry(wait for the start producer task is finish).
      356                       * Note: If the producer always start fail, the start producer task will always retry until the
      357                       *   state changed to {@link State.Terminated}.
      358                       *   Nit: The better solution is creating a {@link CompletableFuture} to trace the in-progress
      359                       *     creation and call "inProgressCreationFuture.thenApply(closeProducer())".
      360                       */
      361                      long waitTimeMs = backOff.next();
      362                      brokerService.executor().schedule(() -> closeProducerAsync(true),
      363                              waitTimeMs, TimeUnit.MILLISECONDS);
      364                  } else {
      365                      log.info("[{}] Skip current producer closing since the previous producer has been closed,"
      366                                      + " and trying start a new one, state : {}",
      367                              replicatorId, setDisconnectingRes.getRight());
      368                  }
      369              } else if (setDisconnectingRes.getRight() == State.Disconnected
      370                      || setDisconnectingRes.getRight() == State.Disconnecting) {
      371                  log.info("[{}] Skip current producer closing since other thread did closing, state : {}",
      372                          replicatorId, setDisconnectingRes.getRight());
      373              } else if (setDisconnectingRes.getRight() == State.Terminating
      374                      || setDisconnectingRes.getRight() == State.Terminated) {
      375                  log.info("[{}] Skip current producer closing since other thread is doing termination, state : {}",
      376                          replicatorId, state);
      377              }
      378              log.info("[{}] Skip current termination since other thread is doing close producer or termination,"
      379                              + " state : {}", replicatorId, state);
      380              return CompletableFuture.completedFuture(null);
      381          }
      382  
      383          // Close producer and update state.
      384          return doCloseProducerAsync(producer, () -> {
      385              Pair<Boolean, State> setDisconnectedRes = compareSetAndGetState(State.Disconnecting, State.Disconnected);
      386              if (setDisconnectedRes.getLeft()) {
      387                  this.producer = null;
      388                  // deactivate further read
      389                  disableReplicatorRead();
      390                  return;
      391              }
      392              if (setDisconnectedRes.getRight() == State.Terminating
      393                      || setDisconnectingRes.getRight() == State.Terminated) {
      394                  log.info("[{}] Skip setting state to terminated because it was terminated, state : {}",
      395                          replicatorId, state);
      396              } else {
      397                  // Since only one task can call "doCloseProducerAsync(producer, action)", this scenario is not expected.
      398                  // So print a warn log.
      399                  log.warn("[{}] Other task has change the state to terminated. so skipped current one task."
      400                                  + " State is : {}",
      401                          replicatorId, state);
      402              }
      403          });
      404      }

      406      protected CompletableFuture<Void> doCloseProducerAsync(Producer<byte[]> producer, Runnable actionAfterClosed) {
      407          CompletableFuture<Void> future =
      408                  producer == null ? CompletableFuture.completedFuture(null) : producer.closeAsync();
      409          return future.thenRun(() -> {
      410              actionAfterClosed.run();
      411          }).exceptionally(ex -> {
      412              long waitTimeMs = backOff.next();
      413              log.warn(
      414                      "[{}] Exception: '{}' occurred while trying to close the producer. Replicator state: {}."
      415                              + " Retrying again in {} s.",
      416                      replicatorId, ex.getMessage(), state, waitTimeMs / 1000.0);
      417              // BackOff before retrying
      418              brokerService.executor().schedule(() -> doCloseProducerAsync(producer, actionAfterClosed),
      419                      waitTimeMs, TimeUnit.MILLISECONDS);
      420              return null;
      421          });
