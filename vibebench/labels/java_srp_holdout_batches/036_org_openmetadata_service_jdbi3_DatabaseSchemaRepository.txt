      216    private Map<UUID, EntityReference> batchFetchDatabases(List<DatabaseSchema> schemas) {
      217      var databaseMap = new HashMap<UUID, EntityReference>();
      218      if (schemas == null || schemas.isEmpty()) {
      219        return databaseMap;
      220      }
      221      var relations =
      222          daoCollection
      223              .relationshipDAO()
      224              .findFromBatch(
      225                  entityListToStrings(schemas), Relationship.CONTAINS.ordinal(), Include.ALL);
      226  
      227      var databaseIds =
      228          relations.stream()
      229              .filter(relation -> Entity.DATABASE.equals(relation.getFromEntity()))
      230              .map(relation -> UUID.fromString(relation.getFromId()))
      231              .distinct()
      232              .toList();
      233  
      234      // Batch fetch all database entity references
      235      var databaseRefs = Entity.getEntityReferencesByIds(Entity.DATABASE, databaseIds, Include.ALL);
      236      var databaseRefMap =
      237          databaseRefs.stream().collect(Collectors.toMap(EntityReference::getId, ref -> ref));
      238  
      239      // Map schemas to their databases
      240      relations.forEach(
      241          relation -> {
      242            // Only process records where the from entity is DATABASE
      243            if (Entity.DATABASE.equals(relation.getFromEntity())) {
      244              var schemaId = UUID.fromString(relation.getToId());
      245              var databaseId = UUID.fromString(relation.getFromId());
      246              var databaseRef = databaseRefMap.get(databaseId);
      247              if (databaseRef != null) {
      248                databaseMap.put(schemaId, databaseRef);
      249              }
      250            }
      251          });
      252  
      253      return databaseMap;
      254    }

      256    private void fetchAndSetTables(List<DatabaseSchema> schemas) {
      257      if (schemas == null || schemas.isEmpty()) {
      258        return;
      259      }
      260  
      261      // Get all schema IDs
      262      List<String> schemaIds =
      263          schemas.stream().map(schema -> schema.getId().toString()).distinct().toList();
      264  
      265      // Bulk fetch all table relationships
      266      List<CollectionDAO.EntityRelationshipObject> tableRelations =
      267          daoCollection
      268              .relationshipDAO()
      269              .findToBatch(schemaIds, Relationship.CONTAINS.ordinal(), Entity.DATABASE_SCHEMA, TABLE);
      270  
      271      // Group table IDs by schema
      272      Map<UUID, List<UUID>> schemaToTableIds = new HashMap<>();
      273      for (CollectionDAO.EntityRelationshipObject relation : tableRelations) {
      274        UUID schemaId = UUID.fromString(relation.getFromId());
      275        UUID tableId = UUID.fromString(relation.getToId());
      276        schemaToTableIds.computeIfAbsent(schemaId, k -> new ArrayList<>()).add(tableId);
      277      }
      278  
      279      // Get all unique table IDs
      280      List<UUID> allTableIds =
      281          schemaToTableIds.values().stream().flatMap(List::stream).distinct().toList();
      282  
      283      // Bulk fetch all table references
      284      Map<UUID, EntityReference> tableReferences = new HashMap<>();
      285      if (!allTableIds.isEmpty()) {
      286        List<EntityReference> tableRefs =
      287            Entity.getEntityReferencesByIds(TABLE, allTableIds, Include.ALL);
      288        for (EntityReference ref : tableRefs) {
      289          tableReferences.put(ref.getId(), ref);
      290        }
      291      }
      292  
      293      // Set tables on each schema
      294      for (DatabaseSchema schema : schemas) {
      295        List<UUID> tableIds = schemaToTableIds.get(schema.getId());
      296        if (tableIds != null) {
      297          List<EntityReference> tables =
      298              tableIds.stream()
      299                  .map(tableReferences::get)
      300                  .filter(Objects::nonNull)
      301                  .collect(Collectors.toList());
      302          schema.setTables(tables);
      303        } else {
      304          schema.setTables(Collections.emptyList());
      305        }
      306      }
      307    }

      323    private DatabaseSchemaProfilerConfig getProfilerConfigForSchema(
      324        DatabaseSchema schema, Map<UUID, DatabaseSchemaProfilerConfig> configCache) {
      325      if (schema.getId() == null) {
      326        return null;
      327      }
      328  
      329      // Check if already in cache
      330      if (configCache.containsKey(schema.getId())) {
      331        return configCache.get(schema.getId());
      332      }
      333  
      334      // Fetch and cache the config
      335      DatabaseSchemaProfilerConfig config = fetchProfilerConfig(schema.getId());
      336      configCache.put(schema.getId(), config);
      337      return config;
      338    }
