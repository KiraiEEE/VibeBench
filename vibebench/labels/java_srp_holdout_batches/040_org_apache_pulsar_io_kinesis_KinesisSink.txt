      113      private void sendUserRecord(ProducerSendCallback producerSendCallback) {
      114          ListenableFuture<UserRecordResult> addRecordResult = kinesisProducer.addUserRecord(this.streamName,
      115                  producerSendCallback.partitionedKey, producerSendCallback.data);
      116          addCallback(addRecordResult, producerSendCallback, directExecutor());

      119      @Override
      120      public void write(Record<GenericObject> record) throws Exception {
      121          // kpl-thread captures publish-failure. fail the publish on main pulsar-io-thread to maintain the ordering
      122          if (kinesisSinkConfig.isRetainOrdering() && previousPublishFailed == TRUE) {
      123              LOG.warn("Skip acking message to retain ordering with previous failed message {}-{}", this.streamName,
      124                      record.getRecordSequence());
      125              throw new IllegalStateException("kinesis queue has publish failure");
      126          }
      127          String partitionedKey = record.getKey().orElse(record.getTopicName().orElse(defaultPartitionedKey));
      128          partitionedKey = partitionedKey.length() > maxPartitionedKeyLength
      129                  ? partitionedKey.substring(0, maxPartitionedKeyLength - 1)
      130                  : partitionedKey; // partitionedKey Length must be at least one, and at most 256
      131          ByteBuffer data = createKinesisMessage(kinesisSinkConfig.getMessageFormat(), record);
      132          int size = data.remaining();
      133          sendUserRecord(ProducerSendCallback.create(this, record, System.nanoTime(), partitionedKey, data));
      134          if (sinkContext != null) {
      135              sinkContext.recordMetric(METRICS_TOTAL_INCOMING, 1);
      136              sinkContext.recordMetric(METRICS_TOTAL_INCOMING_BYTES, data.array().length);
      137          }
      138          if (LOG.isDebugEnabled()) {
      139              LOG.debug("Published message to kinesis stream {} with size {}", streamName, size);
      140          }
      141      }

      204          private Record<GenericObject> resultContext;
      205          private long startTime = 0;
      206          private final Handle<ProducerSendCallback> recyclerHandle;
      207          private KinesisSink kinesisSink;
      208          private Backoff backoff;
      209          private String partitionedKey;
      210          private ByteBuffer data;
      211  
      212          private ProducerSendCallback(Handle<ProducerSendCallback> recyclerHandle) {
      213              this.recyclerHandle = recyclerHandle;

      216          static ProducerSendCallback create(KinesisSink kinesisSink, Record<GenericObject> resultContext, long startTime,
      217                                             String partitionedKey, ByteBuffer data) {
      218              ProducerSendCallback sendCallback = RECYCLER.get();
      219              sendCallback.resultContext = resultContext;
      220              sendCallback.kinesisSink = kinesisSink;
      221              sendCallback.startTime = startTime;
      222              sendCallback.partitionedKey = partitionedKey;
      223              sendCallback.data = data;
      224              if (kinesisSink.kinesisSinkConfig.isRetainOrdering() && sendCallback.backoff == null) {
      225                  sendCallback.backoff = new Backoff(kinesisSink.kinesisSinkConfig.getRetryInitialDelayInMillis(),
      226                          TimeUnit.MILLISECONDS, kinesisSink.kinesisSinkConfig.getRetryMaxDelayInMillis(),
      227                          TimeUnit.MILLISECONDS, 0, TimeUnit.SECONDS);
      228              }
      229              return sendCallback;
      230          }

      232          private void recycle() {
      233              resultContext = null;
      234              kinesisSink = null;
      235              startTime = 0;
      236              if (backoff != null) {
      237                  backoff.reset();
      238              }
      239              partitionedKey = null;
      240              data = null;
      241              recyclerHandle.recycle(this);
      242          }

      244          private static final Recycler<ProducerSendCallback> RECYCLER = new Recycler<ProducerSendCallback>() {
      245              @Override
      246              protected ProducerSendCallback newObject(Handle<ProducerSendCallback> handle) {
      247                  return new ProducerSendCallback(handle);
      248              }
      249          };

      251          @Override
      252          public void onSuccess(UserRecordResult result) {
      253              if (LOG.isDebugEnabled()) {
      254                  LOG.debug("Successfully published message for {}-{} with latency {}",
      255                          kinesisSink.streamName, result.getShardId(),
      256                          TimeUnit.NANOSECONDS.toMillis((System.nanoTime() - startTime)));
      257              }
      258              if (kinesisSink.sinkContext != null) {
      259                  kinesisSink.sinkContext.recordMetric(METRICS_TOTAL_SUCCESS, 1);
      260              }
      261              kinesisSink.previousPublishFailed = FALSE;
      262              this.resultContext.ack();
      263              recycle();
      264          }

      266          @Override
      267          public void onFailure(Throwable exception) {
      268              if (exception instanceof UserRecordFailedException) {
      269                  // If the exception is UserRecordFailedException, we need to extract it to see real error messages.
      270                  UserRecordFailedException failedException = (UserRecordFailedException) exception;
      271                  StringBuffer stringBuffer = new StringBuffer();
      272                  failedException.getResult().getAttempts().forEach(attempt ->
      273                          stringBuffer.append(String.format("errorMessage:%s, errorCode:%s, delay:%d, duration:%d;",
      274                                  attempt.getErrorMessage(), attempt.getErrorCode(), attempt.getDelay(), attempt.getDuration())));
      275                  LOG.error("[{}] Failed to published message for replicator of {}-{}: Attempts:{}",
      276                          kinesisSink.streamName, resultContext.getPartitionId(),
      277                          resultContext.getRecordSequence(), stringBuffer);
      278              } else {
      279                  if (StringUtils.isEmpty(exception.getMessage())) {
      280                      LOG.error("[{}] Failed to published message for replicator of {}-{}", kinesisSink.streamName,
      281                          resultContext.getPartitionId(), resultContext.getRecordSequence(), exception);
      282                  } else {
      283                      LOG.error("[{}] Failed to published message for replicator of {}-{}, {} ", kinesisSink.streamName,
      284                          resultContext.getPartitionId(), resultContext.getRecordSequence(), exception.getMessage());
      285                  }
      286              }
      287              kinesisSink.previousPublishFailed = TRUE;
      288              if (kinesisSink.sinkContext != null) {
      289                  kinesisSink.sinkContext.recordMetric(METRICS_TOTAL_FAILURE, 1);
      290              }
      291              if (backoff != null) {
      292                  long nextDelay = backoff.next();
      293                  LOG.info("[{}] Retry to publish message for replicator of {}-{} after {} ms.", kinesisSink.streamName,
      294                          resultContext.getPartitionId(), resultContext.getRecordSequence(), nextDelay);
      295                  kinesisSink.scheduledExecutor.schedule(() -> kinesisSink.sendUserRecord(this),
      296                          nextDelay, TimeUnit.MICROSECONDS);
      297              } else {
      298                  recycle();
      299              }
      300          }
      301      }

      303      public ByteBuffer createKinesisMessage(MessageFormat msgFormat, Record<GenericObject> record)
      304              throws JsonProcessingException {
      305          switch (msgFormat) {
      306              case FULL_MESSAGE_IN_JSON:
      307                  return ByteBuffer.wrap(Utils.serializeRecordToJson(record).getBytes(StandardCharsets.UTF_8));
      308              case FULL_MESSAGE_IN_FB:
      309                  return Utils.serializeRecordToFlatBuffer(record);
      310              case FULL_MESSAGE_IN_JSON_EXPAND_VALUE:
      311                  return ByteBuffer.wrap(
      312                          Utils.serializeRecordToJsonExpandingValue(objectMapper, record,
      313                                  kinesisSinkConfig.isJsonFlatten())
      314                                  .getBytes(StandardCharsets.UTF_8));
      315              default:
      316                  // send raw-message
      317                  return ByteBuffer.wrap(Utils.getMessage(record).getData());
      318          }
      319      }
