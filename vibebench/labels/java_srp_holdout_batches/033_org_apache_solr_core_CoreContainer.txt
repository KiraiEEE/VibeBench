      458    private synchronized void initializeAuthorizationPlugin(Map<String, Object> authorizationConf) {
      459      authorizationConf = Utils.getDeepCopy(authorizationConf, 4);
      460      int newVersion = readVersion(authorizationConf);
      461      // Initialize the Authorization module
      462      SecurityPluginHolder<AuthorizationPlugin> old = authorizationPlugin;
      463      SecurityPluginHolder<AuthorizationPlugin> authorizationPlugin = null;
      464      if (authorizationConf != null) {
      465        String klas = (String) authorizationConf.get("class");
      466        if (klas == null) {
      467          throw new SolrException(
      468              ErrorCode.SERVER_ERROR, "class is required for authorization plugin");
      469        }
      470        if (old != null && old.getZnodeVersion() == newVersion && newVersion > 0) {
      471          log.debug("Authorization config not modified");
      472          return;
      473        }
      474        log.info("Initializing authorization plugin: {}", klas);
      475        authorizationPlugin =
      476            new SecurityPluginHolder<>(
      477                newVersion,
      478                getResourceLoader()
      479                    .newInstance(
      480                        klas,
      481                        AuthorizationPlugin.class,
      482                        null,
      483                        new Class<?>[] {CoreContainer.class},
      484                        new Object[] {this}));
      485  
      486        // Read and pass the authorization context to the plugin
      487        authorizationPlugin.plugin.init(authorizationConf);
      488      } else {
      489        log.debug("Security conf doesn't exist. Skipping setup for authorization module.");
      490      }
      491      this.authorizationPlugin = authorizationPlugin;
      492      if (old != null) {
      493        try {
      494          old.plugin.close();
      495        } catch (Exception e) {
      496          log.error("Exception while attempting to close old authorization plugin", e);
      497        }
      498      }
