      223      private void resumeSwitchBrowser() {
      224          final String methodTag = TAG + ":resumeSwitchBrowser";
      225          try {
      226              if (switchBrowserBundle == null) {
      227                  throw new ClientException(
      228                          ClientException.NULL_OBJECT,
      229                          "No switch browser bundle found to resume the flow."
      230                  );
      231              }
      232              Logger.info(methodTag, "Resuming switch browser flow");
      233              getSwitchBrowserCoordinator().processSwitchBrowserResume(
      234                      mAuthorizationRequestUrl,
      235                      switchBrowserBundle,
      236                      (switchBrowserResumeUri, switchBrowserResumeHeaders) -> {
      237                          launchWebView(switchBrowserResumeUri.toString(), switchBrowserResumeHeaders);
      238                          return null;
      239                      }
      240              );
      241              setSwitchBrowserBundle(null);
      242          } catch (final ClientException e) {
      243              Logger.error(methodTag, "Error processing switch browser resume", e);
      244              sendResult(RawAuthorizationResult.fromException(e));
      245              finish();
      246          }

      428              @Override
      429              public boolean onShowFileChooser(final WebView webView,
      430                                               final ValueCallback<Uri[]> filePathCallback,
      431                                               final FileChooserParams fileChooserParams) {
      432                  if (!CommonFlightsManager.INSTANCE.getFlightsProvider()
      433                          .isFlightEnabled(CommonFlight.ENABLE_WEBVIEW_FILE_CHOOSER)) {
      434                      return false;
      435                  }
      436                  final String fileChooserTag = methodTag + ":onShowFileChooser";
      437                  Logger.info(fileChooserTag, "WebView requested file chooser.");
      438                  // Cancel any previous pending callback to avoid leaking it.
      439                  if (mFilePathCallback != null) {
      440                      mFilePathCallback.onReceiveValue(null);
      441                      mFilePathCallback = null;
      442                  }
      443                  mFilePathCallback = filePathCallback;
      444                  // Build the file chooser intent from the WebView's params (GET_CONTENT).
      445                  final Intent filePickerIntent = fileChooserParams.createIntent();
      446                  // Attempt to add camera capture as an additional option by creating a
      447                  // temporary file in the app's cache directory. Using getCacheDir() ensures
      448                  // the file is stored in an app-private location and the path is valid on
      449                  // all Android versions. The content URI produced by MsalFileProvider is
      450                  // required for Android 7+ to avoid FileUriExposedException.
      451                  Intent cameraIntent = null;
      452                  try {
      453                      final File tempFile = File.createTempFile(
      454                              "msal_image_capture_", ".jpg", requireContext().getCacheDir());
      455                      mCameraImageFile = tempFile;
      456                      mCameraImageUri = MsalFileProvider.getUriForFile(
      457                              requireContext(),
      458                              requireContext().getPackageName() + FILE_PROVIDER_AUTHORITY_SUFFIX,
      459                              tempFile);
      460                      cameraIntent = new Intent(MediaStore.ACTION_IMAGE_CAPTURE);
      461                      cameraIntent.putExtra(MediaStore.EXTRA_OUTPUT, mCameraImageUri);
      462                  } catch (final IOException e) {
      463                      Logger.warn(fileChooserTag,
      464                              "Failed to create temporary file in cache directory for camera capture: "
      465                                      + e.getMessage());
      466                      mCameraImageFile = null;
      467                      mCameraImageUri = null;
      468                  } catch (final IllegalArgumentException e) {
      469                      // FileProvider not configured or authority mismatch; proceed without camera.
      470                      Logger.warn(fileChooserTag,
      471                              "MsalFileProvider configuration issue — verify that the provider "
      472                                      + "authority matches '" + requireContext().getPackageName()
      473                                      + FILE_PROVIDER_AUTHORITY_SUFFIX
      474                                      + "' and that msal_image_capture_paths.xml is referenced "
      475                                      + "correctly in AndroidManifest.xml. Camera capture unavailable: "
      476                                      + e.getMessage());
      477                      mCameraImageFile = null;
      478                      mCameraImageUri = null;
      479                  }
      480                  // Combine file picker and (if available) camera capture into one chooser.
      481                  final Intent chooserIntent;
      482                  if (cameraIntent != null) {
      483                      chooserIntent = Intent.createChooser(filePickerIntent, null);
      484                      chooserIntent.putExtra(Intent.EXTRA_INITIAL_INTENTS,
      485                              new Intent[]{cameraIntent});
      486                  } else {
      487                      chooserIntent = filePickerIntent;
      488                  }
      489                  try {
      490                      mFileChooserLauncher.launch(chooserIntent);
      491                  } catch (final ActivityNotFoundException e) {
      492                      Logger.error(fileChooserTag, "No activity available to handle file chooser.", e);
      493                      mFilePathCallback.onReceiveValue(null);
      494                      mFilePathCallback = null;
      495                      deleteCameraTempFile();
      496                      return false;
      497                  }
      498                  return true;
      499              }
