      212  	private String scanForMatch(int sqlType, int length, int precision, int scale, boolean nullable, List<SQLTypeMapping> l) {
      213  		if(l!=null) {
      214  			for ( SQLTypeMapping element : l ) {
      215  				if ( element.getJDBCType() != sqlType ) return null;
      216  				if ( element.match( sqlType, length, precision, scale, nullable ) ) {
      217  					return element.getHibernateType();
      218  				}
      219  			}
      220  		}
      221  		return null;
      222  	}
