       53  def _split_bucket_id_and_prefix(path: str) -> tuple[str, str]:
       54      """Split 'namespace/name(/optional/prefix)' into ('namespace/name', 'prefix').
       55  
       56      Returns (bucket_id, prefix) where prefix may be empty string.
       57      Raises ValueError if path doesn't contain at least namespace/name.
       58      """
       59      parts = path.split("/", 2)
       60      if len(parts) < 2 or not parts[0] or not parts[1]:
       61          raise ValueError(f"Invalid bucket path: '{path}'. Expected format: namespace/bucket_name")
       62      bucket_id = f"{parts[0]}/{parts[1]}"
       63      prefix = parts[2] if len(parts) > 2 else ""
       64      return bucket_id, prefix
