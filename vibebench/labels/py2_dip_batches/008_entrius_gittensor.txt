       33  async def forward(self: 'BaseValidatorNeuron') -> None:
       34      """Execute the validator's forward pass.
       35  
       36      Performs the core validation cycle every VALIDATOR_STEPS_INTERVAL steps:
       37      1. Get all available miner UIDs
       38      2. Score OSS contributions and get miner evaluations
       39      3. Update scores using exponential moving average
       40      4. Run issue bounties verification (needs tier data from scoring)
       41  
       42      Args:
       43          self: The validator instance containing all necessary state
       44      """
       45  
       46      if self.step % VALIDATOR_STEPS_INTERVAL == 0:
       47          miner_uids = get_all_uids(self)
       48  
       49          # Score OSS contributions - returns evaluations for issue verification
       50          miner_evaluations = await oss_contributions(self, miner_uids)
       51  
       52          # Issue bounties verification
       53          await issues_competition(self, miner_evaluations)
       54  
       55      await asyncio.sleep(VALIDATOR_WAIT)

       58  async def oss_contributions(self: 'BaseValidatorNeuron', miner_uids: set[int]) -> Dict[int, MinerEvaluation]:
       59      """Score OSS contributions and return miner evaluations for downstream use."""
       60      master_repositories = load_master_repo_weights()
       61      programming_languages = load_programming_language_weights()
       62      token_config = load_token_config()
       63  
       64      tree_sitter_count = sum(1 for c in token_config.language_configs.values() if c.language is not None)
       65  
       66      bt.logging.info('***** Starting scoring round *****')
       67      bt.logging.info(f'Total Repositories loaded: {len(master_repositories)}')
       68      bt.logging.info(f'Total Languages loaded: {len(programming_languages)}')
       69      bt.logging.info(f'Token config: {tree_sitter_count} tree-sitter languages')
       70      bt.logging.info(f'Neurons to evaluate: {len(miner_uids)}')
       71  
       72      rewards, miner_evaluations = await get_rewards(
       73          self, miner_uids, master_repositories, programming_languages, token_config
       74      )
       75  
       76      # -------------------------------------------------------------------------
       77      # Issue Bounties Treasury Allocation
       78      # The smart contract neuron (ISSUES_TREASURY_UID) accumulates emissions
       79      # which fund issue bounty payouts. We allocate a fixed percentage of
       80      # total emissions to this treasury by scaling down all miner rewards
       81      # and assigning the remainder to the treasury UID.
       82      # -------------------------------------------------------------------------
       83      if ISSUES_TREASURY_UID > 0 and ISSUES_TREASURY_UID in miner_uids:
       84          treasury_share = ISSUES_TREASURY_EMISSION_SHARE
       85          miner_share = 1.0 - treasury_share
       86  
       87          # rewards array is indexed by position in sorted(miner_uids)
       88          sorted_uids = sorted(miner_uids)
       89          treasury_idx = sorted_uids.index(ISSUES_TREASURY_UID)
       90  
       91          # Scale down all rewards proportionally
       92          rewards *= miner_share
       93  
       94          # Assign treasury's share
       95          rewards[treasury_idx] = treasury_share
       96  
       97          bt.logging.info(
       98              f'Treasury allocation: Smart Contract UID {ISSUES_TREASURY_UID} receives '
       99              f'{treasury_share * 100:.0f}% of emissions, miners share {miner_share * 100:.0f}%'
      100          )
      101  
      102      self.update_scores(rewards, miner_uids)
      103  
      104      return miner_evaluations
