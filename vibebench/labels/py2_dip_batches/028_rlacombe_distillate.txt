       34              pass

       57  def _find_papers(query: str, state) -> list[tuple[str, dict]]:
       58      """Resolve a query to a list of (item_key, doc) matches.
       59  
       60      Tries (in order): index number, exact citekey, citekey substring,
       61      then title substring.
       62      """
       63      query = query.strip().strip('"').strip("'")
       64  
       65      # Try index number
       66      if query.isdigit():
       67          idx = int(query)
       68          key = state.key_for_index(idx)
       69          if key:
       70              doc = state.get_document(key)
       71              if doc:
       72                  return [(key, doc)]
       73  
       74      query_lower = query.lower()
       75      matches = []
       76      for key, doc in state.documents.items():
       77          ck = doc.get("metadata", {}).get("citekey", "")
       78          if ck and ck.lower() == query_lower:
       79              return [(key, doc)]  # exact citekey match
       80          if (query_lower in ck.lower()
       81                  or query_lower in doc.get("title", "").lower()):
       82              matches.append((key, doc))
       83      return matches

       86  def _compute_engagement(
       87      highlights: dict | None, page_count: int,
       88  ) -> int:
       89      """Compute an engagement score (0–100) from highlights and page count.
       90  
       91      Components (weighted):
       92        - Highlight density (30%): highlights per page, saturates at 1 per page
       93        - Page coverage (40%): fraction of pages with at least one highlight
       94        - Highlight volume (30%): absolute count, saturates at 20
       95      """
       96      if not highlights:
       97          return 0
       98      highlight_count = sum(len(v) for v in highlights.values())
       99      highlighted_pages = len(highlights)
      100      pages = max(page_count, 1)
      101  
      102      density = min(highlight_count / pages, 1.0)
      103      coverage = min(highlighted_pages / pages, 1.0)
      104      volume = min(highlight_count / 20, 1.0)
      105  
      106      return round((density * 0.3 + coverage * 0.4 + volume * 0.3) * 100)
