       25  def parse_provider(provider: str, default_trading_type: str = "router") -> Tuple[str, str]:
       26      """
       27      Parse provider string into (dex_name, trading_type) tuple.
       28  
       29      Provider strings are used by executors in format "dex/trading_type".
       30      Gateway HTTP client requires separate dex_name and trading_type parameters.
       31  
       32      Args:
       33          provider: Provider string in format "dex/type" or just "dex"
       34              Examples: "jupiter/router", "meteora/clmm", "orca"
       35          default_trading_type: Default type if not specified in provider
       36              Use "router" for swap operations, "clmm" for LP operations
       37  
       38      Returns:
       39          Tuple of (dex_name, trading_type)
       40  
       41      Examples:
       42          >>> parse_provider("jupiter/router")
       43          ("jupiter", "router")
       44          >>> parse_provider("meteora/clmm")
       45          ("meteora", "clmm")
       46          >>> parse_provider("orca")
       47          ("orca", "router")
       48          >>> parse_provider("orca", default_trading_type="clmm")
       49          ("orca", "clmm")
       50      """
       51      if "/" in provider:
       52          parts = provider.split("/", 1)
       53          return parts[0], parts[1]
       54      return provider, default_trading_type
