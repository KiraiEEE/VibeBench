       16  Ahead-of-time compilation support for CuTe DSL kernels.
       17  
       18  CuTe DSL kernels are normally JIT-compiled via ``cute.compile()`` with
       19  ``--enable-tvm-ffi``.  This module provides utilities to:
       20  
       21  1. **Export** a compiled CuTe DSL kernel to an object file via TVM-FFI ABI
       22     (``compiled_fn.export_to_c()``)
       23  2. **Link** the object file into a shared library (``.so``)
       24  3. **Load** the pre-compiled ``.so`` at runtime via
       25     ``cute.runtime.load_module()``
       26  
       27  The workflow mirrors how FlashInfer's JitSpec-based AOT system works:
       28  build-time compilation produces ``.so`` files that are discovered and
       29  loaded at runtime, skipping the JIT step entirely.
       30  """

       32  import gc
       33  import subprocess
       34  from pathlib import Path
       35  from typing import Any, Callable, Optional
       36

       55  def export_cute_dsl_kernel(
       56      compiled_fn: Any,
       57      func_name: str,
       58      output_dir: Path,
       59  ) -> Path:
       60      """Export a compiled CuTe DSL kernel to a shared library.
       61  
       62      Args:
       63          compiled_fn: The result of ``cute.compile(..., options="--enable-tvm-ffi")``.
       64              Must be a ``TVMFFIJitCompiledFunction`` with ``export_to_c`` method.
       65          func_name: Unique function name for the exported symbol.
       66          output_dir: Directory to write the ``.o`` and ``.so`` files.
       67  
       68      Returns:
       69          Path to the linked ``.so`` file.
       70      """
       71      import cutlass.cute.runtime as rt
       72  
       73      output_dir = Path(output_dir)
       74      output_dir.mkdir(parents=True, exist_ok=True)
       75  
       76      obj_file = output_dir / f"{func_name}.o"
       77      so_file = output_dir / f"{func_name}.so"
       78  
       79      # Export to object file using TVM-FFI ABI
       80      compiled_fn.export_to_c(str(obj_file), function_name=func_name)
       81  
       82      # Link into shared library
       83      runtime_libs = rt.find_runtime_libraries(enable_tvm_ffi=True)
       84      link_cmd = ["gcc", "-shared", "-o", str(so_file), str(obj_file)] + runtime_libs
       85      subprocess.run(link_cmd, check=True)
       86  
       87      logger.info(f"CuTe DSL AOT: exported {func_name} -> {so_file}")
       88      return so_file
       89  
       90  
       91  def load_cute_dsl_module(so_path: Path) -> Any:
       92      """Load a pre-compiled CuTe DSL shared library via TVM-FFI.

       94      Args:
       95          so_path: Path to the ``.so`` file.
       96  
       97      Returns:
       98          A TVM-FFI module with the exported function(s) as attributes.
       99      """

      102      return rt.load_module(str(so_path), enable_tvm_ffi=True)
