       46      def open(self, file_path: str, run_auto_analysis: bool = False) -> dict:
       47          """Open a binary for analysis. Auto-closes any previously open database.
       48  
       49          Returns a status dict on success.  Raises :class:`IDAError` on failure.
       50          """
       51          path = os.path.abspath(os.path.expanduser(file_path))
       52          if not os.path.isfile(path):
       53              raise IDAError(f"File not found: {path}", error_type="FileNotFoundError")
       54  
       55          if self.is_open():
       56              self.close(save=True)
       57  
       58          result = idapro.open_database(path, run_auto_analysis)
       59          if result != 0:
       60              raise IDAError(
       61                  f"Failed to open database: error code {result}", error_type="RuntimeError"
       62              )
       63  
       64          self._current_path = path
       65          self.capabilities = self._probe_capabilities()
       66          log.info("Opened database: %s (capabilities: %s)", path, self.capabilities)
       67          return {"status": "ok", "path": path}
