       29      normalized = state_name.casefold()

       31          return "⬜"

       40      """Handle /tasks command and show latest Linear issues."""
       41      del key, text
       42  
       43      team_id = orch.config.linear.default_team_id.strip()

       46              text="Linear is not configured: set linear.default_team_id in config.json",
       47          )
       48  
       49      try:
       50          issues = await orch.linear_client.list_recent_issues(team_id=team_id, limit=15)
       51      except (RuntimeError, ValueError, TypeError) as exc:
       52          return OrchestratorResult(text=f"Failed to load Linear issues: {exc}")
       53  
       54      if not issues:
       55          return OrchestratorResult(text="No Linear issues found for the configured team.")
       56  
       57      lines = ["Latest Linear issues:"]
       58      rows: list[list[Button]] = []
       59      for index, issue in enumerate(issues, start=1):
       60          lines.append(f"{index}. {_status_emoji(issue.state_name)} {issue.identifier} {issue.title}")
       61          rows.append(
       62              [Button(text=issue.identifier, callback_data=f"linear:task:{issue.identifier}")]
       63          )
       64  
       65      return OrchestratorResult(
       66          text="\n".join(lines),
