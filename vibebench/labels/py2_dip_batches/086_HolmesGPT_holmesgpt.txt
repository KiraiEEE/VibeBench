       12  def _truncate_to_depth(value: Any, max_depth: Optional[int], current_depth: int = 0):
       13      """Recursively truncate dictionaries/lists beyond the requested depth."""
       14      if max_depth is None or max_depth < 0:
       15          return value
       16  
       17      if current_depth >= max_depth:
