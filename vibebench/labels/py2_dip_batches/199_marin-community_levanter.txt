       82  def _convert_value_to_loggable_rec(value: Any):
       83      if isinstance(value, (list, tuple)):
       84          return [_convert_value_to_loggable_rec(v) for v in value]
       85      elif isinstance(value, typing.Mapping):
       86          return {k: _convert_value_to_loggable_rec(v) for k, v in value.items()}
       87      elif isinstance(value, jax.Array):
       88          if value.ndim == 0:
       89              return value.item()
       90          else:
       91              return np.array(value)
       92      elif isinstance(value, Histogram):
       93          counts, limits = value.to_numpy_histogram()
       94          return {"counts": counts.tolist(), "limits": limits.tolist()}
       95      else:
       96          return value
