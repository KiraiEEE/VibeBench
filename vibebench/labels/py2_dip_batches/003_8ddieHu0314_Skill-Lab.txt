       26  def _build_diff_text(raw_diff: str) -> Text:
       27      """Build a Rich Text from a unified diff with human-readable formatting.
       28  
       29      - Strips redundant file headers (--- / +++)
       30      - Skips @@ hunk markers (line numbers shown inline instead)
       31      - Prepends the file line number to each removed/added line
       32      - Colors removed lines red, added lines green
       33      - Each line is appended separately so word-wrap never bleeds between lines
       34      """
       35      text = Text()
       36      old_line = 0
       37      new_line = 0
       38      first_hunk = True
       39      for line in raw_diff.splitlines():
       40          if line.startswith("--- ") or line.startswith("+++ "):
       41              continue
       42          m = _HUNK_RE.match(line)
       43          if m:
       44              old_line = int(m.group(1))
       45              new_line = int(m.group(2))
       46              if not first_hunk:
       47                  text.append("\n")
       48              first_hunk = False
       49          elif line.startswith("-"):
       50              text.append(f"{old_line:4d} ", style="dim red")
       51              text.append(line + "\n", style="red")
       52              old_line += 1
       53          elif line.startswith("+"):
       54              text.append(f"{new_line:4d} ", style="dim green")
       55              text.append(line + "\n", style="green")
       56              new_line += 1
       57          else:
       58              text.append(f"     {line}\n")
       59              old_line += 1
       60              new_line += 1
       61      return text
