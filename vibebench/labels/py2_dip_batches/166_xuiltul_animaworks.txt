       31  def _calc_effective_max_turns(
       32      base_max_turns: int,
       33      activity_level: int,
       34      hb_max_turns: int | None = None,
       35  ) -> int | None:
       36      """Calculate effective max_turns for heartbeat based on activity level.
       37  
       38      When *hb_max_turns* is provided (from ``config.heartbeat.max_turns``),
       39      it is used as the base instead of the per-anima chat ``max_turns``.
       40  
       41      Below 100%: linear scale (floor 3). At/above 100%: return None (use base).
       42      """
       43      base = hb_max_turns if hb_max_turns is not None else base_max_turns
       44      if activity_level >= 100:
       45          if hb_max_turns is not None:
       46              return hb_max_turns
       47          return None
       48      scaled = max(3, math.ceil(base * activity_level / 100))
       49      return scaled
