       13      """Infer the type of a field from sample values."""
       14      if not values:
       15          return "unknown"
       16  
       17      # Remove None values
       18      non_null = [v for v in values if v is not None]
       19      if not non_null:
       20          return "null"
       21  
       22      # Check if numeric
       23      try:
       24          numeric_values = [float(v) for v in non_null if v != ""]
       25          if len(numeric_values) > len(non_null) * 0.8:  # 80% numeric
       26              # Check if integer
       27              if all(v == int(v) for v in numeric_values):
       28                  return "integer"
       29              return "number"
       30      except (ValueError, TypeError):
       31          pass
       32  
       33      # Check for lat/lon patterns

       35      if any(
       36          keyword in sample_str for keyword in ["lat", "latitude", "lng", "lon", "longitude", "coord"]
       37      ):
       38          return "coordinate"
       39  
       40      return "string"

       45      stats = {"count": len(values), "null_count": sum(1 for v in values if v is None)}
