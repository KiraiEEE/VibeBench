       84          if not os.path.exists(index_path):
       85              from huggingface_hub import hf_hub_download

       87              try:
       88                  index_path = hf_hub_download(model_or_path, "model_index.json")
       89              except Exception:
       90                  return None
       91  
       92          import json
       93  
       94          with open(index_path) as f:
       95              data = json.load(f)
       96          if "transformer" not in data:
       97              return None
       98          for name in HYBRID_AR_COMPONENTS:
       99              if name in data:
      100                  return name
      101          return None

      104      if hasattr(model_or_path, "transformer"):
      105          for name in HYBRID_AR_COMPONENTS:
      106              if hasattr(model_or_path, name) and getattr(model_or_path, name) is not None:
