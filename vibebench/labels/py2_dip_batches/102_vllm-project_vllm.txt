       26  def if_aiter_supported(func: Callable) -> Callable:
       27      """Decorator that only executes the function if
       28      ROCm AITER package is supported on gfx9 archs.
       29      """
       30  
       31      @functools.wraps(func)
       32      def wrapper(*args, **kwargs):
       33          # checks the platform, device arch and aiter library existence.
       34  
       35          if current_platform.is_rocm() and IS_AITER_FOUND:
       36              from vllm.platforms.rocm import on_gfx9
       37  
       38              if on_gfx9():
       39                  return func(*args, **kwargs)
       40  
       41          return None
       42  
       43      return wrapper
